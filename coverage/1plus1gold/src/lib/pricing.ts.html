/**
 * Shared Our-Pricing display derivation — single source for the dynamic
 * "answer value" (premium range) + the merged comparison data, consumed by:
 *   • src/app/our-pricing/page.tsx              (web page useMemo)
 *   • src/app/embed/our-pricing/market-range    (iOS SDUI chart embed)
 *   • src/app/api/pricing/overview              (read-API for the SDUI spec)
 *
 * Extracted VERBATIM from page.tsx's useMemo (2026-06-03) so there is exactly
 * ONE implementation across page / embed / spec API.
 */
 
import { PREMIUM_DATA, npdRow, type PremiumDatum } from "@/lib/premiumData";
import type { PricingSummary } from "@/lib/pricing/page-summary";
import { formatPremium } from "@/lib/utils";
 
export interface PricingDisplay {
  /** e.g. "+22% → +3%" — max → min published premium across 1–100g. */
  answerValue: string;
  /** PREMIUM_DATA with each row's `ours` pct refreshed from live premiums. */
  comparisonData: PremiumDatum[];
}
 
export function derivePricingDisplay(
  premiums: { g: number; premium_pct: number }[],
): PricingDisplay {
  if (!premiums.length) return { answerValue: "—", comparisonData: PREMIUM_DATA };
 
  const relevant = premiums.filter((p) => p.g >= 1 && p.g <= 100);
  const maxPct = relevant.length ? Math.max(...relevant.map((p) => p.premium_pct)) : 22;
  const minPct = relevant.length ? Math.min(...relevant.map((p) => p.premium_pct)) : 3;
 
  // ours is a percentage (e.g. 20), matching PremiumComparisonChart's data format
  const oursPctByGrams = new Map(premiums.map((p) => [p.g, p.premium_pct]));
  const comparisonData = PREMIUM_DATA.map((row) => {
    const g = parseFloat(row.label);
    const dbPct = oursPctByGrams.get(g);
    return dbPct !== undefined ? { ...row, ours: dbPct } : row;
  });
 
  return {
    answerValue: `${formatPremium(maxPct / 100)} → ${formatPremium(minPct / 100)}`,
    comparisonData,
  };
}
 
/**
 * Add the live layer to the survey rows: the "competitors this week" marker
 * per denomination, and the 1g-NPD row (MAG's market; our marker = the MAG
 * 1g premium when a bar is listed). Shared by the page and the iOS embed.
 */
export function withLiveMarkers(
  rows: PremiumDatum[],
  summary: Pick<PricingSummary, "competitorsThisWeek" | "explainer"> | null,
): PremiumDatum[] {
  const compByG = new Map((summary?.competitorsThisWeek ?? []).map((c) => [c.grams, c.medianPct]));
  const live = rows.map((row) => {
    const c = compByG.get(parseFloat(row.label));
    return c !== undefined ? { ...row, competitor: c } : row;
  });
  const mag1g = summary?.explainer.mag[0]?.pct ?? null;
  return [...live, npdRow(mag1g)];
}
