/**
 * TikTok video publishing for Drumbeat. Server-only.
 * Verified (TikTok API for Business v1.3, 2026-11-17): doc 1762228496095234 "Publish a
 * public video post to an owned account", doc 1816387106635778 "publishing status".
 *
 * Two modes (env TIKTOK_PUBLISH_MODE, default "draft"):
 *   draft  — `upload_to_draft`: the video lands in the account's TikTok inbox and Mukesh
 *            posts it from the app. Drumbeat treats it like WhatsApp Status: the channel
 *            waits on a "Mark as posted" tap ("manual-pending-…").
 *   direct — a public post, labelled "Promotional content" (`is_brand_organic`).
 *
 * Publishing is asynchronous: TikTok returns a `share_id`, downloads the file, then
 * reports PROCESSING_DOWNLOAD → PUBLISH_COMPLETE (post_ids may lag up to 3 min) |
 * SEND_TO_USER_INBOX | FAILED. We poll briefly inline; anything still in flight is
 * recorded as `tt-pending:<share_id>` and finished by the tiktok-publish-status cron.
 */
import { tiktokCall, type Http } from "./connection";
 
export type TikTokPublishMode = "draft" | "direct";
 
export function publishMode(env: Readonly<Record<string, string | undefined>> = process.env): TikTokPublishMode {
  return env.TIKTOK_PUBLISH_MODE?.trim() === "direct" ? "direct" : "draft";
}
 
/** TikTok's caption cap is 2,200 UTF-16 units. Trim on a whole code point, never mid-pair. */
export const CAPTION_MAX = 2200;
export function fitCaption(text: string): string {
  if (text.length <= CAPTION_MAX) return text;
  let cut = text.slice(0, CAPTION_MAX);
  const last = cut.charCodeAt(cut.length - 1);
  if (last >= 0xd800 && last <= 0xdbff) cut = cut.slice(0, -1);
  return cut;
}
 
export const PENDING_PREFIX = "tt-pending:";
/** The id recorded once TikTok has published: `tt:<post id>`. */
export const postIdMarker = (postId: string) => `tt:${postId}`;
export const pendingMarker = (shareId: string) => `${PENDING_PREFIX}${shareId}`;
export const shareIdFromPending = (v: string | undefined | null): string | null =>
  typeof v === "string" && v.startsWith(PENDING_PREFIX) ? v.slice(PENDING_PREFIX.length) : null;
 
/**
 * A TikTok video link → our `tt:<id>` marker + a clean permalink, or null when it is
 * not one (2026-11-26, marking a live post the publish lost track of). Only
 * https://(www.|m.)tiktok.com/@account/video/<digits> is accepted.
 */
export function tiktokPostFromUrl(url: string): { postId: string; permalink: string } | null {
  let u: URL;
  try { u = new URL(url.trim()); } catch { return null; }
  if (u.protocol !== "https:" || !/^(www\.|m\.)?tiktok\.com$/.test(u.hostname)) return null;
  const m = /^\/(@[A-Za-z0-9._]{1,40})\/video\/(\d{6,25})\/?$/.exec(u.pathname);
  if (!m) return null;
  return { postId: postIdMarker(m[2]), permalink: `https://www.tiktok.com/${m[1]}/video/${m[2]}` };
}
 
export function permalinkFor(account: string | null, postId: string): string {
  const handle = account?.startsWith("@") ? account : "@1plus1gold";
  return `https://www.tiktok.com/${handle}/video/${postId}`;
}
 
/**
 * Builds `post_info`. The two disclosure flags go on EVERY request: the docs call them
 * {Required} and say draft mode ignores post_info, but TikTok REFUSES a draft without them
 * (`40002 post_info.is_brand_organic: Missing data for required field`, first live draft
 * on PROD 2026-11-17). Our posts promote our own business: Promotional content, never a
 * paid partnership.
 */
export function buildPostInfo(input: { caption: string; mode: TikTokPublishMode }): Record<string, unknown> {
  const disclosure = { is_brand_organic: true, is_branded_content: false };
  return input.mode === "draft"
    ? { ...disclosure, upload_to_draft: true }
    : { ...disclosure, caption: fitCaption(input.caption), disable_comment: false, disable_duet: false, disable_stitch: false };
}
 
export async function publishVideo(
  auth: { token: string; businessId: string },
  input: { videoUrl: string; caption: string; mode: TikTokPublishMode; coverUrl?: string },
  http: Http = {},
): Promise<string> {
  const body = (await tiktokCall("/business/video/publish/", {
    method: "POST", accessToken: auth.token,
    body: {
      business_id: auth.businessId, video_url: input.videoUrl,
      // The cover: a verified-URL image (JPG/PNG/WebP ≤20 MB). Absent → TikTok's own frame.
      ...(input.coverUrl ? { custom_thumbnail_url: input.coverUrl } : {}),
      post_info: buildPostInfo(input),
    },
  }, http)) as { code?: unknown; message?: unknown; data?: { share_id?: unknown } };
  const shareId = body?.data?.share_id;
  if (String(body?.code) !== "0" || typeof shareId !== "string" || !shareId) {
    throw new TikTokRefusal(`TikTok refused the video: ${String(body?.code ?? "?")} ${String(body?.message ?? "").slice(0, 200)}`.trim());
  }
  return shareId;
}

/**
 * Polls until a terminal state or the budget runs out (then "processing").
 *
 * A status check that THROWS (a dropped connection — "fetch failed" — or a bad
 * reply) is read as "still processing", never as a failure (2026-11-26): by the time
 * we poll, TikTok has ACCEPTED the video, so a lost status reply says nothing about
 * the post. Throwing here marked a reel that DID go live as failed, which put a
 * re-post (a duplicate) one tap away. "processing" leaves `tt-pending:<shareId>` on
 * the piece and the tiktok-publish-status cron resolves the real id + link.
 */
export async function waitForPublish(
  auth: { token: string; businessId: string }, shareId: string,
  opts: { budgetMs?: number; intervalMs?: number; sleep?: (ms: number) => Promise<void> } & Http = {},
): Promise<PublishState> {
  const budget = opts.budgetMs ?? 60_000;
  const interval = opts.intervalMs ?? 5_000;
  const sleep = opts.sleep ?? ((ms: number) => new Promise<void>((r) => setTimeout(r, ms)));
  for (let waited = 0; ; waited += interval) {
    let s: PublishState;
    try {
      s = await publishStatus(auth, shareId, opts);
    } catch (err) {
      console.warn(`[tiktok] publish status check failed for ${shareId}, treating as still processing:`, err instanceof Error ? err.message : err);
      s = { kind: "processing" };
    }
    // A complete post whose id has not landed yet is still worth finishing later.
    if (s.kind !== "processing" && !(s.kind === "complete" && !s.postId)) return s;
    if (waited + interval > budget) return s;
    await sleep(interval);
  }
}
