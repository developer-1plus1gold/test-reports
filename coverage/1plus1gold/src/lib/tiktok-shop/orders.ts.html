/**
 * TikTok Shop order import (step D, 2026-11-17) — TikTok's READER for the shared engine
 * (lib/marketplace/order-import), which owns the switch, the cut-over, the ledger, the
 * issue cards and createExternalOrder (the same path as the manual External sales form).
 *
 * Get Order List (POST /order/202309/orders/search, doc 650aa8094a0bb702c06df242): query
 * page_size (≤ 100), sort_field, sort_order, page_token; body update_time_ge (UNIX seconds).
 * Each order carries its line_items, so one call per page is enough.
 *
 * TikTok rules (from that doc):
 *   • each line item is ONE unit (no quantity field) — same as Lazada's item rows;
 *   • order status UNPAID → not ours yet; ON_HOLD → paid but the buyer may still cancel
 *     without our approval — imported anyway, because taking the stock early is what stops
 *     an oversell, and a later cancel becomes a card like Lazada's;
 *     AWAITING_SHIPMENT / AWAITING_COLLECTION / PARTIALLY_SHIPPING / IN_TRANSIT / DELIVERED /
 *     COMPLETED → paid; CANCELLED → cancelled;
 *   • IN_TRANSIT → our order picked up; DELIVERED / COMPLETED → delivered (forward only);
 *   • revenue per unit = the line's original_price; seller-funded product discounts
 *     (line seller_discount) and seller-funded shipping discounts are OUR cost, as on Lazada;
 *     platform discounts are TikTok's money and do not touch our numbers.
 * Returns are not an order status on TikTok (they arrive by the return webhook / Search
 * Returns) — handled with the return flow, not here.
 * Buyer name / phone / address are not stored; only the postal code (the CoA claim key).
 */

const money = (v: unknown) => { const n = Number(v); return Number.isFinite(n) && n > 0 ? n : 0; };
const r2 = (n: number) => Math.round(n * 100) / 100;

const PAID = new Set(["ON_HOLD", "AWAITING_SHIPMENT", "PARTIALLY_SHIPPING", "AWAITING_COLLECTION", "IN_TRANSIT", "DELIVERED", "COMPLETED"]);
/** TikTok statuses once the shipment is arranged (Ship Package done). */
const ARRANGED = new Set(["AWAITING_COLLECTION", "IN_TRANSIT", "DELIVERED", "COMPLETED"]);
const isCancelled = (s: string | undefined) => /^CANCEL/.test((s ?? "").toUpperCase());
 
/** Pure. */
export function ttsOrderState(o: TtsOrder): OrderState {
  const s = (o.status ?? "").toUpperCase();
  if (isCancelled(s)) return "canceled";
  return PAID.has(s) ? "paid" : "unpaid";
}
 
/** Pure. */
export function ttsShipStage(o: TtsOrder): ShipStage {
  const s = (o.status ?? "").toUpperCase();
  if (s === "DELIVERED" || s === "COMPLETED") return "delivered";
  if (s === "IN_TRANSIT") return "shipped";
  return null;
}
 
/** "2026-11-17" in SGT from a UNIX-seconds time. Pure. */
export function sgtDay(unix: number | undefined): string | undefined {
  const d = fromUnix(unix);
  return d ? new Date(d.getTime() + 8 * 3600_000).toISOString().slice(0, 10) : undefined;
}
 
/**
 * Pure: a paid TikTok order → the createExternalOrder input. Units of the same SKU at the
 * same price are grouped; cancelled lines and free gifts are dropped. The channel fee is
 * an ESTIMATE from the TikTok Shop pricing settings until the finance read replaces it.
 */
export function mapTtsOrder(o: TtsOrder, productBySku: Map<string, string>, cfg: MarketplaceCostConfig): MapResult {
  const live = (o.line_items ?? []).filter((l) => !isCancelled(l.display_status) && !l.is_gift);
  const unknownSkus = [...new Set(live.map((l) => l.seller_sku ?? "").filter((s) => !productBySku.has(s)))];
  if (unknownSkus.length) return { ok: false, unknownSkus: unknownSkus.map((s) => s || "(blank SKU)") };
 
  const lines = new Map<string, { productId: string; quantity: number; priceSgd: number }>();
  for (const l of live) {
    const productId = productBySku.get(l.seller_sku!)!;
    const price = r2(money(l.original_price) || money(l.sale_price));
    const key = `${productId}|${price}`;
    const line = lines.get(key) ?? { productId, quantity: 0, priceSgd: price };
    line.quantity += 1;
    lines.set(key, line);
  }
  const gross = [...lines.values()].reduce((s, l) => s + l.priceSgd * l.quantity, 0);
  const pct = (cfg.commissionPct + cfg.otherPct) / 100;
  return {
    ok: true,
    input: {
      channel: CHANNEL,
      externalOrderId: String(o.id),
      buyerLabel: "TikTok Shop buyer",
      orderDate: sgtDay(o.create_time),
      channelFeeSgd: r2(gross * pct),
      sellerVoucherSgd: r2(live.reduce((s, l) => s + money(l.seller_discount), 0)),
      absorbedShippingSgd: r2(money(o.payment?.shipping_fee_seller_discount) + money(o.payment?.shipping_fee_cofunded_discount)),
      shipTo: { postal_code: o.recipient_address?.postal_code?.replace(/\*/g, "").trim() || null },
      items: [...lines.values()],
    },
  };
}
 
/**
 * Pure: the buyer's checkout on TikTok. List / seller / platform discounts are summed over the
 * live lines (as the mapper does); shipping and other charges are TikTok's order-level
 * figures; `total_amount` is TikTok's own "buyer paid".
 */
export function ttsBuyerFigures(o: TtsOrder): BuyerFigures {
  const live = (o.line_items ?? []).filter((l) => !isCancelled(l.display_status) && !l.is_gift);
  const p = o.payment ?? {};
  const reported = Number(p.total_amount);
  return {
    listSgd: r2(live.reduce((s, l) => s + (money(l.original_price) || money(l.sale_price)), 0)),
    sellerVoucherSgd: r2(live.reduce((s, l) => s + money(l.seller_discount), 0)),
    platformVoucherSgd: r2(live.reduce((s, l) => s + money(l.platform_discount), 0)),
    platformShippingDiscountSgd: r2(money(p.shipping_fee_platform_discount)),
    buyerShippingSgd: r2(money(p.shipping_fee)),
    otherBuyerChargesSgd: r2(money(p.tax) + money(p.small_order_fee) + money(p.buyer_service_fee) + money(p.handling_fee)),
    reportedPaidSgd: p.total_amount != null && Number.isFinite(reported) ? r2(reported) : null,
    source: {
      payment: p,
      lines: live.map((l) => ({ original_price: l.original_price, sale_price: l.sale_price, seller_discount: l.seller_discount, platform_discount: l.platform_discount })),
    },
  };
}
 
/** Pure: one TikTok order as the engine sees it. */
export function toChannelOrder(o: TtsOrder): ChannelOrder {
  return {
    externalId: String(o.id ?? ""),
    createdAt: fromUnix(o.create_time),
    updatedAt: fromUnix(o.update_time),
    state: ttsOrderState(o),
    statusText: o.status ?? "",
    shipStage: ttsShipStage(o),
    map: (productBySku, cfg) => mapTtsOrder(o, productBySku, cfg),
    buyer: () => ttsBuyerFigures(o),
    arranged: ARRANGED.has((o.status ?? "").toUpperCase()),
    buyerId: o.user_id ? String(o.user_id).trim() || null : null,
  };
}

export async function syncTikTokShopOrders(db: SupabaseClient, now: Date, mode: ImportMode, http: Http = {}, opts: ImportOptions = {}): Promise<OrderSyncResult> {
  const r = await runOrderImport(db, now, mode, TIKTOK_SHOP_ORDERS, http, opts);
  // Then arrange the shipment for paid orders, when the channel's auto-dispatch is on.
  if (r.ran) {
    r.autoDispatch = await runAutoDispatch(db, now, TIKTOK_DISPATCH, http).catch((e: unknown) => {
      console.error("[auto-dispatch]", e instanceof Error ? e.message : e);
      return undefined;
    });
  }
  return r;
}
