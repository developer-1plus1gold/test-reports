/**
 * TikTok Shop connection — app settings, Connect states, tokens, the shop cipher.
 * Server-only. Built on the shared OAuth store (src/lib/oauth), like Lazada.
 *
 * App secret and tokens are stored encrypted (secret-box) and decrypted only in memory
 * for a call. Nothing here logs a token, the secret or a state value.
 * Design: docs/tiktok-shop-integration.md (step B).
 */
import type { SupabaseClient } from "@supabase/supabase-js";
import { encryptSecret, decryptSecret } from "@/lib/crypto/secret-box";
import {
  issueOAuthState, consumeOAuthState, loadConnectionRow, saveConnectionTokens,
  markConnectionExpired, recordRefreshFailure, disconnectChannel, type ConnectionRow,
} from "@/lib/oauth/connection-store";
import { ttsCall, ttsTokenCall } from "./client";
import { parseTokenResponse, needsRefresh, type TtsTokenSet } from "./tokens";
 
const CHANNEL = "tiktok_shop" as const;
 
export interface TtsAppConfig { appKey: string | null; serviceId: string | null; appSecret: string | null }
 
export async function loadAppConfig(db: SupabaseClient): Promise<TtsAppConfig> {
  const { data } = await db.from("marketplace_app_config")
    .select("app_key, service_id, app_secret_enc").eq("channel", CHANNEL).maybeSingle();
  return {
    appKey: (data?.app_key as string | undefined) ?? null,
    serviceId: (data?.service_id as string | undefined) ?? null,
    appSecret: decryptSecret(data?.app_secret_enc as string | undefined),
  };
}
 
export const APP_KEY_RE = /^[0-9A-Za-z_-]{1,64}$/;
export const SERVICE_ID_RE = /^[0-9]{1,32}$/;
 
/** Saves key + service id; the secret only when a non-empty one is given (an edit never wipes it). */
export async function saveAppConfig(db: SupabaseClient, input: { appKey: string; serviceId: string; appSecret?: string }): Promise<void> {
  if (!APP_KEY_RE.test(input.appKey)) throw new Error("App Key must be letters, digits, - or _ (max 64)");
  if (!SERVICE_ID_RE.test(input.serviceId)) throw new Error("Service ID must be digits");
  const row: Record<string, unknown> = {
    channel: CHANNEL, app_key: input.appKey, service_id: input.serviceId, updated_at: new Date().toISOString(),
  };
  if (input.appSecret) row.app_secret_enc = encryptSecret(input.appSecret);
  const { error } = await db.from("marketplace_app_config").upsert(row, { onConflict: "channel" });
  if (error) throw new Error("Could not save the TikTok Shop app settings");
}
 
export const issueState = (db: SupabaseClient, now: Date): Promise<string> => issueOAuthState(db, CHANNEL, now);
export const consumeState = (db: SupabaseClient, state: string, now: Date): Promise<boolean> =>
  consumeOAuthState(db, CHANNEL, state, now);
 
export interface TtsConnectionRow extends ConnectionRow { shopCipher: string | null; grantedScopes: string[] }
 
export async function loadConnection(db: SupabaseClient): Promise<TtsConnectionRow | null> {
  const row = await loadConnectionRow(db, CHANNEL);
  if (!row) return null;
  const { data } = await db.from("marketplace_connections").select("shop_cipher").eq("channel", CHANNEL).maybeSingle();
  // granted scopes are kept in the (otherwise Lazada-only) report column.
  const report = (row.maskingReport ?? {}) as { grantedScopes?: unknown };
  return {
    ...row,
    shopCipher: (data?.shop_cipher as string | null) ?? null,
    grantedScopes: Array.isArray(report.grantedScopes) ? report.grantedScopes.map(String) : [],
  };
}
 
export interface Http { fetchImpl?: typeof fetch }
 
export interface AuthorizedShop { id: string; name: string; region: string; cipher: string; code: string | null }
 
/** GET /authorization/202309/shops → the shops this seller authorized for the app. */
export async function fetchAuthorizedShops(
  creds: { appKey: string; appSecret: string; accessToken: string }, http: Http = {},
): Promise<AuthorizedShop[]> {
  const data = await ttsCall<{ shops?: Array<Record<string, unknown>> }>("/authorization/202309/shops", {
    appKey: creds.appKey, appSecret: creds.appSecret, accessToken: creds.accessToken, fetchImpl: http.fetchImpl,
  });
  return (data?.shops ?? []).map((s) => ({
    id: String(s.id ?? ""), name: String(s.name ?? ""), region: String(s.region ?? ""),
    cipher: String(s.cipher ?? ""), code: s.code != null ? String(s.code) : null,
  })).filter((s) => s.cipher);
}
 
/** Our one shop: the Singapore one if there are several, else the only one. */
export function pickShop(shops: AuthorizedShop[]): AuthorizedShop | null {
  return shops.find((s) => s.region.toUpperCase() === "SG") ?? (shops.length === 1 ? shops[0] : null);
}
 
/** Callback step: exchange the one-time code, store the tokens, then find the shop cipher. */
export async function completeConnect(db: SupabaseClient, authCode: string, now: Date, http: Http = {}): Promise<{ tokens: TtsTokenSet; shop: AuthorizedShop | null }> {
  const cfg = await loadAppConfig(db);
  if (!cfg.appKey || !cfg.appSecret) throw new Error("TikTok Shop App Key and Secret are not set");
  const tokens = parseTokenResponse(await ttsTokenCall("get", { appKey: cfg.appKey, appSecret: cfg.appSecret, authCode }, http.fetchImpl));
  await saveConnectionTokens(db, CHANNEL, {
    accessToken: tokens.accessToken, refreshToken: tokens.refreshToken,
    accessExpiresAt: tokens.accessExpiresAt, refreshExpiresAt: tokens.refreshExpiresAt,
    sellerId: tokens.openId, account: tokens.sellerName, country: tokens.region,
  }, now, true);
  await db.from("marketplace_connections").update({ masking_report: { grantedScopes: tokens.grantedScopes } }).eq("channel", CHANNEL);
 
  const shop = pickShop(await fetchAuthorizedShops({ appKey: cfg.appKey, appSecret: cfg.appSecret, accessToken: tokens.accessToken }, http));
  if (shop) {
    await db.from("marketplace_connections").update({
      shop_cipher: shop.cipher, account_label: shop.name || tokens.sellerName, seller_id: shop.id || tokens.openId,
      country: shop.region || tokens.region,
    }).eq("channel", CHANNEL);
  }
  return { tokens, shop };
}
 
export type RefreshOutcome = "refreshed" | "not_needed" | "not_connected" | "failed";
 
/** Cron step (every 6 h). As Lazada: a failure is recorded; the connection goes to
 *  `error` (which escalates) only within 6 h of the access token lapsing. */
export async function refreshIfNeeded(db: SupabaseClient, now: Date, http: Http = {}): Promise<RefreshOutcome> {
  const c = await loadConnection(db);
  if (!c || c.status === "disconnected" || !c.refreshTokenEnc) return "not_connected";
  if (c.refreshExpiresAt && c.refreshExpiresAt.getTime() <= now.getTime()) {
    if (c.status !== "expired") await markConnectionExpired(db, CHANNEL, now);
    return "failed";
  }
  if (!needsRefresh(c, now)) return "not_needed";
  const cfg = await loadAppConfig(db);
  const refreshToken = decryptSecret(c.refreshTokenEnc);
  try {
    if (!cfg.appKey || !cfg.appSecret || !refreshToken) throw new Error("app settings or refresh token unreadable");
    const t = parseTokenResponse(await ttsTokenCall("refresh", { appKey: cfg.appKey, appSecret: cfg.appSecret, refreshToken }, http.fetchImpl));
    // Store whatever refresh token comes back (it may rotate).
    await saveConnectionTokens(db, CHANNEL, {
      accessToken: t.accessToken, refreshToken: t.refreshToken, accessExpiresAt: t.accessExpiresAt, refreshExpiresAt: t.refreshExpiresAt,
    }, now, false);
    return "refreshed";
  } catch (err) {
    const msg = err instanceof Error ? err.message.slice(0, 300) : "refresh failed";
    const nearLapse = !c.accessExpiresAt || c.accessExpiresAt.getTime() - now.getTime() < 6 * 3_600_000;
    await recordRefreshFailure(db, CHANNEL, msg, nearLapse, now);
    return "failed";
  }
}
 
/** For the API steps (listings, prices, orders): token + app creds + shop cipher, or null. */
export async function getShopCredentials(db: SupabaseClient, now: Date): Promise<{ token: string; appKey: string; appSecret: string; shopCipher: string } | null> {
  const c = await loadConnection(db);
  if (!c || c.status !== "active" || !c.shopCipher || !c.accessExpiresAt || c.accessExpiresAt.getTime() <= now.getTime()) return null;
  const cfg = await loadAppConfig(db);
  const token = decryptSecret(c.accessTokenEnc);
  if (!token || !cfg.appKey || !cfg.appSecret) return null;
  return { token, appKey: cfg.appKey, appSecret: cfg.appSecret, shopCipher: c.shopCipher };
}
