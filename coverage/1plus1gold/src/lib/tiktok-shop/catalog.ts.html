/**
 * TikTok Shop catalog sync (step C groundwork, 2026-11-17) — READ-ONLY.
 *
 * Reads what the listing work needs and keeps it current, the way "Load Lazada
 * products" does, and saves it on the connection row (`api_shapes.tiktokCatalog`)
 * so it can be reviewed from the database:
 *   - the category tree under "Jewelry Accessories & Derivatives" (TikTok approved our
 *     invite-only application for it on 17 Nov) and any gold / bullion / coin leaf;
 *   - for each such leaf: its rules (certifications, size chart, package dimensions)
 *     and attributes (required ones, allowed values) — what a listing must carry;
 *   - the brands available for our refiners (PAMP, Valcambi, Nadir, …);
 *   - the shop's warehouses (stock is pushed per warehouse);
 *   - the shop's existing products and SKUs.
 *
 * Endpoints, all verified on partner.tiktokshop.com (doc ids in
 * docs/tiktok-shop-integration.md): GET /product/202309/categories,
 * …/categories/{id}/rules, …/categories/{id}/attributes, GET /product/202309/brands,
 * GET /logistics/202309/warehouses, POST /product/202502/products/search.
 * ▶ SEA shops MUST pass category_version=v2 (the 7-level tree) — TikTok's docs.
 * Each section fails on its own: one refused call never hides the others.
 */
import type { SupabaseClient } from "@supabase/supabase-js";
import { ttsCall, type TtsCallOptions } from "./client";
import { getShopCredentials } from "./connection";
 
type Creds = { token: string; appKey: string; appSecret: string; shopCipher: string };
type Http = Pick<TtsCallOptions, "fetchImpl">;
 
export interface CategoryNode { id: string; parent_id: string; local_name: string; is_leaf: boolean; permission_statuses?: string[] }
 
export interface CatalogSnapshot {
  at: string;
  approvedTree: string;
  categories: { total: number; matched: MatchedCategory[] };
  leaves: Array<{ id: string; path: string; rules: unknown; attributes: unknown; error?: string }>;
  brands: Array<{ query: string; found: Array<{ id: string; name: string; authorized_status?: string }>; error?: string }>;
  warehouses: unknown;
  products: { total: number; items: unknown[] } | null;
  errors: Record<string, string>;
}
 
const APPROVED_TREE = /jewel+e?ry accessories/i;
const GOLD_WORDS = /gold|bullion|precious|investment|ingot|coin|bar\b|silver/i;
const BRAND_QUERIES = ["PAMP", "Valcambi", "Nadir", "Royal Canadian Mint", "Argor", "Perth Mint", "Heraeus", "Austrian Mint"];
const MAX_LEAVES = 12;
 
export interface MatchedCategory { id: string; path: string; isLeaf: boolean; permission: string[]; inApprovedTree: boolean }
 
/** Pure: every node under the approved tree, plus any gold-ish node anywhere, with its path. */
export function matchCategories(nodes: CategoryNode[]): MatchedCategory[] {
  const byId = new Map(nodes.map((n) => [n.id, n]));
  const pathOf = (n: CategoryNode): string => {
    const parts: string[] = [];
    let cur: CategoryNode | undefined = n;
    for (let i = 0; cur && i < 10; i++) { parts.unshift(cur.local_name); cur = byId.get(cur.parent_id); }
    return parts.join(" > ");
  };
  const roots = nodes.filter((n) => APPROVED_TREE.test(n.local_name)).map((n) => n.id);
  const under = (n: CategoryNode): boolean => {
    let cur: CategoryNode | undefined = n;
    for (let i = 0; cur && i < 10; i++) { if (roots.includes(cur.id)) return true; cur = byId.get(cur.parent_id); }
    return false;
  };
  return nodes
    .filter((n) => under(n) || GOLD_WORDS.test(n.local_name))
    .map((n) => ({ id: n.id, path: pathOf(n), isLeaf: !!n.is_leaf, permission: n.permission_statuses ?? [], inApprovedTree: under(n) }))
    .sort((a, b) => a.path.localeCompare(b.path));
}
 
/** Pure: the leaves worth reading rules for, within the cap. Order: gold leaves in the
 *  APPROVED tree (where we may list), other approved-tree leaves, then gold leaves
 *  elsewhere (for comparison only — we are not approved to list there). */
export function pickLeaves(matched: MatchedCategory[]): Array<{ id: string; path: string }> {
  const isGold = (m: MatchedCategory) => GOLD_WORDS.test(m.path.split(" > ").slice(-1)[0] ?? "");
  const rank = (m: MatchedCategory) => (m.inApprovedTree ? (isGold(m) ? 0 : 1) : 2);
  return matched.filter((m) => m.isLeaf && (m.inApprovedTree || isGold(m)))
    .sort((a, b) => rank(a) - rank(b) || a.path.localeCompare(b.path))
    .slice(0, MAX_LEAVES).map(({ id, path }) => ({ id, path }));
}
 
const msg = (e: unknown) => (e instanceof Error ? e.message : String(e)).slice(0, 300);
 
export async function fetchCatalog(creds: Creds, now: Date, http: Http = {}): Promise<CatalogSnapshot> {
  const base = { appKey: creds.appKey, appSecret: creds.appSecret, accessToken: creds.token, shopCipher: creds.shopCipher, fetchImpl: http.fetchImpl };
  const errors: Record<string, string> = {};
  const snap: CatalogSnapshot = {
    at: now.toISOString(), approvedTree: "Jewelry Accessories & Derivatives",
    categories: { total: 0, matched: [] }, leaves: [], brands: [], warehouses: null, products: null, errors,
  };
 
  try {
    const data = await ttsCall<{ categories?: CategoryNode[] }>("/product/202309/categories", {
      ...base, query: { category_version: "v2", locale: "en-US" },
    });
    const nodes = data?.categories ?? [];
    snap.categories = { total: nodes.length, matched: matchCategories(nodes) };
  } catch (e) { errors.categories = msg(e); }
 
  for (const leaf of pickLeaves(snap.categories.matched)) {
    const entry: CatalogSnapshot["leaves"][number] = { ...leaf, rules: null, attributes: null };
    try {
      entry.rules = await ttsCall(`/product/202309/categories/${encodeURIComponent(leaf.id)}/rules`, { ...base, query: { category_version: "v2" } });
      entry.attributes = await ttsCall(`/product/202309/categories/${encodeURIComponent(leaf.id)}/attributes`, { ...base, query: { category_version: "v2", locale: "en-US" } });
    } catch (e) { entry.error = msg(e); }
    snap.leaves.push(entry);
  }
 
  const goldLeaf = pickLeaves(snap.categories.matched)[0]?.id;
  for (const q of BRAND_QUERIES) {
    try {
      const data = await ttsCall<{ brands?: Array<{ id: string; name: string; authorized_status?: string }> }>("/product/202309/brands", {
        ...base, query: { brand_name: q, page_size: "20", category_version: "v2", ...(goldLeaf ? { category_id: goldLeaf } : {}) },
      });
      snap.brands.push({ query: q, found: (data?.brands ?? []).map((b) => ({ id: String(b.id), name: String(b.name), authorized_status: b.authorized_status })) });
    } catch (e) { snap.brands.push({ query: q, found: [], error: msg(e) }); }
  }
 
  try { snap.warehouses = await ttsCall("/logistics/202309/warehouses", base); }
  catch (e) { errors.warehouses = msg(e); }
 
  try {
    const data = await ttsCall<{ total_count?: number; products?: unknown[] }>("/product/202502/products/search", {
      ...base, method: "POST", query: { page_size: "100" }, body: { status: "ALL" },
    });
    snap.products = { total: Number(data?.total_count ?? 0), items: data?.products ?? [] };
  } catch (e) { errors.products = msg(e); }
 
  return snap;
}
 
/** Runs the read and stores the snapshot on the connection row. Null when not connected. */
export async function syncCatalog(db: SupabaseClient, now: Date, http: Http = {}): Promise<CatalogSnapshot | null> {
  const creds = await getShopCredentials(db, now);
  if (!creds) return null;
  const snap = await fetchCatalog(creds, now, http);
  const { data } = await db.from("marketplace_connections").select("api_shapes").eq("channel", "tiktok_shop").maybeSingle();
  const shapes = { ...((data?.api_shapes as Record<string, unknown> | null) ?? {}), tiktokCatalog: snap };
  const { error } = await db.from("marketplace_connections").update({ api_shapes: shapes }).eq("channel", "tiktok_shop");
  if (error) throw new Error(`Could not save the TikTok Shop catalog: ${error.message}`);
  return snap;
}
