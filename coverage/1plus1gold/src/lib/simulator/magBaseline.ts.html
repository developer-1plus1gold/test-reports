 * Pure helpers that turn live MAG data into the MAG pool simulator's starting
 * state and its "from app" reference values (docs/mag-simulator-design.md §2).
 *
 * The route fetches; this decides. Stored defaults live in
 * `logistics_cost_settings` under `mag.<field>` keys so they never collide with
 * the P&L simulator's keys in the same table.
 */
 
import { MAG_BRIEF_STATE, type MagPool, type MagState } from "@/lib/simulator/magEngine";
 
/** MagState fields persisted as defaults. Spot is always live, so it is not stored. */
export const MAG_FIELDS = [
  "barCostPct", "capRatePct", "premOpenPct", "premGroupPct", "premReservedPct",
  "barsPerYear", "n100", "windowMonths", "fillMonths",
  "payFeePct", "sellBackSharePct", "sellBackCostPct", "priceMovePct", "dropOutPct", "vaultPerBarMonth",
] as const satisfies ReadonlyArray<keyof MagState>;
export type MagField = (typeof MAG_FIELDS)[number];

export const MAG_SETTING_KEYS: readonly string[] = MAG_FIELDS.map((f) => `${MAG_KEY_PREFIX}${f}`);
 
/** Bar cost assumed for a bar we have not bought yet. */
export const PLANNED_BAR_COST_PCT = MAG_BRIEF_STATE.barCostPct;
 
const DAYS_PER_MONTH = 365.25 / 12;
 
/** Overlay stored `mag.*` settings on the brief defaults, with live spot. */
export function magStateFromSettings(settings: Partial<Record<string, number>>, spot: number): MagState {
  const s: MagState = { ...MAG_BRIEF_STATE, spot };
  for (const f of MAG_FIELDS) {
    const v = settings[`${MAG_KEY_PREFIX}${f}`];
    if (typeof v === "number" && Number.isFinite(v)) s[f] = v;
  }
  return s;
}

const ACCESS_TO_POOL: Record<MagLiveListing["accessType"], MagPool> = { public: "open", group: "group", reserved: "reserved" };
 
/** Headline (smallest-quantity) tier rate, %. null when the bar type has no tiers. */
export function headlineTierPct(tiers: MagLiveTier[]): number | null {
  if (tiers.length === 0) return null;
  return [...tiers].sort((a, b) => a.min_qty - b.min_qty)[0].premium_pct;
}

const weighted = (rows: Array<{ w: number; v: number | null }>): number | null => {
  const use = rows.filter((r): r is { w: number; v: number } => r.v != null && r.w > 0);
  const w = use.reduce((a, r) => a + r.w, 0);
  return w > 0 ? use.reduce((a, r) => a + r.w * r.v, 0) / w : null;
};
 
export function magReferences(listings: MagLiveListing[], spot: number, now = new Date()): MagReferences {
  const refs: MagListingRef[] = listings.map((l) => {
    const pool = ACCESS_TO_POOL[l.accessType] ?? "open";
    const created = Date.parse(l.createdAt);
    const daysListed = Number.isFinite(created) ? Math.max(0, (now.getTime() - created) / 86_400_000) : 0;
    const gramsSold = Math.max(0, l.gramsTotal - l.gramsRemaining);
    const windowEnd = l.windowEndsAt ? Date.parse(l.windowEndsAt) : NaN;
    const perDay = daysListed > 0 ? gramsSold / daysListed : 0;
    return {
      id: l.id,
      serial: l.serial,
      label: l.label,
      pool,
      gramsTotal: l.gramsTotal,
      gramsSold,
      barCostPct: l.costPerGram > 0 && spot > 0 ? (l.costPerGram / spot - 1) * 100 : null,
      tierPct: headlineTierPct(l.tiers),
      windowMonths: pool !== "open" && Number.isFinite(windowEnd) && Number.isFinite(created)
        ? Math.max(0, (windowEnd - created) / 86_400_000 / DAYS_PER_MONTH) : null,
      observedFillMonths: pool === "open" && perDay > 0 ? l.gramsTotal / perDay / DAYS_PER_MONTH : null,
      daysListed,
    };
  });
 
  const barCost = weighted(refs.map((r) => ({ w: r.gramsTotal, v: r.barCostPct })));
  const tierFor = (p: MagPool) => weighted(refs.filter((r) => r.pool === p).map((r) => ({ w: r.gramsTotal, v: r.tierPct })));
  const n100Count = refs.filter((r) => r.gramsTotal === 100).length;
  const n50Count = refs.filter((r) => r.gramsTotal === 50).length;
  const windows = refs.map((r) => r.windowMonths).filter((v): v is number => v != null);
 
  return {
    listings: refs,
    barCostPct: barCost ?? PLANNED_BAR_COST_PCT,
    barCostSource: barCost == null ? "planned" : "listings",
    tierPct: { open: tierFor("open"), group: tierFor("group"), reserved: tierFor("reserved") },
    n100: n100Count + n50Count > 0 ? (12 * n100Count) / (n100Count + n50Count) : null,
    windowMonths: windows.length ? windows.reduce((a, b) => a + b, 0) / windows.length : null,
  };
}
