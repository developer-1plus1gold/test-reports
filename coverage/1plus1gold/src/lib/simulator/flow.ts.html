/**
 * Swimlane view of the P&L simulator — pure.
 *
 * Re-cuts one `SimResult` into the fulfilment day as the operator lives it
 * (docs/pl-simulator-design.md § Flow view): who acts, at what time, and what
 * each step costs. It adds NO new arithmetic of its own beyond attributing GST
 * to the Airpak/Loomis lines it was charged on, so every lane reconciles to the
 * Monthly P&L to the cent (pinned by simulator-flow.test.ts).
 */
import type { SimResult } from "./plEngine";
 
export type FlowKind = "income" | "variable" | "fixed";
/** Input group on the Detail view a card links back to (anchor id). */
export type FlowGroup = "days" | "loomis" | "airpak-col" | "airpak-deliv" | "airpak-fixed" | "stonex" | "packaging" | "payments" | "products" | "opex";
 
export interface FlowItem {
  key: string;
  label: string;
  /** S$ per month. Costs are positive numbers; `kind` says which way they flow. */
  monthly: number;
  kind: FlowKind;
  group: FlowGroup;
  /** Shown with a dashed border — price not yet agreed. */
  tbc?: boolean;
  note?: string;
}
 
export interface FlowStep {
  key: string;
  label: string;
  /** Time slot from the fulfilment flow (column on desktop). */
  slot: FlowSlot;
  items: FlowItem[];
}
 
export type FlowSlot = "order" | "release" | "collect" | "pack" | "deliver" | "nextday" | "monthly";
 
export const FLOW_SLOTS: Array<{ key: FlowSlot; label: string }> = [
  { key: "order", label: "By 1 PM" },
  { key: "release", label: "2:30 PM" },
  { key: "collect", label: "2:30–3 PM" },
  { key: "pack", label: "3–5 PM" },
  { key: "deliver", label: "5:30–9:30 PM" },
  { key: "nextday", label: "Next day" },
  { key: "monthly", label: "Every month" },
];
 
export interface FlowLane {
  key: string;
  label: string;
  note?: string;
  steps: FlowStep[];
  /** Σ cost items (S$/month). */
  costM: number;
  /** Σ income items (S$/month). */
  incomeM: number;
}
 
export interface WaterfallBar {
  key: string;
  label: string;
  /** Signed S$/month (positive adds, negative subtracts). */
  value: number;
  /** Running total after this bar. */
  running: number;
  total?: boolean;
}
 
export interface FlowModel {
  lanes: FlowLane[];
  waterfall: WaterfallBar[];
  parcelsM: number;
  salesM: number;
  /** Σ every cost item across lanes = fulM + payM + opexM. */
  totalCostM: number;
}
 
export interface ItemView {
  perParcel: number | null;
  pctSales: number | null;
}
 
/** Per-parcel and % of sales for a card, given the mode. */
export function itemView(monthly: number, parcelsM: number, salesM: number): ItemView {
  return {
    perParcel: parcelsM > 0 ? monthly / parcelsM : null,
    pctSales: salesM > 0 ? monthly / salesM : null,
  };
}
 
function lane(key: string, label: string, steps: FlowStep[], note?: string): FlowLane {
  let costM = 0;
  let incomeM = 0;
  for (const st of steps) for (const it of st.items) {
    if (it.kind === "income") incomeM += it.monthly;
    else costM += it.monthly;
  }
  return { key, label, note, steps, costM, incomeM };
}
 
export function buildFlow(r: SimResult): FlowModel {
  const g = r.s.gst / 100;
  const withGst = (v: number) => v * (1 + g);
  const eveM = (r.free + r.paid) * r.s.aEve * r.parcelsM;
  const saverM = r.saver * r.s.aSaver * r.parcelsM;
  const sx = r.s.sxShare > 0;
  const own = r.s.sxShare < 100;
 
  const lanes: FlowLane[] = [
    lane("customer", "Customer & 1Plus1Gold", [
      { key: "order", label: "Order placed · pick list by 1 PM", slot: "order", items: [
        { key: "sales", label: "Sales", monthly: r.salesM, kind: "income", group: "products" },
        { key: "cogs", label: "Cost of gold", monthly: r.costM, kind: "variable", group: "products" },
        { key: "fees", label: "Delivery fees charged", monthly: r.feeM, kind: "income", group: "payments" },
        { key: "pay", label: "Payment fees absorbed", monthly: r.payM, kind: "variable", group: "payments" },
      ] },
    ]),
    lane("stonex", "Sourcing — StoneX", [
      { key: "sx", label: "Release to Loomis", slot: "release", items: sx
        ? [{ key: "sxRelease", label: "StoneX release fees", monthly: r.sxRelease, kind: "fixed", group: "stonex", note: "per collection day" }]
        : [] },
    ]),
    lane("loomis", "Loomis vault", [
      { key: "release", label: "Release at 2:30 PM", slot: "release", items: [
        { key: "lPicks", label: "Picks", monthly: withGst(r.loomisPicks), kind: "fixed", group: "loomis", note: `${r.picksPerDay}/collection day` },
        ...(own ? [
          { key: "lXfer", label: "Transfers in", monthly: withGst(r.loomisXfer), kind: "fixed" as const, group: "loomis" as const },
          { key: "lStore", label: "Storage", monthly: withGst(r.loomisStore), kind: "fixed" as const, group: "loomis" as const },
        ] : []),
      ] },
    ], "No weekend release — weekend orders ship Monday."),
    lane("warehouse", "Airpak — collection & warehouse", [
      { key: "collect", label: "Collect 2:30–3 PM → 45 Changi South", slot: "collect", items: [
        { key: "aCol", label: "Collection run", monthly: withGst(r.airCol), kind: "fixed", group: "airpak-col" },
      ] },
      { key: "pack", label: "Pick & pack 3–5 PM", slot: "pack", items: [
        { key: "aPack", label: "Airpak packing", monthly: withGst(r.airPackM), kind: "variable", group: "airpak-col" },
        { key: "pkg", label: "Our packaging", monthly: r.pkgM, kind: "variable", group: "packaging" },
      ] },
    ]),
    lane("delivery", "Airpak — delivery", [
      { key: "eve", label: "Evening run 5:30–9:30 PM (POD)", slot: "deliver", items: [
        { key: "aEve", label: "Evening delivery", monthly: withGst(eveM), kind: "variable", group: "airpak-deliv" },
        { key: "aRe", label: "Re-attempts", monthly: withGst(r.airFailM), kind: "variable", group: "airpak-deliv", note: "failed parcels held overnight at Airpak" },
      ] },
      { key: "saver", label: "Saver — next day", slot: "nextday", items: r.saver > 0
        ? [{ key: "aSaver", label: "Saver delivery", monthly: withGst(saverM), kind: "variable", group: "airpak-deliv", tbc: true, note: "rate to be quoted" }]
        : [] },
    ]),
    lane("fixed", "Fixed / monthly", [
      { key: "monthly", label: "Every month, whatever the volume", slot: "monthly", items: [
        { key: "aFlat", label: "Airpak flat fee", monthly: withGst(r.s.aFlat), kind: "fixed", group: "airpak-fixed" },
        { key: "aStore", label: "Airpak storage", monthly: withGst(r.s.aStore), kind: "fixed", group: "airpak-fixed" },
        { key: "opex", label: "Operating costs (opex)", monthly: r.opexM, kind: "fixed", group: "opex" },
      ] },
    ]),
  ].filter((l) => l.steps.some((s) => s.items.length > 0));
 
  const laneCost = (k: string) => lanes.find((l) => l.key === k)?.costM ?? 0;
  const airFixedM = withGst(r.airFlat);
  const bars: Array<Omit<WaterfallBar, "running">> = [
    { key: "gp", label: "Gross profit", value: r.gpM, total: true },
    { key: "fees", label: "Delivery fees", value: r.feeM },
    { key: "pay", label: "Payment fees", value: -r.payM },
    { key: "stonex", label: "StoneX", value: -laneCost("stonex") },
    { key: "loomis", label: "Loomis", value: -laneCost("loomis") },
    { key: "warehouse", label: "Collection & packing", value: -laneCost("warehouse") },
    { key: "delivery", label: "Delivery", value: -laneCost("delivery") },
    { key: "airFixed", label: "Airpak fixed fees", value: -airFixedM },
    { key: "contrib", label: "Contribution", value: 0, total: true },
    { key: "opex", label: "Opex", value: -r.opexM },
    { key: "net", label: "Net profit", value: 0, total: true },
  ];
  let running = 0;
  const waterfall: WaterfallBar[] = bars.map((b) => {
    if (b.key === "gp") running = b.value;
    else if (!b.total) running += b.value;
    return { ...b, value: b.total ? running : b.value, running };
  });
 
  const totalCostM = lanes.reduce((a, l) => a + l.costM, 0) - r.costM;
  return { lanes, waterfall, parcelsM: r.parcelsM, salesM: r.salesM, totalCostM };
}
