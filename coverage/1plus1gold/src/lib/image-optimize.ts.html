/**
 * Customer-facing image optimization — the upload standard.
 *
 * Every product/shop image is stored as WebP, resized to a sane max edge.
 * WebP is ~90% smaller than raw PNG/JPEG at visually-lossless quality and
 * handles transparency + smooth gradients (gold bars) without the banding
 * that palette-PNG introduces. One helper so web upload, any future Android
 * upload, and batch re-encodes all produce identical output (no drift).
 *
 * Adopted 2026-06-10 after the Home/Shop images were found to be served raw
 * (2.81MB of product PNGs → 278KB WebP). Wired into the product media upload
 * route so NEW uploads are optimized automatically, not just this batch.
 *
 * Animated GIFs are preserved as animated WebP. Non-image bytes must not be
 * passed here — the caller validates the magic bytes first.
 */
 
import sharp from "sharp";
 
export interface OptimizeResult {
  data: Buffer;
  contentType: "image/webp";
  ext: "webp";
  width: number | null;
  height: number | null;
  bytes: number;
}
 
/**
 * Re-encode an image to WebP, resized to fit within `maxDim`×`maxDim`
 * (aspect preserved, never upscaled). Throws if the bytes aren't a
 * decodable image — callers that must not fail the upload should catch and
 * fall back to the original bytes.
 */
export async function optimizeImageToWebp(
  input: Uint8Array | Buffer,
  opts: { maxDim?: number; quality?: number } = {},
): Promise<OptimizeResult> {
  const maxDim = opts.maxDim ?? 1600;
  const quality = opts.quality ?? 88;
 
  const meta = await sharp(input).metadata();
  const animated = (meta.pages ?? 1) > 1;
 
  const data = await sharp(input, animated ? { animated: true } : {})
    .rotate() // honor EXIF orientation before we drop the metadata
    .resize(maxDim, maxDim, { fit: "inside", withoutEnlargement: true })
    .webp({ quality, effort: 4 })
    .toBuffer();
 
  // Re-read dimensions from the output (resize may have changed them).
  const outMeta = await sharp(data).metadata();
  return {
    data,
    contentType: "image/webp",
    ext: "webp",
    width: outMeta.width ?? null,
    height: outMeta.height ?? null,
    bytes: data.byteLength,
  };
}
 
/**
 * Re-encode to WebP onto a FIXED aspect-ratio canvas (letterbox/matte), so a
 * set of heterogeneous source images renders consistently inside a fixed-aspect
 * frame (e.g. the home carousel's 16:9 slides). `fit:"contain"` preserves the
 * WHOLE image — the remainder is filled with `background` (brand dark by
 * default); nothing is cropped. The output is always exactly width×height, so
 * an `object-cover` (or `object-contain`) frame of the same ratio shows it with
 * zero distortion. Alpha is flattened onto the matte so transparent PNGs don't
 * render see-through. Throws on undecodable bytes.
 */
export async function padToAspectWebp(
  input: Uint8Array | Buffer,
  opts: { width?: number; height?: number; background?: string; quality?: number } = {},
): Promise<OptimizeResult> {
  const width = opts.width ?? 1600;
  const height = opts.height ?? 900;        // 16:9
  const quality = opts.quality ?? 88;
  const background = opts.background ?? "#16110A"; // --color-bg
 
  const data = await sharp(input)
    .rotate() // honor EXIF orientation before drop
    .resize(width, height, { fit: "contain", background })
    .flatten({ background }) // drop alpha onto the matte (else PNGs go see-through)
    .webp({ quality, effort: 4 })
    .toBuffer();
 
  const outMeta = await sharp(data).metadata();
  return {
    data,
    contentType: "image/webp",
    ext: "webp",
    width: outMeta.width ?? null,
    height: outMeta.height ?? null,
    bytes: data.byteLength,
  };
}
