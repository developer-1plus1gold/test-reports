/**
 * Signing links for a co-signer with no admin access (Meena, 2026-11-17).
 *
 * The flow, and what each step proves:
 *   1. An admin creates a request for ONE document, ONE signer, and the phone
 *      the code goes to. The link carries a 256-bit token; only its sha256 is
 *      stored, so the database cannot be read back into a working link.
 *   2. Whoever opens the link can ask for a code — but the code is sent to the
 *      phone on the REQUEST, never to a number the visitor types. Holding the
 *      link is not enough; you must also hold that phone.
 *   3. The correct code opens a 30-minute signing session for that request only
 *      (an HMAC cookie). It grants the document and the Sign button, nothing else.
 *   4. The signature must be the signer's own name, and is accepted only while
 *      the document is unchanged since the link was made (`content_sha256`).
 *
 * Secrets: OTP_SECRET, the same HMAC key as WhatsApp sign-in, with distinct
 * message prefixes (`gov:` / `gov-sign:`) so a sign-in code can never be
 * replayed here or the other way round.
 */

/** Pure: is this request still usable at `now`? */
export function classifyRequest(row: SignRequestRow | null, now = Date.now()): RequestLookup {
  if (!row) return { ok: false, reason: "not_found" };
  if (row.revoked_at) return { ok: false, reason: "revoked" };
  if (row.used_at) return { ok: false, reason: "used" };
  if (new Date(row.expires_at).getTime() <= now) return { ok: false, reason: "expired" };
  return { ok: true, row };
}
 
export async function findRequestByToken(token: string): Promise<RequestLookup> {
  if (!isTokenShaped(token)) return { ok: false, reason: "not_found" };
  const { data, error } = await db().from("governance_sign_requests").select(COLUMNS)
    .eq("token_sha256", hashToken(token)).maybeSingle();
  if (error) throw new Error(`governance_sign_requests read failed: ${error.message}`);
  return classifyRequest((data as SignRequestRow | null) ?? null);
}
 
export async function createSignRequest(args: {
  docKey: string; signerKey: string; phone: string; contentSha256: string; docVersion: string; createdBySession: string | null;
}): Promise<{ token: string; expiresAt: string }> {
  const token = newToken();
  const expiresAt = new Date(Date.now() + SIGN_LINK_TTL_MS).toISOString();
  const client = db();
  // One live link per document and signer: creating a new one revokes the old.
  const { error: revokeErr } = await client.from("governance_sign_requests")
    .update({ revoked_at: new Date().toISOString() })
    .eq("doc_key", args.docKey).eq("signer_key", args.signerKey)
    .is("used_at", null).is("revoked_at", null);
  if (revokeErr) throw new Error(`governance_sign_requests revoke failed: ${revokeErr.message}`);
  const { error } = await client.from("governance_sign_requests").insert({
    doc_key: args.docKey, signer_key: args.signerKey, token_sha256: hashToken(token),
    phone_e164: args.phone, content_sha256: args.contentSha256, doc_version: args.docVersion,
    expires_at: expiresAt, created_by_session: args.createdBySession,
  });
  if (error) throw new Error(`governance_sign_requests insert failed: ${error.message}`);
  return { token, expiresAt };
}
