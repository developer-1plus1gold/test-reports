/**
 * The internal-governance document registry — ONE list, shared.
 *
 * Before 2026-09-08 the admin page held its own array of documents and the
 * serving route held an independent allow-list, and a test existed purely to
 * assert the two agreed (a row added to one and not the other was a dead link
 * the UI rendered happily). They are now the same object: the page renders what
 * this file declares and the route serves what this file declares, so the two
 * cannot disagree.
 *
 * This module is PURE DATA — no `node:fs`, no server-only import — because the
 * admin page is a client component and imports it directly for labels.
 *
 * ⚠️ `lastUpdated` for a `raw` document is DECLARED here, and for an `html`
 * document it is read from the file's own frontmatter at request time. It is
 * never taken from the filesystem mtime: a fresh clone or a Vercel build stamps
 * every file with the build time, so an mtime-derived "last updated" would read
 * as "today" forever and tell the reader nothing.
 */
 
import type { SignerKey } from "./signers";
 
export interface GovernanceDoc {
  /** `?doc=` key. Also the allow-list — nothing outside this file is servable. */
  key: string;
  label: string;
  /** One line under the label in the hub, so the operator knows what it is. */
  blurb: string;
  /**
   * Body fragment + frontmatter in content/governance/, wrapped at request time
   * in the house letterhead. Every document, one renderer — there is no second
   * shape (operator decision, 2026-09-11). Before that the IPPC was served as
   * its own bytes and the AML reference downloaded as a .docx, which meant
   * three code paths, three CSPs and, for the .docx, a format no grep could
   * read. Adding a kind here would reopen all of it.
   */
  file: string;
  version?: string;
  /**
   * Named signers this document REQUIRES (2026-11-17). Absent → the usual single
   * signature by any admin. Present → "signed" only when every listed signer
   * has signed the current text; see `resolveCosignState`.
   */
  signers?: SignerKey[];
}

/**
 * "2026-09-08" → "8 September 2026". Anything unparseable passes through
 * unchanged, so a malformed date is visible rather than silently blanked.
 * Lives here (pure data, no `node:fs`) because the admin hub is a client
 * component and must be able to import it without pulling the document
 * letterhead's stylesheet into the browser bundle.
 */
export function humanDate(iso: string | null | undefined): string {
  if (!iso || !/^\d{4}-\d{2}-\d{2}$/.test(iso)) return iso ?? "";
  const [y, m, d] = iso.split("-").map(Number);
  const months = ["January", "February", "March", "April", "May", "June",
    "July", "August", "September", "October", "November", "December"];
  return `${d} ${months[m - 1]} ${y}`;
}
 
/**
 * The customer-facing policies the hub lists. The text lives in content/<slug>.md
 * and is changed via code review; this is only the display registry.
 *
 * ⚠️ `consentBearing` is not decoration. `getCurrentConsentVersions()` reads each
 * policy's `last_updated` frontmatter and stamps THAT string onto a profile at
 * first sign-in — so for Terms and Privacy the "last updated" date IS the consent
 * version in force, and editing the .md is the version bump. Showing the date
 * here is therefore showing the live consent version, not a nicety. The refund
 * policy carries no consent stamp.
 */
export interface PolicySpec {
  slug: string;
  label: string;
  href: string;
  purpose: string;
  consentBearing: boolean;
}
