/**
 * Vault liquidation — shared helpers for the "sell vaulted gold back to us" flow.
 *
 * A vault holding that is still in our custody (stored / mid-withdrawal /
 * mid-liquidation) must NOT appear as sellable in the ordinary pickup-based
 * buyback (we can't pick up what we already hold, and double-counting the bar
 * would corrupt availability). `IN_CUSTODY_STATUSES` is that set; terminal
 * states (`withdrawn` = physically returned → eligible again; `liquidated` =
 * already sold back) are excluded.
 *
 * The reconciliation helpers fire from the admin buyback lifecycle:
 *   - completed  → holding becomes `liquidated` and the bar comes back into our
 *                  sellable stock as a NEW inventory row (source 'buyback',
 *                  cost basis = buyback payout/g). The original row stays
 *                  'sold' with its original cost and its link to the customer's
 *                  order. See `completeVaultLiquidation` for why.
 *   - rejected / cancelled → holding reverts to `stored`.
 */
 
import type { SupabaseClient } from "@supabase/supabase-js";
 
// Vault statuses that mean "the bar is ours to hold for the customer" → counts
// toward their custody balance and blocks the ordinary pickup buyback for that
// unit. Includes 'awaiting_storage' (ordered + bar allocated, physical storage
// not yet completed) — the customer owns it, so it shows in My Vault from
// purchase and can't be double-sold via the ordinary pickup buyback.
export const IN_CUSTODY_STATUSES = [
  "awaiting_storage",
  "stored",
  "withdraw_requested",
  "withdraw_dispatched",
  "liquidation_requested",
] as const;
 
/**
 * Count active (in-custody) vault holdings per order_item_id, so the buyback
 * eligibility paths can subtract them from available quantity — the same
 * partial-qty accounting used for buybacks and conversions.
 */
export async function getInCustodyVaultCounts(
  client: SupabaseClient,
  orderItemIds: string[],
): Promise<Map<string, number>> {
  const counts = new Map<string, number>();
  if (!orderItemIds.length) return counts;
 
  const { data } = await client
    .from("vault_holdings")
    .select("order_item_id, status")
    .in("order_item_id", orderItemIds)
    .in("status", [...IN_CUSTODY_STATUSES]);
 
  for (const row of (data ?? []) as { order_item_id: string }[]) {
    counts.set(row.order_item_id, (counts.get(row.order_item_id) ?? 0) + 1);
  }
  return counts;
}
