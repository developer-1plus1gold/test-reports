/**
 * Spot price fetcher — inlined from the retired Railway price service.
 *
 * Triple-tier resilience:
 *   Tier 1  XAU: goldapi.io      (paid unlimited plan, primary)
 *   Tier 2  XAU: metals-api.com  (graceful no-op if METALS_API_KEY unset)
 *   Tier 3  any: Supabase singleton `spot_price_cache` table (last-known-good)
 *
 *   Tier 1  FX:  fastforex.io    (premium interbank, primary if FASTFOREX_KEY)
 *   Tier 2  FX:  open.er-api.com (free, no key, bank-aggregated)
 *
 * The lib is the single source of truth for spot data inside Next.js. Routes
 * import `getSpotPayload()` directly; HTTP fan-out happens at the edge cache
 * layer via `/api/spot-price` and `/api/price` (legacy-compat shape).
 *
 * Caching strategy:
 *   - Module-level in-memory cache per warm function instance (~30s)
 *   - Vercel edge cache via response Cache-Control headers
 *   - Supabase singleton-row cache as Tier 3 — survives cold starts and
 *     total upstream outage (up to the payment-route staleness gate window:
 *     XAU 5 min hard block, FX 15 min hard block).
 *
 * Order routes still hard-gate stale data via `getPriceStaleness()` — so a
 * Tier 3 cache hit that's older than the block thresholds will correctly
 * surface as a 503 on `/api/payments/.../create-intent`. Catalogue browsing
 * tolerates older cache values (read-only, no commitment).
 */
// NOTE: `server-only` was tried here but breaks vitest's transform layer.
// The lib is naturally server-only via its `@supabase/supabase-js` service-role
// client + `process.env` access; no client component imports it.
import { createClient, type SupabaseClient } from "@supabase/supabase-js";
 
const TROY_OZ = 31.1035;
const STALE_THRESHOLD_MS = parseInt(process.env.PRICE_STALE_THRESHOLD_MS || "60000", 10);
const XAU_BLOCK_THRESHOLD_MS = parseInt(process.env.XAU_BLOCK_THRESHOLD_MS || String(5 * 60 * 1000), 10);
const FX_BLOCK_THRESHOLD_MS = parseInt(process.env.FX_BLOCK_THRESHOLD_MS || String(15 * 60 * 1000), 10);
 
const XAU_FETCH_TTL_MS = 5_000;
const FX_FETCH_TTL_MS = 60_000;
const CHP_FETCH_TTL_MS = 15 * 60_000;
 
// The currencies we fetch live FX rates for (USD-quoted source, so USD itself
// is the implicit base and not listed). MUST cover every CURRENCIES key except
// USD — a currency in CURRENCIES but missing here has no rate → shows "Loading…"
// forever (the TWD-launch gap, 2026-06-30). Guarded by a unit test.
export const FX_CURRENCIES = [
  "SGD", "EUR", "GBP", "AUD", "JPY", "MYR", "CNY", "INR",
  "HKD", "AED", "CHF", "CAD", "THB", "KRW", "PHP", "IDR", "BDT", "LKR", "TWD",
];
 
type XauTick = {
  priceOz: number;
  priceGram: number;
  chp: number;
  ch: number;
  timestamp: number;
  source: "goldapi" | "metals_api" | "cache";
};
type FxRates = {
  rates: Record<string, number>;
  timestamp: number;
  source: "fastforex" | "metals_dev" | "open_er_api" | "cache";
};
 
const cache: {
  xau: XauTick | null;
  xauInflight: Promise<void> | null;
  fx: FxRates | null;
  fxInflight: Promise<void> | null;
  chp: Record<string, number>;
  chpAt: number;
  chpInflight: Promise<void> | null;
  supabaseHydrated: boolean;
} = {
  xau: null,
  xauInflight: null,
  fx: null,
  fxInflight: null,
  chp: {},
  chpAt: 0,
  chpInflight: null,
  supabaseHydrated: false,
};
 
// ─── Supabase singleton-row cache (Tier 3) ─────────────────────────────────────
 
type SupabaseCacheRow = {
  xau: XauTick | null;
  fx: FxRates | null;
  chp: Record<string, number>;
};
 
let supabaseClient: SupabaseClient | null = null;
function getSupabase(): SupabaseClient | null {
  if (supabaseClient) return supabaseClient;
  const url = process.env.NEXT_PUBLIC_SUPABASE_URL;
  const key = process.env.SUPABASE_SERVICE_KEY;
  if (!url || !key) return null;
  supabaseClient = createClient(url, key, { auth: { persistSession: false } });
  return supabaseClient;
}
 
async function hydrateFromSupabase(): Promise<void> {
  if (cache.supabaseHydrated) return;
  cache.supabaseHydrated = true; // attempt once per warm instance
  const sb = getSupabase();
  if (!sb) return;
  try {
    const { data } = await sb
      .from("spot_price_cache")
      .select("payload")
      .eq("id", 1)
      .maybeSingle();
    if (!data?.payload) return;
    const payload = data.payload as SupabaseCacheRow;
    if (!cache.xau && payload.xau) cache.xau = { ...payload.xau, source: "cache" };
    if (!cache.fx && payload.fx) cache.fx = { ...payload.fx, source: "cache" };
    if (Object.keys(cache.chp).length === 0 && payload.chp) cache.chp = payload.chp;
  } catch (err) {
    console.warn("[spot] supabase hydrate failed:", err instanceof Error ? err.message : err);
  }
}
 
async function persistToSupabase(): Promise<void> {
  const sb = getSupabase();
  if (!sb) return;
  if (!cache.xau || !cache.fx) return;
  // Only persist live (non-cache-source) data so we don't loop cache→cache.
  if (cache.xau.source === "cache" && cache.fx.source === "cache") return;
  try {
    const payload: SupabaseCacheRow = {
      xau: cache.xau.source === "cache" ? cache.xau : cache.xau,
      fx: cache.fx.source === "cache" ? cache.fx : cache.fx,
      chp: cache.chp,
    };
    await sb.from("spot_price_cache").upsert(
      { id: 1, payload, updated_at: new Date().toISOString() },
      { onConflict: "id" },
    );
  } catch (err) {
    console.warn("[spot] supabase persist failed:", err instanceof Error ? err.message : err);
  }
}
 
// ─── XAU/USD: goldapi (Tier 1) → metals-api (Tier 2) ──────────────────────────
 
async function tryGoldApi(): Promise<XauTick | null> {
  const key = process.env.GOLDAPI_KEY;
  if (!key) return null;
  try {
    const res = await fetch("https://www.goldapi.io/api/XAU/USD", {
      headers: { "x-access-token": key, "Content-Type": "application/json" },
      signal: AbortSignal.timeout(8_000),
      cache: "no-store",
    });
    if (res.status === 429) { console.warn("[spot] goldapi 429"); return null; }
    if (!res.ok) throw new Error(`HTTP ${res.status}`);
    const data = await res.json();
    const priceOz = data.price;
    if (!priceOz || isNaN(priceOz)) throw new Error("no valid price");
    return {
      priceOz,
      priceGram: priceOz / TROY_OZ,
      chp: typeof data.chp === "number" ? data.chp : 0,
      ch: typeof data.ch === "number" ? data.ch : 0,
      timestamp: Date.now(),
      source: "goldapi",
    };
  } catch (err) {
    console.warn("[spot] goldapi failed:", err instanceof Error ? err.message : err);
    return null;
  }
}
 
async function tryMetalsApi(): Promise<XauTick | null> {
  const key = process.env.METALS_API_KEY;
  if (!key) return null;
  try {
    // metals.dev /v1/latest returns spot directly: metals.gold = USD per troy
    // ounce (when unit=toz). No inversion needed. Free tier 100 req/mo or
    // paid 10k for $19/mo — sized to absorb a multi-hour goldapi outage.
    const res = await fetch(`https://api.metals.dev/v1/latest?api_key=${key}&currency=USD&unit=toz`, {
      signal: AbortSignal.timeout(8_000),
      cache: "no-store",
    });
    if (!res.ok) throw new Error(`HTTP ${res.status}`);
    const data = await res.json();
    if (data.status !== "success") throw new Error(data.error_message || "metals.dev unsuccessful");
    const priceOz = data.metals?.gold;
    if (typeof priceOz !== "number" || priceOz <= 0) throw new Error("no gold price");
    // metals.dev /latest doesn't include 24h change; keep chp/ch as
    // previously known so the UI doesn't flicker to 0%.
    return {
      priceOz,
      priceGram: priceOz / TROY_OZ,
      chp: cache.xau?.chp ?? 0,
      ch: cache.xau?.ch ?? 0,
      timestamp: Date.now(),
      source: "metals_api",
    };
  } catch (err) {
    console.warn("[spot] metals.dev failed:", err instanceof Error ? err.message : err);
    return null;
  }
}
 
async function refreshXau(): Promise<void> {
  const tick = (await tryGoldApi()) ?? (await tryMetalsApi());
  if (tick) {
    cache.xau = tick;
    // Persist whenever we got a live tick (fire-and-forget).
    await persistToSupabase();
  }
  // Otherwise preserve cache.xau (memory) or fall through to Supabase hydrate on read.
}
 
// ─── FX rates: fastforex (Tier 1) → metals.dev (Tier 2) → open.er-api (Tier 3)
 
async function tryFastForexFx(): Promise<FxRates | null> {
  const key = process.env.FASTFOREX_KEY;
  if (!key) return null;
  try {
    const targets = FX_CURRENCIES.join(",");
    const res = await fetch(`https://api.fastforex.io/fetch-multi?from=USD&to=${targets}`, {
      headers: { Authorization: `Bearer ${key}` },
      signal: AbortSignal.timeout(8_000),
      cache: "no-store",
    });
    if (!res.ok) throw new Error(`HTTP ${res.status}`);
    const data = await res.json();
    const results = data.results;
    if (!results || typeof results !== "object" || Object.keys(results).length === 0) {
      throw new Error("empty results");
    }
    const rates: Record<string, number> = { USD: 1 };
    for (const cc of FX_CURRENCIES) {
      const v = results[cc];
      if (typeof v !== "number" || !isFinite(v) || v <= 0) throw new Error(`bad rate for ${cc}: ${v}`);
      rates[cc] = v;
    }
    return { rates, timestamp: Date.now(), source: "fastforex" };
  } catch (err) {
    console.warn("[spot] fastforex failed:", err instanceof Error ? err.message : err);
    return null;
  }
}
 
async function tryMetalsDevFx(): Promise<FxRates | null> {
  // metals.dev /v1/latest returns gold + 174 currencies in one call.
  // Values are USD-per-1-unit (inverse of fastforex's USD-to-X). Update
  // cadence ~120s — slots BETWEEN fastforex (60s) and open.er-api (24h
  // daily). Shares METALS_API_KEY with the XAU Tier-2 path.
  const key = process.env.METALS_API_KEY;
  if (!key) return null;
  try {
    const res = await fetch(`https://api.metals.dev/v1/latest?api_key=${key}&currency=USD&unit=toz`, {
      signal: AbortSignal.timeout(8_000),
      cache: "no-store",
    });
    if (!res.ok) throw new Error(`HTTP ${res.status}`);
    const data = await res.json();
    if (data.status !== "success") throw new Error(data.error_message || "metals.dev unsuccessful");
    const cc = data.currencies;
    if (!cc || typeof cc !== "object") throw new Error("no currencies in response");
    const rates: Record<string, number> = { USD: 1 };
    for (const code of FX_CURRENCIES) {
      const usdPerUnit = cc[code];
      if (typeof usdPerUnit !== "number" || !isFinite(usdPerUnit) || usdPerUnit <= 0) {
        throw new Error(`bad currency for ${code}: ${usdPerUnit}`);
      }
      rates[code] = 1 / usdPerUnit;
    }
    return { rates, timestamp: Date.now(), source: "metals_dev" };
  } catch (err) {
    console.warn("[spot] metals.dev FX failed:", err instanceof Error ? err.message : err);
    return null;
  }
}
 
async function tryOpenErApiFx(): Promise<FxRates | null> {
  try {
    const res = await fetch("https://open.er-api.com/v6/latest/USD", {
      signal: AbortSignal.timeout(8_000),
      cache: "no-store",
    });
    if (!res.ok) throw new Error(`HTTP ${res.status}`);
    const data = await res.json();
    if (!data.rates || typeof data.rates !== "object") throw new Error("no rates in response");
    const rates: Record<string, number> = { USD: 1 };
    for (const cc of FX_CURRENCIES) {
      const v = data.rates[cc];
      if (typeof v === "number" && isFinite(v) && v > 0) rates[cc] = v;
    }
    if (Object.keys(rates).length <= 1) throw new Error("no usable currencies");
    return { rates, timestamp: Date.now(), source: "open_er_api" };
  } catch (err) {
    console.error("[spot] open.er-api failed:", err instanceof Error ? err.message : err);
    return null;
  }
}
 
async function refreshFx(): Promise<void> {
  const rates =
    (await tryFastForexFx()) ??
    (await tryMetalsDevFx()) ??
    (await tryOpenErApiFx());
  if (rates) {
    cache.fx = rates;
    await persistToSupabase();
  }
  // Otherwise preserve cache.fx — Tier 4 (Supabase hydrate on read) handles
  // the case where even cache.fx is empty.
}
 
// ─── Per-currency 24h chp (goldapi only) ───────────────────────────────────────
 
async function refreshChp(): Promise<void> {
  const key = process.env.GOLDAPI_KEY;
  if (!key) return;
  const updated: Record<string, number> = {};
  for (const cc of FX_CURRENCIES) {
    try {
      const res = await fetch(`https://www.goldapi.io/api/XAU/${cc}`, {
        headers: { "x-access-token": key, "Content-Type": "application/json" },
        signal: AbortSignal.timeout(8_000),
        cache: "no-store",
      });
      if (res.ok) {
        const data = await res.json();
        if (typeof data.chp === "number") updated[cc] = data.chp;
      }
    } catch {
      // per-currency failure is non-fatal
    }
  }
  if (Object.keys(updated).length > 0) {
    cache.chp = updated;
    cache.chpAt = Date.now();
    await persistToSupabase();
  }
}
 
// ─── Refresh orchestration ─────────────────────────────────────────────────────
 
async function ensureXau(): Promise<void> {
  if (cache.xau && cache.xau.source !== "cache" && Date.now() - cache.xau.timestamp < XAU_FETCH_TTL_MS) return;
  if (cache.xauInflight) return cache.xauInflight;
  cache.xauInflight = refreshXau().finally(() => { cache.xauInflight = null; });
  return cache.xauInflight;
}
 
async function ensureFx(): Promise<void> {
  if (cache.fx && cache.fx.source !== "cache" && Date.now() - cache.fx.timestamp < FX_FETCH_TTL_MS) return;
  if (cache.fxInflight) return cache.fxInflight;
  cache.fxInflight = refreshFx().finally(() => { cache.fxInflight = null; });
  return cache.fxInflight;
}
 
async function ensureChp(): Promise<void> {
  if (cache.chpAt && Date.now() - cache.chpAt < CHP_FETCH_TTL_MS) return;
  if (cache.chpInflight) return cache.chpInflight;
  cache.chpInflight = refreshChp().finally(() => { cache.chpInflight = null; });
}
 
// ─── Public payload ────────────────────────────────────────────────────────────
 
export type SpotPayload = {
  xauUsd: number;
  xauUsdGram: number;
  chp: number;
  ch: number;
  perCurrencyChp: Record<string, number>;
  usdRates: Record<string, number>;
  currencies: Record<string, { gram: number; oz: number }>;
  timestamp: number;
  ratesSource: "fastforex" | "metals_dev" | "open_er_api" | "cache";
  stale: boolean;
  xauAgeMs: number;
  fxAgeMs: number;
  xauStale: boolean;
  fxStale: boolean;
  source: "goldapi" | "metals_api" | "cache";
};
 
// Freshness window for the Supabase-hydrated cache fast-path. If both XAU
// + FX values from Supabase are younger than this, we serve them directly
// and kick a background refresh — the upstream goldapi/fastforex calls
// don't block the response. Drops cold-start /api/spot-price from
// 500-2000ms to ~50-100ms when Supabase has recent data. Wider than the
// ensureXau/Fx in-memory TTLs (XAU 5s, FX 60s) because Supabase data is
// known to be at most ~10s stale from the writer's perspective.
const SUPABASE_FAST_PATH_XAU_MS = 30_000;
const SUPABASE_FAST_PATH_FX_MS  = 90_000;
 
export async function getSpotPayload(): Promise<SpotPayload | { error: "no_data" | "rates_unavailable" }> {
  // On a cold instance, try to seed cache from Supabase first so even total
  // upstream failure produces a usable response.
  await hydrateFromSupabase();
 
  const now = Date.now();
  const xauFresh = cache.xau != null && now - cache.xau.timestamp < SUPABASE_FAST_PATH_XAU_MS;
  const fxFresh  = cache.fx  != null && now - cache.fx.timestamp  < SUPABASE_FAST_PATH_FX_MS;
 
  if (xauFresh && fxFresh) {
    // Fast path — Supabase had fresh values. Kick non-blocking refreshes
    // so the next request benefits from upstream-fresh data; this one
    // returns immediately.
    void ensureXau();
    void ensureFx();
  } else {
    // Slow path (older cache or first-ever cold start). Await upstream
    // so we don't serve stale data to the caller.
    await Promise.all([ensureXau(), ensureFx()]);
  }
  void ensureChp();
 
  if (!cache.xau) return { error: "no_data" };
  if (!cache.fx) return { error: "rates_unavailable" };
 
  const xauAgeMs = now - cache.xau.timestamp;
  const fxAgeMs = now - cache.fx.timestamp;
  const stale = xauAgeMs > STALE_THRESHOLD_MS;
  const xauStale = xauAgeMs > XAU_BLOCK_THRESHOLD_MS;
  const fxStale = fxAgeMs > FX_BLOCK_THRESHOLD_MS;
 
  const currencies: Record<string, { gram: number; oz: number }> = {};
  for (const [cc, rate] of Object.entries(cache.fx.rates)) {
    currencies[cc] = {
      gram: parseFloat((cache.xau.priceGram * rate).toFixed(6)),
      oz: parseFloat((cache.xau.priceOz * rate).toFixed(4)),
    };
  }
 
  return {
    xauUsd: cache.xau.priceOz,
    xauUsdGram: cache.xau.priceGram,
    chp: cache.xau.chp,
    ch: cache.xau.ch,
    perCurrencyChp: { USD: cache.xau.chp, ...cache.chp },
    usdRates: cache.fx.rates,
    currencies,
    timestamp: cache.xau.timestamp,
    ratesSource: cache.fx.source,
    stale,
    xauAgeMs,
    fxAgeMs,
    xauStale,
    fxStale,
    source: cache.xau.source,
  };
}
