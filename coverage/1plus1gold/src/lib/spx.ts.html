 * SPX (Shopee Xpress) Open Platform client — Singapore.
 *
 * Full contract + the VERIFIED signing algorithm: docs/spx-api-reference.md.
 * SPX is functionally NinjaVan-like (fixed price, no live quote) but authenticates
 * per-request with an HMAC-SHA256 signature (no OAuth token step, so this is
 * simpler than src/lib/ninjavan.ts).
 *
 * Auth (verified against SPX's published test vector — see spx-signing.test.ts):
 *   check-sign = HMAC_SHA256(app_secret, `${app_id}_${timestamp}_${random_num}_${payload}`)  (lowercase hex)
 *   payload    = JSON.stringify(body)   // the EXACT bytes POSTed; sign once, send those bytes
 *   timestamp  = Unix SECONDS
 * Headers on every call: app-id, timestamp, random-num, check-sign, Content-Type: application/json.
 * Body of every order/account/fee call also carries user_id + user_secret.
 *
 * Env vars (Vercel DEV=sandbox, PROD=live; never committed):
 *   SPX_BASE_URL   — https://test-stable.spx.sg (sandbox) | https://spx.sg (live)
 *   SPX_APP_ID     — integer, request signing + webhook verification
 *   SPX_APP_SECRET — HMAC key (also verifies inbound webhooks)
 *   SPX_USER_ID    — integer, order/account body auth
 *   SPX_USER_SECRET— string, order/account body auth
 *   SPX_SENDER_*   — dispatch origin (see senderInfo)
 */

/**
 * check-sign = HMAC_SHA256(app_secret, `${appId}_${timestamp}_${randomNum}_${payload}`) → lowercase hex.
 * `payload` MUST be the exact JSON string sent as the request body.
 */
export function generateCheckSign(
  appId: number,
  appSecret: string,
  timestampSec: number,
  randomNumValue: number,
  payload: string,
): string {
  const message = `${appId}_${timestampSec}_${randomNumValue}_${payload}`;
  return createHmac("sha256", appSecret).update(message, "utf8").digest("hex");
}
 
// Random signed-64-bit-safe integer for `random-num` (anti-replay/dedup). Only
// needs to vary per request, not be cryptographically secure.
function randomNum(): number {
  // eslint-disable-next-line no-restricted-properties -- non-crypto nonce, only needs to differ per request
  return Math.floor(Math.random() * 9_000_000_000_000) + 1_000_000_000_000;
}

/**

  if (!o) return null;
  return { trackingNo: o.tracking_no, statusCode: o.status_code, status: o.status, trackingLink: o.tracking_link };
}

/**

 * Unknown/future codes → 'in_transit' (don't fail-closed on an unrecognized code).
 */
export function mapSpxEvent(statusCode: string | number | null | undefined): string {
  if (statusCode == null) return "in_transit";
  return SPX_STATUS_MAP[String(statusCode)] ?? "in_transit";
}
 
// ─── Webhook signature verification ────────────────────────────────────────────
//
// SPX signs the webhook the SAME way as outbound (per the docs): check-sign =
// HMAC(app_secret, `${app-id}_${timestamp}_${random-num}_${rawBody}`), using the
// app-id/timestamp/random-num it sends in the request headers. Verify over the
// RAW received body (never re-serialize). ⚠️ The exact inbound input set is not
// fully spelled out in the docs — confirm against a real sandbox webhook; until
// then we also accept a body-only signature as a fallback. Timing-safe compare.
/** Which HMAC construction matched an inbound webhook signature. */
export type SpxSigMethod = "app_id_ts_rnd_body" | "body_only";
 
/**
 * Verify the inbound webhook signature AND report which construction matched.
 * Same logic as verifyWebhookSignature but returns the method so the webhook
 * route can record it in spx_webhook_captures — the first time a real webhook
 * verifies, `method` tells us SPX's actual signing format for good.
 */
export function verifyWebhookSignatureDetailed(
  rawBody: string,
  headers: { checkSign?: string | null; appId?: string | null; timestamp?: string | null; randomNum?: string | null },
): { ok: boolean; method: SpxSigMethod | null } {
  const provided = headers.checkSign?.trim();
  if (!provided) return { ok: false, method: null };
  let appSecret: string;
  try {
    appSecret = getCreds().appSecret;
  } catch {
    return { ok: false, method: null };
  }
  const candidates: { method: SpxSigMethod; message: string }[] = [];
  // Primary: same construction as outbound, using the headers SPX sent.
  if (headers.appId && headers.timestamp && headers.randomNum) {
    candidates.push({
      method:  "app_id_ts_rnd_body",
      message: `${headers.appId.trim()}_${headers.timestamp.trim()}_${headers.randomNum.trim()}_${rawBody}`,
    });
  }
  // Fallback: body-only (in case SPX signs just the payload for callbacks).
  candidates.push({ method: "body_only", message: rawBody });
 
  const providedBuf = Buffer.from(provided, "utf8");
  for (const { method, message } of candidates) {
    const expected = createHmac("sha256", appSecret).update(message, "utf8").digest("hex");
    const expectedBuf = Buffer.from(expected, "utf8");
    if (expectedBuf.length === providedBuf.length && timingSafeEqual(expectedBuf, providedBuf)) {
      return { ok: true, method };
    }
  }
  return { ok: false, method: null };
}
 
/** Boolean convenience wrapper — the trust gate only needs pass/fail. */
export function verifyWebhookSignature(
  rawBody: string,
  headers: { checkSign?: string | null; appId?: string | null; timestamp?: string | null; randomNum?: string | null },
): boolean {
  return verifyWebhookSignatureDetailed(rawBody, headers).ok;
}
