/** One facts block per distinct report (1Y and All often coincide). */
export function factsByPeriod(data: RightTimePeriods): { key: string; periods: RtPeriod[]; facts: Record<InsightSection, string | null> }[] {
  const groups = new Map<string, { key: string; periods: RtPeriod[]; facts: Record<InsightSection, string | null> }>();
  for (const p of RT_PERIODS) {
    const facts = factsFor(data.periods[p]);
    const key = createHash("sha256").update(JSON.stringify(facts)).digest("hex").slice(0, 16);
    const g = groups.get(key) ?? { key, periods: [], facts };
    g.periods.push(p);
    groups.set(key, g);
  }
  return [...groups.values()];
}
 
/** Hash of all facts — the insight cache key. */
export function insightsKey(data: RightTimePeriods): string {
  return createHash("sha256").update(JSON.stringify(factsByPeriod(data).map((g) => g.key))).digest("hex").slice(0, 16);
}
 
export function fallbackAll(data: RightTimePeriods): RightTimeInsights {
