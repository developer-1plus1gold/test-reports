/**
 * Shared shop-catalog fetch — the SINGLE source for the shop-visible product
 * SELECT + media-ordering + empty-guard + L1 cache. Consumed by:
 *   - GET /api/products            (the storefront client SPA)
 *   - lib/structured-data.ts       (server-rendered JSON-LD / product pages)
 *   - app/sitemap.ts               (product-page URLs)
 *
 * Extracted 2026-06-15 so the product SELECT lives in exactly one place — the
 * route used to inline it, and the new SEO surfaces would otherwise have
 * duplicated (and inevitably drifted from) it.
 */
 
import { createClient } from "@supabase/supabase-js";
import { getProductsCache, setProductsCache } from "@/lib/products-cache";
 
const supabaseAdmin = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_KEY!,
);
 
/** Raw shop product as returned by the catalog SELECT. Satisfies `ShopPriceable`
 *  (lib/shop-pricing) so it can be passed straight to `priceShopCatalog`. */
export interface RawShopProduct {
  id: string;
  grams: number;
  label: string;
  is_assay: boolean;
  dimensions: string | null;
  product_type: "curve" | "standard" | "accessory";
  price_sgd: number | null;
  discount_rules: unknown;
  status: "active" | "presale" | "sold_out";
  available_date: string | null;
  stock_qty: number | null;
  low_stock_threshold: number | null;
  sort_order: number | null;
  shop_visible: boolean;
  updated_at: string;
  fineness: number | null;
  refinery: string | null;
  is_lbma: boolean;
  gst_exempt: boolean;
  weight_display: string | null;
  // Eligibility flags consumed by the cart kernel (deriveCart → isGyEligibleLine
  // / isVaultEligibleLine). The iOS on-device engine builds its catalog from this
  // endpoint, so omitting these made vaultEligible/gyEligible always false on iOS
  // → the "Store in Vault" + GY checkout cards never showed (web uses the
  // /api/checkout/context catalog, which has them). 2026-06-30.
  gy_eligible: boolean;
  vault_eligible: boolean;
  standard_premium_tiers: { min_qty: number; max_qty: number | null; premium_pct: number }[] | null;
  product_media: { url: string; kind?: string; alt_text?: string | null; sort_order?: number | null }[];
}
 
const CATALOG_SELECT =
  "id, grams, label, is_assay, dimensions, product_type, price_sgd, discount_rules, status, available_date, stock_qty, low_stock_threshold, sort_order, shop_visible, updated_at, fineness, refinery, is_lbma, gst_exempt, weight_display, gy_eligible, vault_eligible, standard_premium_tiers(min_qty, max_qty, premium_pct), product_media(url, kind, alt_text, sort_order)";
 
/**
 * Fetch the shop-visible catalog (sorted), with each product's media ordered by
 * sort_order. Uses the L1 module cache (5-min TTL, invalidated on admin product
 * mutations). NEVER caches an empty result — a transient zero-row read would
 * otherwise pin an empty storefront for the whole TTL (#13, 2026-06-12).
 *
 * @throws on a real DB error (caller decides how to degrade).
 */
export async function fetchShopCatalogRaw(): Promise<{ products: RawShopProduct[]; ts: number; cached: boolean }> {
  const cached = getProductsCache();
  if (cached) return { products: cached.data as RawShopProduct[], ts: cached.ts, cached: true };
 
  const { data, error } = await supabaseAdmin
    .from("products")
    .select(CATALOG_SELECT)
    .eq("shop_visible", true)
    .order("sort_order", { ascending: true });
 
  if (error) {
    console.error("[shop-catalog] DB error:", error);
    throw new Error("catalog_fetch_failed");
  }
 
  const enriched = ((data ?? []) as unknown as RawShopProduct[]).map((p) => ({
    ...p,
    // grams is numeric(10,4) in DB → PostgREST returns it as a string; coerce so
    // all downstream pricing math sees a real number (oz products are fractional).
    grams: Number(p.grams),
    product_media: Array.isArray(p.product_media)
      ? [...p.product_media].sort((a, b) => Number(a.sort_order ?? 0) - Number(b.sort_order ?? 0))
      : [],
  }));
 
  if (enriched.length === 0) return { products: [], ts: Date.now(), cached: false };
 
  setProductsCache(enriched);
  return { products: enriched, ts: Date.now(), cached: false };
}
