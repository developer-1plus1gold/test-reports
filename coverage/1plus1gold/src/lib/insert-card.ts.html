/**
 * The parcel insert card — the printed A6 card that goes in every box.
 *
 * ▶ WHAT IT IS AND WHY IT IS A CONSTANT. Four artefacts already travel with a
 * parcel: the CoA (per serial), the receipt (per order), the packing slip and
 * the shipping label (both per order). This is the fifth and the only one that
 * is the SAME for every customer — it carries no order number, no serial, no
 * name, no amount. That is deliberate, and it is the single most useful fact
 * about this file: with no dynamic input there is nothing to escape, nothing to
 * leak and nothing to get wrong per order. ▶ If anyone ever adds an order
 * detail here, that stops being true and every interpolation needs escaping
 * (see `escapeHtml` in lib/packing-slip.ts) — the card is rendered into an
 * `iframe srcdoc`, which is SAME-ORIGIN and therefore a real sink.
 *
 * ▶ WHAT IT SAYS, and the operator decisions behind it (Mukesh, 2026-09-16/17):
 *   - Thank you for the purchase.
 *   - How to check the gold is real — points at the CoA's QR rather than
 *     carrying a second one, so there is one verification story, not two.
 *   - Sell-back, stated as our philosophy: accumulate, and sell when you need
 *     to, not to trade. ▶ The wording was LIGHTENED on purpose — an earlier
 *     draft leaned on "we hope that day never comes" twice, which made a
 *     service we genuinely offer read as something to feel bad about. The
 *     hinge is "Whenever that day comes", not "if".
 *   - The sell-back caveat points at /contact, NOT a WhatsApp number: a number
 *     printed on paper is permanent and unfixable, and the contact page can
 *     route wherever we want it to later.
 *   - A review ask, with the QR — and it DISCLOSES that the link leaves our
 *     site. Trustpilot is independent; pretending otherwise would undercut the
 *     one thing an independent review is worth.
 *   - The regulatory line reads the LICENCE FROM BUSINESS_CONFIG rather than
 *     typing it. A stale licence number on a printed artefact is exactly the
 *     kind of wrong nobody notices, because it still looks like a licence.
 *   - "Singapore", never "Republic of Singapore" (operator's call; the long
 *     form survives only in BUSINESS_CONFIG.jurisdiction, where an accounting
 *     convention wants it).
 *   - The SME 500 mark as the SEAL, not a sentence about the award.
 *   - NO vault mention: the vault is family-and-friends today, not a public
 *     offer, so the card must not advertise it.
 *   - SINGLE SIDED. A second side is a second thing to get wrong at the printer
 *     for no gain.
 *
 * ▶ THE SEAL SITS ON A DARK FOOT BAND, and that is a constraint rather than a
 * flourish. `public/images/seals/sme500.webp` is the REVERSED lockup — white
 * type with the red lion, 74.2% white / 25.8% red / 0.0% black by pixel. On the
 * ivory stock it is very nearly invisible. The issuer ships a full-colour
 * variant for light grounds, but the two are separate artworks and must never
 * be simulated from each other (a CSS `invert()` turns the lion cyan). The
 * dark band is the ground this file has, used as the issuer intended.
 *
 * ▶ THE LOGO IS THE PRE-RENDERED PNG, and this is NON-NEGOTIABLE per
 * [[feedback-logo-use-receipt-pdf-spec]]: `public/email-logo.png`, the same
 * asset the HTML emails use, for the same reason — this is the one surface
 * where we never let typography fail. It is also the correct artwork, verified
 * against the official vector rather than assumed: the vector is live TYPE (no
 * paths at all), "1" and "1" at 16.5pt in #C9A84C with "+" at 12.75pt and
 * "G O L D" at 5.25pt in #A07830 — so a plus:numeral FONT-SIZE ratio of 0.773
 * and a word ratio of 0.318. The PNG reproduces both (its + is 0.366 of the
 * numeral's INK height, which is what a 0.77 font-size ratio draws) and carries
 * exactly those two colours. Do not rebuild it from type here.
 *
 * ▶ FONTS ARE SELF-HOSTED, unlike the packing slip, which still `<link>`s
 * Google Fonts. A print artefact should not depend on a third-party CDN being
 * up at the moment the operator presses Print — and we already self-host these
 * three families (2026-08-12, after gstatic flaked two prod builds in a day).
 * Same-origin woff2, no network beyond our own.
 *
 * ▶ THE FOOT BAND IS 78px, NOT THE 90px THE DESIGN WAS APPROVED AT, and the
 * 12px is the fix for a defect rather than a restyle. Measured in Chromium at
 * true A6: with a 90px band the body's four blocks plus its own padding need
 * 361.2px against 353px of box, so the content eats the body's 8px bottom
 * padding and the regulatory line's descenders land 0.2px off the dark band —
 * touching it. Nothing clips and nothing compresses, which is exactly why it
 * survived approval: at preview scale a fifth of a pixel is invisible, and the
 * first thing to make it visible would be a printer's font substitution
 * pushing that line INTO the band. 78px restores real clearance and leaves the
 * seal at 56px with 11px above and below. ▶ The guard that catches this class
 * sums `height + marginTop + marginBottom`; an earlier version summed
 * `getBoundingClientRect().height` alone, reported "fits, 4px slack", and was
 * blind to margins while a line was cut 15px behind the band.
 *
 * ▶ WHY AN HTML STRING AND NOT A PAGE. The house pattern for parcel artefacts
 * is build-a-string then hand it to <DocumentPreviewModal> via the doc-preview
 * store — see lib/packing-slip.ts. That modal exists BECAUSE `window.open`
 * silently does nothing inside iOS WKWebView, which is where the operator's
 * admin PWA runs. A `?print` page (the CoA's pattern) is right for something a
 * CUSTOMER can also visit; nobody visits this.
 */
import { BUSINESS_CONFIG } from "@/lib/business-config";
import { qrGeometry, assertPathSafe } from "@/lib/qr";
import { useDocPreviewStore } from "@/store/docPreviewStore";
 
/**
 * Pinned to production for the same reason the CoA's origin is: this is printed
 * onto paper and put in a customer's box, where it is permanent and unfixable.
 * A card printed from dev must never carry a preview host.
 */
const REVIEW_URL = "https://1plus1gold.com/review";
/** The human-readable fallback under the QR — scheme stripped, as on the CoA. */
const REVIEW_URL_DISPLAY = REVIEW_URL.replace(/^https:\/\//, "");
 
/** A6 at 96 dpi, which is exactly the canvas the design was approved on. */
const A6 = { w: 397, h: 559, mm: "105mm 148mm" } as const;
 
/**
 * Error correction Q (25%) — free here. The payload is short enough that Q and
 * M both land on version 3 / 29 modules, so we buy the paper-handling
 * robustness at no cost in density. At 80 px across 37 modules (29 + two
 * 4-module quiet zones) each module is ~2.2 px ≈ 0.57 mm printed, comfortably
 * above the 40 px floor measured for the denser CoA code.
 */
const QR_PX = 80;
 
/** The printable card as a standalone HTML document. Pure — no DOM, no I/O. */
export function buildInsertCardHtml(): string {
  const qr = qrGeometry(REVIEW_URL, "Q");
  const path = assertPathSafe(qr.path);
 
  return `<!doctype html>
<html lang="en">
<head>
<meta charset="utf-8"/>
<title>Parcel insert card</title>
<style>
  /* Self-hosted so pressing Print never waits on a third-party CDN. */
  @font-face { font-family: "Cinzel"; src: url("/fonts/Cinzel-Regular.woff2") format("woff2"); font-weight: 400; font-display: swap; }
  @font-face { font-family: "Cinzel"; src: url("/fonts/Cinzel-SemiBold.woff2") format("woff2"); font-weight: 600; font-display: swap; }
  @font-face { font-family: "Montserrat"; src: url("/fonts/Montserrat-Regular.woff2") format("woff2"); font-weight: 400; font-display: swap; }
  @font-face { font-family: "Montserrat"; src: url("/fonts/Montserrat-Medium.woff2") format("woff2"); font-weight: 500; font-display: swap; }
 
  /* A6 portrait, no printer margin — the card's own 10px sheet padding is the
     trim allowance, so the frame never sits on the paper's edge. */
  @page { size: ${A6.mm}; margin: 0; }
 
  html, body { margin: 0; padding: 0; background: #ffffff; }
  body { -webkit-print-color-adjust: exact; print-color-adjust: exact; }
 
  .sheet {

    background: #ffffff; padding: 10px;

    margin: 0 auto;
  }
  .frame {
    flex-grow: 1; min-height: 0; box-sizing: border-box;

    border-radius: 16px;
    display: flex; flex-direction: column; overflow: hidden;
  }
  .head { padding: 13px 26px 10px; text-align: center; border-bottom: 1px solid #e6ddc4; flex-shrink: 0; }
  .head img { display: block; height: 44px; width: auto; margin: 0 auto; }
  .thanks { margin: 10px 0 0; font-family: Cinzel, Georgia, serif; font-size: 10px; letter-spacing: 0.34em; text-transform: uppercase; color: #7a6a45; }

  .eyebrow { font-family: Cinzel, Georgia, serif; font-size: 10px; letter-spacing: 0.28em; text-transform: uppercase; color: #9a7a2e; margin: 0; }

  .fine    { font-size: 9px; line-height: 1.5; color: #7a6a45; margin: 5px 0 0; }

  .ink     { color: #2a2114; }
 
  .review  { display: flex; align-items: center; gap: 14px; }

  .plate svg { display: block; }
  .rv-note { margin: 5px 0 0; font-size: 9px; line-height: 1.45; color: #7a6a45; }

  .foot { flex-shrink: 0; height: 78px; background: #16110A; display: flex; align-items: center; justify-content: center; }
  .foot img { display: block; height: 56px; width: auto; }
</style>
</head>
<body>
  <div class="sheet">
    <div class="frame">
 
      <div class="head">
        <img src="/email-logo.png" alt="1+1 GOLD"/>
        <p class="thanks">Thank you for your purchase</p>

      </div>

/**
 * Show the card in <DocumentPreviewModal>. One paper variant only — the card
 * is A6 and nothing else, so no size toggle is offered (the modal renders the
 * 4x6/A4 switch only when `htmlVariants` is supplied).
 */
export function openInsertCard(): void {
  useDocPreviewStore.getState().show({
    title: "PARCEL INSERT CARD",
    html:  buildInsertCardHtml(),
  });
}
