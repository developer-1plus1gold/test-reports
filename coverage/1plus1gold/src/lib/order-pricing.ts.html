//
// order-pricing.ts — THE single server-side authoritative cart pricing.
//
// Every server money path (orders/create, payments/stripe/create-intent,
// hitpay-order, the Stripe webhook) prices the gold items through THIS — which
// runs the SAME `deriveCart` kernel the cart display + the iOS engine use. There
// is no other "price a cart" composition: the curve-only `computeOrderTotal`
// that each route used to call directly mispriced standard/wholesale lines and
// drifted from the cart. Single composition → drift is structurally impossible.
//
// Authoritative: callers pass DB-resolved tiers (never client-supplied). For a
// wholesale-enabled customer the better-of standard vs wholesale tiers wins at
// the line qty — mirroring the cart's "better premium wins".
//
// Gold items only (curve/standard). Accessories are priced separately by
// computeAccessoryTotalSgd / order-accessories (they always have).
//
 
import { deriveCart, pickBetterTiers, wholesaleTiersIfEligible, type CartLine, type CartContext } from "./cart-core";
import type { StandardPremiumTier } from "./gold";
import type { PremiumAnchor } from "./shop-pricing";
 
// ── Locked-total reconciliation ──────────────────────────────────────────────
//
// The amount CHARGED is always the server's own `serverTotalSgd` (priceOrder),
// never the client's `lockedSubtotalSgd` — see orders/create + hitpay/create,
// which persist `serverTotalSgd` as `total_sgd`. The client lock is only used to
// decide whether the customer's *view* is still current.
//
// The only direction that can harm the customer is the server total being
// HIGHER than what they locked/saw (we'd charge more than the cart showed). So
// the reconciliation is ONE-SIDED: reject (→ refresh) when the server total
// exceeds the locked total beyond a small rounding margin; proceed silently when
// the server total is equal-or-lower (we charge the lower number — the 10-minute
// lock honoured as a goodwill promise). This replaces the prior symmetric
// S$0.50 band that the security review flagged (finding TOTAL_TOLERANCE_SGD_0.50)
// as an over-wide two-sided tolerance. Spot itself is separately bounded
// (orders/create SPOT_STALENESS_PCT 1%; both routes reprice AT lockedSpotSgd,
// which the schema already constrains to `.positive()`).
//
// 0.05 absorbs multi-line cent-rounding only; legitimate price movement within
// the lock window comes from curve-accumulation-position drift, which can only
// raise the premium (→ correctly forces a refresh so the customer sees it).
export const TOTAL_OVERCHARGE_TOLERANCE_SGD = 0.05;
 
/**
 * Round an SGD money value to 2 decimal places. EVERY authoritative money value
 * a charge path produces (gold subtotal, discounts, grand total, charged total)
 * must pass through this before it is stored or forwarded to a payment gateway.
 *
 * Why: `deriveCart`/`priceOrder` return full-precision floats. When a 100%-off
 * promo caps at the *unrounded* gold subtotal, `gold + ship − gold` leaves an
 * IEEE-754 residue (e.g. 14.419999999999987 instead of 14.42). HitPay then
 * floors that to 14.41 — the customer is charged a cent less than the button
 * showed. The client already rounds its locked subtotal to 2dp (checkout
 * page.tsx, "so the value passed to the payment API is already clean"); the
 * server must mirror that discipline so the forwarded amount equals the display.
 */
export function roundSgd(n: number): number {
  return Math.round(n * 100) / 100;
}
 
/**
 * True when the server-derived subtotal is materially HIGHER than the client's
 * locked subtotal — i.e. proceeding would charge more than the customer saw, so
 * the caller must 422/refresh. Equal-or-lower server totals return false
 * (proceed; the server charges its own lower number).
 */
export function exceedsOverchargeTolerance(serverTotalSgd: number, lockedSubtotalSgd: number): boolean {
  const serverRounded = Math.round(serverTotalSgd * 100) / 100;
  const lockedRounded = Math.round(lockedSubtotalSgd * 100) / 100;
  return serverRounded - lockedRounded > TOTAL_OVERCHARGE_TOLERANCE_SGD;
}
 
/**
 * TWO-SIDED tolerance gate for charging the customer the exact total shown on the
 * "Proceed to Payment" button. We forward the client's DISPLAYED total to the
 * gateway — but only after the server's own authoritative recompute agrees within
 * the cent-rounding margin. Both directions are guarded:
 *   - client total materially BELOW server's  → a tampered/stale low total would
 *     underpay us → reject (refresh).
 *   - client total materially ABOVE server's  → the customer would pay more than
 *     the current fair price (stale lock) → reject (refresh).
 * Within the margin → safe to charge the displayed total, so the charge equals
 * the button exactly. The gold/promo/referral/fee math stays server-authoritative
 * (it's what bounds the displayed total here); we only PIN the final charge to
 * what the customer saw.
 */
export function displayedTotalWithinTolerance(displayedTotalSgd: number, serverChargedSgd: number): boolean {
  // Compare in integer cents — `Math.abs(...) <= 0.05` fails at the exact boundary
  // (0.05000000000000071 > 0.05) on IEEE-754. 5 cents = TOTAL_OVERCHARGE_TOLERANCE_SGD.
  const diffCents = Math.round(Math.abs(roundSgd(displayedTotalSgd) - roundSgd(serverChargedSgd)) * 100);
  return diffCents <= Math.round(TOTAL_OVERCHARGE_TOLERANCE_SGD * 100);
}
 
/**
 * Decide what to charge, given the client's DISPLAYED ("Proceed to Payment") total
 * and the server's own authoritative recompute. Money-safe and ASYMMETRIC — this is
 * the single decision point all charge paths use:
 *
 *   • displayed == null                      → charge the server amount.
 *   • within the cent margin                 → charge the DISPLAYED amount, so the
 *                                              charge equals the button exactly
 *                                              (fixes the float-residue cent gap).
 *   • displayed ABOVE server beyond margin    → the client priced HIGHER than our
 *                                              authoritative recompute — typically a
 *                                              stale client snapshot (e.g. the
 *                                              checkout cached accumulated-curve-grams
 *                                              at mount; the server reads them fresh,
 *                                              sees a lower premium, computes a lower
 *                                              gold). Charging OUR lower number can
 *                                              never underpay us and never
 *                                              surprise-overcharges the customer →
 *                                              charge the server amount, do NOT refresh.
 *   • displayed BELOW server beyond margin    → charging the displayed total would
 *                                              underpay us (tamper / stale-low) →
 *                                              refresh so the client re-syncs the lock.
 *
 * The earlier two-sided reject-on-mismatch (refresh in BOTH directions) made every
 * legitimately-stale-high order 422 — a customer mid-session whose accumulation grew
 * (a prior order completed) could never check out. The displayed-total pin is only a
 * cent-fix; for any larger divergence we fall back to the server-authoritative price,
 * never blocking a legitimate order that pays us at least our fair amount.
 */
export function resolveChargedTotalSgd(
  displayedTotalSgd: number | null | undefined,
  serverChargedSgd: number,
): { chargeSgd: number; refresh: boolean } {
  const server = roundSgd(serverChargedSgd);
  if (displayedTotalSgd == null) return { chargeSgd: server, refresh: false };
  const displayed = roundSgd(displayedTotalSgd);
  if (displayedTotalWithinTolerance(displayed, server)) return { chargeSgd: displayed, refresh: false };
  if (displayed > server) return { chargeSgd: server, refresh: false };
  return { chargeSgd: server, refresh: true };
}
 
export interface OrderPricingItem {
  productId:       string | null;
  productType:     "curve" | "standard" | "accessory";
  grams:           number;
  quantity:        number;
  label?:          string;
  isAssay?:        boolean;
  /** DB standard_premium_tiers (authoritative; null/empty for curve). */
  standardTiers?:  StandardPremiumTier[] | null;
  /** DB wholesale_premium_tiers, if the product has them. */
  wholesaleTiers?: StandardPremiumTier[] | null;
}
 
export interface OrderPricingInput {
  items:                 OrderPricingItem[];
  spotSgd:               number;
  premiums:              PremiumAnchor[];
  accumulatedCurveGrams: number;
  /** Customer eligibility — gates whether wholesale tiers may apply. */
  wholesaleEnabled:      boolean;
}
 
export interface PricedOrderLine {
  productId:   string | null;
  productType: "curve" | "standard" | "accessory";
  grams:       number;
  quantity:    number;
  priceEach:   number;   // SGD/unit — tier-correct (standard) / position-correct (curve)
  lineSgd:     number;
  premiumPct:  number;   // applied premium %, for the order_items snapshot column
}
 
export interface OrderPricingResult {
  goldSubtotalSgd: number;   // curve + standard (matches the cart's goldSubtotalSgd)
  curveRate:       number;   // blended curve rate (for accumulation/GY snapshots)
  lines:           PricedOrderLine[];   // gold lines, in input order
}
 
export function priceOrder(input: OrderPricingInput): OrderPricingResult {
  const { items, spotSgd, premiums, accumulatedCurveGrams, wholesaleEnabled } = input;
 
  const goldItems = items.filter((i) => i.productType === "curve" || i.productType === "standard");
 
  const lines: CartLine[] = goldItems.map((i) => {
    let tiers: StandardPremiumTier[] | undefined;
    if (i.productType === "standard") {
      const std = i.standardTiers && i.standardTiers.length > 0 ? i.standardTiers : undefined;
      // Wholesale only when enabled AND the qty meets the wholesale minimum — below
      // it the customer gets standard (Mukesh 2026-06-07). SAME gate the cart uses.
      const whl = wholesaleEnabled ? wholesaleTiersIfEligible(i.wholesaleTiers, i.quantity) : undefined;
      tiers = whl ? pickBetterTiers(std, whl, i.quantity) : std;
    }
    return {
      productType: i.productType,
      productId:   i.productId,
      label:       i.label ?? "",
      g:           i.grams,
      isAssay:     i.isAssay ?? false,
      count:       i.quantity,
      priceEach:   0,        // recomputed by deriveCart (curve from spot, standard from tiers)
      imageUrl:    null,
      tiers,
    };
  });
 
  const ctx: CartContext = {
    spotSgd,
    premiums,
    accumulatedCurveGrams,
    accumulatedCurveGramsLoaded: true,
    ownedGrams: 0,
    catalog: [],   // lines carry their own tiers → no hydrate/catalog lookup needed
  };
 
  const view = deriveCart(lines, ctx);
 
  const plines: PricedOrderLine[] = view.lines.map((pl, idx) => {
    const src       = goldItems[idx];
    const priceEach = pl.priceEach ?? 0;
    const premiumPct = spotSgd > 0 && src.grams > 0 ? (priceEach / (spotSgd * src.grams) - 1) * 100 : 0;
    return {
      productId:   src.productId,
      productType: src.productType,
      grams:       src.grams,
      quantity:    src.quantity,
      priceEach,
      lineSgd:     pl.lineSgd ?? priceEach * src.quantity,
      premiumPct,
    };
  });
 
  return {
    goldSubtotalSgd: view.summary.goldSubtotalSgd,
    curveRate:       view.summary.blendedPremiumRate,
    lines:           plines,
  };
}
