/**
 * Cancellation teardown helpers.
 *
 * When an order is cancelled/refunded the ad-hoc admin status route already
 * releases reserved inventory, voids the courier booking, and tears down vault
 * holdings. But it did NOT reverse the customer-side ledger — so a cancelled
 * order's grams stayed in `accumulation` and inflated `customer_gold_holdings`
 * (the view behind the family tree, dashboard, owned-grams store, curve pricing,
 * share page and newsletter). This reverses it, ledger-style: a compensating
 * negative-grams row (entry_kind='cancelled') per order line, nets to zero.
 * Idempotent — never double-reverses a line already reversed.
 */
 
import type { SupabaseClient } from "@supabase/supabase-js";
 
/**
 * Reverse a cancelled/refunded order's accumulation via compensating negative
 * 'cancelled' rows. Returns the number of reversal rows written.
 */
export async function reverseAccumulationForOrder(
  client: SupabaseClient,
  orderId: string,
): Promise<number> {
  const { data: order } = await client
    .from("orders")
    .select("customer_id")
    .eq("id", orderId)
    .maybeSingle();
  if (!order?.customer_id) return 0;
 
  const { data: items } = await client
    .from("order_items")
    .select("id")
    .eq("order_id", orderId);
  const itemIds = (items ?? []).map((i) => i.id as string);
  if (itemIds.length === 0) return 0;
 
  const { data: accRows } = await client
    .from("accumulation")
    .select("grams, order_item_id, entry_kind")
    .in("order_item_id", itemIds);
 
  // A line already carrying a 'cancelled' reversal is left alone (idempotent).
  const alreadyReversed = new Set(
    (accRows ?? []).filter((r) => r.entry_kind === "cancelled").map((r) => r.order_item_id as string),
  );
 
  // Net the line's live grams (all non-'cancelled' entry kinds) and reverse it.
  const netByItem = new Map<string, number>();
  for (const r of accRows ?? []) {
    if (r.entry_kind === "cancelled") continue;
    const k = r.order_item_id as string;
    netByItem.set(k, (netByItem.get(k) ?? 0) + Number(r.grams ?? 0));
  }
 
  const inserts: Array<{ customer_id: string; grams: number; order_item_id: string; entry_kind: string }> = [];
  for (const [itemId, grams] of netByItem) {
    if (alreadyReversed.has(itemId) || grams === 0) continue;
    inserts.push({ customer_id: order.customer_id as string, grams: -grams, order_item_id: itemId, entry_kind: "cancelled" });
  }
  if (inserts.length === 0) return 0;
 
  const { error } = await client.from("accumulation").insert(inserts);
  if (error) throw new Error(`reverseAccumulationForOrder: ${error.message}`);
  return inserts.length;
}
 
/**
 * Release the promo redemption a cancelled order consumed: drop the redemption
 * row, re-derive uses_count from the surviving redemptions (self-healing), and
 * clear the legacy single-use slots only if THIS order set them. Idempotent —
 * once the redemption row is gone it's a no-op.
 */
export async function reversePromoForOrder(client: SupabaseClient, orderId: string): Promise<boolean> {
  const { data: order } = await client.from("orders").select("promo_code").eq("id", orderId).maybeSingle();
  const code = order?.promo_code as string | null | undefined;
  if (!code) return false;
 
  const { data: pc } = await client
    .from("promo_codes")
    .select("id, max_uses, used_order_id")
    .eq("code", code)
    .maybeSingle();
  if (!pc?.id) return false;
 
  const { data: deleted } = await client
    .from("promo_redemptions")
    .delete()
    .eq("promo_code_id", pc.id)
    .eq("order_id", orderId)
    .select("id");
  if (!deleted || deleted.length === 0) return false; // already reversed / never redeemed here
 
  // Re-derive the counter from surviving redemptions (idempotent, self-healing).
  const { count } = await client
    .from("promo_redemptions")
    .select("id", { count: "exact", head: true })
    .eq("promo_code_id", pc.id);
  const uses = count ?? 0;
  await client.from("promo_codes").update({
    uses_count: uses,
    used: pc.max_uses != null && uses >= (pc.max_uses as number),
  }).eq("id", pc.id);
 
  // Clear the single-use slots only when this order owned them.
  if (pc.used_order_id === orderId) {
    await client.from("promo_codes")
      .update({ used_by: null, used_at: null, used_order_id: null })
      .eq("id", pc.id).eq("used_order_id", orderId);
  }
  return true;
}
 
/**
 * Reverse a referral tied to a cancelled order. Returns what happened so the
 * caller can alert an admin for the cases we must NOT auto-reverse (already-paid
 * influencer cash, or an already-redeemed reward code).
 */
export async function reverseReferralForOrder(
  client: SupabaseClient,
  orderId: string,
): Promise<{ reversed: boolean; needsManual: boolean; reason?: string }> {
  const { data: ref } = await client
    .from("referrals")
    .select("id, status, reward_promo_code")
    .eq("first_order_id", orderId)
    .maybeSingle();
  if (!ref?.id || ref.status === "reversed") return { reversed: false, needsManual: false };
 
  let needsManual = false;
  let reason: string | undefined;
 
  if (ref.reward_promo_code) {
    // Standard tier: reward is a promo code in the referrer's wallet.
    const { data: rc } = await client
      .from("promo_codes")
      .select("id, used, uses_count, voided_at")
      .eq("code", ref.reward_promo_code)
      .maybeSingle();
    const redeemed = !!rc && (rc.used === true || Number(rc.uses_count ?? 0) > 0);
    if (rc?.id && !redeemed && !rc.voided_at) {
      await client.from("promo_codes")
        .update({ voided_at: new Date().toISOString(), voided_by: "cancel-teardown", void_reason: "referring order cancelled" })
        .eq("id", rc.id);
    } else if (redeemed) {
      needsManual = true;
      reason = `referrer reward ${ref.reward_promo_code} already redeemed`;
    }
  } else {
    // Influencer tier: cash via month-end PayNow — no per-referral paid flag.
    needsManual = true;
    reason = "influencer referral reward (PayNow) — verify payout not yet made";
  }
 
  if (needsManual) {
    // Keep the row (status='reversed') so the admin can see + act; UNIQUE(referred_id) stays.
    await client.from("referrals").update({ status: "reversed" }).eq("id", ref.id);
  } else {
    // Fully reversible → delete, freeing the referred customer's one-referral slot.
    await client.from("referrals").delete().eq("id", ref.id);
  }
  return { reversed: true, needsManual, reason };
}
 
/**
 * Void Gold-Yield enrolment for a cancelled order — but only bars that are still
 * un-received/un-sold. Reverts the GY-consumed inventory (sold→available + drops
 * its order_fulfilments link) and deletes the bar. Live bars (active/sold/exited)
 * are left for manual admin handling.
 */
export async function voidGoldYieldForOrder(
  client: SupabaseClient,
  orderId: string,
): Promise<{ voided: number; needsManual: number }> {
  const { data: bars } = await client
    .from("gold_yield_bars")
    .select("id, order_item_id, status, grams_sold")
    .eq("order_id", orderId);
  if (!bars || bars.length === 0) return { voided: 0, needsManual: 0 };
 
  let voided = 0, needsManual = 0;
  for (const bar of bars) {
    const safeStatus = bar.status === "pending_receipt" || bar.status === "received";
    const noSold = Number(bar.grams_sold ?? 0) === 0;
    let hasSales = false;
    if (safeStatus && noSold) {
      const { data: sales } = await client.from("gold_yield_sales").select("id").eq("bar_id", bar.id).limit(1);
      hasSales = !!sales && sales.length > 0;
    }
    if (!safeStatus || !noSold || hasSales) { needsManual++; continue; }
 
    // Revert the GY-consumed inventory (it was set to 'sold', not 'reserved', so
    // releaseInventoryForOrder does NOT cover it). Find the fulfilment link(s).
    if (bar.order_item_id) {
      const { data: fulfils } = await client
        .from("order_fulfilments")
        .select("id, inventory_id")
        .eq("order_item_id", bar.order_item_id);
      for (const f of fulfils ?? []) {
        if (f.inventory_id) await client.from("inventory").update({ status: "available" }).eq("id", f.inventory_id).eq("status", "sold");
        await client.from("order_fulfilments").delete().eq("id", f.id);
      }
    }
    await client.from("gold_yield_bars").delete().eq("id", bar.id);
    voided++;
  }
  return { voided, needsManual };
}
 
/**
 * Free NFC serials assigned to a cancelled order back to 'unassigned' so the
 * physical card can be re-linked to a future order (the card was never shipped).
 * Idempotent — only touches still-assigned serials for this order.
 */
export async function freeSerialsForOrder(client: SupabaseClient, orderId: string): Promise<number> {
  const { data } = await client
    .from("nfc_serials")
    .update({
      status: "unassigned",
      order_id: null,
      order_item_id: null,
      assigned_at: null,
      assigned_by: null,
      gift_message: null,
      current_owner_customer_id: null,
    })
    .eq("order_id", orderId)
    .eq("status", "assigned")
    .select("serial");
  return data?.length ?? 0;
}
