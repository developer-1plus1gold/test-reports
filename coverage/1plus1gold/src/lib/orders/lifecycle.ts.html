/**
 * Order lifecycle — the canonical "where is this order" state (order-status
 * rewrite, 2026-07-03; see memory project-order-status-rewrite-2026-07-03).
 *
 * ONE source of truth for order progress. The other two dimensions stay
 * separate and orthogonal: `payment_status` = money, `shipper_status` = courier
 * tracking detail (for the delivery tracker only).
 *
 *   placed → preparing → fulfilled → completed   (+ cancelled / refunded / failed)
 *
 * Recognition (revenue + COGS) happens at 'completed' = DELIVERY (customer
 * received: courier delivered / self-pickup collected / vault stored).
 *
 * Step 1 (migration 20260703_order_lifecycle_step1) added orders.lifecycle and
 * backfilled it from the legacy (status, shipper_status). `deriveLifecycle`
 * below is the CODE MIRROR of that SQL CASE — they MUST stay byte-identical so
 * the Step 2 write-both agrees with the backfill. Nothing reads lifecycle for
 * behaviour yet (reads flip in Step 4).
 */
 
export const ORDER_LIFECYCLES = [
  "placed", "preparing", "fulfilled", "completed", "cancelled", "refunded", "failed",
] as const;
export type OrderLifecycle = (typeof ORDER_LIFECYCLES)[number];
 
/** Customer has RECEIVED the goods — the recognition point. */
const RECEIVED_SHIPPER = new Set(["delivered", "stored"]);
/** In the hands of a courier, not yet received. */
const IN_TRANSIT_SHIPPER = new Set([
  "picked_up", "out_for_delivery", "in_transit", "on_going", "assigning_driver", "booked",
]);
 
/**
 * Derive the canonical lifecycle from the legacy (status, shipper_status).
 * MUST match migration 20260703's CASE arm-for-arm (exceptions → received →
 * in-transit → shipped → preparing → placed). Keep the two in lockstep.
 */
export function deriveLifecycle(
  status: string | null | undefined,
  shipperStatus: string | null | undefined,
): OrderLifecycle {
  if (status === "cancelled") return "cancelled";
  if (status === "refunded") return "refunded";
  if (status === "error") return "failed";
  if (status === "completed" || status === "delivered") return "completed";
  if (shipperStatus && RECEIVED_SHIPPER.has(shipperStatus)) return "completed";
  if (shipperStatus && IN_TRANSIT_SHIPPER.has(shipperStatus)) return "fulfilled";
  if (status === "shipped") return "fulfilled";
  if (status === "processing" || status === "ready") return "preparing";
  return "placed";
}
 
/** Human-readable label for each lifecycle state (admin + customer surfaces). */
export const LIFECYCLE_LABELS: Record<OrderLifecycle, string> = {
  placed:    "Placed",
  preparing: "Preparing",
  // 'fulfilled' spans the whole courier journey (booked → picked up → out for
  // delivery), so a specific "Out for Delivery" mislabels a just-booked order.
  // Keep this coarse; the precise courier stage is shown separately from
  // shipper_status (the delivery tracker / the order's "Status" row).
  fulfilled: "Shipping",
  completed: "Completed",
  cancelled: "Cancelled",
  refunded:  "Refunded",
  failed:    "Failed",
};
 
/** Coerce any (possibly null/legacy) lifecycle value to a known state, deriving
 *  from the legacy (status, shipper_status) when the column is absent. */
export function resolveLifecycle(
  lifecycle: string | null | undefined,
  status: string | null | undefined,
  shipperStatus: string | null | undefined,
): OrderLifecycle {
  if (lifecycle && (ORDER_LIFECYCLES as readonly string[]).includes(lifecycle)) {
    return lifecycle as OrderLifecycle;
  }
  return deriveLifecycle(status, shipperStatus);
}
 
/** Terminal states — no further transitions expected. */
export const TERMINAL_LIFECYCLES: ReadonlySet<OrderLifecycle> = new Set([
  "completed", "cancelled", "refunded", "failed",
]);
 
/** Whether revenue should be recognised for an order in this lifecycle
 *  (delivery = 'completed'). Accepts a plain string since the DB column is free
 *  text read as string. */
export function isRevenueRecognised(lifecycle: string | null | undefined): boolean {
  return lifecycle === "completed";
}
