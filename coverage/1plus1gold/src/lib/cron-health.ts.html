/**
 * Cron health — is this job actually alive? Pure, no I/O.
 *
 * ▶ WHY THIS EXISTS. `/admin/crons` is the screen we use to answer "is anything
 * silently dead?", and it could not answer it. Two separate failures:
 *
 *   1. Of 59 registered crons, 12 wrote a `cron_runs` heartbeat. The other 47
 *      rendered an empty Last-run cell forever, so a dead cron and a healthy one
 *      were indistinguishable — the standing question aimed at the dashboard we
 *      use to ask it.
 *   2. Worse, the 12 that DID report were rendered by recency alone: a job on a
 *      five-minute schedule whose last heartbeat was three days old showed
 *      **"✓ ok · 3d ago" in green**. A monitor that reports success on a corpse
 *      is worse than no monitor, because it is trusted.
 *
 * (2) is what this module fixes. A heartbeat timestamp is not health — health is
 * that timestamp measured against the schedule the job claims to keep. So the
 * expected interval is derived FROM the cron expression rather than configured
 * per job: there is no second number to keep in step, and a schedule change
 * automatically moves its own staleness bar.
 */
 
import { nextRunIso } from "./crons-catalog";
 
/**
 * The true period of a cron expression at this point in time.
 *
 * Derived as the gap between the NEXT two fire times, not between now and the
 * next one — for a monthly job the distance to the next 1st says nothing about
 * its period, and would make a job that ran yesterday look 29 days overdue.
 *
 * The scan cap is deliberately generous (70 days): a monthly expression in a
 * 31-day month needs 44,640 minutes to find its second fire, which sits a
 * handful of minutes inside the default cap in `nextRunIso`. Relying on that
 * margin would make February fine and March silently return null.
 */
export function expectedIntervalMs(expr: string, fromMs: number): number | null {
  const CAP = 70 * 24 * 60; // minutes
  const first = nextRunIso(expr, fromMs, CAP);
  if (!first) return null;
  const firstMs = Date.parse(first);
  const second = nextRunIso(expr, firstMs + 60_000, CAP);
  Iif (!second) return null;
  const gap = Date.parse(second) - firstMs;
  return gap > 0 ? gap : null;
}
 
/**
 * How long silence is tolerated before a job is called late.
 *
 * One full missed run, plus half a period or 20 minutes — whichever is larger:
 *
 *   every 5 min  -> 25 min      hourly -> 90 min
 *   every 10 min -> 30 min      daily  -> 36 h
 *   every 30 min -> 50 min      weekly -> 10.5 days
 *
 * The two ends need different treatment and this is the one rule that gives it.
 * A flat multiple is wrong at both: doubling makes a daily job look healthy for
 * two days, while 1.5x flags a five-minute job after a single skipped tick. The
 * 20-minute floor absorbs ordinary scheduler drift (GitHub's own schedules
 * routinely lag ~15 min) without which every fast cron would cry wolf.
 */
export function staleAfterMs(intervalMs: number): number {
  return intervalMs + Math.max(intervalMs * 0.5, 20 * 60_000);
}
 
/**
 * A job that fires several times a day and has NEVER written a heartbeat is
 * dead, not young. Below this period, "no runs yet" is treated as late.
 *
 * Above it the opposite is true: a monthly job deployed last week has correctly
 * never run, and flagging that would teach the operator to ignore the colour —
 * which is how the screen got into this state.
 */
export const NEVER_RAN_IS_LATE_BELOW_MS = 6 * 60 * 60_000;
 
export type CronHealth =
  /** Heartbeat present, recent, and the run succeeded. */
  | "ok"
  /** The job ran and reported failure. */
  | "error"
  /** Silent for longer than its own schedule allows. THE signal this file adds. */
  | "late"
  /** Instrumented, but no heartbeat yet — benign on an infrequent job. */
  | "never"
  /** Schedule could not be parsed, so no staleness claim can be made. */
  | "unknown";
 
export interface CronHealthInput {
  schedule:      string;
  lastRunAt:     string | null;
  lastRunStatus: string | null;
  nowMs:         number;
}
 
/**
 * ▶ ORDER MATTERS, and `error` before `late` is the load-bearing choice: a job
 * that reported an error and then stopped running entirely is both, and the
 * error is the actionable fact — "late" would send the operator looking at the
 * scheduler instead of at the failure it already told them about.
 */
export function cronHealth(input: CronHealthInput): CronHealth {
  const { schedule, lastRunAt, lastRunStatus, nowMs } = input;
  const interval = expectedIntervalMs(schedule, nowMs);
 
  if (lastRunAt === null) {
    Iif (interval === null) return "unknown";
    return interval <= NEVER_RAN_IS_LATE_BELOW_MS ? "late" : "never";
  }
  if (lastRunStatus === "error") return "error";
 
  const ranMs = Date.parse(lastRunAt);
  if (Number.isNaN(ranMs)) return "unknown";
  if (interval === null) return "unknown";
 
  // A clock-skewed future timestamp is not evidence of health, but it is also
  // not staleness — don't manufacture a "late" from a negative age.
  const age = nowMs - ranMs;
  if (age < 0) return "ok";
 
  return age > staleAfterMs(interval) ? "late" : "ok";
}
 
/** Health states that mean somebody should look. */
export function isCronUnhealthy(h: CronHealth): boolean {
  return h === "late" || h === "error";
}
 
/**
 * Attach health to catalogue rows.
 *
 * Lives here rather than in `buildCronCatalog` because this module already
 * imports `nextRunIso` from the catalogue — computing health there instead would
 * make the two files import each other, and a cycle that happens to work under
 * ESM today is a trap for whoever touches it next. The dependency runs one way:
 * catalogue knows schedules, this knows what a schedule implies.
 */
export function withHealth<
  T extends { schedule: string; lastRunAt: string | null; lastRunStatus: string | null },
>(rows: T[], nowMs: number): Array<T & { health: CronHealth; intervalMs: number | null }> {
  return rows.map((r) => ({
    ...r,
    health: cronHealth({
      schedule:      r.schedule,
      lastRunAt:     r.lastRunAt,
      lastRunStatus: r.lastRunStatus,
      nowMs,
    }),
    intervalMs: expectedIntervalMs(r.schedule, nowMs),
  }));
}
 
/** Shape the morning digest needs to name an unhealthy job. */
export interface LateCron { name: string; health: CronHealth; lastRunAt: string | null; intervalMs: number | null; }
 
/** Max jobs named in one message — a mass outage must not send a wall of text. */
export const LATE_ALERT_MAX_NAMED = 8;
 
/**
 * One WhatsApp line for the crons that are not doing their job, or null when
 * there is nothing to say.
 *
 * ▶ ONE MESSAGE, NEVER ONE PER JOB. The per-row escalate path in
 * `automation-monitor` would have sent 24 separate WhatsApps the morning after
 * this shipped. An alert that arrives 24 times is an alert that gets muted, and
 * a muted alert is the failure this whole change exists to remove.
 */
export function formatLateCronAlert(rows: LateCron[]): string | null {
  const bad = rows.filter((r) => isCronUnhealthy(r.health));
  if (bad.length === 0) return null;
 
  const named = bad.slice(0, LATE_ALERT_MAX_NAMED).map((r) => {
    if (r.health === "error") return `${r.name} (last run failed)`;
    const every = r.intervalMs ? `${Math.round(r.intervalMs / 60_000)}m` : "schedule";
    return r.lastRunAt
      ? `${r.name} (silent ${Math.round((Date.now() - Date.parse(r.lastRunAt)) / 60_000)}m, runs every ${every})`
      : `${r.name} (never run, runs every ${every})`;
  });
  const rest = bad.length - named.length;
  return `${bad.length} scheduled job${bad.length === 1 ? " is" : "s are"} not reporting on schedule: `
    + named.join("; ")
    + (rest > 0 ? `; +${rest} more` : "")
    + ". Check Control Tower → Scheduled Jobs.";
}
