/**
 * MAG pool offers (owner 2026-11-14) — the product × duration → premium table a
 * customer chooses from when asking for their own bar or a group pool, and the
 * quote that request carries. Pure: shared by the admin + public routes, the web
 * form and the tests. Migration 20261114c_mag_pool_offers.
 *
 * Decisions (Mukesh): any catalog gold bar may be offered; one flat % per product
 * + duration; the duration starts when the bar is SET UP (not on the request
 * date); the same % for every group member; an unfinished bar at its closing date
 * keeps today's behaviour. On approval the admin creates the bar's own MAG bar
 * type at the quoted % — checkout prices a MAG line from its bar type, so the
 * customer is charged exactly the quote with no change to the pricing kernel.
 */
 
export type DurationUnit = "weeks" | "months";
 
export interface PoolOffer {
  id: string;
  productId: string;
  durationValue: number;
  durationUnit: DurationUnit;
  /** Percent over spot (2.5 = 2.5%) — the unit of standard_premium_tiers. */
  premiumPct: number;
  active: boolean;
}
 
/** A product as the customer form shows it, with its active durations. */
export interface OfferProduct {
  productId: string;
  label: string;
  /** "100 g Valcambi" / "1 oz PAMP" — fits the one-line dropdown; the full
   *  catalog name is too long for a card a third of the page wide. */
  shortLabel: string;
  grams: number;
  /** The bar's first catalog photo, if any (owner 2026-11-14: show the bar). */
  imageUrl?: string | null;
  /** What this bar costs bought outright on the shop today, % over spot (owner
   *  2026-11-14). Null when the bar itself is not on the shop. */
  outrightPct?: number | null;
  durations: { offerId: string; durationValue: number; durationUnit: DurationUnit; premiumPct: number }[];
}
 
/**
 * Sizes a customer may ask us to quote, with any duration (owner 2026-11-14):
 * "we quote the premium". No outright comparison: no bar is chosen yet.
 */
export const QUOTE_SIZES: readonly { grams: number; label: string }[] = [
  { grams: 20, label: "20 g" },
  { grams: 31.1035, label: "1 oz" },
  { grams: 50, label: "50 g" },
  { grams: 100, label: "100 g" },
];
 
/** The durations offered on a quote request. */
export const QUOTE_DURATIONS: readonly { durationValue: number; durationUnit: DurationUnit }[] = [
  { durationValue: 2, durationUnit: "weeks" },
  { durationValue: 1, durationUnit: "months" },
  { durationValue: 2, durationUnit: "months" },
  { durationValue: 3, durationUnit: "months" },
  { durationValue: 6, durationUnit: "months" },
  { durationValue: 12, durationUnit: "months" },
];
 
/** The quote size for `grams` (± 0.01 g), or null when it is not one we quote. */
export function quoteSize(grams: unknown): { grams: number; label: string } | null {
  const g = Number(grams);
  if (!Number.isFinite(g)) return null;
  return QUOTE_SIZES.find(s => Math.abs(s.grams - g) < 0.01) ?? null;
}
 
/** True for a duration on the quote list. */
export function isQuoteDuration(value: unknown, unit: unknown): boolean {
  return QUOTE_DURATIONS.some(d => d.durationValue === Number(value) && d.durationUnit === unit);
}
 
/** A bar we sell outright today and its premium, % over spot. */
export interface OutrightRate { productId: string; grams: number; pct: number }
 
/**
 * The outright comparison for a bar (owner 2026-11-14): what this SAME bar
 * costs bought on the shop today. Null when the bar is not on the shop — no
 * stand-in from another bar of the same weight (Mukesh: "only if the same
 * product is visible on the shop").
 */
export function outrightPctFor(productId: string | null, rates: readonly OutrightRate[]): number | null {
  if (!productId) return null;
  return rates.find(r => r.productId === productId)?.pct ?? null;
}
 
export const MAX_DURATION_VALUE = 120;
export const MAX_PREMIUM_PCT = 50;
/** How long a quote holds for the admin to set the bar up. After this the
 *  customer is re-quoted from the current table. */
export const QUOTE_VALID_DAYS = 7;
 
export function durationLabel(value: number, unit: DurationUnit): string {
  const one = unit === "weeks" ? "week" : "month";
  return `${value} ${value === 1 ? one : `${one}s`}`;
}
 
/** "2.5%" — at most two decimals, no trailing zeros. */
export function premiumLabel(pct: number): string {
  return `${Number(pct.toFixed(2))}%`;
}
 
/**
 * The bar's closing date when it is set up `from` (decision 2a): N weeks, or N
 * calendar months (clamped to the month's last day, so 31 Jan + 1 month = 28/29
 * Feb, never 3 Mar).
 */
export function windowEndFrom(from: Date, value: number, unit: DurationUnit): Date {
  if (unit === "weeks") return new Date(from.getTime() + value * 7 * 24 * 60 * 60 * 1000);
  const d = new Date(from.getTime());
  const day = d.getUTCDate();
  d.setUTCDate(1);
  d.setUTCMonth(d.getUTCMonth() + value);
  const last = new Date(Date.UTC(d.getUTCFullYear(), d.getUTCMonth() + 1, 0)).getUTCDate();
  d.setUTCDate(Math.min(day, last));
  return d;
}
 
/** True while a quote made at `quotedAt` still holds. */
export function quoteIsValid(quotedAt: string | Date | null | undefined, now: Date = new Date()): boolean {
  if (!quotedAt) return false;
  const t = typeof quotedAt === "string" ? Date.parse(quotedAt) : quotedAt.getTime();
  if (!Number.isFinite(t)) return false;
  return now.getTime() - t <= QUOTE_VALID_DAYS * 24 * 60 * 60 * 1000;
}
 
export type OfferInput = { productId: string; durationValue: number; durationUnit: DurationUnit; premiumPct: number; active: boolean };
 
const UUID_RE = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;
 
/** Validate an admin create (all fields) or edit (`partial`: premium / active only). */
export function validateOfferInput(body: unknown, partial = false):
  { ok: true; value: Partial<OfferInput> } | { ok: false; error: string } {
  const b = (body && typeof body === "object" ? body : {}) as Record<string, unknown>;
  const out: Partial<OfferInput> = {};
  if (!partial) {
    if (typeof b.productId !== "string" || !UUID_RE.test(b.productId)) return { ok: false, error: "Pick a product" };
    out.productId = b.productId;
    const v = Number(b.durationValue);
    if (!Number.isInteger(v) || v < 1 || v > MAX_DURATION_VALUE) {
      return { ok: false, error: `Duration must be a whole number from 1 to ${MAX_DURATION_VALUE}` };
    }
    out.durationValue = v;
    if (b.durationUnit !== "weeks" && b.durationUnit !== "months") return { ok: false, error: "Duration unit must be weeks or months" };
    out.durationUnit = b.durationUnit;
  }
  if (!partial || b.premiumPct !== undefined) {
    const p = Number(b.premiumPct);
    if (b.premiumPct === null || b.premiumPct === "" || !Number.isFinite(p) || p < 0 || p > MAX_PREMIUM_PCT) {
      return { ok: false, error: `Premium must be a % from 0 to ${MAX_PREMIUM_PCT}` };
    }
    out.premiumPct = Math.round(p * 1000) / 1000;
  }
  if (b.active !== undefined) {
    if (typeof b.active !== "boolean") return { ok: false, error: "active must be true or false" };
    out.active = b.active;
  } else if (!partial) {
    out.active = true;
  }
  if (partial && Object.keys(out).length === 0) return { ok: false, error: "Nothing to change" };
  return { ok: true, value: out };
}
 
/** "1 oz" for a troy ounce (± 0.01 g), else "100 g" / "2.5 g". */
export function weightLabel(grams: number): string {
  if (Math.abs(grams - 31.1035) < 0.01) return "1 oz";
  return `${Number(grams.toFixed(2))} g`;
}
 
/** Group active offers by product for the customer form: products by weight,
 *  durations shortest first (weeks before months at equal length is irrelevant —
 *  compared in days). Inactive offers and products without a label are dropped. */
export function groupOffersForCustomer(
  offers: readonly PoolOffer[],
  products: ReadonlyMap<string, { label: string; grams: number; refinery?: string | null; imageUrl?: string | null; outrightPct?: number | null }>,
): OfferProduct[] {
  const days = (v: number, u: DurationUnit) => (u === "weeks" ? v * 7 : v * 30.44);
  const byProduct = new Map<string, OfferProduct>();
  for (const o of offers) {
    if (!o.active) continue;
    const p = products.get(o.productId);
    if (!p) continue;
    const shortLabel = p.refinery ? `${weightLabel(p.grams)} ${p.refinery}` : p.label;
    const entry = byProduct.get(o.productId) ?? {
      productId: o.productId, label: p.label, shortLabel, grams: p.grams,
      imageUrl: p.imageUrl ?? null, outrightPct: p.outrightPct ?? null, durations: [],
    };
    entry.durations.push({ offerId: o.id, durationValue: o.durationValue, durationUnit: o.durationUnit, premiumPct: o.premiumPct });
    byProduct.set(o.productId, entry);
  }
  const list = [...byProduct.values()];
  for (const p of list) p.durations.sort((a, b) => days(a.durationValue, a.durationUnit) - days(b.durationValue, b.durationUnit));
  return list.sort((a, b) => a.grams - b.grams || a.label.localeCompare(b.label));
}
 
/** The dedicated MAG bar type's name: "<product> — own bar for <name>" or
 *  "<product> — group <group name>". Admin-facing; capped at 120 characters. */
export function barTypeLabel(productLabel: string, requestType: string, customerName: string, groupName: string | null): string {
  const who = requestType === "group_pool" ? `group ${groupName ?? customerName}` : `own bar for ${customerName}`;
  return `${productLabel} — ${who}`.slice(0, 120);
}
 
/** Map a DB row to a PoolOffer. */
export function offerFromRow(r: {
  id: string; product_id: string; duration_value: number | string; duration_unit: string;
  premium_pct: number | string; active: boolean;
}): PoolOffer {
  return {
    id: r.id, productId: r.product_id, durationValue: Number(r.duration_value),
    durationUnit: r.duration_unit === "months" ? "months" : "weeks",
    premiumPct: Number(r.premium_pct), active: !!r.active,
  };
}
