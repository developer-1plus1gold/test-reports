/** A legacy bar has no stored schedule: the nearest quote duration to its opening → closing span. */
export function derivedSchedule(l: { grams_total: number; created_at: string; window_ends_at: string | null }): ReservedSchedule | null {
  if (!l.window_ends_at) return null;
  const spanDays = (Date.parse(l.window_ends_at) - Date.parse(l.created_at)) / 86400000;
  const days = (d: { durationValue: number; durationUnit: DurationUnit }) => (d.durationUnit === "weeks" ? 7 : 30.44) * d.durationValue;
  const best = [...QUOTE_DURATIONS].sort((a, b) => Math.abs(days(a) - spanDays) - Math.abs(days(b) - spanDays))[0];
  return buildSchedule({ grams: l.grams_total, durationValue: best.durationValue, durationUnit: best.durationUnit, openedAt: l.created_at });
}

/** The 2-week quote premium for the bar on the day (outright % + 2 weeks' cost of capital). */
async function twoWeekPremium(db: SupabaseClient, l: ListingRow): Promise<number> {
  const { data: inv } = l.inventory_id ? await db.from("inventory").select("product_id").eq("id", l.inventory_id).maybeSingle() : { data: null };
  const physical = (inv as { product_id?: string } | null)?.product_id;
  // The bar's own one-bar premium (lowest-quantity tier) — read directly, so a bar the shop
  // no longer shows can still be priced when it is released.
  const [tier, rate] = await Promise.all([
    physical
      ? db.from("standard_premium_tiers").select("premium_pct").eq("product_id", physical).order("min_qty", { ascending: true }).limit(1).maybeSingle()
      : Promise.resolve({ data: null }),
    readCostOfCapitalPct(db),
  ]);
  const raw = (tier.data as { premium_pct?: unknown } | null)?.premium_pct;
  const outright = raw == null ? null : Number(raw);
  if (outright == null || !Number.isFinite(outright)) throw new Error("No outright premium for the bar — cannot price the released bar; left reserved for Admin");
  return suggestedQuotePct(outright, rate, 2, "weeks");
}

  return rows.map((l) => {
    const schedule = parseSchedule(l.rhythm_schedule) ?? derivedSchedule(l);
    const enforced = enabled && !l.rhythm_legacy && consented.has(l.id);
    const state = parseState(enforced ? l.rhythm_state : l.rhythm_dry_state);
    const h = held.get(l.id)?.held ?? 0;
    const steps = schedule ? [schedule.commitment, ...schedule.checkpoints].map((s) => ({
      key: s.key, dueAt: s.dueAt, grams: s.grams,
      status: (state.missed.includes(s.key) ? "missed" : state.judged.includes(s.key) ? "met"
        : Date.parse(s.dueAt) <= now.getTime() ? "due" : "upcoming") as RhythmBarReport["steps"][number]["status"],
    })) : [];
    return {
      listingId: l.id, serial: l.serial, label: l.products?.label ?? "", accessType: l.access_type, legacy: l.rhythm_legacy,
      mode: l.rhythm_legacy ? "legacy" : enforced ? "enforced" : "dry_run",
      grams: l.grams_total, held: h, windowEndsAt: l.window_ends_at,
      steps, strikes: state.strikes, catchUp: state.catchUp, ended: state.ended, events: evBy.get(l.id) ?? [],
    };
  });
}
