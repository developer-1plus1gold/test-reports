/**
 * MAG reserved bars — the buying schedule and its rules (Mukesh 2026-11-27,
 * docs/mag-reserved-bar-rhythm-design.md §2–§4, Agreement §1B). Pure: the quote,
 * the account card, the admin view and the daily cron all read it, so a rule is
 * the same everywhere.
 *
 *   periods   2 weeks → 2 × 7 days · 1 month → 2 × 15 days · N months → N calendar months
 *   commit    10 % of the bar, rounded UP to whole grams, within 5 days of opening
 *   share     the rest ÷ periods, rounded UP to whole grams (MAG sells whole grams)
 *   check k   by opening + k periods, held ≥ min(commit + k × share, grams)  (k = 1 … periods−1)
 *   close     by the bar's CLOSING date (windowEndFrom), the whole bar
 *
 * Mukesh 2026-11-28 (DEV walkthrough), replacing the first rule:
 *  - the commitment is a modest 10 % starter on every duration (it was one full share:
 *    half the bar within 3 days on a 2-week bar), within 5 days on every duration;
 *  - each period's share is due by the end of that period and the last at closing
 *    (the first rule had the whole bar due one period before closing).
 * A missed CLOSING step releases the bar at once — no catch-up past closing — and
 * the release counts toward the 6-month pause (Mukesh 2026-11-28).
 *
 * Fixed when the bar opens and stored on the listing: a later rule change never
 * moves an existing customer's goalposts. "Held" = grams the customer (group: all
 * members) holds on the bar now; grams sold back no longer count.
 */

function periods(value: number, unit: DurationUnit): { count: number; at: (opened: Date, k: number) => Date } {
  if (unit === "weeks") return { count: value, at: (o, k) => new Date(o.getTime() + k * 7 * DAY) };
  if (value === 1) return { count: 2, at: (o, k) => new Date(o.getTime() + k * 15 * DAY) };
  return {
    count: value,
    at: (o, k) => {
      const d = new Date(o.getTime());
      const day = d.getUTCDate();
      d.setUTCDate(1);
      d.setUTCMonth(d.getUTCMonth() + k);
      // 31 Jan + 1 month → 28/29 Feb, never 3 Mar.
      const last = new Date(Date.UTC(d.getUTCFullYear(), d.getUTCMonth() + 1, 0)).getUTCDate();
      d.setUTCDate(Math.min(day, last));
      return d;
    },
  };
}

/**
 * What KEEPS the bar by `until`: the larger of `base` (the catch-up amount) and every
 * step not yet judged that falls due on or before `until` (Mukesh 2026-11-28, DEV
 * walkthrough: on a 2-week bar the catch-up deadline IS the closing date, so "catch up
 * to 18 g" kept nothing — the whole bar, 31 g, was due the same moment). The strike
 * notice, the catch-up reminder and the account card all quote this one number.
 */
export function keepTarget(schedule: ReservedSchedule, state: Pick<RhythmState, "judged">, until: string, base: number): number {
  const by = Date.parse(until);
  return [schedule.commitment, ...schedule.checkpoints]
    .filter((st) => !state.judged.includes(st.key) && Date.parse(st.dueAt) <= by)
    .reduce((m, st) => Math.max(m, st.grams), base);
}

/**

 * events (oldest step first) and the next state. Idempotent: running it twice
 * on the same day yields no new events the second time. Stops at the first

 * not struck while their payment settles.
