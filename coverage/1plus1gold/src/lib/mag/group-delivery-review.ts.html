/**
 * MAG group-bar delivery: the Compliance Officer's review before dispatch
 * (IPPC v1.6 §2.3 + §9.2, AML Technical Reference v1.5 §4.3; Mukesh 2026-11-12).
 *
 * Why. When a group bar is delivered, every member's grams pass to the recipient
 * in one step (mag_complete_delivery), but the delivery ORDER carries only the
 * unpaid remainder plus the courier fee. So order-level transaction monitoring
 * never sees the value that actually moves between people. A group bar of
 * 100 g is about S$17k passing to one customer, and that is the case the
 * group-transfer red flag exists for.
 *
 * What. On completion of a GROUP delivery we file one `escalate` decision in
 * Control Tower (and page the admin team) naming the recipient, every other
 * member, their grams and what they paid. Every step that would send the
 * parcel out (packing, booking any courier, hand-over, vault storage) goes
 * through `dispatchBlocker` (lib/shipping/dispatch-blocker.ts), which refuses
 * until that decision is approved.
 *
 * Fail closed, by construction. The gate does NOT look for "a decision that says
 * stop". It asks "is this a group delivery?" and then requires an APPROVED
 * decision. A decision that failed to insert, or a lookup error, therefore blocks
 * dispatch instead of letting it through. Reserved (one-person) bars and
 * ordinary orders are unaffected: no value moves between customers there.
 */

const round2 = (n: number) => Math.round(n * 100) / 100;
const round4 = (n: number) => Math.round(n * 10_000) / 10_000;
 
/**
 * The group listing delivered on this order, or `undefined` when the order is
 * not a group delivery. Throws on a lookup error so callers can fail closed.
 */
async function findGroupListing(db: SupabaseClient, orderId: string): Promise<ListingRow | undefined> {
  const { data, error } = await db
    .from("mag_listings")
    .select("id, serial, access_type, group_name, recipient_customer_id")
    .eq("delivered_order_id", orderId)
    .maybeSingle();
  if (error) throw new Error(`mag_listings lookup failed: ${error.message}`);
  const row = data as ListingRow | null;
  return row && row.access_type === "group" ? row : undefined;
}
 
/** Pure: fold the delivered allocations into per-member totals and the transfer. */
export function summariseGroupDelivery(
  listing: { id: string; serial: string; group_name: string | null; recipient_customer_id: string },
  allocations: { customer_id: string; grams: number | string; price_per_g_sgd: number | string | null }[],
  names: Map<string, string>,
): GroupDeliveryFacts {
  const byMember = new Map<string, { grams: number; paid: number }>();
  for (const a of allocations) {
    const g = Number(a.grams) || 0;
    const m = byMember.get(a.customer_id) ?? { grams: 0, paid: 0 };
    m.grams += g;
    m.paid += g * (Number(a.price_per_g_sgd) || 0);
    byMember.set(a.customer_id, m);
  }
  const recipientId = listing.recipient_customer_id;
  const members: GroupDeliveryMember[] = [...byMember.entries()]
    .map(([customerId, m]) => ({
      customerId,
      name: names.get(customerId) ?? "Unknown customer",
      grams: round4(m.grams),
      paidSgd: round2(m.paid),
      isRecipient: customerId === recipientId,
    }))
    .sort((a, b) => Number(b.isRecipient) - Number(a.isRecipient) || b.grams - a.grams);
  const others = members.filter((m) => !m.isRecipient);
  return {
    listingId: listing.id,
    serial: listing.serial,
    groupName: listing.group_name,
    recipientId,
    recipientName: names.get(recipientId) ?? "Unknown customer",
    members,
    transferredGrams: round4(others.reduce((s, m) => s + m.grams, 0)),
    transferredPaidSgd: round2(others.reduce((s, m) => s + m.paidSgd, 0)),
  };
}
 
/** The one-line Control Tower summary (also the WhatsApp page body). */
export function groupDeliverySummary(f: GroupDeliveryFacts): string {
  const others = f.members.filter((m) => !m.isRecipient);
  const pool = f.groupName ? ` (${f.groupName})` : "";
  return (
    `Group MAG bar ${f.serial}${pool} delivered to ${f.recipientName}: ` +
    `${others.length} other member${others.length === 1 ? "" : "s"} pass ${f.transferredGrams} g ` +
    `(paid S$${f.transferredPaidSgd.toFixed(2)}) to the recipient. ` +
    `Review the group-transfer red flag (IPPC §9.2) and approve before dispatch.`
  );
}

/**
 * Gate for every dispatch step: a 409-ready message while a group delivery waits
 * on (or was declined by) the compliance review, else null. Fails closed.
 */
export async function groupDeliveryHoldBlocker(db: SupabaseClient, orderId: string): Promise<string | null> {
  try {
    const listing = await findGroupListing(db, orderId);
    if (!listing) return null;
    const { data, error } = await db
      .from("agent_decisions")
      .select("status")
      .eq("agent_name", MAG_GROUP_DELIVERY_AGENT)
      .eq("related_entity_id", orderId);
    if (error) throw new Error(error.message);
    const statuses = ((data ?? []) as { status: string }[]).map((d) => d.status);
    if (statuses.some((s) => CLEARED.has(s))) return null;
    if (statuses.includes("denied")) {
      return `Compliance declined the delivery of group MAG bar ${listing.serial}. Do not dispatch — reverse or refund the order.`;
    }
    return `Group MAG bar ${listing.serial} is waiting for compliance approval. Approve it in Control Tower before dispatch.`;
  } catch (e) {
    console.error(`[mag-group-review] hold check failed for order ${orderId}:`, e);
    return "Could not confirm the compliance review for this MAG delivery — try again.";
  }
}
