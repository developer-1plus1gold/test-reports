/**
 * The quote form's pre-fill (docs/mag-quote-automation-design.md §3.1–3.2): every
 * catalog bar of the requested size that BullionKing can supply, cheapest
 * BullionKing landed cost first, each with our outright % (one bar) and the
 * suggested % for every quotable duration.
 *
 * Mukesh 2026-11-27: "we might not have a right size bar in stock … look at the
 * bar of the size requested that we can acquire at the lowest cost from
 * bullionking. We use that bar in our quote." Stock is shown, never the ranking.
 *
 * The cost is the Pricing Strategy cost source (assembleSupplierComparison →
 * BullionKing's cell): StoneX-indexed or agreed price, 2 % card fee excluded, as
 * in every margin comparison (standing decision 2026-11-14).
 */
import type { SupabaseClient } from "@supabase/supabase-js";
import { assembleSupplierComparison, type ComparisonRow, type SupplierComparison } from "@/lib/pricing/supplier-comparison";
import { quoteSize } from "./pool-offers";
import { quoteOptionsFor, parseCostOfCapitalPct, type QuoteOptionCalc } from "./quote-pricing";
 
export const BULLIONKING_SLUG = "bullionking";
 
export interface QuoteBarCandidate {
  productId: string;
  label: string;
  grams: number;
  /** BullionKing landed cost, % over spot. */
  bkCostPct: number;
  /** Our shop premium for ONE bar, % over spot; null when the bar has no outright price. */
  outrightPct: number | null;
  /** Bars of this product we hold, available now. */
  inStock: number;
  options: QuoteOptionCalc[];
}
 
export interface QuoteSuggestion {
  costOfCapitalPct: number;
  spotSgdPerGram: number | null;
  /** Cheapest BullionKing bar first; empty when BullionKing has none of this size. */
  candidates: QuoteBarCandidate[];
}
 
/**
 * Pure ranking from a supplier comparison. Excludes split parents (CombiBars are
 * sold as 1 g cards, not as a bar) and bars without an outright price (the
 * suggestion is outright % + capital, so there is nothing to start from).
 */
export function rankBullionKingBars(
  cmp: Pick<SupplierComparison, "suppliers" | "rows">,
  grams: number,
  ratePct: number,
  stockByProduct: ReadonlyMap<string, number> = new Map(),
): QuoteBarCandidate[] {
  const size = quoteSize(grams);
  const bk = cmp.suppliers.find((s) => s.slug === BULLIONKING_SLUG);
  if (!size || !bk) return [];
  const out: QuoteBarCandidate[] = [];
  for (const r of cmp.rows as ComparisonRow[]) {
    if (r.isSplitParent || quoteSize(r.grams)?.grams !== size.grams) continue;
    Iif (r.productType !== "standard" && r.productType !== "curve") continue;
    const cell = r.cells.find((c) => c.supplierId === bk.id);
    const cost = cell?.premiumOverSpotPct;
    if (cost == null || !Number.isFinite(cost) || r.retailPremiumPct == null) continue;
    out.push({
      productId: r.productId,
      label: r.label,
      grams: r.grams,
      bkCostPct: Math.round(cost * 1000) / 1000,
      outrightPct: r.retailPremiumPct,
      inStock: stockByProduct.get(r.productId) ?? 0,
      options: quoteOptionsFor(r.retailPremiumPct, cost, ratePct),
    });
  }
  return out.sort((a, b) => a.bkCostPct - b.bkCostPct || a.label.localeCompare(b.label));
}
 
/** The admin's cost-of-capital setting (mag_settings.cost_of_capital_pct), 7 when unset. */
export async function readCostOfCapitalPct(db: SupabaseClient): Promise<number> {
  const { data } = await db.from("mag_settings").select("cost_of_capital_pct").eq("id", 1).maybeSingle();
  return parseCostOfCapitalPct((data as { cost_of_capital_pct?: unknown } | null)?.cost_of_capital_pct);
}
 
export async function buildQuoteSuggestion(db: SupabaseClient, grams: number): Promise<QuoteSuggestion> {
  const [cmp, ratePct] = await Promise.all([
    assembleSupplierComparison(db, { includeHidden: true }),
    readCostOfCapitalPct(db),
  ]);
  const ids = cmp.rows.filter((r) => quoteSize(r.grams)?.grams === quoteSize(grams)?.grams).map((r) => r.productId);
  const stock = new Map<string, number>();
  if (ids.length) {
    const { data } = await db.from("inventory").select("product_id").eq("status", "available").in("product_id", ids).limit(5000);
    for (const r of (data ?? []) as { product_id: string }[]) stock.set(r.product_id, (stock.get(r.product_id) ?? 0) + 1);
  }
  return { costOfCapitalPct: ratePct, spotSgdPerGram: cmp.spotSgdPerGram, candidates: rankBullionKingBars(cmp, grams, ratePct, stock) };
}
