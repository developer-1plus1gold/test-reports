/**
 * MAG — what a customer is told about a piece's grams (Mukesh, 2026-11-10).
 *
 * A 1 oz piece is sold as 31 whole grams + a 0.1035 g fraction (D1). While that
 * fraction is unsold it is part of what is LEFT to buy — "Buy the rest" takes it.
 * Showing only whole grams ("28g of 31g available", "28g to go") next to a price
 * that includes the fraction is misleading, so every customer-facing figure uses
 * these: the piece's true weight, and what is left including an unsold fraction.
 * Whole-gram bars are unchanged (no fraction, same numbers as before).
 *
 * The quantity stepper still moves in whole grams — only Buy the rest takes the
 * fraction — so the stepper's "Max" stays a whole number.
 *
 * Pure: shared by the web shop and pinned against the native app (MagShopLogic).
 */
import type { MagListingPublic } from "./listings";
 
type PieceInput = Pick<MagListingPublic, "gramsTotal" | "gramsRemaining" | "pieceGrams" | "houseGrams" | "houseAvailable">;
 
const round4 = (v: number) => Math.round(v * 10_000) / 10_000;
 
/** The fraction still for sale (0 once someone owns it, or for a whole-gram bar). */
export function unsoldFraction(l: Partial<Pick<PieceInput, "houseGrams" | "houseAvailable">>): number {
  const h = Number(l.houseGrams ?? 0);
  return l.houseAvailable && h > 0 ? h : 0;
}
 
/** The piece's true weight: 31.1035 for 1 oz, else the bar's grams. */
export function pieceTotalGrams(l: PieceInput): number {
  return l.pieceGrams != null && Number(l.pieceGrams) > 0 ? Number(l.pieceGrams) : Number(l.gramsTotal);
}
 
/** Grams left to buy on the piece, including an unsold fraction. */
export function pieceLeftGrams(l: PieceInput): number {
  return round4(Math.max(0, Number(l.gramsRemaining)) + unsoldFraction(l));
}
 
/** Percent of the piece already bought (the progress bar). */
export function pieceSoldPct(l: PieceInput): number {
  const total = pieceTotalGrams(l);
  return total > 0 ? Math.round(((total - pieceLeftGrams(l)) / total) * 100) : 0;
}
 
/** "28.1035" / "31" — up to 4 decimals, no trailing zeros. */
export function fmtPieceGrams(g: number): string {
  return String(round4(g));
}
