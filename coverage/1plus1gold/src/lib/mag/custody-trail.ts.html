/**
 * MAG custody trail for one customer (Mukesh 2026-11-24): "Tinuttha bought 5 g,
 * sold 3 g and now has 2 g, but that trail is not available, so I can't tell if
 * the 2 g is correct."
 *
 * Why the screen alone can't show it: a partial sell-back SPLITS an allocation
 * row (20261110c) — the original row is reduced in place (5 g → 2 g) and a new
 * row carries the sold grams. Nothing on the row says it was ever 5 g.
 *
 * The trail is rebuilt from the records that DID keep the history, and it is a
 * real reconciliation, not a tautology:
 *   + Bought      — from the ORDER LINES (quantity × grams each), the
 *                   independent record of what was purchased.
 *   − Sold back   — allocation rows `sold_back` (+ `sell_requested` = in progress),
 *                   grouped by their buyback request.
 *   − Used to pay — rows `redeemed` (MAG spent on another order).
 *   − Delivered   — rows `delivered` (taken as a physical piece).
 *   ↔ Moved       — mag_transfers between bars (net zero for the customer).
 *   = Held        — rows `allocated` + `sell_requested`, which is what the
 *                   custody screen shows.
 * `reconciles` is true when bought − sold − used − delivered = held.
 * Checkout holds (`held`) and abandoned ones (`released`) are not custody.
 */
 
export type TrailEventKind = "bought" | "sold_back" | "sell_in_progress" | "used_to_pay" | "delivered" | "moved";
 
export interface TrailEvent {
  at: string;
  kind: TrailEventKind;
  /** Signed effect on the customer's grams (0 for a move between bars). */
  grams: number;
  /** Money for the line: what was paid, or the sell-back payout; null if unknown. */
  amountSgd: number | null;
  /** Human reference: order / buyback short id, bars. */
  ref: string;
  /** Running grams after this event. */
  balance: number;
}
 
export interface TrailInput {
  allocations: Array<{
    id: string; listing_id: string; order_item_id: string | null; grams: number;
    price_per_g_sgd: number | string | null; status: string; buyback_request_id: string | null;
    allocated_at: string | null; closed_at: string | null; created_at: string; updated_at: string;
  }>;
  orderItems: Array<{ id: string; order_id: string; quantity: number; grams_each: number | string; created_at: string }>;
  buybacks: Array<{ id: string; status: string; submitted_at: string | null; paid_at: string | null; total_payout_sgd: number | string | null; payout_method: string | null }>;
  transfers: Array<{ created_at: string; grams: number; from_serial: string | null; to_serial: string | null; reason: string | null }>;
  serialByListing: Record<string, string | null>;
}
 
export interface CustodyTrail {
  events: TrailEvent[];
  boughtGrams: number;
  outGrams: number;
  heldGrams: number;
  reconciles: boolean;
}
 
const CUSTODY = new Set(["allocated", "sell_requested", "sold_back", "redeemed", "delivered"]);
const short = (id: string) => id.slice(0, 8).toUpperCase();
// `+ 0` turns -0 into 0 (a customer with nothing out would read "−0 g").
const round = (n: number) => Math.round(n * 10000) / 10000 + 0;
 
export function buildCustodyTrail(input: TrailInput): CustodyTrail {
  const rows = input.allocations.filter((a) => CUSTODY.has(a.status));
  const raw: Array<Omit<TrailEvent, "balance">> = [];
 
  // + Bought: one event per order line the customer's MAG grams came from.
  const itemIds = new Set(rows.map((r) => r.order_item_id).filter((x): x is string => !!x));
  for (const item of input.orderItems.filter((i) => itemIds.has(i.id))) {
    const grams = Number(item.quantity) * Number(item.grams_each);
    const priced = rows.filter((r) => r.order_item_id === item.id && r.price_per_g_sgd != null);
    const perG = priced.length ? Number(priced[0].price_per_g_sgd) : null;
    const bars = [...new Set(rows.filter((r) => r.order_item_id === item.id).map((r) => input.serialByListing[r.listing_id]).filter(Boolean))];
    raw.push({
      at: item.created_at, kind: "bought", grams,
      amountSgd: perG != null ? Math.round(perG * grams * 100) / 100 : null,
      ref: `Order ${short(item.order_id)}${bars.length ? ` · bar ${bars.join(", ")}` : ""}`,
    });
  }
 
  // − Sold back / used to pay: grouped by the buyback request the rows point at.
  const byRequest = new Map<string, typeof rows>();
  for (const r of rows) {
    if (!["sold_back", "sell_requested", "redeemed"].includes(r.status)) continue;
    const key = r.buyback_request_id ?? `row:${r.id}`;
    byRequest.set(key, [...(byRequest.get(key) ?? []), r]);
  }
  for (const [key, group] of byRequest) {
    const req = input.buybacks.find((b) => b.id === key);
    const grams = group.reduce((s, r) => s + r.grams, 0);
    const redeemed = group.every((r) => r.status === "redeemed") || req?.payout_method === "mag_redeem";
    const inProgress = group.some((r) => r.status === "sell_requested");
    raw.push({
      at: req?.submitted_at ?? group[0].updated_at,
      kind: redeemed ? "used_to_pay" : inProgress ? "sell_in_progress" : "sold_back",
      // A sell-back in progress is still in our custody (still counted in held).
      grams: inProgress ? 0 : -grams,
      amountSgd: req?.total_payout_sgd != null && !redeemed ? Number(req.total_payout_sgd) : null,
      ref: `${redeemed ? "Redeem" : "Buyback"} ${req ? short(req.id) : "—"} · ${grams} g${inProgress ? " in progress" : ""}${req?.paid_at ? " · paid" : ""}`,
    });
  }
 
  // − Delivered.
  for (const r of rows.filter((x) => x.status === "delivered")) {
    raw.push({ at: r.closed_at ?? r.updated_at, kind: "delivered", grams: -r.grams, amountSgd: null,
      ref: `Delivered · bar ${input.serialByListing[r.listing_id] ?? "—"}` });
  }
 
  // ↔ Moves between bars (no change to the customer's total).
  for (const t of input.transfers) {
    raw.push({ at: t.created_at, kind: "moved", grams: 0, amountSgd: null,
      ref: `${t.grams} g moved ${t.from_serial ?? "?"} → ${t.to_serial ?? "?"}${t.reason ? ` (${t.reason})` : ""}` });
  }
 
  raw.sort((a, b) => a.at.localeCompare(b.at));
  let balance = 0;
  const events = raw.map((e) => ({ ...e, balance: (balance = round(balance + e.grams)) }));
 
  const boughtGrams = round(raw.filter((e) => e.kind === "bought").reduce((s, e) => s + e.grams, 0));
  const outGrams = round(raw.filter((e) => e.grams < 0).reduce((s, e) => s - e.grams, 0));
  const heldGrams = round(rows.filter((r) => r.status === "allocated" || r.status === "sell_requested").reduce((s, r) => s + r.grams, 0));
  return { events, boughtGrams, outGrams, heldGrams, reconciles: round(boughtGrams - outGrams) === heldGrams };
}
