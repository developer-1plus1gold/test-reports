/** Customer-readable text for a mag_delivery_refusal code. */
export function deliveryRefusalMessage(code: string | null | undefined): string {
  switch (code) {
    case "not_active":      return "This piece can no longer be delivered — please refresh.";
    case "locked":          return "This piece is already being checked out for delivery.";
    case "busy":            return "Part of this piece is in a checkout or a sell-back right now — please try again later.";
    case "incomplete":      return "Every gram of this piece must be owned before it can be delivered.";
    case "no_recipient":    return "This group bar has no delivery recipient yet — please contact us.";
    case "not_recipient":   return "Only the group's delivery recipient can take delivery of this bar.";
    case "terms_pending":   return "Every member must agree who receives this bar before it can be delivered.";
    case "not_sole_owner":  return "Only the customer who owns the whole piece can take delivery.";
    default:                return "This piece cannot be delivered right now — please refresh.";
  }
}

/** Courier states after which the bar has left us — never put back on the books automatically. */
const BAR_LEFT_US = new Set(["picked_up", "in_transit", "out_for_delivery", "delivered", "completed", "stored"]);
 
/**
 * May a cancelled / refunded delivery order be reversed automatically? Only
 * while the bar is still with us: the order was 'paid' (not completed) and no
 * courier has collected it. Anything else is an admin's call (the bar is with
 * the customer or on its way).
 */
export function canReverseDelivery(previousStatus: string, shipperStatus: string | null | undefined): boolean {
  if (previousStatus !== "paid") return false;
  return !shipperStatus || !BAR_LEFT_US.has(shipperStatus);
}
