/**
 * The MAG block on the account dashboard, pre-rendered for the native
 * SDUI Dashboard (/api/account/dashboard → `mag*` keys). It mirrors the web
 * MagHoldings card (components/account/MagHoldings.tsx): two tiles (grams held,
 * value at spot), the latest bar-audit line, and one row per bar.
 *
 * Strings are composed here, server-side, because the SDUI renderer has no
 * conditional interpolation or date formatting — and composing them once keeps
 * the native and web wording from drifting.
 */
import type { MagHoldingRow } from "@/lib/mag/listings";
import type { DeliverablePiece, MagDeliveredRecord } from "@/lib/mag/delivery";
import { formatBarSerial } from "@/lib/serials/bar-serial";
import { magTopUpSgd } from "@/lib/mag/completion";
import { recipientDeliveredLine } from "@/lib/mag/recipient-copy";
import { transferLine, type MagTransferRecord } from "@/lib/mag/transfer";
 
export interface MagLastAudit { signedAt: string; signedBy: string; serial: string }
 
export interface MagDashboardRow {
  listingId: string;
  label: string;
  serialText: string;
  heldGrams: number;
  paidSgd: number;
  hasPending: boolean;
  pendingText: string;
}
 
/** A completed piece the customer can take home (docs/mag-piece-delivery-design.md §12). */
export interface MagDeliverableRow {
  listingId: string;
  productId: string;
  label: string;
  serial: string;
  /** "Your 31.1035g 1 oz PAMP Cast Bar is complete" */
  titleText: string;
  /** What is left to pay for the gold before delivery (the unowned fraction); null = no live price. */
  topUpSgd: number | null;
  /** True when a fraction is still to pay (then `topUpSgd` is shown), false when fully paid. */
  hasTopUp: boolean;
  houseGrams: number;
  /** The bar type's tiers as JSON ({min_qty,max_qty,premium_pct}[]), for the native delivery line. */
  tiersJson: string;
}
 
export interface MagDashboardBlock {
  /** Show the MAG section at all: grams held, a piece to take home, or a delivered record. */
  magShowSection: boolean;
  magHasHoldings: boolean;
  magTotalGrams: number;
  magSpotValueSgd: number | null;
  magHasAudit: boolean;
  magAuditText: string;
  magRows: MagDashboardRow[];
  magHasDeliverable: boolean;
  magDeliverable: MagDeliverableRow[];
  magHasDelivered: boolean;
  /** One line per group bar delivered to someone else (§3 rule 5). */
  magDelivered: { listingId: string; text: string }[];
  magHasMoves: boolean;
  /** One dated line per holding we moved to another bar (MAG Agreement §4A). */
  magMoves: { id: string; text: string }[];
}
 
/** Same format as the web card: "5 October 2026" in Singapore time. */
export const fmtMagAuditDate = (iso: string): string =>
  new Date(iso).toLocaleDateString("en-SG", { day: "numeric", month: "long", year: "numeric", timeZone: "Asia/Singapore" });
 
export function magAuditSentence(a: MagLastAudit): string {
  return `Last bar audit ${fmtMagAuditDate(a.signedAt)} by ${a.signedBy} — bar ${a.serial} checked against its serial.`;
}
 
export function magDashboardBlock(
  holdings: MagHoldingRow[],
  lastAudit: MagLastAudit | null,
  spotSgdPerG: number | null,
  pieces: { deliverable?: DeliverablePiece[]; delivered?: MagDeliveredRecord[]; moves?: MagTransferRecord[] } = {},
): MagDashboardBlock {
  // Held = free grams + grams locked in an open sell-back (still theirs).
  const rows: MagDashboardRow[] = holdings
    .map((h) => ({
      listingId:  h.listingId,
      label:      h.label,
      serialText: formatBarSerial(h.serial, h.manufacturerSerial),
      heldGrams:  h.grams + h.pendingSaleGrams,
      paidSgd:    Math.round((h.paidSgd + (h.pendingSalePaidSgd ?? 0)) * 100) / 100,
      hasPending: h.pendingSaleGrams > 0,
      pendingText: h.pendingSaleGrams > 0 ? `${h.pendingSaleGrams}g sell-back in progress` : "",
    }))
    .filter((r) => r.heldGrams > 0);
  const total = rows.reduce((s, r) => s + r.heldGrams, 0);
  const deliverable: MagDeliverableRow[] = (pieces.deliverable ?? []).map((p) => ({
    listingId:  p.listingId,
    productId:  p.productId,
    label:      p.label,
    serial:     p.serial,
    titleText:  `Your ${p.pieceGrams}g ${p.label} is complete`,
    topUpSgd:   p.houseGrams > 0 && spotSgdPerG != null ? magTopUpSgd(p.houseGrams, spotSgdPerG, p.tiers) : null,
    hasTopUp:   p.houseGrams > 0,
    houseGrams: p.houseGrams,
    tiersJson:  JSON.stringify(p.tiers), // snake_case — the shape CartTier decodes
  }));
  const delivered = (pieces.delivered ?? []).map((d) => ({
    listingId: d.listingId,
    text: recipientDeliveredLine({ grams: d.grams, piece: d.piece, recipientName: d.recipientName, deliveredOn: fmtMagAuditDate(d.deliveredAt) }),
  }));
  const moves = (pieces.moves ?? []).map((m) => ({ id: m.id, text: transferLine(m) }));
  return {
    magShowSection:  total > 0 || deliverable.length > 0 || delivered.length > 0,
    magHasHoldings:  total > 0,
    magTotalGrams:   total,
    magSpotValueSgd: spotSgdPerG != null && total > 0 ? Math.round(total * spotSgdPerG * 100) / 100 : null,
    magHasAudit:     total > 0 && lastAudit != null,
    magAuditText:    lastAudit ? magAuditSentence(lastAudit) : "",
    magRows:         rows,
    magHasDeliverable: deliverable.length > 0,
    magDeliverable:  deliverable,
    magHasDelivered: delivered.length > 0,
    magDelivered:    delivered,
    // Shown with the holdings: a move never empties an account of grams.
    magHasMoves:     total > 0 && moves.length > 0,
    magMoves:        moves,
  };
}
