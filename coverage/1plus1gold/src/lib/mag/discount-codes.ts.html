/**
 * 1+1Gold MAG — promo and referral codes do not apply (testing item 15,
 * operator 2026-09-28). MAG is priced from the bar's own cost and its gram
 * premium tiers; a percentage or fixed promo would sell allocated gold below
 * that, and the referral discount is written for physical first-order bars.
 *
 * Refused, never silently dropped: a client that still shows a discount would
 * otherwise be charged more than it displayed. The cart and checkout hide the
 * code fields for MAG, so this only fires for a stale or crafted request.
 */
export const MAG_CODES_REFUSED =
  "Promo and referral codes can't be used on 1+1Gold MAG. Remove the code from your cart to continue.";
 
const present = (v: unknown) => typeof v === "string" && v.trim() !== "";
 
/** Returns the refusal message when a MAG order carries a promo or referral code. */
export function magDiscountCodeRefusal(isMag: boolean, promoCode: unknown, referralCode: unknown): string | null {
  return isMag && (present(promoCode) || present(referralCode)) ? MAG_CODES_REFUSED : null;
}
