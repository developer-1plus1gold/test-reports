 * docs/mag-reserved-bar-rhythm-design.md §3). Pure; one wording for email,
 * WhatsApp and the admin preview. Wording rule (lib/mag/wording.ts): grams and
 * premium %, never a value.

import { sgLongDate } from "./request-flow";
import type { RhythmEvent } from "./reserved-schedule";
 
export interface RhythmNotice { subject: string; lines: string[]; whatsappLine: string }
 
const g = (n: number) => `${n} g`;
const pct = (n: number) => `${Number(n.toFixed(2))}%`;
 
export interface ReleaseOutcome {
  /** Where THIS holder's grams are now ("none": they held no grams on the bar). */
  kind: "moved" | "stayed_open" | "none";
  oldPct: number;
  newPct: number;
  /** Their grams on the released bar. */
  grams: number;
  /** The open bar the grams moved to (moved), by serial and catalog name. */
  toSerial?: string | null;
  toLabel?: string | null;
  /** Moved to a bar of the same refinery, weight and fineness (else: same fineness only). */
  sameType?: boolean;
}
