/**
 * Where a released reservation's grams go (Mukesh 2026-11-28, Agreement §1B.5 draft). Each
 * holder's grams move — all of them, to one bar (§4A.3) — to the first open bar on sale that
 * can take them, in this order:
 *
 *   1. same refinery, same weight
 *   2. another refinery, same weight
 *   3. same refinery, another weight
 *   4. another refinery, another weight
 *
 * — the fineness always the same. The customer keeps a bar of the same size first (Mukesh
 * 2026-11-28, option 2: the first cut ranked "same refinery" above "same weight", and a 1 oz
 * Valcambi went to a 100 g Valcambi Combibar ahead of a 1 oz PAMP).
 *
 * Within a step: the nearest weight first, then the lowest MAG premium. Fineness must always
 * match and be known (it keeps the GST treatment). Only if no bar can take a holder's grams
 * does the released bar itself stay open with them on it (lib/mag/reserved-rhythm.ts). Pure.
 */
 
export interface OpenBarCandidate {
  listingId: string;
  serial: string;
  productId: string;
  /** The bar type's catalog name ("100 Gram Argor-Heraeus Cast Bar"). */
  label: string;
  refinery: string | null;
  grams: number;
  fineness: number | null;
  /** Grams nobody holds (mag_listing_availability). */
  gramsRemaining: number;
  /** The bar type's MAG premium % (its lowest-quantity tier). */
  pct: number;
}
 
export interface ReleaseSource { refinery: string | null; grams: number; fineness: number | null }
 
export type ReleaseTier = 1 | 2 | 3 | 4;
 
const norm = (s: string | null | undefined) => String(s ?? "").trim().toLowerCase();
const sameWeight = (a: number, b: number) => Math.abs(a - b) < 0.01;
 
/** The step a candidate falls in for this source bar, or null when it may not take the grams. */
export function releaseTier(src: ReleaseSource, c: Pick<OpenBarCandidate, "refinery" | "grams" | "fineness">): ReleaseTier | null {
  if (src.fineness == null || c.fineness == null || !(Number(src.fineness) > 0) || Number(src.fineness) !== Number(c.fineness)) return null;
  const sameRefinery = !!norm(src.refinery) && norm(src.refinery) === norm(c.refinery);
  const weight = sameWeight(Number(src.grams), Number(c.grams));
  if (weight) return sameRefinery ? 1 : 2;
  return sameRefinery ? 3 : 4;
}
 
/** The candidates that could take `grams`, best first. */
export function rankReleaseDestinations(
  src: ReleaseSource, candidates: readonly OpenBarCandidate[], grams: number,
): (OpenBarCandidate & { tier: ReleaseTier })[] {
  return candidates
    .map((c) => ({ ...c, tier: releaseTier(src, c) }))
    .filter((c): c is OpenBarCandidate & { tier: ReleaseTier } => c.tier != null && c.gramsRemaining >= grams)
    .sort((a, b) =>
      a.tier - b.tier
      || Math.abs(a.grams - src.grams) - Math.abs(b.grams - src.grams)
      || a.pct - b.pct
      || a.serial.localeCompare(b.serial));
}
