/**
 * MAG — the home-page copy, in ONE place.
 *
 * Read by BOTH the web home page (src/app/page.tsx) and the iOS home SDUI spec
 * (src/app/spec.ts), so the two surfaces cannot say different things about
 * MAG. Operator wording rules (2026-09-23):
 *   - always name it "MAG" where "My Allocated Gold" appears;
 *   - never describe MAG by reference to standard bars (its premiums are its
 *     own and will be lower, with or without tiers);
 *   - never "buy whole grams" (reads as "buy the whole bar");
 *   - never "the bar is never cut";
 * Operator corrections (2026-09-28, MAG testing round 1):
 *   - never "allocated to you by name" — the bar is SHARED; a customer owns a
 *     recorded share of it, not the bar;
 *   - no "no other dealer in Singapore offers" claim;
 *   - say what redemption actually is: the BUYBACK value of the grams is
 *     applied against the CURRENT price of a physical bar, balance paid
 *     (src/lib/mag/redeem.ts);
 *   - the point to lead with: you can move into a physical bar at our
 *     SMALLEST size, without first accumulating to 50g or 100g;
 *   - grams are bought "in 1-gram steps", never "whole grams".
 */

/** A step's points as one bulleted string — for renderers that take a single
 *  text (the iOS step_card's `description`). */
export function stepPointsText(points: readonly string[]): string {
  return points.map((p) => `• ${p}`).join("\n");
}
