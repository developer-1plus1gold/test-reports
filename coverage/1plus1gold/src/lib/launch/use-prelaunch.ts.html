"use client";
 
/**
 * usePrelaunchActive — the RUNTIME pre-launch flag for client chrome
 * (PrelaunchBanner + Navigation's safe-area inset). Both must agree on whether
 * the banner is shown, or the iOS notch inset gets doubled / orphaned.
 *
 * Initial value = the build-time env (NEXT_PUBLIC_PRELAUNCH_NOTICE) so SSR and
 * the first client render match (no hydration mismatch, no flash during normal
 * pre-launch). It then confirms against /api/config — which returns
 * prelaunch.active=false once the launch ceremony completes — and both
 * consumers flip together. If the env already says "live", we never override.
 */
 
import { useEffect, useState } from "react";
 
const ENV_ACTIVE = process.env.NEXT_PUBLIC_PRELAUNCH_NOTICE === "true";
 
let cached: boolean | null = null;
let inflight: Promise<boolean> | null = null;
 
function fetchActive(): Promise<boolean> {
  if (cached !== null) return Promise.resolve(cached);
  if (inflight) return inflight;
  inflight = fetch("/api/config", { cache: "no-store" })
    .then((r) => (r.ok ? r.json() : null))
    .then((j) => {
      const a = j?.prelaunch?.active;
      cached = typeof a === "boolean" ? a : ENV_ACTIVE;
      return cached;
    })
    .catch(() => {
      cached = ENV_ACTIVE; // fail-closed: stay in the env's pre-launch state
      return cached;
    })
    .finally(() => { inflight = null; });
  return inflight;
}
 
export function usePrelaunchActive(): boolean {
  const [active, setActive] = useState<boolean>(cached ?? ENV_ACTIVE);
  useEffect(() => {
    // Env already live → initial state is already false; nothing to override.
    if (!ENV_ACTIVE) return;
    let alive = true;
    void fetchActive().then((a) => { if (alive) setActive(a); });
    return () => { alive = false; };
  }, []);
  return active;
}
