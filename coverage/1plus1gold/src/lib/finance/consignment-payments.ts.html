/**
 * Consignment supplier payments as CASH movements.
 *
 * WHY THIS EXISTS (2026-08-25, spotted by Mukesh): a normal PO payment writes a
 * `po_payment` ledger row and shows up in the books. A CONSIGNMENT payment
 * writes no ledger row at all — deliberately, because the payable is an accrual
 * on `consignment_procurements` and a `po_payment` row would double-count the
 * liability (the PO payments route blocks consignment POs for exactly this
 * reason). The consequence nobody intended: S$13,132.93 of real supplier
 * payments had no line item anywhere in the cash view, so there was nothing to
 * reconcile a bank statement against.
 *
 * ▶ The BALANCE SHEET was never wrong. `impliedCash` is solved from the identity
 * Assets = Liabilities + Equity, so relieving the consignment payable drops
 * implied cash by the same amount automatically. Only the cash-MOVEMENT view was
 * blind. That is why the fix is derived at read time and feeds the cash report
 * ONLY — writing a real ledger row instead would drop cash twice.
 *
 * Same pattern as `gateway-fee-expenses.ts`: a real cost the operator should not
 * have to log by hand, derived from the row that already records it.
 */
 
// eslint-disable-next-line @typescript-eslint/no-explicit-any
type SupabaseClient = any;
 
/** The subset of a procurement that makes it a cash movement. */
export interface ConsignmentPaidRow {
  supplier_paid_at:         string | null;
  supplier_paid_amount_sgd: number | string | null;
  confirmed_cost_sgd?:      number | string | null;
  /** `swap` = settled by handing BullionKing a StoneX bar, so NO cash moved. */
  settlement_method?:       string | null;
}
 
/** Shape the cash-flow engine already understands (`LedgerCashRow`). */
export interface DerivedCashRow {
  entry_type:     string;
  entry_date:     string;
  amount_sgd:     number;
  funding_source: string;
}
 
/**
 * Map paid consignment obligations to cash-out rows.
 *
 * Typed as `po_payment` because that is what they ARE — money paid to a supplier
 * for stock — so the existing cash-flow switch books them under supplier
 * payments with no special-casing. `funding_source: "company"` because the
 * company settles these directly (unlike personally-fronted expenses, which the
 * engine excludes from company cash).
 *
 * Falls back to `confirmed_cost_sgd` when the paid amount was never stamped:
 * every row on prod has them equal to the cent, and a payment with a known
 * obligation but a missing amount should still appear rather than vanish.
 */
export function consignmentPaymentsToCashRows(rows: ConsignmentPaidRow[]): DerivedCashRow[] {
  const out: DerivedCashRow[] = [];
  for (const r of rows ?? []) {
    if (!r?.supplier_paid_at) continue;
    // A swap settles the payable with a bar, not money — it is not a cash
    // movement, and booking it as one would show a supplier payment the bank
    // statement never carries. Its offsetting swap SALE is excluded from
    // collected cash for the same reason (fetchSwapSalesOrderIds).
    if (r.settlement_method === "swap") continue;
    const amount = Number(r.supplier_paid_amount_sgd ?? r.confirmed_cost_sgd ?? 0);
    if (!Number.isFinite(amount) || amount <= 0) continue;
    const entryDate = String(r.supplier_paid_at).slice(0, 10);
    Iif (!/^\d{4}-\d{2}-\d{2}$/.test(entryDate)) continue;
    out.push({
      entry_type:     "po_payment",
      entry_date:     entryDate,
      amount_sgd:     amount,
      funding_source: "company",
    });
  }
  return out;
}
 
/**
 * Fetch consignment supplier payments settled within [startDate, endDate] as
 * cash rows. Soft-fails to an empty list — the cash report must still render if
 * this read hiccups, exactly as it would have before this source existed.
 */
export async function fetchConsignmentPaymentCashRows(
  db: SupabaseClient,
  startDate: string,
  endDate: string,
): Promise<DerivedCashRow[]> {
  try {
    const { data, error } = await db
      .from("consignment_procurements")
      .select("supplier_paid_at, supplier_paid_amount_sgd, confirmed_cost_sgd, settlement_method")
      .not("supplier_paid_at", "is", null)
      .gte("supplier_paid_at", `${startDate}T00:00:00.000Z`)
      .lte("supplier_paid_at", `${endDate}T23:59:59.999Z`);
    if (error) {
      console.warn("[finance/consignment-payments] read failed (non-fatal):", error.message);
      return [];
    }
    return consignmentPaymentsToCashRows((data ?? []) as ConsignmentPaidRow[]);
  } catch (err) {
    console.warn("[finance/consignment-payments] read threw (non-fatal):", err);
    return [];
  }
}
 
/**
 * Order ids of the swap SALES orders — the sale to BullionKing that settles a
 * consignment line with a StoneX bar. The order is `payment_status='paid'` (the
 * receivable is settled by the bar BullionKing gave us earlier, netting), so the
 * cash engine would otherwise count it as money collected. No money moved, so it
 * must be excluded from the cash view. Revenue and COGS are UNAFFECTED — the
 * swap sale is real revenue that offsets the consignment cost (design §3-C).
 *
 * Soft-fails to an empty set, logging loudly: a failed read overstates
 * collected cash by the swap amount rather than taking the report down.
 */
export async function fetchSwapSalesOrderIds(db: SupabaseClient): Promise<Set<string>> {
  try {
    const { data, error } = await db
      .from("consignment_procurements")
      .select("swap_sales_order_id")
      .eq("settlement_method", "swap")
      .not("swap_sales_order_id", "is", null);
    if (error) {
      console.error("[finance/consignment-payments] swap-order read failed — swap sales may show as collected cash:", error.message);
      return new Set();
    }
    return new Set(((data ?? []) as Array<{ swap_sales_order_id: string | null }>)
      .map((r) => r.swap_sales_order_id)
      .filter((id): id is string => typeof id === "string"));
  } catch (err) {
    console.error("[finance/consignment-payments] swap-order read threw:", err);
    return new Set();
  }
}
 
/**
 * Order ids paid with STORE CREDIT from a vault buyback — a customer's vaulted
 * gold bought back at its original price and the credit spent on a new order
 * (e.g. the 2026-10-25 vault → MAG conversions). Each buyback row carries
 * `payout_method='store_credit'` and points at the order it paid for via
 * `mag_redeem_order_id`. The order reads `payment_status='paid'`, but no money
 * came in, so the cash view must not count it as collected. Revenue and the
 * buyback cost are unaffected.
 *
 * Soft-fails to an empty set, logging loudly — same trade as the swap helper.
 */
export async function fetchStoreCreditOrderIds(db: SupabaseClient): Promise<Set<string>> {

    const { data, error } = await db
      .from("buyback_requests")

    if (error) {
      console.error("[finance/consignment-payments] store-credit order read failed — credit-paid orders may show as collected cash:", error.message);
      return new Set();

  } catch (err) {
    console.error("[finance/consignment-payments] store-credit order read threw:", err);
    return new Set();
  }
}
