/**
 * Balance Sheet aggregation — pure logic, unit-testable without DB.
 *
 * Accounting identity: Assets = Liabilities + Equity.
 *
 * Components:
 *   Assets
 *     - Inventory (on-hand) — landed value of gold still held, i.e. rows in
 *       status IN ('available', 'reserved_for_order'), each valued at its OWN
 *       cost_per_g × the product's grams. Supplied pre-aggregated as
 *       `goldInventoryValueSgd` (v_product_stock.held_cost_sgd) — this module
 *       no longer walks inventory rows. NOTE: still a point-in-time snapshot of
 *       the inventory table TODAY, not a true historical reconstruction.
 *       Acceptable v1 limitation; documented on the page so the operator knows.
 *     - Receivables (AR) — buybacks received but not yet paid? For us
 *       these are LIABILITIES (we owe the customer), not AR. Real AR for
 *       1Plus1Gold is effectively 0 today.
 *     - Implied cash — residual that makes the identity balance. Lets
 *       Mukesh sanity-check against his actual bank balance.
 *
 *   Liabilities
 *     - Outstanding loans = sum(loan_in) − sum(loan_repayment) up to as_of
 *     - Buyback payable = sum(total_payout_sgd) for buyback_requests where
 *       status IN ('received', 'approved') AND paid_at IS NULL
 *     - Reimbursements payable (Due to funder) = sum(amount_sgd) for
 *       expense + po_payment rows with funding_source='personal', LESS
 *       sum(amount_sgd) for 'reimbursement' rows. Grouped per funder. A
 *       personally-funded cost is economically (funder lends $X) + (company
 *       pays $X): the expense/inventory leg is already counted; this is the
 *       matching liability that stops the implied-cash plug from silently
 *       assuming company cash paid for it.
 *
 *   Equity
 *     - Capital injected (cumulative) − Owner draws (cumulative)
 *     - Retained earnings = cumulative net income from the P&L
 *       (revenue − COGS − expenses) up to as_of
 */
 
export interface LedgerCumRow {
  /** Inclusive at-or-before as_of */
  entry_type: string;
  amount_sgd: number | null;
  entry_date: string;
  /** 'company' | 'personal' — only 'personal' on expense/po_payment accrues
   *  a reimbursable payable. Optional for back-compat with older callers. */
  funding_source?: string | null;
  /** Who fronted the money (multi-funder). Null → grouped as 'Unattributed'. */
  funder?: string | null;
}

  lifecycle:              string | null;

  payment_status:         string | null;

  subtotal_ipm_sgd:       number | null;
  subtotal_taxable_sgd:   number | null;
  shipping_fee_sgd:       number | null;

export interface AccessoryCogsCumRow {

/** A consignment procurement. Once RECEIVED, its confirmed cost is a payable we
 *  owe the supplier until supplier_paid_at is set (standard AP). Caller passes
 *  received rows with received_at ≤ as_of; COGS rides the inventory path. */
export interface ConsignmentPayableCumRow {
  status:             string;      // gated on 'received' here
  confirmed_cost_sgd: number | null;
  supplier_paid_at:   string | null;
}
 
export interface BalanceSheet {
  asOf:        string;
  assets: {
    /** Gold bullion on hand (serialized `inventory` table). */
    inventory_sgd:     number;
    /** Non-gold counter stock (products.stock_qty × avg landed cost). Previously
     *  hidden inside implied cash — now surfaced explicitly per category. */
    accessories_sgd:           number;
    consumables_sgd:           number;
    /** Equipment NET of accumulated depreciation — what is still on the books. */
    equipment_sgd:             number;
    /** Landed cost as received, before depreciation. Disclosure. */
    equipmentGross_sgd:        number;
    /** Accumulated depreciation netted off, capped at the gross. Disclosure. */
    equipmentDepreciation_sgd: number;
    fulfilmentConsumables_sgd: number;
    /** accessories + consumables + equipment + fulfilment_consumables. */
    nonGoldInventory_sgd:      number;
    receivables_sgd:   number;
    /** Cost paid for but not yet consumed — the unexpired half of an amortised
     *  entry. Without it, implied cash absorbs the remainder. */
    prepaidExpenses_sgd: number;
    impliedCash_sgd:   number;
    total_sgd:         number;
  };
  liabilities: {
    outstandingLoans_sgd:    number;
    buybackPayable_sgd:      number;
    /** Confirmed consignment cost we owe suppliers, not yet paid. */
    consignmentPayable_sgd:  number;
    reimbursementsPayable_sgd: number;
    /** Cash received for orders paid but not yet delivered (revenue recognised at
     *  delivery). Earned → moves to retained earnings when the order completes. */
    deferredRevenue_sgd:     number;
    /** Net amount still owed per funder (accrued − reimbursed), descending. */
    reimbursementsByFunder:  Array<{ funder: string; amount_sgd: number }>;
    total_sgd:               number;
  };
  equity: {
    capitalIn_sgd:         number;
    ownerDraw_sgd:         number;
    retainedEarnings_sgd:  number;
    total_sgd:             number;
  };
  /**
   * Memo lines — disclosure, NOT part of any total.
   *
   * Deliberately outside `assets`: this is a statement about the QUALITY of the
   * inventory figure, not a competing measurement of it. Adding it anywhere in
   * the arithmetic would break the accounting identity and, worse, imply we have
   * a better number than stock_qty. We don't — we just know how much of it the
   * book cannot yet explain.
   */
  memo: {
    /**
     * Non-gold stock value with no supporting derivation (Σ delta × landed cost
     * across the four counter categories). POSITIVE means assets are overstated
     * by this much until it is explained; negative means the counter sits below
     * what the derivation predicts, which is not an audit risk.
     */
    unreconciledNonGoldInventory_sgd: number;
    /** Counter SKUs carrying a non-zero reconciliation delta. */
    unreconciledSkus:                 number;
  };
  /** Sanity check — should always be balanced; delta exposes any rounding drift. */
  identity: {
    assets_sgd:                number;
    liabilitiesPlusEquity_sgd: number;
    delta_sgd:                 number;
    balanced:                  boolean;
  };
}
 
const BUYBACK_PAYABLE_STATUSES   = new Set(["received", "approved"]);
 
export function buildBalanceSheet(args: {
  asOf:           string; // YYYY-MM-DD
  ledger:         LedgerCumRow[];
  orders:         PaidOrderCumRow[];
  conversions:    PaidConversionCumRow[];
  buybacks:       BuybackPayableRow[];
  /** Confirmed consignment procurements with confirmed_at ≤ as_of. Their cost is
   *  a COGS (folded into `cogs` below) AND, until paid, a liability. */
  consignmentPayables?: ConsignmentPayableCumRow[];
  /**
   * Landed value of gold we still hold (available + reserved_for_order),
   * aggregated in SQL by `v_product_stock.held_cost_sgd`.
   *
   * Was an array of every inventory row, summed here. That is a database's job:
   * the select was unbounded, so past PostgREST's row cap it would have silently
   * dropped rows and understated the gold asset with no error to notice. The
   * valuation itself is unchanged — still per-lot, since each inventory row
   * carries its own cost_per_g_sgd (proven identical before/after, 20260720d).
   */
  goldInventoryValueSgd: number;
  /** Cumulative gold COGS as of the date — `gold_cogs_as_of()`, same reasoning. */
  goldCogsSgd:    number;
  /** Cumulative accessory COGS rows. */
  accessoryCogs:  AccessoryCogsCumRow[];
  /** On-hand holding value (stock_qty × avg landed cost) per non-gold counter
   *  category. A point-in-time snapshot, same caveat as gold inventory. */
  nonGoldInventory: {
    accessories:           number;
    consumables:           number;
    equipment:             number;
    fulfilmentConsumables: number;
  };
  /** Disclosure only — reported as a memo, never folded into any total. */
  unreconciledNonGold?: { value_sgd: number; skus: number };
  /** Cumulative SGD of inventory written off / sampled via stock_adjustments up
   *  to as_of. The counterpart to the asset those adjustments already removed
   *  from nonGoldInventory — omit it and the sheet stops balancing. */
  stockWriteOffs?: number;
  /**
   * Cumulative equipment depreciation up to as_of (see
   * lib/finance/equipment-depreciation).
   *
   * ⚠️ ONE number, used TWICE below — added to `expenses` and subtracted from
   * the equipment asset — because both halves must move together and passing
   * them separately is how they drift. Equipment is bought through a PO
   * (`po_payment`, not a P&L expense) and capitalised as non-gold inventory, so
   * writing it down is a genuine double entry: equity falls and the asset falls.
   *
   * Do only the first half and NOTHING FAILS. `impliedCash` is SOLVED from the
   * identity, so the sheet still reports `balanced: true` while the cash plug
   * silently absorbs the whole charge. That is the same trap `stockWriteOffs`
   * exists for, arriving from the other direction.
   */
  equipmentDepreciation?: number;
  /**
   * Unexpired portion of amortised expenses as of the date — the PREPAID ASSET
   * (see lib/finance/amortisation).
   *
   * ⚠️ The mirror image of `equipmentDepreciation`, and omitting it fails the
   * same silent way. A personally-funded prepaid puts the FULL amount into the
   * director's-loan liability the day it is paid, while only one month's slice
   * reaches the P&L. Without this asset the cash plug absorbs the difference and
   * implied cash is OVERSTATED by the unexpired remainder — and the sheet still
   * reports `balanced: true`, because it balances by construction.
   */
  prepaidExpenses?: number;
}): BalanceSheet {
  const { asOf, ledger, orders, conversions, buybacks, goldInventoryValueSgd, goldCogsSgd, accessoryCogs, nonGoldInventory } = args;
 
  // Equity components (cumulative, up to as_of — caller filters).
  // `stockWriteOffs` (2026-07-19) is seeded from inventory written off via
  // stock_adjustments. It MUST be here, not only in the P&L: nonGoldInventory
  // values the asset from products.stock_qty, which an adjustment already
  // lowered. Without the matching expense, assets would fall while equity
  // stayed put and the sheet would silently stop balancing — the loss absorbed
  // by nobody. Same rows, same landed-cost basis, as the P&L expense line.
  let capitalIn = 0, ownerDraw = 0, loanIn = 0, loanRepay = 0;
  let expenses = (args.stockWriteOffs ?? 0) + (args.equipmentDepreciation ?? 0);
  // Reimbursements payable, netted per funder: expense + po_payment rows that
  // were funded personally ACCRUE; 'reimbursement' rows RELIEVE. Note expense
  // rows still flow into `expenses` (P&L) above regardless of funding source —
  // the payable is the SEPARATE balance-sheet leg, so no double counting.
  const payableByFunder = new Map<string, number>();
  for (const e of ledger) {
    const amt = Number(e.amount_sgd ?? 0);
    if (e.entry_type === "capital_in")     capitalIn  += amt;
    else if (e.entry_type === "owner_draw") ownerDraw  += amt;
    else if (e.entry_type === "loan_in")    loanIn     += amt;
    else if (e.entry_type === "loan_repayment") loanRepay += amt;
    else if (e.entry_type === "expense")    expenses   += amt;
 
    const funder = (e.funder && e.funder.trim()) ? e.funder.trim() : "Unattributed";
    if ((e.entry_type === "expense" || e.entry_type === "po_payment") && e.funding_source === "personal") {
      payableByFunder.set(funder, (payableByFunder.get(funder) ?? 0) + amt);
    } else if (e.entry_type === "reimbursement") {
      payableByFunder.set(funder, (payableByFunder.get(funder) ?? 0) - amt);
    }
  }
 
  // Revenue is recognised at DELIVERY (lifecycle='completed' on-or-before as_of).
  // Cash received for orders not yet delivered is DEFERRED REVENUE — a liability,
  // not equity. When the order completes it moves from deferred → retained
  // earnings. This keeps the identity balanced (the cash sits in implied cash
  // either way; only its counterparty moves from equity to liability).
  const asOfTs = `${asOf}T23:59:59.999Z`;
  let revenue = 0, deferredRevenue = 0;
  for (const o of orders) {
    const value = Number(o.subtotal_ipm_sgd     ?? 0)
               +  Number(o.subtotal_taxable_sgd ?? 0)
               +  Number(o.shipping_fee_sgd     ?? 0)
               +  Number(o.processing_fee_sgd   ?? 0);
    const recognised = o.lifecycle === "completed"
      && !!o.lifecycle_completed_at && o.lifecycle_completed_at <= asOfTs;
    if (recognised) {
      revenue += value;
    } else if (o.payment_status === "paid"
      && o.lifecycle !== "cancelled" && o.lifecycle !== "refunded" && o.lifecycle !== "failed") {
      // Cash in, not yet delivered as of this date → deferred revenue.
      deferredRevenue += value;
    }
  }
  for (const c of conversions) {
    if (!c.paid_at) continue;
    revenue += Number(c.conversion_fee_sgd ?? 0);
  }
 
  // COGS (cumulative) — landed cost from inventory + PO lookup for accessories.
  // Consignment rides the SAME inventory path: its row (created at receipt) is
  // counted by gold_cogs_as_of when it sells at delivery, matched to revenue. So
  // there is NO consignment COGS booked here — only the payable below.
  let cogs = Number(goldCogsSgd ?? 0);
  for (const a of accessoryCogs) {
    cogs += Number(a.quantity ?? 0) * Number(a.landed_cost_sgd ?? 0);
  }
 
  // Consignment payable — standard accounts-payable: the obligation is booked
  // when we RECEIVE the goods (inventory + payable together), and relieved when
  // paid. A confirmed-but-unreceived line is a mere purchase commitment, not yet
  // a liability, so it does NOT count here.
  let consignmentPayable = 0;
  for (const c of args.consignmentPayables ?? []) {
    if (c.status !== "received") continue;
    if (c.supplier_paid_at) continue;
    consignmentPayable += Number(c.confirmed_cost_sgd ?? 0);
  }
 
  const retainedEarnings = revenue - cogs - expenses;
 
  // Inventory — point-in-time snapshot (gold, serialized). The available +
  // reserved_for_order filter now lives in v_product_stock.held_cost_sgd, so
  // "what counts as on hand" is defined once for the whole codebase.
  const inventoryValue = Number(goldInventoryValueSgd ?? 0);
 
  // Non-gold counter inventory (accessories / consumables / equipment /
  // fulfilment consumables). Bought for cash (po_payment — not a P&L expense), so
  // its value was previously buried in the implied-cash plug. Surfacing it as an
  // explicit asset AND subtracting it from implied cash below is a pure
  // reclassification: total assets and the identity are unchanged, but implied
  // cash now tracks the real bank balance more closely.
  const accessories           = Number(nonGoldInventory.accessories ?? 0);
  const consumables           = Number(nonGoldInventory.consumables ?? 0);
  // Equipment is carried NET of accumulated depreciation — the other half of the
  // expense added to `expenses` above. Floored at zero rather than capping the
  // depreciation itself: if a write-off already removed some of this asset via
  // stock_adjustments, that loss is expensed by stock-adjustment-expenses and
  // depreciating the full received cost on top would charge the same value
  // twice. The floor bounds that residual at the value actually there.
  const equipmentGross        = Number(nonGoldInventory.equipment ?? 0);
  const equipmentDepreciation = Number(args.equipmentDepreciation ?? 0);
  const equipment             = Math.max(0, equipmentGross - equipmentDepreciation);
  const fulfilmentConsumables = Number(nonGoldInventory.fulfilmentConsumables ?? 0);
  const nonGoldInventoryValue = accessories + consumables + equipment + fulfilmentConsumables;
 
  // Buyback payable.
  let buybackPayable = 0;
  for (const b of buybacks) {
    if (!BUYBACK_PAYABLE_STATUSES.has(b.status)) continue;
    if (b.paid_at) continue;
    buybackPayable += Number(b.total_payout_sgd ?? 0);
  }
 
  const outstandingLoans = loanIn - loanRepay;
 
  // Reimbursements payable — net per funder, then sum. A fully-reimbursed (or
  // over-reimbursed) funder nets to ≤0 and drops out of the breakdown, but
  // still contributes to the total so an overpayment correctly reduces the
  // liability (an over-repaid funder is a small receivable the operator settles).
  const reimbursementsByFunder = [...payableByFunder.entries()]
    .map(([funder, amount]) => ({ funder, amount_sgd: round2(amount) }))
    .filter((r) => Math.abs(r.amount_sgd) >= 0.01)
    .sort((a, b) => b.amount_sgd - a.amount_sgd);
  const reimbursementsPayable = reimbursementsByFunder.reduce((s, r) => s + r.amount_sgd, 0);
 
  const totalLiabilities = outstandingLoans + buybackPayable + consignmentPayable + reimbursementsPayable + deferredRevenue;
  const capitalEquity    = capitalIn - ownerDraw;
  const totalEquity      = capitalEquity + retainedEarnings;
 
  // Implied cash: solve identity Assets = Liabilities + Equity.
  // assets = gold inventory + non-gold inventory + receivables + impliedCash;
  // receivables = 0 in v1.
  const receivables   = 0;
  // Prepaid expenses — the unexpired half of an amortised entry. It is a REAL
  // asset (cost paid for, not yet consumed), so it comes OUT of the cash plug
  // rather than being left inside it. Floored at zero: a negative prepaid would
  // be an invented liability.
  const prepaid       = Math.max(0, Number(args.prepaidExpenses ?? 0));
  const totalAssetsTarget = totalLiabilities + totalEquity;
  const impliedCash   = totalAssetsTarget - inventoryValue - nonGoldInventoryValue - receivables - prepaid;
  const totalAssets   = inventoryValue + nonGoldInventoryValue + receivables + prepaid + impliedCash;
 
  return {
    asOf,
    assets: {
      inventory_sgd:             round2(inventoryValue),
      accessories_sgd:           round2(accessories),
      consumables_sgd:           round2(consumables),
      equipment_sgd:             round2(equipment),
      equipmentGross_sgd:              round2(equipmentGross),
      equipmentDepreciation_sgd:       round2(Math.min(equipmentDepreciation, equipmentGross)),
      fulfilmentConsumables_sgd: round2(fulfilmentConsumables),
      nonGoldInventory_sgd:      round2(nonGoldInventoryValue),
      receivables_sgd: round2(receivables),
      prepaidExpenses_sgd: round2(prepaid),
      impliedCash_sgd: round2(impliedCash),
      total_sgd:       round2(totalAssets),
    },
    liabilities: {
      outstandingLoans_sgd:      round2(outstandingLoans),
      buybackPayable_sgd:        round2(buybackPayable),
      consignmentPayable_sgd:    round2(consignmentPayable),
      reimbursementsPayable_sgd: round2(reimbursementsPayable),
      deferredRevenue_sgd:       round2(deferredRevenue),
      reimbursementsByFunder,
      total_sgd:                 round2(totalLiabilities),
    },
    equity: {
      capitalIn_sgd:        round2(capitalIn),
      ownerDraw_sgd:        round2(ownerDraw),
      retainedEarnings_sgd: round2(retainedEarnings),
      total_sgd:            round2(totalEquity),
    },
    // Memo — reported, never summed. See the `memo` docs on BalanceSheet.
    memo: {
      unreconciledNonGoldInventory_sgd: round2(args.unreconciledNonGold?.value_sgd ?? 0),
      unreconciledSkus:                 args.unreconciledNonGold?.skus ?? 0,
    },
    identity: {
      assets_sgd:                round2(totalAssets),
      liabilitiesPlusEquity_sgd: round2(totalAssetsTarget),
      delta_sgd:                 round2(totalAssets - totalAssetsTarget),
      balanced:                  Math.abs(totalAssets - totalAssetsTarget) < 0.01,
    },
  };
}
 
function round2(v: number): number {
  return Math.round(v * 100) / 100;
}
