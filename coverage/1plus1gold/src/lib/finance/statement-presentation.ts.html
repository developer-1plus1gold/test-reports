/**
 * How the Statement of Financial Position is PRESENTED — pure, no I/O. The engine
 * (buildBalanceSheet) measures; this decides the rows a reader sees.
 *
 * Two decisions (Mukesh 2026-11-17, preparing the StoneX APAC onboarding pack):
 *
 *  1. Partner funding is CAPITAL, not a loan. 1Plus1Gold is a partnership — there are
 *     no directors — and what the partner paid in personally is his capital. The
 *     engine books it as `reimbursementsPayable` (money owed back to the funder); here
 *     it moves to equity as "Partners' capital — <name>". Totals still balance: the
 *     same amount leaves liabilities and enters equity.
 *
 *  2. Cash is the REAL bank balance when one is given. The engine's cash is a residual
 *     that balances the sheet ("implied"), so it silently absorbs anything the books
 *     do not record — in our case personal payments made from the business accounts.
 *     With `cashAtBankSgd`, cash is that figure and the difference is shown as what it
 *     is: net drawings by the partner, deducted in his capital account. Without it,
 *     the internal statement still shows the residual, labelled as such; the external
 *     statement refuses to render without a bank figure (see the route).
 *
 * The EXTERNAL variant (for a counterparty such as a supplier's compliance team) also
 * folds internal creditor classes into "Other payables" when non-zero — a liability is
 * never hidden, only named generically — and drops zero rows.
 */
import type { BalanceSheet } from "./balance-sheet";
 
export type StatementVariant = "internal" | "external";
 
export interface PresentedRow { label: string; v: number }
 
export interface PresentedBalanceSheet {
  assets: PresentedRow[];
  totalAssets: number;
  liabilities: PresentedRow[];
  totalLiabilities: number;
  equity: PresentedRow[];
  totalEquity: number;
  /** Partner(s) named in the capital rows. */
  partners: string[];
  /** Funding the partner(s) put in (engine reimbursements + capital_in). */
  partnerContributed: number;
  /** Net drawings = recorded owner draws + (implied cash − actual cash) when a bank
   *  figure is given. Positive = taken out of the business. */
  partnerDrawings: number;
  cash: { value: number; basis: "bank" | "implied" };
  /** Assets − (liabilities + equity); 0 by construction, exposed for tests/guards. */
  delta: number;
}
 
const r2 = (n: number) => Math.round(n * 100) / 100;
const clean = (s: string) => s.replace(/[\u0000-\u001F\u007F‎‏‪-‮]/g, "").trim().slice(0, 60);
 
export function presentBalanceSheet(
  bs: BalanceSheet,
  opts: { variant: StatementVariant; cashAtBankSgd?: number | null },
): PresentedBalanceSheet {
  const ext = opts.variant === "external";
  const A = bs.assets, L = bs.liabilities, E = bs.equity;
 
  const partners = L.reimbursementsByFunder.filter((f) => Math.abs(f.amount_sgd) > 0.005).map((f) => clean(f.funder) || "Partner");
  const funding = L.reimbursementsPayable_sgd;
  const contributed = funding + E.capitalIn_sgd;
 
  const hasBank = typeof opts.cashAtBankSgd === "number" && Number.isFinite(opts.cashAtBankSgd);
  const cash = hasBank ? (opts.cashAtBankSgd as number) : A.impliedCash_sgd;
  // What the books expected in the bank but is not there = taken out by the partner.
  const unrecordedDrawings = hasBank ? A.impliedCash_sgd - cash : 0;
  const drawings = E.ownerDraw_sgd + unrecordedDrawings;
 
  const assets: PresentedRow[] = [
    { label: "Gold inventory, at landed cost", v: A.inventory_sgd },
    { label: "Other inventory", v: A.nonGoldInventory_sgd },
    { label: "Prepaid expenses", v: A.prepaidExpenses_sgd },
    { label: "Conversion fee credit receivable", v: A.receivables_sgd },
    { label: hasBank ? "Cash at bank" : "Cash and equivalents (implied — not reconciled to bank)", v: cash },
  ].filter((r, i, all) => i === all.length - 1 || Math.abs(r.v) > 0.005);
 
  const internalCreditors: PresentedRow[] = [
    { label: "Loans", v: L.outstandingLoans_sgd },
    { label: "Consignment & buyback payable", v: L.consignmentPayable_sgd + L.buybackPayable_sgd },
    { label: "Gold Yield payable", v: L.goldYieldPayable_sgd },
    { label: "Conversion payouts owed", v: L.conversionPayoutPayable_sgd },
  ];
  const deferred: PresentedRow = { label: "Deferred revenue (paid, not yet delivered)", v: L.deferredRevenue_sgd };
  const nonZero = (r: PresentedRow) => Math.abs(r.v) > 0.005;
  const otherPayables: PresentedRow = { label: "Other payables", v: internalCreditors.reduce((t, r) => t + r.v, 0) };
  const liabilities: PresentedRow[] = ext
    // External: never hide a liability, only name it generically; drop zero rows.
    ? [deferred, otherPayables].filter(nonZero)
    // Internal: every creditor class by name (the operator wants to see the zeros).
    : [deferred, ...internalCreditors.filter((r) => nonZero(r) || r.label === "Consignment & buyback payable" || r.label === "Gold Yield payable")];
  const totalLiabilities = L.total_sgd - funding;
 
  const who = partners.length === 1 ? partners[0] : "partners";
  const equity: PresentedRow[] = [
    { label: `Partners' capital contributed — ${who}`, v: contributed },
    ...(Math.abs(drawings) > 0.005 ? [{ label: `Less: partner drawings, net — ${who}`, v: -drawings }] : []),
    { label: E.retainedEarnings_sgd < 0 ? "Accumulated deficit" : "Accumulated profit", v: E.retainedEarnings_sgd },
  ];
  const totalEquity = contributed - drawings + E.retainedEarnings_sgd;
  const totalAssets = A.total_sgd - A.impliedCash_sgd + cash;
 
  return {
    assets: assets.map((x) => ({ ...x, v: r2(x.v) })), totalAssets: r2(totalAssets),
    liabilities: liabilities.map((x) => ({ ...x, v: r2(x.v) })), totalLiabilities: r2(totalLiabilities),
    equity: equity.map((x) => ({ ...x, v: r2(x.v) })), totalEquity: r2(totalEquity),
    partners, partnerContributed: r2(contributed), partnerDrawings: r2(drawings),
    cash: { value: r2(cash), basis: hasBank ? "bank" : "implied" },
    delta: r2(totalAssets - (totalLiabilities + totalEquity)),
  };
}
