/**
 * THE balance sheet as at a date — one assembly for every surface (Mukesh 2026-11-18:
 * "all of these use the same functions so they don't differ").
 *
 * Callers: the Balance Sheet screen (/api/admin/finance/balance-sheet), the financial
 * statements PDF and the monthly archive (assemble-financials). Before this, the screen had
 * its own copy of this code and the two disagreed on retained earnings (S$4,693 at 31 Oct),
 * gold inventory (a "today" view snapshot vs a point-in-time reconstruction) and accessory
 * cost (latest PO vs average landed cost).
 *
 * The basis, once:
 *  - RETAINED EARNINGS = the P&L's own net income from BOOKS_START to `asOf`
 *    (buildPnlForWindow — the same engine as the P&L screen), so equity and the income
 *    statement can never disagree. Callers that already built that P&L pass it in.
 *  - GOLD INVENTORY = gold_held_cost_as_of(asOf): a point-in-time reconstruction, Gold
 *    Yield participants' metal excluded at source, unsold MAG grams included, and (since
 *    20261118) a received consignment bar counted until it is delivered — the same window
 *    in which its supplier payable is on the books.
 *  - ACCESSORY COGS = the shared detector + average landed cost (lib/finance/accessory-cogs).
 *  - A failed read THROWS. The sheet balances by construction (cash is the plug), so a
 *    silently empty read would still report `balanced: true` with wrong numbers.
 */
// eslint-disable-next-line @typescript-eslint/no-explicit-any
type Db = any;
 
import { getInternalCustomerIds, excludeInternalOrders } from "@/lib/finance/internal-customers";
import { getStockAdjustmentExpenses } from "@/lib/finance/stock-adjustment-expenses";
import { getCategoryStock } from "@/lib/inventory/category-stock";
import { accumulatedDepreciation, fetchEquipmentReceipts } from "@/lib/finance/equipment-depreciation";
import { prepaidAssetAsOf } from "@/lib/finance/amortisation";
import { isAccessoryLine, fetchAccessoryUnitCosts } from "@/lib/finance/accessory-cogs";
import type { Jurisdiction } from "@/lib/finance/pnl";
import { buildBalanceSheet, type BalanceSheet, type AccessoryCogsCumRow, type PaidConversionCumRow } from "@/lib/finance/balance-sheet";
import { gyLiability, type GyObligationSaleRow, type GyPayoutRow } from "@/lib/finance/gold-yield-obligations";
import { buildPnlForWindow } from "@/lib/finance/pnl-window";
import { sgtDayEnd } from "@/lib/finance/sgt-window";
 
/** First day the books can hold activity — the business started in 2026; a year
 *  earlier is a safe floor for the cumulative (inception-to-date) P&L. */
export const BOOKS_START = "2025-01-01";
 
export interface BalanceSheetAsOf {
  sheet: BalanceSheet;
  /** Cumulative expense on the books at `asOf` (drives the retained-earnings note). */
  expensesToDate: number;
  /** The P&L net income BOOKS_START → asOf that the sheet carries as retained earnings. */
  retainedEarningsSgd: number;
}
 
function ok<T>(label: string, r: { data: T; error: { message?: string } | null }): T {
  if (r.error) throw new Error(`balance sheet: ${label} read failed — ${r.error.message ?? "unknown error"}`);
  return r.data;
}
 
/** Inception-to-date P&L net income — the retained earnings every balance sheet carries. */
export async function retainedEarningsAsOf(db: Db, internalIds: string[], asOf: string, jurisdiction: Jurisdiction = "sg"): Promise<number> {
  const pnl = await buildPnlForWindow(db, {
    period: "year", jurisdiction, internalIds,
    at: new Date(sgtDayEnd(asOf)),
    window: { start: BOOKS_START, end: asOf },
  });
  return pnl.netIncome_sgd;
}
 
export async function balanceSheetAsOf(
  db: Db, asOf: string,
  opts: { internalIds?: string[]; retainedEarningsSgd?: number; jurisdiction?: Jurisdiction } = {},
): Promise<BalanceSheetAsOf> {
  const internalIds = opts.internalIds ?? await getInternalCustomerIds(db);
  const asOfTs = sgtDayEnd(asOf);
  const [ledgerRes, ordersRes, conversionsRes, heldRes, buybacksRes, cogsRes, retained] = await Promise.all([
    db.from("ledger_entries").select("entry_type, amount_sgd, entry_date, funding_source, funder, amortise_months, amortise_start").lte("entry_date", asOf),
    excludeInternalOrders(db.from("orders").select("id, status, lifecycle, lifecycle_completed_at, payment_status, created_at, subtotal_ipm_sgd, subtotal_taxable_sgd, shipping_fee_sgd, processing_fee_sgd, discount_ipm_apportioned_sgd, discount_taxable_apportioned_sgd, promo_type, promo_discount_sgd").lte("created_at", asOfTs), internalIds),
    db.from("conversions").select("paid_at, conversion_fee_sgd, excess_grams, excess_value_sgd, fee_credit_sgd, excess_payout_sgd, approved_at, rejected_at, spot_sgd_at_approval, payout_paid_at").lte("paid_at", asOfTs),
    db.rpc("gold_held_cost_as_of", { p_as_of: asOfTs }),
    db.from("buyback_requests").select("status, paid_at, total_payout_sgd"),
    db.rpc("gold_cogs_as_of", { p_as_of: asOfTs }),
    opts.retainedEarningsSgd ?? retainedEarningsAsOf(db, internalIds, asOf, opts.jurisdiction ?? "sg"),
  ]);
 
  // eslint-disable-next-line @typescript-eslint/no-explicit-any
  const ledgerAll = (ok("ledger", ledgerRes) ?? []) as any[];
  const orderRows = (ok("orders", ordersRes) ?? []) as Array<{ id: string } & Record<string, unknown>>;
  const conversions = (ok("conversions", conversionsRes) ?? []) as PaidConversionCumRow[];
  const goldInventoryValueSgd = Number(ok("gold held", heldRes) ?? 0);
  const buybacks = (ok("buybacks", buybacksRes) ?? []) as Array<{ status: string; paid_at: string | null; total_payout_sgd: number | null }>;
  const goldCogsSgd = Number(ok("gold COGS", cogsRes) ?? 0);
 
  const accessoryCogs: AccessoryCogsCumRow[] = [];
  if (orderRows.length > 0) {
    const orderIds = orderRows.map((o) => o.id);
    // The shared accessory detector (lib/finance/accessory-cogs) — the P&L uses the same one.
    const accRes = await db.from("order_items").select("product_id, quantity, product_type, grams_each, products(grams)").in("order_id", orderIds);
    // eslint-disable-next-line @typescript-eslint/no-explicit-any
    const lines = ((ok("order items", accRes) ?? []) as any[])
      .map((r) => { const pr = Array.isArray(r.products) ? r.products[0] : r.products;
        return { product_id: (r.product_id ?? null) as string | null, quantity: r.quantity as number | null,
                 product_type: (r.product_type ?? null) as string | null,
                 grams_each: Number(r.grams_each ?? pr?.grams ?? 0) }; })
      .filter(isAccessoryLine);
    if (lines.length > 0) {
      const unitCost = await fetchAccessoryUnitCosts(db, lines.map((l) => l.product_id));
      for (const l of lines) if (l.product_id) accessoryCogs.push({ quantity: Number(l.quantity ?? 0), landed_cost_sgd: unitCost.get(l.product_id) ?? 0 });
    }
  }
 
  const [accStock, consStock, equipStock, fulfStock] = await Promise.all([
    getCategoryStock(db, "accessory"), getCategoryStock(db, "consumable"), getCategoryStock(db, "equipment"), getCategoryStock(db, "fulfilment_consumable"),
  ]);
  const stockWriteOffs = (await getStockAdjustmentExpenses(db, "1970-01-01", asOf)).reduce((s, e) => s + Number(e.amount_sgd ?? 0), 0);
  // The other half of the depreciation expense — see buildBalanceSheet.
  const equipmentDepreciation = accumulatedDepreciation(await fetchEquipmentReceipts(db), asOf);
  // Amortised expenses already PAID as of this date: the P&L recognises a monthly
  // slice, this carries what is left as an asset.
  const prepaidExpenses = prepaidAssetAsOf(ledgerAll.filter((e) => e.entry_type === "expense"), asOf);
 
  // RECEIVED consignment procurements — the supplier payable, booked at receipt (standard AP).
  // Its asset is the received bar, inside gold_held_cost_as_of until delivery (20261118).
  const consRes = await db.from("consignment_procurements").select("status, confirmed_cost_sgd, supplier_paid_at")
    .eq("status", "received").not("received_at", "is", null).lte("received_at", asOfTs);
  const consignmentPayables = (ok("consignment", consRes) ?? []) as Array<{ status: string; confirmed_cost_sgd: number | null; supplier_paid_at: string | null }>;
 
  // Gold Yield: the held asset needs no subtraction (gold_held_cost_as_of excludes it at
  // source, 20260922c); the paired obligation/payout input is still needed.
  const [gySalesRes, gyPayoutsRes] = await Promise.all([
    db.from("gold_yield_sales").select("sold_at, grams_sold, participant_share, principal_per_g").lte("sold_at", asOfTs),
    db.from("gold_yield_payouts").select("amount, paid_at").lte("paid_at", asOfTs),
  ]);
  const gyLiab = gyLiability(
    (ok("gold yield sales", gySalesRes) ?? []) as GyObligationSaleRow[],
    (ok("gold yield payouts", gyPayoutsRes) ?? []) as GyPayoutRow[],
  );
  // A sale with no recorded principal makes BOTH the expense and the payable a
  // floor, in the direction that understates what we owe. Said, not hidden.
  if (!gyLiab.complete) {
    console.warn("[balance-sheet-as-of] Gold Yield obligation/payable are a FLOOR — at least one sale carries no recorded principal.");
  }
 
  const ledger = ledgerAll as Array<{ entry_type: string; amount_sgd: number | null; entry_date: string; funding_source: string | null; funder: string | null }>;
  const expensesToDate = ledger.filter((e) => e.entry_type === "expense").reduce((s, e) => s + Number(e.amount_sgd ?? 0), 0) + stockWriteOffs;
  const retainedEarningsSgd = Number(retained);
 
  const sheet = buildBalanceSheet({
    asOf, ledger, stockWriteOffs, equipmentDepreciation, prepaidExpenses, retainedEarningsSgd,
    // eslint-disable-next-line @typescript-eslint/no-explicit-any
    orders: orderRows as any,
    conversions,
    goldInventoryValueSgd,
    buybacks,
    consignmentPayables,
    goldCogsSgd,
    // ONE input, two legs — see buildBalanceSheet's arg doc.
    goldYield: { obligation_sgd: gyLiab.owed_sgd, payouts_sgd: gyLiab.paid_sgd },
    accessoryCogs,
    nonGoldInventory: { accessories: accStock.totals.stockValue, consumables: consStock.totals.stockValue, equipment: equipStock.totals.stockValue, fulfilmentConsumables: fulfStock.totals.stockValue },
    unreconciledNonGold: {
      value_sgd: [accStock, consStock, equipStock, fulfStock].reduce((s, c) => s + c.totals.unreconciledValue, 0),
      skus:      [accStock, consStock, equipStock, fulfStock].reduce((s, c) => s + c.totals.unreconciledSkus, 0),
    },
  });
  return { sheet, expensesToDate, retainedEarningsSgd };
}
