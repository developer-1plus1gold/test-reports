/**
 * Combibar conversions → P&L expense lines (derived, not logged).
 *
 * ── What a conversion actually costs us ─────────────────────────────────────
 * A customer hands back 50 × 1g and receives one 50g bar. The metal is
 * like-for-like — 50g in, 50g out — so the gold nets to nothing. What we give
 * up is the PREMIUM embedded in the bar that leaves: a bar costs spot plus a
 * premium, and the cards come back carrying none.
 *
 *   cost of conversion = Σ(cost basis of the bars that leave)
 *                      − (grams × spot pinned at approval)
 *
 * One formula, both sourcing paths, no branch. On a freshly procured bar the
 * cost basis is what we just paid the supplier, so the residual IS the supplier
 * premium. On a bar already in stock it is that bar's historical procurement
 * cost, so the residual is its premium measured against the pinned spot.
 *
 * ── Why the books close ─────────────────────────────────────────────────────
 * At approval inventory rises by `grams × spot_pinned` (the cards, booked at
 * metal value by the approval route). At dispatch it falls by the outbound
 * bar's cost basis. The net asset change is exactly the negative of this
 * expense, so the balance sheet closes by construction — provided the outbound
 * bar is NOT also counted in gold COGS. See `isConversionSoldBar` below.
 *
 * ── Dated by the bar's sold_at, deliberately ────────────────────────────────
 * That is the same date the excluded gold-COGS row would have carried, so the
 * exclusion and this line always land in the SAME accounting period and a
 * conversion can never straddle a month boundary in two directions.
 *
 * ── It can be negative, and that is correct ─────────────────────────────────
 * A bar bought when spot was lower has a cost basis below the pinned metal
 * value, so the line comes out negative — a realised holding gain on that bar,
 * legitimate because we disposed of an asset at fair value. NOT clamped: a
 * floor at zero would silently drop real gains and break the asset identity
 * above. Splitting it into premium-given-away versus holding-gain would need
 * spot-at-procurement, whose prod coverage is thin (13/53 PO lines) — see §3.6
 * of docs/conversion-books-and-standard-bar-design.md.
 *
 * ── What this line deliberately does NOT capture ────────────────────────────
 * The returned cards come in at spot, carrying no premium. When they later sell
 * at the 1g retail premium that spread is earned in full, and it belongs to the
 * period they sell in — not to the conversion. Keeping it out is the point.
 *
 * ⚠️ `cost_of_conversion` is a DERIVED category. It is deliberately absent from
 * EXPENSE_CATEGORIES, so no manual ledger row can be filed against it and there
 * is nothing to double-count.
 */
 
import { categoryToSgTaxLine, categoryToUsTaxLine } from "@/lib/ledger";
import type { ExpenseRow } from "@/lib/finance/pnl";
 
/** A conversion, only the fields the derivation needs. */
export interface ConversionCostRow {
  id: string;
  /** SGD/gram pinned at approval. Null = never approved → no swap to book. */
  spot_sgd_at_approval: number | null;
  /** Grams handed back to the customer = grams received from them. */
  total_target_grams: number | null;
}
 
/** One outbound bar consumed for a conversion (inventory ⋈ order_fulfilments). */
export interface ConversionOutboundBarRow {
  conversion_id: string;
  /** Stamped at the status transition to 'sold' (20260723d). Null = not dispatched. */
  sold_at: string | null;
  cost_per_g_sgd: number | null;
  grams: number | null;
}
 
function round2(v: number): number {
  return Math.round(v * 100) / 100;
}
 
/**
 * Is this sold inventory row a conversion's outbound bar?
 *
 * The gold-COGS loop MUST skip these: their cost is expressed as
 * cost-of-conversion instead, and counting both charges the whole bar twice —
 * the single highest-risk line in this change.
 *
 * A conversion-sold bar has no order link, which `pnl.ts` already surfaces as a
 * null `source_channel`. That is a weaker signal than the conversion id itself,
 * so callers pass the id when they have it.
 */
export function isConversionSoldBar(row: { conversion_id?: string | null }): boolean {
  return Boolean(row.conversion_id);
}
 
/**
 * One `cost_of_conversion` expense row per dispatched conversion.
 *
 * Pure: the caller passes the already-fetched conversions and their outbound
 * bars, so this cannot drift from the set the P&L recognised and is
 * unit-testable without a DB.
 *
 * Skipped, each for a reason:
 *   • no pinned spot   — never approved, so no swap to book (and booking against
 *                        zero would charge the entire bar as premium)
 *   • no dispatched bar — the outbound leg has not happened; nothing has left
 *   • non-positive grams — a malformed conversion; better absent than wrong
 */
export function deriveCostOfConversionExpenses(
  conversions: ConversionCostRow[],
  bars:        ConversionOutboundBarRow[],
): ExpenseRow[] {
  const sgTax = categoryToSgTaxLine("cost_of_conversion");
  const usTax = categoryToUsTaxLine("cost_of_conversion");
 
  const barsByConversion = new Map<string, ConversionOutboundBarRow[]>();
  for (const b of bars) {
    if (!b.conversion_id || !b.sold_at) continue; // not dispatched yet
    const list = barsByConversion.get(b.conversion_id);
    if (list) list.push(b);
    else barsByConversion.set(b.conversion_id, [b]);
  }
 
  const out: ExpenseRow[] = [];
  for (const c of conversions) {
    const spot  = Number(c.spot_sgd_at_approval ?? 0);
    const grams = Number(c.total_target_grams ?? 0);
    if (!(spot > 0) || !(grams > 0)) continue;
 
    const mine = barsByConversion.get(c.id);
    if (!mine?.length) continue;
 
    const costOut = mine.reduce(
      (sum, b) => sum + Number(b.cost_per_g_sgd ?? 0) * Number(b.grams ?? 0),
      0,
    );
 
    // Latest sold_at among the bars — one dispatch in practice, but a conversion
    // with several bars must land in ONE period, and the last one to leave is
    // when the swap actually completed.
    const soldAt = mine
      .map((b) => b.sold_at as string)
      .sort()
      .at(-1) as string;
 
    out.push({
      entry_date:      soldAt.slice(0, 10),
      amount_sgd:      round2(costOut - grams * spot),
      category:        "cost_of_conversion",
      sg_tax_category: sgTax,
      us_tax_category: usTax,
    });
  }
  return out;
}
