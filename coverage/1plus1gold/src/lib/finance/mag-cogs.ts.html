/**
 * 1+1Gold MAG — cost of goods sold (design: docs/allocated-gold-design.md §6).
 *
 * WHY THIS EXISTS: a MAG sale never moves an inventory row. The physical bar
 * stays `mag_listed` for its whole life while customers buy grams of it, so
 * every surface that builds COGS from `inventory.status = 'sold'` saw a MAG
 * order as revenue with NO cost — pure margin — while the bar's cost sat
 * nowhere on the balance sheet (`v_product_stock` counts only available /
 * reserved rows).
 *
 * The cost of a MAG sale is the allocation itself: `grams × cost_per_g_sgd`,
 * where the cost was copied from the listing at the moment the grams were
 * allocated. It is recognised at `allocated_at` — the same moment the vault
 * order completes and its revenue is recognised (shipper `vault` / status
 * `stored` derives lifecycle `completed` immediately).
 *
 * Every status a sale ever passed through counts — `allocated`,
 * `sell_requested`, `sold_back`, `redeemed`: a later sell-back or redemption
 * does not unwind the original sale, it is a separate purchase from the
 * customer that puts the grams back into the listing's cost basis (see
 * `mag_complete_sell_back`). `held` and `released` never sold.
 *
 * The SQL mirror for cumulative / point-in-time figures is the MAG term in
 * `gold_cogs_as_of` (migration 20260923d) — these two must agree.
 */
import type { SupabaseClient } from "@supabase/supabase-js";
 
/** Statuses representing grams that were SOLD to a customer. */
export const MAG_SOLD_STATUSES = ["allocated", "sell_requested", "sold_back", "redeemed"] as const;
 
export interface MagCogsRow {
  /** Recognition date — the allocation time. Named `sold_at` so the row slots
   *  straight into the P&L's sold-inventory list. */
  sold_at:           string;
  cost_per_g_sgd:    number;
  grams:             number;
  source_channel:    string | null;
  order_item_id:     string | null;
  /** Always null: a MAG sale is never a conversion or a Gold Yield bar. Present
   *  so the row has the full SoldInventoryRow shape. */
  conversion_id:     null;
  gold_yield_bar_id: null;
  serial_number:     string | null;
  product_label:     string | null;
  /** The listing's physical inventory row, for supplier resolution. */
  inventory_id:      string | null;
  source_type:       string | null;
  source_id:         string | null;
}
 
const one = <T,>(v: T | T[] | null | undefined): T | null => (Array.isArray(v) ? v[0] ?? null : v ?? null);
 
/**
 * MAG grams sold in [startTs, endTs]. Throws on a query error — a silently
 * empty result would put the missing-cost defect straight back.
 */
export async function fetchMagCogsRows(
  db: SupabaseClient,
  startTs: string,
  endTs: string,
): Promise<MagCogsRow[]> {
  const { data, error } = await db
    .from("mag_allocations")
    .select(
      "allocated_at, grams, cost_per_g_sgd, order_item_id, " +
      "orders(source_channel), " +
      "mag_listings(serial, products(label), inventory(id, source_type, source_id))",
    )
    .in("status", [...MAG_SOLD_STATUSES])
    .gte("allocated_at", startTs)
    .lte("allocated_at", endTs);
  if (error) throw new Error(`mag_cogs_query_failed: ${error.message}`);
 
  // eslint-disable-next-line @typescript-eslint/no-explicit-any
  return ((data ?? []) as any[]).map((a) => {
    const order   = one(a.orders);
    const listing = one(a.mag_listings);
    const product = one(listing?.products);
    const inv     = one(listing?.inventory);
    return {
      sold_at:           String(a.allocated_at),
      cost_per_g_sgd:    Number(a.cost_per_g_sgd ?? 0),
      grams:             Number(a.grams ?? 0),
      source_channel:    (order?.source_channel ?? null) as string | null,
      order_item_id:     (a.order_item_id ?? null) as string | null,
      conversion_id:     null,
      gold_yield_bar_id: null,
      serial_number:     (listing?.serial ?? null) as string | null,
      product_label:     (product?.label ?? null) as string | null,
      inventory_id:      (inv?.id ?? null) as string | null,
      source_type:       (inv?.source_type ?? null) as string | null,
      source_id:         (inv?.source_id ?? null) as string | null,
    };
  });
}
 
/** Total cost of a set of MAG rows, summed raw and rounded once. */
export function magCogsTotal(rows: Array<Pick<MagCogsRow, "grams" | "cost_per_g_sgd">>): number {
  const raw = rows.reduce((s, r) => s + r.grams * r.cost_per_g_sgd, 0);
  return Math.round(raw * 100) / 100;
}
 
/** The P&L's sold-inventory row shape (`SoldInventoryRow` in pnl.ts). */
export function toSoldInventoryRow(r: MagCogsRow): {
  sold_at: string; cost_per_g_sgd: number | null; grams: number | null;
  source_channel: string | null; conversion_id: string | null; gold_yield_bar_id: string | null;
} {
  return {
    sold_at: r.sold_at, cost_per_g_sgd: r.cost_per_g_sgd, grams: r.grams,
    source_channel: r.source_channel, conversion_id: null, gold_yield_bar_id: null,
  };
}
