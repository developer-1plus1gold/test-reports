/**
 * Gold Yield → the books. Derived at read time, never written to the ledger.
 *
 * ▶ THE FINDING (2026-09-22). `gold_yield` appeared NOWHERE in `src/lib/finance`
 * or `src/app/api/admin/finance`. The consequence is not a missing expense line;
 * it is that **a participant's gold was booked as ours.**
 *
 * `gold_held_cost_as_of` — the balance-sheet gold asset — excludes consignment
 * explicitly and says why: it "carries a matching payable". Gold Yield is the
 * same shape (metal on our premises that somebody else owns, against which we
 * owe them when it sells) and was never brought into line. Measured on DEV:
 * S$4,105.53 of participant gold counted as our owned asset with no payable,
 * and S$11,810.24 of it flowed through our COGS.
 *
 * Three consequences, all previously invisible:
 *
 *   1. GROSS PROFIT WAS OVERSTATED ON EVERY GY RESALE, by exactly the
 *      `net_position_sgd` the participant's own card shows. COGS used the
 *      inventory row's `cost_per_g_sgd`, which for a GY split child is the
 *      participant's PAID basis — so the principal was covered by accident,
 *      while their 50% share and the un-hedged revaluation were expensed
 *      nowhere at all.
 *   2. NO LIABILITY. What we owe participants on sold grams was on no balance
 *      sheet.
 *   3. THE PAYOUT WAS NOT A CASH LINE. Recording one writes a row to
 *      `gold_yield_payouts` and nothing else — the identical shape to the
 *      consignment finding of 2026-08-25, where S$13,132.93 of real bank
 *      outflow had no line item to reconcile a statement against.
 *
 * ▶ WHY DERIVED RATHER THAN REAL LEDGER ENTRIES (operator decision, option 1:
 * mirror consignment). Double-entry is the conventional answer and it was
 * declined for a concrete reason: the cost of these grams is ALREADY derived
 * from the inventory row, so a ledger entry for the same economics
 * double-counts. That is the trap the 2026-08-25 consignment note states in
 * terms — "do NOT write a `po_payment` ledger row. The balance-sheet plug
 * already accounts for it, so a cash-out row would drop implied cash twice."
 * Deriving keeps one writer per number and self-corrects the moment a sale row
 * is fixed.
 *
 * ▶ AND IT IS THE OBLIGATION, NOT THE INVENTORY COST. The two are different
 * numbers and the difference is the whole point: the obligation is
 * `participant_share + principal_per_g × grams_sold`, which carries the
 * participant's profit share AND the un-hedged revaluation. The inventory row
 * knows neither.
 */
 
import type { ExpenseRow } from "@/lib/finance/pnl";
import type { LedgerCashRow } from "@/lib/finance/reconciliation";
import { settlementNumber } from "@/lib/gold-yield/settlement";
 
/** One accrued Gold Yield sale, as the books need to read it. */
export interface GyObligationSaleRow {
  /** When the gram sold. Dates the obligation into the same period as its revenue. */
  sold_at:           string | null;
  grams_sold:        number | string | null;
  /** Their half of the margin. A dollar total for the sale, NOT a per-gram rate. */
  participant_share: number | string | null;
  /** Their capital, PER GRAM — multiplied by `grams_sold` below. */
  principal_per_g:   number | string | null;
}
 
/** A recorded payout — money that has actually left the bank. */
export interface GyPayoutRow {
  amount:  number | string | null;
  paid_at: string | null;
}
 
export interface GyObligation {
  /** Their profit share — money we owe on the trade. */
  profit_sgd:    number;
  /** Their own capital going back. Not earnings, and much the larger term. */
  principal_sgd: number;
  /** The two above. What the sale actually cost us. */
  total_sgd:     number;
  /**
   * Sales whose principal is not recorded, so `total_sgd` is a FLOOR.
   *
   * ▶ IT IS A FLOOR IN THE DIRECTION THAT UNDERSTATES COST, which is the
   * unsafe direction for a P&L — so the caller must surface it rather than
   * quietly publishing the number. Same posture as the payout gate, which
   * refuses outright; the books cannot refuse to render, so they say so.
   */
  unknown_rows:  number;
}
 
function round2(v: number): number {
  return Math.round(v * 100) / 100;
}
 
/** `YYYY-MM-DD` in UTC, the shape `ExpenseRow.entry_date` and the ledger use. */
function isoDate(ts: string): string {
  return ts.slice(0, 10);
}
 
/**
 * What the Gold Yield grams sold in `[from, to]` cost us — end INCLUSIVE.
 *
 * ▶ A ROW WITH NO PRINCIPAL CONTRIBUTES ITS SHARE AND IS COUNTED AS UNKNOWN,
 * rather than being dropped whole. The share is owed whether or not we can
 * price the capital beside it, and discarding the row would throw away a cost
 * that is not in doubt — the same term-by-term rule `owedBreakdown` follows,
 * for the same reason.
 */
export function gyObligationForPeriod(
  rows: GyObligationSaleRow[] | null | undefined,
  /** `YYYY-MM-DD`, inclusive. */ from: string,
  /** `YYYY-MM-DD`, INCLUSIVE — the whole day. */ to: string,
): GyObligation {
  let profit = 0;
  let principal = 0;
  let unknown = 0;
 
  // ▶ DATE STRINGS, AND THE END IS INCLUSIVE, because that is what the callers'
  // SQL already did: `.gte("sold_at", from + T00:00:00Z).lte("sold_at", to +
  // T23:59:59.999Z)`. A half-open Date window here would disagree with the query
  // that fetched the rows and silently drop the last day of every period — and
  // the register would still render, just short. Same string convention as
  // `deriveEquipmentDepreciation`, which also takes `{ from, to }` as dates.
  const lo = `${from}T00:00:00.000Z`;
  const hi = `${to}T23:59:59.999Z`;
 
  for (const r of rows ?? []) {
    if (!r.sold_at) { unknown += 1; continue; }
    // Lexicographic on ISO-8601 UTC is chronological, and avoids constructing a
    // Date per row purely to compare it.
    const t = new Date(r.sold_at).toISOString();
    if (t < lo || t > hi) continue;
 
    const share = settlementNumber(r.participant_share);
    const cap   = settlementNumber(r.principal_per_g) * settlementNumber(r.grams_sold);
 
    Eif (Number.isFinite(share)) profit += share;
    if (Number.isFinite(cap))   principal += cap;
    if (!Number.isFinite(share) || !Number.isFinite(cap)) unknown += 1;
  }
 
  const p = round2(profit);
  const c = round2(principal);
  return { profit_sgd: p, principal_sgd: c, total_sgd: round2(p + c), unknown_rows: unknown };
}
 
/**
 * The obligation as a P&L expense row.
 *
 * ▶ ONE ROW FOR THE PERIOD, not one per sale. The register is read by a human
 * and fifty identical 1 g lines is noise, not detail; the per-sale record lives
 * in `gold_yield_sales`, which is where anybody auditing it should look.
 *
 * ⚠️ It is dated on the LAST DAY of the period rather than per sale, so a
 * caller must only ever pass a window that matches the report's own. The
 * alternative — a row per sale date — was rejected for the noise above.
 */
export function gyObligationExpense(
  ob: GyObligation,
  /** `YYYY-MM-DD` — the last day of the window the obligation was totalled over. */
  periodEndInclusive: string,
): ExpenseRow[] {
  if (ob.total_sgd === 0) return [];
  return [{
    entry_date:      isoDate(periodEndInclusive),
    amount_sgd:      ob.total_sgd,
    category:        "gold_yield_obligation",
    // Cost of metal that was never ours. "Other operating expenses" is where
    // `cost_of_conversion` — the same shape, premium given away on a swap —
    // already lands, so the two read together on the SG Form P.
    sg_tax_category: "Other operating expenses",
    us_tax_category: "Other deductions",
  }];
}
 
export interface GyLiability {
  owed_sgd:   number;
  paid_sgd:   number;
  /** Owed − paid, floored at zero. What we still owe participants. */
  unpaid_sgd: number;
  /** False when a sale has no recorded principal, so `owed_sgd` is a floor. */
  complete:   boolean;
}
 
/**
 * What we owe participants, and what we have already sent them.
 *
 * ▶ FLOORED AT ZERO ON PURPOSE. A negative would mean we have paid out more
 * than we owe, which is a data problem rather than a negative liability, and
 * rendering it as a negative creditor on a balance sheet would present that
 * problem as a tidy asset.
 */
export function gyLiability(
  sales: GyObligationSaleRow[] | null | undefined,
  payouts: GyPayoutRow[] | null | undefined,
): GyLiability {
  let owed = 0;
  let complete = true;
 
  for (const r of sales ?? []) {
    const share = settlementNumber(r.participant_share);
    const cap   = settlementNumber(r.principal_per_g) * settlementNumber(r.grams_sold);
    if (Number.isFinite(share)) owed += share; else Ecomplete = false;
    if (Number.isFinite(cap))   owed += cap;   else complete = false;
  }
 
  const paid = (payouts ?? []).reduce((s, p) => {
    const v = settlementNumber(p.amount);
    return Number.isFinite(v) ? s + v : s;
  }, 0);
 
  return {
    owed_sgd:   round2(owed),
    paid_sgd:   round2(paid),
    unpaid_sgd: round2(Math.max(0, owed - paid)),
    complete,
  };
}
 
/**
 * Recorded payouts as cash-out rows for the reconciliation view.
 *
 * ▶ WHY `expense` AND NOT A NEW TYPE. `buildCashFlowReport` classifies by
 * `entry_type`, and a type it does not know would be silently dropped —
 * re-creating the invisibility this exists to remove. `funding_source:
 * "company"` because the money leaves the business account.
 *
 * ⚠️ THIS FEEDS THE CASH REPORT ONLY. It must never reach the balance sheet's
 * ledger input: that sheet SOLVES for cash, so the payout is already implicit
 * in the plug, and a second cash-out row would drop implied cash twice. Exactly
 * the trap the 2026-08-25 consignment note records.
 */
export function gyPayoutCashRows(payouts: GyPayoutRow[] | null | undefined): LedgerCashRow[] {
  return (payouts ?? [])
    .filter((p) => p.paid_at && Number.isFinite(settlementNumber(p.amount)))
    .map((p) => ({
      entry_type:     "expense",
      entry_date:     isoDate(p.paid_at as string),
      amount_sgd:     round2(settlementNumber(p.amount)),
      is_capex:       false,
      funding_source: "company",
    }));
}
 
/**
 * Is this inventory row a participant's gold?
 *
 * ▶ KEYED ON `gold_yield_bar_id`, NOT ON `source_type`. The whole-bar path
 * mints a row with `source_type='gold_yield'`, but the SPLIT path mints the 1 g
 * children through the ordinary breakdown, so they carry
 * `source_type='breakdown'` and are invisible to a source-type test. Measured
 * on DEV: 50 of the 56 GY-tagged rows are `breakdown`. The bar id is on all of
 * them, by construction — it is what makes them a participant's.
 */
export function isGoldYieldInventoryRow(row: { gold_yield_bar_id?: string | null }): boolean {
  return Boolean(row.gold_yield_bar_id);
}
