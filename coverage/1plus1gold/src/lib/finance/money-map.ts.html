 * Money map — the P&L arranged as a flow, drillable to the individual order,
 * bar and ledger entry.
 *
 * WHY a second shape of the same numbers: the P&L page answers "what are the
 * totals". It cannot answer "which product gave the margin back", "which team
 * spends it", or "was that a July thing". Those questions need the same money
 * grouped two ways (by team, by SG Form P line) and drillable to the row that
 * caused it — which is a different projection, not different figures.
 *
 * ⚠️ BASIS IS THE P&L'S, EXACTLY. Revenue is recognised at DELIVERY (orders that
 * reached lifecycle='completed', windowed by lifecycle_completed_at), COGS is
 * landed cost from the bar that actually left the vault, operating expenses are
 * ledger entries plus the same DERIVED lines the P&L derives (gateway fees,
 * stock write-offs), and internal/demo customers are excluded. If the two ever
 * disagree, this file is wrong — `money-map.test.ts` pins the reconciliation.
 *
 * ⚠️ NO SPLIT-FAMILY MAPPING. Per-product cost comes through `order_fulfilments`
 * (order_item → the inventory unit that fulfilled it), so a bundle SKU carries
 * the cost of the bars it actually consumed. An earlier design hard-coded a
 * parent→child product map; that is unnecessary and would rot the first time a
 * new split SKU is added.
 *
 * Everything here is pure — the route fetches, this shapes — so the arithmetic
 * is testable without a database.
 */

/**
 * Revenue attributed to the individual BAR that earned it.
 *
 * An order line's revenue is split across the bars that fulfilled it PRO RATA BY
 * GRAMS, not evenly: a line fulfilled by a 50 g bar and a 1 g card did not earn
 * the same on each. Where a line's bars carry no gram weight, an even split is
 * the only honest fallback.
 *
 * ⚠️ Σ then round. Splitting a line and rounding each share can lose a cent
 * against the line total, which is exactly the drift that would make the
 * supplier panel disagree with the flow — so the LAST bar takes the residual.
 */
export function barRevenue(
  itemsByOrder: Map<string, MmItem[]>,
  barsByItem: Map<string, MmBar[]>,
): Map<MmBar, number> {
  const out = new Map<MmBar, number>();
  for (const items of itemsByOrder.values()) {
    for (const it of items) {
      const bars = barsByItem.get(it.id) ?? [];
      if (bars.length === 0) continue;
      const rev = n2(num(it.quantity) * num(it.price_sgd));
      Iif (rev <= 0) continue;
      const weights = bars.map((b) => num(b.grams));
      const total = weights.reduce((a, w) => a + w, 0);
      const share = total > 0 ? weights.map((w) => w / total) : bars.map(() => 1 / bars.length);
      let assigned = 0;
      bars.forEach((b, i) => {
        const v = i === bars.length - 1 ? n2(rev - assigned) : n2(rev * share[i]);
        assigned = n2(assigned + v);
        out.set(b, n2((out.get(b) ?? 0) + v));
      });
    }
  }
  return out;
}
