/**
 * Product-scoped promo codes — the ONE place that decides what a scoped code's
 * discount is computed on, and whether its minimum order is met.
 *
 * Added 2026-11-01 for occasion / sale campaigns: a campaign may limit its code
 * to the campaign's products (`promo_codes.eligible_product_ids`). Every charge
 * path (orders/create, HitPay/Razorpay gateway-create, Stripe create-intent),
 * the validate endpoint and the client checkout display call this, so the
 * number a customer sees and the number they are charged cannot diverge.
 *
 * Rules:
 *  - `eligible_product_ids` NULL → unscoped, exactly the pre-existing behaviour:
 *    the discount runs on the whole merchandise subtotal and the minimum order is
 *    checked against the caller's existing whole-order figure.
 *  - Scoped → the discount base is ONLY the gold lines whose product is listed,
 *    and the minimum order is checked against that same base ("spend S$500 on
 *    Diwali products"). Accessories and shipping are never in a scoped base.
 *  - A scoped code with NO eligible product in the cart is refused (even a
 *    free-delivery one) — otherwise "free delivery on Diwali gold" would waive
 *    the fee on an order with no Diwali gold in it.
 *
 * Pure + dependency-free so it can ship in the on-device JS engine as well.
 */
 
export interface ScopeLine {
  productId: string | null | undefined;
  lineSgd: number;
}
 
const round2 = (n: number) => Math.round(n * 100) / 100;
 
/** Normalise a DB/JSON value to a list of ids, or null for "unscoped". */
export function normaliseEligibleIds(v: unknown): string[] | null {
  if (!Array.isArray(v)) return null;
  const ids = v.filter((x): x is string => typeof x === "string" && x.length > 0);
  return ids.length > 0 ? ids : null;
}
 
/**
 * The subtotal a scoped promo is computed on, or `null` when the promo is not
 * scoped (the caller then uses its whole merchandise subtotal, as before).
 */
export function promoDiscountBaseSgd(eligibleProductIds: unknown, lines: ScopeLine[]): number | null {
  const ids = normaliseEligibleIds(eligibleProductIds);
  if (!ids) return null;
  const allowed = new Set(ids);
  let sum = 0;
  for (const l of lines) {
    if (l.productId && allowed.has(l.productId)) sum += Math.max(0, Number(l.lineSgd) || 0);
  }
  return round2(sum);
}
 
export type PromoScopeCheck =
  | { ok: true; baseSgd: number | null }
  | { ok: false; error: string };
 
export const SCOPED_NO_ELIGIBLE_MESSAGE =
  "This code only applies to selected products. Add one of them to your cart to use it.";
 
/**
 * Scope + minimum-order gate. `wholeOrderMinBasisSgd` is whatever figure the
 * caller compared the minimum against before scoping existed, so unscoped codes
 * keep their exact behaviour.
 */
export function checkPromoScope(args: {
  eligibleProductIds: unknown;
  minOrderSgd: number | null | undefined;
  lines: ScopeLine[];
  wholeOrderMinBasisSgd: number;
}): PromoScopeCheck {
  const base = promoDiscountBaseSgd(args.eligibleProductIds, args.lines);
  const min = args.minOrderSgd != null ? Number(args.minOrderSgd) : 0;
  if (base === null) {
    if (min > 0 && args.wholeOrderMinBasisSgd < min) {
      return { ok: false, error: `Minimum order of S$${min.toFixed(2)} required for this promo code.` };
    }
    return { ok: true, baseSgd: null };
  }
  if (base <= 0) return { ok: false, error: SCOPED_NO_ELIGIBLE_MESSAGE };
  if (min > 0 && base < min) {
    return { ok: false, error: `Spend at least S$${min.toFixed(2)} on the eligible products to use this code.` };
  }
  return { ok: true, baseSgd: base };
}
