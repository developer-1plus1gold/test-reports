/**
 * Does a compliance assessment need a HUMAN, or only a record?
 *
 * ▶ THE DISTINCTION THIS MODULE EXISTS TO MAKE. Every KYC must be LOGGED — the
 * case row is the CDD risk-assessment record, retained five years, and it
 * carries `next_review_at`, which is the only thing driving ongoing AML
 * monitoring. But logging is not the same as ACTIONING, and the code used to
 * conflate them: every assessment landed at `status='open'`, which means "a
 * Compliance Officer must decide something". For a LOW assessment with no
 * sanctions hit and no flags the agent has already decided and there is nothing
 * a human can add, so the register claimed work that did not exist.
 *
 * ▶ THE THREE OUTCOMES, and the middle one is the new part:
 *   escalated  a critical assessment — pages, and waits for a human.
 *   open       something to weigh: an elevated risk level, a sanctions hit, or
 *              any flag the agent raised. A human decides.
 *   cleared    assessed, nothing to decide, AND STILL MONITORED. Not `reviewed`
 *              — see below, that distinction is the whole design.
 *
 * ▶ WHY `cleared` AND NOT `reviewed`. `compliance-review` picks up cases where
 *       next_review_at <= now() AND status NOT IN ('closed','reviewed')
 * so filing a clean case as `reviewed` would quietly drop that customer out of
 * ongoing monitoring for good. `cleared` sits outside that exclusion, so the
 * scheduled re-review still happens. A clean customer today is exactly who
 * ongoing monitoring is for.
 *
 * ▶ FAIL TOWARD THE HUMAN. Anything unrecognised — an unknown risk level, an
 * absent one, a flag list we cannot read — resolves to `open`. The cost of an
 * unnecessary review is a minute of someone's time; the cost of auto-clearing
 * something that needed looking at is a regulatory one. They are not
 * comparable, so the default is not symmetric.
 */
 
/** Statuses a case can be BORN with. `reviewed`/`closed` are human acts. */
export type InitialCaseStatus = "escalated" | "open" | "cleared";
 
export interface AssessmentOutcome {
  riskLevel:     string | null | undefined;
  sanctionsHit:  boolean | null | undefined;
  /** Whatever the agent raised. Any flag at all means a human looks. */
  flags?:        unknown;
}
 
/**
 * The ONLY risk level that carries no human decision on its own. Deliberately a
 * single value rather than "not high and not critical": a level we do not
 * recognise must not fall through to clear.
 */
const CLEARABLE_RISK = "low";
 
export function initialCaseStatus(o: AssessmentOutcome): InitialCaseStatus {
  if (o.riskLevel === "critical") return "escalated";
 
  const level     = typeof o.riskLevel === "string" ? o.riskLevel.trim().toLowerCase() : "";
  const flagCount = Array.isArray(o.flags) ? o.flags.filter((f) => String(f ?? "").trim().length > 0).length : 0;
 
  // A sanctions hit is never clearable whatever the level says, and an
  // unreadable `flags` (not an array) is treated as "we do not know", which is
  // not the same as "none" — so it does not clear either.
  const flagsUnreadable = o.flags != null && !Array.isArray(o.flags);
 
  if (level === CLEARABLE_RISK && o.sanctionsHit !== true && flagCount === 0 && !flagsUnreadable) {
    return "cleared";
  }
  return "open";
}
 
/** True when the case was born needing nobody. Used to keep the Control Tower
 *  mirror out of the action queue while still writing it for the audit trail. */
export function isAutoCleared(status: string | null | undefined): boolean {
  return status === "cleared";
}
 
/**
 * What the register should say about a case nobody had to touch. Written into
 * `notes` at insert, so the row explains ITSELF — a status a reader has to look
 * up in code is a status that will be misread.
 */
export const AUTO_CLEARED_NOTE =
  "Auto-cleared: the assessment returned LOW risk with no sanctions hit and no flags, so there was nothing for a " +
  "Compliance Officer to decide. Logged in full and retained for the five-year AML period, and STILL SCHEDULED for " +
  "ongoing review — see next_review_at. A human may still escalate this case at any time.";
