/**
 * Consignment Queue — presentation helpers (PURE).
 *
 * The queue screen (`/admin/consignment/queue`) and its tests share these so the
 * on-screen figures, the profitability math, and the forwardable supplier
 * message can never drift from one another.
 *
 * TWO AUDIENCES, ONE SOURCE:
 *   • On-screen (internal) shows the full line: the message fields PLUS our
 *     revenue and profit — margin is ours to see.
 *   • The supplier message shows ONLY procurement fields (date, grams, qty,
 *     total grams, our indicative spot, and the est. cost at THEIR premium) plus
 *     a grand total. It MUST NEVER carry our selling price or profit — that is
 *     our margin and is never sent to a supplier. `buildSupplierMessage` is the
 *     single place that guarantee lives.
 *
 * "Cost at supplier premium" is the already-booked provisional cost (our spot at
 * sale, plus the supplier's spread and per-piece premium) until the supplier
 * confirms their own locked spot, after which the confirmed cost governs. It is
 * labelled "est." in the outbound message because the supplier locks their own
 * spot — see the two-spots note in `cost.ts`.
 */
 
import { orderGatewayFeeSgd, type GatewayFeeConfigRow } from "@/lib/finance/gateway-fee-expenses";
 
/** The subset of a queue line these helpers read. Mirrors `QueueLine` on the page. */
export interface ConsignmentLineView {
  grams: number;              // grams in ONE piece
  quantity: number;
  spotAtSaleSgd: number;      // our locked spot at sale — the spot the supplier honors
  provisionalCostSgd: number; // cost at OUR spot + supplier terms (spot×g + spread×g + premium)
  confirmedCostSgd: number | null; // cost at the supplier's locked spot (only if they didn't honor ours)
  premiumSgd: number;         // per-PIECE fixed premium (terms snapshot, e.g. 45)
  spreadPerGSgd: number;      // per-GRAM adder (terms snapshot, e.g. 0.40)
  revenueSgd: number | null;  // our selling price for the line (order_items.price × qty)
  createdAt: string;          // ISO — when the obligation landed
  /**
   * The line's share of the ORDER-level fees we absorb (payment-gateway fee +
   * marketplace commission), so profit reflects our NET take, not the pre-fee
   * gross. Populated by the route via `absorbedFeeForLine`; defaults to 0 when
   * fee inputs are unavailable so profit degrades to the old gross figure rather
   * than erroring.
   */
  absorbedFeesSgd?: number;
}
 
/** Order-level inputs needed to allocate the absorbed fees to a line. */
export interface OrderFeeContext {
  grandTotalSgd:          number;
  paymentMethod:          string | null;
  capturedGatewayFeeSgd:  number | null;   // orders.payment_gateway_fee_sgd (actual, if captured)
  externalChannelFeeSgd:  number;          // orders.external_channel_fee_sgd (marketplace commission)
  processingFeeChargedSgd: number;         // orders.processing_fee_sgd (what the CUSTOMER paid — offsets the gateway cut)
}
 
/** The raw `orders` columns (snake_case) an OrderFeeContext is built from. */
export interface OrderFeeRow {
  grand_total_sgd?:         unknown;
  payment_method?:          unknown;
  payment_gateway_fee_sgd?: unknown;
  external_channel_fee_sgd?: unknown;
  processing_fee_sgd?:      unknown;
}
 
/**
 * Map a raw `orders` row → OrderFeeContext. ONE place so the queue route and the
 * analytics route can never disagree on which columns feed the absorbed-fee calc
 * (before this, both hand-built the object and one could silently miss a field).
 */
export function orderFeeContext(ord: OrderFeeRow | null | undefined): OrderFeeContext {
  const o = ord ?? {};
  return {
    grandTotalSgd:           Number(o.grand_total_sgd ?? 0),
    paymentMethod:           o.payment_method == null ? null : String(o.payment_method),
    capturedGatewayFeeSgd:   o.payment_gateway_fee_sgd != null ? Number(o.payment_gateway_fee_sgd) : null,
    externalChannelFeeSgd:   Number(o.external_channel_fee_sgd ?? 0),
    processingFeeChargedSgd: Number(o.processing_fee_sgd ?? 0),
  };
}
 
/**
 * The fees we genuinely ABSORB on a consignment sale, attributed to ONE line.
 *
 * The gateway (HitPay/Stripe) deducts its cut from EVERY payout. But we also
 * CHARGE the customer a processing fee (`orders.processing_fee_sgd`) sized to
 * cover it — so the fee we truly absorb is only the shortfall:
 *
 *   • CARD  — customer is charged a matching processing fee → cost − charged ≈ 0,
 *             so the line keeps essentially its full gross (we lose ~nothing).
 *   • PAYNOW — customer is charged 0 → cost − 0 = the whole gateway cut is a real
 *             subsidy we eat. THIS is what the net-profit figure is meant to show.
 *
 * This is the same "payment-processing cushion" (charged − cost) the Books P&L
 * books; netting here keeps the two consistent. Plus any marketplace commission
 * (`external_channel_fee_sgd`), which is a real cost and is NOT passed to the
 * customer. Both are ORDER-level, allocated to the line by its revenue share.
 *
 * ⚠️ Earlier this subtracted the FULL gateway cut without crediting the
 * processing fee the customer paid — which double-hit every card order (a card
 * consignment sale showed a large false loss). Fixed 2026-08-11.
 */
export function absorbedFeeForLine(
  revenueSgd: number | null,
  order: OrderFeeContext,
  feeCfg: Map<string, GatewayFeeConfigRow>,
): number {
  if (revenueSgd == null || order.grandTotalSgd <= 0) return 0;
  const gatewayCost = order.paymentMethod
    ? orderGatewayFeeSgd(feeCfg.get(order.paymentMethod), order.grandTotalSgd, order.capturedGatewayFeeSgd)
    : 0;
  // The fee we NET absorb: the gateway's cut minus the processing fee the
  // customer already paid to cover it (clamped at 0 — an over-recovering fee is
  // separate P&L, not a negative cost on this line).
  const netGatewayAbsorbed = Math.max(0, gatewayCost - Math.max(0, order.processingFeeChargedSgd));
  const orderAbsorbed = netGatewayAbsorbed + Math.max(0, order.externalChannelFeeSgd);
  const share = Math.min(1, revenueSgd / order.grandTotalSgd);
  return round2(orderAbsorbed * share);
}
 
/** Total grams the line represents: one-piece grams × quantity. */
export function lineTotalGrams(l: Pick<ConsignmentLineView, "grams" | "quantity">): number {
  return round4(l.grams * l.quantity);
}
 
/**
 * The ACTUAL cost we book for the line (COGS): the supplier's confirmed cost
 * once they lock their spot, otherwise the provisional cost at our spot with
 * their premium. Used on screen (Cost column) + for profit. When the supplier
 * honors our spot — the normal case — this equals the our-spot cost below.
 */
export function lineCostSgd(l: Pick<ConsignmentLineView, "provisionalCostSgd" | "confirmedCostSgd">): number {
  return l.confirmedCostSgd != null ? l.confirmedCostSgd : l.provisionalCostSgd;
}
 
/**
 * Cost at OUR spot + the supplier's premium — what we quote the supplier, who
 * honors our spot in the majority case. Equals the stored provisional cost
 * (spot_at_sale × totalGrams + spread × totalGrams + premium × qty), so the
 * message ties exactly to what we booked.
 */
export function lineOurSpotCostSgd(l: Pick<ConsignmentLineView, "provisionalCostSgd">): number {
  return l.provisionalCostSgd;
}
 
/** The spot-metal portion at our spot: spot_at_sale × totalGrams. */
export function lineMetalSgd(l: Pick<ConsignmentLineView, "spotAtSaleSgd" | "grams" | "quantity">): number {
  return round2(l.spotAtSaleSgd * l.grams * l.quantity);
}
 
/**
 * The full premium on the line = spread × totalGrams + per-piece premium × qty.
 * Derived from the stored our-spot cost minus the metal so it ties to the cent
 * with what we booked (never recomputed independently). Worked, 2.5g×1 at
 * spot 167.14, spread 0.40, premium 45: 463.85 − 417.85 = 46.00.
 */
export function linePremiumSgd(l: Pick<ConsignmentLineView, "provisionalCostSgd" | "spotAtSaleSgd" | "grams" | "quantity">): number {
  return round2(l.provisionalCostSgd - l.spotAtSaleSgd * l.grams * l.quantity);
}
 
/** Human-readable premium breakdown, e.g. "S$45/pc + S$0.40/g". */
export function premiumBreakdown(l: Pick<ConsignmentLineView, "premiumSgd" | "spreadPerGSgd">): string {
  return `${sgd(l.premiumSgd)}/pc + ${sgd(l.spreadPerGSgd)}/g`;
}
 
/**
 * NET profit = revenue − supplier cost − the fees we absorb (gateway +
 * marketplace commission), or null when the line has no known revenue. The
 * absorbed fee defaults to 0 when the route couldn't supply the inputs, so this
 * degrades to the gross figure rather than erroring.
 */
export function lineProfitSgd(l: ConsignmentLineView): number | null {
  if (l.revenueSgd == null) return null;
  return round2(l.revenueSgd - lineCostSgd(l) - (l.absorbedFeesSgd ?? 0));
}

  totalOurSpotCostSgd: number;  // at our honored spot — what the supplier message quotes

export interface ConsignmentQueueLineInput extends ConsignmentLineView {

/**
 * Consolidated totals for the WHOLE open queue — drives the summary tiles at the
 * top of the Consignment Queue screen. Distinct from `summariseLines` (which
 * rolls up a selected subset for the supplier message): this rolls up every open
 * obligation and adds the operational counts (owed to suppliers, awaiting
 * confirmation, underwater, oldest age).
 */
export function summariseQueue(lines: ConsignmentQueueLineInput[]): {
  openLines:           number;
  pieces:              number;
  totalGrams:          number;
  totalPremiumSgd:     number;   // premium component of what we pay suppliers (over metal spot)
  owedToSuppliersSgd:  number;   // Σ cost of lines not yet paid to the supplier (the payable)
  awaitingConfirm:     number;   // lines still awaiting_supplier (need chasing)
  totalRevenueSgd:     number | null;
  totalAbsorbedFeesSgd: number;
  netProfitSgd:        number | null;
  netMarginPct:        number | null;
  underwaterCount:     number;
  oldestAgeDays:       number;
} {
  const base = summariseLines(lines);
  let owed = 0;
  let awaitingConfirm = 0;
  let underwaterCount = 0;
  let oldestAgeDays = 0;
  for (const l of lines) {
    if (!l.supplierPaid) owed += lineCostSgd(l);
    if (l.status === "awaiting_supplier") awaitingConfirm += 1;
    if (l.underwater) underwaterCount += 1;
    if (l.ageDays > oldestAgeDays) oldestAgeDays = l.ageDays;
  }
  const netMarginPct =
    base.totalRevenueSgd != null && base.totalRevenueSgd !== 0 && base.totalProfitSgd != null
      ? round1((base.totalProfitSgd / base.totalRevenueSgd) * 100)
      : null;
  return {
    openLines:            lines.length,
    pieces:               base.pieces,
    totalGrams:           base.totalGrams,
    totalPremiumSgd:      base.totalPremiumSgd,
    owedToSuppliersSgd:   round2(owed),
    awaitingConfirm,
    totalRevenueSgd:      base.totalRevenueSgd,
    totalAbsorbedFeesSgd: base.totalAbsorbedFeesSgd,
    netProfitSgd:         base.totalProfitSgd,
    netMarginPct,
    underwaterCount,
    oldestAgeDays,
  };
}
 
/** S$ with two decimals, e.g. "S$1,234.50". */
export function sgd(n: number): string {
  return `S$${n.toLocaleString("en-SG", { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`;
}
 
/** Grams, trimmed of trailing zeros: 5 → "5g", 2.5 → "2.5g", 31.1035 → "31.1035g". */
export function grams(n: number): string {
  return `${Number(n.toFixed(4))}g`;
}
 
/** Sale date as "10 Aug" in Singapore time (deterministic — fixed locale + TZ). */
export function saleDate(iso: string): string {
  const d = new Date(iso);
  if (Number.isNaN(d.getTime())) return "—";
  return d.toLocaleDateString("en-GB", { day: "2-digit", month: "short", timeZone: "Asia/Singapore" });
}
 
/**
 * The consignment SETTLEMENT ledger for a set of SELECTED lines — a statement
 * of what we owe the supplier, priced at OUR spot (which they honor) + their
 * premium, sent so they can settle.
 *
 * Greets `contactName` (the supplier's contact person). Each line shows the
 * date, grams, our spot, and the premium calc; the amount is the our-spot cost
 * (`spot × totalGrams + premium`). It deliberately omits our revenue and profit
 * — that is our margin and is never sent to a supplier.
 */
export function buildSupplierMessage(
  contactName: string,
  lines: ConsignmentLineView[],
): string {
  const who = contactName.trim() || "there";
  if (lines.length === 0) {
    return `Hi ${who}, (no lines selected).`;
  }
  const rows = lines.map((l, i) => {
    const totG = lineTotalGrams(l);
    return (
      `${i + 1}. ${saleDate(l.createdAt)} · ${grams(l.grams)} × ${l.quantity} = ${grams(totG)}`
      + ` · spot ${sgd(l.spotAtSaleSgd)}/g`
      + ` · premium ${sgd(linePremiumSgd(l))} (${premiumBreakdown(l)})`
      + ` · ${sgd(lineOurSpotCostSgd(l))}`
    );
  });
  const t = summariseLines(lines);
  const totalLine =
    `Total: ${t.pieces} ${t.pieces === 1 ? "pc" : "pc"} · ${grams(t.totalGrams)}`
    + ` · premium ${sgd(t.totalPremiumSgd)} · ${sgd(t.totalOurSpotCostSgd)}`;
  return (
    `Hi ${who}, consignment settlement:\n`
    + `${rows.join("\n")}\n\n`
    + `${totalLine}\nThank you.`
  );
}
 
function round1(n: number): number { return Math.round(n * 10) / 10; }
function round2(n: number): number { return Math.round(n * 100) / 100; }
function round4(n: number): number { return Math.round(n * 10000) / 10000; }
