/**
 * Dispatch lanes (B5 — design: docs/fulfilment-lanes-design.md §7).
 *
 * The Dispatch screen used to be organised by COURIER. Next-day evening orders
 * are booked through Lalamove, so every evening order appeared twice — on the
 * evening board AND in "Express (Lalamove) — Awaiting Booking" — and the
 * consolidate control let an operator put today's express parcels on the same
 * trip as tomorrow evening's. Organising by DELIVERY TYPE fixes both:
 *
 *   - a lane is what the customer bought (same-day express, next-day evening),
 *     read from `orders.delivery_option`, not from the courier;
 *   - a slot is one lane + one Singapore delivery date + one promised window;
 *   - one trip may only ever carry orders from ONE slot — enforced here, and on
 *     the server by `prepareBatch`, so the UI cannot be the only check.
 *
 * Pure, so the grouping the operator acts on and the server guard share one
 * definition and are unit-tested.
 */
 
export type DeliveryLane = "express" | "evening";
 
/**
 * The courier that books each lane. Lane membership on the Dispatch screen is
 * keyed on `delivery_option` + this map, never on a hardcoded courier test, so
 * moving a lane to another courier is a change here (one line) once that
 * courier has a booking integration.
 *
 * Deliberately a code constant, not a database setting: a settings switch that
 * can point a lane at a courier we cannot book through would be a control that
 * leads nowhere — the order would match no booking path. When a second courier
 * for a lane is integrated, promote this to `shipper_settings` in the same PR.
 */
export const LANE_COURIER: Record<DeliveryLane, string> = {
  express: "lalamove",
  evening: "lalamove",
};
 
export interface LaneOrder {
  id: string;
  shipper?: string | null;
  delivery_option?: string | null;
  delivery_window_start?: string | null;
  delivery_window_end?: string | null;
}
 
/** Singapore calendar date (YYYY-MM-DD) of an instant. */
export function sgDate(iso: string): string {
  return new Date(new Date(iso).getTime() + 8 * 3600_000).toISOString().slice(0, 10);
}
 
function validIso(s: string | null | undefined): s is string {
  return !!s && !Number.isNaN(new Date(s).getTime());
}
 
/** The delivery type the customer bought. Anything not evening is express. */
export function deliveryLane(o: Pick<LaneOrder, "delivery_option">): DeliveryLane {
  return o.delivery_option === "evening" ? "evening" : "express";
}
 
/** True when the order belongs to `lane` AND is booked by that lane's courier. */
export function inLane(o: LaneOrder, lane: DeliveryLane): boolean {
  return o.shipper === LANE_COURIER[lane] && deliveryLane(o) === lane;
}
 
/**
 * Stable slot identity: lane + SGT date + window start + window end. An order
 * with no (valid) window is an "as soon as possible" slot for its lane.
 */
export function slotKey(o: LaneOrder): string {
  const lane = deliveryLane(o);
  if (!validIso(o.delivery_window_start)) return `${lane}|asap`;
  const start = new Date(o.delivery_window_start).toISOString();
  const end = validIso(o.delivery_window_end) ? new Date(o.delivery_window_end).toISOString() : "";
  return `${lane}|${sgDate(start)}|${start}|${end}`;
}
 
export interface LaneSlot<T extends LaneOrder> {
  key: string;
  lane: DeliveryLane;
  /** SGT delivery date (YYYY-MM-DD); null for an ASAP slot. */
  date: string | null;
  windowStart: string | null;
  windowEnd: string | null;
  orders: T[];
  /** Packed + gate-passed — bookable now. */
  ready: T[];
  /** Not yet bookable (not packed, or gold still with the supplier). */
  waiting: T[];
}
 
/**
 * Group orders of ONE lane into slots. ASAP first (most urgent), then windowed
 * slots soonest first. Every order of the lane lands in exactly one slot, in its
 * original relative order; orders of other lanes are ignored.
 */
export function groupLaneSlots<T extends LaneOrder>(
  orders: readonly T[],
  lane: DeliveryLane,
  isReady: (o: T) => boolean,
): LaneSlot<T>[] {
  const slots = new Map<string, LaneSlot<T>>();
  for (const o of orders) {
    if (deliveryLane(o) !== lane) continue;
    const key = slotKey(o);
    let s = slots.get(key);
    if (!s) {
      const asap = key.endsWith("|asap");
      const start = asap ? null : new Date(o.delivery_window_start as string).toISOString();
      const end = !asap && validIso(o.delivery_window_end) ? new Date(o.delivery_window_end).toISOString() : null;
      s = { key, lane, date: start ? sgDate(start) : null, windowStart: start, windowEnd: end, orders: [], ready: [], waiting: [] };
      slots.set(key, s);
    }
    s.orders.push(o);
    (isReady(o) ? s.ready : s.waiting).push(o);
  }
  return [...slots.values()].sort((a, b) => {
    if (!a.windowStart) return -1;
    Iif (!b.windowStart) return 1;
    return a.windowStart.localeCompare(b.windowStart) || (a.windowEnd ?? "").localeCompare(b.windowEnd ?? "");
  });
}
 
/**
 * Server guard: a single trip must carry ONE slot. Returns a human message when
 * the selection spans more than one lane, date or window, else null.
 */
export function mixedSlotError(orders: readonly LaneOrder[]): string | null {
  const keys = new Set(orders.map(slotKey));
  if (keys.size <= 1) return null;
  const lanes = new Set(orders.map(deliveryLane));
  if (lanes.size > 1) {
    return "Can't put express and evening orders on one trip — book each delivery type separately";
  }
  return "Can't put orders from different delivery dates or windows on one trip — book each window separately";
}
