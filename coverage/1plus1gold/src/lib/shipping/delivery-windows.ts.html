/**
 * Customer-chosen Lalamove delivery windows — the ONE slot algorithm.
 * Design: docs/lalamove-delivery-window-design.md §4.
 *
 * Pure (no I/O). The same function feeds /api/shipping-options (web + iOS
 * pickers) and the server-side re-check at order creation, so a client can
 * never book a slot the server would not offer.
 *
 * All wall-clock maths is Asia/Singapore, which is UTC+8 with NO daylight
 * saving — so "SGT local → UTC" is a fixed −8h shift, done with Date.UTC
 * rather than a timezone library.
 *
 * Rules:
 *   • Today: a window is offered when its start is ≥ now + minLead, the order
 *     is placed before today's next_day_cutoff, today's schedule is enabled,
 *     and the window sits inside today's open/close hours.
 *   • Tomorrow: every enabled window that fits tomorrow's schedule.
 *   • Default: the first today-window starting inside [now+minLead, now+maxLead];
 *     otherwise the first tomorrow window.
 *   • ASAP ("As soon as possible, 1–2 h"): offered while the shipper is open now
 *     and before today's cutoff — today's instant product, kept above the windows.
 */

/** The SGT calendar date (y, m0, d), day-of-week and minutes-since-midnight of `now`. */
export function sgtParts(now: Date): { y: number; m0: number; d: number; dow: number; minutes: number } {
  const shifted = new Date(now.getTime() + SGT_OFFSET_MIN * 60_000);
  return {
    y: shifted.getUTCFullYear(),
    m0: shifted.getUTCMonth(),
    d: shifted.getUTCDate(),
    dow: shifted.getUTCDay(),
    minutes: shifted.getUTCHours() * 60 + shifted.getUTCMinutes(),
  };
}
 
/** UTC instant of an SGT wall-clock time on the SGT date `dayOffset` days from `now`. */
function sgtInstant(now: Date, dayOffset: number, minutes: number): Date {
  const p = sgtParts(now);
  return new Date(Date.UTC(p.y, p.m0, p.d + dayOffset, 0, minutes - SGT_OFFSET_MIN));
}

  return `${DOW[p.dow]} ${p.d} ${MON[p.m0]}`;
}

function windowsForDay(
  input: WindowInput,
  dayOffset: 0 | 1,
): OfferedWindow[] {
  const { now, windows, scheduleByDow, minLeadMinutes } = input;
  const today = sgtParts(now);
  const dow = (today.dow + dayOffset) % 7;
  const sched = scheduleByDow[dow];
  if (sched && sched.enabled === false) return [];
  const open = hhmmToMinutes(sched?.open_time) ?? 0;
  const close = hhmmToMinutes(sched?.close_time) ?? 24 * 60;
 
  if (dayOffset === 0) {
    const cutoff = hhmmToMinutes(sched?.next_day_cutoff);
    if (cutoff != null && today.minutes >= cutoff) return [];
  }
 
  const earliest = now.getTime() + minLeadMinutes * 60_000;
  return windows
    .filter((w) => w.enabled !== false && (w.day_of_week == null || w.day_of_week === dow))
    .map((w) => ({ w, s: hhmmToMinutes(w.start_time), e: hhmmToMinutes(w.end_time) }))
    .filter((x): x is { w: DeliveryWindowDef; s: number; e: number } => x.s != null && x.e != null && x.e > x.s)
    .filter((x) => x.s >= open && x.e <= close)
    .sort((a, b) => a.s - b.s || (a.w.sort_order ?? 0) - (b.w.sort_order ?? 0))
    .map((x) => {
      const start = sgtInstant(now, dayOffset, x.s);
      const end = sgtInstant(now, dayOffset, x.e);
      const label = `${hhmm(x.s)}–${hhmm(x.e)}`;
      const dateLabel = sgtDateLabel(start);
      return {
        start: start.toISOString(),
        end: end.toISOString(),
        day: dayOffset === 0 ? ("today" as const) : ("tomorrow" as const),
        label,
        dateLabel,
        fullLabel: `${dateLabel}, ${label}`,
      };
    })
    .filter((o) => dayOffset === 1 || new Date(o.start).getTime() >= earliest)
    // de-dupe identical slots (an every-day row + a same-time weekday row)
    .filter((o, i, arr) => arr.findIndex((x) => x.start === o.start && x.end === o.end) === i);
}
 
export function availableWindows(input: WindowInput): WindowAvailability {
  const { now, scheduleByDow, minLeadMinutes, maxLeadMinutes } = input;

  const tomorrow = windowsForDay(input, 1);

  const hi = now.getTime() + Math.max(minLeadMinutes, maxLeadMinutes) * 60_000;

    today.find((w) => { const t = new Date(w.start).getTime(); return t >= lo && t <= hi; }) ??
    tomorrow[0] ?? null;
 
  const p = sgtParts(now);
  const sched = scheduleByDow[p.dow];
  const open = hhmmToMinutes(sched?.open_time) ?? 0;
  const close = hhmmToMinutes(sched?.close_time) ?? 24 * 60;
  const cutoff = hhmmToMinutes(sched?.next_day_cutoff);
  const asapAvailable =
    sched?.enabled !== false && p.minutes >= open && p.minutes < close && (cutoff == null || p.minutes < cutoff);

}
 
/**
 * Server-side re-check of a window the client submitted. `graceMinutes` keeps
 * a customer who spent a few minutes on the payment page from being refused a
 * slot that was valid when they picked it.
 */
export function isWindowOffered(
  input: WindowInput,
  startIso: string,
  endIso: string,
  graceMinutes = 15,
): boolean {
  const relaxed = { ...input, minLeadMinutes: Math.max(0, input.minLeadMinutes - graceMinutes) };
  const { today, tomorrow } = availableWindows(relaxed);
  const s = new Date(startIso).getTime();
  const e = new Date(endIso).getTime();
  if (!Number.isFinite(s) || !Number.isFinite(e)) return false;
  return [...today, ...tomorrow].some((w) => new Date(w.start).getTime() === s && new Date(w.end).getTime() === e);
}

/**
 * When to schedule a consolidated multi-stop trip. One trip has one scheduleAt,
 * so it is the EARLIEST window start — and only when every stop has a window.
 * A trip that includes an ASAP order goes now (scheduling would delay it).
 */
export function batchScheduleStart(stops: Array<{ windowStart: string | null }>): string | null {
  if (stops.length === 0 || stops.some((d) => !d.windowStart)) return null;
  return stops.map((d) => d.windowStart as string).sort()[0];
}
