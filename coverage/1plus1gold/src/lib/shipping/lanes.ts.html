/**
 * Checkout delivery LANES — the one promise engine (Mukesh 2026-11-21,
 * docs/singpost-integration-plan.md → "Delivery lanes — design"; port diff:
 * SDUI contract §253).
 *
 * A lane is ONE delivery promise shown at checkout ("Tomorrow 3–6 pm · Free"),
 * fully described by its settings row in Admin → Shippers → Delivery lanes:
 *
 *   • Day rule. The DISPATCH day is today when today is a dispatch day, not a
 *     holiday (if the lane observes them) and now is before today's
 *     `ordersUntil` (null = all day); else the next dispatch day. The DELIVERY
 *     day is the first delivery day (not a holiday) on or after dispatch +
 *     `daysAfter`. A `todayOnly` lane is offered only when that date is today.
 *   • Window per delivery weekday: start–end ("3–6 pm"), end only ("by 6 pm"),
 *     or none (that day, no time promised).
 *   • Price: the carrier cost, then a discount or premium % (netShipperFee —
 *     the same model every carrier row uses); minimum and maximum order.
 *   • Default carrier + service — what Dispatch books (Dispatch can still switch
 *     one order). The customer's promise never depends on the carrier.
 *
 * Pure (no I/O) and constant-free: every time, day and price is an input. All
 * wall-clock maths is Asia/Singapore (UTC+8, no DST). Each lane yields AT MOST
 * ONE offer; the picker lists them earliest delivery first, nothing pre-selected.
 */
import { clockLabel, hhmmToMinutes, relativeDayPrefix, sgtDateLabel, sgtParts, timeRangeLabel } from "@/lib/shipping/delivery-windows";
import { meetsMinOrder } from "@/lib/shipping/min-order";
import { BRAND_ICONS, courierIconKey } from "@/lib/admin/brand-icons";
 
const SGT_OFFSET_MIN = 8 * 60;
/** How far ahead a dispatch or delivery day is searched (two weeks covers any sane calendar). */
const LOOKAHEAD_DAYS = 14;
 
/** Carriers a lane can be booked with (Dispatch). */
export const LANE_CARRIERS = ["lalamove", "grab", "spx", "singpost"] as const;
export type LaneCarrier = (typeof LANE_CARRIERS)[number];
 
/** SingPost services a lane can book (singpost carrier only). */
export const SINGPOST_LANE_SERVICES = ["PRIORITY_SAMEDAY", "PRIORITY_NEXTDAY", "PRIME", "STANDARD", "SAVER"] as const;
export type SingpostLaneService = (typeof SINGPOST_LANE_SERVICES)[number];

export const LANE_DISPATCH_KINDS = ["afternoon", "evening"] as const;
export type LaneDispatchKind = (typeof LANE_DISPATCH_KINDS)[number];
 
export interface LaneDay {

  dispatch: boolean;
  /** Pay before this time (HH:MM) for THIS day to be the dispatch day. null = all day. */
  ordersUntil: string | null;
  /** The customer can receive on this weekday. */
  delivers: boolean;
  /** Window on this delivery weekday. Both null = no time promised; start null + end = "by <end>". */
  windowStart: string | null;
  windowEnd: string | null;
}

function sgtInstant(now: Date, dayOffset: number, minutes: number): Date {
  const p = sgtParts(now);
  return new Date(Date.UTC(p.y, p.m0, p.d + dayOffset, 0, minutes - SGT_OFFSET_MIN));
}
 
/** SGT calendar date (YYYY-MM-DD) `dayOffset` days from `now`. */
export function sgtDateIso(now: Date, dayOffset: number): string {
  return new Date(sgtInstant(now, dayOffset, 12 * 60).getTime() + SGT_OFFSET_MIN * 60_000).toISOString().slice(0, 10);
}
 
function dayTitle(dayOffset: number, dateLabel: string): string {
  if (dayOffset === 0) return `Today · ${dateLabel}`;
  if (dayOffset === 1) return `Tomorrow · ${dateLabel}`;
  return dateLabel;
}
 
/** The lane's dispatch and delivery day offsets for an order placed at `now`; null = no day within the lookahead. */
export function laneDates(cfg: LaneConfig, now: Date, holidays: ReadonlySet<string>): { dispatch: number; delivery: number } | null {
  const today = sgtParts(now);
  const dow = (offset: number) => (today.dow + offset) % 7;
  const closed = (offset: number) => cfg.observesHolidays && holidays.has(sgtDateIso(now, offset));
  const isDispatch = (offset: number) => cfg.days[dow(offset)]?.dispatch === true && !closed(offset);
 
  let dispatch: number | null = null;
  const until = hhmmToMinutes(cfg.days[today.dow]?.ordersUntil);
  if (isDispatch(0) && (until == null || today.minutes < until)) dispatch = 0;
  for (let o = 1; dispatch == null && o <= LOOKAHEAD_DAYS; o++) if (isDispatch(o)) dispatch = o;
  if (dispatch == null) return null;
 
  const from = dispatch + Math.max(0, cfg.daysAfter);
  for (let o = from; o <= from + LOOKAHEAD_DAYS; o++) {
    if (cfg.days[dow(o)]?.delivers === true && !closed(o)) return { dispatch, delivery: o };
  }
  return null;
}
 
/** The lane's one offer for an order placed at `now`, or null (off, past cut-off, window started, no day). */
export function laneOffer(cfg: LaneConfig, now: Date, holidays: ReadonlySet<string> = new Set()): LaneOffer | null {
  if (!cfg.enabled) return null;
  const d = laneDates(cfg, now, holidays);
  if (!d) return null;
  if (cfg.todayOnly && d.delivery !== 0) return null;
 
  const day = cfg.days[(sgtParts(now).dow + d.delivery) % 7];
  const s = hhmmToMinutes(day?.windowStart);
  const e = hhmmToMinutes(day?.windowEnd);
  const hasWindow = s != null && e != null && e > s;
  const start = hasWindow ? sgtInstant(now, d.delivery, s) : null;
  const end = e != null ? sgtInstant(now, d.delivery, e) : null;
  // A promise for today whose window has already started (or whose "by" time has passed) is not offered.
  const startsAt = start ?? end;
  if (d.delivery === 0 && startsAt && startsAt.getTime() <= now.getTime()) return null;
 
  const timeLabel = hasWindow ? timeRangeLabel(s, e) : e != null ? `by ${clockLabel(e)}` : null;
  const dateLabel = sgtDateLabel(sgtInstant(now, d.delivery, 12 * 60));
  const date = sgtDateIso(now, d.delivery);
  return {
    id: `${cfg.key}:${date}`,
    lane: cfg.key,
    date,
    dayOffset: d.delivery,
    isToday: d.delivery === 0,
    start: start?.toISOString() ?? null,
    end: end?.toISOString() ?? null,
    timeLabel,
    dateLabel,
    dayTitle: dayTitle(d.delivery, dateLabel),
    fullLabel: `${relativeDayPrefix(d.delivery)}${dateLabel}${timeLabel ? `, ${timeLabel}` : ""}`,
    sortAt: (end ?? sgtInstant(now, d.delivery, 24 * 60)).toISOString(),
    displayName: cfg.displayName,
    notes: cfg.notes,
    fee: cfg.fee.fee,
    grossFee: cfg.fee.grossFee,
    discountPct: cfg.fee.discountPct,
    premiumPct: cfg.fee.premiumPct,
    minOrderSgd: cfg.minOrderSgd,
    maxOrderSgd: cfg.maxOrderSgd,
    carrierLogo: cfg.showCarrierLogo ? carrierLogo(cfg.carrier) : null,
  };
}
 
/** The carrier's logo (public/images/brands, the same marks Admin and Dispatch show). */
export function carrierLogo(carrier: LaneCarrier): { src: string; alt: string } | null {
  const key = courierIconKey(carrier);
  return key ? { src: BRAND_ICONS[key].src, alt: BRAND_ICONS[key].alt } : null;
}
 
/**
 * Every lane's offer, grouped by day (Mukesh 2026-11-22, SDUI §255 P1): the delivery
 * DAY first, then the time it is promised for — a window's start, a "by" lane's time,
 * or (no time at all: SPX, Express) the end of its day — then the lane order. Checkout draws a date header over
 * each day's group, so this order is what makes "Today · afternoon, evening;
 * Tomorrow · afternoon, evening; Tue · SPX" read top to bottom. Installed apps render
 * the server's order, so they get it without a build.
 */
export function laneOffers(cfgs: readonly LaneConfig[], now: Date, holidays: ReadonlySet<string> = new Set()): LaneOffer[] {
  const startKey = (o: LaneOffer) => Date.parse(o.start ?? o.sortAt);
  return [...cfgs]
    .sort((a, b) => a.sortOrder - b.sortOrder)
    .map((c) => laneOffer(c, now, holidays))
    .filter((o): o is LaneOffer => o !== null)
    .sort((a, b) => a.dayOffset - b.dayOffset || startKey(a) - startKey(b));
}
 
/** The order value qualifies for the offer (configured minimum and maximum). */
export function offerFitsOrder(o: Pick<LaneOffer, "minOrderSgd" | "maxOrderSgd">, orderValueSgd: number): boolean {
  return meetsMinOrder(o.minOrderSgd, orderValueSgd) && (o.maxOrderSgd == null || orderValueSgd <= o.maxOrderSgd);
}
 
/** The line shown instead of the offers an order is below the minimum for. */
export function laneMinOrderNote(minOrderSgd: number): string {
  const amount = Number.isInteger(minOrderSgd) ? String(minOrderSgd) : minOrderSgd.toFixed(2);
  return `More delivery options are available on orders from S$${amount}.`;
}
 
/** What the picker lists for this order value, and one note for what the minimum hides. */
export function visibleLaneOffers(offers: readonly LaneOffer[], orderValueSgd: number): { shown: LaneOffer[]; note: string | null } {
  const shown = offers.filter((o) => offerFitsOrder(o, orderValueSgd));
  const hiddenMins = offers
    .filter((o) => !meetsMinOrder(o.minOrderSgd, orderValueSgd))
    .map((o) => o.minOrderSgd)
    .filter((m): m is number => m != null);
  return { shown, note: hiddenMins.length > 0 ? laneMinOrderNote(Math.min(...hiddenMins)) : null };
}
 
/**
 * Server re-check of a submitted lane offer: the lane's offer now, or
 * `graceMinutes` ago (picked at 10:58, paid at 11:03 keeps the promise they
 * saw), must be for `date`. When both match, the CHEAPER fee wins. null = not
 * promised.
 */
export function findPromisedLaneOffer(
  cfg: LaneConfig,
  now: Date,
  date: string,
  holidays: ReadonlySet<string> = new Set(),
  graceMinutes = 15,
): LaneOffer | null {
  const earlier = new Date(now.getTime() - graceMinutes * 60_000);
  const matches = [laneOffer(cfg, now, holidays), laneOffer(cfg, earlier, holidays)]
    .filter((o): o is LaneOffer => o !== null && o.date === date);
  if (matches.length === 0) return null;
  return matches.reduce((a, b) => (b.fee < a.fee ? b : a));
}
 
/** Lane key format: lower-case letters, digits, underscore (it is stored on orders). */
export function isLaneKey(v: unknown): v is string {
  return typeof v === "string" && /^[a-z][a-z0-9_]{1,39}$/.test(v);
}
 
/**
 * A lane's delivery days + windows in the shape Admin → "Change delivery date"
 * reads (lib/shipping/reschedule.ts `rescheduleWindows`): one window row per
 * delivery weekday that has a start AND end, and those weekdays switched on.
 * Pure. Holidays are filtered by the caller (`notOnHoliday`).
 */
export function laneRescheduleInput(cfg: Pick<LaneConfig, "days">): {
  windows: Array<{ day_of_week: number; start_time: string; end_time: string; enabled: true }>;
  scheduleByDow: Partial<Record<number, { enabled: boolean; open_time: string; close_time: string; next_day_cutoff: null }>>;
} {
  const windows: Array<{ day_of_week: number; start_time: string; end_time: string; enabled: true }> = [];
  // open/close span the whole day: a lane's windows are delivery times, not hours it is sold.
  const scheduleByDow: Partial<Record<number, { enabled: boolean; open_time: string; close_time: string; next_day_cutoff: null }>> = {};
  for (let dow = 0; dow < 7; dow++) {
    const d = cfg.days[dow];
    scheduleByDow[dow] = { enabled: d?.delivers === true, open_time: "00:00", close_time: "23:59", next_day_cutoff: null };
    if (d?.delivers && d.windowStart && d.windowEnd) {
      windows.push({ day_of_week: dow, start_time: d.windowStart, end_time: d.windowEnd, enabled: true });
    }
  }
  return { windows, scheduleByDow };
}
 
/** A slot (ISO start) falls on a holiday the lane observes. */
export function onLaneHoliday(cfg: Pick<LaneConfig, "observesHolidays">, startIso: string, holidays: ReadonlySet<string>): boolean {
  if (!cfg.observesHolidays) return false;
  return holidays.has(new Date(Date.parse(startIso) + SGT_OFFSET_MIN * 60_000).toISOString().slice(0, 10));
}
