/**
 * Checkout order of the delivery options (Mukesh, 2026-10-08).
 *
 * The picker used to list the carriers in a fixed order with Lalamove first and
 * pre-select Lalamove. The order is now a setting
 * (`shipper_settings.checkout_sort_order`): the lowest number is listed first,
 * and the first option that is AVAILABLE today is pre-selected.
 *
 * One definition, read by `/api/shipping-options` (which ships the order to
 * every client) and by the web picker (which renders and defaults from it), so
 * the list and the default can never disagree.
 */
 
/** Customer-facing carriers that have a settings row and therefore a position. */
export const CHECKOUT_CARRIERS = ["lalamove", "ninjavan", "evening", "spx"] as const;
export type CheckoutCarrier = (typeof CHECKOUT_CARRIERS)[number];
 
/** Every option the picker can list. Self-pickup is a per-customer privilege
 *  with no settings row, so it has no position and always comes last. */
export type CheckoutOption = CheckoutCarrier | "self_pickup";
 
export interface CarrierPositionRow {
  shipper: string;
  checkout_sort_order?: number | null;
}
 
/**
 * The carriers in checkout order, followed by `self_pickup`.
 *
 * - Lower `checkout_sort_order` first; a missing/invalid position sorts last.
 * - Ties (and missing positions) fall back to the built-in order above, so the
 *   result is deterministic and matches the old fixed order when unset.
 * - Every carrier is ALWAYS present, whether or not it has a row: availability
 *   is decided by the client from the options payload, never by omission here.
 */
export function checkoutOrder(rows: readonly CarrierPositionRow[] | null | undefined): CheckoutOption[] {
  const pos = new Map<string, number>();
  for (const r of rows ?? []) {
    const n = r.checkout_sort_order;
    if (typeof n === "number" && Number.isFinite(n)) pos.set(r.shipper, n);
  }
  const ordered = [...CHECKOUT_CARRIERS].sort((a, b) => {
    const pa = pos.get(a) ?? Number.POSITIVE_INFINITY;
    const pb = pos.get(b) ?? Number.POSITIVE_INFINITY;
    if (pa !== pb) return pa - pb;
    return CHECKOUT_CARRIERS.indexOf(a) - CHECKOUT_CARRIERS.indexOf(b);
  });
  return [...ordered, "self_pickup"];
}
 
/**
 * Normalise an order received from the API: keep known options, drop
 * duplicates, and append anything missing in the built-in order. A client must
 * never lose an option because the server sent a short or stale list.
 */
export function normaliseCheckoutOrder(order: readonly unknown[] | null | undefined): CheckoutOption[] {
  const known: readonly CheckoutOption[] = [...CHECKOUT_CARRIERS, "self_pickup"];
  const out: CheckoutOption[] = [];
  for (const k of order ?? []) {
    if (typeof k === "string" && (known as readonly string[]).includes(k) && !out.includes(k as CheckoutOption)) {
      out.push(k as CheckoutOption);
    }
  }
  for (const k of CHECKOUT_CARRIERS) if (!out.includes(k)) out.push(k);
  // Self-pickup is pinned last regardless of where the list put it.
  return [...out.filter((k) => k !== "self_pickup"), "self_pickup"];
}
 
// No default option: checkout never pre-selects a shipper (Mukesh 2026-10-09,
// #483 B3). The order above only decides the sequence the options are shown in.
