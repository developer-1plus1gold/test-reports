/**
 * Grab Express — a MANUAL courier (2026-11-06, Mukesh: "move to Grab Express and
 * I will book those manually and add the tracking url").
 *
 * The admin books the trip in the Grab app, then:
 *   0. unmove — the reverse of move (back to Lalamove), only while no Grab trip
 *               link, cost or live booking is saved (2026-11-13);
 *   1. move   — a Lalamove order that is NOT yet with a Lalamove driver becomes
 *               shipper='grab' (shipper_status 'pending', no trip id);
 *   2. track  — pastes the Grab share link (shipper_share_link) and, optionally,
 *               what the trip cost (shipper_actual_fee_sgd) → status 'booked';
 *   3. deliver — marks it delivered (order completed, bars sold), like mark-pickup.
 *
 * Pure rules, shared by the route and the Dispatch screen so they cannot drift.
 * The customer's shipping fee is never touched: a Grab trip that costs more is
 * absorbed, exactly as a Lalamove upgrade is.
 *
 * ▶ 2026-11-07: the GrabExpress API (lib/grab-express.ts) can book the same
 * order instead of step 2, once its credentials are set. An API trip carries
 * Grab's deliveryID in shipper_order_id; a hand-booked one never has one. The
 * manual steps stay as the fallback for both.
 */
 
export const GRAB_SHIPPER = "grab";
export const GRAB_SERVICE = "EXPRESS";

  shipper?:          string | null;
  shipper_status?:   string | null;
  shipper_order_id?: string | null;
  status?:           string | null;

const DEAD_ORDER = new Set(["cancelled", "refunded"]);
 
/** Why a Lalamove order cannot move to Grab, or null when it can. A booked
 *  Lalamove trip must be cancelled first (Cancel booking), so the parcel is
 *  never with two couriers. A failed trip (no driver) can move as-is. */
export function moveToGrabBlocker(o: GrabOrderState): string | null {
  if (DEAD_ORDER.has(o.status ?? "")) return `This order is ${o.status}`;
  if (o.shipper !== "lalamove") return `Only Lalamove orders can be moved to Grab Express (this one is "${o.shipper ?? "unset"}")`;
  const st = o.shipper_status ?? "";
  if (st === "delivered") return "This order is already delivered";
  if (o.shipper_order_id && st !== "failed") {
    return "This order has a live Lalamove booking — cancel it first (Cancel booking), then move it to Grab Express";
  }
  return null;
}

/** Why a Grab order cannot go back to Lalamove, or null when it can — the mirror
 *  of `moveToGrabBlocker` (Mukesh, 2026-11-12). Only while nothing about a Grab
 *  trip is saved: a share link or a trip cost means a driver was (or is being)
 *  booked, and a live API booking must be cancelled first. A failed API trip
 *  had no driver, so it can go back as-is. */
export function moveBackToLalamoveBlocker(o: MoveBackState): string | null {
  if (DEAD_ORDER.has(o.status ?? "")) return `This order is ${o.status}`;
  if (o.shipper !== GRAB_SHIPPER) return "Only Grab Express orders can be moved back to Lalamove";
  const st = o.shipper_status ?? "";
  if (st === "delivered") return "This order is already delivered";
  if (o.shipper_order_id && st !== "failed") {
    return "This order has a live Grab booking — cancel it first (Cancel booking), then move it back to Lalamove";
  }
  if (o.shipper_share_link) return "A Grab trip link is saved — the trip was booked in the Grab app, so it cannot go back to Lalamove";
  if (o.shipper_actual_fee_sgd != null && o.shipper_actual_fee_sgd !== "") {
    return "A Grab trip cost is saved — the trip was booked in the Grab app, so it cannot go back to Lalamove";
  }
  return null;
}

/** Why a Grab order cannot be updated (tracking / delivered), or null. */
export function grabUpdateBlocker(o: GrabOrderState): string | null {
  if (DEAD_ORDER.has(o.status ?? "")) return `This order is ${o.status}`;
  if (o.shipper !== GRAB_SHIPPER) return "This is not a Grab Express order";
  if ((o.shipper_status ?? "") === "delivered") return "Already marked delivered";
  return null;
}
 
/** A Grab share link: https on grab.com or a grab.com subdomain. Returns the
 *  normalised URL, or null. Anything else is refused — the link is shown to the
 *  customer as their "Track" button, so it must not point anywhere else. */
export function normaliseGrabTrackingUrl(raw: unknown): string | null {
  if (typeof raw !== "string") return null;
  const s = raw.trim();
  if (!s || s.length > 1000) return null;
  let u: URL;
  try { u = new URL(s); } catch { return null; }
  if (u.protocol !== "https:") return null;
  const host = u.hostname.toLowerCase();
  if (host !== "grab.com" && !host.endsWith(".grab.com")) return null;
  return u.toString();
}
 
/** Optional trip cost in SGD: a positive number under S$500, 2 dp; null when
 *  absent; undefined when present but invalid. */
export function parseGrabFee(raw: unknown): number | null | undefined {
  if (raw === undefined || raw === null || raw === "") return null;
  const n = typeof raw === "number" ? raw : Number(String(raw).trim());
  if (!Number.isFinite(n) || n <= 0 || n >= 500) return undefined;
  return Math.round(n * 100) / 100;
}
 
/** Why the GrabExpress API cannot book this order, or null when it can. Only
 *  a Grab order with no live trip: never booked, or its last API trip failed.
 *  A hand-booked trip (it has a share link) is never re-booked by the API —
 *  that would send two drivers. */
export function grabApiBookBlocker(o: GrabOrderState & { shipper_share_link?: string | null }): string | null {
  const base = grabUpdateBlocker(o);
  if (base) return base;
  const failed = (o.shipper_status ?? "") === "failed";
  if (o.shipper_order_id && !failed) return "This order already has a live Grab booking";
  if (o.shipper_share_link && !failed) return "This order was booked by hand in the Grab app (it has a tracking link)";
  return null;
}

/** Dispatch buckets for Grab orders. "Awaiting" = nothing booked yet, or the
 *  last trip failed (it needs re-booking — before 2026-11-07 a failed trip fell
 *  out of both buckets). "On the way" = a live trip, booked by API or by hand. */
export function isGrabAwaitingTracking(o: GrabTile): boolean {
  if (!live(o)) return false;
  if ((o.shipper_status ?? "") === "failed") return true;
  return !o.shipper_share_link && !o.shipper_order_id;
}
export function isGrabOnTheWay(o: GrabTile): boolean {
  if (!live(o) || (o.shipper_status ?? "") === "failed") return false;
  return !!o.shipper_share_link || !!o.shipper_order_id;
}
