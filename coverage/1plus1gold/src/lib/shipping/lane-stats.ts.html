/**
 * Delivery-option uptake + evening cost per stop (design:
 * docs/fulfilment-lanes-design.md §5 piece 6).
 *
 * The question this answers: is FREE next-day evening delivery sustainable?
 * Two numbers decide it — how many customers move to it (and from which lane),
 * and what each evening stop actually costs us, since we absorb all of it.
 *
 * Pure: the route fetches orders, this groups them. Weeks are Monday-start in
 * Singapore time, keyed by the Monday's SGT date.
 */
 
export const LANES = ["express", "evening", "spx", "ninjavan", "self_pickup"] as const;
export type Lane = (typeof LANES)[number];
 
export interface LaneOrder {
  created_at:             string;
  shipper:                string | null;
  delivery_option:        string | null;
  shipper_actual_fee_sgd: number | string | null;
}
 
/** The customer-facing delivery option an order was placed on, or null. */
export function laneOf(o: Pick<LaneOrder, "shipper" | "delivery_option">): Lane | null {
  switch (o.shipper) {
    case "lalamove":    return o.delivery_option === "evening" ? "evening" : "express";
    case "spx":         return "spx";
    case "ninjavan":    return "ninjavan";
    case "self_pickup": return "self_pickup";
    // vault, gold_yield, marketplace carriers: not a delivery choice at checkout.
    default:            return null;
  }
}
 
/** Monday (SGT) of the week an instant falls in, as YYYY-MM-DD. */
export function sgWeekStart(iso: string): string {
  const sg = new Date(new Date(iso).getTime() + 8 * 3600_000);
  const dow = (sg.getUTCDay() + 6) % 7; // Mon = 0
  sg.setUTCDate(sg.getUTCDate() - dow);
  return sg.toISOString().slice(0, 10);
}
 
export interface WeekLaneStats {
  week:     string;                 // Monday, YYYY-MM-DD (SGT)
  counts:   Record<Lane, number>;
  total:    number;                 // orders on any customer delivery option
  /** Evening stops whose trip cost is recorded (booked). */
  eveningCostedStops: number;
  eveningCostSgd:     number;
  /** Mean cost per evening stop; null until one is booked. */
  eveningCostPerStopSgd: number | null;
}
 
const emptyCounts = (): Record<Lane, number> =>
  Object.fromEntries(LANES.map((l) => [l, 0])) as Record<Lane, number>;
 
/**
 * One row per week, newest first. Weeks with no orders are omitted. An evening
 * order counts toward cost only once its trip is booked (the batch book splits
 * the trip fee across its stops into `shipper_actual_fee_sgd`); unbooked ones
 * count for uptake only, so the average is never diluted by zeros.
 */
export function weeklyLaneStats(orders: readonly LaneOrder[]): WeekLaneStats[] {
  const byWeek = new Map<string, WeekLaneStats>();
  for (const o of orders) {
    const lane = laneOf(o);
    if (!lane || !o.created_at || !Number.isFinite(new Date(o.created_at).getTime())) continue;
    const week = sgWeekStart(o.created_at);
    let w = byWeek.get(week);
    if (!w) {
      w = { week, counts: emptyCounts(), total: 0, eveningCostedStops: 0, eveningCostSgd: 0, eveningCostPerStopSgd: null };
      byWeek.set(week, w);
    }
    w.counts[lane]++;
    w.total++;
    if (lane === "evening" && o.shipper_actual_fee_sgd != null) {
      const fee = Number(o.shipper_actual_fee_sgd);
      Eif (Number.isFinite(fee) && fee >= 0) {
        w.eveningCostedStops++;
        w.eveningCostSgd += fee;
      }
    }
  }
  const rows = [...byWeek.values()].sort((a, b) => (a.week < b.week ? 1 : -1));
  for (const w of rows) {
    w.eveningCostSgd = Math.round(w.eveningCostSgd * 100) / 100;
    w.eveningCostPerStopSgd = w.eveningCostedStops > 0
      ? Math.round((w.eveningCostSgd / w.eveningCostedStops) * 100) / 100
      : null;
  }
  return rows;
}
