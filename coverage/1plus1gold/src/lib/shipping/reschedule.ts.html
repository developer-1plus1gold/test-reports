/**
 * Admin "Change delivery date" (2026-11-11, Mukesh: a customer asked to postpone
 * order C7EC1AED from 12 to 14 Nov, which had to be done by hand in SQL).
 *
 * Pure (no I/O). The route (`/api/admin/orders/[id]/delivery-date`) loads the
 * shipper's window rows and schedule with `loadWindowInput`, then uses:
 *   • `rescheduleBlocker(order)` — why this order's date cannot be changed (or null);
 *   • `rescheduleLane(order)`    — whose windows apply: the evening product's or Lalamove's;
 *   • `rescheduleWindows(...)`   — the slots the admin can pick, over the next N days.
 *
 * Only the WINDOW changes. It never books, cancels or re-prices: a booked trip
 * must be cancelled first (Dispatch → Cancel booking), the same rule as Move to
 * ASAP. Dispatch, the unbooked-window alert and "Book with Lalamove" all read
 * `delivery_window_start`, so moving the window moves all of them.
 *
 * Which slots are offered: every enabled window on a day the shipper's schedule
 * does not switch off, starting at least `minLeadMinutes` from now. Unlike the
 * checkout rules (delivery-windows.ts / evening.ts), there is no order cutoff
 * and no "today/tomorrow only" limit: the admin is moving an order that is
 * already paid for, usually to a later day the customer asked for.
 * Lalamove windows must also sit inside that day's open/close hours. For the
 * evening product, open/close are the hours the OPTION is sold at checkout, not
 * delivery hours (evening.ts), so they are not applied.
 *
 * All wall-clock maths is Asia/Singapore (UTC+8, no DST).
 */
import {
  deliveryWindowLabel, hhmmToMinutes, sgtDateLabel, sgtParts, timeRangeLabel,
  type DeliveryWindowDef, type WindowInput,
} from "@/lib/shipping/delivery-windows";
 
const SGT_OFFSET_MIN = 8 * 60;
export const RESCHEDULE_DAYS = 14;
/** A slot must start at least this far ahead, so there is time to pack and book. */
export const RESCHEDULE_MIN_LEAD_MINUTES = 60;
 
export type RescheduleLane = "evening" | "lalamove";
 
export interface ReschedulableOrder {
  status?: string | null;
  shipper?: string | null;
  shipper_status?: string | null;
  shipper_order_id?: string | null;
  shipper_batch_id?: string | null;
  delivery_option?: string | null;
}
 
export interface RescheduleSlot {
  start: string;     // ISO UTC
  end: string;       // ISO UTC
  dateLabel: string; // "Fri 14 Nov"
  label: string;     // "7:30–9 pm"
  fullLabel: string; // "Fri 14 Nov, 7:30–9 pm"
}
 
/** Whose windows apply. Evening is its own product with its own window rows. */
export function rescheduleLane(o: ReschedulableOrder): RescheduleLane | null {
  if (o.delivery_option === "evening") return "evening";
  if (o.shipper === "lalamove") return "lalamove";
  return null;
}
 
/** Why the date cannot be changed, as the admin should read it, or null when it can. */
export function rescheduleBlocker(o: ReschedulableOrder): string | null {
  if (o.status !== "paid") return "Only a paid order that is not yet delivered can be moved";
  if (!rescheduleLane(o)) return "Only Lalamove and next-day evening deliveries have a delivery date to change";
  // Same rule as Move to ASAP: a live booking (or a consolidated trip) carries its time at
  // Lalamove, so moving only our copy would leave the two disagreeing. A failed attempt can move.
  const liveBooking = !!o.shipper_order_id && o.shipper_status !== "failed";
  if (liveBooking || o.shipper_batch_id) {
    return "The trip is already booked: cancel the booking first, then change the date";
  }
  return null;
}
 
function sgtInstant(now: Date, dayOffset: number, minutes: number): Date {
  const p = sgtParts(now);
  return new Date(Date.UTC(p.y, p.m0, p.d + dayOffset, 0, minutes - SGT_OFFSET_MIN));
}
 
/** The slots an admin can move an order to, earliest first. */
export function rescheduleWindows(
  input: Pick<WindowInput, "now" | "windows" | "scheduleByDow">,
  lane: RescheduleLane,
  days: number = RESCHEDULE_DAYS,
  minLeadMinutes: number = RESCHEDULE_MIN_LEAD_MINUTES,
): RescheduleSlot[] {
  const earliest = input.now.getTime() + minLeadMinutes * 60_000;
  const today = sgtParts(input.now);
  const out: RescheduleSlot[] = [];
  for (let offset = 0; offset < days; offset++) {
    const dow = (today.dow + offset) % 7;
    const sched = input.scheduleByDow[dow];
    if (sched && sched.enabled === false) continue;
    const open = lane === "lalamove" ? (hhmmToMinutes(sched?.open_time) ?? 0) : 0;
    const close = lane === "lalamove" ? (hhmmToMinutes(sched?.close_time) ?? 24 * 60) : 24 * 60;
    const day = input.windows
      .filter((w: DeliveryWindowDef) => w.enabled !== false && (w.day_of_week == null || w.day_of_week === dow))
      .map((w) => ({ w, s: hhmmToMinutes(w.start_time), e: hhmmToMinutes(w.end_time) }))
      .filter((x): x is { w: DeliveryWindowDef; s: number; e: number } => x.s != null && x.e != null && x.e > x.s)
      .filter((x) => x.s >= open && x.e <= close)
      .sort((a, b) => a.s - b.s || (a.w.sort_order ?? 0) - (b.w.sort_order ?? 0))
      .map((x) => {
        const start = sgtInstant(input.now, offset, x.s);
        const end = sgtInstant(input.now, offset, x.e);
        const dateLabel = sgtDateLabel(start);
        const label = timeRangeLabel(x.s, x.e);
        return { start: start.toISOString(), end: end.toISOString(), dateLabel, label, fullLabel: `${dateLabel}, ${label}` };
      })
      .filter((s) => new Date(s.start).getTime() >= earliest);
    for (const s of day) {
      if (!out.some((x) => x.start === s.start && x.end === s.end)) out.push(s);
    }
  }
  return out;
}

export const NOTE_MAX = 200;
 
/**
 * The customer's message, sent through the approved order-status template
 * ("Your order {ref} is now {status}." + one free line). The time comes from the
 * stored window, never from a client label.
 */
export function rescheduleMessage(slot: Pick<RescheduleSlot, "start" | "end">, note?: string | null): { statusLabel: string; reason: string } {
  const when = deliveryWindowLabel(slot.start, slot.end) ?? "";
  const extra = (note ?? "").replace(/\s+/g, " ").trim().slice(0, NOTE_MAX);
  return {
    statusLabel: "Rescheduled",
    reason: `Your new delivery time is ${when}.${extra ? ` ${extra}` : ""}`,
  };
}
