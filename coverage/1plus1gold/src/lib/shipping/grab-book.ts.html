/**
 * Book a Grab order through the GrabExpress API (2026-11-07).
 *
 * The API counterpart of the manual "track" step in grab-manual.ts, with the
 * same gates as a Lalamove booking (book-delivery): packed, no bar still
 * awaiting gold, a complete address. Called by POST /api/admin/orders/[id]/grab
 * { action: "book" }.
 *
 * Money: the trip's price is what Grab charges us, recorded on
 * shipper_actual_fee_sgd. The customer's fee is never touched (as for every
 * courier). COD orders are refused: Grab cash collection needs a COD service
 * agreement we do not have, so a COD parcel is booked by hand.
 *
 * Service + promo codes (2026-11-13, lib/shipping/grab-options.ts): the trip is
 * booked as the service the operator picked in Dispatch, else the lane default
 * from Admin → Shippers. `quoteGrabOptions` prices INSTANT and SAME_DAY, each
 * with and without our active promo codes, so the operator can compare before
 * booking. A booking quotes first with the codes (Grab: an untested code on
 * Create is a hard 4xx) and uses the one that saves most; if Create still
 * refuses the code, it books once more without it.
 */

/** One service priced plain and with each of the first codes; the cheapest code wins. */
async function quoteService(r: GrabDeliveryRequest, service: GrabServiceType, codes: string[]): Promise<GrabServiceQuote> {
  const out: GrabServiceQuote = {
    service, priceSgd: null, promoCode: null, promoPriceSgd: null, discountSgd: null, pickupEta: null, dropoffEta: null, error: null,
  };
  try {
    const [plain] = await getQuotes(quoteBody(r, service));
    out.priceSgd = discountedQuote(plain).priceSgd;
    out.pickupEta = plain?.estimatedTimeline?.pickup ?? null;
    out.dropoffEta = plain?.estimatedTimeline?.dropoff ?? null;
  } catch (err) {
    out.error = err instanceof Error ? err.message : String(err);
    return out;
  }
  for (const code of codes.slice(0, MAX_PROMO_CODES_PER_QUOTE)) {
    try {
      const [q] = await getQuotes(quoteBody(r, service, code));
      const { priceSgd, discountSgd } = discountedQuote(q);
      if (discountSgd !== null && priceSgd !== null && (out.promoPriceSgd === null || priceSgd < out.promoPriceSgd)) {
        out.promoCode = code; out.promoPriceSgd = priceSgd; out.discountSgd = discountSgd;
      }
    } catch { /* a code Grab refuses on a quote is simply not used */ }
  }
  return out;
}
 
/**
 * Price both services for an order, with and without our codes, for the
 * operator to choose in Dispatch. Nothing is booked.
 */
export async function quoteGrabOptions(db: SupabaseClient, orderId: string): Promise<BookResult> {
  const p = await prepare(db, orderId, { forBooking: false });
  if (!p.ok) return p.result;
  const codes = (await usableCodes(db)).map((c) => c.code);
  const quotes = await Promise.all(GRAB_SERVICE_TYPES.map((s) => quoteService(p.request, s, codes)));
  return {
    status: 200,
    body: {
      quotes,
      default_service: defaultGrabService(p.order, p.settings.options),
      vehicle_type: p.request.vehicleType,
      scheduled: !!p.request.schedule,
    },
  };
}
