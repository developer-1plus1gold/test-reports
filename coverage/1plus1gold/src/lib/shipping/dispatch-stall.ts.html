/**
 * Dispatch-stall facts — pure, no env, no I/O, no carrier SDK.
 *
 * ▶ WHY THIS EXISTS. On 2026-09-14 order `07a9dfca` (S$2,103.02) sat in
 * `assigning_driver` for two hours, Lalamove auto-cancelled it, and it was
 * rebooked seventeen seconds later — restarting the same two-hour wait. Nothing
 * alerted at any point. `shipping_monitor` was watching the two conditions that
 * did NOT apply (un-booked, and `shipper_status='failed'`) and had no concept of
 * "booked, but the carrier cannot find a driver".
 *
 * Worse, it was LOUD about the benign case: it fired *"Lalamove unbooked 35min"*
 * at 21:10 for an order placed at 20:35, after the day's dispatch cutoff, that
 * was never going out that night. Every historical alert of that shape (30, 35,
 * 36, 37, 37 min) is the same non-event. So the monitor was inverted — noisy
 * where nothing was wrong, silent where something was.
 *
 * ▶ THE THRESHOLD IS MEASURED, NOT CHOSEN. Across the full Lalamove history on
 * PROD at the time of writing:
 *   - 28 assignments that succeeded: mean 9 min, **max 40 min**.
 *   -  2 that failed:                **120 min and 120 min** — zero variance,
 *      which is Lalamove's own auto-cancel, not a distribution.
 * So 45 minutes sits just above every observed success and still leaves 75
 * minutes of runway before the carrier gives up. Alerting at 120 would only
 * report what had already happened.
 *
 * ▶ THE CLOCK IS THE LATEST `assigning_driver`, AND BOTH HALVES OF THAT MATTER.
 *   - NOT `created_at`: that order was placed 20:35 and dispatched 09:26 the
 *     next morning. Thirteen hours apart. A clock on `created_at` reports a
 *     thirteen-hour stall for a carrier that has had the job for four minutes.
 *   - NOT the FIRST `assigning_driver`: a rebook re-enters the state, so a clock
 *     on the first entry never stops growing and the alert can never clear.
 */
 
/** One entry of `orders.shipper_status_events`. */
export interface ShipperEvent {
  status: string;
  at: string;
}
 
/**
 * DEFAULT minutes in `assigning_driver` before an assignment counts as stalled.
 *
 * ▶ This is a fallback, not the setting. The live value is per carrier in
 * `shipper_settings.stall_alert_minutes`, because the number is a fact about the
 * CARRIER — 45 is derived from Lalamove's own 40-minute worst observed
 * assignment and its 120-minute auto-cancel, and GrabExpress will differ. This
 * constant is what an unconfigured carrier gets, so a new carrier is never
 * silently un-alerted.
 */
export const STALL_THRESHOLD_MIN = 45;
 
/** Bounds for an operator-supplied threshold. */
export const STALL_THRESHOLD_MIN_FLOOR = 5;
export const STALL_THRESHOLD_MIN_CEIL = 240;
 
/**
 * Resolve the configured threshold, falling back to the default.
 *
 * ▶ CLAMPED, AND THE FLOOR IS THE POINT. A threshold of 0 or 1 would page on
 * every booking — the operator would mute the alert within a day and the whole
 * mechanism would be worse than not having it. A value above the carrier's own
 * auto-cancel window is equally useless: the alert would arrive after the
 * carrier had already given up. Out-of-range values fall back rather than being
 * honoured, and an unparseable one does the same.
 */
export function resolveStallThreshold(configured: number | null | undefined): number {
  if (typeof configured !== "number" || !Number.isFinite(configured)) return STALL_THRESHOLD_MIN;
  const n = Math.round(configured);
  if (n < STALL_THRESHOLD_MIN_FLOOR || n > STALL_THRESHOLD_MIN_CEIL) return STALL_THRESHOLD_MIN;
  return n;
}
 
/**
 * When Lalamove gives up on its own. Not ours to set — observed twice at exactly
 * 120 minutes. Kept here because it is what makes 45 a useful number rather than
 * an arbitrary one, and because an alert at or past this is worthless.
 */
export const LALAMOVE_AUTO_CANCEL_MIN = 120;
 
/**
 * DEFAULT time the operator is back at their desk.
 *
 * ▶ Also a fallback: the live value is `shipper_settings.alert_resume_time`. It
 * sits beside the threshold so a carrier's whole alerting window — start from
 * `shipper_schedules.next_day_cutoff`, end here — is edited in one place rather
 * than split between a schedule row and a constant nobody can reach.
 */
export const DEFAULT_OPERATOR_RESUME_SGT = "08:30";
 
/**
 * Parse `shipper_status_events` tolerantly.
 *
 * The column is jsonb and every writer appends rather than replaces, so a
 * malformed entry from any past writer must not take down the monitor for every
 * other order. Unparseable entries are dropped, not thrown on.
 */
export function readShipperEvents(raw: unknown): ShipperEvent[] {
  if (!Array.isArray(raw)) return [];
  const out: ShipperEvent[] = [];
  for (const e of raw) {
    if (!e || typeof e !== "object") continue;
    const status = (e as Record<string, unknown>).status;
    const at = (e as Record<string, unknown>).at;
    if (typeof status !== "string" || typeof at !== "string") continue;
    if (Number.isNaN(Date.parse(at))) continue;
    out.push({ status, at });
  }
  return out;
}
 
/**
 * How long the order has been in `status` since it LAST entered it, or null if
 * it is not currently in that state.
 *
 * "Currently" is decided by the last event in time order, not by the array's
 * order — appends are chronological today, but a clock that depends on insertion
 * order is one careless writer away from being wrong.
 */
export function minutesInCurrentState(
  events: ShipperEvent[],
  status: string,
  now: Date,
): number | null {
  if (events.length === 0) return null;
  const sorted = [...events].sort((a, b) => Date.parse(a.at) - Date.parse(b.at));
  const last = sorted[sorted.length - 1];
  if (last.status !== status) return null;
 
  // Walk back to the START of this run of the state: a poll that re-writes the
  // same status must not look like a fresh entry and reset the clock.
  let enteredAt = last.at;
  for (let i = sorted.length - 2; i >= 0; i--) {
    if (sorted[i].status !== status) break;
    enteredAt = sorted[i].at;
  }
  return Math.floor((now.getTime() - Date.parse(enteredAt)) / 60_000);
}
 
/**
 * Did the carrier give up and did we put the order straight back in the queue?
 *
 * ▶ THIS READS THE EVENT ARRAY BECAUSE THE LIVE COLUMN CANNOT SHOW IT. The
 * failure on 2026-09-14 existed in `orders.shipper_status` for SEVENTEEN
 * SECONDS before the rebook overwrote it, against a monitor that polls every ten
 * minutes. A poll against a transient state is blind by construction — it can
 * only ever catch failures the operator has not already reacted to, which is the
 * opposite of useful. The transition survives in the array permanently.
 */
export function hasRebookAfterFailure(events: ShipperEvent[]): boolean {
  const sorted = [...events].sort((a, b) => Date.parse(a.at) - Date.parse(b.at));
  for (let i = 1; i < sorted.length; i++) {
    if (sorted[i - 1].status === "failed" && sorted[i].status === "assigning_driver") return true;
  }
  return false;
}
 
/** How many times the carrier has given up on this order. */
export function failedAttempts(events: ShipperEvent[]): number {
  return events.filter((e) => e.status === "failed").length;
}
 
/** `HH:MM` (or `HH:MM:SS`) → minutes past midnight, or null if unparseable. */
export function parseClockMinutes(hhmm: string | null | undefined): number | null {
  if (!hhmm) return null;
  const m = /^(\d{1,2}):(\d{2})(?::\d{2})?$/.exec(hhmm.trim());
  if (!m) return null;
  const h = Number(m[1]);
  const min = Number(m[2]);
  if (h > 23 || min > 59) return null;
  return h * 60 + min;
}
 
/**
 * Day of week in Singapore, 0=Sunday — the key `shipper_schedules` is stored on.
 *
 * Read through Intl rather than `getDay()`, which answers in the RUNNER's zone:
 * the cron runs on UTC infrastructure, so between 16:00 and 24:00 UTC the UTC
 * date is already the previous Singapore day and every schedule lookup in that
 * window would read the wrong row — which is exactly the evening this monitor
 * exists to get right.
 */
export function sgtDayOfWeek(now: Date): number {
  const short = new Intl.DateTimeFormat("en-US", { timeZone: "Asia/Singapore", weekday: "short" }).format(now);
  return ["Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"].indexOf(short);
}
 
/** Minutes past midnight in Singapore for an instant. */
export function sgtMinutesOfDay(now: Date): number {
  // Read the wall clock through Intl rather than by adding 8h to UTC — the
  // offset is a fact about the zone, not a constant to hardcode.
  const parts = new Intl.DateTimeFormat("en-GB", {
    timeZone: "Asia/Singapore",
    hour: "2-digit",
    minute: "2-digit",
    hour12: false,
  }).formatToParts(now);
  const h = Number(parts.find((p) => p.type === "hour")?.value ?? "0");
  const m = Number(parts.find((p) => p.type === "minute")?.value ?? "0");
  return h * 60 + m;
}
 
/**
 * Is it outside the hours the operator can act on a dispatch problem?
 *
 * ▶ THE START COMES FROM `shipper_schedules.next_day_cutoff`, DELIBERATELY.
 * That column already exists, is already in the admin schedule editor, already
 * has a PATCH route — and drove nothing: it was NULL on all seven days while the
 * rule it represents lived only in the free-text customer note ("Orders after
 * 7:00 PM will be shipped next morning"). That is exactly how the note and the
 * operator's own understanding came to disagree about whether the cutoff is 7:00
 * or 7:30. Reading the field means the customer-facing copy and the alert cannot
 * drift apart: there is one number and both derive from it.
 *
 * ▶ A NULL CUTOFF MEANS NO QUIET WINDOW — it fails toward alerting. A missing
 * config value must never be able to silence a money-path alarm; the failure we
 * can afford is a ping at an awkward hour, not a stalled parcel nobody hears
 * about.
 */
export function isQuietHour(
  now: Date,
  cutoffSgt: string | null | undefined,
  resumeSgt: string | null | undefined = DEFAULT_OPERATOR_RESUME_SGT,
): boolean {
  const start = parseClockMinutes(cutoffSgt);
  if (start === null) return false;
  const end = parseClockMinutes(resumeSgt) ?? parseClockMinutes(DEFAULT_OPERATOR_RESUME_SGT)!;
  const t = sgtMinutesOfDay(now);
  // The window wraps midnight (19:30 → 08:30), so "inside" is a union, not a range.
  return start > end ? t >= start || t < end : t >= start && t < end;
}
