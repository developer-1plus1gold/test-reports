/**
 * Next-day evening delivery — the ONE promise calculation.
 * Design: docs/fulfilment-lanes-design.md (v2) §1–§3.
 *
 * Pure (no I/O). The same function feeds /api/shipping-options (what the
 * checkout shows) and the server re-check in the order routes (the 409), so a
 * customer can never be charged for a promise the server would not make.
 *
 * Rule:
 *   • The ORDER cutoff is today's `next_day_cutoff` on the evening schedule
 *     (19:00 by default). Before it → the earliest delivery day is tomorrow;
 *     at or after it → the day after tomorrow.
 *   • From that day, take the first day whose schedule is not disabled AND has
 *     at least one enabled window. A disabled day is a day we do not deliver, so
 *     the promise rolls forward past it.
 *   • Today's `open_time`–`close_time` on the evening schedule are the hours
 *     the option is offered at checkout (same shape as every other shipper's
 *     operating hours). Outside them there is no promise. Today's `enabled`
 *     flag is NOT consulted here — for evening it means "we deliver that day".
 *   • The customer does not pick the date. They pick a window on that date
 *     (one window today; more slots later are just more rows).
 *
 * All wall-clock maths is Asia/Singapore (UTC+8, no DST).
 */
import type { DeliveryWindowDef, WindowInput } from "@/lib/shipping/delivery-windows";
import { clockLabel, hhmmToMinutes, relativeDayPrefix, sgtDateLabel, sgtParts, timeRangeLabel } from "@/lib/shipping/delivery-windows";
 
export { clockLabel };
 
const SGT_OFFSET_MIN = 8 * 60;
/** How far ahead to look for an enabled delivery day before giving up. */
const LOOKAHEAD_DAYS = 8;
 
export interface EveningWindow {
  start: string;     // ISO UTC
  end: string;       // ISO UTC
  label: string;     // "5–9 pm"
  dateLabel: string; // "Thu 9 Oct"
  fullLabel: string; // "tomorrow, Thu 9 Oct, 5–9 pm" (no lead beyond tomorrow)
}
 
export interface EveningPromise {
  /** SGT date we deliver on, e.g. "Thu 9 Oct". */
  dateLabel: string;
  /** Days from today (1 = tomorrow). */
  dayOffset: number;
  windows: EveningWindow[];
  defaultWindow: EveningWindow;
  /** "7 pm" while today's cutoff is still ahead; null once it has passed (or none is set). */
  orderByLabel: string | null;
}
 
export type EveningInput = Pick<WindowInput, "now" | "windows" | "scheduleByDow">;
 
function sgtInstant(now: Date, dayOffset: number, minutes: number): Date {
  const p = sgtParts(now);
  return new Date(Date.UTC(p.y, p.m0, p.d + dayOffset, 0, minutes - SGT_OFFSET_MIN));
}
 
function windowsOn(input: EveningInput, dayOffset: number): EveningWindow[] {
  const dow = (sgtParts(input.now).dow + dayOffset) % 7;
  const sched = input.scheduleByDow[dow];
  if (sched && sched.enabled === false) return [];
  return input.windows
    .filter((w: DeliveryWindowDef) => w.enabled !== false && (w.day_of_week == null || w.day_of_week === dow))
    .map((w) => ({ w, s: hhmmToMinutes(w.start_time), e: hhmmToMinutes(w.end_time) }))
    .filter((x): x is { w: DeliveryWindowDef; s: number; e: number } => x.s != null && x.e != null && x.e > x.s)
    .sort((a, b) => a.s - b.s || (a.w.sort_order ?? 0) - (b.w.sort_order ?? 0))
    .map((x) => {
      const start = sgtInstant(input.now, dayOffset, x.s);
      const end = sgtInstant(input.now, dayOffset, x.e);
      const label = timeRangeLabel(x.s, x.e);
      const dateLabel = sgtDateLabel(start);
      return { start: start.toISOString(), end: end.toISOString(), label, dateLabel, fullLabel: `${relativeDayPrefix(dayOffset)}${dateLabel}, ${label}` };
    })
    .filter((o, i, arr) => arr.findIndex((x) => x.start === o.start && x.end === o.end) === i);
}
 
/** The delivery promise for an order placed at `input.now`, or null when no day in range delivers. */
export function eveningPromise(input: EveningInput): EveningPromise | null {
  const today = sgtParts(input.now);
  const todayRow = input.scheduleByDow[today.dow];
  const open = hhmmToMinutes(todayRow?.open_time) ?? 0;
  const close = hhmmToMinutes(todayRow?.close_time) ?? 24 * 60;
  if (today.minutes < open || today.minutes >= close) return null;
  const cutoff = hhmmToMinutes(todayRow?.next_day_cutoff);
  const beforeCutoff = cutoff == null || today.minutes < cutoff;
  const first = beforeCutoff ? 1 : 2;
  for (let offset = first; offset < first + LOOKAHEAD_DAYS; offset++) {
    const windows = windowsOn(input, offset);
    if (windows.length > 0) {
      return {
        dateLabel: windows[0].dateLabel,
        dayOffset: offset,
        windows,
        defaultWindow: windows[0],
        orderByLabel: cutoff != null && beforeCutoff ? clockLabel(cutoff) : null,
      };
    }
  }
  return null;
}
 
/**
 * Server-side re-check of the window a client submitted. A customer who picked
 * the option at 18:58 and paid at 19:03 is still held to the promise they saw:
 * the window is accepted if it matches the promise at now OR at now − grace.
 */
export function isEveningWindowPromised(
  input: EveningInput,
  startIso: string,
  endIso: string,
  graceMinutes = 15,
): boolean {
  const s = new Date(startIso).getTime();
  const e = new Date(endIso).getTime();
  if (!Number.isFinite(s) || !Number.isFinite(e)) return false;
  const earlier = { ...input, now: new Date(input.now.getTime() - graceMinutes * 60_000) };
  return [eveningPromise(input), eveningPromise(earlier)].some((p) =>
    !!p && p.windows.some((w) => new Date(w.start).getTime() === s && new Date(w.end).getTime() === e),
  );
}
