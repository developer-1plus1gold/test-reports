/**
 * Delivery-window reminders — one admin WhatsApp per Lalamove delivery window,
 * sent ~15 minutes before the window opens (Mukesh, 2026-10-26).
 *
 * Covers both lanes that carry a window: next-day EVENING and same-day EXPRESS.
 * One message per window, never one per order, listing every order in it with
 * its booking state, so the operator sees the whole slot at a glance:
 *
 *   Evening window 7:30–9:30 pm (Mon 26 Oct) starts in 15 min — 3 orders:
 *   #A1B2C3D4 driver assigned · #C3D4E5F6 waiting for driver ·
 *   #E5F6A7B8 NOT BOOKED ⚠️
 *
 * ▶ ONE LINE ON PURPOSE. The text goes into a WhatsApp TEMPLATE variable, and
 * Meta rejects a template parameter containing a newline — the whole send would
 * fail. Orders are separated by " · " instead.
 *
 * Pure: no I/O, no clock. The cron owns the query, the dedup row and the send.
 */
 
import { deliveryLane, sgDate, slotKey, type DeliveryLane } from "./dispatch-lanes";
 
/** How long before a window opens the reminder fires. */
export const WINDOW_REMINDER_LEAD_MIN = 15;
 
export interface WindowOrder {
  id: string;
  delivery_option?: string | null;
  delivery_window_start: string;
  delivery_window_end?: string | null;
  shipper_order_id?: string | null;
  shipper_status?: string | null;
}
 
export type BookingState = "not_booked" | "booking_failed" | "waiting_for_driver" | "driver_assigned";
 
/**
 * Booking state for the reminder. No carrier booking id means NOT booked,
 * whatever the status column says — the id is the proof a booking exists.
 */
export function bookingState(o: Pick<WindowOrder, "shipper_order_id" | "shipper_status">): BookingState {
  if (o.shipper_status === "failed") return "booking_failed";
  if (!o.shipper_order_id) return "not_booked";
  if (o.shipper_status === "assigning_driver" || o.shipper_status === "pending" || !o.shipper_status) {
    return "waiting_for_driver";
  }
  return "driver_assigned";
}
 
const STATE_TEXT: Record<BookingState, string> = {
  not_booked: "NOT BOOKED ⚠️",
  booking_failed: "BOOKING FAILED ⚠️",
  waiting_for_driver: "waiting for driver",
  driver_assigned: "driver assigned",
};
 
/** True when an order in this state needs the operator to act now. */
export function isUrgent(state: BookingState): boolean {
  return state === "not_booked" || state === "booking_failed";
}
 
/**
 * True when the window opens within the lead time and has not opened yet.
 * Windows already open are excluded — a reminder after the start is noise.
 */
export function windowIsDue(windowStartIso: string, now: Date, leadMin = WINDOW_REMINDER_LEAD_MIN): boolean {
  const start = new Date(windowStartIso).getTime();
  if (Number.isNaN(start)) return false;
  const msUntil = start - now.getTime();
  return msUntil > 0 && msUntil <= leadMin * 60_000;
}
 
export interface WindowGroup {
  key: string;
  lane: DeliveryLane;
  windowStart: string;
  windowEnd: string | null;
  orders: WindowOrder[];
}
 
/** Group due orders by window (lane + SGT date + start + end). */
export function groupWindows(orders: readonly WindowOrder[]): WindowGroup[] {
  const map = new Map<string, WindowGroup>();
  for (const o of orders) {
    const key = slotKey(o);
    let g = map.get(key);
    if (!g) {
      g = {
        key,
        lane: deliveryLane(o),
        windowStart: new Date(o.delivery_window_start).toISOString(),
        windowEnd: o.delivery_window_end ? new Date(o.delivery_window_end).toISOString() : null,
        orders: [],
      };
      map.set(key, g);
    }
    g.orders.push(o);
  }
  return [...map.values()].sort((a, b) => a.windowStart.localeCompare(b.windowStart));
}
 
// Fixed tables, not Intl month/day names: Node ICU and browsers spell them
// differently (the 2026-08-23 "Sept" lesson). SGT has no DST, so +8h is exact.
const MONTHS = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
const DAYS = ["Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"];
 
function sgt(iso: string): Date {
  return new Date(new Date(iso).getTime() + 8 * 3600_000);
}
 
/** "7:30" / "9" style clock, 12-hour, with the am/pm of the instant. */
function clock(iso: string): { text: string; ampm: "am" | "pm" } {
  const d = sgt(iso);
  const h24 = d.getUTCHours();
  const m = d.getUTCMinutes();
  const h12 = h24 % 12 === 0 ? 12 : h24 % 12;
  return { text: m === 0 ? `${h12}` : `${h12}:${String(m).padStart(2, "0")}`, ampm: h24 < 12 ? "am" : "pm" };
}
 
/** "7:30–9:30 pm" — the am/pm is stated once when both ends share it. */
export function windowLabel(startIso: string, endIso: string | null): string {
  const s = clock(startIso);
  if (!endIso) return `${s.text} ${s.ampm}`;
  const e = clock(endIso);
  return s.ampm === e.ampm ? `${s.text}–${e.text} ${e.ampm}` : `${s.text} ${s.ampm}–${e.text} ${e.ampm}`;
}
 
function dayLabel(iso: string): string {
  const d = sgt(iso);
  return `${DAYS[d.getUTCDay()]} ${d.getUTCDate()} ${MONTHS[d.getUTCMonth()]}`;
}
 
/** The one-line WhatsApp detail for a window. Urgent orders are listed first. */
export function reminderMessage(g: WindowGroup, now: Date): string {
  const laneText = g.lane === "evening" ? "Evening" : "Same-day";
  const mins = Math.max(1, Math.round((new Date(g.windowStart).getTime() - now.getTime()) / 60_000));
  const rank: Record<BookingState, number> = { not_booked: 0, booking_failed: 1, waiting_for_driver: 2, driver_assigned: 3 };
  const lines = g.orders
    .map((o) => ({ o, s: bookingState(o) }))
    .sort((a, b) => rank[a.s] - rank[b.s] || a.o.id.localeCompare(b.o.id))
    .map(({ o, s }) => `#${o.id.slice(0, 8).toUpperCase()} ${STATE_TEXT[s]}`);
  const urgent = g.orders.filter((o) => isUrgent(bookingState(o))).length;
  const n = g.orders.length;
  const head = `${laneText} window ${windowLabel(g.windowStart, g.windowEnd)} (${dayLabel(g.windowStart)}) starts in ${mins} min — ${n} order${n === 1 ? "" : "s"}`;
  const tail = urgent > 0 ? ` — ${urgent} need${urgent === 1 ? "s" : ""} booking now` : "";
  return `${head}: ${lines.join(" · ")}${tail}`;
}
 
/** SGT date of the window — exposed for the dedup row. */
export function windowDate(g: WindowGroup): string {
  return sgDate(g.windowStart);
}
