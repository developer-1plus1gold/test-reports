/**
 * Canonical brand positioning — the ONE place the business describes itself.
 *
 * Source of truth: `docs/brand-positioning.md` (v2.0, approved by Mukesh
 * 2026-10-13). This module is that document's §4 copy blocks in code, so the
 * meta description, PWA manifest, JSON-LD, llms.txt, agent facts, FAQ and
 * Guides IMPORT the positioning instead of each retyping it.
 *
 * ▶ Why (audit 2026-10-13): ~22 independently written versions of "what we
 * sell" had drifted — most still described a 1g-split / Valcambi business, and
 * the first gold.com.sg directory draft inherited that. Retyped copy drifts;
 * imported copy cannot. When the positioning changes, change the doc, then
 * this file, and every surface follows.
 *
 * ▶ Rules baked in here (each one was a real defect in the old copy):
 *   1. Three offerings, ALWAYS in this order: The Premium Curve® · Standard
 *      Gold Bars and Coins · MAG · My Allocated Gold™. Never fold Standard into a generic
 *      phrase (an early draft read as if we sold only the Curve and MAG).
 *   2. The LBMA / 999.9 / GST line is stated ONCE for the whole range — never
 *      attached to one offering. Scoping it to Standard implied the Curve and
 *      MAG were not LBMA.
 *   3. No product-specific exceptions in the range line or any customer copy.
 *      The few remaining Puregold-Singapore gift bars (non-LBMA, non-IPM) are
 *      being run down and are deliberately NOT named in copy (Mukesh
 *      2026-10-13). Their own product pages carry their per-bar details.
 *   4. MAG is "recorded against a specific serial-numbered bar" — never
 *      "allocated to you by name" (the MAG Agreement + APP_FACTS say share,
 *      not by-name allocation).
 *   5. Gold Yield is invite-only and stays OUT of public copy.
 *   6. No "cheapest" / "lowest in Singapore" claims, and no specific premium %
 *      in static copy — premiums move; point to Our Pricing.
 *   7. Name refiners ONLY from REFINER_EXAMPLES — the list approved for Terms
 *      §3 (Wave D, Mukesh 2026-10-15). A mint we sell nothing from kept
 *      reappearing from an older hand-typed list; the guard test
 *      `refiner-naming.test.ts` now fails on any retyped refiner list.
 */

/** The three offerings as one inline sentence fragment: "A; B; and C". */
export function offeringsSentence(): string {
  const [curve, standard, mag] = OFFERINGS;
  return `${curve.name} — ${curve.phrase}; ${standard.name} — ${standard.phrase}; and ${mag.name} — ${mag.phrase}`;
}
 
/** ≈90 words — directory listings (gold.com.sg and similar). */

}
 
/** Plain-text bullet list of the three offerings ("- Name — one line"). */

}

// THE STORYLINE (docs/brand-positioning.md §0, v2.1 — Mukesh 2026-10-13)
//
// How we introduce ourselves, in this ORDER, said with pride but humbly:
//   1. Who we are   — a licensed, registered Singapore gold dealer.
//   2. Our edge     — three ways to price gold, every premium published;
//                     The Premium Curve® is the signature.
//   3. What we stand on — the seven trust pillars (the drumbeat pillars).
//   4. Our strength — Technology: an AI-native platform, built and run here.
//   5. Recognition  — the Singapore SME 500 Award 2026, from ATC.
//
// ▶ The order is the point. A list of everything at once reads as "stuffed"
//   (Mukesh's word); a story that builds reads as earned.
//
// ⚠️ ATC's five criteria do NOT include technology. Technology is OUR claim
//   (the strongest of our own pillars); the award is ATC's independent
//   recognition on THEIR criteria. Never write that ATC judged, assessed or
//   approved our technology. Quote ATC_CRITERIA verbatim — ATC's own wording.
//
// Award rules (src/lib/newsroom/atc.ts): the name "Singapore SME 500 Award
// 2026" is never altered; we "received" it (never "won", never "award-
// winning" about a product); the seal is never paired with a bar.
// ───────────────────────────────────────────────────────────────────────────

/** The pillar names as a sentence fragment: "a, b, … and g". */
export function trustPillarNames(): string {
  const names = TRUST_PILLARS.map((p) => p.name.toLowerCase());
  return `${names.slice(0, -1).join(", ")} and ${names[names.length - 1]}`;
}

/**
 * About-us for partner profiles and outreach (200–800 chars, pinned by test).
 * Follows the storyline order exactly. Recognition comes last, said humbly.
 */
export function aboutUs(pspmLicence: string): string {
  const [curve, standard, mag] = OFFERINGS;
  return (
    `1Plus1Gold is a licensed Singapore gold dealer (PSPM Licence ${pspmLicence}, Ministry of Law), selling gold bars online. ` +
    `Our edge is how we price: three ways to buy, each premium published — ${curve.name}, our signature, where your premium falls as your grams add up; ` +
    `${standard.name}; and ${mag.name}. ` +
    `We stand on seven trust pillars: ${trustPillarNames()}. ` +
    "Technology is our strongest: an AI-native platform we built and run, with a person on every decision that matters. " +
    `In our first months, ATC gave us the ${AWARD_NAME}. We are grateful, and still building.`
  );
}
 
/**
 * The storyline as agent-facing prose (APP_FACTS "OUR STORY"). Kept here so

  return [

    `1. Who we are: a licensed, registered Singapore gold dealer (PSPM Licence ${pspmLicence}, Ministry of Law; AML/CFT supervised by MAS), selling gold bars online.`,
