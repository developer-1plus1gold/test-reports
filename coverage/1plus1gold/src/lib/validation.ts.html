// ─────────────────────────────────────────────────────────────────────────────
// Shared cart body schema
//
// Three routes (orders/create, payments/hitpay/create, payments/stripe/
// create-intent) accept the same checkout body shape. Defining it once
// avoids drift when fields are added or constraints change. Each route
// extends with its own payment-method narrowing and may add route-
// specific fields via `.merge()`.
//
// `.passthrough()` is intentional — the cart frontend sometimes ships
// transitional UI-only fields ahead of the server consuming them; we
// don't want a typo in *new* checkout-step data to break *existing*
// checkouts. Numeric guards still hold on the fields we DO read.
// ─────────────────────────────────────────────────────────────────────────────
 
export const cartItemSchema = z.object({
  productId: z.string().nullish(),
  grams:     z.number().positive(),
  quantity:  z.number().int().positive().max(500),
  // Gift flow (Phase 3 2026-05-18; per-card variant added 2026-05-19) —
  // two mutually-exclusive shapes:
  //   • giftMessage  (string)   — same message replicated × quantity
  //   • giftMessages (string[]) — per-card override; length === quantity
  // 240-char hard cap on each entry. Empty string rejected — gift requires
  // intent. Refinements below pin the mutual-exclusion + length invariants.
  giftMessage:  z.string().trim().min(1).max(240).optional(),
  giftMessages: z.array(z.string().trim().min(1).max(240)).max(500).optional(),
}).passthrough().refine(
  (v) => !(v.giftMessage && v.giftMessages && v.giftMessages.length > 0),
  { message: "Send giftMessage or giftMessages, not both",  path: ["giftMessages"] },
).refine(
  (v) => !v.giftMessages || v.giftMessages.length === v.quantity,
  { message: "giftMessages length must equal quantity",     path: ["giftMessages"] },
);
 
export const cartAccessorySchema = z.object({
  productId: z.string(),
  quantity:  z.number().int().positive(),
}).passthrough();
 
export const cartBodySchema = z.object({
  // Messages chosen to match the legacy error strings the cart frontend +
  // existing test suite already assert against (see b2-payment-routes,
  // c9-payment-routes-early). Don't change without updating those tests.
  // Gold items MAY be empty for an accessory-only order — the "at least one
  // line" rule is enforced across items+accessories by the refine below, so a
  // truly empty cart is still rejected with "Cart is empty". (Accessories ship
  // as their own array; see the accessories-in-cart gotcha.)
  items:                z.array(cartItemSchema),
  accessories:          z.array(cartAccessorySchema).optional(),
  lockedSpotSgd:        z.number({ error: "Invalid locked prices" }).positive("Invalid locked prices"),
  // nonnegative (was positive): an accessory-only cart locks a $0 gold subtotal.
  // A GOLD cart must still be positive — enforced conditionally in the refine
  // below so the "attacker claims $0 gold subtotal" case stays rejected.
  lockedSubtotalSgd:    z.number({ error: "Invalid locked prices" }).nonnegative("Invalid locked prices"),
  shipperType:          z.string().optional(),
  shippingFee:          z.number().nonnegative().optional(),
  // The exact total shown on the "Proceed to Payment" button. When present and
  // within the server's cent-tolerance recompute, it is what we charge — so the
  // gateway amount equals the displayed amount. Optional for back-compat.
  // Upper-bounded (defence-in-depth, security_agent_pr 2026-06-20): the charge
  // is already clamped to within ±0.05 of the server's own recompute, but cap
  // the raw input at a generous ceiling so an absurd value can never reach a
  // gateway even via a server-side miscompute. S$1M is far above any real order.
  displayedTotalSgd:    z.number().nonnegative().max(1_000_000).optional(),
  deliveryAddress:      z.record(z.string(), z.unknown()).optional(),
  quotationId:          z.string().optional(),
  quotationExpiresAt:   z.string().optional(),
  senderStopId:         z.string().optional(),
  stopId:               z.string().optional(),
  serviceType:          z.string().optional(),
  recipientName:        z.string().optional(),
  recipientPhone:       z.string().optional(),
  paymentMethod:        z.string().optional(),
  promoCode:            z.string().optional(),
  referralCode:         z.string().optional(),
  refundPolicyAccepted: z.boolean().optional(),
  // First-party analytics visitor id, attribution-ONLY. The web sends its
  // p1g_aid via the cookie (read server-side); the NATIVE iOS app has no
  // cookie jar for our domain, so it threads its persistent visitorId here
  // instead. Routes read it strictly as a FALLBACK to the cookie and never let
  // it touch money/auth (see orders/create + hitpay/create).
  // ⚠️ Keep this LENIENT (length cap only) — do NOT add `.uuid()`/`.regex()`
  // here. A hard schema reject on a malformed visitorId would 400 an
  // already-paid checkout, which is far worse than an unattributed one. The
  // UUID format gate lives at the READ site (resolveAttributionVisitorId), where
  // a bad value is simply ignored instead of failing the order.
  visitorId:            z.string().max(64).optional(),
}).passthrough().superRefine((v, ctx) => {
  const goldCount = v.items?.length ?? 0;
  const accCount  = v.accessories?.length ?? 0;
  // A cart must have at least one line total — gold OR accessory. Preserves the
  // "Cart is empty" guard (previously items.min(1)) now that gold may be empty
  // for an accessory-only order.
  if (goldCount + accCount < 1) {
    ctx.addIssue({ code: "custom", message: "Cart is empty", path: ["items"] });
  }
  // A GOLD cart must lock a positive gold subtotal — this is the "attacker
  // claims $0 gold subtotal to underpay" guard that lockedSubtotalSgd.positive()
  // used to enforce. Only accessory-only carts (no gold) may lock $0.
  if (goldCount > 0 && !(v.lockedSubtotalSgd > 0)) {
    ctx.addIssue({ code: "custom", message: "Invalid locked prices", path: ["lockedSubtotalSgd"] });
  }
});
