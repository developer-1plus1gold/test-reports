/**
 * Newsroom — the shared vocabulary. One module so the page, the API routes, the
 * admin editor and the tests cannot drift on what a post IS.
 *
 * Design: docs/newsroom-design.md.
 *
 * ▶ THE SPLIT THAT DRIVES EVERYTHING. A post is one of two shapes:
 *
 *   AUTHORED  (announcement · article · newsletter) — we wrote it, we host it,
 *             it has a markdown body and a page at /newsroom/<slug>.
 *   EXTERNAL  (press · market_news)                 — somebody else wrote it,
 *             we link out, there is NO body and NO page of ours.
 *
 * The external rule is a copyright rule, not a stylistic one: reproducing a
 * third party's article at our URL is republishing their work on our domain.
 * It is enforced by a CHECK constraint in migration 20260915 as well as here,
 * so an external row carrying a body cannot be written even by a hand-rolled
 * SQL statement.
 */
 
export const AUTHORED_KINDS = ["announcement", "article", "newsletter"] as const;
export const EXTERNAL_KINDS = ["press", "market_news"] as const;
 
export type AuthoredKind = (typeof AUTHORED_KINDS)[number];
export type ExternalKind = (typeof EXTERNAL_KINDS)[number];
export type PostKind = AuthoredKind | ExternalKind;
 
export const POST_KINDS: readonly PostKind[] = [...AUTHORED_KINDS, ...EXTERNAL_KINDS];
 
export const POST_STATUSES = ["draft", "published", "archived"] as const;
export type PostStatus = (typeof POST_STATUSES)[number];
 
/** Reader-facing label for a kind. Never the raw enum — "market_news" is ours. */
export const KIND_LABEL: Record<PostKind, string> = {
  announcement: "Announcement",
  article:      "Article",
  newsletter:   "Newsletter",
  press:        "In the press",
  market_news:  "Market news",
};
 
export function isAuthoredKind(kind: string): kind is AuthoredKind {
  return (AUTHORED_KINDS as readonly string[]).includes(kind);
}
 
export function isExternalKind(kind: string): kind is ExternalKind {
  return (EXTERNAL_KINDS as readonly string[]).includes(kind);
}
 
/** A published post as the public surfaces see it. */
export interface NewsroomPost {
  id:             string;
  slug:           string;
  kind:           PostKind;
  title:          string;
  summary:        string | null;
  body_md:        string | null;
  hero_image_url: string | null;
  source_name:    string | null;
  source_url:     string | null;
  published_at:   string;
}
 
/** The same row plus the editorial fields only the admin surface needs. */
export interface NewsroomPostAdmin extends Omit<NewsroomPost, "published_at"> {
  status:       PostStatus;
  published_at: string | null;
  created_at:   string;
  updated_at:   string;
}
 
/**
 * Where a post sends the reader. Authored posts open our own page; external
 * ones leave the site. Centralised because getting it wrong on ONE surface is
 * how an external item silently acquires a body page it must not have.
 */
export function postHref(post: Pick<NewsroomPost, "kind" | "slug" | "source_url">): string {
  return isExternalKind(post.kind) ? (post.source_url ?? "#") : `/newsroom/${post.slug}`;
}
 
export function isOutboundPost(post: Pick<NewsroomPost, "kind">): boolean {
  return isExternalKind(post.kind);
}
 
/**
 * Slug rules. Lowercase, a-z0-9 and single hyphens, 3-80 chars. Deliberately
 * narrow: the slug is permanent once published (a rename becomes a redirect),
 * so it should not be able to contain anything that later needs escaping in a
 * URL, a sitemap entry or an OpenGraph tag.
 */
export const SLUG_RE = /^[a-z0-9]+(?:-[a-z0-9]+)*$/;
 
export function isValidSlug(slug: string): boolean {
  return slug.length >= 3 && slug.length <= 80 && SLUG_RE.test(slug);
}
 
/** Derive a candidate slug from a title. Editable before first publish. */
export function slugify(title: string): string {
  return title
    .toLowerCase()
    .normalize("NFKD")
    .replace(/[̀-ͯ]/g, "")   // strip diacritics
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "")
    .slice(0, 80)
    .replace(/-+$/, "");
}
 
/**
 * Only http(s) may reach an anchor's href.
 *
 * Parsed with `URL`, NOT a `startsWith("http")` prefix test — that accepts
 * `http-evil:` and is defeated by the control characters browsers strip before
 * resolving a scheme. A `javascript:` URI rendered into an admin-facing anchor
 * was a real finding on 2026-09-08; this is the same guard, applied at the
 * projection layer so it also covers a row written directly to the database.
 */
export function safeExternalUrl(raw: string | null | undefined): string | null {
  if (!raw) return null;
  try {
    const u = new URL(raw.trim());
    return u.protocol === "http:" || u.protocol === "https:" ? u.toString() : null;
  } catch {
    return null;
  }
}
 
/**
 * Only a site-relative path or an http(s) URL may reach an image `src` — or,
 * more to the point, the `href` of the anchor that wraps it.
 *
 * ▶ WHY THIS IS NOT `safeExternalUrl`. That guard parses with bare `new URL()`,
 * which THROWS on a relative path and so returns null for one. Our own hero
 * images are served from `/public`, and the only live value on prod today is
 * `/images/newsroom/sme500-certificate-2026.webp` — so reusing `safeExternalUrl`
 * here would have blanked the certificate on the SME500 announcement. The
 * pre-existing data is part of the contract: a stricter rule that rejects the
 * shape your own table is in is an outage, not a hardening (the lesson from the
 * 2026-09-11 `country='Singapore'` case, where moving a write behind an API
 * moved it behind that API's validation and broke the profile save for 81% of
 * customers).
 *
 * ▶ WHY IT IS NEEDED AT ALL. The post page renders the hero inside
 * `<a href={hero_image_url} target="_blank">` so a reader can open the asset at
 * full size. A `javascript:` URI in that href EXECUTES on click, and
 * `rel="noopener noreferrer"` does nothing about it. The write path is
 * admin-gated, so this is not remotely exploitable — but `docs/attacker-playbook.md`
 * **F5** disposes every CSP-chain finding on the standing fact that `src/` holds
 * no XSS sink, and a DB-sourced `javascript:` href IS one. On 2026-09-14 a
 * `dangerouslySetInnerHTML` was removed from the money map specifically to keep
 * that premise true; leaving this open would have spent the same premise a week
 * later. Guarded at BOTH layers (write + projection) because a row can be
 * written straight to the database, bypassing the route entirely.
 *
 * Rejected: every non-http(s) scheme, and the PROTOCOL-RELATIVE form `//host`,
 * which reads like a path and resolves to a third-party origin.
 */
export function safeImageUrl(raw: string | null | undefined): string | null {
  if (!raw) return null;
  const s = raw.trim();
  if (!s) return null;
  // `//host/path` is protocol-relative — an external origin wearing a path's
  // clothes. Checked before the leading-slash branch, which would admit it.
  if (s.startsWith("//")) return null;
  if (s.startsWith("/")) return s;
  return safeExternalUrl(s);
}
