/**
 * Projecting a published market brief onto the public newsroom (Phase 4).
 *
 * ▶ WHAT GOES PUBLIC, AND WHAT DELIBERATELY DOES NOT.
 * ONLY `brief_english_md` — the market commentary. The founder note is
 * EXCLUDED, and not for tone:
 *   • It carries `{{customer_activity}}`, an unrendered per-customer
 *     personalisation token. Verified 2026-09-16 across all 10 stored issues:
 *     `brief_english_md` contains ZERO placeholders, `founder_note_md` contains
 *     that one. Publishing it raw would print the braces on a public page.
 *   • It is written to an audience that has already bought — "Dear Customers",
 *     "Thanks for being our customer" — and August's carries qualitative
 *     business commentary ("orders have come in steadily, and customers have
 *     kept signing on week after week").
 *
 * That split also dissolves the gating question docs/newsroom-design.md §6a
 * deferred, rather than deciding it: the public gets the brief, the customer
 * gets brief + founder note + their own activity. Different artefacts, so
 * publishing publicly takes nothing away from being a customer.
 *
 * ▶ NO BUSINESS DATA, EVER. Operator decision 2026-09-16: "No data to be shown
 * on order volume in a newsletter." Phase 4 ships with no stats engine at all —
 * the suppression rule in §6b is specified and stays unbuilt. Nothing here
 * computes or copies a figure; it moves prose that a human already approved
 * through the compliance agent.
 *
 * ▶ WHY A COPY IS SAFE HERE, when §4a refused one for the social wall. That
 * refusal was about a table maintained by a DIFFERENT writer (the marketing
 * publisher), where a copy drifts. Here the copy is written by the one route
 * that owns the source's published state, and a correction re-publishes through
 * that same route — so the upsert is keyed on the period and UPDATES the same
 * row. There is exactly one place that can ever change it, so it cannot drift.
 */
 
/** Months spelled out. Fixed table, NEVER Intl: Node's ICU renders en-GB
 *  September as "Sept" where a browser renders "September", so an Intl label
 *  differs between the test runner and the page. Paid for once already. */
const MONTHS = [
  "January", "February", "March", "April", "May", "June",
  "July", "August", "September", "October", "November", "December",
] as const;
 
/** Stable, sortable, and derived purely from the period — so a correction
 *  re-publishing the same month lands on the SAME row rather than a second one. */
export function newsletterSlug(year: number, month: number): string {
  return `market-brief-${year}-${String(month).padStart(2, "0")}`;
}
 
export function newsletterTitle(year: number, month: number): string {
  const name = MONTHS[month - 1] ?? String(month);
  return `Market Brief — ${name} ${year}`;
}
 
/**

 * cannot contradict the body.
 *
 * Markdown chrome is stripped because the index renders plain text: a heading's
 * `##` or a `**bold**` lede would otherwise show its syntax on the card.
 */
export function newsletterSummary(briefMd: string): string | null {
  const lines = briefMd
    .split("\n")
    .map((l) => l.trim())
    .filter((l) => l.length > 0 && !l.startsWith("|") && !l.startsWith("---"));
 
  // Skip a leading heading. Every real brief opens with "## Gold market in
  // <Month> <Year>", which under a card titled "Market Brief — <Month> <Year>"
  // is the same words twice. The line AFTER it is the actual lede
  // ("**Gold slipped 2.3% in April as …**"). Falls back to the heading if a
  // brief is nothing but headings, so the card is never blank.

export interface NewsletterPostRow {
  slug: string;
  kind: "newsletter";
  title: string;
  summary: string | null;
  body_md: string;
  source_url: null;
  status: "published";
  published_at: string;
}
 
/**
 * Build the newsroom row for a published brief. Pure — no I/O — so the shape
 * that reaches a public page is testable without a database.
 *
 * `source_url: null` is explicit rather than omitted: the table's shape CHECK
 * requires an authored kind to carry `body_md` AND a null `source_url`, so
 * stating it keeps the row valid by construction instead of by omission.
 */
export function buildNewsletterPost(input: {
  periodYear: number;
  periodMonth: number;
  briefEnglishMd: string;
  publishedAt: string;
}): NewsletterPostRow {
  return {
    slug: newsletterSlug(input.periodYear, input.periodMonth),
    kind: "newsletter",
    title: newsletterTitle(input.periodYear, input.periodMonth),
    summary: newsletterSummary(input.briefEnglishMd),
    body_md: input.briefEnglishMd,
    source_url: null,
    status: "published",
    published_at: input.publishedAt,
  };
}
