/** Strip control characters from a short single-line field. */
function cleanLine(v: unknown, max: number): string | null {
  if (typeof v !== "string") return null;
  // Written as \u escapes, NEVER as literal control bytes. The first cut of this
  // file embedded the raw range, which made the whole file BINARY TO GREP - so
  // our own source sweeps skipped it, and the security agent, shown a copy with
  // the control bytes stripped in transport, read the class as `[ --]`
  // (space-to-hyphen) and filed it as a malformed range. The regex was right and
  // the ENCODING was the defect. Same family as the NUL byte that hid two real
  // issues on 2026-09-15: a format our tools cannot read is a place facts hide.
  // eslint-disable-next-line no-control-regex -- stripping controls is the point
  const s = v.replace(/[\u0000-\u001F\u007F-\u009F]/g, " ").trim();
  return s ? s.slice(0, max) : null;
}
