/**
 * MAG own-bar / group-pool request emails (2026-11-27,
 * docs/mag-quote-automation-design.md §6): the quote (and its one reminder), and
 * "your bar is open". Same words as the WhatsApp templates
 * (scripts/submit-mag-request-templates.mjs). Factual UTILITY tone.
 */
import { renderEmail } from "./layout";
 
export interface MagQuoteEmailData {
  firstName?: string;
  /** "your own 20 g bar" / "your group pool (Tan family)". */
  what: string;
  /** The quote number, "MQ-BC21F52E". */
  ref?: string;
  /** The bar quoted, catalog name. */
  bar: string;
  /** One line per duration: "6 months — 4.75% over spot". */
  options: string[];
  /** "4 December 2026" (SGT). */
  validUntil: string;
  accountUrl: string;
  reminder?: boolean;
  /** "Buy it outright today: S$3,141.50 for the whole bar (3% over spot) …" — null when not on the shop. */
  outright?: string | null;
  /** Amounts are at today's spot; MAG grams are priced on the day each is bought. */
  disclaimer?: string | null;
  /** The reserved-bar buying-schedule rules are on (Agreement §1B): each option carries its schedule. */
  schedule?: boolean;
}
 
export function magQuoteEmail(d: MagQuoteEmailData): { subject: string; html: string; text: string } {
  const greeting = d.firstName ? `Hi ${d.firstName},` : "Hi,";
  const intro = d.reminder
    ? `A reminder: our quote for ${d.what} is valid until ${d.validUntil}.`
    : `Here is our quote for ${d.what}: a ${d.bar}, priced per gram over the live spot price.`;
  const how = `To answer, tap the button below and sign in with the email or WhatsApp number you use with us: your quote is at the top of your account, under "Your MAG requests". Choose the duration that suits you and tap Accept by ${d.validUntil}. ${d.schedule
    ? "Accepting reserves the bar for you with no deposit, and agrees to the buying schedule shown for that duration (MAG Agreement, Section 1B)."
    : "Accepting reserves the bar for you; there is no deposit and no obligation to buy."}`;
  const after = "Once you accept, we set the bar up and tell you as soon as you can start buying grams from it.";
  const list = d.options.map((o) => `<li style="margin: 0 0 4px;">${escapeHtml(o)}</li>`).join("");
  const outrightHtml = d.outright
    ? `<p style="margin: 0 0 12px;">${escapeHtml(d.outright)}</p><p style="margin: 0 0 8px;">Or buy it gram by gram with MAG:</p>`
    : "";
  const disclaimerHtml = d.disclaimer ? `<p style="margin: 0 0 16px; font-size: 12px; color: #8a7f66;">${escapeHtml(d.disclaimer)}</p>` : "";
  const content = `
    <p style="margin: 0 0 16px;">${escapeHtml(greeting)}</p>
    ${outrightHtml}
    <p style="margin: 0 0 12px;">${escapeHtml(intro)}</p>
    <ul style="margin: 0 0 16px; padding-left: 20px;">${list}</ul>
    ${disclaimerHtml}
    <p style="margin: 0 0 16px;">${escapeHtml(how)}</p>
    <p style="margin: 0;">${escapeHtml(after)}</p>
  `;
  return {
    subject: d.reminder
      ? `Reminder: your MAG quote${d.ref ? ` ${d.ref}` : ""} is valid until ${d.validUntil}`
      : `Your MAG quote${d.ref ? ` ${d.ref}` : ""} is ready`,
    html: renderEmail({ preheader: intro, content, cta: { label: "Open your quote →", url: d.accountUrl } }),
    text: [greeting, "", ...(d.outright ? [d.outright, "Or buy it gram by gram with MAG:", ""] : []), intro, ...d.options.map((o) => `• ${o}`),
      ...(d.disclaimer ? ["", d.disclaimer] : []), "", how, after, "", `Your account: ${d.accountUrl}`, "", "1Plus1Gold · Singapore"].join("\n"),
  };
}
 
export interface MagBarOpenEmailData {
  firstName?: string;
  /** "your own 20 g Valcambi bar" / "the Tan family group bar (20 g Valcambi)". */
  bar: string;
  premiumPct: string;
  closingDate: string;
  accountUrl: string;
  /** A group pool: the link the requester shares with their group. */
  shareUrl?: string | null;
  /** The bar is under the buying schedule (Agreement §1B). */
  schedule?: boolean;
}
 
export function magBarOpenEmail(d: MagBarOpenEmailData): { subject: string; html: string; text: string } {
  const greeting = d.firstName ? `Hi ${d.firstName},` : "Hi,";
  const what = `${capitalise(d.bar)} is now set up in MAG.`;
  const how = d.schedule
    ? `You can buy grams from it at ${d.premiumPct} over spot until ${d.closingDate}, keeping to the buying schedule shown in your account — the first 10% of the bar within 5 days.`
    : `You can buy grams from it at ${d.premiumPct} over spot, at your own pace, until ${d.closingDate}.`;
  const share = d.shareUrl
    ? "Share this link with the people in your group. Each of them signs in and taps Join, and can then buy their own grams from the bar:"
    : null;
  const content = `
    <p style="margin: 0 0 16px;">${escapeHtml(greeting)}</p>
    <p style="margin: 0 0 16px;">${escapeHtml(what)}</p>
    <p style="margin: 0;">${escapeHtml(how)}</p>
    ${share ? `<p style="margin: 16px 0 4px;">${escapeHtml(share)}</p><p style="margin: 0; word-break: break-all;"><a href="${escapeHtml(d.shareUrl!)}">${escapeHtml(d.shareUrl!)}</a></p>` : ""}
  `;
  return {
    subject: "Your MAG bar is open",
    html: renderEmail({ preheader: what, content, cta: { label: "Open your bar →", url: d.accountUrl } }),
    text: [greeting, "", what, how, ...(share ? ["", share, d.shareUrl!] : []), "", `Your account: ${d.accountUrl}`, "", "1Plus1Gold · Singapore"].join("\n"),
  };
}
 
export interface MagScheduleEmailData {
  firstName?: string;
  subject: string;
  /** The message, one paragraph per line (lib/mag/rhythm-copy.ts). */
  lines: string[];
  accountUrl: string;
}
 
/** A buying-schedule notice on a reserved / group bar (Agreement §1B). */
export function magScheduleEmail(d: MagScheduleEmailData): { subject: string; html: string; text: string } {
  const greeting = d.firstName ? `Hi ${d.firstName},` : "Hi,";
  const content = [greeting, ...d.lines].map((l, i, a) => `<p style="margin: 0 0 ${i === a.length - 1 ? 0 : 16}px;">${escapeHtml(l)}</p>`).join("\n");
  return {
    subject: d.subject,
    html: renderEmail({ preheader: d.lines[0] ?? d.subject, content, cta: { label: "See your bar →", url: d.accountUrl } }),
    text: [greeting, "", ...d.lines.flatMap((l) => [l, ""]), `Your account: ${d.accountUrl}`, "", "1Plus1Gold · Singapore"].join("\n"),
  };
}
 
function capitalise(s: string): string {
  return s.charAt(0).toUpperCase() + s.slice(1);
}
 
function escapeHtml(s: string): string {
  return s.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&#39;");
}
