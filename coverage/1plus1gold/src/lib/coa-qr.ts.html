/**
 * QR code for the Certificate of Authenticity.
 *
 * ▶ WHY THIS EXISTS. The printed CoA closes with "Verify this card any time at
 * 1plus1gold.com/verify/1G-XXXX-XXXX" — 42 characters the holder has to type
 * correctly. For a CURVE/split card that is only a convenience, because the
 * card carries an NFC tag they can tap. For a STANDARD bar it is the whole
 * story: standard bars carry no NFC tag (the certificate deliberately never
 * tells that holder to scan one), so the typed URL is the ONLY way they can
 * ever check their own certificate. On PROD that was 49 assigned cards at the
 * time this shipped. The QR is not a nicety for those holders; it is the
 * missing half of the feature.
 *
 * ▶ THE HOST IS PINNED TO PRODUCTION ON PURPOSE — do not "fix" this to use
 * SITE_URL. Everywhere else in the app an env-driven base URL is right,
 * because a link rendered on dev SHOULD point at dev. A certificate is
 * different in kind: it is printed onto paper, put in a customer's box, and
 * is then permanent and unfixable. If the CoA were ever printed from
 * dev.1plus1gold.com — and `?print` is reachable from any environment with an
 * admin session — an env-driven URL would bake a preview host into a
 * customer's permanent record of ownership, both in the QR and in the human
 * fallback beneath it. The asymmetry is total: pinning costs nothing, and
 * getting it wrong cannot be undone after the print.
 *
 * ▶ ERROR CORRECTION IS Q (25%), not the usual M. Paper gets folded, handled
 * and smudged, and this particular payload is short enough (~42 chars) that Q
 * still fits in a low version — 33x33 modules — so the code stays chunky and
 * easy to scan rather than dense. We buy real robustness for no legibility
 * cost, which is not usually the trade.
 *
 * ▶ THE QUIET ZONE COMES FROM THE LIBRARY, not from our CSS. `margin: 4` emits
 * the 4-module border the QR spec requires as part of the SVG itself, so
 * correctness does not depend on a padding value in a component someone may
 * later restyle. The plate padding around it is purely aesthetic.
 *
 * ▶ PROVEN END-TO-END, not just unit-tested (2026-09-14). The real markup was
 * rendered in Chromium on BOTH grounds — the dark on-screen card (#1A1810) and
 * the ivory print stock (#fffdf6) — screenshotted to pixels and decoded by an
 * independent decoder (jsQR). Both returned exactly
 * "https://1plus1gold.com/verify/1G-ZPM2-GET2". A round-trip test proves the
 * path matches the library's matrix; only a real renderer proves the markup
 * produces a scannable image, and that is the claim that matters for something
 * printed onto a customer's permanent record.
 * ▶ And the SIZE FLOOR was measured rather than assumed: it still decodes down
 * to 40 CSS px (~10.6 mm printed). The certificate renders it at 116 px
 * (~30.7 mm), so there is roughly a 3x margin — worth knowing before anyone
 * shrinks it, and evidence the current size is not a marginal call.
 */
import QRCode from "qrcode";
 
/**
 * The canonical production origin. Deliberately NOT `SITE_URL` — see the
 * header. A printed certificate always speaks for the production brand.
 */
const COA_ORIGIN = "https://1plus1gold.com";
 
/** The URL a holder lands on — what the QR encodes and what we print beneath it. */
export function coaVerifyUrl(serial: string): string {
  return `${COA_ORIGIN}/verify/${encodeURIComponent(serial)}`;
}
 
/** The same URL without its scheme, for the human-readable line on the card. */
export function coaVerifyUrlDisplay(serial: string): string {
  return coaVerifyUrl(serial).replace(/^https:\/\//, "");
}
 
/** Quiet zone in modules. 4 is the QR spec's minimum; below it, scanners fail. */
const QUIET_ZONE = 4;
 
export interface CoaQr {
  /** SVG path data for every dark module, in a `viewBox="0 0 size size"` space. */
  path: string;
  /** Side of the square viewBox, in modules — code width plus both quiet zones. */
  size: number;
}
 
/**
 * The certificate's QR as SVG PATH DATA, for the caller to render as real JSX.
 *
 * ▶ WHY A PATH AND NOT AN SVG STRING. The obvious implementation is
 * `QRCode.toString({type:"svg"})` piped into `dangerouslySetInnerHTML`. Do not
 * do that. The attacker playbook's F5 disposes every CSP-chain finding on the
 * standing fact that `src/` carries no XSS sink beyond JsonLd.tsx, whose
 * `application/ld+json` payload is non-executable. Adding an injection site to
 * sell a decorative square would invalidate a standing security disposition for
 * a page that does not need it. Returning geometry lets the caller render an
 * ordinary <svg><path/></svg>, which has no injection surface at all.
 * ▶ The inventory that premise depends on is now asserted by
 * `src/__tests__/dangerously-set-inner-html-inventory.test.ts`, not counted by
 * hand — when this file first said "exactly ONE", there were in fact two.
 *
 * SVG rather than a PNG data-URI for the same reason plus a better one: it is
 * vector, so it prints sharp at any size on any printer.
 *
 * Colour is applied by the caller, deliberately: near-black ink on a white
 * plate. NOT brand gold — tinting a QR is the classic way to trade scan
 * reliability for decoration, and a code nobody can scan fails at the one job
 * it has.
 */
export async function coaQr(serial: string): Promise<CoaQr> {
  const qr = QRCode.create(coaVerifyUrl(serial), { errorCorrectionLevel: "Q" });
  const n = qr.modules.size;
  const d = qr.modules.data;
 
  // Emit one path command per RUN of adjacent dark modules rather than one per
  // module — same rendering, materially smaller document, and it keeps the
  // printed certificate's markup from being mostly QR.
  const parts: string[] = [];
  for (let y = 0; y < n; y++) {
    let run = 0;
    for (let x = 0; x <= n; x++) {
      const dark = x < n && d[y * n + x] === 1;
      if (dark) { run++; continue; }
      if (run > 0) {
        parts.push(`M${x - run + QUIET_ZONE} ${y + QUIET_ZONE}h${run}v1h-${run}z`);
        run = 0;
      }
    }
  }
 
  return { path: parts.join(""), size: n + QUIET_ZONE * 2 };
}
