/**
 * Who is NOT a real customer, for sales and customer analytics.
 *
 * One fetch, three sets, so every report draws the line in the same place:
 *   • internal — `customers.is_internal`: our own, demo and test accounts.
 *   • supplier — `suppliers.customer_id`: a supplier's own customer account.
 *                BullionKing swap settlements are sales orders to it
 *                (docs/stonex-dual-sourcing-design.md §5: in revenue, but
 *                "excluded from customer analytics only").
 *   • house    — `channel_house_customers`: one synthetic owner per marketplace
 *                (Shopee, Lazada, in-person). Its orders ARE real sales, but one
 *                house row is many unrelated buyers, so it is excluded from any
 *                per-customer metric (repeat rate, cohorts).
 *
 * Callers decide which sets apply: sales counts drop internal + supplier;
 * per-customer metrics drop all three.
 */
import type { SupabaseClient } from "@supabase/supabase-js";
 
export interface SalesExclusions {
  internal: Set<string>;
  supplier: Set<string>;
  house:    Set<string>;
}
 
export async function fetchSalesExclusions(db: SupabaseClient): Promise<SalesExclusions> {
  const [internal, supplier, house] = await Promise.all([
    db.from("customers").select("id").eq("is_internal", true),
    db.from("suppliers").select("customer_id").not("customer_id", "is", null),
    db.from("channel_house_customers").select("customer_id"),
  ]);
  if (internal.error || supplier.error || house.error) throw new Error("sales_exclusions_fetch_failed");
  const ids = (rows: unknown[] | null, key: string) =>
    new Set((rows ?? []).map((r) => (r as Record<string, string>)[key]).filter(Boolean));
  return {
    internal: ids(internal.data, "id"),
    supplier: ids(supplier.data, "customer_id"),
    house:    ids(house.data, "customer_id"),
  };
}
 
/** Accounts whose orders are not customer sales (internal + supplier). */
export function nonSalesCustomerIds(x: SalesExclusions): Set<string> {
  return new Set([...x.internal, ...x.supplier]);
}
 
/** Accounts excluded from per-customer metrics (all three). */
export function nonPersonCustomerIds(x: SalesExclusions): Set<string> {
  return new Set([...x.internal, ...x.supplier, ...x.house]);
}
