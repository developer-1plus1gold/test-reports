/**
 * NFC / product serial — format: <TOKEN>-XXXX-XXXX
 *
 * TOKEN is a short weight token (1G, 5G, 10G, 100G, 1OZ, the 2.5g override 2G5,
 * …) — it conveys the bar's weight. Historically this was the fixed string "1G"
 * (only 1g splits existed); it now derives from the product's weight so a 5g or
 * 1oz bar doesn't carry a misleading "1G" (= "1 gram") prefix. Every legacy
 * "1G-…" serial stays valid — the format below is a strict superset.
 *
 * Body: 8 chars from a 30-char alphabet (uppercase A–Z + 2–9, minus the
 * visually ambiguous I, O, L, 1, 0). 30^8 ≈ 6.6 × 10¹¹ combinations per token.
 * The TOKEN may contain 0/1 (10G, 1OZ) — only the BODY uses the restricted
 * charset, so a dashless serial can always be split (last 8 = body).
 *
 * The DB CHECK constraint (nfc_serials_format) mirrors SERIAL_REGEX — keep the
 * two in sync (migration 20260630c_standard_serials_provenance.sql).
 */
import { randomBytes } from "node:crypto";
 
const ALPHABET = "ABCDEFGHJKMNPQRSTUVWXYZ23456789"; // length 30 (body charset)
const BODY = "[A-HJ-NP-Z2-9]{4}";
const SERIAL_REGEX = new RegExp(`^[0-9A-Z]{2,5}-${BODY}-${BODY}$`);
const TROY_OZ_G = 31.1035;
 
/**
 * Derive the weight token for a product. Integer grams → `<n>G`; troy ounces →
 * `<n>OZ`; fractional grams → the decimal point encoded as `G` (2.5 → `2G5`,
 * since `.` is illegal in a serial). The operator can override at generation
 * time for any odd case. Always uppercase, ≤5 chars.
 */
export function deriveWeightToken(grams: number, weightDisplay?: string | null): string {
  const wd = (weightDisplay ?? "").toLowerCase();
  if (Number.isInteger(grams) && grams > 0 && grams < 1000) return `${grams}G`;
  const oz = grams / TROY_OZ_G;
  if (wd.includes("oz") || (oz >= 0.99 && Math.abs(oz - Math.round(oz)) < 0.02)) {
    return `${Math.max(1, Math.round(oz))}OZ`;
  }
  return `${grams}`.replace(".", "G");
}
 
/** Is `token` a usable serial prefix (the chars the DB CHECK allows)? */
export function isValidWeightToken(token: string): boolean {
  return /^[0-9A-Z]{2,5}$/.test(token);
}
 
/** Generate a single serial under `prefix` (default "1G" for back-compat).
 *  Caller must check uniqueness against the DB. */
export function generateSerial(prefix = "1G"): string {
  // Rejection sampling → exactly-uniform alphabet distribution.
  const out: string[] = [];
  while (out.length < 8) {
    const buf = randomBytes(16);
    for (const byte of buf) {
      if (byte >= ALPHABET.length * Math.floor(256 / ALPHABET.length)) continue;
      out.push(ALPHABET[byte % ALPHABET.length]);
      if (out.length === 8) break;
    }
  }
  return `${prefix}-${out.slice(0, 4).join("")}-${out.slice(4, 8).join("")}`;
}
 
/** Generate a batch under `prefix` — caller inserts + handles collision retries. */
export function generateSerialBatch(n: number, prefix = "1G"): string[] {
  const seen = new Set<string>();
  while (seen.size < n) seen.add(generateSerial(prefix));
  return [...seen];
}
 
/** Format check — does NOT confirm existence in DB. */
export function isValidSerialFormat(serial: string): boolean {
  return SERIAL_REGEX.test(serial);
}
 
/** Normalise a user-typed OUR-format serial (verify lookup, curve pack entry):
 *  strip whitespace, uppercase, re-insert dashes if missing. The body is always
 *  the trailing 8 chars (restricted charset); everything before is the token, so
 *  a dashless serial of any token length is unambiguous. Returns null if it
 *  isn't one of our serials. (Manufacturer serials are NOT normalised here —
 *  they're arbitrary; the pack route looks those up raw.) */
export function normaliseSerial(input: string): string | null {
  const cleaned = input.replace(/\s+/g, "").toUpperCase();
  if (cleaned.includes("-")) {
    return isValidSerialFormat(cleaned) ? cleaned : null;
  }
  // Dashless: split off the trailing 8-char body; the rest is the token.
  if (cleaned.length < 10 || cleaned.length > 13) return null;
  const token = cleaned.slice(0, cleaned.length - 8);
  const body  = cleaned.slice(cleaned.length - 8);
  const formatted = `${token}-${body.slice(0, 4)}-${body.slice(4, 8)}`;
  return isValidSerialFormat(formatted) ? formatted : null;
}
 
/**
 * PostgREST `or()` filter for the admin serial-list search box.
 *
 * Operators search by whatever identifier is physically in front of them, so the
 * box matches OUR serial AND the MANUFACTURER serial — the same two identifiers
 * the packing-time assign route accepts. Before 2026-07-21 it matched only ours,
 * so a PAMP E-number or a Nadir SG00xxx returned nothing.
 *
 * Two normalisations, deliberately:
 *  • RAW input for a full pasted serial and for manufacturer serials. Stripping
 *    hyphens broke both — "%1G4VDBCS%" cannot match the stored "1G-4VDB-CSPD"
 *    (so pasting a full serial found NOTHING; only segment searches like "4VDB"
 *    worked), and a manufacturer serial can contain meaningful hyphens.
 *  • HYPHEN-LESS, uppercased, first 8 chars — preserves segment search.
 *
 * Commas and parens are stripped from values: PostgREST delimits or() with them,
 * so an unescaped one would break out of the filter expression.
 *
 * @returns the or() argument, or null when the input has no searchable content.
 */
export function buildSerialSearchFilter(search: string): string | null {
  const safe    = (s: string) => s.replace(/[,()]/g, "").trim();
  const raw     = safe(search).slice(0, 64);
  const cleaned = safe(search).replace(/[\s-]/g, "").toUpperCase().slice(0, 8);
 
  const parts: string[] = [];
  if (raw) parts.push(`serial.ilike.*${raw}*`);
  if (cleaned && cleaned !== raw.toUpperCase()) parts.push(`serial.ilike.*${cleaned}*`);
  if (raw) parts.push(`manufacturer_serial.ilike.*${raw}*`);
 
  return parts.length ? parts.join(",") : null;
}
