// ─── Per-bar metrics from live spot + selected qty (page.tsx:93-113) ────────────
 
export function barMetrics(bar: EligibleBar, spotToday: number | null, sellQty?: number): BarMetrics {
  // Default qty = whatever's still available for sale (post buyback/conversion
  // subtraction). totalCost denominator stays quantity_ordered because price_sgd
  // reflects what was paid for ALL originally-purchased units — the correct cost
  // basis to pro-rate.
  const qty        = sellQty ?? bar.quantity_available;
  const totalGrams = bar.grams_each * qty;
  const totalCost  = bar.price_sgd * (qty / bar.quantity_ordered);
  const sellPrice  = spotToday !== null
    ? parseFloat((spotToday * totalGrams * (1 + bar.buyback_premium_pct / 100)).toFixed(2))
    : null;
  const pnl      = sellPrice !== null ? sellPrice - totalCost : null;
  const pnlPct   = pnl !== null && totalCost > 0 ? (pnl / totalCost) * 100 : null;
  const spotChangePct = spotToday !== null && bar.spot_at_purchase
    ? ((spotToday - bar.spot_at_purchase) / bar.spot_at_purchase) * 100
    : null;
  const receivePerG = spotToday !== null
    ? spotToday * (1 + bar.buyback_premium_pct / 100)
    : null;
  return { sellPrice, pnl, pnlPct, spotChangePct, totalGrams, totalCost, receivePerG };
}
 
// ─── Effective premium at purchase (page.tsx:117-120) ──────────────────────────
 
export function effectivePurchasePremium(bar: EligibleBar): number | null {
  if (bar.cost_per_g === null || bar.spot_at_purchase === null || bar.spot_at_purchase === 0) return null;
  return ((bar.cost_per_g - bar.spot_at_purchase) / bar.spot_at_purchase) * 100;
}

export function computeTotals(
  selectedBars: EligibleBar[],
  selectedQuantities: Record<string, number>,
  spotSgd: number | null,
): BuybackTotals {
  let grams = 0;
  let payout: number | null = 0;
  for (const bar of selectedBars) {
    const qty = selectedQuantities[bar.order_item_id] ?? bar.quantity_available;
    const { totalGrams, sellPrice } = barMetrics(bar, spotSgd, qty);
    grams += totalGrams;
    if (payout !== null && sellPrice !== null) payout += sellPrice;
    else payout = null;
  }
  return { grams, payout };
}
