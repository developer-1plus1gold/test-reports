/**
 * Credential seals — ONE registry, read by both surfaces that show them.
 *
 * ▶ WHY THIS EXISTS. Adding a credential used to be a code task with real
 * research attached: edit the web component, edit the SDUI band, work out how
 * the issuer serves the badge, and check whether the CSP permits it. Four
 * places to get right, two of which fail SILENTLY — a `frame-src` miss renders
 * nothing with no error, and a seal added to one surface and not the other is
 * exactly the drift that left the whole band desktop-only for three months.
 * A credential arrives maybe twice a year, so that knowledge is always stale by
 * the time it is next needed. Here it is written down once and the adding is a
 * row.
 *
 * ▶ THE STANDARD SHAPE IS `image + href`, AND THAT CHOICE IS THE WHOLE POINT.
 * It is the one shape that needs NO `next.config.ts` change:
 *
 *     img-src     'self' data: blob: https://*.supabase.co www.combibar.com
 *     frame-src   'self' js.stripe.com *.dnb.com *.dunsregistered.com
 *     form-action 'self' profiles.dunsregistered.com
 *
 * A same-origin image is already permitted and a plain link is not governed by
 * CSP at all. An <iframe> from a new host is not, and neither is a <script>
 * badge — either of those is a CSP edit, i.e. a code change, i.e. the thing
 * this registry exists to avoid. So we HOST the artwork (issuers licence the
 * mark; that is what a licence is for) and link to their page.
 *
 * ▶ WHAT ADDING A SEAL COSTS, precisely, so the claim is checkable:
 *   1. Put the artwork at `public/images/seals/<id>.webp` (WebP is the house
 *      standard for customer-facing images — see CLAUDE.md).
 *   2. Add a row below and set `live: true`.
 *   3. Merge. The web footer and the native Home band both pick it up.
 *
 * ▶ AND IT IS NEVER AN APP STORE BUILD. The native band is server-authored
 * (`trustBandSection()` in src/app/spec.ts, served over /api/spec/*), and it
 * renders every seal with node types already in every shipped binary —
 * `button` + `image`, the same two the D&B seal and the social marks have used
 * since 2026-09-10. So a seal added here reaches installed apps on their next
 * spec fetch, with no submission and no 24-48h Apple review. That property is
 * what let iOS 1.0.6 ship before the SME500 artwork existed.
 *
 * ▶ THE ONE EXCEPTION, DECLARED RATHER THAN IMPLIED: D&B on the web keeps its
 * live <iframe> (`webEmbed`). It is animated, genuinely interactive, it
 * authenticates by DOMAIN, and it works — replacing a working surface with a
 * flat image to tidy the model would be a regression. Native cannot host it
 * (their profile is POST-only; see @/lib/dnb-seal and /dnb-profile), so native
 * uses the standard shape via our own composite route. `webEmbed` is therefore
 * "this seal renders differently on the web", not "seals may be iframes".
 *
 * ⚠️ BEFORE ADDING A ROW, CHECK ONE ASSUMPTION rather than inheriting it. An
 * award or accreditation mark is almost always a licensed logo plus a link to
 * a verification page, which is the standard shape. If an issuer instead
 * requires an embedded widget, that is a CSP change + a new `webEmbed` + a
 * native fallback — a real piece of work, not a row. Read their embed code;
 * do not assume it matches D&B's, and do not assume it matches this one's.
 */
import { DNB_SEAL_SANDBOX, DNB_SEAL_SRC } from "@/lib/dnb-seal";
 
export type TrustSeal = {
  /** Stable id. Also the artwork basename under public/images/seals/. */
  id: string;
  /** Alt text on the web, `accessibility_label` natively. SwiftUI reads
   *  nothing off a remote image, so without this the native control is
   *  announced as an unlabelled button. */
  label: string;
  /** Absolute destination. Absolute because the native opener
   *  (`UIApplication.shared.open`) has no origin to resolve a relative path
   *  against. */
  href: string;
  /** SAME-ORIGIN artwork path — `img-src 'self'`. A hotlinked issuer URL is
   *  blocked on the web and would be a CSP edit to permit. */
  image: string;
  /** The DISPLAY box in CSS px / native points — not necessarily the file's
   *  pixel size; see `assetScale`. The aspect must match the artwork, because a
   *  fixed native frame at the wrong ratio distorts the mark, and a credential
   *  rendered wrong is worse than one absent. Read the file rather than guess:
   *    node -e "require('sharp')('public/images/seals/<id>.webp').metadata().then(m=>console.log(m.width,m.height))" */
  width: number;
  height: number;
  /**
   * Pixels per display unit in the FILE. Default 1 (file is exactly
   * width x height). Set 2 to ship a retina asset: the file is 2x and the
   * renderer draws it into the 1x box, which is all `@2x` has ever meant and
   * needs no srcset machinery on either surface.
   *
   * ⚠️ NEVER upscale the source to reach a scale — enlarging a raster IS an
   * alteration of the mark. Only declare 2 when the licensed original is at
   * least twice the display box, which for a modern kit it invariably is
   * (SME500's is 1933px wide against a 179px box). The registry guard enforces
   * both halves: file == display x scale, and file >= display.
   */
  assetScale?: 1 | 2;
  /** false = the row is authored but the artwork is not in hand yet. Rendered
   *  by nothing until it flips, so an incomplete seal cannot half-ship as a
   *  broken image. Flip it in the SAME commit that adds the artwork — that way
   *  the two can never land apart. */
  live: boolean;
  /**
   * WEB-ONLY override: render the issuer's own live embed instead of our
   * hosted image. The D&B exception above. Requires the host in `frame-src`.
   * Native ignores this entirely and uses `image` + `href`.
   */
  webEmbed?: {
    src: string;
    sandbox: string;
    /** The seal authenticates by domain, so `no-referrer` breaks it. */
    referrerPolicy: "strict-origin";
  };
};

/** The seals that actually render. Both surfaces call this, never TRUST_SEALS,
 *  so a pending row cannot reach a customer from either one. */
export function activeTrustSeals(): TrustSeal[] {
  return TRUST_SEALS.filter((s) => s.live);
}
