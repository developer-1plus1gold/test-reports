/**
 * Send the delivered-order documents PDF to the customer (Mukesh 2026-11-27): the
 * receipt, packing list and every Certificate of Authenticity — the paper in the
 * parcel, delivered digitally by WhatsApp + email once the order is delivered.
 *
 * Driven by ONE job (/api/cron/order-documents) rather than by each of the ~dozen
 * code paths that can mark an order delivered (courier webhooks + polls, pickup,
 * Grab, marketplace sync…): the database stamps `lifecycle_completed_at` whichever
 * path did it, and the job sends for every delivered order whose documents are due.
 */
import type { SupabaseClient } from "@supabase/supabase-js";
import { isProd } from "@/lib/env";
import { generatePdfToken } from "@/lib/pdf-token";
import { deliveryOutcome } from "@/lib/notifications/delivery-outcome";
import { notifyOrderDocuments, type NotifyResult } from "@/lib/notifications/notify-customer";
import { loadOrderDocuments, renderOrderDocumentsPdf } from "./order-documents";
 
/**
 * The master switch. ON since 2026-11-27 — Mukesh approved the documents' look
 * (D03B9321) and said "switch it on". Set false to pause: the cron then sends and
 * marks nothing, so orders delivered while paused are still due (within
 * DOCUMENTS_SEND_WINDOW_MS) when it is switched back on.
 */
export const DOCUMENTS_DIGITAL_ENABLED = true;
 
/** Only orders delivered from this moment on — the feature never back-sends history. */
export const DOCUMENTS_DIGITAL_FROM = "2026-11-27T04:00:00Z";
/** A delivered order is sent within this window, or not at all. */
export const DOCUMENTS_SEND_WINDOW_MS = 3 * 24 * 60 * 60 * 1000;
/** Failed tries before the job gives up on an order. */
export const DOCUMENTS_MAX_ATTEMPTS = 3;
/** The email's download link outlives the WhatsApp fetch by weeks (receipt precedent). */
const LINK_TTL_S = 60 * 60 * 24 * 90;
 
/** Gold that stays with us is not a parcel — no documents to send. */
export const NO_PARCEL_SHIPPERS: ReadonlySet<string> = new Set(["vault"]);
 
export function publicOrigin(prod: boolean = isProd()): string {
  return prod ? "https://1plus1gold.com" : "https://dev.1plus1gold.com";
}
 
export type DocumentsSendResult =
  | { status: "delivered" | "unreachable"; channels: NotifyResult }
  /** Not finished: try again on a later run (a provider failed, or the WhatsApp
   *  template is still awaiting Meta approval for a customer with no email). */
  | { status: "retry"; reason: string }
  | { status: "skipped"; reason: string };
 
export async function sendOrderDocuments(
  supabase: SupabaseClient,
  order: { id: string; customer_id: string },
): Promise<DocumentsSendResult> {
  const docs = await loadOrderDocuments(supabase, order.id);
  if (!docs) return { status: "skipped", reason: "no_documents" };
 
  const pdf = await renderOrderDocumentsPdf(docs, "screen");
  const orderRef = order.id.slice(0, 8).toUpperCase();
  const token = generatePdfToken({ userId: order.customer_id, resource: "order_documents", resourceId: order.id }, LINK_TTL_S);
  const mediaUrl = `${publicOrigin()}/api/orders/documents-pdf/${order.id}.pdf?t=${token}`;
 
  const channels = await notifyOrderDocuments(
    order.customer_id, orderRef, order.id, mediaUrl, token, pdf.toString("base64"), docs.coas.length,
  );
  const outcome = deliveryOutcome(channels);
  if (outcome === "failed") return { status: "retry", reason: `${channels.whatsapp.reason ?? ""} / ${channels.email.reason ?? ""}` };
  // WhatsApp-only customer while the template awaits approval: nothing went out,
  // but approval will change that — keep it due rather than closing it.
  if (outcome === "unreachable" && channels.whatsapp.reason === "no_template") return { status: "retry", reason: "template_pending" };
  return { status: outcome, channels };
}
