/**
 * ORDER DOCUMENTS — one PDF holding everything that goes in the parcel (Mukesh
 * 2026-11-27): the receipt, the packing list, and one Certificate of Authenticity
 * per card, in that order.
 *
 *   • Dispatch prints it with ONE button (admin route, theme "print": the receipt's
 *     ink-light look) — plain A4, actual size, no settings.
 *   • The customer receives the SAME file digitally once the order is delivered —
 *     the paper and the digital copy can never disagree because they are one build.
 *
 * Built from the same sources as each document on its own: the receipt from
 * lib/receipt/load-receipt-data (the receipt route's assembly), the list from the
 * packing record (buildSlipExtras — supplies never listed), each certificate from
 * lookupSerial (the /verify page's loader) and coaQr.
 */

/** One certificate page's data — the same rows, in the same order, as /verify. */
export async function loadCoaPage(supabase: SupabaseClient, serial: string): Promise<CoaPageData | null> {
  const v = await lookupSerial(serial, supabase);
  if (!v) return null;
  const [qr, photo] = await Promise.all([coaQr(v.serial), photoDataUri(v.image_url)]);
  const vis = giftBlockVisibility({ hasOwner: !!v.current_owner_customer_id, viewerIsOwner: false, isOutstandingGift: v.is_pending_gift, isPrint: true });
  return {
    serial:    v.serial,
    verifyUrl: coaVerifyUrl(v.serial),
    photo,
    qr,
    rows: [
      { label: "Serial Number", value: v.serial },
      { label: "Packed by",     value: "1Plus1Gold" },
      { label: "Content",       value: v.product_label },
      { label: "Purity",        value: fmtFineness(v.fineness) },
      ...(v.product_dimensions ? [{ label: "Dimensions", value: v.product_dimensions }] : []),
      { label: "Source",        value: v.source_label },
      { label: "Date Packed",   value: fmtDate(v.assigned_at) },
    ],
    hasNfc: v.product_type !== "standard",
    gift: vis.show
      ? { from: v.current_owner_first_name, message: vis.showNote ? v.gift_message : null, weightPhrase: goldWeightPhrase(v.product_grams) }
      : null,
  };
}

/** The single PDF: receipt, packing list, one certificate per card. */
export async function renderOrderDocumentsPdf(docs: OrderDocuments, theme: ReceiptTheme): Promise<Buffer> {
  const ref = docs.orderId.slice(0, 8).toUpperCase();
  return renderToBuffer(
    <Document title={`1Plus1Gold order documents — ${ref}`}>
      <ReceiptPage receipt={docs.receipt} theme={theme} />
      <PackingListPage data={docs.packingList} />
      {docs.coas.map((c) => <CoaPage key={c.serial} data={c} />)}
    </Document>,
  );
}
