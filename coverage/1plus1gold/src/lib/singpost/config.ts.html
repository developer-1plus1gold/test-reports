/**
 * SingPost API environment + credentials (2026-11-21). Env only — never stored,
 * never committed, never returned by any route.
 *
 * Env (Vercel DEV = sandbox, PROD = production):
 *   SINGPOST_ENV                "production" for live; anything else → sandbox
 *                               (fails closed: a missing or mistyped value never hits live)
 *   SINGPOST_BOOKING_API_KEY    ezy2ship `x-api-key` header
 *   SINGPOST_BOOKING_TICKET     ezy2ship `<Ticket>` in every SOAP body
 *   SINGPOST_ACCOUNT_NUMBER     ezy2ship `AccountNumber`
 *   SINGPOST_TRACKING_API_KEY   Tracking API `x-api-key` (UAT and production keys differ)
 *   SINGPOST_CONTACT_EMAIL      our ops inbox — sender AND recipient email on the
 *                               booking (as GrabExpress does), so SingPost never
 *                               emails a customer from our account
 * Optional:
 *   SINGPOST_SOAP_NAMESPACE     the ezy2ship namespace / SOAPAction base. The
 *                               guide (v1.0.3) shows the same staging URI for both
 *                               environments; override if SingPost gives another.
 *
 * The sender phone is our dispatch phone, shared with every courier
 * (lib/shipping/shop-origin.ts).
 */
import "server-only";
 
export type SingpostEnv = "sandbox" | "production";
 
export function singpostEnv(): SingpostEnv {
  return process.env.SINGPOST_ENV?.trim().toLowerCase() === "production" ? "production" : "sandbox";
}
 
/** ezy2ship Shipment Booking API v1.0.3. */
export const SINGPOST_BOOKING_URL: Record<SingpostEnv, string> = {
  sandbox:    "https://api.qa.cce.singpost.com/ezy2shipWS/v1",
  production: "https://api.cce.singpost.com/ezy2shipWS/v1",
};
 
/** Tracking API v3.0. */
export const SINGPOST_TRACKING_URL: Record<SingpostEnv, string> = {
  sandbox:    "https://apispnet-uat.singpost.com/tnt/v1_0/customer/trackitems/",
  production: "https://apispnet.singpost.com/tnt/v1_0/customer/trackitems/",
};
 
/** The ezy2ship SOAP namespace, as printed in the guide's every example. Not a URL we call. */
export const DEFAULT_SINGPOST_SOAP_NAMESPACE = "https://singpostuat2-staging.oss.neopost-id.com/modules/oss/api/ship/sz_Client";
 
function env(name: string): string | undefined {
  const v = process.env[name]?.trim();
  return v ? v : undefined;
}
 
export interface SingpostBookingCreds {
  apiKey: string;
  ticket: string;
  accountNumber: string;
  contactEmail: string;
}
 
/** True when the booking calls can be made. Callers branch on this rather than catching a throw. */
export function isSingpostBookingConfigured(): boolean {
  return !!(env("SINGPOST_BOOKING_API_KEY") && env("SINGPOST_BOOKING_TICKET") && env("SINGPOST_ACCOUNT_NUMBER") && env("SINGPOST_CONTACT_EMAIL"));
}
 
/** True when the tracking poll can run. */
export function isSingpostTrackingConfigured(): boolean {
  return !!env("SINGPOST_TRACKING_API_KEY");
}
 
export function singpostBookingCreds(): SingpostBookingCreds {
  const apiKey = env("SINGPOST_BOOKING_API_KEY");
  const ticket = env("SINGPOST_BOOKING_TICKET");
  const accountNumber = env("SINGPOST_ACCOUNT_NUMBER");
  const contactEmail = env("SINGPOST_CONTACT_EMAIL");
  if (!apiKey || !ticket || !accountNumber || !contactEmail) {
    throw new Error("SingPost booking is not configured (SINGPOST_BOOKING_API_KEY / SINGPOST_BOOKING_TICKET / SINGPOST_ACCOUNT_NUMBER / SINGPOST_CONTACT_EMAIL)");
  }
  return { apiKey, ticket, accountNumber, contactEmail };
}
 
export function singpostTrackingKey(): string {
  const key = env("SINGPOST_TRACKING_API_KEY");
  if (!key) throw new Error("SingPost tracking is not configured (SINGPOST_TRACKING_API_KEY)");
  return key;
}
 
export function singpostSoapNamespace(): string {
  return env("SINGPOST_SOAP_NAMESPACE") ?? DEFAULT_SINGPOST_SOAP_NAMESPACE;
}
