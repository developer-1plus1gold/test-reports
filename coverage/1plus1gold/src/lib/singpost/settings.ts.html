/**
 * SingPost (Speedpost) courier settings — what can change without a deploy
 * (2026-11-21). Pure: shared by the Admin → Shippers panel, the settings route
 * and the booking, so the three can never disagree about what is valid.
 *
 * Stored in `shipper_settings.singpost_config` (jsonb, row 'singpost'; migration
 * 20261121_singpost_courier.sql). Secrets are NOT here — API keys, the Ticket
 * and the account number are env vars (lib/singpost/config.ts). That row is
 * readable by anyone while it is enabled ("Public read enabled shippers"), so
 * nothing in this config may be more private than what SingPost prints on every
 * parcel label: codes, the collection choice, parcel defaults and our sender
 * address. Our contact phone and email come from env for the same reason.
 *
 * Every field has a value SingPost has to give us (carrier / service codes,
 * collection type) or a sensible default (parcel size, sender address). Until
 * the SingPost-given ones are set, `missingSettings` names them and booking is
 * refused — the Dispatch button does not even show.
 */
import { z } from "zod";
import { SINGPOST_LANE_SERVICES, type SingpostLaneService } from "@/lib/shipping/lanes";
 
export const SINGPOST_SHIPPER = "singpost";
 
/**
 * The SingPost services Dispatch can book. Keys are ours; the codes are SingPost's.
 * Prime + Priority from 2026-11-20; Standard + Saver added 2026-11-22 (Mukesh: every
 * SingPost lane must be bookable, not only the two Lalamove replacements).
 */
export const SINGPOST_SERVICES = ["PRIME", "PRIORITY", "STANDARD", "SAVER"] as const;
export type SingpostService = (typeof SINGPOST_SERVICES)[number];
 
export const SINGPOST_SERVICE_LABEL: Record<SingpostService, string> = {
  PRIME:    "Speedpost Prime",
  PRIORITY: "Speedpost Priority",
  STANDARD: "Speedpost Standard",
  SAVER:    "Speedpost Saver",
};
 
/** SingPost's rate codes from Lisa Then's offer (2026-11-20) — for reference in
 *  the admin panel only; the API ServiceCode is what SingPost assigns with the account. */
export const SINGPOST_RATE_CODE: Partial<Record<SingpostService, string>> = { PRIME: "SAMPU1", PRIORITY: "PSDPU1" };
 
export function isSingpostService(v: unknown): v is SingpostService {
  return typeof v === "string" && (SINGPOST_SERVICES as readonly string[]).includes(v);
}
 
/**
 * First-mile options assignShipments accepts that need nothing per parcel.
 * POPSTATION (we drop at a kiosk) needs a kiosk code per call and is not
 * offered: Mukesh wants one fixed collection a day.
 */
export const SINGPOST_COLLECTION_TYPES = ["FIXED_SCHEDULED", "COLLECTION_REQUEST", "SELF_LODGE"] as const;
export type SingpostCollectionType = (typeof SINGPOST_COLLECTION_TYPES)[number];
 
export const SINGPOST_COLLECTION_LABEL: Record<SingpostCollectionType, string> = {
  FIXED_SCHEDULED:    "Fixed daily collection",
  COLLECTION_REQUEST: "Collection on request (date + window)",
  SELF_LODGE:         "We drop it at SingPost (self lodge)",
};
 
const HHMM = /^([01]\d|2[0-3]):[0-5]\d$/;
// Codes are SingPost's: letters, digits, underscore or hyphen, within the spec's lengths.
const code = (max: number) => z.string().trim().regex(new RegExp(`^[A-Za-z0-9_-]{1,${max}}$`), `1–${max} letters or digits`);
// Address text: printable, within SingPost's 35-character line limit.
// eslint-disable-next-line no-control-regex
const line = z.string().trim().max(35, "at most 35 characters").regex(/^[^<>\u0000-\u001F]*$/, "no control characters or < >");
 
export const singpostConfigSchema = z.object({
  carrierCode: code(3).nullable(),
  serviceCodes: z.object({
    PRIME:    code(10).nullable(),
    PRIORITY: code(10).nullable(),
    // Added 2026-11-22. Defaulted so a config saved before them still reads — a strict
    // object would otherwise reject it and reset the Prime/Priority codes to blank.
    STANDARD: code(10).nullable().default(null),
    SAVER:    code(10).nullable().default(null),
  }).strict(),
  collectionType: z.enum(SINGPOST_COLLECTION_TYPES).nullable(),
  /** Only for COLLECTION_REQUEST: the window asked for, SGT, on the booking day. */
  collectionWindow: z.object({
    from: z.string().regex(HHMM, "HH:MM"),
    to:   z.string().regex(HHMM, "HH:MM"),
  }).strict().refine((w) => w.from < w.to, "the window must end after it starts").nullable(),
  /** The box we ship in. SingPost charges the higher of actual and L×W×H/5000. */
  parcel: z.object({
    weightKg: z.number().gt(0).max(30),
    lengthCm: z.number().gt(0).max(200),
    widthCm:  z.number().gt(0).max(200),
    heightCm: z.number().gt(0).max(200),
  }).strict(),
  /**
   * SingPost's booking cut-off per service and weekday (index 0 = Sunday; null =
   * no booking that day), from its service pages. Not used to book — only to
   * WARN in Admin → Shippers → Delivery lanes when a lane's pay-by time leaves no
   * time to book (lib/shipping/lane-settings.ts). Keys = the lane services.
   */
  bookingCutoffs: z.object(Object.fromEntries(SINGPOST_LANE_SERVICES.map((k) => [k,
    z.array(z.string().regex(HHMM, "HH:MM").nullable()).length(7)])) as Record<SingpostLaneService, z.ZodArray<z.ZodNullable<z.ZodString>>>).strict(),
  /** Printed on the label. Keep it low-profile — never "gold" (carrier-sender-anonymity). */
  sender: z.object({
    name:     line.min(1, "required"),
    line1:    line.min(1, "required"),
    line2:    line,
    line3:    line,
    postcode: z.string().trim().regex(/^\d{6}$/, "6 digits"),
  }).strict(),
}).strict();
 
export type SingpostConfig = z.infer<typeof singpostConfigSchema>;

/**
 * The stored config with defaults filled in. Anything that does not validate
 * falls back to the default for that section — a bad row must never book a
 * parcel with half a sender address; the panel then shows the defaults and the
 * admin re-saves.
 */
export function resolveSingpostConfig(raw: unknown): SingpostConfig {
  const obj = raw && typeof raw === "object" && !Array.isArray(raw) ? (raw as Record<string, unknown>) : {};
  const merged: Record<string, unknown> = { ...DEFAULT_SINGPOST_CONFIG };
  for (const key of Object.keys(DEFAULT_SINGPOST_CONFIG) as Array<keyof SingpostConfig>) {
    if (!(key in obj)) continue;
    const part = singpostConfigSchema.shape[key].safeParse(obj[key]);
    if (part.success) merged[key] = part.data;
  }
  return singpostConfigSchema.parse(merged);
}
 
/** Validate an admin edit. Returns the config or every problem, field by field. */
export function parseSingpostConfig(raw: unknown):
  | { ok: true; config: SingpostConfig }
  | { ok: false; errors: string[] } {
  const r = singpostConfigSchema.safeParse(raw);
  if (r.success) {
    if (r.data.collectionType === "COLLECTION_REQUEST" && !r.data.collectionWindow) {
      return { ok: false, errors: ["collectionWindow: required for a collection on request"] };
    }
    return { ok: true, config: r.data };
  }
  return { ok: false, errors: r.error.issues.map((i) => `${i.path.join(".") || "config"}: ${i.message}`) };
}
 
/**
 * What is still missing before `service` can be booked — empty when it can.
 * Read by the status route (is the Dispatch button shown?) and the booking.
 */
export function missingSettings(c: SingpostConfig, service?: SingpostService): string[] {
  const out: string[] = [];
  if (!c.carrierCode) out.push("carrier code");
  const services = service ? [service] : SINGPOST_SERVICES;
  for (const s of services) if (!c.serviceCodes[s]) out.push(`${SINGPOST_SERVICE_LABEL[s]} service code`);
  if (!c.collectionType) out.push("collection type");
  if (c.collectionType === "COLLECTION_REQUEST" && !c.collectionWindow) out.push("collection window");
  return out;
}
 
/** The services that can be booked with this config. */
export function bookableServices(c: SingpostConfig): SingpostService[] {
  return SINGPOST_SERVICES.filter((s) => missingSettings(c, s).length === 0);
}
