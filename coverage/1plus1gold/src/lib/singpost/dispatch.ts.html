/**
 * SingPost on the Dispatch screen — which orders may be booked with it, and
 * where a SingPost order is listed (2026-11-21). Pure and client-safe: the
 * Dispatch page and the booking route share these, so the button can never
 * offer what the server refuses.
 *
 * ▶ HOW AN ORDER BECOMES A SINGPOST ORDER. SingPost is not a checkout option
 * yet (that, and what customers are promised, is Mukesh's decision once
 * SingPost answers). The operator books an existing delivery order with it
 * from Dispatch: "Book with SingPost" on an unbooked Lalamove or SPX order. The
 * booking moves the order to shipper 'singpost' in the same write that saves
 * the tracking number — the order is never half-moved. The customer's delivery
 * fee is never touched, as for every courier change.
 *
 * ▶ A BOOKING IS TWO SINGPOST CALLS. createShipment gives the tracking number;
 * assignShipments confirms it for collection. If the second fails, the order is
 * saved on SingPost with the tracking number, status 'failed' and the stage
 * "created" in its event, so the next tap only CONFIRMS it — creating a second
 * shipment would leave an orphan parcel record on SingPost's side.
 */
import { needsCourierBooking, isCourierDispatched } from "@/lib/admin/dispatch-buckets";
import { SINGPOST_SHIPPER, isSingpostService, type SingpostService } from "@/lib/singpost/settings";
 
/**
 * The SingPost service to book for an order, from what it was SOLD as
 * (orders.shipper_service). A delivery lane stores its SingPost service there
 * (lanes-server laneOrderFields): PRIME / STANDARD / SAVER as-is, and both
 * Priority lanes (same-day, next-day) book the Priority service. null = the order
 * names no SingPost service (a Lalamove / SPX order being moved), so Dispatch
 * offers the operator's choice. Without this, a Standard or Saver lane order fell
 * to the first bookable service and could have been booked as Prime.
 */
export function singpostServiceForOrder(shipperService: string | null | undefined): SingpostService | null {
  if (shipperService === "PRIORITY_SAMEDAY" || shipperService === "PRIORITY_NEXTDAY") return "PRIORITY";
  return isSingpostService(shipperService) ? shipperService : null;
}
 
/** Couriers whose unbooked orders can be moved to SingPost. NinjaVan is no longer used (2026-11-20). */
export const SINGPOST_SOURCE_SHIPPERS = ["lalamove", "spx"] as const;
 
const DEAD = new Set(["cancelled", "refunded"]);
 
export interface SingpostOrderState {
  status?: string | null;
  shipper?: string | null;
  shipper_status?: string | null;
  shipper_order_id?: string | null;
  source_channel?: string | null;
  delivery_address?: unknown;
  shipper_status_events?: unknown;
}
 
/** The stage a SingPost booking reached, recorded on its event. */
export type SingpostBookingStage = "created" | "assigned";
 
interface SingpostEventShape { singpost_reference?: unknown; singpost_stage?: unknown }
 
function events(o: SingpostOrderState): SingpostEventShape[] {
  return Array.isArray(o.shipper_status_events) ? (o.shipper_status_events as SingpostEventShape[]).filter((e) => e && typeof e === "object") : [];
}
 
/** The latest SingPost booking event on the order, or null. */
function lastSingpostEvent(o: SingpostOrderState): SingpostEventShape | null {
  const ev = events(o).filter((e) => typeof e.singpost_reference === "string");
  return ev.length ? ev[ev.length - 1] : null;
}
 
/** How many SingPost shipments this order has had — the next one's SenderReference suffix. */
export function singpostAttemptNumber(o: SingpostOrderState): number {
  // Distinct references: a "confirm for collection" event repeats its shipment's reference.
  return new Set(events(o).map((e) => e.singpost_reference).filter((r) => typeof r === "string")).size + 1;
}
 
/**
 * A SingPost shipment exists (tracking number saved) but was not confirmed for
 * collection: the next action is "Confirm for collection", not a new booking.
 */
export function needsSingpostConfirm(o: SingpostOrderState): boolean {
  return o.shipper === SINGPOST_SHIPPER && !!o.shipper_order_id && (o.shipper_status ?? "") === "failed"
    && lastSingpostEvent(o)?.singpost_stage === "created";
}
 
/** Why this order cannot be booked with SingPost now, or null when it can. */
export function singpostBookBlocker(o: SingpostOrderState): string | null {
  if (DEAD.has(o.status ?? "")) return `This order is ${o.status}`;
  if (o.source_channel && !["website", "web", "app"].includes(String(o.source_channel).toLowerCase())) {
    return "A marketplace order ships with the marketplace's own courier";
  }
  const st = o.shipper_status ?? "";
  if (st === "delivered") return "This order is already delivered";
  const addr = (o.delivery_address ?? null) as { line1?: unknown; postal_code?: unknown } | null;
  if (!addr || !addr.line1 || !addr.postal_code) return "This order has no delivery address";
  const live = !!o.shipper_order_id && st !== "failed";
  if (o.shipper === SINGPOST_SHIPPER) return live ? "This order already has a live SingPost booking" : null;
  if (!(SINGPOST_SOURCE_SHIPPERS as readonly string[]).includes(o.shipper ?? "")) {
    return `Only Lalamove or SPX delivery orders can be booked with SingPost (this one is "${o.shipper ?? "unset"}")`;
  }
  if (live) return "This order has a live courier booking — cancel it first (Cancel booking), then book with SingPost";
  return null;
}
 
/** Dispatch buckets for SingPost orders, on the shared rules (dispatch-buckets.ts). */
export function isSingpostAwaitingBooking(o: SingpostOrderState): boolean {
  return o.shipper === SINGPOST_SHIPPER && !DEAD.has(o.status ?? "") && needsCourierBooking(o);
}
export function isSingpostBooked(o: SingpostOrderState): boolean {
  return o.shipper === SINGPOST_SHIPPER && !DEAD.has(o.status ?? "") && isCourierDispatched(o);
}
