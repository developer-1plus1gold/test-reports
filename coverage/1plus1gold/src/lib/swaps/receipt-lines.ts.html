/**
 * A swapped order line on the receipt as THREE lines (Mukesh, 2026-11-09):
 *
 *   1 gram Split-Valcambi · Serial 1G-685P-3RGN                          S$ 204.90
 *   Returned: 1 gram Split-Valcambi · Serial 1G-685P-3RGN               −S$ 204.90
 *   Replacement: 1 gram Gold Bar - Nadir · Bar no. A910001 · Serial 1G-DT3B-8HQE  S$ 204.90
 *
 * Serials use the receipt's one format (lib/serials/bar-serial.ts): the refinery
 * bar number first when the bar has one (2026-11-11 — the swap lines had printed
 * the bare NFC serial only).
 *
 * The total is untouched — the two swap lines net to zero — and they only appear
 * once the returned bar is RECEIVED (SWAP_RECEIPT_STATUSES), which is also when
 * the order line itself is re-pointed at the replacement product. Line 1 therefore
 * prints from the swap's snapshot (what was sold), never from the order line's
 * current product.
 *
 * Applied by BOTH receipt writers (receipt-pdf + send-receipt) to their items
 * BEFORE the cache key is built, so the key changes with the swap and the two
 * writers keep computing the same key.
 */

/** The receipt-line fields this transform reads and writes. */
export interface SwappableLine {
  label:                string;
  serials?:             string | null;
  grams:                number;
  quantity:             number;
  price_sgd:            number;
  premium_applied_pct?: number | null;
  line_total_sgd?:      number | null;
}
 
/**
 * Expand each swapped line into its three receipt lines. `orderItemIds[i]` is
 * the order item of `items[i]`. Pure; input is not mutated.
 */
export function applySwapsToReceiptItems<T extends SwappableLine>(
  items: readonly T[],
  orderItemIds: readonly string[],
  swaps: ReadonlyMap<string, ReceiptSwap>,
): T[] {
  if (swaps.size === 0) return [...items];
  const out: T[] = [];
  items.forEach((item, i) => {
    const swap = swaps.get(orderItemIds[i] ?? "");
    if (!swap) { out.push(item); return; }
    const amount = item.line_total_sgd ?? item.price_sgd * item.quantity;
    out.push({ ...item, label: swap.returnedLabel, serials: swap.returnedSerials });
    out.push({
      ...item, label: `Returned: ${swap.returnedLabel}`, serials: swap.returnedSerials,
      quantity: 1, price_sgd: -amount, line_total_sgd: -amount, premium_applied_pct: null,
    });
    out.push({
      ...item, label: `Replacement: ${swap.replacementLabel}`, serials: swap.replacementSerial,
      quantity: 1, price_sgd: amount, line_total_sgd: amount, premium_applied_pct: null,
    });
  });
  return out;
}
