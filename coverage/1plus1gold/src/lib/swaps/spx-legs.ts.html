/**

 *   return   — customer's address → shop (sender = the customer, deliver = shopContact())
 *   outbound — shop → customer's address (the normal outbound shape)
 *
 * Booking mirrors /api/spx/book: 'pickup' takes SPX's earliest pickup slot,
 * 'dropoff' carries none; declaring a value is opt-in per booking (SPX refuses a
 * value without insurance collection — see createOrder). Polling mirrors
 * spx-status-poll and is called by it, so a swap converges without a webhook.

/** Columns of one leg, so both legs share one code path. */
function legColumns(leg: SwapLeg) {
  return leg === "return"
    ? { collect: "return_collect_type", tracking: "return_tracking_no", link: "return_share_link",
        status: "return_status", events: "return_status_events", bookedAt: "return_booked_at", error: "return_error" }
    : { collect: "outbound_collect_type", tracking: "outbound_tracking_no", link: "outbound_share_link",
        status: "outbound_status", events: "outbound_status_events", bookedAt: "outbound_booked_at", error: "outbound_error" };
}

export async function cancelSwapLeg(supabase: SupabaseClient, swap: OrderSwap, leg: SwapLeg): Promise<CancelLegResult> {
  const cols = legColumns(leg);
  const tracking = leg === "return" ? swap.return_tracking_no : swap.outbound_tracking_no;
  const status = leg === "return" ? swap.return_status : swap.outbound_status;
  if (!tracking) return { ok: false, status: 409, error: `No ${leg} parcel is booked.` };
  if (status !== "booked") return { ok: false, status: 409, error: `The ${leg} parcel is already ${status} — SPX only cancels before collection.` };
  try {
    await cancelOrder(tracking);
  } catch (err) {
    const msg = err instanceof Error ? err.message : "SPX cancel failed";
    console.error(`[swaps] cancel ${leg} ${tracking}:`, msg);
    return { ok: false, status: 502, error: "SPX would not cancel that parcel — it may already be collected." };
  }
  const { error } = await supabase.from("order_swaps").update({
    [cols.collect]: null, [cols.tracking]: null, [cols.link]: null, [cols.status]: null,
    [cols.bookedAt]: null, updated_at: new Date().toISOString(),
    // A cancelled return parcel's label is dead — a rebooked one is sent afresh.
    ...(leg === "return" ? { return_label_sent_at: null, return_label_sent_channels: null } : {}),
    status: leg === "return" ? "awaiting_return" : "received",
  }).eq("id", swap.id);
  if (error) return { ok: false, status: 500, error: "Cancelled on SPX, but saving that failed — refresh." };
  return { ok: true };
}
