/**

 * replacement over SPX (Mukesh, 2026-11-09; migration 20261107e_order_swaps).
 *
 * Lifecycle (order_swaps.status):
 *   awaiting_return → return_booked → received → dispatched → completed
 *   (cancelled only before the returned bar is received)
 *
 * No money moves: the order stays the sale, there is no swap or shipping fee,
 * and the receipt shows the swap as three lines (lib/swaps/receipt-lines.ts).
 * The stock / COGS / order-line / NFC re-pointing happens in ONE SQL transaction
 * (order_swap_receive) when the returned bar is received.

 * This file holds the pure rules; the SPX legs are in lib/swaps/spx-legs.ts.

/** Why a swap cannot be created, or null when it can. Every rule is checked
 *  before a row is written, so a refused swap leaves nothing behind. */
export function swapIneligibility(i: SwapEligibilityInput): string | null {
  if (i.orderPaymentStatus !== "paid") return "Only a paid order can be swapped.";
  if (i.orderStatus === "cancelled" || i.orderStatus === "refunded") return "This order is cancelled.";
  if (i.hasOpenSwap) return "This item already has an open swap — finish or cancel it first.";
  if (i.lineQuantity !== 1) return "Only a single-bar line can be swapped (this line has more than one bar).";
  Iif (!i.lineProductId) return "This order line has no product.";
  if (i.lineProductId === i.replacementProductId) return "Pick a different product for the replacement.";
  if (!(i.lineGrams > 0)) return "Only a gold bar line can be swapped.";
  if (Math.abs(i.replacementGrams - i.lineGrams) > 1e-9) {
    return `The replacement must weigh the same (${i.lineGrams}g) — a swap is bar for bar at the same price.`;
  }
  if (i.replacementAvailable < 1) return "No bar of the replacement product is in stock.";
  const missing = addressMissing(i.address);
  return missing ? `The order's delivery address is incomplete (${missing}) — fix it before booking the return.` : null;
}

/**
 * SPX `order_id` for one leg: alphanumeric, ≤ 32 chars, DETERMINISTIC (a retried
 * booking sends the same id → SPX 13103 duplicate, never a second parcel), and
 * disjoint from order ids (those are 32 bare hex chars; these start "sw").
 */
export function swapSpxOrderId(swapId: string, leg: SwapLeg): string {
  const hex = swapId.replace(/[^a-fA-F0-9]/g, "").toLowerCase().slice(0, 29);
  return `sw${leg === "return" ? "r" : "o"}${hex}`;
}
 
/** Street line SPX prints — line1 + line2, as the outbound booking does. */

/**
 * The swap status after a leg's courier status changes. Only the outbound leg's
 * delivery finishes a swap; the return leg's delivery does NOT mark it received
 * — the bar is inspected first and "Received" is an admin step.
 */
export function swapStatusAfterLeg(current: SwapStatus, leg: SwapLeg, courierStatus: string): SwapStatus {
  if (leg === "outbound" && courierStatus === "delivered" && current === "dispatched") return "completed";
  return current;
}

  returnedCostPerG:    number | null;
  replacementCostPerG: number | null;
  grams:               number;
  returnFeeSgd:        number | null;
  outboundFeeSgd:      number | null;
}
 
export interface SwapCost {
  /** Replacement bar cost minus the returned bar's — what the higher-cost bar costs us (can be negative). */
  goldDifferenceSgd: number | null;
  /** SPX cost of both parcels, as entered (null legs count as unknown, not zero). */
  shippingSgd:       number;
  shippingComplete:  boolean;
  totalSgd:          number | null;
}
 
/**
 * The swap's cost to the business (no fee is charged to the customer). Display
 * only: the P&L already carries both parts on its own — the gold difference
 * through COGS (order_swap_receive re-points the order at the replacement bar)
 * and the SPX parcels through the SPX invoice logged in the ledger — so this is
 * never booked as a separate expense, which would count them twice.
 */
export function swapCost(i: SwapCostInput): SwapCost {
  const round2 = (n: number) => Math.round(n * 100) / 100;
  const gold = i.returnedCostPerG != null && i.replacementCostPerG != null
    ? round2((i.replacementCostPerG - i.returnedCostPerG) * i.grams)
    : null;
  const shipping = round2(Number(i.returnFeeSgd ?? 0) + Number(i.outboundFeeSgd ?? 0));
  const complete = i.returnFeeSgd != null && i.outboundFeeSgd != null;
  return { goldDifferenceSgd: gold, shippingSgd: shipping, shippingComplete: complete, totalSgd: gold == null ? null : round2(gold + shipping) };
}
 
/** Parse an admin-entered courier cost: null clears it; otherwise 0–1000 S$, 2dp. */
export function parseFeeInput(v: unknown): { ok: true; value: number | null } | { ok: false } {
  if (v === null || v === "") return { ok: true, value: null };
  const n = Number(v);
  if (!Number.isFinite(n) || n < 0 || n > 1000) return { ok: false };
  return { ok: true, value: Math.round(n * 100) / 100 };
}
