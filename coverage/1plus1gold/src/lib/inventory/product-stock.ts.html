/**
 * The three quantities a product can have, named.
 *
 * Backed by the `v_product_stock` SQL view — see
 * 20260720b_v_product_stock_single_source.sql for the full reasoning. Short
 * version: `1 gram Split-Valcambi` legitimately reads 18 on one screen and 68
 * on another, because "available" was being used for three different things.
 * All three numbers were correct, which is exactly what made it dangerous — a
 * collision like that never shows up as a bug, it shows up as a decision made
 * on the wrong number.
 *
 * Pick deliberately:
 *
 *   freeToSell      unreserved, on the shelf. Reconciliation's "Showing".
 *   physicallyHeld  + reserved. Custody — what an auditor counting the vault
 *                   would find, and the right basis for an asset figure.
 *   sellableNow     what the shop may sell, pool-aware. The ONLY one that can
 *                   exceed the physical bar count, and only for CombiBar pool
 *                   members (18 loose splits + 50 grams still inside an
 *                   unbroken CombiBar = 68).
 *
 * Valuation is deliberately NOT here — gold values grams × cost_per_g_sgd from
 * serialized rows, non-gold values stock_qty × avg landed cost from PO receipts
 * (`category-stock.ts`). Two different, already-correct computations; folding
 * them in would muddy a module whose job is to be unambiguous about quantity.
 */
 
import type { SupabaseClient } from "@supabase/supabase-js";
 
export interface ProductStock {
  product_id:         string;
  product_type:       string;
  serialized:         boolean;
  bars_available:     number;
  bars_reserved:      number;
  bars_sold:          number;
  bars_broken:        number;
  free_to_sell:       number;
  physically_held:    number;
  sellable_now:       number;
  pool_member:        boolean;
  pool_key:           string | null;
  has_inventory_rows: boolean;
  /** Supplier stock we may sell but do NOT hold (procure-on-sale). Included in
   *  `sellable_now`, deliberately absent from `physically_held`. */
  consignment_available: number;
  /** When the supplier last confirmed that quantity. Availability is a claim
   *  with an age — stale is how you oversell. */
  consignment_as_of:  string | null;
  /** WHY sellable_now diverges from free_to_sell. Read this instead of
   *  inferring: before consignment, pool was the only possible cause. */
  divergence_reason:  "pool" | "consignment" | "both" | null;
}
 
/**
 * True when this product's shop availability differs from the bars we hold —
 * i.e. the number a reader is most likely to misinterpret. Only CombiBar pool
 * members can do this, and it is correct when they do: a bundle SKU can show 13
 * sellable against ZERO physical bars, because it is floor(pool grams / 5).
 *
 * Surface this wherever both numbers are visible, so the difference explains
 * itself instead of looking like drift.
 */
export function hasPoolDivergence(s: ProductStock): boolean {
  return s.sellable_now !== s.free_to_sell
      && (s.divergence_reason === "pool" || s.divergence_reason === "both");
}
 
/** True when ANY cause makes shop availability exceed the bars we hold. */
export function hasStockDivergence(s: ProductStock): boolean {
  return s.sellable_now !== s.free_to_sell && s.divergence_reason !== null;
}
 
/** One-line explanation of a divergence, for a tooltip or sub-label. */
export function poolDivergenceNote(s: ProductStock): string | null {
  return hasPoolDivergence(s) ? divergenceNote(s) : null;
}
 
/**
 * One-line explanation of why sellable exceeds physical, for a tooltip.
 *
 * Branches on `divergence_reason` rather than assuming pool. Before consignment
 * existed, pool was the only cause and the note could hardcode CombiBar
 * language — the moment a second cause appeared that text became confidently
 * wrong on any consignment product, which is the failure this replaces.
 */
export function divergenceNote(s: ProductStock): string | null {
  if (!hasStockDivergence(s)) return null;
  const lead = `${s.free_to_sell} physical · ${s.sellable_now} sellable`;
 
  switch (s.divergence_reason) {
    case "pool":
      return `${lead} — the rest is uncommitted grams inside an unbroken ` +
             `CombiBar (${s.pool_key} pool)`;
    case "consignment":
      return `${lead} — ${s.consignment_available} held by the supplier, ` +
             `procured when sold`;
    case "both":
      return `${lead} — ${s.consignment_available} from the supplier, the rest ` +
             `uncommitted grams inside an unbroken CombiBar (${s.pool_key} pool)`;
    default:
      return null;
  }
}
 
/** Whole-catalogue stock, keyed by product_id. */
export async function getProductStock(
  db: SupabaseClient,
): Promise<Map<string, ProductStock>> {
  const { data, error } = await db.from("v_product_stock").select("*");
  if (error) throw new Error(`v_product_stock query failed: ${error.message}`);
  return new Map((data ?? []).map((r) => [r.product_id as string, r as ProductStock]));
}
