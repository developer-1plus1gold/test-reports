/**
 * Counter-based stock valuation for a non-serialized product category
 * (accessory / consumable / fulfilment_consumable / equipment).
 *
 * Gold bars are serialized (one `inventory` row each, valued by cost_per_g_sgd ×
 * grams). Everything else is a COUNTER: on-hand qty in `products.stock_qty`,
 * valued at the avg landed cost/unit derived from PO receipts × current stock.
 *
 * ONE source of truth shared by the per-category admin views AND the balance
 * sheet's non-gold inventory asset lines, so a holding value never disagrees
 * between the two. Extracted from /api/admin/accessories (2026-07-03).
 *
 * 2026-07-20 — each row also reports how much of its value is UN-RECONCILED.
 * We value the counter (stock_qty), which is correct: it is our best estimate of
 * what is physically on the shelf and it is what a physical count is checked
 * against. But the counter does not justify itself. `inventory_reconciliation`
 * independently derives what should be there from received − consumed − sold ±
 * documented adjustments, and any gap is value on the balance sheet that nothing
 * explains. That gap used to be invisible — an unexplained unit was capitalised
 * at full landed cost and looked identical to a verified one. The asset figure
 * is deliberately UNCHANGED; the point is only that the unexplained portion is
 * now visible instead of passing as audited.
 */

/** Non-serialized categories that use the counter (stock_qty) model. */
export type CategoryProductType = "accessory" | "consumable" | "fulfilment_consumable" | "equipment";
 
export interface CategoryStockRow {
  id:                    string;
  label:                 string;
  sku:                   string | null;
  stock_qty:             number | null;
  low_stock_threshold:   number;
  low_stock:             boolean;
  avg_cost_per_unit_sgd: number;
  stock_value_sgd:       number;
  total_sold:            number;
  total_revenue_sgd:     number;
  last_sold_at:          string | null;
  last_received_at:      string | null;
  /**
   * Reconciliation delta in units (showing − expected), or null if this product
   * isn't in the reconciliation view. See `unreconciled_value_sgd`.
   */
  reconciliation_delta:  number | null;
  /**
   * The slice of `stock_value_sgd` the book cannot explain — delta × avg landed
   * cost, signed the same way as the delta.
   *
   * POSITIVE is the direction that matters: we are carrying (and capitalising)
   * more units than received-minus-consumed accounts for, so the asset is
   * overstated by this much until someone explains it. Negative means the
   * counter is BELOW what the derivation predicts, which understates rather
   * than overstates and is not an audit risk.
   */
  unreconciled_value_sgd: number;
}
 
export interface CategoryStockTotals {
  skus:         number;
  stockValue:   number;   // Σ stock_value_sgd — the balance-sheet asset figure
  totalRevenue: number;
  lowStock:     number;
  /** Σ unreconciled_value_sgd — the portion of stockValue nothing explains. */
  unreconciledValue: number;
  /** SKUs carrying a non-zero delta. */
  unreconciledSkus:  number;
}
 
export interface CategoryStock {
  rows:   CategoryStockRow[];
  totals: CategoryStockTotals;
}
