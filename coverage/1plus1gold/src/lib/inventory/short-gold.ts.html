/**
 * Gold lines that are SHORT of a bar (Mukesh 2026-11-25, Lazada FEA14523).
 *
 * A paid order line for a gold product is fulfilled by bars linked through
 * order_fulfilments, or by consignment lines the supplier will deliver. When a sale
 * takes more units than are free (two marketplaces selling the last bar minutes apart),
 * consumeStockForOrderItem reserves what it can — and until now the rest was only a log
 * line: no alert, Dispatch showed the order as ready, the packing slip printed one serial
 * for two bars, and bars received later never reached the order. This module is the ONE
 * definition of "short", used by:
 *   - the post-sale check (alertIfOrderShort): a short order files a Control Tower card
 *     and a WhatsApp the moment it is created;
 *   - the dispatch gate (awaiting-gold.ts): a short line blocks packing and booking, and
 *     the slip says which bar is missing;
 *   - PO receipt (fillShortLinesFromArrival): bars just received go to short paid orders,
 *     oldest first, before they become free stock;
 *   - Dispatch → Attach bar (attachBarToLine): the operator attaches a free bar by hand.
 *
 * NOT short, by design: pool products (1 g splits / CombiBars — their grams come from the
 * pool, often with no row of their own), Gold Yield enrolment lines (gold_yield_bars), MAG
 * and accessories (not bars), and orders from before the first trading day (pre-launch test
 * orders still read 'paid' with no bars). Only open orders (status 'paid') are considered.
 */

/** One candidate line with everything that can cover it. Pure input for shortOf. */
export interface LineCoverage {
  orderId:        string;
  orderItemId:    string;
  productId:      string;
  productLabel:   string;
  quantity:       number;
  orderCreatedAt: string;
  fulfilments:    number;

/** Pure: how many units of this line have nothing behind them (0 when not applicable). */
export function missingUnits(l: LineCoverage): number {
  if (l.isPoolProduct || l.isGoldYield) return 0;
  return Math.max(0, Math.round(l.quantity) - l.fulfilments - l.consigned);
}
 
/** Pure: the short lines among candidates, oldest order first. */
export function shortOf(lines: LineCoverage[]): ShortGoldLine[] {
  return lines
    .map((l) => ({ l, missing: missingUnits(l) }))
    .filter((x) => x.missing > 0)
    .sort((a, b) => a.l.orderCreatedAt.localeCompare(b.l.orderCreatedAt) || a.l.orderItemId.localeCompare(b.l.orderItemId))
    .map(({ l, missing }) => ({
      orderId: l.orderId, orderItemId: l.orderItemId, productId: l.productId, productLabel: l.productLabel,
      quantity: l.quantity, covered: l.fulfilments + l.consigned, missing, orderCreatedAt: l.orderCreatedAt,
    }));
}

/**
 * Pure: the identity of ONE shortage on an order — which lines, how many bars missing.
 * Every alert path (the sale, a packed card taking a bar, the hourly sweep) files under
 * this key, so the same shortage pages once; a different shortage on the same order
 * (another line, another bar missing) is a new key and pages again.
 */
export function shortGoldAlertKey(orderId: string, lines: ShortGoldLine[]): string {
  const parts = lines.map((l) => `${l.orderItemId}x${l.missing}`).sort().join(",");
  return `short-gold:${orderId}:${parts}`;
}
