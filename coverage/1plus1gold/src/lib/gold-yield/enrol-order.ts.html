/**
 * Enrol every Gold Yield bar an order paid for — one bar per UNIT.
 *
 * ▶ WHY THIS IS A MODULE AND NOT A BLOCK IN A ROUTE.
 *
 * Until 2026-09-20 this logic lived inline in `/api/orders/create`, which meant
 * it only ever ran for COD and bank-transfer orders. `hitpay-order.ts` and the
 * Stripe webhook both set `shipper: "gold_yield"` on the order and both SKIP
 * `consumeStockForOrder` with a comment saying GY "goes through the GY
 * enrolment path" — and neither one called it. So a participant who chose
 * "Enrol in Gold Yield Programme" and paid by card got an order marked
 * `enrolled`, NO bar, and NO stock consumed: we took the money and left the
 * gold on the shelf.
 *
 * Measured on production 2026-09-20: 14 paid Gold Yield orders, ZERO with a
 * bar, every one an online payment. All 14 are the Demo Account and pre-date
 * the public launch, so no customer money was ever affected — but nothing made
 * it unreachable. `PaymentMethodSelector` accepts `shipperType` and never reads
 * it, so every card and PayNow option is offered on a Gold Yield order.
 *
 * One home, called by both doors, so the two cannot drift again.
 *
 * ▶ IT DERIVES EVERYTHING FROM THE DATABASE, on purpose. The caller passes an
 * order id and nothing else. Both doors already persist every input this needs
 * — `orders.spot_sgd`, `order_items.quantity/grams_each/price_sgd/is_presale`,
 * `products.gy_eligible/is_split_parent` — so there is no per-door argument
 * list that could be built two different ways. It also means a future reconcile
 * can replay an order that missed enrolment with no extra plumbing.
 *
 * ▶ ONE BAR PER UNIT. `denomination_g` is the PER-UNIT weight, so a qty-3 line
 * is three bars, three serials and three consumed inventory units — not one.
 * See the 20260920 migration for what replaced `UNIQUE (order_item_id)`.
 *
 * ▶ IDEMPOTENT IN BOTH DIRECTIONS, because it now runs AFTER the HTTP response
 * (waitUntil) and can therefore be retried by a reconcile without a human
 * deciding whether it already ran:
 *   bars      — only the unit indices not already present are inserted;
 *   inventory — consumption is driven by how many `order_fulfilments` already
 *               exist for the line, NOT by whether a bar was created, so a run
 *               that wrote the bars and then failed to consume tops itself up
 *               instead of leaving stock on the shelf forever.
 *
 * ▶ IT NEVER THROWS. Both callers invoke it outside the request/response path,
 * where an escaping exception is invisible. Every failure is collected into the
 * returned summary and the caller pages a human on it.
 */

/**
 * How many per-bar WhatsApps one order LINE may send before the rest are
 * summarised instead.
 *
 * A bar is a serialized asset and the per-bar message carries its serial, so
 * one each is the honest default. But quantity is customer-supplied and the
 * cart's only ceiling is 500 items per order — without a cap, one order could
 * fire hundreds of messages at the participant and at us. Five covers every
 * realistic Gold Yield order (a 50 g bar is five figures) and turns the
 * pathological case into a single summary line rather than a flood.
 */
export const GY_NOTIFY_PER_LINE_CAP = 5;
 
export interface GyEnrolSummary {
  /** False when the order is not a Gold Yield order — the common case. */
  applicable: boolean;
  barsCreated: number;
  /** Bars that already existed. A replay of a completed enrolment reports 0 created and N existing. */
  barsAlreadyPresent: number;
  unitsConsumed: number;
  /** Bars written with no derivable premium — enrolled, but not settleable until a human sets one. */
  barsWithoutBasis: number;
  /** Every failure, in words a human can act on. Non-empty means page somebody. */
  errors: string[];
}

async function enrolOneLine(
  client: Client,
  order: { id: string; customer_id: string },
  line: GyLine,
  spotAtEnrol: number,
  participant: { full_name?: string | null; phone?: string | null } | null,
  out: GyEnrolSummary,
): Promise<void> {

  // read as data rather than relied on as a constraint violation, so a partial
  // previous run completes instead of failing.
  const { data: existing, error: existErr } = await client
    .from("gold_yield_bars")
    .select("unit_index")
    .eq("order_item_id", line.orderItemId);
 
  Iif (existErr) {
    out.errors.push(`could not read existing bars for order item ${line.orderItemId}: ${existErr.message}`);
    return;

  out.barsAlreadyPresent += present.size;
 
  const missing: number[] = [];
  for (let u = 1; u <= line.quantity; u++) if (!present.has(u)) missing.push(u);

  //
  // The dashboard enrol door REFUSES this case (409) — a gift line, a
  // promotion, or a data error. This door cannot refuse: the customer has
  // already paid, and failing after payment is the 2026-09-03 fractional-gram
  // incident class.
  //
  // But it must not write 0 either. A stored premium of 0 says "bought at
  // spot", which puts the settlement basis exactly ON spot — the whole premium
  // the buyer pays becomes "profit" and the participant is handed half of it on
  // top of their principal. NULL is the honest answer: the bar enrols,
  // `settleSale` fails closed on a non-finite premium, and the basis can be set
  // by hand.
  //
  // ▶ A NEGATIVE premium is stored AS IS, deliberately — not collapsed to NULL.
  // `settleSale` refuses `premiumPct < 0` through the same `isNonNegative` guard
  // it refuses a non-finite one, so NULL and −20% fail settlement identically:
  // collapsing buys no safety and destroys the one fact an operator needs, which
  // is how far below spot the line was sold. `basisMissing` below still covers
  // both, so both flag the bar and page. (Proposed as a fix in the 2026-09-20
  // security review and declined for this reason; pinned by a test so the
  // "obvious" simplification cannot be re-applied silently.)
  const premiumPct = premiumPctFromPaid(line.paidPerG, spotAtEnrol);
  const basisMissing = premiumPct === null || premiumPct < 0;
 
  if (missing.length > 0) {
    const rows: Record<string, unknown>[] = [];
    for (const unitIndex of missing) {

        order_id:       order.id,
        order_item_id:  line.orderItemId,

        denomination_g: line.grams,
        grams_total:    line.grams,
        // What they PAID, not the spot — the audit trail the locked premium
        // derives from. Writing the bare spot here made a checkout-enrolled bar
        // carry a lower basis than the same bar enrolled from the dashboard.
        bar_cost_per_g: line.paidPerG.toFixed(4),
        earning_path:   resolveEarningPath({ isSplitParent: line.isSplitParent }),
        // Hedged. There is deliberately no un-hedged option at the checkout
        // door: taking spot risk is a choice made per bar on the enrolment
        // screen, not something to fall into while buying gold.
        hedged:         true,
        premium_pct:    premiumPct === null ? null : premiumPct.toFixed(5),
        spot_at_enrol:  spotAtEnrol.toFixed(4),
        status:         line.isPresale ? "pending_receipt" : "received",
        // The door. A checkout bar never leaves our custody, so there is no
        // receipt leg to run and no pickup address to be missing — the admin
        // panel hides the Receipt controls on this value. Written explicitly
        // rather than left to the column default so the two doors can never
        // drift into agreeing by accident.
        enrolment_source: "checkout",
        ...(line.isPresale ? {} : { received_at: new Date().toISOString() }),
      });
    }
 
    // ONE multi-row insert, so it is atomic: there is no half-enrolled line to
    // reason about, and a concurrent duplicate run collides on the composite
    // unique key rather than writing a second set of bars.

      .from("gold_yield_bars")
      .insert(rows)
      .select("id, inventory_id, unit_index");

  // the bars and then died before consuming, "did we create a bar?" says no and

  // only from `consumeInventoryForGyEnrolment`, so the count is unambiguous.
  const { count: alreadyConsumed, error: fulErr } = await client
    .from("order_fulfilments")
    .select("id", { count: "exact", head: true })
    .eq("order_item_id", line.orderItemId);

    out.errors.push(`could not count fulfilments for order item ${line.orderItemId}: ${fulErr.message}`);
    return;
  }

      );
      break;
    }
    out.unitsConsumed += 1;
  }
}
 
async function notifyLine(
  line: GyLine,
  created: { inventory_id: string }[],
  participant: { full_name?: string | null; phone?: string | null } | null,
  basisMissing: boolean,
  spotAtEnrol: number,
  out: GyEnrolSummary,
): Promise<void> {
  const adminPhones = getAdminPhonesCsv();
  const named = created.slice(0, GY_NOTIFY_PER_LINE_CAP);
  const grams = String(line.grams);
 
  if (adminPhones) {
    for (const bar of named) {
      sendWhatsAppTemplate(adminPhones, GY_ADMIN_ENROLLED, {
        "1": bar.inventory_id,
        "2": participant?.full_name ?? "Unknown",
        "3": grams,
      }).catch((e) => console.error("[gy-enrol] admin notify failed:", e));
    }
  } else E{
    console.warn("[gy-enrol] no admin phones configured — skipping enrolment alert");
  }
 
  // A bar with no settlement basis is enrolled but cannot be paid out. It is
  // reported through `errors` so the caller pages, rather than as a separate
  // free-form WhatsApp that needs an open 24h session window to deliver.
  if (basisMissing) {
    out.errors.push(
      `order item ${line.orderItemId}: ${created.length} bar(s) enrolled with NO settlement basis ` +
      `(paid S$${line.paidPerG.toFixed(4)}/g against spot S$${spotAtEnrol.toFixed(4)}). ` +
      `Any sale will be refused until a premium is set by hand.`,
    );
  }
 
  if (participant?.phone && !line.isPresale) {
    for (const bar of named) {
      sendWhatsAppTemplate(
        participant.phone,
        templateOrFallback(GY_RECEIPT_RECEIVED, BUYBACK_RECEIVED),
        { "1": bar.inventory_id, "2": grams },
      ).catch((e) => console.error("[gy-enrol] participant notify failed:", e));
    }
  }
 
  const unnamed = created.length - named.length;
  if (unnamed > 0) {
    console.warn(
      `[gy-enrol] order item ${line.orderItemId}: ${created.length} bars enrolled, ` +
      `${unnamed} beyond the per-line notify cap of ${GY_NOTIFY_PER_LINE_CAP} were not messaged individually`,
    );
    out.errors.push(
      `order item ${line.orderItemId}: ${created.length} bars enrolled — only the first ` +
      `${GY_NOTIFY_PER_LINE_CAP} were notified individually; confirm the remaining ${unnamed} in admin`,
    );
  }
}
