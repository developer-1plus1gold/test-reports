/**
 * Which purchased bars may be enrolled in the Gold Yield Programme, and how
 * each one earns.
 *
 * ▶ ONE GATE, TWO PATHS. Design of record: docs/gold-yield-redesign.md §1.
 *
 *   gate  — `products.gy_eligible`. The operator's programme decision, and the
 *           ONLY thing that decides whether a bar may be enrolled.
 *   path  — `products.is_split_parent`. A physical fact about the bar, and it
 *           decides how that bar earns, not whether it qualifies.
 *
 * ▶ WHAT CHANGED ON 2026-09-19, and why the old rule was wrong in BOTH
 * directions. It gated on weight — `grams_each ∈ {50, 100}` — plus
 * `gy_eligible`:
 *
 *   too narrow  — a 20g solid bar the operator wants in the programme was
 *                 refused for being the wrong weight, though nothing about the
 *                 programme cares. The operator's words: any bar that cannot be
 *                 split "will be intact and be sold complete REGARDLESS OF ITS
 *                 WEIGHT".
 *   too wide    — it let a SOLID 50g bar through on weight alone if someone
 *                 ever flagged it, and the breakdown mechanism is inoperable on
 *                 one. That half was closed on 2026-09-18 by adding
 *                 `gy_eligible`; this pass removes the weight half and replaces
 *                 the assumption underneath it — that every enrolled bar gets
 *                 broken down — with an explicit second earning path.
 *
 * ⚠️ Weight could never have expressed either rule, because there is no product
 * type to lean on: EVERY 50g and 100g product is `product_type = 'standard'`,
 * CombiBars included (measured on production, 2026-09-18).
 *
 * ▶ WHY `gy_eligible` AND NOT `is_split_parent` AS THE GATE, now that both are
 * consulted: they mean different things and are load-bearing separately.
 * `is_split_parent` is physical and cannot be withdrawn; `gy_eligible` is the
 * programme lever and can. Gating on the physical fact would quietly take that
 * lever away, and would also re-exclude every whole bar — which is the case
 * this pass exists to admit.
 */
 
export type GyEarningPath = "split" | "whole";
 
export interface GyBarEligibilityInput {
  /**
   * `products.gy_eligible` for the product the purchased line points at. `null`
   * or `undefined` means the product could not be read — treated as NOT
   * eligible, because the programme gate must fail closed.
   */
  gyEligible: boolean | null | undefined;
}
 
export interface GyEarningPathInput {
  /**
   * `products.is_split_parent`. `null`/`undefined` is read as `false` — a bar
   * we cannot confirm is scored is sold whole, which is the recoverable
   * mistake. Breaking down a bar that turns out to be solid is not.
   */
  isSplitParent: boolean | null | undefined;
}
 
/**
 * The single predicate the listing route and the enrolment route share, so the
 * two cannot drift into offering what the server then rejects.
 */
export function isGyEligibleBar({ gyEligible }: GyBarEligibilityInput): boolean {
  return gyEligible === true;
}
 
/**
 * How an enrolled bar earns. Stamped onto `gold_yield_bars.earning_path` at
 * enrolment and never re-derived afterwards, because a product's
 * `is_split_parent` can be edited later and an enrolled bar's physics cannot.
 */
export function resolveEarningPath({ isSplitParent }: GyEarningPathInput): GyEarningPath {
  return isSplitParent === true ? "split" : "whole";
}
 
/** The refusal an ineligible bar gets, so both routes word it identically. */
export function gyIneligibleReason(input: GyBarEligibilityInput): string | null {
  if (isGyEligibleBar(input)) return null;
  return "This product is not enrolled in the Gold Yield Programme";
}
 
/**
 * ▶ THE QUANTITY GATE THAT USED TO LIVE HERE IS GONE (removed 2026-09-20,
 * operator decision). It exported GY_MAX_UNITS_PER_LINE = 1,
 * isGyEnrollableQuantity() and gyQuantityReason(), and both enrolment doors
 * refused any line with quantity > 1.
 *
 * The refusal was right for the code underneath it and wrong as a destination.
 * The enrolment created exactly ONE `gold_yield_bars` row per order ITEM at the
 * PER-UNIT weight and consumed exactly one inventory unit, so a qty-3 line took
 * payment for three bars and enrolled one — 10 g the participant owned, had
 * paid for, and could not reach.
 *
 * The fix is the fan-out, not the refusal. `gold_yield_bars` now carries
 * `unit_index` under `UNIQUE (order_item_id, unit_index)` (migration
 * 20260920_gold_yield_bars_unit_index), which is both the fan-out key and the
 * replay guard the old single-column unique constraint was. Enrolment creates
 * one bar per UNIT and consumes one inventory unit per bar:
 *   - checkout    → src/lib/gold-yield/enrol-order.ts (both rails)
 *   - dashboard   → src/app/api/gold-yield/enrol/route.ts
 *
 * ⚠️ CORRECTION to what stood here before: this file claimed replacing
 * `UNIQUE (order_item_id)` was the hard part. It was one guarded migration and
 * it was a no-op on every existing row. The claim was an overstatement and is
 * recorded as such rather than quietly deleted — the same overstatement was
 * written into src/app/api/orders/create/route.ts on the same day.
 *
 * Quantity is not a gate on any Gold Yield surface any more. It is a delivery
 * choice, and it has no more business restricting quantity than Vault does.
 */
