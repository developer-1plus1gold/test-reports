/**
 * Which purchased bars may be enrolled in the Gold Yield Programme.
 *
 * ▶ THE RULE HAS TWO HALVES AND THE SECOND ONE WAS MISSING UNTIL 2026-09-18.
 *
 * Both `/api/gold-yield/eligible-bars` (what the customer is offered) and
 * `/api/gold-yield/enrol` (what the server accepts) gated on WEIGHT alone —
 * `grams_each ∈ {50, 100}` — while their own comments said "CombiBar". Weight
 * is not the property the programme depends on.
 *
 * The programme's revenue mechanism is that the bar is BROKEN DOWN into 1g/5g/
 * 10g splits which are then sold off (`/api/admin/gold-yield/bars/[id]/
 * breakdown`, and the `grams_sold` / `grams_remaining` columns it draws down).
 * A Valcambi CombiBar is scored into cards and can be broken that way. A solid
 * 50g PAMP Fortuna Veriscan in assay cannot. Enrolling one produces a yield bar
 * that can never be drawn down — the mechanism is inoperable on it, and nothing
 * downstream would have said so.
 *
 * ⚠️ Weight could never have expressed this, because there is no separate
 * product type to lean on: EVERY 50g and 100g product is `product_type =
 * 'standard'`, CombiBars included. What distinguishes them is the per-product
 * flag the operator already sets — `products.gy_eligible`, true on exactly the
 * two CombiBars and false on the seven solid bars (measured on production,
 * 2026-09-18).
 *
 * ▶ WHY `gy_eligible` AND NOT `is_split_parent`, when today they are true on
 * the same two rows: they mean different things. `is_split_parent` is a
 * PHYSICAL fact (this bar is scored into cards) and is a necessary
 * precondition; `gy_eligible` is the PROGRAMME decision (this product may be
 * enrolled) and is the operator's lever — it can be withdrawn from a bar that
 * is still physically splittable. Gating on the programme flag keeps that lever
 * real. Gating on the physical flag would quietly take it away.
 */
 
/** The denominations the programme is written around. */
export const GY_ELIGIBLE_DENOMINATIONS = [50, 100] as const;
 
export interface GyBarEligibilityInput {
  /** `order_items.grams_each` for the purchased line. */
  grams: number;
  /**
   * `products.gy_eligible` for the product that line points at. `null` or
   * `undefined` means the product could not be read — treated as NOT eligible,
   * because the programme gate must fail closed: offering a bar we cannot
   * break down is worse than failing to offer one we could.
   */
  gyEligible: boolean | null | undefined;
}
 
/**
 * The single predicate both the listing route and the enrolment route use, so
 * the two cannot drift into offering what the server then rejects — or, as
 * before this existed, into accepting what the programme cannot operate on.
 */
export function isGyEligibleBar({ grams, gyEligible }: GyBarEligibilityInput): boolean {
  if (gyEligible !== true) return false;
  return (GY_ELIGIBLE_DENOMINATIONS as readonly number[]).includes(Number(grams));
}
 
/** The refusal an ineligible bar gets, so both routes word it identically. */
export function gyIneligibleReason({ grams, gyEligible }: GyBarEligibilityInput): string | null {
  if (isGyEligibleBar({ grams, gyEligible })) return null;
  if (!(GY_ELIGIBLE_DENOMINATIONS as readonly number[]).includes(Number(grams))) {
    return "Only 50g and 100g CombiBars are eligible";
  }
  return "This bar cannot be enrolled — the Gold Yield Programme needs a bar that can be broken into 1g cards";
}
