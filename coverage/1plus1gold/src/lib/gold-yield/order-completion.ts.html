/**
 * When has a Gold Yield ORDER reached its destination?
 *
 * ▶ WHY THE QUESTION EXISTS. A Gold Yield order is written at checkout as
 * `shipper='gold_yield', shipper_status='enrolled'`, and until 2026-09-20
 * `derive_order_lifecycle` had no arm for it — so it sat at lifecycle 'placed'
 * for ever. Revenue recognises at 'completed' (src/lib/finance/pnl.ts), so no
 * Gold Yield order had ever been recognised: PROD held 14 of them, S$81,584.10.
 * Same class as the 2026-07-21 `in_person` incident, where a channel matching no
 * dispatch bucket could never reach 'completed'.
 *
 * Operator: "FOr gold yield order completion in order history - simialr ot the
 * vault..only 2 status, but in this case GY specific status and also on
 * enrolment activation."
 *
 * ▶ WHY IT IS A LIB AND NOT A HELPER IN THE ROUTE. Two reasons, and the second
 * is the binding one: a decision that moves revenue deserves its own tests, and
 * a Next route file may only export handlers and segment config — exporting
 * anything else fails `next build` but NOT `npm run verify` or CI Tests, so it
 * reaches Deploy (PR #259, guarded since #260 by route-exports-valid.test.ts).
 *
 * ⚠️ Do not write that route glob here: a doubled star followed by a slash ends
 * this block comment early, and the file then fails to parse. It did.
 */
 
/** Bar statuses that mean "this bar is IN the programme". */
const IN_PROGRAMME: ReadonlySet<string> = new Set(["active", "partially_sold", "fully_sold"]);
 
/** Bar statuses that mean "this bar has not been accepted yet". */
const AWAITING: ReadonlySet<string> = new Set(["pending_receipt", "received"]);
 
/**
 * Should the order carrying these bars now complete?
 *
 * ▶ THE RULE IS NOT "ALL BARS ACTIVE", and the difference matters on a qty > 1
 * order. A rejected bar is going back to the customer and will NEVER activate,
 * so "all" would leave such an order un-completable for ever — its revenue
 * unrecognised with nothing left to trigger it. So: nothing still AWAITING a
 * decision, and at least one bar actually in the programme.
 *
 * An order whose every bar was rejected therefore does NOT complete, which is
 * the right answer — nothing went into the programme, so there is nothing to
 * recognise. An order with no bars at all does not complete either: that is the
 * enrolment having failed (see the 2026-09-20 `COUNT(*) + 1` defect, where a
 * qty-2 order created zero bars), and completing it would recognise revenue for
 * gold that never entered the programme.
 */
export function shouldCompleteGoldYieldOrder(barStatuses: readonly string[]): boolean {
  if (barStatuses.some((s) => AWAITING.has(s))) return false;
  return barStatuses.some((s) => IN_PROGRAMME.has(s));
}
