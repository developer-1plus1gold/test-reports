/**
 * Gold Yield — a gold line's price per gram, NET of its share of the
 * order-level IPM discount. Used on BOTH sides of the trade.
 *
 * ▶ IT IS NOT ENROLMENT-ONLY, despite this file's name, and that matters
 * because the one-sided-ness below was itself the defect. Consumers:
 *
 *   enrolment side — what the PARTICIPANT paid, stamped as the bar's basis:
 *     api/gold-yield/enrol, api/gold-yield/eligible-bars, lib/gold-yield/enrol-order
 *   sale side      — what the BUYER paid, settled against that basis:
 *     lib/inventory-fulfilment (accrueGoldYieldForOrderItem),
 *     api/admin/gold-yield/sales (the manual route's default)
 *
 * Until 2026-09-22 only the first group called it, so the promo was netted out
 * of the cost basis and NOT out of the sale price — which overstates the spread
 * and therefore overpays the participant AND overstates our platform share.
 * Operator, 2026-09-22, on the worked example: "$102 buying value of gold and
 * $104 selling value of gold i.e. Profit of $2". The sale side was reading 105.
 * `gold-yield-enrol-cost-wiring.test.ts` now pins both groups.
 *
 * ▶ KEEP THE FILE NAME OR RENAME IT WHOLESALE — do NOT add a second,
 * sale-side copy of this arithmetic. Two implementations of one discount rule
 * is how the two sides drifted apart in the first place.
 *
 * ── Why it exists at all ──────────────────────────────────────────────────
 *
 * ▶ THE DEFECT THIS EXISTS FOR, found 2026-09-21 on the §9 DEV test.
 *
 * Both enrolment doors stamped the bar's basis as `order_items.price_sgd /
 * grams_each` — the CATALOGUE line price, before any order-level discount. A
 * promo code therefore never reached the bar.
 *
 * Measured on the reported order (DEV `d4676ed1`):
 *
 *   line price          964.09 for 5 g   =  192.8180/g
 *   promo LW1+1H8YQ    −19.28            =   −3.8560/g   (2%, all of it IPM)
 *   actually paid       944.81 for 5 g   =  188.9620/g   ← the true basis
 *
 * The bar was stamped 192.8180/g and premium 7.99980%. It then sold at
 * 192.7180/g and settled to EXACTLY ZERO — because the settlement carries the
 * bar at the participant's locked premium over spot-at-sale, and a premium
 * derived from the undiscounted price says the participant paid the same
 * margin the buyer is paying. The real margin — the discount they were given —
 * was invisible to the arithmetic.
 *
 *   with the true premium (5.83965%): basis 188.8640/g, profit 3.8540/g,
 *   S$19.27 on 5 g, split S$9.64 / S$9.63.
 *
 * ▶ IT IS A RATE, NOT AN ALLOCATION, AND THAT IS DELIBERATE. `discounts.ts`
 * splits a whole-order discount between the IPM and taxable subtotals
 * pro-rata, with no product-level scoping — so within IPM every gold line bore
 * the same proportional reduction. Applying the same proportion here is
 * faithful to how the discount was computed, needs no residual-to-last
 * bookkeeping, and gives a per-gram rate that sits beside `premium_pct`, which
 * is also a rate.
 *
 * ▶ THE BASE IS THE LINES, NOT `orders.subtotal_ipm_sgd`. The stored subtotal
 * is nullable and is written by four different order-creation paths; the lines
 * are the thing being apportioned across, so deriving from them guarantees the
 * shares sum to the whole discount. Verified on DEV that the two agree on every
 * discounted order (964.09 = 964.09; 31,603.49 = 31,603.49).
 */
 
/** One order line, as both enrolment doors already have it. */
export interface IpmLineInput {
  /** `order_items.price_sgd` — PER UNIT, not the line total. */
  pricePerUnitSgd: number | string | null;
  /** `order_items.quantity`. */
  quantity: number | string | null;
  /** `order_items.grams_each`. Zero on an accessory, which is not IPM. */
  gramsEach: number | string | null;
}
 
const num = (v: number | string | null | undefined): number =>
  v === null || v === undefined ? NaN : Number(v);
 
/**
 * Σ (price × quantity) over the order's IPM lines — the base the IPM discount
 * was apportioned against.
 *
 * ▶ EVERY gold line counts, not just the Gold-Yield-eligible ones. The discount
 * was spread across all of them, so measuring the base on a subset would
 * inflate this line's share.
 */
export function ipmSubtotalFromLines(lines: IpmLineInput[]): number {
  let total = 0;
  for (const l of lines) {
    const grams = num(l.gramsEach);
    const price = num(l.pricePerUnitSgd);
    const qty   = num(l.quantity);
    if (!(grams > 0) || !Number.isFinite(price) || !Number.isFinite(qty)) continue;
    total += price * Math.max(1, Math.trunc(qty));
  }
  return total;
}
 
/**
 * The bar's true cost basis per gram, net of this line's share of the
 * order-level IPM discount.
 *
 * ▶ RETURNS NULL RATHER THAN A GUESS, because this number is stamped onto the
 * bar and paid out against years from now. Null on: a non-positive price or
 * weight (already the refusing case today — `premiumPctFromPaid` rejects it and
 * the bar is created without a basis), and on a discount that is positive but
 * cannot be coherently apportioned. A discount at or beyond the whole gold
 * subtotal would put the basis on or below zero, which turns the buyer's entire
 * premium into "profit" and pays the participant half of it — the same failure
 * mode `settlementNumber` exists to stop, one input earlier.
 *
 * ▶ NO DISCOUNT IS NOT AN ERROR. A null/absent/zero `discount_ipm_apportioned_sgd`
 * is the ordinary case and returns exactly `price / grams`, byte-identical to
 * the behaviour before this function existed.
 */
export function netPaidPerGram(args: {
  pricePerUnitSgd: number | string | null;
  gramsEach:       number | string | null;
  /** `orders.discount_ipm_apportioned_sgd`. Nullable on legacy orders. */
  ipmDiscountSgd:  number | string | null;
  /** From `ipmSubtotalFromLines` over the whole order. */
  ipmSubtotalSgd:  number;
}): number | null {
  const price = num(args.pricePerUnitSgd);
  const grams = num(args.gramsEach);
  if (!Number.isFinite(price) || !(price > 0)) return null;
  if (!Number.isFinite(grams) || !(grams > 0)) return null;
 
  const discount = num(args.ipmDiscountSgd);
  // Absent or zero — the ordinary, un-discounted order.
  if (!Number.isFinite(discount) || discount <= 0) return price / grams;
 
  const base = args.ipmSubtotalSgd;
  if (!Number.isFinite(base) || base <= 0) return null;   // a discount against nothing
  if (discount >= base) return null;                      // basis would be <= 0
 
  return (price * (1 - discount / base)) / grams;
}
