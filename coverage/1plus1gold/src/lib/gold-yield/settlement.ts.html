/**
 * Gold Yield — THE settlement rule. One function, three consumers.
 *
 * Design of record: docs/gold-yield-redesign.md §2 and §5.2.
 *
 * ▶ WHY THIS FILE EXISTS. Until 2026-09-19 the repo held THREE different
 * answers to "what does the participant earn":
 *
 *   1. the Yield Simulator          — the agreement's formula, but fed the
 *                                     premium derived against TODAY's spot,
 *                                     which collapses it into (3)
 *   2. /api/admin/gold-yield/sales  — frozen-dollar basis, costs deducted
 *   3. accrueGoldYieldSale          — frozen-dollar basis, costs at 0
 *
 * A participant was shown (1) and would be paid (2) or (3). All three now call
 * this, so the projection someone sees and the amount we actually pay are the
 * same arithmetic.
 *
 * ▶ THE MODEL. The quantity locked at enrolment is the participant's PREMIUM
 * PERCENTAGE, never a dollar figure:
 *
 *     net profit/g = spot × (buyer premium − your bar's premium)
 *
 * and nothing else. Both premiums are taken NET of any order-level discount on
 * their own side of the trade — see `enrol-cost.ts`, which both the enrolment
 * and the sale path call.
 *
 * Agreement v1.1 §7 already matches: "Spot-price movement is hedged by
 * 1Plus1Gold and does not affect your earnings or your bar's cost basis."
 *
 * ▶ GOLD ONLY — NO SHIPPING, PAYMENT-PROCESSING OR PACKING DEDUCTION.
 * Operator decision, 2026-09-22: "for 3 and for 4 we will change the agreement
 * and make it simple and remove anything around payment processing, shipping
 * and packing. The agreement for GY will be made simple and only around Gold."
 *
 * Three reasons it is the right call, not just the simple one:
 *   1. Those costs are incurred at ORDER level. One buyer's order can draw
 *      grams from several participants' bars, so charging a whole order's
 *      shipping to one participant's grams was arbitrary — the automatic path
 *      already passed 0 for exactly that reason, while the manual admin route
 *      let an operator enter real figures. The same sale therefore settled
 *      differently depending on which path recorded it.
 *   2. The processing fee largely washes: the buyer pays one and we pay the
 *      gateway one. Deducting the gross fee double-counts. (The books measured
 *      78.3% recovery on 2026-08-10 — real, but an order-level subsidy, not a
 *      per-participant one.)
 *   3. A participant can verify a gold-only rule from spot and two premiums.
 *      They cannot audit our packing cost, so a deduction they cannot check is
 *      a deduction they have to take on trust.
 *
 * ⚠️ THE AGREEMENT TEXT STILL SAYS OTHERWISE, DELIBERATELY. v1.1 §4 lists the
 * three deductions. Operator, same message: "leave the changes to agreement
 * aside. We will work on that once we have finished the GY work technically and
 * then ensure that the agreement matches the code." So the code is now the
 * narrower of the two — it can only ever pay the participant MORE than §4
 * permits, never less, which is the safe direction to be out of step in. The
 * agreement must be re-issued to a gold-only §4 before the programme takes real
 * money. `gold-yield-settlement.test.ts` pins that no cost term returns.
 *
 * ▶ THE COLUMNS STAY. `gold_yield_sales.shipping_cost_per_g` /
 * `payment_cost_per_g` / `packing_cost_per_g` are NOT NULL DEFAULT 0 and hold
 * real figures on historical rows. Dropping them would erase the record of how
 * those sales were actually settled; they are simply no longer written.
 *
 * Read that way the two clauses agree and earnings become almost
 * spot-independent, which is what "hedged" means. The previous code locked a
 * DOLLAR cost against a FLOATING sale price, so a 2.2% move in spot swung the
 * payout 8.5× — the precise opposite of §7 — and it was asymmetric in a way
 * that cost us in both directions (a zero payout on a healthy spread when spot
 * fell; four times the spread when spot rose).
 *
 * ▶ THE ONE ALGEBRAIC STEP worth stating, because it means a real settlement
 * never needs to know the buyer's premium:
 *
 *     spot × (buyerPrem − barPrem)
 *       = spot × ((salePrice/spot − 1) − barPrem)
 *       = salePrice − spot × (1 + barPrem)
 *
 * So `revaluedBasisPerG = spotAtSale × (1 + barPrem)` — the participant's bar
 * carried at THEIR locked premium over TODAY's spot — is all that is needed,
 * and it does double duty as the un-hedged principal.
 *
 * ▶ HEDGED vs UN-HEDGED differ in ONE input, and earnings are identical:
 *
 *     principal/g = (hedged ? spotAtEnrol : spotAtSale) × (1 + barPrem)
 *
 *   hedged     — principal returns at what they PAID; we absorb the spot move,
 *                which is what our hedge is for.
 *   un-hedged  — principal returns at today's spot carrying their premium; they
 *                take the risk and the reward and we place no hedge.
 *
 * ⚠️ THE HEDGE IS NOT CODE. Nothing in this application places, tracks or
 * reconciles a hedge position. On a hedged bar, if the hedge is not genuinely
 * on, we eat the spot move — on the worked example a sale that looks profitable
 * loses S$7.65. That is an offline treasury discipline; this file cannot verify
 * it and does not pretend to.
 */
 
/** Participant's half of net profit. Agreement §4. */
export const PARTICIPANT_SHARE = 0.5;
 
export interface SettlementInput {
  /** Grams changing hands in THIS sale. Fractional is legitimate (2.5g, 1oz). */
  grams: number;
  /** What the buyer paid per gram, SGD. */
  salePricePerG: number;
  /** Spot SGD/g at the moment of sale. */
  spotAtSale: number;
  /** Spot SGD/g when the participant bought the bar. */
  spotAtEnrol: number;
  /** The participant's locked premium, as a PERCENTAGE (5 for 5%). */
  premiumPct: number;
  /** false = the participant took the spot risk. Stamped on the bar. */
  hedged: boolean;
  /** Override only for modelling; production is always PARTICIPANT_SHARE. */
  sharePct?: number;
}
 
export interface Settlement {
  /** The bar carried at the participant's premium over spot-at-sale. */
  revaluedBasisPerG: number;
  /** Clamped at zero — agreement §4: "your share is always zero or positive". */
  netProfitPerG: number;
  /** What the participant gets BACK per gram. NEVER clamped. */
  principalPerG: number;
  participantSharePerG: number;
  /** Totals for this sale, rounded to cents exactly once, at the end. */
  participantShare: number;
  platformShare: number;
  principal: number;
  /** principal + their share — what actually leaves us for this sale. */
  participantTotal: number;
  /** What they originally paid for these grams. */
  investedPerG: number;
  invested: number;
  /** (total back − invested) ÷ invested × 100. MAY BE NEGATIVE — see below. */
  returnPct: number;
}
 
export type SettlementResult =
  | { ok: true; settlement: Settlement }
  | { ok: false; reason: string };
 
const isPositive = (n: unknown): n is number =>
  typeof n === "number" && Number.isFinite(n) && n > 0;
 
const isNonNegative = (n: unknown): n is number =>
  typeof n === "number" && Number.isFinite(n) && n >= 0;
 
/** Round to cents ONCE, at the end. Never round an intermediate per-gram rate. */
const cents = (n: number): number => Number(n.toFixed(2));
 
/**
 * ▶ FAILS CLOSED, and returns a reason rather than throwing.
 *
 * A missing spot or premium cannot be defaulted: every plausible default is a
 * wrong payout, and this runs on a path where the customer has already paid.
 * The caller decides whether that is a 500, an alert, or a skipped accrual —
 * but it never silently computes against a guess.
 */
/**
 * Read a NULLABLE database numeric for settlement.
 *
 * ▶ NEVER use `Number()` on one of these, and this is not a style preference.
 * `Number(null)` is **0**, and 0 is a LEGITIMATE premium here — it is the claim
 * "this bar was bought at spot", which `settleSale` accepts on purpose. So a
 * NULL `premium_pct` coerced with `Number()` does NOT fail closed: it settles
 * at zero premium, which puts the basis exactly on spot, turns the buyer's
 * ENTIRE premium into "profit", and pays the participant half of it on top of
 * their full principal. That is an overpayment on a money path.
 *
 * It was reachable by design, not in theory: the checkout door writes
 * `premium_pct: null` with a VALID `spot_at_enrol` when the premium cannot be
 * derived (it cannot refuse — the customer has already paid), so the
 * `isPositive(spotAtEnrol)` guard that catches a fully-unstamped legacy bar
 * does not fire, and nothing else did.
 *
 * `NaN` is the honest reading of "not recorded": it fails every `isPositive` /
 * `isNonNegative` guard below, so `settleSale` refuses and the caller pages.
 */
export const settlementNumber = (v: unknown): number =>
  v === null || v === undefined ? NaN : Number(v);
 
export function settleSale(input: SettlementInput): SettlementResult {
  const {
    grams, salePricePerG, spotAtSale, spotAtEnrol, premiumPct, hedged,
    sharePct = PARTICIPANT_SHARE,
  } = input;
 
  if (!isPositive(grams))         return { ok: false, reason: "grams must be > 0" };
  if (!isPositive(salePricePerG)) return { ok: false, reason: "salePricePerG must be > 0" };
  if (!isPositive(spotAtSale))    return { ok: false, reason: "spotAtSale must be > 0" };
  if (!isPositive(spotAtEnrol))   return { ok: false, reason: "spotAtEnrol must be > 0" };
  if (!isNonNegative(premiumPct)) return { ok: false, reason: "premiumPct must be >= 0" };
  if (typeof hedged !== "boolean") return { ok: false, reason: "hedged must be set on the bar" };
  if (!(sharePct >= 0 && sharePct <= 1)) return { ok: false, reason: "sharePct must be within 0..1" };
 
  const premium = premiumPct / 100;
 
  // The bar at THEIR premium over TODAY's spot. Doubles as the un-hedged
  // principal — see the algebra in the header.
  const revaluedBasisPerG = spotAtSale * (1 + premium);
 
  // Clamped at zero: fires only when we sell BELOW their locked premium, which
  // is the one case the participant is protected from.
  const netProfitPerG = Math.max(0, salePricePerG - revaluedBasisPerG);
 
  const participantSharePerG = netProfitPerG * sharePct;
 
  // NEVER clamped. Under the un-hedged model this can be less than they paid;
  // that is the risk they chose and the agreement must say so in those words.
  const principalPerG = hedged ? spotAtEnrol * (1 + premium) : revaluedBasisPerG;
 
  const investedPerG = spotAtEnrol * (1 + premium);
  const invested     = investedPerG * grams;
  const principal    = principalPerG * grams;
 
  const participantShare = cents(participantSharePerG * grams);
  const platformShare    = cents(netProfitPerG * (1 - sharePct) * grams);
  const participantTotal = cents(principal + participantSharePerG * grams);
 
  // ⚠️ MAY BE NEGATIVE, deliberately. The legacy `effectiveYieldPct` could not
  // be, because it divided clamped earnings by a cost — so in the un-hedged
  // spot-down case it reported +0.71% while the participant was actually
  // −0.40%. That is the most misleading number the simulator could carry on a
  // screen where someone is choosing to take spot risk.
  const returnPct = invested > 0 ? ((principal + participantSharePerG * grams) - invested) / invested * 100 : 0;
 
  return {
    ok: true,
    settlement: {
      revaluedBasisPerG,
      netProfitPerG,
      principalPerG,
      participantSharePerG,
      participantShare,
      platformShare,
      principal: cents(principal),
      participantTotal,
      investedPerG,
      invested: cents(invested),
      returnPct,
    },
  };
}
 
/**
 * Projection form: the simulator knows the buyer's PREMIUM (from the curve),
 * not a concrete sale price. Converts, then defers to the one rule above so a
 * projection can never diverge from a settlement.
 */
export function projectSale(
  input: Omit<SettlementInput, "salePricePerG"> & { buyerPremiumPct: number },
): SettlementResult {
  const { buyerPremiumPct, ...rest } = input;
  if (!Number.isFinite(buyerPremiumPct) || buyerPremiumPct < 0) {
    return { ok: false, reason: "buyerPremiumPct must be >= 0" };
  }
  if (!isPositive(rest.spotAtSale)) return { ok: false, reason: "spotAtSale must be > 0" };
  return settleSale({ ...rest, salePricePerG: rest.spotAtSale * (1 + buyerPremiumPct / 100) });
}
 
/**
 * The premium a bar was bought at, recovered from what was paid.
 *
 * ⚠️ The divisor is the ENROLMENT spot, never today's. Dividing by today's spot
 * is precisely the defect that made the Yield Simulator compute the
 * frozen-dollar model while its own banner told the participant their bar was
 * hedged — see docs/gold-yield-redesign.md §5.1.
 */
export function premiumPctFromPaid(paidPerG: number, spotAtEnrol: number): number | null {
  if (!isPositive(paidPerG) || !isPositive(spotAtEnrol)) return null;
  const pct = (paidPerG / spotAtEnrol - 1) * 100;
  return Number.isFinite(pct) ? pct : null;
}
 
/**
 * What this bar owes its participant: their profit share PLUS the principal
 * returned as grams sold.
 *
 * ▶ ONE definition, used by the exit gate AND by the exit-complete message.
 * They were computed independently under the same variable name and drifted
 * the moment the gate learned about principal: the gate summed both, the
 * message summed the profit share alone — and the template's variable is
 * documented as `payoutSgd`, not "earnings". On a hedged bar the principal is
 * the larger of the two by an order of magnitude, so the participant was told
 * they had been paid a fraction of what they actually received.
 *
 * ▶ Returns NaN — never a number — when any input is missing or unparseable,
 * via `settlementNumber`, this codebase's existing vocabulary for "unknown".
 * An earlier version of THIS comment claimed a null principal was "not coerced
 * to zero". That was false of the code: `Number(null)` is 0, so a missing
 * principal silently vanished from the total, in the one direction that
 * understates what is owed. Both callers now have to decide what to do with an
 * unknown, and both refuse to state a figure rather than state a wrong one.
 *
 * ▶ IT LIVES HERE RATHER THAN IN THE ROUTE SO IT CAN BE TESTED. It was a
 * module-scope helper inside `api/admin/gold-yield/bars/[id]/route.ts`, and
 * `route-exports-valid.test.ts` forbids a route exporting anything but
 * handlers + segment config — so there was no way to assert on it at all. The
 * exit gate it feeds has now carried four separate defects (a reserved-row
 * window, a per-user payout denominator, a coerced null principal, and a
 * present-but-unparseable share passing `NaN > 0`), every one of them pushing
 * the same way: pending too small, exit through, participant still owed. A
 * function with that history should not be untestable.
 */
export type GyOwedRow = {
  /**
   * ▶ A DOLLAR TOTAL FOR THAT SALE, NOT A PER-GRAM RATE — the two terms below
   * are added, so getting this wrong would silently under-count by a factor of
   * `grams_sold`. `settleSale` writes it as `cents(participantSharePerG *
   * grams)` (settlement.ts:195) and the accrual persists that value verbatim
   * (inventory-fulfilment.ts:876). The sibling `principal_per_g` IS per-gram,
   * which is exactly why the column names have to be read rather than assumed:
   * one row carries both conventions.
   */
  participant_share: number | string;
  /** Per-gram principal for that sale — multiplied by `grams_sold` below. */
  principal_per_g:   number | string | null;
  grams_sold:        number | string;
};
 
export function barOwedToParticipant(rows: GyOwedRow[] | null | undefined): number {
  return (rows ?? []).reduce(
    (s, r) => s + settlementNumber(r.participant_share)
                + settlementNumber(r.principal_per_g) * settlementNumber(r.grams_sold), 0);
}
 
/**
 * The money story of ONE bar, from the sale rows it has accrued.
 *
 * ▶ WHY THIS EXISTS. The admin panel showed a bar's `participant_earned_sgd`
 * and `platform_earned_sgd` and nothing else — no cost basis, no sale value, no
 * principal — so a settled bar rendered as "earned 0.00 / 0.00" with no way to
 * see whether that was a defect or the truth. It was the truth (operator,
 * 2026-09-21: "no details of the participant earning, platform earning and the
 * participant principal ... in both the admin screen and customer dashboard,
 * it's very unclear in terms of cost of acquiring and value of sale and thus
 * the profit (50/50+principal)"), but a screen that cannot distinguish a real
 * zero from a broken accrual is not telling anyone anything.
 *
 * Both surfaces now render THIS, so the participant and the operator are
 * reading one arithmetic rather than two.
 *
 * ▶ THE FIGURE THAT MATTERS MOST IS `net_position_sgd` — what they get back
 * minus what they put in. Earnings alone can be a perfectly healthy 0.00 while
 * the participant is down on the trade, because on an un-hedged bar the
 * principal is revalued at today's spot. Showing the two halves without their
 * difference is how that stays invisible.
 *
 * ▶ NULL, NEVER 0, when a term cannot be computed — `settlementNumber`'s
 * "unknown" vocabulary, carried through. A legacy sale row with no
 * `principal_per_g` makes the principal and everything downstream of it
 * unknown; the caller renders "—" and says so, rather than publishing a
 * confident figure built on a missing input.
 */
export type GyBarSaleRow = {
  grams_sold:         number | string;
  sale_price_per_g:   number | string | null;
  bar_cost_per_g?:    number | string | null;
  principal_per_g:    number | string | null;
  participant_share:  number | string;
  platform_share:     number | string;
};
 
export type GyBarSettlement = {
  grams_sold:             number;
  /** What the participant paid for the grams that have since sold. */
  cost_of_sold_grams_sgd: number | null;
  /**
   * What the participant ACTUALLY PAID for the grams that have sold — their
   * bar's basis (already net of promo, see enrol-cost.ts) × grams sold.
   *
   * ▶ NOT the same as `cost_of_sold_grams_sgd`, and conflating them was a bug.
   * That figure is the REVALUED basis the sale settled against (their premium
   * over spot-at-sale); this one is what left their pocket at enrolment. The two
   * differ by the whole spot move.
   */
  invested_sgd:           number | null;
  /** Gross proceeds of those grams. */
  sale_value_sgd:         number | null;
  /** The pot that gets split — sale value less the revalued basis and costs. */
  net_profit_sgd:         number;
  participant_share_sgd:  number;
  platform_share_sgd:     number;
  /** Their own capital coming back. Null if any sale lacks a principal. */
  principal_returned_sgd: number | null;
  /**
   * Principal back MINUS what they paid for those grams — the spot movement
   * they carried, on its own.
   *
   * ▶ WHY IT IS A FIELD RATHER THAN A SUBTRACTION AT THE RENDER SITE. Operator,
   * 2026-09-22: "the last row 'Participant net' is a bit confusing … the
   * participant should be able to clearly see what came back to them from their
   * principal amount in both cases of hedged v/s not-hedged and what they made
   * in profit from premium."
   *
   * `net_position_sgd` is two unrelated quantities added together:
   *
   *     net_position = share + (principal − invested)
   *                      ↑            ↑
   *              premium profit   spot movement on capital
   *
   * On a HEDGED bar `principal == invested`, so this term is zero and
   * `net_position` is numerically IDENTICAL to `participant_share_sgd` — a
   * figure presented as a conclusion that merely restates a row already shown.
   * On an UN-HEDGED bar the two differ and nothing said by how much. Splitting
   * the term out is what makes the hedge visible as a number rather than a badge.
   *
   * ▶ DERIVED FROM THE ROUNDED INPUTS, deliberately. These three figures are
   * rendered as a column that must add up on screen; deriving from the raw sums
   * lets a half-cent make the column appear not to total. Rounding once at the
   * end is right for a lone figure and wrong for one that has to reconcile.
   *
   * Null, never 0, when either input is unknown — a zero would assert "spot did
   * not move", which is a claim, not an absence.
   */
  principal_movement_sgd: number | null;
  /** Principal + their half — the cash they actually receive. */
  total_owed_sgd:         number | null;
  /**
   * Received − paid, i.e. `participant_share_sgd + principal_movement_sgd`.
   * Negative is a real outcome, not an error.
   */
  net_position_sgd:       number | null;
  avg_sale_price_per_g:   number | null;
  avg_cost_per_g:         number | null;
};
 
export function summariseBarSettlement(
  rows: GyBarSaleRow[] | null | undefined,
  /**
   * `gold_yield_bars.bar_cost_per_g` — what the participant PAID per gram, net
   * of promo. TWO jobs, deliberately one parameter:
   *
   *   1. the `invested_sgd` basis, which nothing on a sale row carries (a sale
   *      row's `principal_per_g` equals it on a HEDGED bar but not an un-hedged
   *      one, where principal is revalued at spot-at-sale);
   *   2. the fallback settled basis for a legacy row with no `bar_cost_per_g`
   *      of its own — under the retired frozen-dollar model the paid figure WAS
   *      the settled basis, so the same value is correct for both.
   *
   * Omitted → `invested_sgd` and `net_position_sgd` are null rather than wrong.
   */
  barPaidPerG?: number | string | null,
): GyBarSettlement {
  const sales = rows ?? [];
 
  let grams = 0;
  let saleValue = 0;
  let costOfSold = 0;
  let principal = 0;
  let participant = 0;
  let platform = 0;
 
  for (const r of sales) {
    const g = settlementNumber(r.grams_sold);
    grams += g;
    saleValue  += settlementNumber(r.sale_price_per_g) * g;
    // The sale's OWN frozen cost basis where it has one — a later change to the
    // bar must not rewrite the economics of a sale already settled.
    costOfSold += settlementNumber(r.bar_cost_per_g ?? barPaidPerG) * g;
    principal  += settlementNumber(r.principal_per_g) * g;
    // These two are DOLLAR TOTALS on the row, not per-gram rates — see GyOwedRow.
    participant += settlementNumber(r.participant_share);
    platform    += settlementNumber(r.platform_share);
  }
 
  const fin = (n: number): number | null => (Number.isFinite(n) ? Number(n.toFixed(2)) : null);
 
  const principalOut = fin(principal);
  const participantOut = Number.isFinite(participant) ? Number(participant.toFixed(2)) : 0;
  const costOut = fin(costOfSold);
  const owed = principalOut === null ? null : Number((principalOut + participantOut).toFixed(2));
 
  // ▶ WHAT THEY PUT IN, not what the sale settled against — this is the fix.
  //
  // `net_position_sgd` used to be `owed − costOfSold`, and `costOfSold` is the
  // REVALUED basis (their premium over spot-at-sale). On a HEDGED bar in a
  // rising market that made the headline figure wrong by the entire spot move,
  // in the direction that alarms the participant:
  //
  //   paid 102/g (spot 100, premium 2%), spot rises to 110, buyer pays 115.50
  //     revalued basis  110 × 1.02 = 112.20   → cost_of_sold
  //     profit          115.50 − 112.20 = 3.30 → participant 1.65
  //     principal (hedged) 100 × 1.02 = 102
  //     owed            103.65
  //     OLD net_position 103.65 − 112.20 = −8.55   ← they are actually +1.65
  //
  // The module header calls this "the figure that matters most"; it has to be
  // measured against the number the participant recognises, which is the one
  // they paid.
  const investedPerG = settlementNumber(barPaidPerG);
  const invested = Number.isFinite(investedPerG) ? investedPerG * grams : NaN;
  const investedOut = fin(invested);
 
  // ▶ FROM THE ROUNDED PAIR, so the rendered column reconciles. See the field's
  // own doc comment: these three lines are shown together as
  // `share + movement = net`, and a raw-sum derivation can leave them a cent
  // apart on screen for no reason the reader can see.
  const movementOut =
    principalOut === null || investedOut === null
      ? null
      : Number((principalOut - investedOut).toFixed(2));
 
  return {
    grams_sold:             Number(grams.toFixed(4)),
    cost_of_sold_grams_sgd: costOut,
    sale_value_sgd:         fin(saleValue),
    net_profit_sgd:         Number(((Number.isFinite(participant) ? participant : 0) + (Number.isFinite(platform) ? platform : 0)).toFixed(2)),
    participant_share_sgd:  participantOut,
    platform_share_sgd:     Number.isFinite(platform) ? Number(platform.toFixed(2)) : 0,
    principal_returned_sgd: principalOut,
    principal_movement_sgd: movementOut,
    total_owed_sgd:         owed,
    invested_sgd:           investedOut,
    net_position_sgd:       owed === null || investedOut === null ? null : Number((owed - investedOut).toFixed(2)),
    avg_sale_price_per_g:   grams > 0 ? fin(saleValue / grams) : null,
    avg_cost_per_g:         grams > 0 ? fin(costOfSold / grams) : null,
  };
}
 
/**
 * The same total as `barOwedToParticipant`, but split into what CAN be computed
 * and what is blocking a firm answer.
 *
 * ▶ WHY BOTH EXIST. `barOwedToParticipant` is deliberately all-or-nothing: it
 * returns NaN if any sale lacks a principal, and the exit gate refuses on that
 * — correct, because an exit recorded as settled while a participant is still
 * owed is unrecoverable. But the admin earnings screen inherited the same
 * all-or-nothing and rendered the participant's WHOLE balance as "—" because a
 * handful of legacy rows predate the `principal_per_g` column. One unknowable
 * row erased S$121.22 that is perfectly well known, on the screen an operator
 * reads to decide what to pay.
 *
 * So the screen gets the known part plus an explicit count of what is missing —
 * "at least X, and N sales have no recorded principal" — while `pending_sgd`
 * stays null so the payout form still refuses to prefill. Honest about the
 * floor, honest about the gap, and the existing safety is untouched: the
 * operator can see the money without the form quietly proposing to underpay.
 */
export type GyOwedBreakdown = {
  /** Sum over the rows whose principal IS recorded. A floor, not the answer. */
  computable_sgd: number;
  /**
   * `computable_sgd` split into the two things a participant is owed, because
   * they answer different questions and the operator asked for both to be shown
   * (2026-09-22: "no clarity for the participant on what are they getting back
   * in payout (how much principal and how much profits)").
   *
   *   profit_sgd    — their half of the premium spread. Money they MADE.
   *   principal_sgd — their own capital coming back. Money they PUT IN.
   *
   * ▶ THEY SUM TO `computable_sgd` BY CONSTRUCTION — same loop, same
   * finite-ness branches — so a screen can show the split beside the total
   * without the three figures ever disagreeing.
   *
   * ▶ THIS IS THE SPLIT OF WHAT IS OWED, NOT OF WHAT IS PENDING. Once a part
   * payment exists there is no honest way to say which half it settled: any
   * allocation (profit-first, pro-rata) would be an invention presented as a
   * fact. Callers show the split against the total owed and report payments
   * separately.
   */
  profit_sgd:     number;
  principal_sgd:  number;
  /** Rows with no usable principal — the reason a firm total cannot be given. */
  unknown_rows:   number;
  /** Grams sitting behind those rows, so the gap has a magnitude. */
  unknown_grams:  number;
  /** True when nothing was missing, i.e. `computable_sgd` IS the whole answer. */
  complete:       boolean;
};
 
export function owedBreakdown(rows: GyOwedRow[] | null | undefined): GyOwedBreakdown {
  let computable = 0;
  let profit = 0;
  let principalTotal = 0;
  let unknownRows = 0;
  let unknownGrams = 0;
 
  for (const r of rows ?? []) {
    const share     = settlementNumber(r.participant_share);
    const grams     = settlementNumber(r.grams_sold);
    const principal = settlementNumber(r.principal_per_g);
    const rowTotal  = share + principal * grams;
 
    if (Number.isFinite(rowTotal)) {
      computable     += rowTotal;
      profit         += share;
      principalTotal += principal * grams;
      continue;
    }
 
    unknownRows += 1;
    Eif (Number.isFinite(grams)) unknownGrams += grams;
 
    // ▶ A FLOOR MUST CARRY EVERY TERM WE ARE CERTAIN OF, TERM BY TERM — not
    // just every row. Discarding the whole row when ONE term is unknown threw
    // away S$121.22 of participant share on DEV that is not in doubt at all,
    // purely because the principal beside it predates the column. The share is
    // owed whether or not we can price the capital, so it belongs in the floor;
    // only the unknown term is left out.
    // The principal term is deliberately NOT added here even when it alone is
    // finite — that is the pre-existing contract this split has to preserve, so
    // `profit_sgd + principal_sgd` keeps equalling `computable_sgd`.
    if (Number.isFinite(share)) {
      computable += share;
      profit     += share;
    }
  }
 
  return {
    computable_sgd: Number(computable.toFixed(2)),
    profit_sgd:     Number(profit.toFixed(2)),
    principal_sgd:  Number(principalTotal.toFixed(2)),
    unknown_rows:   unknownRows,
    unknown_grams:  Number(unknownGrams.toFixed(4)),
    complete:       unknownRows === 0,
  };
}
