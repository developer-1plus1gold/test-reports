// Map the courier statuses ALREADY written (Lalamove + NinjaVan) onto the
// existing 5-step bar (Ordered · Arranging · Picked Up · Out for Delivery ·
// Delivered). Previously only picked_up/delivered were mapped, so every
// intermediate status collapsed to "Ordered" and the bar jumped (audit D11).
export function deliveryCompletedCount(status: string | null, shipper: string | null): number {
  switch (status) {
    case "delivered":        return 5;
    case "out_for_delivery": return 4;
    case "picked_up":
      // Lalamove is same-day: once picked up it's effectively out for delivery
      // (it emits no separate out_for_delivery). NinjaVan does emit that step,
      // so a NinjaVan pickup stays at "Picked Up" until out_for_delivery.
      return shipper === "lalamove" ? 4 : 3;
    case "on_going":         // Lalamove: driver en route
    case "assigning_driver": // Lalamove: finding a driver
    case "in_transit":       // NinjaVan: moving through the network
    case "booked":           // NinjaVan: booking accepted
      return 2;              // Arranging
    default:                 return 1; // Ordered (pending / unknown)
  }
}
 
/**
 * The furthest step this delivery has EVER reached, not the one it is on.
 *
 * A carrier can move a parcel BACKWARDS. SPX does it routinely: a parcel dropped
 * at one of their service points is marked "In Transit" (2001) on receipt, and if
 * their own logistics partner then fails to collect it from that point, SPX emits
 * "Pending Pickup" (1001) again — observed on SPXSG063526130219 (2026-09-11) and
 * SPXSG061644870198 (2026-08-29), both collected the next morning.
 *
 * The ADMIN view mirrors that faithfully, because the operator needs carrier truth.
 * The CUSTOMER bar must not: a progress bar that goes backwards reads as "something
 * happened to my order", and the customer can neither see why nor act on it. So the
 * customer's step is monotonic — it holds at the furthest rung reached, and the
 * exception is explained by pickupExceptionNote() instead of by a silent regression.
 *
 * Happy path is unaffected: with no regression in the history, the furthest step IS
 * the current step, so this returns exactly what deliveryCompletedCount() returns.
 */
export function deliveryCompletedCountMonotonic(
  status:  string | null,
  events:  { status: string; at: string }[] | null | undefined,
  shipper: string | null,
): number {
  const current = deliveryCompletedCount(status, shipper);
  let furthest = current;
  for (const e of events ?? []) {
    const n = deliveryCompletedCount(e.status ?? null, shipper);
    if (n > furthest) furthest = n;
  }
  return furthest;
}
 
/**
 * Customer-facing explanation for a parcel the carrier has un-picked-up, or null.
 *
 * Returns a string ONLY in the one exception we have actually seen: the parcel
 * reached "picked up" and the carrier has since moved it back below that. On the
 * happy path this is null and nothing about the order row changes.
 *
 * It is DERIVED, never stored — so it appears the moment the carrier regresses and
 * clears itself the moment they move the parcel forward again. Nothing to reset, no
 * cleanup job, and no way for a stale note to outlive the condition it describes.
 *
 * Scoped to SPX because the copy names their drop-off point. Another carrier
 * regressing would hold the bar (above) but say nothing, which is the safe default:
 * better silent than wrong about someone else's process.
 */
export function pickupExceptionNote(
  status:  string | null,
  events:  { status: string; at: string }[] | null | undefined,
  shipper: string | null,
): string | null {
  if (shipper !== "spx") return null;
  const pickedUp = deliveryCompletedCount("picked_up", shipper);
  const everPickedUp = (events ?? []).some(
    e => deliveryCompletedCount(e.status ?? null, shipper) >= pickedUp,
  );
  if (!everPickedUp) return null;
  if (deliveryCompletedCount(status, shipper) >= pickedUp) return null;  // moved on again
  return "Your parcel is at the SPX drop-off point. SPX didn't collect today — collection usually follows the next working day.";
}

/** Lowercased haystack the Order History search box matches against — id8,

/** Layer Gold-Yield flags onto a derived row. Kept separate from
 *  deriveOrderRow so the core view-model stays GY-agnostic. */
export function enrichGoldYield(
  row: OrderRowVM,
  gyEnrolled: boolean,           // canonical enrolled status (participant + current agreement)
  enrolledItemIds: Set<string>,
): OrderRowVM {
  const enrollable = ["completed", "paid", "delivered"].includes(row.status);
  const items = row.items.map((it) => ({
    ...it,
    // Eligible to enrol = enrolled in the programme AND a gy_eligible product
    // (flag-keyed via products.gy_eligible — single source with cart/checkout;
    // replaces the old [50,100].includes(grams) inference). #2 2026-06-12.
    gyEligible:   gyEnrolled && it.gyEligibleProduct === true,
    gyEnrolled:   enrolledItemIds.has(it.id),
    orderId:      row.id,
    gyEnrollable: enrollable,
  }));
  return {
    ...row,
    items,
    gyEnrollable:   enrollable,
    hasEnrolledBar: gyEnrolled && items.some((it) => it.gyEnrolled),
  };
}

  // marketplace/seller, NOT our own dispatch — the buyer already received the
  // item through the marketplace before ever claiming it here. Our 5-step
  // courier journey (Arranging → Picked Up → Out for Delivery → Delivered)
  // describes a delivery we didn't run, and a claimed order often carries no /
  // partial shipper_status, so the bar contradicts the status pill (a
  // "Completed" order rendering "Arranging"). Suppress the tracker; the row still
  // shows the status pill, the channel chip ("Shopee"), items, and address.
  if (o.source_channel && isExternalChannel(o.source_channel)) return null;
  // A cancelled/refunded order has no live delivery — don't show a progress bar
  // frozen at "Arranging"/"Out for delivery" (cancellation teardown, item v).
  if (o.status === "cancelled" || o.status === "refunded") return null;
  const isSelfPickup = o.shipper === "self_pickup";
  const isVault      = o.shipper === "vault";
  const steps     = isVault ? VAULT_STEPS : isSelfPickup ? SELF_PICKUP_STEPS : DELIVERY_STEPS;
  const completed = isVault
    ? vaultCompletedCount(o.shipper_status)
    : isSelfPickup
    ? selfPickupCompletedCount(o.shipper_status)
    // Monotonic on purpose — the customer bar must not follow a carrier rollback
    // backwards. See deliveryCompletedCountMonotonic().
    : deliveryCompletedCountMonotonic(o.shipper_status, o.shipper_status_events, o.shipper);
  const lastStepIdx = steps.length - 1;
  const fillPct     = completed > 1 ? ((completed - 1) / lastStepIdx) * 100 : 0;
  const isFailed    = !isSelfPickup && !isVault && o.shipper_status === "failed" && !!o.shipper_order_id;
  const events      = o.shipper_status_events ?? [];
  const stepTimestamps = steps.map((_, i) =>
    (isSelfPickup || isVault)
      // step 0 = order date; vault has no pod_at so its "Stored" step carries no
      // precise timestamp (renderer handles null, as courier steps do).
      ? (i === 0 ? o.created_at : (isVault ? null : o.pod_at ?? null))
      : deliveryStepTimestamp(events, i, o.created_at, o.pod_at),
  );
  return { variant: isVault ? "vault" : isSelfPickup ? "self_pickup" : "courier", steps, completed, lastStepIdx, fillPct, isFailed, stepTimestamps };
}
