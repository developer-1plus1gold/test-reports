/**
 * The My Account review prompt — who sees it, and when it stops.
 *
 * ▶ THE SHAPE OF THE ASK WAS THE OPERATOR'S DECISION, and it rules out the
 * obvious implementations. Mukesh, 2026-09-16, on why there is no WhatsApp
 * campaign: "I do not want to be like a spam on whatsapp, as people treat it
 * very personally and in frustration end up locking the chats." And on the
 * account surface instead: "while the person is browsing through their account,
 * they are at a calm state and while seeing their data, can possibly feel like
 * leaving a review. Overall, we do not want to force the reviews, we want it to
 * come genuinely from our customers."
 *
 * So, concretely: an inline CARD in the dashboard's own rhythm — never a modal,
 * never a popup, never an interstitial, and nothing that fires on a timer or on
 * page load. The customer is already looking at their own gold; the ask sits
 * beside it and waits.
 *
 * ▶ DISMISSAL IS PERMANENT. A prompt that reappears after a month is the
 * nagging that was ruled out, just on a slower clock. One "not now" is an
 * answer, and we take it.
 *
 * ▶ ELIGIBILITY IS DELIVERY, NOT PURCHASE. Asking at checkout would be asking
 * someone to review a parcel they have not opened. `deriveLifecycle` is the
 * single definition of "completed" in this codebase and already knows the thing
 * that catches people out — a delivered order very often still reads
 * `status: "paid"`, so the carrier state is what settles it.
 *
 * ▶ WHAT THIS DELIBERATELY DOES NOT DO: branch on sentiment. There is no
 * "were you happy?" step, and the review link is shown to every eligible
 * customer regardless of anything we know about their orders. That is not
 * squeamishness — selectively soliciting only customers you expect to be happy
 * ("review gating") is against Trustpilot's own guidelines, and it is also the
 * thing that makes a review score worthless. The card offers BOTH a review link
 * and a "tell us first" contact link, unconditionally and side by side; that is
 * a choice we give everyone, not a funnel that sorts them.
 */
import { deriveLifecycle } from "@/lib/orders/lifecycle";
 
/** `ask` — show the card. `none` — show nothing at all. */
export type ReviewPromptState = "ask" | "none";
 
export interface ReviewPromptInput {
  /** Every order belonging to this customer — status + carrier status only. */
  orders: Array<{ status: string | null; shipper_status?: string | null }>;
  /** `profiles.review_prompt_dismissed_at`. NULL = never dismissed. */
  dismissedAt: string | null;
}
 
export interface ReviewPromptView {
  state: ReviewPromptState;
  /** How many orders have actually reached the customer. Surfaced for the
   *  copy's sake and because "why am I seeing this?" should be answerable. */
  deliveredCount: number;
}
 
export function deriveReviewPrompt({ orders, dismissedAt }: ReviewPromptInput): ReviewPromptView {
  const deliveredCount = orders.filter(
    (o) => deriveLifecycle(o.status ?? "", o.shipper_status ?? null) === "completed",
  ).length;
 
  // Dismissal wins over eligibility: once answered, the question is closed.
  const state: ReviewPromptState = dismissedAt ? "none" : deliveredCount > 0 ? "ask" : "none";
  return { state, deliveredCount };
}
 
/**

 * SDUI card and the printed insert card cannot drift into three different
 * voices. It is the same sentence the parcel card carries, on purpose: the card
 * in the box and the card in the account should sound like the same company.
 */

  body:      "If we did well by you, a couple of lines helps the next person decide.",
  /**
   * For a surface where the platform is NOT otherwise named — the email, which
   * renders this as plain text with no artwork beside it.
   */
  cta:       "Write a review on Trustpilot →",
  /**
   * For a surface that draws Trustpilot's OWN lockup beside it — the account

   *

  ctaWithMark: "Write a review →",

/**
 * The `/verify` line — the same ask, at a different moment, in the same voice.
 *
 * ▶ WHY IT IS A SEPARATE CONSTANT RATHER THAN A REUSE. The account card speaks
 * to someone reviewing their own holdings; this speaks to whoever is holding
 * the card, which is often NOT the buyer — a gift recipient, or a marketplace
 * buyer who has not claimed yet. "If we did well by you" presumes a purchase
 * they may not have made. The design doc's own sentence for this surface is
 * `Glad it checks out? Tell someone.`, and it earns its place by naming the
 * thing that just happened rather than the relationship.
 *
 * ▶ WHY IT LIVES IN AN `account/` MODULE. This file is the review-copy module;
 * the folder name is where it started, not what it is. Keeping every rendered
 * review sentence in one file is the whole point of the note above — a fourth
 * surface with its own copy elsewhere is exactly the drift that note prevents.
 *
 * ▶ THE DISCLOSURE IS NOT OPTIONAL and is deliberately the account card's
 * wording verbatim. Trustpilot is independent; a customer clicking from our own
 * certificate page should be told they are leaving it, in the same words we use
 * everywhere else.
 */
/**
 * The humble ask itself — ONE sentence, shared by the pushed message and the
 * `/verify` card, because the operator asked for exactly that on 2026-09-18:
 * *"also the text in /verify for trustpilot - make it similar to what we are
 * sending in the whatsapp message."*
 *
 * ▶ WHY IT IS A CONSTANT AND NOT A COPIED STRING. It is now rendered on two
 * surfaces that a single customer can meet within minutes of each other — the
 * WhatsApp message about their delivery, and the certificate page they scan off
 * the card in the same box. Two near-identical sentences that drift apart read
 * as two different companies, which is the exact failure this module's header
 * exists to prevent.
 *
 * ▶ AND IT IS THE ONE HALF OF THE MESSAGE THAT PORTS. The rest of the WhatsApp
 * body is order-specific (*"your order {{2}} has been delivered"*) and cannot
 * appear here: `/verify` is often read by somebody who did NOT buy — a gift
 * recipient, or a marketplace buyer who has not claimed yet — so presuming a
 * purchase would be wrong on this surface. This sentence presumes nothing.
 */

  "If you have a moment, we would be grateful for a few words — it helps us know how we are doing.";

/**
 * The post-delivery review REQUEST — the one message we push, across both
 * channels, in the same voice.
 *
 * ▶ REWRITTEN 2026-09-18 ON THE OPERATOR'S INSTRUCTION, and the change is the
 * point: *"I don't want to be defensive for contact us first. It should be
 * simple providing the review url only… keep the request very simple and about
 * us (for us to know how we are doing) and overall tone to be humble request."*
 *
 * So the earlier two-sided shape — "if anything is not right, tell us first at
 * /contact" ahead of the ask — is GONE from the pushed message. What replaced
 * it is shorter, asks for one thing, and frames the reason as OURS (knowing how
 * we are doing) rather than as a service to a future stranger.
 *
 * ▶ WHAT THAT DOES AND DOES NOT CHANGE, stated so nobody re-adds the old line
 * believing it was load-bearing. It was originally justified as what kept this
 * from being REVIEW GATING. On re-reading Trustpilot's rule that was wrong:
 * gating is selectively soliciting only the customers you expect to be happy,
 * and we ask EVERY delivered customer — no sentiment branch, enforced in code
 * and asserted by test. Removing a sentence from the body cannot create gating.
 * The real trade is a business one the operator took knowingly: an unhappy
 * customer now goes straight to a public review rather than to us first.
 *
 * ▶ THE "TELL US FIRST" ROUTE STILL EXISTS, just not as an interruption — the
 * account card carries it unconditionally beside the review link
 * (REVIEW_PROMPT_COPY.fallback), which is a place someone chose to visit.
 *
 * ▶ `whatsappBody` IS DOCUMENTATION, NOT THE SEND. Once Meta approves it the
 * authoritative body lives in Twilio's Content API and we supply only the
 * variables — editing this string then changes nothing that goes out. It is
 * here so the email can be written to match, and so a guard can assert the two
 * channels say the same thing rather than drifting into two companies.
 *
 * ▶ META VARIABLE RULES ARE BAKED INTO THE STRING, not left to the submitter:
 * `{{1}}` is not at the start (the body opens with "Hi ") and the body ends on
 * static text, because Meta rejects a template whose body begins or ends with a
 * variable — this repo has already eaten that rejection once
 * (PHONE_CHANGE_REJECTED v1). The URL is STATIC rather than a variable: fewer
 * placeholders is fewer ways to be rejected, and `/review` is a permanent
 * vanity redirect (next.config.ts, pinned by vanity-redirects.test.ts).
 *
 * ▶ THE ORDER REFERENCE STAYS. It is what makes this a post-delivery check-in
 * about a NAMED order rather than a marketing blast — which is both the honest
 * reading and the one that fits Meta's UTILITY category.
 */

  whatsappBody:

  ask:        REVIEW_ASK_SENTENCE,
  cta:        REVIEW_PROMPT_COPY.cta,
  disclosure: REVIEW_PROMPT_COPY.disclosure,
} as const;
