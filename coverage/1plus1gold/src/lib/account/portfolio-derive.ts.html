/**
 * Portfolio derivation — the dashboard-specific math that sits on top of the
 * already-shared aggregators (`aggregateOrders` from order-aggregations,
 * `nextPurchaseRate` from gold). EXTRACTED (moved, not copied) from the inline
 * computation in src/app/account/page.tsx 2026-06-01 so the PWA dashboard page
 * AND the new /api/account/dashboard aggregator consume ONE implementation —
 * no drift (the SDUI-default rule's "expose, don't duplicate" discipline).
 *
 * Pure function: same raw inputs → same numbers. Data FETCHING stays with each
 * caller (the PWA fetches via the browser supabase client; the API via
 * service-role). Only the math lives here.
 */
 
import { aggregateOrders, PAID_STATUSES, type OrderAgg } from "@/lib/order-aggregations";
import { nextPurchaseRate } from "@/lib/gold";
 
export const CURVE_MILESTONE = 50;
 
/** Minimal order shape the derivation needs (superset of OrderAgg). */
export interface DeriveOrder extends OrderAgg {
  shipping_fee_sgd?:   number | string | null;
  processing_fee_sgd?: number | string | null;
}
 
export interface DeriveBuyback {
  status:             string;
  paid_at?:           string | null;
  total_payout_sgd?:  number | string | null;
  shipping_fee_sgd?:  number | string | null;
  buyback_request_lines: { grams: number | string }[];
}
 
export interface DeriveConversion {
  status:               string;
  total_target_grams?:  number | string | null;
  target_100g_count?:   number | string | null;
  target_50g_count?:    number | string | null;
  total_charged_sgd?:   number | string | null;
  conversion_fee_sgd?:  number | string | null;
}
 
export interface DeriveInput {
  orders:              DeriveOrder[];
  /** Lifetime curve-only grams (customer_gold_holdings.curve_grams_purchased). */
  purchasedCurveGrams: number;
  /** customer_gold_holdings.yielded_grams (for the composition donut). */
  yieldedGrams:        number;
  buybacks:            DeriveBuyback[];
  conversions:         DeriveConversion[];
  /** position → premium% map for the next-purchase rate (optional). */
  premiumMap?:         Map<number, number>;
}
 
const num = (v: unknown): number => (v == null ? 0 : Number(v));
 
export function derivePortfolio(input: DeriveInput) {
  const { orders, purchasedCurveGrams, yieldedGrams, buybacks, conversions, premiumMap } = input;
 
  const completedOrders = orders.filter((o) => PAID_STATUSES.includes(o.status));
  const totalSgd            = completedOrders.reduce((s, o) => s + num(o.total_sgd), 0);
  const totalShippingFees   = completedOrders.reduce((s, o) => s + num(o.shipping_fee_sgd), 0);
  const totalProcessingFees = completedOrders.reduce((s, o) => s + num(o.processing_fee_sgd), 0);
 
  // ── Sell-to-Us ──────────────────────────────────────────────────────────
  const paidBuybacks = buybacks.filter((bb) => bb.status === "completed" || bb.status === "paid" || !!bb.paid_at);
  const pendingBuybackCount = buybacks.filter((bb) => ["submitted", "received", "approved"].includes(bb.status)).length;
  const buybackTotalGrams = paidBuybacks.reduce((s, bb) => s + bb.buyback_request_lines.reduce((g, l) => g + num(l.grams), 0), 0);
  const buybackTotalPayout = paidBuybacks.reduce((s, bb) => s + Math.max(0, num(bb.total_payout_sgd) - num(bb.shipping_fee_sgd)), 0);
  const buybackPickupFees = paidBuybacks.reduce((s, bb) => s + num(bb.shipping_fee_sgd), 0);
  const buybackAvgPerGram = buybackTotalGrams > 0 && buybackTotalPayout > 0 ? buybackTotalPayout / buybackTotalGrams : null;
 
  // ── Conversions ─────────────────────────────────────────────────────────
  const completedConversions = conversions.filter((c) => c.status === "completed" || c.status === "delivered");
  const pendingConversionCount = conversions.filter((c) => !["completed", "delivered", "rejected", "cancelled"].includes(c.status)).length;
  const convertedGrams = completedConversions.reduce((s, c) => s + num(c.total_target_grams), 0);
  const completed100gCount = completedConversions.reduce((s, c) => s + num(c.target_100g_count), 0);
  const completed50gCount  = completedConversions.reduce((s, c) => s + num(c.target_50g_count), 0);
  // "2 × 100g + 1 × 50g" — only non-zero parts, "—" when none. Derived here so
  // the PWA tile and the SDUI dashboard render one label (bindings can't do
  // the conditional join).
  const combibarsReceivedText = (() => {
    const parts: string[] = [];
    Iif (completed100gCount > 0) parts.push(`${completed100gCount} × 100g`);
    if (completed50gCount  > 0) parts.push(`${completed50gCount} × 50g`);
    return parts.length ? parts.join(" + ") : "—";
  })();
  const conversionFeesPaid = conversions
    .filter((c) => !["rejected", "cancelled"].includes(c.status))
    .reduce((s, c) => s + (c.total_charged_sgd != null ? num(c.total_charged_sgd) : num(c.conversion_fee_sgd)), 0);
 
  // ── Curve / Standard / Accessory (shared aggregator) ──────────────────────
  const aggregated = aggregateOrders(completedOrders);
  const completedItems = completedOrders.flatMap((o) => o.items);
 
  const standardGrams   = aggregated.byType.standard.grams;
  const standardSgd     = aggregated.byType.standard.revenue;
  const standardAvgCost = aggregated.byType.standard.avgCostPerGram;
  const standardAvgPrem = aggregated.byType.standard.avgPremium;
 
  const curveGrams   = purchasedCurveGrams;
  const curveSgd     = aggregated.byType.curve.revenue;
  const curveAvgCost = curveGrams > 0 && curveSgd > 0 ? curveSgd / curveGrams : null;
  const curveAvgPrem = aggregated.byType.curve.avgPremium;
 
  const accessoryRevenue = aggregated.byType.accessory.revenue;
  const accessoryItems   = aggregated.byType.accessory.itemCount;
  const accessoryOrders  = aggregated.byType.accessory.orderCount;
 
  const portfolioGrams = curveGrams + standardGrams;
  // Overall avg cost/gram (Overall Portfolio §3 tile). Derived here so the PWA
  // JSX and the SDUI dashboard read one value — no inline-arithmetic drift.
  const overallAvgCost = portfolioGrams > 0 && totalSgd > 0 ? totalSgd / portfolioGrams : null;
 
  // ── 50g cycle (lifetime — GY must not roll back loyalty) ──────────────────
  const curveCyclesCompleted = Math.floor(purchasedCurveGrams / CURVE_MILESTONE);
  const curvePositionInCycle = purchasedCurveGrams % CURVE_MILESTONE;
 
  // ── Combibar composition (splits vs already-sealed) ───────────────────────
  const already50Count = completedItems
    .filter((i) => num(i.grams_each) === 50 && (i.product_type ?? "curve") === "curve")
    .reduce((s, i) => s + num(i.quantity), 0);
  const already100Count = completedItems
    .filter((i) => num(i.grams_each) === 100 && (i.product_type ?? "curve") === "curve")
    .reduce((s, i) => s + num(i.quantity), 0);
  const physicalGrams = already50Count * 50 + already100Count * 100;
  const splitGrams = Math.max(purchasedCurveGrams - physicalGrams, 0);
 
  const blendedPremium = curveAvgPrem;
 
  const currentCurveRate = premiumMap && premiumMap.size
    ? nextPurchaseRate(curvePositionInCycle, premiumMap) * 100
    : null;
 
  return {
    // Hero / composition (chart-panel props + Overall)
    totalSgd, totalShippingFees, totalProcessingFees, portfolioGrams, yieldedGrams,
    // Curve journey
    curveGrams, curveSgd, curveAvgCost, curveAvgPrem, blendedPremium,
    curveCyclesCompleted, curvePositionInCycle, currentCurveRate,
    physicalGrams, splitGrams, already50Count, already100Count,
    overallAvgCost,
    // Standard
    standardGrams, standardSgd, standardAvgCost, standardAvgPrem,
    // Accessories
    accessoryRevenue, accessoryItems, accessoryOrders,
    // Sell-to-Us
    paidBuybackCount: paidBuybacks.length, pendingBuybackCount,
    buybackTotalGrams, buybackTotalPayout, buybackPickupFees, buybackAvgPerGram,
    // Conversions
    completedConversionCount: completedConversions.length, pendingConversionCount,
    convertedGrams, completed100gCount, completed50gCount, combibarsReceivedText, conversionFeesPaid,
  };
}
 
export type DerivedPortfolio = ReturnType<typeof derivePortfolio>;
