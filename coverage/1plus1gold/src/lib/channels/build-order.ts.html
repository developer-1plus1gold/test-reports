/**
 * PURE builder: NormalizedExternalOrder → create_order_atomic RPC input.
 *
 * No I/O — unit-testable in isolation (mirrors buildShopeeListings). The API route
 * resolves products (grams, stock) and the house customer, then hands a fully-formed
 * NormalizedExternalOrder here; this maps it to the exact { p_order, p_items } jsonb
 * the RPC expects. Externals are PRE-PAID (money collected by the marketplace) → they
 * enter status='paid', payment_status='paid', payment_method='external', and are
 * fulfilled by the marketplace (shipper='external').
 */
 
import type { NormalizedExternalOrder, ExternalOrderTotals } from "./types";
 
export interface OrderRpcInput {
  pOrder: Record<string, unknown>;
  pItems: Array<Record<string, unknown>>;
}
 
/** Round to 2dp SGD (money) — matches roundSgd used across the pricing kernel. */
function round2(n: number): number {
  return Math.round((n + Number.EPSILON) * 100) / 100;
}
 
/**
 * Resolve an external order's money into the two totals that actually differ.
 *
 * Worked example — the Shopee order that exposed this (2026-07-21):
 *   merchandise 246.00, seller voucher 10.00, absorbed shipping 4.29, fees 7.72
 *   → buyer paid  246.00 − 10.00            = 236.00   (Shopee's "Total Buyer Payment")
 *   → net payout  236.00 − 7.72 − 4.29      = 223.99   (Shopee's "Our Order Income")
 *
 * Only `buyerPaidSgd` may appear on a customer receipt. The voucher is visible to
 * the buyer (it reduced their bill); the fees and absorbed shipping are ours alone
 * and must never surface on their document.
 *
 * Exported so the admin form can show the same arithmetic live and the operator can
 * reconcile it against the marketplace screen before saving.
 */
export function computeExternalTotals(order: {
  grossTotalSgd: number;
  channelFeeSgd: number;
  sellerVoucherSgd?: number;
  absorbedShippingSgd?: number;
}): ExternalOrderTotals {
  const merchandiseSgd     = round2(order.grossTotalSgd);
  const voucherSgd         = round2(Math.max(0, order.sellerVoucherSgd ?? 0));
  const channelFeeSgd      = round2(Math.max(0, order.channelFeeSgd ?? 0));
  const absorbedShippingSgd = round2(Math.max(0, order.absorbedShippingSgd ?? 0));
  // A voucher larger than the merchandise would imply a negative charge; clamp
  // rather than emit a nonsense receipt.
  const buyerPaidSgd       = round2(Math.max(0, merchandiseSgd - voucherSgd));
  const netPayoutSgd       = round2(buyerPaidSgd - channelFeeSgd - absorbedShippingSgd);
 
  return { merchandiseSgd, voucherSgd, buyerPaidSgd, channelFeeSgd, absorbedShippingSgd, netPayoutSgd };
}
 
/** Gold (curve/standard) is investment precious metal → GST-exempt; everything
 *  else is taxable. Falls back to "has grams" when the caller didn't resolve the
 *  product type, which is true for every gold line. */
function isIpmLine(it: { productType?: string; gramsEach: number }): boolean {
  if (it.productType) return it.productType === "curve" || it.productType === "standard";
  return it.gramsEach > 0;
}
 
/** Today in Singapore as yyyy-mm-dd. Shared by the admin form (to default and
 *  bound the picker) and the route (to validate), so the two cannot disagree
 *  about what "today" means for an admin working late in another timezone. */
export function todaySgt(now: Date = new Date()): string {
  return now.toLocaleDateString("en-CA", { timeZone: "Asia/Singapore" });
}
 
/**
 * Resolve the operator's chosen order date into a `created_at` for the RPC.
 *
 * Returns undefined for "no opinion" — an omitted date, or a date that IS today.
 * Today is deliberately treated as no-opinion so a sale recorded at 18:00 is
 * stamped 18:00 rather than being flattened to a synthetic noon, which would
 * scramble the ordering of several sales entered on the same day.
 *
 * A genuinely backdated day is anchored at 12:00 SGT — the middle of the
 * Singapore business day the sale belongs to.
 *
 * Note what midday does and does NOT buy. It gives 12 hours of slack against a
 * mis-set clock or an off-by-a-few-hours offset, where 00:00 SGT would tip into
 * the previous day at the slightest nudge. It does NOT make the timestamp
 * render as the same calendar day everywhere: noon SGT is 04:00 UTC, so any
 * zone west of UTC-4 (most of the Americas) shows the previous date under a
 * naive local render. The invariant we actually rely on is that every surface
 * renders order dates with an explicit `timeZone: "Asia/Singapore"` — this is a
 * Singapore business and the SGT day is the business day. Keep that pinned in
 * any new renderer; the anchor alone will not save you.
 */
export function resolveOrderCreatedAt(
  orderDate: string | undefined,
  now: Date = new Date(),
): string | undefined {
  if (!orderDate) return undefined;
  if (orderDate === todaySgt(now)) return undefined;
  return `${orderDate}T12:00:00+08:00`;
}
 
export function buildExternalOrderRpcInput(order: NormalizedExternalOrder): OrderRpcInput {
  const totalGrams = order.items.reduce((sum, it) => sum + it.gramsEach * it.quantity, 0);
  const totals     = computeExternalTotals(order);
  const hasPresale = order.items.some((it) => it.isPresale === true);
 
  // IRAS IPM/taxable split. These columns — NOT total_sgd — are what the P&L
  // recognises revenue from, and externals left them null: an external sale
  // therefore contributed ZERO revenue while its COGS landed, showing a real
  // sale as a pure loss the moment its lifecycle reached 'completed'.
  const ipmSubtotal = round2(order.items
    .filter(isIpmLine)
    .reduce((s, it) => s + it.priceSgd * it.quantity, 0));
  const taxableSubtotal = round2(order.items
    .filter((it) => !isIpmLine(it))
    .reduce((s, it) => s + it.priceSgd * it.quantity, 0));
 
  const pOrder: Record<string, unknown> = {
    // Amounts. `total_sgd` is the MERCHANDISE subtotal; `grand_total_sgd` is what
    // the buyer actually paid after the seller voucher. These were previously the
    // same number, so a receipt printed the pre-voucher figure and overstated the
    // customer's payment.
    total_sgd:        totals.merchandiseSgd,
    total_grams:      totalGrams,
    grand_total_sgd:  totals.buyerPaidSgd,
    payment_amount:   totals.buyerPaidSgd,
    payment_currency: "SGD",
    fx_rate:          1.0,
    processing_fee_sgd: 0,
    // The buyer is charged nothing for delivery. What the marketplace bills US for
    // shipping goes to shipper_actual_fee_sgd below — same meaning it carries for
    // our own couriers: what the carrier actually charged us.
    shipping_fee_sgd:   0,
    shipper_actual_fee_sgd: totals.absorbedShippingSgd,
 
    // Seller voucher — the one absorbed cost the buyer SEES, so it renders as a
    // discount line on the receipt. promo_code stays null (there is no code of
    // ours to redeem); promo_type labels the source so reporting can tell a
    // marketplace voucher apart from one of our campaigns.
    ...(totals.voucherSgd > 0 && {
      promo_code:         null,
      promo_type:         "external_voucher",
      promo_discount_sgd: totals.voucherSgd,
    }),
 
    // IRAS IPM/taxable split — also what the P&L recognises revenue from.
    subtotal_ipm_sgd:                 ipmSubtotal,
    subtotal_taxable_sgd:             taxableSubtotal,
    discount_ipm_apportioned_sgd:     totals.voucherSgd,
    discount_taxable_apportioned_sgd: 0,
    gst_amount_sgd:                   0,
 
    // Books immediately (status='paid', like COD), but the marketplace PAYOUT to us
    // is not received yet → payment_status='unpaid'. The Payments tab "External
    // Payouts" section flips it to paid when the payout lands (mirrors COD).
    status:         "paid",
    payment_status: "unpaid",
    payment_method: "external",
    payment_gateway: order.channel,
 
    // An external order is a NORMAL order that happens to originate on a
    // marketplace. The CHANNEL lives in source_channel (below) — Dispatch buckets
    // on that — so shipper/shipper_service are left free for the PHYSICAL carrier
    // + tracking, filled in when the operator adds tracking (exactly like our own
    // courier orders: shipper=carrier, shipper_order_id=tracking#). No channel is
    // written into shipper anymore, so external and web/iOS orders are uniform.
    shipper_status:  "pending",
    fulfillment_type: "delivery",
    has_presale_items: hasPresale,
 
    delivery_address: {
      full_name:   order.shipTo.full_name ?? null,
      phone:       order.shipTo.phone ?? null,
      line1:       order.shipTo.line1 ?? null,
      line2:       order.shipTo.line2 ?? null,
      postal_code: order.shipTo.postal_code ?? null,
    },
 
    // Channel attribution (20260705 columns).
    source_channel:           order.channel,
    external_order_id:        order.externalOrderId,
    external_channel_fee_sgd: totals.channelFeeSgd,
    external_buyer_label:     order.buyerLabel,
 
    // The marketplace's own T&Cs cover refunds; consent is satisfied for the RPC gate.
    refund_policy_accepted: true,
  };
 
  // Market spot on the sale date. Key omitted entirely when unknown, so the RPC
  // stores NULL rather than a fabricated figure.
  if (order.spotSgd != null && Number.isFinite(order.spotSgd)) {
    pOrder.spot_sgd = order.spotSgd;
  }
 
  // Business date. Set ONLY when the operator backdated the sale — the key is
  // omitted otherwise so the RPC falls through to NOW(), exactly as before.
  const createdAt = resolveOrderCreatedAt(order.orderDate);
  if (createdAt) pOrder.created_at = createdAt;
 
  const pItems = order.items.map((it) => ({
    product_id:              it.productId,
    quantity:                it.quantity,
    price_sgd:               round2(it.priceSgd),
    grams_each:              it.gramsEach,
    is_presale:              it.isPresale ?? false,
    needs_stock_reservation: it.needsStockReservation ?? false,
  }));
 
  return { pOrder, pItems };
}
