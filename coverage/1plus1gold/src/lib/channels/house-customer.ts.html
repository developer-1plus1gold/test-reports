/**
 * Synthetic per-channel "house" customer provisioning.
 *
 * External orders need an owner (orders.customer_id → auth.users FK), but a marketplace
 * buyer has no account (and may never make one). So each external channel gets ONE
 * synthetic house customer that owns all its UNCLAIMED orders. The real buyer's identity
 * lives on orders.external_buyer_label + delivery_address. When the buyer later claims the
 * purchase by serial (Phase 1b), their tree blooms under their OWN account via an
 * accumulation delta — the order stays house-owned so Shopee channel revenue is preserved.
 *
 * Mirrors the WhatsApp-OTP synthetic-email pattern: a real auth.users row with a synthetic
 * email → the on_auth_user_created trigger provisions customers + profiles automatically.
 * Idempotent + cached via channel_house_customers (20260705c).
 */
 
import { createClient, type SupabaseClient } from "@supabase/supabase-js";
import type { ExternalChannelName } from "./types";
import { getChannel } from "./index";
 
const HOUSE_EMAIL = (channel: string) => `channel_${channel}@house.1plus1gold.com`;
 
function admin(): SupabaseClient {
  return createClient(process.env.NEXT_PUBLIC_SUPABASE_URL!, process.env.SUPABASE_SERVICE_KEY!);
}
 
/**
 * Returns the house customer_id for a channel, provisioning it on first use.
 * @throws if the channel is unknown or the auth user can't be created/found.
 */
export async function getOrCreateChannelHouseCustomer(
  channel: ExternalChannelName,
  client: SupabaseClient = admin(),
): Promise<string> {
  const ch = getChannel(channel);
  if (!ch) throw new Error(`unknown sales channel: ${channel}`);
 
  // 1. Cached mapping?
  const { data: existing } = await client
    .from("channel_house_customers")
    .select("customer_id")
    .eq("channel", channel)
    .maybeSingle();
  if (existing?.customer_id) return existing.customer_id as string;
 
  // 2. Provision the synthetic auth user (trigger creates customers + profiles).
  const email = HOUSE_EMAIL(channel);
  const label = `${ch.label} Marketplace`;
  let customerId: string | null = null;
 
  const created = await client.auth.admin.createUser({
    email,
    email_confirm: true,
    user_metadata: { house_channel: channel, display_name: label },
  });
  if (created.data?.user?.id) {
    customerId = created.data.user.id;
  } else {
    // Already exists (race / prior partial run) → find it by paging auth users.
    // Rare path; the mapping row below makes it a one-time cost per channel, and
    // 20260722_in_person_house_customer_mapping pre-seeds the row so it should
    // never fire in practice.
    //
    // This runs on the service-role client and nothing but the matched id ever
    // leaves this function, so no user data crosses a trust boundary. The defect
    // a single `perPage: 1000` call DID have was correctness, not disclosure:
    // past 1000 auth users the house account silently falls off the end of page
    // one, `.find()` returns undefined, and we throw below — breaking external
    // order creation for a customer who has already paid. Page until found.
    const PER_PAGE = 200;
    const MAX_PAGES = 50; // 10k users — a bounded loop, never an unbounded scan
    for (let page = 1; page <= MAX_PAGES && !customerId; page++) {
      const { data: list, error } = await client.auth.admin.listUsers({ page, perPage: PER_PAGE });
      if (error) break;
      const users = list?.users ?? [];
      customerId = users.find((u) => u.email === email)?.id ?? null;
      if (users.length < PER_PAGE) break; // last page reached
    }
  }
  if (!customerId) {
    throw new Error(`could not provision house customer for channel ${channel}`);
  }
 
  // 3. Give the profile a friendly name so admin/customer screens read "Shopee Marketplace".
  await client.from("profiles").update({ full_name: label }).eq("id", customerId);
 
  // 4. Cache the mapping (idempotent — a concurrent creator wins the PK, we re-read).
  const { error: mapErr } = await client
    .from("channel_house_customers")
    .insert({ channel, customer_id: customerId });
  if (mapErr) {
    const { data: reread } = await client
      .from("channel_house_customers")
      .select("customer_id")
      .eq("channel", channel)
      .maybeSingle();
    if (reread?.customer_id) return reread.customer_id as string;
    throw mapErr;
  }
 
  return customerId;
}
