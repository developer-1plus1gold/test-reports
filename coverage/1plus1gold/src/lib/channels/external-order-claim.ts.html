// ─────────────────────────────────────────────────────────────────────────
// external-order-claim — shared matching for the customer-facing "claim my
// marketplace order" funnel (Phase 1c). A buyer proves a Shopee/Carousell/
// Lazada purchase by matching its marketplace order number + delivery postal
// code + buyer id against our recorded external sales; the match is
// admin-approved before the order is re-homed (see 20260807_external_order_claims.sql).
//
// The order number is the strong anchor (unique per channel), the delivery
// postal code is the reproducible secondary check (amount was dropped — Shopee
// vouchers/coins make the paid figure unreliable), buyer id is a soft signal
// shown to the reviewer. The real gate is operator approval — a marketplace
// order number is not a secret, so a match alone must never move a gold asset.
//
// SECURITY: the customer-facing preview deliberately carries NO buyer-label
// match verdict (that would be a soft oracle on the stored username). The
// admin reviewer route recomputes the buyer-label comparison independently.
//
// Pure helpers here are unit-tested without a DB; findClaimableExternalOrder
// takes a service-role client and is shared by the lookup + submit routes so
// they can never diverge.
// ─────────────────────────────────────────────────────────────────────────
 
import type { SupabaseClient } from "@supabase/supabase-js";
import { isExternalChannel } from "./index";
 
export function normalizeOrderNumber(s: string): string {
  return s.trim();
}
 
/** Canonical form for a delivery postal code — case-fold + strip all whitespace,
 *  so "S 088 539" and "088539" compare equal. */
export function normalizePostal(s: string | null | undefined): string {
  return (s ?? "").replace(/\s+/g, "").toUpperCase();
}
 
/** Delivery postal code matches, exactly, after normalisation. Used as the
 *  reproducible secondary check in place of the buyer-paid amount, which Shopee
 *  vouchers / coins make unreliable. Fails closed when the order carries no stored
 *  postal (nothing to verify against → not self-service claimable). */
export function postalMatches(
  storedPostal: string | null | undefined,
  submitted: string,
): boolean {
  const a = normalizePostal(storedPostal);
  const b = normalizePostal(submitted);
  return a.length > 0 && a === b;
}
 
/** A generic/default buyer label ("Shopee buyer") or an empty one carries no
 *  identity signal — we never captured the real username, so we can't hard-match
 *  on it. Returns true for those so the reviewer decides instead. */
export function isGenericBuyerLabel(stored: string | null | undefined): boolean {
  const s = (stored ?? "").trim().toLowerCase();
  return s === "" || /\bbuyer$/.test(s);
}
 
/** Soft buyer-label comparison, surfaced to the admin reviewer (never a hard gate
 *  on its own). Distinguishes an exact hit from a mere substring overlap so a
 *  single-character/suffix guess can't read as a full "match":
 *   "exact"    — case-insensitive equality
 *   "partial"  — one string contains the other (weaker; reviewer judges)
 *   "mismatch" — a real stored label that neither equals nor overlaps
 *   null       — no usable stored label (generic/empty) */
export type BuyerLabelMatch = "exact" | "partial" | "mismatch" | null;
export function buyerLabelMatches(
  stored: string | null | undefined,
  submitted: string,
): BuyerLabelMatch {
  if (isGenericBuyerLabel(stored)) return null;
  const a = (stored ?? "").trim().toLowerCase();
  const b = submitted.trim().toLowerCase();
  if (!b) return "mismatch";
  if (a === b) return "exact";
  return a.includes(b) || b.includes(a) ? "partial" : "mismatch";
}
 
export interface ClaimOrderItemPreview {
  label: string;
  quantity: number;
  grams: number;
}
 
export interface ClaimablePreview {
  orderId: string;
  channel: string;
  createdAt: string;
  // Only the fields the confirm UI actually renders. Deliberately NO money
  // figure: the paid amount was removed from the confirmation screen (Mukesh),
  // and it is a private figure — so it must not ship in the response either
  // (a raw-JSON reader would otherwise still get it). `items` (product + grams)
  // is the minimum a buyer needs to recognise their own order.
  items: ClaimOrderItemPreview[];
  // NOTE: no buyer-label match verdict here on purpose — exposing exact/partial
  // to the client would be a soft oracle on the stored username. The mismatch
  // case is already folded into `no_match` above; the admin route recomputes the
  // comparison for the reviewer.
}
 
export type ClaimLookupResult =
  | { status: "ok"; preview: ClaimablePreview }
  | { status: "no_match" }
  | { status: "already_owned" }         // an order matched but a real customer already owns it
  | { status: "already_claimed" }       // an active order-claim / serial-claim by someone else
  | { status: "already_claimed_by_you" }; // the caller already has an active claim on this order
 
interface FindParams {
  channel: string;
  orderNumber: string;
  postalCode: string;
  buyerLabel: string;
  /** The signed-in caller, if known. Lets the caller be told "you already claimed
   *  this" (not an oracle — they know their own claims) while a claim by someone
   *  else stays the generic already_claimed. */
  callerCustomerId?: string;
}
 
/**
 * Resolve a claimable external order from customer-supplied details. Requires a
 * service-role client (it reads house-owned orders the caller does not own yet).
 * Returns a discriminated result the routes map to HTTP + messaging.
 */
export async function findClaimableExternalOrder(
  db: SupabaseClient,
  { channel, orderNumber, postalCode, buyerLabel, callerCustomerId }: FindParams,
): Promise<ClaimLookupResult> {
  if (!isExternalChannel(channel)) return { status: "no_match" };
 
  // The house customer that owns unclaimed orders for this channel. No mapping →
  // nothing has ever been sold on this channel → no match.
  const { data: house } = await db
    .from("channel_house_customers")
    .select("customer_id")
    .eq("channel", channel)
    .maybeSingle();
  const houseId = (house as { customer_id?: string } | null)?.customer_id ?? null;
 
  // Unique (source_channel, external_order_id) index → at most one row.
  const { data: order } = await db
    .from("orders")
    .select(
      "id, customer_id, source_channel, external_order_id, external_buyer_label, created_at, delivery_address, order_items(quantity, grams_each, products(label, grams))",
    )
    .eq("source_channel", channel)
    .eq("external_order_id", normalizeOrderNumber(orderNumber))
    .maybeSingle();
 
  if (!order) return { status: "no_match" };
  const o = order as Record<string, unknown>;
 
  // Secondary check: delivery postal code (exact) — the reproducible signal in
  // place of the voucher/coins-distorted amount. Fail closed (generic "no match",
  // no field-level oracle); an order with no stored postal is not claimable here.
  const storedPostal = (o.delivery_address as { postal_code?: string } | null)?.postal_code ?? null;
  if (!postalMatches(storedPostal, postalCode)) return { status: "no_match" };
 
  // Soft buyer-label check: only a real, non-overlapping stored label rejects
  // (exact / partial / no-stored-label all pass to the admin reviewer).
  const labelMatch = buyerLabelMatches(o.external_buyer_label as string | null, buyerLabel);
  if (labelMatch === "mismatch") return { status: "no_match" };
 
  // Ownership: only a still-house-owned order is claimable.
  if (!houseId || o.customer_id !== houseId) return { status: "already_owned" };
 
  // A card from this order already serial-claimed, or an active order-claim
  // exists → not claimable through this funnel.
  const [{ data: serialClaim }, { data: orderClaim }] = await Promise.all([
    db.from("external_claims").select("serial").eq("order_id", o.id as string).limit(1).maybeSingle(),
    db
      .from("external_order_claims")
      .select("id, claimant_customer_id")
      .eq("order_id", o.id as string)
      .in("status", ["pending", "approved"])
      .limit(1)
      .maybeSingle(),
  ]);
  if (serialClaim) return { status: "already_claimed" };
  if (orderClaim) {
    const byCaller = callerCustomerId && (orderClaim as { claimant_customer_id?: string }).claimant_customer_id === callerCustomerId;
    return { status: byCaller ? "already_claimed_by_you" : "already_claimed" };
  }
 
  const items = (o.order_items as Array<Record<string, unknown>> | null) ?? [];
  return {
    status: "ok",
    preview: {
      orderId: o.id as string,
      channel,
      createdAt: o.created_at as string,
      items: items.map((it) => {
        const p = it.products as { label?: string; grams?: number } | null;
        return {
          label: p?.label ?? "Product",
          quantity: Number(it.quantity ?? 0),
          grams: Number(p?.grams ?? it.grams_each ?? 0),
        };
      }),
    },
  };
}
