/**
 * Marketplace price + stock push — the ONE engine for every marketplace (Lazada
 * 2026-11-16, TikTok Shop 2026-11-17: "exactly the way we did it for Lazada").
 *
 * The guards live here so no channel can skip them:
 *   - the automatic switch gates scheduled and stock-only runs;
 *   - a stale gold price (> 15 min, or flagged stale) sends no prices;
 *   - a scheduled run pushes prices only when due (30 min, or gold moved ≥ 0.5 %);
 *   - a move > 5 % waits for the admin's Approve; below restock cost after fees is never sent;
 *   - only exact-SKU / confirmed mappings are touched; hidden products never (the export
 *     list is shop_visible only).
 * A channel supplies only the TRANSPORT (`PushSender`): how its API takes the updates and
 * which SKUs it refused. Cards for Control Tower are per channel.
 */

/** Whether a scheduled run should push now: 30 min since the last push, or spot moved ≥ 0.5 %. */
export function pushIsDue(last: { at: Date | null; spot: number | null }, spotNow: number, now: Date): { due: boolean; reason: string } {
  if (!last.at || last.spot == null || !(last.spot > 0)) return { due: true, reason: "first push" };
  if (now.getTime() - last.at.getTime() >= PUSH_INTERVAL_MS) return { due: true, reason: "30 minutes since the last push" };
  const move = Math.abs(spotNow - last.spot) / last.spot;
  if (move >= SPOT_MOVE_TRIGGER) return { due: true, reason: `gold moved ${(move * 100).toFixed(2)}%` };
  return { due: false, reason: `last push ${Math.round((now.getTime() - last.at.getTime()) / 60_000)} min ago, gold moved ${(move * 100).toFixed(2)}%` };
}

function cardId(channel: string, key: string): string {
  // Lazada's ids were hashed from "lazada_price_push:<key>" before this was shared — kept, so open cards still match.
  const h = createHash("sha256").update(`${channel}_price_push:${key}`).digest("hex");
  return `${h.slice(0, 8)}-${h.slice(8, 12)}-${h.slice(12, 16)}-${h.slice(16, 20)}-${h.slice(20, 32)}`;
}
 
/** The Control Tower cards a full run leaves open (waiting for approval / below cost; refused by the marketplace). */
export function pushCards(r: PushResult, ch: Pick<PushChannel, "channel" | "label" | "page">): MonitorMatch[] {
  const cards: MonitorMatch[] = [];
  const waiting = [...r.held, ...r.belowCost];
  if (waiting.length) {
    cards.push({
      id: cardId(ch.channel, "waiting"),
      summary: `${ch.label}: ${r.held.length} price(s) waiting for your approval, ${r.belowCost.length} below cost not sent — ${waiting.slice(0, 5).map((w) => w.name).join(", ")}${waiting.length > 5 ? "…" : ""}`,
      payload: { page: ch.page, held: r.held.map((h) => ({ sku: h.sellerSku, price: h.plan.price })), belowCost: r.belowCost.map((h) => h.sellerSku) },
    });
  }
  if (r.failed.length) {
    cards.push({
      id: cardId(ch.channel, "failed"),
      summary: `${ch.label} refused ${r.failed.length} update(s) on the last push — ${r.failed.slice(0, 3).map((f) => `${f.name}: ${f.error}`).join("; ")}`,
      payload: { page: ch.page, failed: r.failed.map((f) => ({ sku: f.sellerSku, error: f.error })) },
    });
  }
  return cards;
}
 
/**
 * Keep the two cards in step with a run. Only a FULL run (scheduled or Push now) knows the
 * current state: a skipped tick or an Approve of a few SKUs leaves the cards as they are.
 */
export async function syncMarketplacePushCards(db: SupabaseClient, r: PushResult, ch: PushChannel, mode: PushMode["kind"] = "scheduled") {
  if (!r.ran || mode === "approve" || mode === "stock") return null;
  return runMonitor({ sb: db, agentName: ch.agentName, severity: "approve", entityType: `${ch.channel}_price_push`, matches: pushCards(r, ch) });
}
