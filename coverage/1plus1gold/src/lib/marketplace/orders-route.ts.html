/**
 * Admin order-import handlers for a marketplace — ONE implementation for Lazada and TikTok
 * Shop (2026-11-17), so the switch, the cut-over and the dismiss rule cannot drift.
 *
 * GET  → { enabled, importFrom, lastSync, lastFinanceSync, imports[] (latest 30) }
 * POST { action: "import", enabled }          → switch the import on/off. The FIRST switch-on
 *        sets order_import_from = now: only orders created from then on are imported, and a
 *        later off/on keeps that cut-over, so nothing in between is skipped or doubled.
 * POST { action: "sync" }                     → Import now (same rules as the cron)
 * POST { action: "dismiss", externalOrderId } → close an open problem by hand
 * POST { action: "fees" }                     → read the actual fees now (channels that have it)
 * POST { action: "refill" }                   → re-read every order since the cut-over and fill
 *        the buyer views the regular run can no longer see (creates nothing new twice)
 * POST { action: "auto_dispatch", mode }       → off | drop_off | pickup (lib/marketplace/auto-dispatch)
 * GET also returns `autoDispatch` (the mode) and `priceGap` — the buyer's price vs our site price (buyer-view.ts).
 */

/** Last 30 days and everything since the cut-over. Pure. */
export function priceGapOf(rows: Array<{ first_seen_at: string; buyer_view: BuyerView }>, since30: string) {
  return {
    last30: summarisePriceGap(rows.filter((r) => new Date(r.first_seen_at).getTime() >= new Date(since30).getTime()).map((r) => r.buyer_view)),
    all: summarisePriceGap(rows.map((r) => r.buyer_view)),
  };
}
