/**
 * Marketplace auto-dispatch (Mukesh 2026-11-19): as soon as a paid order is imported, arrange
 * its shipment on the marketplace — the buyer then meets an arranged shipment rather than an
 * open order, and the drop-off deadline starts at once. Physical packing comes after: the
 * operator packs the bars, prints the label from the Dispatch tile and drops the parcel off.
 *
 *   TikTok Shop: Ship Package. TikTok picks the carrier AND the handover itself — for SG
 *     local sellers the collection method is the shop's Seller Center setting (ours: pickup),
 *     and `handover_method` is ignored (lib/tiktok-shop/fulfil). So TikTok's only mode is "on",
 *     stored as `pickup`. Only once TikTok has released the order (AWAITING_SHIPMENT) —
 *     during ON_HOLD the buyer may still cancel freely and TikTok will not ship it.
 *   Lazada: Pack, then Ready to ship (Mukesh: "consider it as packed — that is just a Lazada
 *     step"). Drop-off is how the store is set up on Lazada (delivery_type "dropship").
 *
 * The SAME fulfil functions as the tile's buttons do the work (lib/lazada/fulfil,
 * lib/tiktok-shop/fulfil), so a manual and an automatic arrangement are one code path.
 *
 * Grace (Mukesh 2026-11-20): an order is arranged only once it has been ours for
 * GRACE_MINUTES. Lazada 176203723521250 was packed + ready-to-ship 9 s after payment and the
 * buyer asked to cancel 22 s later; a short wait lets most change-of-mind cancels land before
 * a parcel is arranged. Each run checks again, so it is arranged on the first run after that.
 *
 * Safety: per channel and OFF by default (marketplace_connections.auto_dispatch). Each order
 * gets at most MAX_ATTEMPTS tries, RETRY_MINUTES apart; the attempt is counted BEFORE the
 * call, so a crash can never loop. After the last failed try: one Control Tower card + one
 * WhatsApp, and the order waits for a person (the tile's buttons still work).
 */

export type AutoDispatchMode = "off" | "drop_off" | "pickup";
export const AUTO_DISPATCH_MODES: readonly AutoDispatchMode[] = ["off", "drop_off", "pickup"];
export const MAX_ATTEMPTS = 3;
export const RETRY_MINUTES = 20;
/** Minutes after we recorded the paid order before it is arranged automatically. */
export const GRACE_MINUTES = 15;
 
/** Pure: has the order been ours long enough to arrange automatically? */
export function pastDispatchGrace(orderCreatedAt: string | null | undefined, now: Date): boolean {
  const t = orderCreatedAt ? Date.parse(orderCreatedAt) : NaN;
  if (!Number.isFinite(t)) return false; // unknown age → wait (fails toward not arranging)
  return now.getTime() - t >= GRACE_MINUTES * 60_000;
}

export function lazadaEligible(status: string | null): boolean {

export interface AutoDispatchResult {
  mode: AutoDispatchMode;
  arranged: Array<{ externalOrderId: string; how: string }>;
  failed: Array<{ externalOrderId: string; error: string; final: boolean }>;
}
 
/** Pure: one line for the cron log ("off", or "drop_off: arranged 1, failed 1 (reason)"). */
export function dispatchSummary(r: { mode: string; arranged: unknown[]; failed: AutoDispatchResult["failed"] } | undefined): string {
  if (!r) return "not run";
  if (r.mode === "off") return "off";
  const why = r.failed.map((f) => `${f.externalOrderId}: ${f.error}`).join("; ").slice(0, 300);
  return `${r.mode}: arranged ${r.arranged.length}, failed ${r.failed.length}${why ? ` (${why})` : ""}`;
}
 
/** Pure: may this ledger row be tried now? */
export function dueForAttempt(row: Pick<LedgerRow, "auto_dispatch_attempts" | "auto_dispatch_at">, now: Date): boolean {
  if (row.auto_dispatch_attempts >= MAX_ATTEMPTS) return false;
  if (!row.auto_dispatch_at) return true;
  return now.getTime() - new Date(row.auto_dispatch_at).getTime() >= RETRY_MINUTES * 60_000;
}
