/**
 * The BUYER'S view of a marketplace order (Mukesh 2026-11-19) — kept beside, never instead
 * of, our books.
 *
 *   Our view (orders.*): list price − OUR voucher − the shipping we absorb − fees = our net.
 *     Platform-funded vouchers never touch it: the marketplace pays us that money.
 *   Buyer's view (marketplace_order_imports.buyer_view): what the buyer actually paid, and
 *     who paid for each discount — ours, or the marketplace's.
 *   Site comparison: the same products at the same quantities priced through OUR checkout
 *     engine (priceOrder → deriveCart) at the order's spot, as a NEW customer (curve
 *     position 0, no wholesale). That is what a direct buyer would have paid us that day —
 *     the price gap, not proof the buyer would have come direct.
 *
 * "Buyer paid" is the marketplace's OWN figure where it reports one. We also add it up from
 * the parts (list − our voucher − platform voucher + shipping + other buyer charges); when
 * the two disagree the view says so (`check: "mismatch"`), so a misread field can never
 * quietly print a wrong number on a receipt.
 */

export interface PriceGapSummary {
  orders: number;
  /** Orders with a site quote (the rest are counted but not in the averages). */
  compared: number;
  siteSgd: number;
  buyerPaidSgd: number;
  gapSgd: number;
  gapPct: number | null;
  sellerVoucherSgd: number;
  platformVoucherSgd: number;
  mismatches: number;
}
 
/** Pure. Sums over the views given (the caller picks the window). */
export function summarisePriceGap(views: BuyerView[]): PriceGapSummary {
  const compared = views.filter((v) => v.site && v.gapSgd != null && v.check !== "mismatch");
  const site = r2(compared.reduce((s, v) => s + v.site!.totalSgd, 0));
  const paid = r2(compared.reduce((s, v) => s + (v.buyerPaidSgd - v.buyerShippingSgd - v.otherBuyerChargesSgd), 0));
  const gap = r2(site - paid);
  return {
    orders: views.length,
    compared: compared.length,
    siteSgd: site,
    buyerPaidSgd: paid,
    gapSgd: gap,
    gapPct: site > 0 ? r2((gap / site) * 100) : null,
    sellerVoucherSgd: r2(views.reduce((s, v) => s + v.sellerVoucherSgd, 0)),
    platformVoucherSgd: r2(views.reduce((s, v) => s + v.platformVoucherSgd, 0)),
    mismatches: views.filter((v) => v.check === "mismatch").length,
  };
}
