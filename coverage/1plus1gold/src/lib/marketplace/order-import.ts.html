/**
 * Marketplace order import — ONE engine for every marketplace (Lazada 2026-11-16, moved here
 * unchanged on 2026-11-17 when TikTok Shop joined, "exactly the way we did it for Lazada").
 *
 * The poll is the convergence spine (a 10-minute cron per channel, only while that channel's
 * import switch is ON; a channel's webhook only triggers the same run sooner). For each
 * order created after `order_import_from` (the cut-over — nothing older is ever touched):
 *   • new + paid  → the channel maps its SKUs to our products via the CONFIRMED listing
 *                   mapping, then the order is created through createExternalOrder — the SAME
 *                   path as the manual External sales form (stock taken, Dispatch tile) —
 *                   and the admins get the same "New order" WhatsApp as a checkout;
 *   • unmapped SKU → NOT created; issue `unknown_sku`, retried every run until mapped;
 *   • already recorded and the marketplace now says cancelled / returned → an issue. Never
 *     automatic: a cancel or a return needs a human.
 *   • already recorded and the marketplace says shipped / delivered → our order moves forward
 *     only (picked up → delivered + completed + bars sold), never back.
 * Every order is recorded in `marketplace_order_imports`; open issues drive the Control Tower
 * cards, so a card persists across runs that no longer see the order and closes once the
 * issue is resolved (our order cancelled, the SKU mapped, or dismissed in Admin).
 *
 * A channel supplies only the READER (`OrderChannel.connect` → `list(since)`), which returns
 * each order already classified and with a `map()` that builds the createExternalOrder input.
 */

export async function rememberBuyerId(db: SupabaseClient, orderId: string, buyerId: string | null | undefined): Promise<boolean> {
  const id = (buyerId ?? "").trim();

function cardId(channel: string, key: string): string {
  // Lazada's ids were hashed from "lazada_order_import:<key>" before this was shared — kept.
  const h = createHash("sha256").update(`${channel}_order_import:${key}`).digest("hex");
  return `${h.slice(0, 8)}-${h.slice(8, 12)}-${h.slice(12, 16)}-${h.slice(16, 20)}-${h.slice(20, 32)}`;
}
