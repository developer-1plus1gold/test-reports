 *
 * Every bar we sell carries OUR NFC serial (e.g. 100G-4FWJ-7XGM). Most also carry
 * the refinery's own stamped serial (`nfc_serials.manufacturer_serial`, e.g.
 * B240888) — the number a customer can read off the metal. Wherever we name a bar
 * to a customer we now show both, in ONE format, from this ONE module:
 *
 *     Bar no. B240888 · Serial 100G-4FWJ-7XGM
 *
 * Falls back to "Serial <nfc>" when there is no manufacturer serial. That is the
 * Valcambi 1g Premium Curve card case by DATA, not by special-casing: cards cut
 * from a CombiBar carry no stamped serial of their own (verified on prod
 * 2026-10-25 — every "1 gram Split-Valcambi" row has manufacturer_serial NULL),
 * so they stay NFC-only exactly as Mukesh asked. A manufacturer_serial equal to
 * the NFC serial is the old blank-fallback self-copy (2026-09-14) and is treated
 * as absent, never printed as a fake bar number.
 */

/** A real refinery serial, or null (blank, or a self-copy of the NFC serial). */
export function realManufacturer(nfc: string, manufacturer: string | null | undefined): string | null {
  const m = (manufacturer ?? "").trim();
  if (!m || m.toUpperCase() === nfc.trim().toUpperCase()) return null;
  return m;
}
 
/** "Bar no. B240888 · Serial 100G-4FWJ-7XGM", or "Serial 100G-4FWJ-7XGM". */
export function formatBarSerial(nfc: string, manufacturer?: string | null): string {
  const m = realManufacturer(nfc, manufacturer);
  return m ? `Bar no. ${m} · Serial ${nfc}` : `Serial ${nfc}`;
}
 
/**
 * Prose form for a sentence that already names the NFC serial
 * ("your gift card 1G-ABCD-1234 (bar no. B240888)"). Emails AND the WhatsApp
 * templates use it: Meta approves a template's TEXT, not its variable values, so
 * carrying the bar number inside the existing {{serial}} var needs no re-approval
 * (Mukesh 2026-10-25, option 1).
 */
export function serialWithBarNo(nfc: string, manufacturer?: string | null): string {
  const m = realManufacturer(nfc, manufacturer);
  return m ? `${nfc} (bar no. ${m})` : nfc;
}
 
/**
 * Bracket-free form, for a template that already wraps the var in brackets
 * ("your vaulted bar {{1}} ({{2}}g)") — "100G-X / bar no. B240888".
 */
export function serialSlashBarNo(nfc: string, manufacturer?: string | null): string {
  const m = realManufacturer(nfc, manufacturer);
  return m ? `${nfc} / bar no. ${m}` : nfc;
}

/**
 * Several bars on one line (qty > 1). Bars with a refinery serial are named one
 * by one; NFC-only cards are collapsed into a single "Serial A, B, C" so a
 * ten-card Valcambi line does not repeat the word ten times.
 */
export function formatBarSerials(bars: { nfc: string; manufacturer?: string | null }[]): string {
  const seen = new Set<string>();
  const named: string[] = [];
  const bare: string[] = [];
  for (const b of bars) {
    const nfc = b.nfc?.trim();
    if (!nfc || seen.has(nfc)) continue;
    seen.add(nfc);
    if (realManufacturer(nfc, b.manufacturer)) named.push(formatBarSerial(nfc, b.manufacturer));
    else bare.push(nfc);
  }
  if (bare.length) named.push(`Serial ${bare.join(", ")}`);
  return named.join("; ");
}
