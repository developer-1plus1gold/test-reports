/**
 * Turning one resolved `ShowroomPiece` into geometry.
 *
 * Every piece is drawn at its TRUE millimetres (§3.5) — that is the whole
 * argument of the room, so nothing here normalises, fits-to-box or scales for
 * appearance. A 1 g bar is 8.9 mm wide and looks it.
 *
 * The four forms (§3.6):
 *   bar      extruded rounded rect, photo face, gold sides
 *   card     the same slab at card millimetres, matte sides
 *   standee  the face mounted upright on a small easel — the honest answer for
 *            a photo that is not a flat-on cut-out. It LOOKS like a picture
 *            because it is one, rather than pretending to be an object.
 *   box      procedural, no face at all
 */
 
import * as THREE from "three";
import type { ShowroomPiece } from "@/lib/showroom/pieces";
import { GOLD, canvas2d, canvasTexture, fitFaceUv, type SceneMaterials } from "@/lib/showroom/scene/materials";
 
/** Rounded-rectangle profile, centred on the origin, for extrusion. */
function roundedRect(w: number, h: number, r: number): THREE.Shape {
  const s = new THREE.Shape();
  const x = -w / 2;
  const y = -h / 2;
  s.moveTo(x + r, y);
  s.lineTo(x + w - r, y);
  s.quadraticCurveTo(x + w, y, x + w, y + r);
  s.lineTo(x + w, y + h - r);
  s.quadraticCurveTo(x + w, y + h, x + w - r, y + h);
  s.lineTo(x + r, y + h);
  s.quadraticCurveTo(x, y + h, x, y + h - r);
  s.lineTo(x, y + r);
  s.quadraticCurveTo(x, y, x + r, y);
  return s;
}
 
/**
 * A slab with a bevelled rim.
 *
 * The bevel is what makes a bar read as a minted object rather than a printed
 * rectangle — it is the edge that catches the light. It is also the reason a
 * bounding-box density calculation OVERSTATES a real bar, which is why
 * `densityCheck` accepts a small positive delta and not a negative one.
 */
function slab(w: number, h: number, t: number, r: number): THREE.ExtrudeGeometry {
  const bevelT = Math.min(t * 0.22, 0.35);
  const bevelS = Math.min(0.7, Math.min(w, h) * 0.05);
  const g = new THREE.ExtrudeGeometry(roundedRect(w, h, r), {
    depth: Math.max(t - 2 * bevelT, 0.01),
    bevelEnabled: true,
    bevelThickness: bevelT,
    bevelSize: bevelS,
    bevelOffset: -bevelS,
    bevelSegments: 3,
    curveSegments: 10,
  });
  g.translate(0, 0, bevelT);
  return g;
}
 
/** Shrink a font size until the text fits `maxW`, the way the prototype does. */
function fitText(x: CanvasRenderingContext2D, txt: string, maxW: number, size: number, weight: number, fam: string): void {
  x.font = `${weight} ${size}px ${fam}`;
  const m = x.measureText(txt).width;
  if (m > maxW) x.font = `${weight} ${(size * maxW) / m}px ${fam}`;
}
 
const SANS = "Montserrat, Arial, sans-serif";
 
/**
 * The engraved brass placard.
 *
 * Two lines only, and baked once (§5.1): the weight, and the first clause of
 * the description. The PRICE is deliberately not here — it floats above as
 * live DOM, because a baked price is a photograph of a price and re-baking
 * fifteen textures a minute is a visible hitch on the phone the quality bar
 * names.
 */
function placard(lines: [string, string]): THREE.Mesh {
  const W = 26;
  const H = 8;
  const u = 26; // px per mm — enough to stay legible on a tilted plate
  const [cv, x] = canvas2d(W * u, H * u);
 
  const g = x.createLinearGradient(0, 0, 0, H * u);
  g.addColorStop(0, "#9c7a3a");
  g.addColorStop(1, "#6a4f1f");
  x.fillStyle = g;
  x.fillRect(0, 0, W * u, H * u);
  x.strokeStyle = "rgba(40,28,10,.55)";
  x.lineWidth = 0.25 * u;
  x.strokeRect(0.6 * u, 0.6 * u, (W - 1.2) * u, (H - 1.2) * u);
 
  x.fillStyle = "#140d05";
  x.textAlign = "center";
  x.textBaseline = "middle";
  fitText(x, lines[0], (W - 3) * u, 3.1 * u, 600, SANS);
  x.fillText(lines[0], (W / 2) * u, 3.2 * u);
  if (lines[1]) {
    fitText(x, lines[1], (W - 3) * u, 1.7 * u, 500, SANS);
    x.fillText(lines[1], (W / 2) * u, 6.0 * u);
  }
 
  const face = new THREE.MeshStandardMaterial({ map: canvasTexture(cv, true), metalness: 0.35, roughness: 0.5 });
  const edge = new THREE.MeshStandardMaterial({ color: 0x7d5f27, metalness: 1, roughness: 0.4 });
  const m = new THREE.Mesh(new THREE.BoxGeometry(W, H, 0.8), [edge, edge, edge, edge, face, edge]);
  // Tilted toward the viewer so it can be read from standing height.
  m.rotation.x = -1.0;
  m.castShadow = true;
  return m;
}
 
/**
 * A blank gold face, for a piece with no photograph.
 *
 * Deliberately NOT a stamped imitation of a real bar. The prototype draws
 * "FINE GOLD / 999.9 / 1P 004264" onto the metal, and §3.7 is explicit that
 * those serials are fabricated: a photoreal render carrying an invented serial
 * on a site whose whole proposition is verifiable provenance is the one place
 * the brandkit's conventions must not be carried across. A plain brushed face
 * says "we have no photograph of this yet", which is true.
 */
function blankFace(w: number, h: number, aniso: number): THREE.MeshStandardMaterial {
  const ppm = Math.max(8, 180 / Math.min(w, h));
  const [cv, x] = canvas2d(w * ppm, h * ppm);
  x.fillStyle = "rgb(40,40,40)";
  x.fillRect(0, 0, cv.width, cv.height);
  for (let i = 0; i < cv.height; i += 2) {
    x.fillStyle = `rgba(255,255,255,${Math.random() * 0.05})`;
    x.fillRect(0, i, cv.width, 1);
  }
  const rough = fitFaceUv(canvasTexture(cv, false, aniso), w, h);
  return new THREE.MeshStandardMaterial({
    color: GOLD,
    metalness: 1,
    roughness: 1,
    roughnessMap: rough,
    envMapIntensity: 1.25,
  });
}
 
/** The photographic face, mapped to the whole slab front. */
function photoFace(
  url: string,
  w: number,
  h: number,
  loader: THREE.TextureLoader,
  aniso: number,
  overrides?: Partial<THREE.MeshStandardMaterialParameters>,
): THREE.MeshStandardMaterial {
  const t = loader.load(url);
  t.colorSpace = THREE.SRGBColorSpace;
  t.anisotropy = aniso;
  fitFaceUv(t, w, h);
  return new THREE.MeshStandardMaterial({
    map: t,
    // A flat photograph under a warm spot goes muddy; a small emissive
    // contribution from the map itself keeps the artwork readable without
    // washing out the metal around it.
    emissive: 0xffffff,
    emissiveMap: t,
    emissiveIntensity: 0.12,
    metalness: 0.55,
    roughness: 0.36,
    envMapIntensity: 0.55,
    ...overrides,
  });
}
 
export interface BuiltPiece {
  group: THREE.Group;
  /** Every mesh that should answer a click, tagged with the piece id. */
  pickables: THREE.Mesh[];
}
 
/**
 * Build one piece, standing on the bench at local y = 0, its own centre at
 * local x = 0. The caller positions the group within the case.
 */
export function buildPieceMesh(
  piece: ShowroomPiece,
  mats: SceneMaterials,
  loader: THREE.TextureLoader,
  aniso: number,
): BuiltPiece {
  const group = new THREE.Group();
  const pickables: THREE.Mesh[] = [];
  const { widthMm: w, heightMm: h, thicknessMm: t } = piece;
 
  if (piece.form === "box") {
    // Accessories: a procedural presentation box with a gold band. No face —
    // a photograph of a box lying flat on velvet is wrong however well shot.
    const body = new THREE.Mesh(
      new THREE.BoxGeometry(w, t, h),
      new THREE.MeshStandardMaterial({ color: 0x241c12, roughness: 0.85, metalness: 0 }),
    );
    body.position.y = t / 2;
    body.castShadow = true;
    body.receiveShadow = true;
    group.add(body);
 
    const band = new THREE.Mesh(new THREE.BoxGeometry(w * 1.004, t * 0.1, h * 0.18), mats.goldSide);
    band.position.y = t / 2;
    group.add(band);
    pickables.push(body, band);
  } else if (piece.form === "standee") {
    // Upright on a small easel. The face is a picture and is shown as one.
    const faceMat = piece.faceUrl
      ? photoFace(piece.faceUrl, w, h, loader, aniso, { metalness: 0.1, roughness: 0.5, emissiveIntensity: 0.3 })
      : blankFace(w, h, aniso);
    const board = new THREE.Mesh(slab(w, h, Math.max(t, 1.2), Math.min(w, h) * 0.06), [
      faceMat,
      new THREE.MeshStandardMaterial({ color: 0x1c1b19, roughness: 0.6 }),
    ]);
    // Leaned back a little, standing on its bottom edge.
    board.position.y = h / 2;
    board.rotation.x = -0.18;
    board.castShadow = true;
    group.add(board);
 
    const foot = new THREE.Mesh(new THREE.BoxGeometry(w * 0.5, 1.6, h * 0.35), mats.bronze);
    foot.position.set(0, 0.8, h * 0.12);
    group.add(foot);
    pickables.push(board, foot);
  } else {
    // `bar` and `card` are the same construction at different millimetres and
    // with different side material — the metal of a bar, the matte edge of a
    // laminated card.
    const isCard = piece.form === "card";
    const faceMat = piece.faceUrl
      ? photoFace(
          piece.faceUrl,
          w,
          h,
          loader,
          aniso,
          isCard ? { metalness: 0.15, roughness: 0.42, envMapIntensity: 0.9, emissiveIntensity: 0.35 } : undefined,
        )
      : blankFace(w, h, aniso);
    const sideMat = isCard
      ? new THREE.MeshStandardMaterial({ color: 0x1c1b19, roughness: 0.6 })
      : mats.goldSide;
 
    const mesh = new THREE.Mesh(slab(w, h, t, Math.min(w, h) * (isCard ? 0.037 : 0.1)), [faceMat, sideMat]);
    // Extruded along +z, so lay it flat: face up, height running away from the
    // viewer. Same transform the prototype uses.
    mesh.rotation.x = -Math.PI / 2;
    mesh.castShadow = true;
    mesh.receiveShadow = isCard;
    group.add(mesh);
    pickables.push(mesh);
  }
 
  const plate = placard(piece.placard);
  // In FRONT of the piece (toward the viewer), clear of its footprint.
  plate.position.set(0, (8 / 2) * Math.cos(1.0) + 0.4, h / 2 + 14);
  group.add(plate);
  pickables.push(plate);
 
  for (const m of pickables) m.userData.pieceId = piece.id;
  return { group, pickables };
}
 
/**
 * Lay a case's pieces out along its bench.
 *
 * ▶ The gap is a CONSTRAINT, not styling (§4.1). The brandkit measured it: at
 * 15 bars a row *"fused into a single gold slab at 132px and only read as
 * fifteen objects at 88px."* So the gap is a floor, never squeezed to fit more
 * in — a case that outgrows its bench is a membership problem for §6.1 to
 * solve by reflowing or spilling, which is what `crowded` flags.
 */
export const MIN_PIECE_GAP_MM = 26;
 
export function layOutPieces(pieces: ShowroomPiece[], benchWidthMm: number): number[] {
  if (pieces.length === 0) return [];
  const widths = pieces.map((p) => p.widthMm);
  const total = widths.reduce((a, b) => a + b, 0);
  // Share out whatever room is left, but never below the floor.
  const slack = benchWidthMm - total;
  const gap = pieces.length > 1 ? Math.max(MIN_PIECE_GAP_MM, slack / (pieces.length + 1)) : 0;
  const span = total + gap * (pieces.length - 1);
 
  const xs: number[] = [];
  let x = -span / 2;
  for (const wMm of widths) {
    xs.push(x + wMm / 2);
    x += wMm + gap;
  }
  return xs;
}
