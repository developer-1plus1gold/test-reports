/**
 * Shared materials, textures and the reflection environment.
 *
 * One module so the gold in one case is the same gold as the gold in the next,
 * and so the PMREM environment — the expensive thing — is built once per
 * renderer rather than once per case.
 *
 * ▶ Gold is a METAL, so almost all of what you see is reflection. With no
 * environment map a `metalness: 1` material renders near-black under any
 * number of lights, which is the second way a ported scene comes out dark. The
 * generated environment below (a dark box with a few warm panels) is therefore
 * not decoration: it is where the metal's colour comes from.
 */
 
import * as THREE from "three";
 
/** Working-space gold. Constructed from floats, so it is already in the
 *  renderer's working colour space and must NOT be sRGB-converted. */
export const GOLD = new THREE.Color(1.0, 0.74, 0.34);
 
export interface SceneMaterials {
  environment: THREE.Texture;
  goldSide: THREE.MeshStandardMaterial;
  bronze: THREE.MeshStandardMaterial;
  glass: THREE.MeshStandardMaterial;
  velvet: THREE.MeshStandardMaterial;
  counterBody: THREE.MeshStandardMaterial;
  counterTop: THREE.MeshStandardMaterial;
  dispose(): void;
}
 
/** A small canvas, sized in device pixels, for procedural textures. */
export function canvas2d(w: number, h: number): [HTMLCanvasElement, CanvasRenderingContext2D] {
  const cv = document.createElement("canvas");
  cv.width = Math.max(2, Math.round(w));
  cv.height = Math.max(2, Math.round(h));
  const ctx = cv.getContext("2d");
  if (!ctx) throw new Error("2d context unavailable");
  return [cv, ctx];
}
 
/**
 * Wrap a canvas as a texture.
 *
 * `srgb` must be TRUE for anything that carries colour and FALSE for anything
 * that carries data (roughness, bump, alpha). Tagging a data map as sRGB
 * silently applies a transfer curve to numbers that are not colours — the
 * surface then looks subtly wrong in a way that is very hard to attribute.
 */
export function canvasTexture(cv: HTMLCanvasElement, srgb: boolean, aniso = 1): THREE.CanvasTexture {
  const t = new THREE.CanvasTexture(cv);
  if (srgb) t.colorSpace = THREE.SRGBColorSpace;
  t.anisotropy = aniso;
  return t;
}
 
/**
 * Map the WHOLE image onto an extruded face.
 *
 * ⚠️ This STRETCHES — it does not letterbox. `ExtrudeGeometry` generates face
 * UVs in world units rather than 0..1, so `repeat = 1/size` is what brings the
 * image back to exactly one tile across the face. The consequence is that a
 * face image whose aspect does not match the piece's millimetres is distorted,
 * silently and plausibly. That is precisely why the uploader validates aspect
 * to ±5% of the declared dimensions (§3.4) rather than trusting the file.
 */
export function fitFaceUv(t: THREE.Texture, widthMm: number, heightMm: number): THREE.Texture {
  t.repeat.set(1 / widthMm, 1 / heightMm);
  t.offset.set(0.5, 0.5);
  return t;
}
 
/** A vertical stripe used as a bump map for fluted joinery. */
function fluteCanvas(): HTMLCanvasElement {
  const [cv, x] = canvas2d(64, 4);
  const g = x.createLinearGradient(0, 0, 64, 0);
  g.addColorStop(0, "#222");
  g.addColorStop(0.5, "#ddd");
  g.addColorStop(1, "#222");
  x.fillStyle = g;
  x.fillRect(0, 0, 64, 4);
  return cv;
}
 
export function fluteTexture(repeat: number): THREE.CanvasTexture {
  const t = canvasTexture(fluteCanvas(), false);
  t.wrapS = t.wrapT = THREE.RepeatWrapping;
  t.repeat.set(repeat, 1);
  return t;
}
 
/**
 * The reflection environment: a dark box with warm softboxes, rendered once to
 * a PMREM cubemap. Ported from the prototype unchanged in geometry — only the
 * `lin()` wrapper is gone, per `units.ts` §2.
 */
function buildEnvironment(renderer: THREE.WebGLRenderer): THREE.Texture {
  const s = new THREE.Scene();
  s.add(
    new THREE.Mesh(
      new THREE.BoxGeometry(10, 10, 10),
      new THREE.MeshBasicMaterial({ color: new THREE.Color(0.018, 0.013, 0.008), side: THREE.BackSide }),
    ),
  );
 
  const panel = (w: number, h: number, pos: [number, number, number], rot: [number, number, number], k: number, col: [number, number, number]) => {
    const m = new THREE.Mesh(
      new THREE.PlaneGeometry(w, h),
      new THREE.MeshBasicMaterial({ color: new THREE.Color(col[0] * k, col[1] * k, col[2] * k), side: THREE.DoubleSide }),
    );
    m.position.set(...pos);
    m.rotation.set(...rot);
    s.add(m);
  };
 
  const warm: [number, number, number] = [1, 0.9, 0.76];
  const soft: [number, number, number] = [1, 0.95, 0.88];
  panel(5, 2.2, [0, 4.9, 0.4], [Math.PI / 2, 0, 0], 5.5, soft);
  panel(7, 2.4, [0, 2.6, -4.9], [0, 0, 0], 3.2, warm);
  panel(1.6, 4, [-4.9, 1.2, 0], [0, Math.PI / 2, 0], 2.2, warm);
  panel(1.6, 4, [4.9, 1.2, 0], [0, -Math.PI / 2, 0], 1.4, warm);
  panel(8, 0.4, [0, 0.6, 4.9], [0, Math.PI, 0], 1.2, warm);
 
  const pm = new THREE.PMREMGenerator(renderer);
  const tex = pm.fromScene(s, 0.03).texture;
  pm.dispose();
  s.traverse((o) => {
    if (o instanceof THREE.Mesh) {
      o.geometry.dispose();
      (o.material as THREE.Material).dispose();
    }
  });
  return tex;
}
 
export function createMaterials(renderer: THREE.WebGLRenderer): SceneMaterials {
  const environment = buildEnvironment(renderer);
 
  const goldSide = new THREE.MeshStandardMaterial({
    color: GOLD,
    metalness: 1,
    roughness: 0.2,
    envMapIntensity: 1.25,
  });
 
  const bronze = new THREE.MeshStandardMaterial({ color: 0x9a7636, metalness: 1, roughness: 0.32 });
 
  // `depthWrite: false` so the glass never occludes the gold behind it in the
  // depth buffer; `renderOrder` on the meshes keeps it drawn last.
  const glass = new THREE.MeshStandardMaterial({
    color: 0xffffff,
    metalness: 0,
    roughness: 0.04,
    transparent: true,
    opacity: 0.07,
    depthWrite: false,
    side: THREE.DoubleSide,
    envMapIntensity: 1.6,
  });
 
  const velvet = new THREE.MeshStandardMaterial({ color: 0x1e1812, roughness: 1, metalness: 0 });
 
  const counterBody = new THREE.MeshStandardMaterial({
    color: 0x2a2219,
    roughness: 0.7,
    bumpMap: fluteTexture(70),
    bumpScale: 1.5,
  });
 
  const counterTop = new THREE.MeshStandardMaterial({ color: 0x120e09, roughness: 0.22, metalness: 0.1 });
 
  return {
    environment,
    goldSide,
    bronze,
    glass,
    velvet,
    counterBody,
    counterTop,
    dispose() {
      environment.dispose();
      for (const m of [goldSide, bronze, glass, velvet, counterBody, counterTop]) {
        if (m.bumpMap) m.bumpMap.dispose();
        m.dispose();
      }
    },
  };
}
