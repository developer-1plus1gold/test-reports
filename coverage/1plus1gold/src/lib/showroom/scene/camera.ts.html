/**
 * The case-relative camera rig (§4.2).
 *
 * ▶ WHY NOT THE PROTOTYPE'S RIG. The prototype orbits the WORLD origin, which
 * is correct when there is exactly one counter sitting on it. With four cases
 * at different positions and rotations it breaks in a specific way: orbiting a
 * left-wall case in world space swings the camera through the wall, and the
 * azimuth bounds that keep you in front of one case are wrong for the next.
 *
 * So the state is expressed in the CASE's own frame — a target, a radius, and
 * two angles about that case — and only converted to world space at the last
 * moment. A case is then framed correctly wherever it stands, bounds are
 * written once, and "in front of the case" means the same thing for all four.
 *
 * Pure except for the Vector3 it writes into: no Three scene, no DOM, no
 * renderer. That is deliberate — framing arithmetic is easy to get subtly
 * wrong and hard to see wrong, so it is testable without a GPU.
 */
 
import * as THREE from "three";
 
export interface RigState {
  /** Orbit centre, in ROOM space. */
  target: THREE.Vector3;
  /** Distance from the target, millimetres. */
  radius: number;
  /** Azimuth RELATIVE TO THE CASE's facing. 0 looks straight at its front. */
  azimuth: number;
  /** Polar angle from +y. Smaller is higher above the bench. */
  polar: number;
  /** The case's own rotation about y, so azimuth can be resolved to world. */
  caseRotation: number;
}
 
/** How far the camera may swing either side of a case's front face. Beyond
 *  this you are looking along the glass rather than into it, and behind a case
 *  against a wall there is nothing to see. */
export const MAX_AZIMUTH = 0.95;
/** Polar bounds: never below the bench, never a plan view. */
export const MIN_POLAR = 0.35;
export const MAX_POLAR = 1.32;
 
export const clamp = (v: number, a: number, b: number) => Math.max(a, Math.min(b, v));
 
export function clampRig(s: RigState, minR: number, maxR: number): void {
  s.azimuth = clamp(s.azimuth, -MAX_AZIMUTH, MAX_AZIMUTH);
  s.polar = clamp(s.polar, MIN_POLAR, MAX_POLAR);
  s.radius = clamp(s.radius, minR, maxR);
}
 
/**
 * Resolve the rig to a world-space eye position.
 *
 * The case's rotation is ADDED to the azimuth, which is the whole trick: a
 * case turned 72° to face the room is orbited about its own front, so azimuth
 * 0 is its glass and not the room's +z.
 */
export function eyePosition(s: RigState, out = new THREE.Vector3()): THREE.Vector3 {
  const a = s.azimuth + s.caseRotation;
  const sinP = Math.sin(s.polar);
  return out.set(
    s.target.x + s.radius * sinP * Math.sin(a),
    s.target.y + s.radius * Math.cos(s.polar),
    s.target.z + s.radius * sinP * Math.cos(a),
  );
}
 
/**
 * The radius at which a span of `spanW` × `spanH` millimetres fills the usable
 * part of the viewport.
 *
 * `usableW/usableH` are the pixels actually free of the overlay panels, not
 * the canvas — framing to the full canvas puts the case behind the info panel
 * on desktop, which is the bug the prototype's view-offset trick exists to
 * avoid. Carried across unchanged because it still applies.
 */
export function fitRadius(
  spanWmm: number,
  spanHmm: number,
  fovDeg: number,
  viewportH: number,
  usableW: number,
  usableH: number,
): number {
  const tanV = Math.tan(THREE.MathUtils.degToRad(fovDeg) / 2);
  // Horizontal half-angle derived from the vertical fov and the USABLE width
  // over the FULL height — that ratio is what the projection actually uses
  // once the view offset is applied.
  const tanH = (tanV * Math.max(usableW, 1)) / Math.max(viewportH, 1);
  const tanU = (tanV * Math.max(usableH, 1)) / Math.max(viewportH, 1);
  return Math.max(spanWmm / 2 / Math.max(tanH, 1e-4), spanHmm / 2 / Math.max(tanU, 1e-4));
}
 
/** Cubic ease, matching the prototype's fly-between feel. */
export const ease = (t: number) => (t < 0.5 ? 4 * t * t * t : 1 - Math.pow(-2 * t + 2, 3) / 2);
 
export interface RigFlight {
  from: RigState;
  to: RigState;
  t0: number;
  durationMs: number;
}
 
/**
 * Interpolate a flight.
 *
 * ▶ The target and the case rotation are interpolated too, which is what makes
 * moving BETWEEN cases a single ~1 s arc rather than a mode (§4.1). Rotation
 * is interpolated on the SHORTEST path — lerping 170° to -170° the long way
 * round would spin the camera three-quarters of the way through the room.
 */
export function applyFlight(f: RigFlight, now: number, out: RigState): boolean {
  const k = Math.min(1, (now - f.t0) / Math.max(f.durationMs, 1));
  const q = ease(k);
  out.target.lerpVectors(f.from.target, f.to.target, q);
  out.radius = f.from.radius + (f.to.radius - f.from.radius) * q;
  out.azimuth = f.from.azimuth + (f.to.azimuth - f.from.azimuth) * q;
  out.polar = f.from.polar + (f.to.polar - f.from.polar) * q;
 
  let d = f.to.caseRotation - f.from.caseRotation;
  while (d > Math.PI) d -= Math.PI * 2;
  while (d < -Math.PI) d += Math.PI * 2;
  out.caseRotation = f.from.caseRotation + d * q;
 
  return k >= 1;
}
