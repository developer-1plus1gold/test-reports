/**
 * How a product presents in a case (docs/showroom-design.md §3.6).
 *
 *   card    flat slab at the card's true mm — the assay range, our signature card
 *   bar     extruded rounded rect at the bar's true mm, gold side material
 *   standee the face mounted upright on a small easel
 *   box     a procedural box with the gold logo — the two accessories
 *
 * ▶ `standee` is what keeps requirement R4 true. If the only face available is
 * a photo taken at an angle, mounting it upright is TRUTHFUL — it looks like a
 * picture because it is one — whereas laying it flat pretends it is an object
 * and fails. So a product can never be blocked from the room for want of a
 * face.
 */
 
export const SHOWROOM_FORMS = ["bar", "card", "standee", "box"] as const;
export type ShowroomForm = (typeof SHOWROOM_FORMS)[number];
 
export function isShowroomForm(v: unknown): v is ShowroomForm {
  return typeof v === "string" && (SHOWROOM_FORMS as readonly string[]).includes(v);
}
 
/**
 * ▶ THE ONE LINE THAT CARRIES AN OPEN OPERATOR DECISION.
 *
 * Today an in-assay product defaults to `card`, because that is what the
 * operator decided and it is what the customer actually receives.
 *
 * The brandkit argues the other way, and it argues with a measurement. Its
 * IRON RULE 6 — "PACKAGING SHRINKS THE GOLD TWICE OVER" — says to use the bare
 * bar whenever a product is scaled DOWN, and to measure gold-pixel share
 * rather than judge by eye. Measured at matched footprint:
 *
 *     bare PAMP bar    95.9%      assay card, navy   25.1%
 *     CombiBar, open   99.1%      our signature card  7.8%
 *
 * A piece on a shelf rendered on a phone IS the scaled-down case, and the
 * kit's own eight PAMP renders are bare bars.
 *
 * The recommendation in §3.6 is to split it the way price and placard copy are
 * already split — bare bar on the shelf where Rule 6 bites, assay card in the
 * focused panel where the product is being examined. That is a product
 * decision, not a refactor, so it is NOT applied here. Flipping it is this one
 * constant.
 */
const ASSAY_DEFAULT_FORM: ShowroomForm = "card";
 
export interface FormInput {
  product_type: "curve" | "standard" | "accessory";
  is_assay: boolean;
  showroom_form: string | null;
}
 
/**
 * Resolve the form. An explicit `showroom_form` always wins — the operator can
 * override any product, including putting a single piece on a standee while
 * its photograph is being made.
 */
export function resolveForm(p: FormInput): ShowroomForm {
  if (isShowroomForm(p.showroom_form)) return p.showroom_form;
  if (p.product_type === "accessory") return "box";
  if (p.is_assay) return ASSAY_DEFAULT_FORM;
  return "bar";
}
