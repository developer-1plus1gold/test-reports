/**
 * SSRF guard for any URL we hand to a server-side fetch / automated runtime.
 *
 * Requires https + a public host: rejects localhost / private / link-local /
 * loopback / cloud-metadata endpoints, incl. IPv4-mapped-IPv6 (::ffff:127.x).
 *
 * Caveat: this is a STRING check only. It does not resolve the host to an IP at
 * fetch time, so it does not defend DNS-rebinding on its own — when a URL comes
 * from a genuinely untrusted source, also re-check the resolved IP at fetch
 * time. For our current callers the inputs are first-party (Replicate output,
 * our own Supabase signed URLs, an admin-entered supplier URL), so the string
 * check is appropriate defense-in-depth.
 *
 * Single source: imported by the supplier fetch-config route (write-time) and
 * marketing storage ingestFromUrl (fetch-time). Do not re-implement locally.
 */
export function isSafeFetchUrl(raw: string): boolean {
  let u: URL;
  try { u = new URL(raw); } catch { return false; }
  if (u.protocol !== "https:") return false;
  const host = u.hostname.toLowerCase().replace(/^\[|\]$/g, "");
  if (host === "localhost" || host.endsWith(".localhost") || host.endsWith(".internal") || host.endsWith(".local")) return false;
  // Cloud metadata endpoints.
  if (host === "metadata.google.internal" || host === "metadata.gcp.internal" || host === "169.254.169.254") return false;
  // IPv4-mapped IPv6 has no legitimate use in a public URL and the WHATWG URL
  // parser normalises the dotted-quad to hex (::ffff:127.0.0.1 → ::ffff:7f00:1),
  // which would slip past a dotted-decimal private-range check — reject outright.
  if (/::ffff:/i.test(host)) return false;
  // (kept for any dotted form that survives) strip the prefix and re-check the v4.
  const v4 = host.replace(/^::ffff:/i, "");
  // Reject literal private / loopback / link-local IPs.
  if (/^127\./.test(v4) || /^10\./.test(v4) || /^192\.168\./.test(v4) ||
      /^172\.(1[6-9]|2[0-9]|3[01])\./.test(v4) || /^169\.254\./.test(v4) ||
      v4 === "0.0.0.0" || host === "::1" || host === "::" ||
      host.startsWith("fc") || host.startsWith("fd") || host.startsWith("fe80")) return false;
  return true;
}
 
/**
 * Stricter than isSafeFetchUrl: requires a safe URL AND a host in an explicit
 * allowlist (exact match or a sub-domain of an allowed suffix).
 *
 * Use this — not the bare isSafeFetchUrl — at any boundary where we either
 * (a) attach a credential to the request (the URL must be the API we own a key
 * for, never an attacker-influenced host = credential exfil), or (b) ingest the
 * bytes of a URL derived from an untrusted third-party response (a poisoned
 * Replicate/Veo `output`/`name` field must not be able to redirect us to an
 * arbitrary public host). isSafeFetchUrl only blocks INTERNAL SSRF; it still
 * allows any public host, so on its own it does not stop public-host exfil.
 */
export function isHostAllowed(raw: string, allowedSuffixes: string[]): boolean {
  if (!isSafeFetchUrl(raw)) return false;
  let host: string;
  try { host = new URL(raw).hostname.toLowerCase(); } catch { return false; }
  return allowedSuffixes.some((s) => {
    const suf = s.toLowerCase();
    return host === suf || host.endsWith("." + suf);
  });
}
 
/** Throwing variant of isHostAllowed — for fetch boundaries that should hard-fail. */
export function assertHostAllowed(raw: string, allowedSuffixes: string[], label: string): void {
  if (!isHostAllowed(raw, allowedSuffixes)) {
    throw new Error(`${label}: refusing URL whose host is not in the allowlist`);
  }
}
 
/** Google APIs — Gemini / Veo endpoints and the file URIs they return (we

export const GOOGLE_API_HOSTS = ["googleapis.com"];
 
/** Replicate prediction output delivery hosts (where finished images/videos
 *  live). A Replicate `output` URL we ingest must be on one of these. */
export const REPLICATE_OUTPUT_HOSTS = ["replicate.delivery", "replicate.com"];
 
/** Our own Supabase project host, derived from the configured URL. Signed
 *  storage URLs we generate ourselves always live here — so when we re-fetch
 *  one server-side we can pin the host instead of trusting any public URL. */
export function ownSupabaseHost(): string | null {
  try { return new URL(process.env.NEXT_PUBLIC_SUPABASE_URL!).hostname.toLowerCase(); }
  catch { return null; }
}
 
/** True if `raw` is one of OUR Supabase signed URLs (own project host). Falls
 *  back to the generic safe-fetch check if the project host can't be derived
 *  (e.g. unit tests without the env), so it never blocks a legitimately-safe
 *  first-party URL just because the env is absent. */
export function isOwnSupabaseUrl(raw: string): boolean {
  const own = ownSupabaseHost();
  return own ? isHostAllowed(raw, [own]) : isSafeFetchUrl(raw);
}
