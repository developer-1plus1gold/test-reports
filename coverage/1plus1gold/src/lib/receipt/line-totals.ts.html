/**
 * Receipt line amounts that add up to what the customer was charged.
 *
 * `order_items.price_sgd` is stored at 2 decimals, but checkout charges the
 * UNROUNDED unit price × qty and stores that sum, rounded once, as the order's
 * gold subtotal (`orders.total_sgd`). So Σ(qty × price_sgd) can differ from
 * `total_sgd` by a few cents (PROD 2026-10-30: 80 of 367 orders, up to 23¢),
 * and the receipt's line amounts then do not add up to its Subtotal/Total.
 * Customers were always charged correctly; this is display only.
 *
 * Fix (parked 2026-09-28, option 1): derive the gold lines from the charged
 * subtotal — each line keeps its rounded amount and the residual cent(s) go to
 * the LAST gold line, so the lines sum to `total_sgd` exactly. Accessory lines
 * (grams 0) are catalogue 2dp prices and already sum exactly; they are left as is.
 *
 * A residual larger than per-unit rounding can explain is not a rounding
 * artefact (e.g. a pre-launch test order whose lines were edited); then nothing
 * is moved and the lines print as stored, rather than hiding real money in one line.
 */
 
export interface LineTotalInput {
  grams:     number;
  quantity:  number;
  price_sgd: number;
}
 
function round2(n: number): number {
  return Math.round(n * 100) / 100;
}
 
/** Residual (cents, signed) that the last gold line absorbs, and its index;
 *  `null` when there is nothing to move or the gap is not a rounding artefact. */
export function goldLineResidual(
  items: readonly LineTotalInput[],
  goldSubtotalSgd: number | null | undefined,
): { index: number; residual: number } | null {
  if (goldSubtotalSgd == null || !Number.isFinite(Number(goldSubtotalSgd))) return null;
  let last = -1;
  let sum = 0;
  let units = 0;
  items.forEach((it, i) => {
    if (Number(it.grams) > 0) {
      last = i;
      sum += round2(Number(it.price_sgd) * Number(it.quantity));
      units += Number(it.quantity);
    }
  });
  if (last < 0) return null;
  const residual = round2(Number(goldSubtotalSgd) - sum);
  if (residual === 0) return null;
  // Each unit price was rounded by at most half a cent, so a true rounding
  // residual is bounded by units × 0.005 (+1¢ for the subtotal's own rounding).
  if (Math.abs(residual) > units * 0.005 + 0.01 + 1e-9) return null;
  return { index: last, residual };
}
 
/** The items with `line_total_sgd` set on every line, gold lines summing to
 *  `goldSubtotalSgd` (see the module comment). Input items are not mutated. */
export function withReconciledLineTotals<T extends LineTotalInput>(
  items: readonly T[],
  goldSubtotalSgd: number | null | undefined,
): Array<T & { line_total_sgd: number }> {
  const adj = goldLineResidual(items, goldSubtotalSgd);
  return items.map((it, i) => {
    const base = round2(Number(it.price_sgd) * Number(it.quantity));
    return { ...it, line_total_sgd: adj && adj.index === i ? round2(base + adj.residual) : base };
  });
}
