function fmtSgd(n: number) {
  return "S$ " + n.toFixed(2).replace(/\B(?=(\d{3})+(?!\d))/g, ",");
}
 
function fmtDateTimeSGT(iso: string) {
  return new Date(iso).toLocaleString("en-SG", {
    day: "2-digit", month: "short", year: "numeric",
    hour: "2-digit", minute: "2-digit", hour12: true,
    timeZone: "Asia/Singapore",
  }) + " SGT";
}
 
 
function formatServiceType(raw: string | null | undefined): string | null {
  if (!raw) return null;
  switch (raw.toUpperCase()) {
    case "MOTORCYCLE": return "Lalamove Express";
    case "SEDAN":      return "Lalamove Express (Car)";
    case "NEXT_DAY":   return "NinjaVan Next Day";
    default:           return raw;
  }
}
 
function sanitizeText(s: string | null | undefined): string | null {
  if (!s) return null;
  return s.replace(/[\x00-\x1F\x7F<>]/g, "").slice(0, 100).trim() || null;
}

/**

 * "TAX INVOICE · Split X of N" (no PAID badge — payment is on the master page)

export function ReceiptPage({ receipt, split }: { receipt: ReceiptData; split?: SplitPageMeta }) {
  const baseRef         = receipt.orderId.slice(0, 8).toUpperCase();
  const ref             = split ? `${baseRef}-S${split.index}` : baseRef;
  const customerName    = sanitizeText(receipt.customerName);
  const customerPhone   = sanitizeText(receipt.customerPhone);
 
  const hasShipping      = receipt.shippingFeeSgd != null && receipt.shippingFeeSgd > 0;
  const hasProcessingFee = receipt.processingFeeSgd != null && receipt.processingFeeSgd > 0;
  const hasPromoDiscount    = (receipt.promoDiscountSgd ?? 0) > 0;
  const hasReferralDiscount = (receipt.referralDiscountSgd ?? 0) > 0;
  const hasAnyDiscount      = hasPromoDiscount || hasReferralDiscount;
 
  // IPM vs taxable split (computed from items — no reliance on stored subtotals)
  const subtotalIpm     = receipt.items.filter((i) => i.gstTreatment === "ipm_exempt")
    .reduce((s, i) => s + i.price_sgd * i.quantity, 0);
  const subtotalTaxable = receipt.items.filter((i) => i.gstTreatment === "standard_rated")
    .reduce((s, i) => s + i.price_sgd * i.quantity, 0);
  const hasMixed        = subtotalIpm > 0 && subtotalTaxable > 0;

                     + (receipt.processingFeeSgd ?? 0);

              <Text style={s.colGrams}>{item.grams === 0 ? "—" : `${(item.grams * item.quantity).toFixed(1)}g`}</Text>

              <Text style={s.grandVal}>{fmtSgd(grandTotal)}</Text>
            </View>
          </View>
        </View>
 
        {/* Spacer: pushes footer to bottom without stretching body (which caused blank page 2) */}
        <View style={{ flex: 1, backgroundColor: cream }} />
 
        <View style={s.goldRule} />

        <View style={s.footer}>
          <Text style={s.footerGold}>{split ? "ELECTRONIC TAX INVOICE" : "ELECTRONIC RECEIPT"} · NO SIGNATURE REQUIRED</Text>

          <Text style={s.footerText}>
            1Plus1Gold sends receipts and account notices via WhatsApp and email only. We will never contact you by SMS or any other channel.

            </Text>
          )}
          <Text style={[s.footerText, { marginBottom: 4 }]}>
            {BUSINESS_CONFIG.gst.isRegistered

          </Text>
          <Text style={s.footerContact}>For support, visit {BUSINESS_CONFIG.contactUrl}</Text>
          <Text style={s.footerReg}>
            {BUSINESS_CONFIG.name} · UEN {BUSINESS_CONFIG.uen} · {regulatoryShortLine()}
          </Text>

        </View>
 
      </Page>

/**
 * Split-invoice document: page 1 is the ORIGINAL invoice (byte-identical to
 * generateReceiptPdf — same master ReceiptData, no `split` prop), followed by
 * one page per distributed invoice. Admin-only, render-only (see
 * src/lib/receipt/split-invoice.ts for how the split ReceiptData set is built).
 */
export async function generateSplitInvoicePdf(
  master: ReceiptData,
  splits: { data: ReceiptData; index: number; total: number; mode: SplitPageMeta["mode"] }[],
): Promise<Buffer> {
  const ref = master.orderId.slice(0, 8).toUpperCase();
  return await renderToBuffer(
    <Document title={`1Plus1Gold Invoice (split) — ${ref}`}>
      <ReceiptPage receipt={master} />
      {splits.map((sp) => (
        <ReceiptPage key={sp.index} receipt={sp.data} split={{ index: sp.index, total: sp.total, mode: sp.mode }} />
      ))}
    </Document>,
  );
}
