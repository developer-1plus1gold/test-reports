/**
 * Stripe is RETIRED (Mukesh, 2026-10-18 — docs/razorpay-integration-design.md D5).
 *
 * Razorpay + HitPay cover every method Stripe offered, so the routes that START
 * a Stripe payment now answer 410 Gone. The code is kept (not deleted) so the
 * rail can be revived without archaeology — flip STRIPE_RETIRED=false.
 *
 * What is deliberately NOT retired:
 *   - /api/webhooks/stripe stays live, so a late event (refund, dispute, a
 *     payment that was already in flight) is still recorded.
 *   - /api/payments/stripe/status stays live (read-only) for the same reason.
 *
 * Fails toward retired: anything other than the exact string "false" is retired.
 */
import { NextResponse } from "next/server";
 
export function isStripeRetired(env: Record<string, string | undefined> = process.env): boolean {
  return env.STRIPE_RETIRED !== "false";
}
 
export function stripeRetiredResponse(): NextResponse {
  return NextResponse.json(
    { error: "stripe_retired", message: "Card payments now go through our other payment providers. Please choose another payment method." },
    { status: 410 },
  );
}
