/**
 * The hosted payment gateways, in ONE place (2026-11-07).
 *
 * Every hosted gateway (HitPay, Razorpay, Airwallex) works the same way: the
 * server creates a hosted checkout, the customer pays on the gateway's page and
 * comes back to /payment/confirming?<refParam>=<reference>, and the order is
 * completed from a server-side fact. What differs is only data — this table.
 *
 * Before this module, every caller asked `startsWith("razorpay_")` /
 * `ref.startsWith("RP-")` with HitPay as the silent default. A third gateway
 * would then have fallen into HitPay's branch everywhere it was not added —
 * the worst case being MAG redeem settlement, which would have looked an
 * Airwallex reference up in HitPay's table, found nothing and RELEASED the
 * held grams of a paid order. Callers now ask this registry instead.
 *
 * Pure: no server imports (checkout + confirming pages import it).
 */
 
export type OnlineGatewayKey = "hitpay" | "razorpay" | "airwallex";
 
export interface OnlineGateway {
  key:          OnlineGatewayKey;
  /** Customer-facing name ("You'll complete payment on HitPay's secure page"). */
  label:        string;
  /** payment_gateway_fees.method_key prefix. */
  methodPrefix: string;
  /** Reference prefix minted by gateway-create (orders), e.g. "HP-…". */
  refPrefix:    string;
  /** Query param /payment/confirming (and the iOS return URL) carries the reference in. */
  refParam:     string;
  createRoute:  string;
  statusRoute:  string;
  /** Query param that asks the status route to check the gateway's API. */
  statusCheckParam: string;
  /** Pending-payments table keyed by reference_number. */
  pendingTable: string;
}
