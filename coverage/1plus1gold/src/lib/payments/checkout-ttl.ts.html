/**
 * How long a hosted checkout (Razorpay link, HitPay payment request, Airwallex
 * session) stays payable. Mukesh 2026-11-23: an abandoned checkout must not stay
 * payable, because checkout locks the price for only 10 minutes
 * (locked-spot-guard). 20 minutes covers the lock plus the time to pay, and
 * Razorpay's `expire_by` needs at least 15.
 *
 * Razorpay links carry it as `expire_by`. HitPay requests and Airwallex sessions
 * have no create-time expiry we can set (HitPay's `expiry_date` is for repeated
 * payments; an Airwallex session closes itself after 1 hour), so the 10-minute
 * reconcile cron deletes or cancels an unattempted checkout once it is this old.
 */
export const CHECKOUT_TTL_MINUTES = 20;
 
/** Pure: has a checkout created at `createdAt` outlived the TTL at `now`? An
 *  unreadable timestamp is treated as NOT expired, so nothing is cancelled on a guess. */
export function isPastCheckoutTtl(createdAt: unknown, now: number = Date.now()): boolean {
  if (typeof createdAt !== "string") return false;
  const t = Date.parse(createdAt);
  return Number.isFinite(t) && now - t >= CHECKOUT_TTL_MINUTES * 60_000;
}
