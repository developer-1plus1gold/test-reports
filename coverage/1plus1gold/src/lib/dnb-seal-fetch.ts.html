/**
 * Reads D&B's live seal page and turns it into the two things the NATIVE iOS
 * surface needs: one composited seal image, and the key that opens our profile.
 *
 * ▶ WHY THIS EXISTS AT ALL — the web embed cannot work inside the app.
 * On the web the seal is an <iframe> (see @/lib/dnb-seal) and its click does:
 *
 *     window.open('', 'formresult', …);   // an EMPTY named window
 *     form2.submit();                     // …then POSTs iSealKey into it
 *
 * Inside the app's WKWebView, `createWebViewWith` (WebViewWrapper.swift) always
 * returns nil and routes an external URL through `UIApplication.shared.open` —
 * a GET, which drops the POST body. MEASURED against their live endpoint
 * (2026-09-10): a GET to TPSING-COMP-004.aspx carrying the key returns a 2.2 KB
 * empty shell titled "D-U-N-S Registered Profile"; the same request as a POST
 * returns 16 KB naming 1PLUS1GOLD and DUNS 659512753. So the profile is
 * POST-ONLY and there is no link to open. The native band therefore renders a
 * composited image and hands the tap to `/dnb-profile`, a same-origin bridge
 * that performs the POST from the system browser, outside our delegate.
 *
 * ▶ WHY PARSE RATHER THAN HARDCODE. The layer filenames are obfuscated
 * (`xdgfkjghdf.png`, `mxfdjhkgd.png`, …) and the seal key is issued by D&B —
 * both are theirs to rotate. Hardcoding either buys a badge that breaks
 * silently on their schedule, which is the failure this whole area keeps
 * producing. Re-reading the page each cache cycle self-heals.
 *
 * ▶ WHY COMPOSITE. The seal is FIVE 114x97 layers stacked by z-index, one of
 * them a 25-frame GIF. VERIFIED by rendering both ways: drop the GIF and the
 * "D-U-N-S® REGISTERED™" band disappears entirely — frame 1 carries it. So the
 * composite is the seal as a browser draws it, flattened, not an edit of it.
 * SwiftUI's AsyncImage cannot stack five layers or animate a GIF; one image can.
 *
 * ▶ The page 500s without a Referer from our own domain — it authenticates on
 * the referring site. Confirmed: no Referer → HTTP 500; with one → 200.
 */
import sharp from "sharp";
 
const SEAL_PAGE = "https://dunsregistered.dnb.com/SealAuthentication.aspx?Cid=1";
const REFERER = "https://1plus1gold.com/";
 
/**
 * Hosts we will follow a parsed URL to. The layer URLs come out of a THIRD
 * PARTY'S HTML, so fetching them unchecked would make this route an SSRF
 * gadget — anything they (or anyone who could tamper with that response) put in
 * an <img src> would be fetched by our server. Allow-list first, fetch second.
 */
const ALLOWED_HOSTS = /(?:^|\.)(?:dnb\.com|dunsregistered\.com)$/i;
 
export interface SealSource {
  /** Layer image URLs in DOM order, which is z-index order. */
  layers: string[];
  /** The key their profile page expects in a POST body. */
  sealKey: string;
}
 
function isAllowed(raw: string): boolean {
  try {
    const u = new URL(raw);
    return u.protocol === "https:" && ALLOWED_HOSTS.test(u.hostname);
  } catch {
    return false;
  }
}
 
/** Reads the live seal page and extracts its layers + key. */
export async function fetchSealSource(signal?: AbortSignal): Promise<SealSource> {
  const res = await fetch(SEAL_PAGE, {
    headers: { Referer: REFERER, "User-Agent": "Mozilla/5.0 (compatible; 1Plus1Gold/1.0)" },
    redirect: "follow",
    signal,
  });
  if (!res.ok) throw new Error(`seal page ${res.status}`);
  const html = await res.text();
 
  // DOM order == z-index order on their page (each layer declares z-index 0..4
  // ascending in source order), so a plain scan preserves the stack.
  const layers = [...html.matchAll(/<img[^>]+src="([^"]+)"/gi)]
    .map((m) => m[1].trim())
    .filter(isAllowed);
 
  const key = html.match(/name="iSealKey"[^>]*value="([^"]+)"/i)?.[1]?.trim() ?? "";
 
  if (layers.length === 0) throw new Error("no seal layers found");
  // Strictly alphanumeric, and that is LOAD-BEARING beyond validation: the
  // /dnb-profile bridge is a route handler that interpolates this straight into
  // hand-written HTML, where — unlike the JSX it replaced — nothing escapes it
  // for us. A key that cannot contain a quote or an angle bracket cannot break
  // out of the attribute. Do not relax this without escaping at that call site.
  if (!/^[A-Za-z0-9]{6,32}$/.test(key)) throw new Error("no usable seal key");
  return { layers, sealKey: key };
}
 
/**
 * Flattens the layers into a single WebP. The GIF layer is read as a still —
 * `animated: false` takes frame 1, which is the frame carrying the REGISTERED
 * band (verified by compositing with and without it).
 */
export async function compositeSeal(layers: string[], signal?: AbortSignal): Promise<Buffer> {
  const buffers = await Promise.all(
    layers.map(async (url) => {
      if (!isAllowed(url)) throw new Error("layer host not allowed");
      const r = await fetch(url, { signal });
      if (!r.ok) throw new Error(`layer ${r.status}`);
      return sharp(Buffer.from(await r.arrayBuffer()), { animated: false }).png().toBuffer();
    }),
  );
 
  const meta = await sharp(buffers[0]).metadata();
  return sharp({
    create: {
      width: meta.width ?? 114,
      height: meta.height ?? 97,
      channels: 4,
      background: { r: 0, g: 0, b: 0, alpha: 0 },
    },
  })
    .composite(buffers.map((input) => ({ input })))
    .webp({ quality: 95, alphaQuality: 100 })
    .toBuffer();
}
