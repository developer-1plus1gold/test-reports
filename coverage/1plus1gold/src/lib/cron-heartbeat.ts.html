import { NextResponse } from "next/server";
import { createClient, type SupabaseClient } from "@supabase/supabase-js";
import { recordCronRun } from "./cron-runs";
 
/**
 * Wrap a cron handler so it reports that it ran — every tick, quiet ones included.
 *
 * ▶ WHY A WRAPPER RATHER THAN A CALL IN EACH ROUTE. Instrumenting was opt-in,
 * and 47 of 59 crons did not opt in: `/admin/crons` showed an empty Last-run
 * cell for four fifths of the catalogue, so a dead job and a healthy one looked
 * identical on the screen we use to tell them apart. Adding 47 hand-written
 * calls would have fixed today and left the same hole open for cron number 60.
 * The wrapper closes it structurally: the heartbeat is part of how a cron is
 * exported, and `cron-heartbeat-coverage.test.ts` fails the PR if a registered
 * cron is neither wrapped nor calling `recordCronRun` itself.
 *
 * ▶ IT REPORTS, IT DOES NOT DECIDE. The handler's response is returned
 * untouched and a throw is re-thrown — the wrapper must be invisible to
 * behaviour, because it is now in the path of money crons (hitpay-reconcile,
 * receipt-retry, spx-status-poll). A heartbeat that failed to write is swallowed
 * by `recordCronRun`; nothing here can change a cron's outcome.
 *
 * ▶ AN UNAUTHORISED CALL IS NOT A RUN. Cron endpoints are public URLs, so a 401
 * or 403 writes nothing. Otherwise anyone could keep a dead cron looking alive
 * by curling it — a liveness signal anybody can forge is not a liveness signal.
 *
 * ▶ THE 12 CRONS THAT ALREADY CALL `recordCronRun` ARE DELIBERATELY NOT WRAPPED.
 * Their details are hand-written and genuinely better than anything derivable
 * (`cutoff=2026-06-18; snapshots_redacted=13`), and a wrapper row written
 * milliseconds later would shadow that with a generic one. They keep one
 * residual gap — a throw before their own call writes nothing — and that is
 * covered by staleness detection in `cron-health.ts` rather than by a second
 * heartbeat: no row eventually reads as `late`, which is the right answer.
 */
 
type Handler<Req extends Request> = (req: Req) => Promise<Response> | Response;
 
let cached: SupabaseClient | null = null;
function client(): SupabaseClient {
  // Lazy so importing this module in a test does not require Supabase env.
  if (!cached) {
    cached = createClient(
      process.env.NEXT_PUBLIC_SUPABASE_URL!,
      process.env.SUPABASE_SERVICE_KEY!,
    );
  }
  return cached;
}
 
/**
 * Turn a cron's own JSON response into a heartbeat detail.
 *
 * The routes already return exactly the right thing — `{scanned, recovered,
 * failed, stillPending}` — so the detail is genuinely informative without any
 * per-route work. Scalars only: an object or array would blow the 300-char cap
 * and bury the counts that matter.
 */
export function detailFromBody(body: unknown): string | undefined {
  if (!body || typeof body !== "object" || Array.isArray(body)) return undefined;
  const parts: string[] = [];
  for (const [k, v] of Object.entries(body as Record<string, unknown>)) {
    if (typeof v === "number" || typeof v === "boolean") parts.push(`${k}=${v}`);
    else if (typeof v === "string" && v.length <= 80) parts.push(`${k}=${v}`);
    if (parts.join(" ").length > 280) break;
  }
  return parts.length > 0 ? parts.join(" ").slice(0, 300) : undefined;
}
 
export function withCronHeartbeat<Req extends Request>(
  cronName: string,
  handler: Handler<Req>,
): (req: Req) => Promise<Response> {
  return async (req: Req): Promise<Response> => {
    let res: Response;
    try {
      res = await handler(req);
    } catch (err) {
      // Record, then re-throw so Next still produces its own 500 — the wrapper
      // must not convert a crash into a success-shaped response.
      await recordCronRun(
        client(),
        cronName,
        "error",
        `threw: ${err instanceof Error ? err.message.slice(0, 160) : "unknown"}`,
      );
      throw err;
    }
 
    if (res.status === 401 || res.status === 403) return res;
 
    let detail: string | undefined;
    try {
      // Clone so the handler's own body is still readable by the framework.
      detail = detailFromBody(await res.clone().json());
    } catch {
      detail = undefined; // non-JSON (a stream, a redirect) — status is enough.
    }
    await recordCronRun(client(), cronName, res.ok ? "ok" : "error",
      detail ?? `http_${res.status}`);
    return res;
  };
}
