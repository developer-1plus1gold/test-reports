/**
 * Cost per order — marketing spend against the orders it is meant to produce
 * (docs/marketing-scorecard-design.md §1, Mukesh 2026-10-14).
 *
 * Pure: the route fetches rows, this decides what they mean. One function, two
 * consumers (the dashboard tab and the Outreach scorecard), so the two numbers
 * can never disagree.
 *
 * HONEST-NUMBER RULES, each one a way a nicer-looking figure would be wrong:
 *  1. BLENDED, NOT ATTRIBUTED. Web cost per order = every web-driving channel's
 *     spend ÷ every real web order. Buyers see an ad and come back directly days
 *     later (agreed 2026-10-14), so crediting a single ad would invent precision.
 *  2. SPREAD, NOT LUMPED. Spend goes through the ledger's own amortisation
 *     `schedule()`, so an annual subscription counts 1/12 a month — the same
 *     recognition the P&L uses.
 *  3. A MISSING INVOICE IS SAID, NOT HIDDEN. In the current month a channel that
 *     had spend last month and none booked yet is flagged `pending`, rather than
 *     quietly making the month look cheap. Bookings are usually entered after the
 *     invoice arrives (Mukesh: "the dashboard looks at live data").
 *  4. NO DIVISION BY ZERO DRESSED AS A NUMBER. Spend with no orders → null, not
 *     0 or Infinity.
 */

/** Map a ledger vendor string to a channel. Case-insensitive, substring-based
 *  because invoices carry legal names ("LAZADA SINGAPORE PTE. LTD."). */
export function channelForVendor(vendor: string | null | undefined): SpendChannel {
  const v = (vendor ?? "").toLowerCase();
  if (/openai|chatgpt/.test(v)) return "openai";
  if (/meta|facebook|instagram/.test(v)) return "meta";
  if (/google/.test(v)) return "google";
  if (/linkedin/.test(v)) return "linkedin";
  if (/shopee/.test(v)) return "shopee";
  if (/lazada/.test(v)) return "lazada";
  if (/carousell/.test(v)) return "carousell";
  return "other";
}

/** Statuses that never became a real sale. */
const NOT_A_SALE = new Set(["cancelled", "refunded", "failed", "pending"]);
 
/** A real order: not cancelled/failed, not an internal/demo account, not a stock swap. */
export function isRealOrder(o: OrderRow, internalIds: ReadonlySet<string>): boolean {
  if (NOT_A_SALE.has((o.status ?? "").toLowerCase())) return false;
  if (o.customer_id && internalIds.has(o.customer_id)) return false;
  // Settle-by-swap invoices to a supplier are booked as in-person orders.
  if ((o.external_order_id ?? "").toUpperCase().startsWith("SWAP-")) return false;
  return true;
}
 
/** "YYYY-MM" of an instant, in Singapore time. */
export function sgtMonth(iso: string): string {
  const ms = Date.parse(iso) + 8 * 3_600_000;
  return new Date(ms).toISOString().slice(0, 7);
}
 
function nextMonth(m: string): string {
  const [y, mo] = m.split("-").map(Number);
  return mo === 12 ? `${y + 1}-01` : `${y}-${String(mo + 1).padStart(2, "0")}`;
}
function prevMonth(m: string): string {
  const [y, mo] = m.split("-").map(Number);
  return mo === 1 ? `${y - 1}-12` : `${y}-${String(mo - 1).padStart(2, "0")}`;
}
