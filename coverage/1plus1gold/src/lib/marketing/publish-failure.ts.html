/**
 * Making a BLOCKED marketing publish visible.
 *
 * ── Why this exists ────────────────────────────────────────────────────────
 * Found 2026-09-02. Drumbeat piece `b0ea1213` ("If Gold were currency, how
 * many grams is daily life?") was scheduled for 2026-08-31 23:30 UTC and never
 * published. The publisher was NOT broken — it refused, correctly: the piece's
 * video asset was still `generated` (never operator-approved) and no brand
 * review existed, so the publish gate rejected it with HTTP 409 on every tick.
 *
 * The defect was that the refusal was SILENT. The `*​/5` cron retried roughly
 * 340 times across 28 hours and wrote nothing — no `cron_runs` heartbeat, no
 * admin alert, no decision row — and it discarded the response BODY, keeping
 * only `HTTP ${status}`, so even its own return value could not say WHICH gate
 * had failed. From the outside, "blocked on an unapproved asset" and "nothing
 * due" looked identical.
 *
 * That is the standing question this codebase keeps asking: *what would this
 * look like if it were silently doing nothing?* Here the answer was "exactly
 * the same", which is the signal that a loud failure path was needed before
 * anything else.
 *
 * ── The terminal/transient split, and why it is load-bearing ───────────────
 * A 4xx from the publish route is a GATE decision: the request is refused for
 * a reason that cannot change on its own, so retrying forever is pointless and
 * a human must act. Those alert immediately — waiting adds nothing but delay
 * on a post that is already late.
 *
 * A 5xx or a network error MIGHT clear on the next tick (an upstream blip, a
 * cold start). Paging on the first one would train the reader to ignore the
 * channel, which is how the original silence survived. Those alert only once
 * the piece is genuinely overdue.
 *
 * Everything here is pure so the decision to page — the part that is easy to
 * get subtly wrong and hard to notice — is unit-testable without a network,
 * a database, or a clock.
 */
 
/** A publish attempt's failure mode. */
export type PublishFailureClass = "terminal" | "transient";
 
/** One dispatch attempt for one scheduled piece. */
export interface PublishAttempt {
  pieceId: string;
  /** The piece's theme, for a human-readable alert. */
  theme?: string | null;
  ok: boolean;
  /** HTTP status from the publish route; 0/undefined when the fetch itself threw. */
  httpStatus?: number;
  /** The route's `error` field (e.g. `assets_not_approved`), when it returned JSON. */
  errorCode?: string;
  /** How long past `scheduled_at` this piece is, in minutes. */
  overdueMinutes: number;
  /** The piece's `scheduled_at`, so a RESCHEDULE re-arms the alert. */
  scheduledAt?: string | null;
}
 
/**
 * Transient failures must persist this long past `scheduled_at` before they
 * page. One `*​/5` tick is 5 minutes, so this is ~3 chances to self-clear.
 * Terminal failures ignore it entirely.
 */
export const TRANSIENT_GRACE_MINUTES = 15;
 
/**
 * A due piece the automated publisher has nothing left to do on, because its
 * only unresolved channels are prepared MANUAL posts waiting on the operator.
 */
export interface AwaitingManualPiece {
  pieceId: string;
  theme?: string | null;
  /** The manual channels still pending (today: `whatsapp_status`). */
  channels: string[];
  /** How long past `scheduled_at` this piece is, in minutes. */
  overdueMinutes: number;
  /** The piece's `scheduled_at`, so a RESCHEDULE re-arms the reminder. */
  scheduledAt?: string | null;
}
 
/**
 * How long a piece may sit awaiting a manual post before we say so again.
 *
 * The publish route already fires a one-shot "Manual WhatsApp Status post
 * needed" alert the moment it prepares the channel, so the FIRST nudge is not
 * this module's job — a second message minutes later would be noise. What was
 * missing is what happens when that single message is missed: the piece then
 * waits forever, `published_at` never lands, and the heartbeat stays green.
 * Six hours is long enough that the original alert has plainly not been acted
 * on, and short enough that a post scheduled in the morning is not still
 * un-finished at night.
 */
export const MANUAL_NUDGE_AFTER_MINUTES = 360;
 
/** After the first reminder, repeat at most once a day. */
export const MANUAL_NUDGE_INTERVAL_MINUTES = 1440;
 
/**
 * 4xx → terminal (the gate refused; retrying cannot change the answer).
 * 5xx, 0, or unknown → transient (may clear on its own).
 *
 * Deliberately keyed on the status CLASS rather than on a list of error codes:
 * a code added to the route later must not silently fall through to
 * "transient" and stop paging. Unknown 4xx codes still alert.
 */
export function classifyPublishFailure(httpStatus?: number): PublishFailureClass {
  if (typeof httpStatus === "number" && httpStatus >= 400 && httpStatus < 500) return "terminal";
  return "transient";
}
 
/**
 * Dedup key for an alert: one page per (piece, blocker, scheduled_at).
 *
 * Keying on the BLOCKER as well as the piece is deliberate. If the operator
 * clears one gate and the piece then fails a different one, that is new
 * information and pages again. If it keeps failing the same way, it stays
 * quiet after the first — which is what stops a `*​/5` cron sending 288
 * messages a day.
 *
 * ⚠️ `scheduledAt` is in the key for a reason that nearly bit us on day one.
 * The expected next action on a blocked piece is to RESCHEDULE it — and with a
 * (piece, blocker) key, a piece rescheduled while still un-approved would fail
 * exactly the same way and be suppressed by its own earlier alert. That is the
 * original silent-failure bug reintroduced on the very piece this machinery was
 * built for. Including the schedule makes each new attempt a new event, while
 * still collapsing the ~288 retries within one schedule into a single page.
 */
export function publishAlertKey(pieceId: string, errorCode?: string, scheduledAt?: string | null): string {
  return `${pieceId}:${errorCode ?? "unknown"}:${scheduledAt ?? "unscheduled"}`;
}
 
/** Whether this failed attempt warrants paging a human now. */
export function shouldAlertPublishFailure(
  attempt: PublishAttempt,
  graceMinutes: number = TRANSIENT_GRACE_MINUTES,
): boolean {
  if (attempt.ok) return false;
  if (classifyPublishFailure(attempt.httpStatus) === "terminal") return true;
  return attempt.overdueMinutes >= graceMinutes;
}
 
/**
 * What each blocker MEANS and what to do about it.
 *
 * The remedy is the whole point: at 07:30 the useful message is not
 * "HTTP 409", it is which button to press. Codes mirror the publish route's
 * `error` values exactly — keep them in step if that route grows a new one.
 */
const BLOCKER_HELP: Record<string, string> = {
  assets_not_approved:
    "the media is not operator-approved. Open the piece in Admin → Marketing and Approve the asset (or run a brand review and let it pass).",
  no_brand_review:
    "no brand sign-off: the media is not operator-approved and no brand review exists. Approve the asset in Admin → Marketing.",
  brand_review_blocking:
    "the latest brand review did not pass. Re-run it, or grant an override, or approve the assets directly.",
  no_publishable_asset:
    "no asset is assigned to the channels being published. Assign media to each target channel.",
  no_approved_caption_for_channel:
    "a target channel has no approved/edited caption. Approve one caption variant per channel.",
  piece_wrong_status:
    "the piece is not in a publishable status (needs `scheduled` or `brand_reviewed`).",
  no_requested_channel_in_plan:
    "the requested channels are not in the piece's plan.",
  piece_not_found:
    "the piece no longer exists.",
  invalid_json:
    "the publish request body was malformed — this is a code defect, not an operator step.",
  agent_failed:
    "the publisher agent threw while calling a channel API. Check the channel credentials (an expired LinkedIn token presents this way).",
};
 
/** Human explanation + remedy for a blocker code; falls back to the raw code. */
export function describePublishBlocker(errorCode?: string): string {
  if (!errorCode) return "the publish attempt failed with no error code returned.";
  return BLOCKER_HELP[errorCode] ?? `the publish route refused with \`${errorCode}\`.`;
}
 
/** The WhatsApp/decision message for a blocked piece. */
export function buildPublishFailureAlert(attempt: PublishAttempt): string {
  const theme = attempt.theme ? `"${attempt.theme}"` : attempt.pieceId.slice(0, 8);
  const overdue =
    attempt.overdueMinutes >= 60
      ? `${Math.floor(attempt.overdueMinutes / 60)}h${attempt.overdueMinutes % 60 ? ` ${attempt.overdueMinutes % 60}m` : ""}`
      : `${attempt.overdueMinutes}m`;
  const kind = classifyPublishFailure(attempt.httpStatus);
  const tail =
    kind === "terminal"
      ? "It will NOT retry into success — this needs a human."
      : "It will keep retrying every 5 minutes.";
  return (
    `Scheduled post ${theme} did not publish (${overdue} overdue). ` +
    `Blocker: ${describePublishBlocker(attempt.errorCode)} ${tail}`
  );
}
 
/**
 * One-line heartbeat detail for `cron_runs`, e.g.
 *   `due=2 published=1 blocked=1 (b0ea1213=assets_not_approved)`
 *   `due=1 published=0 awaiting_manual=1 (b0ea1213=whatsapp_status) blocked=0`
 *
 * The blocked pieces are named inline because a bare count is what made the
 * original outage survivable: "blocked=1" every 5 minutes reads as normal,
 * while a named blocker does not.
 *
 * ⚠️ `awaiting_manual` is counted SEPARATELY from `published`, and that is the
 * whole point of the 2026-09-03 change. A piece waiting on its manual channel
 * used to be re-dispatched every tick and counted as `published=1` — a run that
 * published nothing reporting success, forever, which is the original silent
 * failure with a green badge on it. It is not counted as `blocked` either: it
 * is not a failure and the operator has already been told once.
 */
export function summarisePublishAttempts(
  attempts: PublishAttempt[],
  awaitingManual: AwaitingManualPiece[] = [],
): string {
  const published = attempts.filter((a) => a.ok).length;
  const blocked = attempts.filter((a) => !a.ok);
  const named = blocked
    .map((a) => `${a.pieceId.slice(0, 8)}=${a.errorCode ?? `http_${a.httpStatus ?? 0}`}`)
    .join(", ");
  const waiting = awaitingManual
    .map((w) => `${w.pieceId.slice(0, 8)}=${w.channels.join("+")}`)
    .join(", ");
  return (
    `due=${attempts.length + awaitingManual.length} published=${published}` +
    (awaitingManual.length ? ` awaiting_manual=${awaitingManual.length} (${waiting})` : "") +
    ` blocked=${blocked.length}` +
    (named ? ` (${named})` : "")
  );
}
 
/**
 * Dedup key for a manual-post reminder: one per (piece, schedule, day).
 *
 * `scheduledAt` is in the key for the same reason it is in `publishAlertKey` —
 * rescheduling a piece is the expected operator action, and a reminder muted by
 * its own earlier self is the silent-failure bug wearing a different hat. The
 * day bucket makes it a daily reminder rather than a single shot: the failure
 * this addresses is precisely a one-shot message being missed.
 */
export function manualNudgeKey(item: AwaitingManualPiece): string {
  const day = Math.floor(item.overdueMinutes / MANUAL_NUDGE_INTERVAL_MINUTES);
  return `manual:${item.pieceId}:${item.scheduledAt ?? "unscheduled"}:day${day}`;
}
 
/** Whether a piece has waited long enough on its manual post to say so again. */
export function shouldNudgeManual(
  item: AwaitingManualPiece,
  afterMinutes: number = MANUAL_NUDGE_AFTER_MINUTES,
): boolean {
  return item.channels.length > 0 && item.overdueMinutes >= afterMinutes;
}

/** The WhatsApp/decision message for a piece still awaiting its manual post. */
export function buildManualNudgeAlert(item: AwaitingManualPiece): string {
  const theme = item.theme ? `"${item.theme}"` : item.pieceId.slice(0, 8);
  return (
    `Scheduled post ${theme} is still waiting on you (${humaniseMinutes(item.overdueMinutes)} since it was due). ` +
    `Its other channels went out; ${item.channels.join(", ")} needs a manual post. ` +
    `Open the piece in Admin → Marketing and tap "Mark as posted" — or "Skip channel" if you are not posting it. ` +
    `Until then the piece stays unpublished.`
  );
}
