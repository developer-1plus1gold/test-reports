/**
 * Curation-action vocabulary + resolution — the single source of truth for what
 * an approved guardrail curation action WRITES.
 *
 * Why this module exists (2026-08-27). The guardrail scope/kind allow-lists had
 * drifted into FOUR separate copies, three of which were wrong:
 *   • guardrail-classify.ts  — all 7 scopes (correct; it EMITS 'response'/'brief')
 *   • decisions/[id] route   — 5 scopes, missing 'response'/'brief'
 *   • guardrails POST route  — 4 scopes, missing 'caption' too (the admin UI
 *                              offers 'caption', so that selection was silently
 *                              clamped to 'all')
 *   • guardrail-curator.ts   — 5 scopes in its TypeScript union
 * A clamp is not a rejection: an unrecognised scope fell through to 'all', and
 * because getActiveGuardrails() always includes 'all', that BROADENS a rule to
 * every surface rather than narrowing it. Reply-copy and brief-spec rules are
 * created by guardrail-ingest today, and the curator's documented job is to
 * re-type them on its daily pass — so the narrow lists sat directly on the
 * intended workflow. Everything now imports from here.
 *
 * The second job of this module is `isNoOpCuration`. See its doc comment.
 */
 
import { sanitizeRuleText } from "./sanitize";
 
export const GUARDRAIL_SCOPES = [
  "all", "generate", "edit", "animate", "caption", "response", "brief",
] as const;
export type GuardrailScopeValue = (typeof GUARDRAIL_SCOPES)[number];
 
export const GUARDRAIL_KINDS = [
  "negative_exclusion", "positive_directive", "review_check",
] as const;
export type GuardrailKindValue = (typeof GUARDRAIL_KINDS)[number];
 
export const CURATION_TYPES = [
  "merge", "generalize", "retire", "rewrite", "promote",
] as const;
export type CurationActionType = (typeof CURATION_TYPES)[number];
 
/** Scope/kind a malformed value falls back to. Kept explicit because the
 *  fallback is load-bearing: 'all' is the BROADEST scope, not a safe default. */
export const DEFAULT_SCOPE: GuardrailScopeValue = "all";
export const DEFAULT_KIND: GuardrailKindValue = "negative_exclusion";
 
export function isGuardrailScope(v: unknown): v is GuardrailScopeValue {
  return typeof v === "string" && (GUARDRAIL_SCOPES as readonly string[]).includes(v);
}
export function isGuardrailKind(v: unknown): v is GuardrailKindValue {
  return typeof v === "string" && (GUARDRAIL_KINDS as readonly string[]).includes(v);
}
export function isCurationType(v: unknown): v is CurationActionType {
  return typeof v === "string" && (CURATION_TYPES as readonly string[]).includes(v);
}
 
/** The subset of a curation action that determines the written row. */
export interface CurationWriteInput {
  newRuleText?: unknown;
  newScope?:    unknown;
  newKind?:     unknown;
}
 
export interface CurationWrite {
  /** Sanitised rule text. EMPTY means the apply step writes nothing at all. */
  text:  string;
  scope: GuardrailScopeValue;
  kind:  GuardrailKindValue;
}
 
/**
 * Resolve exactly what an approved action would write, applying the same
 * sanitisation and enum clamping the apply step uses. Proposal-time and
 * apply-time MUST agree, which is why both call this rather than re-deriving.
 */
export function resolveCurationWrite(action: CurationWriteInput): CurationWrite {
  return {
    text:  sanitizeRuleText(action.newRuleText),
    scope: isGuardrailScope(action.newScope) ? action.newScope : DEFAULT_SCOPE,
    kind:  isGuardrailKind(action.newKind) ? action.newKind : DEFAULT_KIND,
  };
}
 
/** A guardrail row as the curator and the apply step see it. */
export interface CuratedRule {
  id:        string;
  scope:     string;
  kind:      string;
  rule_text: string;
  status?:   string;
}
 
/**
 * True when applying this action would leave the ruleset semantically unchanged.
 *
 * ▶ Why this is code and not a prompt rule. Between 2026-07-09 and 2026-08-26 the
 *   curator filed the SAME byte-identical `rewrite` of the SAME rule THIRTEEN
 *   consecutive times — same text, same kind, same scope — each rationale saying
 *   the rule "is correctly typed ... no change needed". Twelve of the last fifteen
 *   proposals overall were no-ops; eight were approved and applied, writing the
 *   identical row back. The model was not malfunctioning: it was asked to check
 *   for miscategorisation, it checked, and it reported the result as an action
 *   because nothing told it that "already correct" is not an action. Whether two
 *   rows are identical is arithmetic, so code decides it — the same correction
 *   made to playbook_curator's running count on 2026-08-26.
 *
 * Definitions, each tied to what the apply step actually does:
 *   • rewrite / single-target merge / single-target generalize — the written
 *     row would equal the current row (and the rule is already active, since the
 *     apply step also forces status='active', which is a real change otherwise).
 *     A single-target merge/generalize retires the row and re-inserts identical
 *     content, so it churns the id and changes nothing else.
 *   • promote — every target is already active.
 *   • retire — never a no-op; the curator only ever loads non-retired rules, so
 *     a retire always changes status.
 * A multi-target merge/generalize is a genuine consolidation and is never a no-op.
 */
/**
 * A stable signature for "the change this action would make".
 *
 * ▶ Why (2026-09-04). `isNoOpCuration` compares the proposed write against the
 *   CURRENT row, so it stops a proposal that is byte-identical to what is stored.
 *   It cannot stop the loop actually observed: the curator holds two synonymous
 *   phrasings of the same rule —
 *       "…a call-to-action that links to https://1plus1gold.com/."
 *       "…a call-to-action linking to https://1plus1gold.com/."
 *   — and alternates. Whichever is stored, the other is not identical to it, so
 *   it passes the no-op filter and costs an operator a tap. The 2026-08-27 fix
 *   therefore did not end the loop; it made it two-cycle instead of one-cycle.
 *   Measured on 2026-09-04: 17 of the agent's 28 lifetime proposals target one
 *   rule, the last 15 consecutively, and decision 7eb9e047 (09-03) is signature-
 *   identical to 9d352e5e, denied 34 days earlier.
 *
 *   Whether two sentences MEAN the same thing is not decidable here. Whether we
 *   have already rejected this exact proposal is arithmetic — so code decides it,
 *   the same reasoning that made isNoOpCuration a code check rather than a prompt
 *   rule. The caller compares this signature against previously DENIED decisions.
 *
 * Shape notes, each load-bearing:
 *   • Target ids are sorted, so the same merge proposed in a different order is
 *     recognised as the same proposal.
 *   • `retire` and `promote` write no rule text, so their signature deliberately
 *     omits the write — otherwise two identical retires would look different
 *     because one happened to carry stray text.
 *   • The write is resolved through `resolveCurationWrite`, so this agrees with
 *     `isNoOpCuration` and with the apply step by construction rather than by
 *     re-deriving sanitisation a third time.
 *
 * Returns null when the action is too malformed to have a meaningful identity
 * (unknown type, or no targets); callers treat null as "cannot match".
 */
export function curationSignature(
  action: { type?: unknown; guardrailIds?: unknown } & CurationWriteInput,
): string | null {
  if (!isCurationType(action.type)) return null;
  const ids = Array.isArray(action.guardrailIds)
    ? action.guardrailIds.filter((i): i is string => typeof i === "string").slice().sort()
    : [];
  if (ids.length === 0) return null;
 
  const writePart =
    action.type === "retire" || action.type === "promote"
      ? ""
      : (() => {
          const w = resolveCurationWrite(action);
          return [w.text, w.scope, w.kind].join("\u0000");
        })();
 
  return [action.type, ids.join(","), writePart].join("\u0001");
}
