/**
 * Scheduler heartbeat for the marketing Planner.
 *
 * Why this exists: the weekly Planner cron (`marketing-weekly-cycle`) silently
 * never executed for 6 weeks — Vercel registered the day-of-week schedule but
 * didn't run it (daily/monthly crons fire fine). We decoupled it to a daily
 * cron + in-code Sunday guard, but a future scheduler gap could recur. This
 * heartbeat runs inside the *reliable* daily ops-summary cron and alerts if the
 * Planner hasn't logged a run in over a week — a direct tripwire for any
 * scheduler regression, independent of the mechanism that broke.
 *
 * "A run" = any `agent_decisions` row with agent_name='marketing_planner'
 * (every planner invocation files one, including the empty-state branches that
 * skip inference — so it's the most reliable run signal).
 */
 
/** Alert if the planner hasn't logged a run in more than this many days. */
export const PLANNER_STALE_DAYS = 8;
 
export interface PlannerHeartbeat {
  /** True when no run, or the last run is older than the threshold. */
  stale: boolean;
  /** Whole days since the last run; null when there has never been a run. */
  daysSince: number | null;
  lastRunMs: number | null;
}
 
/**
 * Pure threshold evaluation — kept separate from the DB query so it's unit
 * testable. `lastRunMs` is the epoch-ms of the most recent planner run, or null.
 */
export function evaluatePlannerHeartbeat(
  lastRunMs: number | null,
  nowMs: number,
  thresholdDays: number = PLANNER_STALE_DAYS,
): PlannerHeartbeat {
  if (lastRunMs === null) {
    return { stale: true, daysSince: null, lastRunMs: null };
  }
  const daysSince = Math.floor((nowMs - lastRunMs) / 86_400_000);
  return { stale: daysSince > thresholdDays, daysSince, lastRunMs };
}
 
/** One-line, newline-free alert body (WhatsApp template rule). */
export function plannerHeartbeatAlert(hb: PlannerHeartbeat): string {
  if (hb.lastRunMs === null) {
    return "Marketing planner has NEVER logged a run. The weekly planner cron may not be firing — check Vercel crons or run /api/admin/agents/marketing-planner manually.";
  }
  const last = new Date(hb.lastRunMs).toISOString().slice(0, 10);
  return `Marketing planner hasn't logged a run in ${hb.daysSince} days (last: ${last}). The weekly planner cron may have stopped firing — check Vercel crons or run it manually.`;
}
