/**
 * draft-text — flatten a marketing_drafts.content jsonb into the plain text a
 * human actually reads, so the generated-vs-shipped diff can be shown to the
 * Writer as prose rather than as JSON.
 *
 * Pure + shape-tolerant on purpose: `content` is jsonb whose shape varies by
 * draft_kind, has changed over time, and is partly operator-authored. An
 * unrecognised shape degrades to a compact JSON rendering rather than throwing —
 * a caption that cannot be flattened must never break a caption generation.
 */
 
/** Hard cap so one enormous draft cannot dominate the prompt. */
const MAX_TEXT = 2_000;
 
function clamp(s: string): string {
  // Angle brackets are stripped HERE, at the single flattening point, rather
  // than left to each interpolation site to remember. Raised in security review
  // on PR #177: containment that depends on every future call site calling
  // asPromptData() is one forgotten call away from a forgeable delimiter.
  // Callers still wrap values in asPromptData() as defence in depth.
  const t = asPromptData(s).trim();
  return t.length > MAX_TEXT ? `${t.slice(0, MAX_TEXT)}…` : t;
}
 
function str(v: unknown): string | null {
  return typeof v === "string" && v.trim() ? v : null;
}
 
/**
 * Best-effort human text for a draft. Mirrors the admin UI's own preview logic
 * so what the Writer is shown to have "shipped" is what the operator saw.
 */
export function draftText(kind: string, content: unknown): string {
  if (content == null) return "";
  Iif (typeof content === "string") return clamp(content);
  if (Array.isArray(content)) return clamp(content.map((x) => draftText(kind, x)).join("\n"));
  if (typeof content !== "object") return clamp(String(content));
 
  const c = content as Record<string, unknown>;
 
  switch (kind) {
    case "caption":
    case "response":
      return clamp(str(c.text) ?? JSON.stringify(c));
    case "hashtags":
      return clamp(Array.isArray(c.tags) ? (c.tags as unknown[]).join(" ") : JSON.stringify(c));
    case "image_prompt":
      return clamp(str(c.prompt) ?? JSON.stringify(c));
    case "brief": {
      // A brief is a multi-field spec; render the fields that carry meaning, in
      // the order the admin editor shows them, skipping empties.
      const fields = [
        ["Hero concept",       c.hero_concept],
        ["Asset specs",        c.asset_specs],
        ["Copy direction",     c.copy_direction],
        ["Music direction",    c.music_direction],
        ["Interaction script", c.interaction_script],
        ["Rationale",          c.rationale],
      ] as const;
      const lines = fields
        .map(([label, v]) => { const t = str(v); return t ? `${label}: ${t.trim()}` : null; })
        .filter((x): x is string => x !== null);
      // A pasted brief carries free text rather than the structured fields.
      if (!lines.length) return clamp(str(c.text) ?? JSON.stringify(c));
      return clamp(lines.join("\n"));
    }
    default:
      return clamp(str(c.text) ?? JSON.stringify(c));
  }
}
 
/**
 * Make a string safe to interpolate inside an XML-style prompt data block.
 *
 * ⚠️ Added 2026-08-26 after a security-agent finding on PR #177 that was RIGHT,
 * though not for the reason it gave. The first cut of the delimiters claimed the
 * tags were "unforgeable by construction, because the sanitizers strip < and >".
 * That claim was FALSE: only `edit_note` passes a sanitizer. Draft `content` is
 * stored raw (`update.content = body.content` in the drafts PATCH route), so an
 * operator edit — or model output — can contain literal angle brackets and could
 * close the block.
 *
 * (The finding's stated mechanism, that JSON.stringify re-introduces \u003c
 * escapes, is not correct — JSON.stringify emits `<` literally. The conclusion
 * was right regardless, which is why it is fixed rather than declined.)
 *
 * Stripping at PROMPT-BUILD time rather than at write time is deliberate: it
 * holds for every value regardless of which path stored it, so the guarantee no
 * longer depends on remembering to sanitize at each of many write sites.
 */
export function asPromptData(s: string): string {
  return s.replace(/[<>]/g, "");
}
 
export interface DraftEditPair {
  generated: string;
  shipped:   string;
  note?:     string;
}
 
/**
 * Build the generated-vs-shipped pairs the Writer learns from.
 *
 * Rows whose two sides flatten to the same text are dropped: an edit that
 * changed only a non-text field (alt_text, char_count) teaches nothing about
 * copy, and a no-op pair would spend prompt budget asserting that nothing
 * changed. Empty-generated rows are dropped for the same reason — a pasted
 * draft has no model output to compare against.
 */
export function buildEditPairs(
  rows: ReadonlyArray<{ draft_kind: string; content: unknown; original_content: unknown; edit_note?: string | null }>,
  limit = 6,
): DraftEditPair[] {
  const out: DraftEditPair[] = [];
  for (const r of rows) {
    if (r.original_content == null) continue;
    const generated = draftText(r.draft_kind, r.original_content);
    const shipped   = draftText(r.draft_kind, r.content);
    if (!generated || !shipped || generated === shipped) continue;
    out.push({ generated, shipped, ...(r.edit_note?.trim() ? { note: r.edit_note.trim() } : {}) });
    if (out.length >= limit) break;
  }
  return out;
}
