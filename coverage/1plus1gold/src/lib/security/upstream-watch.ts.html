/**
 * upstream-watch — pure logic for watching an UPSTREAM fix we are blocked on.
 *
 * ── Why this exists ──────────────────────────────────────────────────────────
 * CSP Phase 2 (the nonce / strict-dynamic flip that would drop `'unsafe-inline'`
 * from script-src, playbook F5) is gated on vercel/next.js PR #92844, which has
 * been open since April 2026. Because the block is entirely outside our repo,
 * the item was tracked as a DATE — "re-evaluate on <date>" — and every time the
 * date arrived the answer was "still open, push the date". By 2026-08-26 that
 * had happened NINE consecutive mornings: each one costs a read, a judgement and
 * a re-write, and produces nothing.
 *
 * A date is the wrong instrument for "someone else's PR merges at a time nobody
 * can predict". This replaces it with a watch: silent while the PR is open, one
 * alert the moment it is not. Same move as `linkedin-token-monitor`, which
 * replaced "remember a calendar entry" with a cron that speaks up on its own.
 *
 * ── Deliberately evidence-based parsing ──────────────────────────────────────
 * The GitHub PR API response shape could NOT be verified live from the build
 * sandbox: its proxy blocks api.github.com for repositories outside this
 * session's scope, and anonymous git read access does not carry PR JSON. Rather
 * than trust a remembered shape — the exact mistake that made `trackOrder`
 * return null for every real SPX response for weeks — `classifyPullState` asserts
 * a terminal state only on strong positive evidence and treats everything else
 * as `unknown`, which never alerts. An unrecognised shape therefore degrades to
 * silence plus a logged cron detail, never to a false "it merged!" page.
 */
 
export type UpstreamState = "open" | "merged" | "closed_unmerged" | "unknown";
 
export interface WatchedUpstream {
  /** Stable id used in the alert dedup key. Never reuse. */
  key:    string;
  owner:  string;
  repo:   string;
  number: number;
  /** What this upstream PR blocks on our side. */
  blocks: string;
  /** What to do once it lands — the runbook, carried in the alert itself. */
  action: string;
}

function asRecord(raw: unknown): Record<string, unknown> | null {
  return raw && typeof raw === "object" && !Array.isArray(raw)
    ? (raw as Record<string, unknown>)
    : null;
}
 
/**
 * Classify a GitHub pull-request payload.
 *
 * Ordering is deliberate: MERGED is checked first and on either of two
 * independent signals, because a merge is the event we must not miss. Every
 * unrecognised payload falls through to `unknown` rather than to `open`, so a
 * shape change surfaces as "we stopped knowing" instead of as a permanent
 * silent "still open".
 */
export function classifyPullState(raw: unknown): UpstreamState {
  const pr = asRecord(raw);
  if (!pr) return "unknown";
 
  // An error envelope ({"message": "Not Found"}) has no `state` — never treat
  // it as an answer.
  if (typeof pr.state !== "string") return "unknown";
 
  const mergedFlag = pr.merged === true;
  const mergedAt   = typeof pr.merged_at === "string" && pr.merged_at.length > 0;
  if (mergedFlag || mergedAt) return "merged";
 
  const state = pr.state.toLowerCase();
  if (state === "open")   return "open";
  // Closed without a merge is also actionable: it means the upstream fix was
  // abandoned, so waiting on it is no longer a plan.
  if (state === "closed") return "closed_unmerged";
  return "unknown";
}
 
/** Only a terminal, actionable outcome alerts. `open` and `unknown` stay silent. */
export function shouldAlert(state: UpstreamState): boolean {
  return state === "merged" || state === "closed_unmerged";
}
 
/**
 * Dedup key. Includes the state so a PR that merges after being closed still
 * alerts once for the merge — and so an `open` run can never consume the key.
 */
export function alertKey(w: WatchedUpstream, state: UpstreamState): string {
  return `${w.key}:${state}`;
}
 
export function buildUpstreamAlert(w: WatchedUpstream, state: UpstreamState): string {
  const ref  = `${w.owner}/${w.repo}#${w.number}`;
  const head = state === "merged"
    ? `Upstream ${ref} has MERGED.`
    : `Upstream ${ref} was CLOSED WITHOUT MERGING.`;
  const tail = state === "merged"
    ? `Unblocked: ${w.action}`
    : `The fix we were waiting on is not coming. Re-assess: either patch it ` +
      `ourselves or formally accept the current posture instead of waiting. ` +
      `Original plan was: ${w.action}`;
  return `${head}\n\nBlocked: ${w.blocks}\n\n${tail}\n\nhttps://github.com/${w.owner}/${w.repo}/pull/${w.number}`;
}
 
/** One-line cron_runs detail, so a silent run still leaves evidence it ran. */
export function summariseStates(
  results: ReadonlyArray<{ key: string; state: UpstreamState }>,
): string {
  return results.map((r) => `${r.key}=${r.state}`).join(" ");
}
