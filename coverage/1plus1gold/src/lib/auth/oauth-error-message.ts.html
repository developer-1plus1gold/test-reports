/**
 * OAuth / auth-callback error-code → customer-facing message.
 *
 * /auth/callback appends the provider's error as `?error=<code>` when sign-in
 * fails (Apple/Google provider error, or our own exchange failure). The sign-in
 * page previously only recognised `auth_failed`, so any other code (e.g. Apple's
 * `server_error`) bounced back to sign-in with NO visible message — the reported
 * "comes back with no information". This maps known codes to friendly copy and
 * gives a safe generic fallback (with a reference) for anything else.
 *
 * Security (LOW-25): never reflect the raw param verbatim. Known codes map to
 * hard-coded strings; unknown codes are charset-guarded before being shown as a
 * reference, and React escapes the interpolation.
 */
 
export const OAUTH_ERROR_MESSAGES: Record<string, string> = {
  auth_failed:             "The sign-in link was invalid or has expired. Please try again.",
  access_denied:           "Sign-in was cancelled before it finished. Please try again.",
  server_error:            "The sign-in provider hit a temporary error. Please try again in a moment.",
  temporarily_unavailable: "The sign-in provider is temporarily unavailable. Please try again shortly.",
  invalid_request:         "Something went wrong starting sign-in. Please try again.",
  invalid_scope:           "That sign-in method returned an unexpected response. Try again or use another method.",
  unauthorized_client:     "This sign-in method isn't set up correctly right now. Please use another method or contact us.",
  invalid_client:          "This sign-in method isn't set up correctly right now. Please use another method or contact us.",
  interaction_required:    "Apple needs another confirmation. Please try signing in again.",
  login_required:          "Please sign in again to continue.",
};
 
const SAFE_ERR_CODE = /^[a-z0-9_]{1,40}$/i;
 
/** Message for an `?error=<code>` param, or null when there is no error. */
export function messageForAuthError(raw: string | null | undefined): string | null {
  if (!raw) return null;
  const code = SAFE_ERR_CODE.test(raw) ? raw : "unknown";
  return OAUTH_ERROR_MESSAGES[code]
    ?? `Sign-in didn't complete (ref: ${code}). Please try again, or use another sign-in method.`;
}
