/**
 * Every Supabase auth user, a page at a time. `auth.admin.listUsers` returns at most 1,000 per
 * call, so a single call silently drops everyone after the first 1,000 (Mukesh 2026-11-24: no
 * admin list may stop at 1,000). Throws `errorCode` on a failed page; callers that treat auth
 * metadata as optional catch it.
 */
import type { SupabaseClient, User } from "@supabase/supabase-js";
 
export const AUTH_USERS_PER_PAGE = 1000;
/** A runaway guard, not a limit we expect to meet: 50 × 1,000 users. */
const MAX_PAGES = 50;
 
export async function listAllAuthUsers(db: SupabaseClient, errorCode: string): Promise<User[]> {
  const out: User[] = [];
  for (let page = 1; page <= MAX_PAGES; page++) {
    const { data, error } = await db.auth.admin.listUsers({ page, perPage: AUTH_USERS_PER_PAGE });
    if (error) throw new Error(errorCode);
    const users = (data?.users ?? []) as User[];
    out.push(...users);
    if (users.length < AUTH_USERS_PER_PAGE) break;
  }
  return out;
}
