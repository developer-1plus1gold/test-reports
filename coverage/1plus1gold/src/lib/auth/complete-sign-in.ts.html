/**
 * Everything after a credential has been accepted: the 2FA gate, then
 * navigation.
 *
 * WHY THIS EXISTS (2026-09-01, reported by a customer on his first sign-in):
 * *"the system was hung for sometime and when I refreshed it showed that I am
 * logged in"*. His session had been created correctly — `verifyOtp` succeeded —
 * and what stalled was what came after it. Two hazards, neither of which had any
 * fallback:
 *
 *   1. `mfa.getAuthenticatorAssuranceLevel()` is an extra NETWORK call made
 *      after the credential is already accepted. If it hangs, sign-in hangs,
 *      even though the user is by then fully authenticated.
 *   2. `router.push()` is a client-side navigation that needs an RSC fetch. A
 *      slow or cold response leaves the user on the sign-in screen with a
 *      spinner and no error.
 *
 * The busy flag was only ever cleared in a `catch`, so neither case surfaced
 * anything — the spinner simply span forever.
 *
 * So the AAL check is time-boxed and fails OPEN to aal1 (the safe direction — a
 * user who genuinely needs 2FA is still gated by the SERVER on every protected
 * route, so skipping the client-side redirect cannot grant access), and a hard
 * navigation backstops the soft one.
 *
 * ▶ WHY IT LIVES HERE rather than inside the page. It was written inline in a
 * 400-line client component and shipped WITHOUT A TEST — a fix for a hang that
 * nothing asserted could not hang again. Its dependencies are injected so the
 * two failure modes it exists for (an AAL call that never returns, a navigation
 * that never lands) can actually be exercised, instead of asserting against a
 * Supabase mock larger than the behaviour under test.
 */
 
/** How long to wait for the post-credential AAL check before assuming aal1. */
export const AAL_TIMEOUT_MS = 4000;
 
/**
 * How long to wait for the soft navigation before forcing a full load.
 *
 * Deliberately LONGER than the AAL leash, and the two are sequential, not
 * concurrent: this timer is scheduled after the AAL race has already settled. A
 * `router.push` on a poor connection can legitimately take several seconds, and
 * firing a hard load over a soft navigation that was about to land just throws
 * away work.
 */
export const NAV_FALLBACK_MS = 6000;
 
export const SIGN_IN_PATH = "/auth/sign-in";
 
export interface SignInAal {
  currentLevel?: string | null;
  nextLevel?: string | null;
}
 
export interface CompleteSignInDeps {
  /** The post-credential 2FA check. May reject, or never settle. */
  getAal: () => Promise<SignInAal | null>;
  /** Soft navigation (router.push). */
  push: (to: string) => void;
  /** Hard navigation (window.location.assign) — the backstop. */
  assign: (to: string) => void;
  /** Where the browser is right now (window.location.pathname). */
  currentPath: () => string;
}
 
export interface CompleteSignInOptions {
  /** Where to land once signed in — already same-origin-validated by the caller. */
  next: string;
  /** The caller has already established multiple factors, so skip the AAL check. */
  skipAal?: boolean;
}
 
export async function completeSignIn(
  { next, skipAal = false }: CompleteSignInOptions,
  deps: CompleteSignInDeps,
): Promise<void> {
  let destination = next;
 
  if (!skipAal) {
    try {
      const aal = await Promise.race([
        deps.getAal(),
        new Promise<null>((resolve) => setTimeout(() => resolve(null), AAL_TIMEOUT_MS)),
      ]);
      if (aal?.nextLevel === "aal2" && aal?.currentLevel !== "aal2") destination = "/auth/2fa";
    } catch {
      // Treat an AAL failure as aal1 — see the note above. Failing open here is
      // safe; failing closed would strand an authenticated user on a spinner.
    }
  }
 
  deps.push(destination);
 
  // Backstop: if the soft navigation has not taken us off this page, force a
  // full load. Harmless when push already won — a hard navigation unmounts the
  // page along with the pending timer, so it cannot loop.
  //
  // Skipped when the destination IS the sign-in page (`?next=/auth/sign-in`
  // survives the caller's same-origin check): there the pathname test can never
  // clear, so the backstop would fire one pointless reload on a navigation that
  // actually succeeded.
  if (destination.split("?")[0] !== SIGN_IN_PATH) {
    setTimeout(() => {
      if (deps.currentPath() === SIGN_IN_PATH) deps.assign(destination);
    }, NAV_FALLBACK_MS);
  }
}
