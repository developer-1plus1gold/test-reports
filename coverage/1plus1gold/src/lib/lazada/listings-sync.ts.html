/** Every Lazada listing (GET /products/get, paged) — shared by the sync and the
 *  reconciliation (lib/marketplace/reconcile). Null when Lazada is not connected. */
export async function readLazadaListings(db: SupabaseClient, now: Date, http: Http = {}): Promise<{ listings: LazadaListing[]; firstRaw: unknown } | null> {
  const auth = await getAccessToken(db, now);
  if (!auth) return null;
  const listings: LazadaListing[] = [];
  let firstRaw: unknown = null;
  for (let page = 0; page < MAX_PAGES; page++) {
    const body = await lazadaCall<{ data?: { products?: unknown[] } }>("/products/get", {
      filter: "all", limit: String(PAGE), offset: String(page * PAGE),
    }, { appKey: auth.appKey, appSecret: auth.appSecret, accessToken: auth.token, method: "GET", fetchImpl: http.fetchImpl });
    const raw = Array.isArray(body.data?.products) ? body.data!.products! : [];
    if (page === 0) firstRaw = raw[0] ?? null;
    listings.push(...parseLazadaProducts(body));
    if (raw.length < PAGE) break;
  }
  return { listings, firstRaw };
}

/** Admin decision on one listing: map it to a product (confirm), or clear it. */
export async function setListingProduct(db: SupabaseClient, sellerSku: string, productId: string | null, now: Date): Promise<boolean> {
  const { data } = await db.from("marketplace_listings").update({
    product_id: productId,
    match_source: productId ? "confirmed" : "none",
    match_reason: productId ? "Confirmed in admin" : "Cleared in admin",
    matched_at: productId ? now.toISOString() : null,
    updated_at: now.toISOString(),
  }).eq("channel", CHANNEL).eq("seller_sku", sellerSku).select("seller_sku");
  return Array.isArray(data) && data.length === 1;
}

/** A listing may be removed from Lazada only when it is not linked to anything we still
 *  sell: unmapped, or mapped to a product that is no longer on our shop. Pure. */
export function canRemoveListing(productId: string | null, shopVisibleIds: ReadonlySet<string>): boolean {
  return !productId || !shopVisibleIds.has(productId);
}
 
/**
 * RemoveProduct parameters. The first live call with `seller_sku_list` (2026-11-17)
 * came back `6 E006 Unexpected internal error`; Lazada's reference gives the id form
 * `sku_id_list = ["SkuId_<item_id>_<sku_id>"]`, which we hold for every synced listing.
 * The seller-SKU form stays only for a row without ids. Pure.
 */
export function removeParams(sellerSku: string, itemId: string | null, skuId: string | null): Record<string, string> {
  return itemId && skuId
    ? { sku_id_list: JSON.stringify([`SkuId_${itemId}_${skuId}`]) }
    : { seller_sku_list: JSON.stringify([sellerSku]) };
}
