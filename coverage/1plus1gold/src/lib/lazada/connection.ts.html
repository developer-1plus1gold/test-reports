export const APP_KEY_RE = /^[0-9A-Za-z_-]{1,64}$/;
 
/** Saves the key; the secret only when a non-empty one is given (an edit never wipes it). */
export async function saveAppConfig(db: SupabaseClient, input: { appKey: string; appSecret?: string }): Promise<void> {
  if (!APP_KEY_RE.test(input.appKey)) throw new Error("App Key must be letters, digits, - or _ (max 64)");
  const row: Record<string, unknown> = { channel: CHANNEL, app_key: input.appKey, updated_at: new Date().toISOString() };
  if (input.appSecret) row.app_secret_enc = encryptSecret(input.appSecret);
  const { error } = await db.from("marketplace_app_config").upsert(row, { onConflict: "channel" });
  if (error) throw new Error("Could not save the Lazada app settings");
}

export async function completeConnect(db: SupabaseClient, code: string, now: Date, http: Http = {}): Promise<LazadaTokenSet> {
  const cfg = await loadAppConfig(db);
  if (!cfg.appKey || !cfg.appSecret) throw new Error("Lazada App Key and Secret are not set");
  const body = await lazadaCall("/auth/token/create", { code }, {
    appKey: cfg.appKey, appSecret: cfg.appSecret, gateway: "auth", fetchImpl: http.fetchImpl,
  });
  const tokens = parseTokenResponse(body, now);
  await saveTokens(db, tokens, now, true);
  return tokens;
}

type FieldState = MaskingReport["postalCode"];
 
export function classifyField(values: Array<unknown>): FieldState {
  const present = values.filter((v) => typeof v === "string" && v.trim() !== "") as string[];
  if (values.length === 0) return "unknown";
  if (present.length === 0) return "absent";
  return present.some((v) => v.includes("*")) ? "masked" : "clear";
}
