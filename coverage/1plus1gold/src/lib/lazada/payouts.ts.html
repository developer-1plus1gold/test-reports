/**
 * Lazada payout batches for reconciliation (lib/marketplace/payout-reconcile).
 *
 * The same per-charge rows the fee sync reads (`/finance/transaction/details/get`) carry the
 * payout STATEMENT each charge belongs to and whether it was PAID (`statement`, `paid_status`).
 * Grouped by statement, they are the batches: each order's revenue, fees, shipping and
 * adjustments, plus charges tied to no order. `/finance/payout/status/get` adds the statement's
 * paid flag, payout amount and date where Lazada reports them.
 *
 * Classification adds one bucket to the fee sync's (lib/lazada/finance bucketOf): REFUNDS,
 * returns, reversals, penalties and compensation are ADJUSTMENTS — counted as a fee they would
 * hide a refund inside "the marketplace's fee", which is exactly what reconciliation must flag.
 * Field names are read defensively; the dry run reports raw samples before anything is written.
 */

  return Number.isNaN(d.getTime()) ? null : d.toISOString();

export function payoutBucketOf(r: FinanceRow): PayoutBucket {
  const n = `${r.fee_name ?? ""} ${r.fee_type ?? ""}`.toLowerCase();
  if (/refund|return|reversal|penalt|compensat|adjust|claim/.test(n)) return "adjustment";
  return bucketOf(r);
}
 
/** "Paid" / "paid" / "1" / true → paid; anything else (incl. "Not paid") → pending. */
export function lazadaPaid(v: unknown): boolean {
  const s = String(v ?? "").trim().toLowerCase();
  return s === "paid" || s === "1" || s === "true" || s === "yes";
}
 
/** Pure: one order's rows within one statement → its line. Revenue counts seller vouchers (they
 *  are already off our order total); fees and shipping come out as positive charges. */
export function lazadaOrderLine(orderNo: string, rows: FinanceRow[]): PayoutOrder {
  const sum = (b: PayoutBucket) => rows.filter((r) => payoutBucketOf(r) === b).reduce((s, r) => s + money(r.amount), 0);
  const revenue = sum("revenue") + sum("voucher");
  return {
    externalOrderId: orderNo,
    revenueSgd: rows.some((r) => payoutBucketOf(r) === "revenue") ? r2(revenue) : null,
    feeSgd: r2(Math.max(0, -sum("fee"))),
    shippingSgd: r2(Math.max(0, -sum("shipping"))),
    adjustmentSgd: r2(sum("adjustment")),
    settlementSgd: r2(rows.reduce((s, r) => s + money(r.amount), 0)),
  };
}
 
const MONTHS: Record<string, string> = { jan: "01", feb: "02", mar: "03", apr: "04", may: "05", jun: "06", jul: "07", aug: "08", sep: "09", oct: "10", nov: "11", dec: "12" };
 
/**
 * The statement's END date (YYYY-MM-DD), the one key both of Lazada's answers share: a charge row
 * names its statement by period ("15 Nov 2026 - 21 Nov 2026"), the statement list by number
 * ("SGZVVA12-2026-1121"). Seen on PROD 11-24: matching the two raw strings never matched.
 */
export function statementEndDate(v: unknown): string | null {
  const s = String(v ?? "").trim();
  const num = /(\d{4})-(\d{2})(\d{2})$/.exec(s);
  if (num) return `${num[1]}-${num[2]}-${num[3]}`;
  const label = /(\d{1,2})\s+([A-Za-z]{3})[A-Za-z]*\s+(\d{4})\s*$/.exec(s.split(" - ").pop() ?? "");
  if (label && MONTHS[label[2].toLowerCase()]) return `${label[3]}-${MONTHS[label[2].toLowerCase()]}-${label[1].padStart(2, "0")}`;
  return null;
}
 
/** "1027.24 SGD" → 1027.24 (Lazada's payout carries its currency; `money` would read 0). */
export function payoutAmount(v: unknown): number | null {
  const m = /-?\d[\d,]*(?:\.\d+)?/.exec(String(v ?? ""));
  return m ? Number(m[0].replace(/,/g, "")) : null;
}
 
/** Pure: all rows → one batch per statement. `statuses` = Lazada's statement list, if read. */
export function lazadaBatches(rows: PayoutRow[], statuses: Array<Record<string, unknown>>): PayoutBatch[] {
  const byStatement = new Map<string, PayoutRow[]>();
  for (const r of rows) {
    const ref = String(r.statement ?? r.statement_number ?? "").trim();
    if (!ref) continue; // not yet in a statement → not part of any payout
    byStatement.set(ref, [...(byStatement.get(ref) ?? []), r]);
  }
  const statusOf = new Map<string, Record<string, unknown>>();
  for (const s of statuses) {
    const key = statementEndDate(s.statement_number ?? s.statement);
    if (key) statusOf.set(key, s);
  }
  const batches: PayoutBatch[] = [];
  for (const [label, list] of byStatement) {
    const byOrder = new Map<string, PayoutRow[]>();
    let other = 0;
    for (const r of list) {
      const k = String(r.order_no ?? "").trim();
      if (!k) { other += -money(r.amount); continue; }
      byOrder.set(k, [...(byOrder.get(k) ?? []), r]);
    }
    const end = statementEndDate(label);
    const st = end ? statusOf.get(end) : undefined;
    // The statement number is the stable reference (the bank line names it); the period is the label.
    const ref = st?.statement_number ? String(st.statement_number) : label;
    const paid = st ? lazadaPaid(st.paid) : list.every((r) => lazadaPaid(r.paid_status));
    const status: PayoutStatus = paid ? "paid" : "pending";
    // Lazada gives no bank date; the statement's last update is when it was settled as paid.
    const paidAtRaw = st ? (st.paid_at ?? st.updated_at ?? null) : null;
    const amount = (st ? payoutAmount(st.payout) : null) ?? list.reduce((s, r) => s + money(r.amount), 0);
    batches.push({
      channel: "lazada", ref, status,
      paidAt: paid && paidAtRaw ? sgtIso(String(paidAtRaw)) : null,
      amountSgd: r2(amount),
      periodLabel: label,
      orders: [...byOrder.entries()].map(([k, rs]) => lazadaOrderLine(k, rs)),
      otherDeductionsSgd: r2(other),
    });
  }
  return batches;
}
