export type LazadaShipState = "shipped" | "delivered" | null;
 
/** shipped / delivered from the items' statuses; anything else → null (no change). */
export function shipState(statuses: string[]): LazadaShipState {
  const s = statuses.map((x) => x.toLowerCase()).filter((x) => x && x !== "canceled" && x !== "cancelled");
  if (!s.length) return null;
  if (s.every((x) => x === "delivered" || x === "confirmed")) return "delivered";
  if (s.every((x) => x === "shipped" || x === "delivered" || x === "confirmed")) return "shipped";
  return null;
}
