/**
 * Lazada token handling — PURE (no I/O): parse the token endpoints' response and
 * decide when to refresh / when to raise a Control Tower card.
 *
 * Token lifetimes for OUR app (Seller In-house, app page read 2026-11-16):
 * access token 7 days, refresh token 30 days. The response's own `expires_in` /
 * `refresh_expires_in` (seconds) are the source of truth — the 7/30 days are only
 * what we expect, never hard-coded into the maths.
 *
 * Response shape: /auth/token/create and /auth/token/refresh on auth.lazada.com/rest
 * (endpoints verified in Lazada's Node + Python SDKs). Field names below are parsed
 * defensively: a missing token or lifetime is an ERROR, never a silent default.
 */
 
export interface LazadaTokenSet {
  accessToken: string;
  refreshToken: string;
  accessExpiresAt: Date;
  refreshExpiresAt: Date;
  /** Our seller id in SG, from country_user_info — null if Lazada omits it. */
  sellerId: string | null;
  /** The seller login (an email) — shown on the admin card. */
  account: string | null;
  country: string | null;
}
 
export class LazadaTokenResponseError extends Error {}
 
function num(v: unknown): number | null {
  const n = typeof v === "string" ? Number(v) : v;
  return typeof n === "number" && Number.isFinite(n) && n > 0 ? n : null;
}
 
export function parseTokenResponse(body: unknown, now: Date): LazadaTokenSet {
  const b = (body ?? {}) as Record<string, unknown>;
  const code = b.code == null ? "0" : String(b.code);
  if (code !== "0") {
    // Lazada's message is safe to show the admin (e.g. "invalid code"); it never echoes secrets.
    throw new LazadaTokenResponseError(`Lazada refused: ${code} ${String(b.message ?? "").slice(0, 200)}`.trim());
  }
  const accessToken = typeof b.access_token === "string" ? b.access_token : "";
  const refreshToken = typeof b.refresh_token === "string" ? b.refresh_token : "";
  const expiresIn = num(b.expires_in);
  const refreshExpiresIn = num(b.refresh_expires_in);
  if (!accessToken || !refreshToken || !expiresIn || !refreshExpiresIn) {
    throw new LazadaTokenResponseError("Lazada's token response was missing a token or its lifetime");
  }
 
  const users = Array.isArray(b.country_user_info) ? (b.country_user_info as Array<Record<string, unknown>>) : [];
  const sg = users.find((u) => String(u.country ?? "").toLowerCase() === "sg") ?? users[0];
  const sellerId = sg?.seller_id != null ? String(sg.seller_id) : null;
 
  return {
    accessToken,
    refreshToken,
    accessExpiresAt: new Date(now.getTime() + expiresIn * 1000),
    refreshExpiresAt: new Date(now.getTime() + refreshExpiresIn * 1000),
    sellerId,
    account: typeof b.account === "string" ? b.account : null,
    country: typeof b.country === "string" ? b.country : (sg?.country != null ? String(sg.country) : null),
  };
}
