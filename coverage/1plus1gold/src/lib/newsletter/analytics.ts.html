/**
 * Newsletter analytics — pure derivation, no I/O.
 *
 * THE CENTRAL PROBLEM THIS SOLVES (2026-09-01):
 * `newsletter_views` serves two masters. It was built as a READ tracker, so its
 * columns carry read-shaped defaults (`view_count DEFAULT 1`,
 * `first_viewed_at/last_viewed_at DEFAULT now()`). Later, `notification_sent_at`
 * was added to the SAME table as the fan-out's idempotency lock — and the
 * fan-out claims that lock by INSERTING a row. With those defaults, the act of
 * SENDING writes a row that reads as "viewed once, just now".
 *
 * The admin Engagement tab counted rows in that table and called them viewers,
 * so it could only ever report a 100% open rate. Measured on prod: July 2026
 * showed 101 "unique viewers" against 101 recipients; the real figure is 18.
 *
 * The honest signal is recoverable because the read path
 * (/api/account/newsletter/[year]/[month]) INCREMENTS an existing row rather
 * than inserting one. So for a row that was sent, a read is the SECOND touch:
 *
 *   sent (notification_sent_at set) + view_count === 1  → never opened
 *   sent                            + view_count >= 2   → opened; reads = count - 1
 *   not sent (notification_sent_at null)                → opened without a send
 *                                                          (organic / pre-fan-out)
 *
 * ⚠️ The discriminator is `notification_sent_at`, NEVER `view_count` alone. The
 * read path's own INSERT branch also lands on view_count = 1, and there it
 * genuinely means one read — treating "view_count === 1" as "unopened" across
 * the board would silently discard every organic reader.
 *
 * Because this is pure derivation over columns that already exist, it applies
 * RETROACTIVELY to every past issue.
 */
 
/** A `newsletter_views` row, narrowed to the fields the derivation reads. */
export interface ViewRow {
  market_brief_id:      string;
  customer_id:          string;
  view_count:           number | null;
  notification_sent_at: string | null;
}
 
/**
 * A `newsletter_reads` row — one append-only event per open.
 *
 * This is the authoritative read record from 2026-09-01 onward. Issues sent
 * before that have no events and fall back to the `view_count` derivation, so
 * every metric sourced only from events reports `null` (unknown) for them
 * rather than 0 (which would claim nobody read it).
 */
export interface ReadRow {
  market_brief_id: string;
  customer_id:     string;
  language_code:   string | null;
  read_at:         string;
}
 
/** A `newsletter_deliveries` row, narrowed likewise. */
export interface DeliveryRow {
  market_brief_id: string;
  channel:         "email" | "whatsapp";
  status:          "sent" | "failed";
}
 
export interface BriefRow {
  id:           string;
  period_year:  number;
  period_month: number;
  status:       string;
  published_at: string | null;
}
 
export interface IssueStats {
  id:           string;
  year:         number;
  month:        number;
  label:        string;          // "Jul 2026"
  publishedAt:  string | null;
  /** Customers the fan-out marked notified for this issue. */
  sent:         number;
  /** Distinct customers who actually opened it (see the module note). */
  opened:       number;
  /** Opens as a share of sends. Null when nothing was sent — NOT 0%, which
   *  would assert "nobody opened it" about an issue that never went out. */
  openRate:     number | null;
  /** Total reads, excluding the phantom read the send row would contribute. */
  reads:        number;
  /** Readers with no send row — organic, or issues that predate the fan-out. */
  organic:      number;
  /** Per-channel delivery counts. `null` (not 0) when this issue predates
   *  delivery capture: zero would claim we sent nothing on that channel. */
  email:        number | null;
  whatsapp:     number | null;
  failed:       number | null;
 
  // ── Engagement depth. Sourced from the read-event log, so all three are
  //    `null` for issues that predate it — unknown, not zero. ───────────────
  /** Readers who opened the issue more than once. */
  repeatReaders:      number | null;
  /** Median hours between the send and that reader's FIRST open. */
  medianHoursToOpen:  number | null;
  /** Language the issue was actually read in, by reader count. */
  readLangCounts:     Record<string, number> | null;
}
 
const MONTHS = ["Jan","Feb","Mar","Apr","May","Jun","Jul","Aug","Sep","Oct","Nov","Dec"];
 
/**
 * Median, not mean. One reader who opens the issue three weeks later would drag
 * a mean well past anything typical; the median answers "when does a normal
 * reader open it".
 */
function median(xs: number[]): number {
  const s = [...xs].sort((a, b) => a - b);
  const mid = s.length >> 1;
  return s.length % 2 ? s[mid] : (s[mid - 1] + s[mid]) / 2;
}
 
export function monthLabel(year: number, month: number): string {
  return `${MONTHS[Math.min(11, Math.max(0, month - 1))]} ${year}`;
}
 
/**
 * Per-issue stats, newest first. Only published issues are included — a draft
 * has no send or read history to report and would just be a row of dashes.
 */
export function deriveIssueStats(
  briefs: BriefRow[],
  views: ViewRow[],
  deliveries: DeliveryRow[],
  reads: ReadRow[] = [],
): IssueStats[] {
  const byBrief = new Map<string, ViewRow[]>();
  for (const v of views) {
    const list = byBrief.get(v.market_brief_id);
    if (list) list.push(v); else byBrief.set(v.market_brief_id, [v]);
  }
 
  const delivByBrief = new Map<string, DeliveryRow[]>();
  for (const d of deliveries) {
    const list = delivByBrief.get(d.market_brief_id);
    if (list) list.push(d); else delivByBrief.set(d.market_brief_id, [d]);
  }
 
  const readsByBrief = new Map<string, ReadRow[]>();
  for (const r of reads) {
    const list = readsByBrief.get(r.market_brief_id);
    if (list) list.push(r); else readsByBrief.set(r.market_brief_id, [r]);
  }
 
  // Send time per (brief, customer) — the baseline for time-to-first-open.
  const sentAt = new Map<string, string>();
  for (const v of views) {
    if (v.notification_sent_at) sentAt.set(`${v.market_brief_id}|${v.customer_id}`, v.notification_sent_at);
  }
 
  return briefs
    .filter((b) => b.status === "published")
    .map((b) => {
      const rows = byBrief.get(b.id) ?? [];
 
      let sent = 0, opened = 0, reads = 0, organic = 0;
      for (const r of rows) {
        // A NULL/absent count is treated as 1 — that is the column default, so
        // a row can only be missing it if something wrote it explicitly null.
        const count = Number(r.view_count ?? 1);
        if (r.notification_sent_at) {
          sent++;
          if (count > 1) { opened++; reads += count - 1; }
        } else {
          organic++; opened++; reads += Math.max(1, count);
        }
      }
 
      const dels = delivByBrief.get(b.id);
      const hasDeliveryData = dels !== undefined && dels.length > 0;
 
      const ev = readsByBrief.get(b.id) ?? [];
      const hasReadEvents = ev.length > 0;
 
      // Per-reader: how many opens, when the FIRST one was, and in what
      // language. Language is taken from the first open — a reader who later
      // switches language should not be counted under two languages for one
      // issue.
      const perReader = new Map<string, { opens: number; first: string; lang: string | null }>();
      for (const r of ev) {
        const prev = perReader.get(r.customer_id);
        if (!prev) perReader.set(r.customer_id, { opens: 1, first: r.read_at, lang: r.language_code });
        else {
          prev.opens++;
          if (r.read_at < prev.first) { prev.first = r.read_at; prev.lang = r.language_code; }
        }
      }
 
      let repeatReaders: number | null = null;
      let medianHoursToOpen: number | null = null;
      let readLangCounts: Record<string, number> | null = null;
 
      if (hasReadEvents) {
        repeatReaders = [...perReader.values()].filter((r) => r.opens > 1).length;
 
        readLangCounts = {};
        for (const r of perReader.values()) {
          const k = r.lang ?? "en";
          readLangCounts[k] = (readLangCounts[k] ?? 0) + 1;
        }
 
        // Only readers we can pair with a send have a meaningful latency; an
        // organic reader has no send to measure from and is excluded rather
        // than counted as an instant open.
        const gaps: number[] = [];
        for (const [customerId, r] of perReader) {
          const sent = sentAt.get(`${b.id}|${customerId}`);
          if (!sent) continue;
          const hrs = (Date.parse(r.first) - Date.parse(sent)) / 3_600_000;
          // A negative gap means the read predates the send row — clock skew or
          // a re-send after an organic read. Dropped rather than clamped to 0,
          // which would drag the median toward a fake "instant open".
          if (Number.isFinite(hrs) && hrs >= 0) gaps.push(hrs);
        }
        medianHoursToOpen = gaps.length > 0 ? median(gaps) : null;
      }
 
      return {
        id:          b.id,
        year:        b.period_year,
        month:       b.period_month,
        label:       monthLabel(b.period_year, b.period_month),
        publishedAt: b.published_at,
        sent,
        opened,
        openRate:    sent > 0 ? opened / sent : null,
        reads,
        organic,
        email:     hasDeliveryData ? dels.filter((d) => d.channel === "email"    && d.status === "sent").length : null,
        whatsapp:  hasDeliveryData ? dels.filter((d) => d.channel === "whatsapp" && d.status === "sent").length : null,
        failed:    hasDeliveryData ? dels.filter((d) => d.status === "failed").length : null,
        repeatReaders,
        medianHoursToOpen,
        readLangCounts,
      };
    })
    .sort((a, b) =>
      b.year - a.year ||
      b.month - a.month ||
      (b.publishedAt ?? "").localeCompare(a.publishedAt ?? ""));
}
 
export interface ReachSummary {
  /** Every customer on the books. */
  totalCustomers: number;
  /** Excluded from the newsletter as internal/demo or house-channel accounts. */
  excluded:       number;
  /** Explicitly unsubscribed. */
  unsubscribed:   number;
  /** The denominator: who a send actually goes to. */
  reachable:      number;
}
 
/**
 * The answer to "out of how many customers". The fan-out already computes this
 * set (all customers, minus internal, minus house/channel accounts, minus
 * unsubscribed) but only returns it in an HTTP response nobody reads.
 *
 * Sets are intersected rather than summed: an unsubscribed internal account
 * would otherwise be subtracted twice and under-report reach.
 */
export function deriveReach(
  allCustomerIds: string[],
  internalIds:    string[],
  houseIds:       string[],
  unsubscribedIds: string[],
): ReachSummary {
  const excludedSet = new Set([...internalIds, ...houseIds]);
  const unsubSet    = new Set(unsubscribedIds);
 
  let excluded = 0, unsubscribed = 0, reachable = 0;
  for (const id of allCustomerIds) {
    if (excludedSet.has(id)) { excluded++; continue; }
    if (unsubSet.has(id))    { unsubscribed++; continue; }
    reachable++;
  }
  return { totalCustomers: allCustomerIds.length, excluded, unsubscribed, reachable };
}

export interface ReadsByMonth {
  /** "2026-09" — the month the READ happened, not the issue's period. */
  month:   string;
  reads:   number;
  readers: number;
}
 
/**
 * Reads over time, oldest first.
 *
 * Keyed on when the read HAPPENED, which is a different question from the
 * per-issue table (that one is keyed on which issue was read). A reader
 * catching up on three back-issues in one evening shows as three reads this
 * month here, and as one read against each of three issues there — both are
 * true, and conflating them is what makes "readers by month" ambiguous.
 *
 * Sourced only from the event log, so it starts from 2026-09-01. Months with no
 * reads are omitted rather than zero-filled: the caller renders a list, and a
 * fabricated run of zero-months before the log existed would read as a
 * collapse in engagement rather than as absence of data.
 */
export function deriveReadsByMonth(reads: ReadRow[]): ReadsByMonth[] {
  const byMonth = new Map<string, Set<string>>();
  const counts  = new Map<string, number>();
  for (const r of reads) {
    // Slice the ISO date rather than constructing a Date: this must bucket by
    // the stored UTC month, and a local-timezone getMonth() would shift reads
    // near a month boundary into the wrong bucket depending on server locale.
    const month = r.read_at.slice(0, 7);
    counts.set(month, (counts.get(month) ?? 0) + 1);
    const set = byMonth.get(month);
    if (set) set.add(r.customer_id); else byMonth.set(month, new Set([r.customer_id]));
  }
  return [...counts.entries()]
    .map(([month, reads]) => ({ month, reads, readers: byMonth.get(month)!.size }))
    .sort((a, b) => a.month.localeCompare(b.month));
}
 
export interface EngagementSegments {
  /** Distinct customers who have opened at least one issue, ever. */
  everRead:      number;
  /** Sent at least one issue and opened none. The re-engagement pool. */
  neverOpened:   number;
  /** Sent 3+ issues and opened none — persistently disengaged, not just new. */
  lapsed:        number;
  /** Opened more than one distinct issue — the habitual readers. */
  habitual:      number;
}
 
/**
 * Reader segments across all issues.
 *
 * Derived from `newsletter_views` rather than the event log, deliberately: this
 * is the one place where the legacy counter is BETTER, because it carries the
 * full June-onward history and the event log starts in September. The
 * send/read discriminator is the same one used per-issue (see the module note):
 * on a sent row, a read is the second touch.
 */
export function deriveSegments(views: ViewRow[]): EngagementSegments {
  const sentCount = new Map<string, number>();
  const readIssues = new Map<string, Set<string>>();
 
  for (const v of views) {
    const count = Number(v.view_count ?? 1);
    const opened = v.notification_sent_at ? count > 1 : true;
    if (v.notification_sent_at) sentCount.set(v.customer_id, (sentCount.get(v.customer_id) ?? 0) + 1);
    if (opened) {
      const set = readIssues.get(v.customer_id);
      if (set) set.add(v.market_brief_id); else readIssues.set(v.customer_id, new Set([v.market_brief_id]));
    }
  }
 
  let neverOpened = 0, lapsed = 0;
  for (const [customerId, sent] of sentCount) {
    if (readIssues.has(customerId)) continue;
    neverOpened++;
    if (sent >= 3) lapsed++;
  }
 
  let habitual = 0;
  for (const set of readIssues.values()) if (set.size > 1) habitual++;
 
  return { everRead: readIssues.size, neverOpened, lapsed, habitual };
}
