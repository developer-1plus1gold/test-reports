/**
 * Translation cache helper. One translation per (brief, language); served
 * from cache thereafter.
 *
 * Tokenises the source markdown to protect numeric/currency/date values from
 * the translation LLM, calls the translator agent, persists the result, and
 * returns it.
 */
 
import type { SupabaseClient } from "@supabase/supabase-js";
import { tokeniseNumbers, substituteValues, type TokenValue } from "./tokenise";
import { NewsletterTranslatorAgent, type LanguageCode } from "@/lib/ai/agents/newsletter-translator";
import { EN_LABELS, type NewsletterLabels } from "./labels";
 
export interface BriefJsonForTranslation {
  market_summary: { headline: string };
  three_drivers: { title: string; explanation: string }[];
  concept_explainer: { term: string; explanation: string };
  things_worth_knowing: { topic: string; explanation: string }[];
}
 
/** Translated counterpart — same shape, target-language strings. */
export type TranslatedBriefJson = BriefJsonForTranslation;
 
export interface TranslatedNewsletter {
  briefMd:        string;     // post-substitution, ready to render
  founderNoteMd:  string | null;
  /** brief_json with all customer-visible strings translated + numbers
   *  re-substituted in viewer's currency. NULL when language=en. */
  briefJson:      TranslatedBriefJson | null;
  /** Static UI labels (section heads, stat labels, footer) translated
   *  to the target language. NULL when language=en. Renderer should
   *  call mergeLabels() to fill any gaps with the English fallback. */
  labels:         Partial<NewsletterLabels> | null;
  fromCache:      boolean;
}
 
interface TranslationRow {
  translated_brief_md:        string;
  translated_founder_note_md: string | null;
  translated_template_strings: Record<string, unknown>;
}
 
// ── brief_json flatten / unflatten helpers ──────────────────────────────
// Flatten the structured brief into a flat list of {id, text} pairs so
// each piece can be tokenised + translated independently. Stable ID
// scheme so unflatten reverses cleanly.
function flattenBrief(brief: BriefJsonForTranslation): { id: string; text: string }[] {
  const out: { id: string; text: string }[] = [];
  out.push({ id: "headline", text: brief.market_summary.headline });
  brief.three_drivers.forEach((d, i) => {
    out.push({ id: `driver_${i}_title`,       text: d.title });
    out.push({ id: `driver_${i}_explanation`, text: d.explanation });
  });
  out.push({ id: "concept_term",        text: brief.concept_explainer.term });
  out.push({ id: "concept_explanation", text: brief.concept_explainer.explanation });
  brief.things_worth_knowing.forEach((t, i) => {
    out.push({ id: `tw_${i}_topic`,       text: t.topic });
    out.push({ id: `tw_${i}_explanation`, text: t.explanation });
  });
  return out;
}
 
function unflattenBrief(
  source:    BriefJsonForTranslation,
  translated: Record<string, string>,
): TranslatedBriefJson {
  return {
    market_summary: {
      headline: translated.headline ?? source.market_summary.headline,
    },
    three_drivers: source.three_drivers.map((d, i) => ({
      title:       translated[`driver_${i}_title`]       ?? d.title,
      explanation: translated[`driver_${i}_explanation`] ?? d.explanation,
    })),
    concept_explainer: {
      term:        translated.concept_term        ?? source.concept_explainer.term,
      explanation: translated.concept_explanation ?? source.concept_explainer.explanation,
    },
    things_worth_knowing: source.things_worth_knowing.map((t, i) => ({
      topic:       translated[`tw_${i}_topic`]       ?? t.topic,
      explanation: translated[`tw_${i}_explanation`] ?? t.explanation,
    })),
  };
}
 
// ── Static UI label flatten / unflatten ────────────────────────────────
// Each label key becomes its own additionalDoc (label_<key>) in the
// translator call. The translator preserves {placeholder} markers
// verbatim — they look like the existing {{TOKEN_N}} markers it already
// knows to leave alone. Falls back to English when a key is missing.
const LABEL_DOC_PREFIX = "label_";
 
function flattenLabels(): { id: string; text: string }[] {
  return Object.entries(EN_LABELS).map(([key, text]) => ({
    id:   `${LABEL_DOC_PREFIX}${key}`,
    text,
  }));
}
 
function unflattenLabels(translated: Record<string, string>): Partial<NewsletterLabels> {
  const out: Partial<NewsletterLabels> = {};
  for (const [id, text] of Object.entries(translated)) {
    if (!id.startsWith(LABEL_DOC_PREFIX)) continue;
    const key = id.slice(LABEL_DOC_PREFIX.length) as keyof NewsletterLabels;
    if (key in EN_LABELS) {
      (out as Record<string, string>)[key] = text;
    }
  }
  return out;
}
 
/**
 * Cached UI-label translations that still match today's English.
 *
 * Labels are translated once per issue + language and cached forever, so an
 * English edit (2026-11-03: dropping "Splits"; the accessory story's new
 * placeholders) would otherwise keep serving the OLD wording, or a template
 * whose placeholders no longer exist, to every non-English reader. Each cache
 * row records the English source it translated (`uiLabelsSource`); a label
 * whose source has since changed is dropped, so mergeLabels falls back to the
 * current English for it. Rows written before the source was recorded have no
 * way to prove freshness and fall back to English entirely.
 */
export function freshCachedLabels(
  cached: Partial<NewsletterLabels> | null,
  source: Record<string, string> | null | undefined,
): Partial<NewsletterLabels> | null {
  if (!cached || !source) return null;
  const out: Partial<NewsletterLabels> = {};
  for (const [key, text] of Object.entries(cached)) {
    if (key in EN_LABELS && source[key] === (EN_LABELS as unknown as Record<string, string>)[key]) {
      (out as Record<string, string>)[key] = text as string;
    }
  }
  return Object.keys(out).length > 0 ? out : null;
}
 
export async function getOrCreateTranslation(
  supabase: SupabaseClient,
  opts: {
    marketBriefId: string;
    language:      LanguageCode;
    briefMd:       string;
    founderNoteMd: string | null;
    /** Structured brief content — drivers, did-you-know, things-worth-
     *  knowing. Translated alongside the markdown so the renderer can
     *  show non-English customers the AI commentary in their language. */
    briefJson?:    BriefJsonForTranslation;
    customerCurrency: string;
    fxRate:           Record<string, number>;
    locale?:          string;
  },
): Promise<TranslatedNewsletter> {
  const { marketBriefId, language, briefMd, founderNoteMd, briefJson, customerCurrency, fxRate, locale } = opts;
 
  if (language === "en") {
    return { briefMd, founderNoteMd, briefJson: null, labels: null, fromCache: false };
  }
 
  // 1) Cache lookup
  const { data: cached } = await supabase
    .from("newsletter_translations")
    .select("translated_brief_md, translated_founder_note_md, translated_template_strings")
    .eq("market_brief_id", marketBriefId)
    .eq("language_code", language)
    .maybeSingle();
 
  if (cached) {
    const c = cached as TranslationRow;
    const tts = c.translated_template_strings ?? {};
    const valueMap = (tts.valueMap ?? {}) as Record<string, TokenValue>;
    const translatedFields = (tts.briefFields ?? {}) as Record<string, string>;
    const cachedLabels = freshCachedLabels(
      (tts.uiLabels ?? null) as Partial<NewsletterLabels> | null,
      (tts.uiLabelsSource ?? null) as Record<string, string> | null,
    );
    const subOpts = { currency: customerCurrency, fxRate, locale };
    // Re-substitute every translated brief_json field with the viewer's
    // currency. Stored values are tokenised; substitute returns the
    // localised display string.
    const substitutedFields: Record<string, string> = {};
    for (const [k, v] of Object.entries(translatedFields)) {
      substitutedFields[k] = substituteValues(v, valueMap, subOpts);
    }
    return {
      briefMd:       substituteValues(c.translated_brief_md, valueMap, subOpts),
      founderNoteMd: c.translated_founder_note_md
        ? substituteValues(c.translated_founder_note_md, valueMap, subOpts)
        : null,
      briefJson:     briefJson ? unflattenBrief(briefJson, substitutedFields) : null,
      labels:        cachedLabels,
      fromCache:     true,
    };
  }
 
  // 2) Tokenise + translate.
  const briefTok = tokeniseNumbers(briefMd);
  const noteTok  = founderNoteMd ? tokeniseNumbers(founderNoteMd) : null;
  const flatBriefDocs = briefJson ? flattenBrief(briefJson).map(f => {
    const tok = tokeniseNumbers(f.text);
    return { id: f.id, tokenised: tok.tokenised, valueMap: tok.valueMap };
  }) : [];
 
  // Static labels — translator preserves {placeholder} markers verbatim
  // in the same way it preserves {{TOKEN_N}}. No tokenisation needed
  // (labels carry no numeric values; placeholders interpolate at render
  // time after the dictionary lookup).
  const flatLabelDocs = flattenLabels();
 
  const combinedValueMap = {
    ...briefTok.valueMap,
    ...(noteTok?.valueMap ?? {}),
    ...flatBriefDocs.reduce((acc, d) => ({ ...acc, ...d.valueMap }), {}),
  };
 
  // ── Chunked translation ────────────────────────────────────────────
  // Single-call mode put 1 brief + 1 founder note + ~13 brief subdocs +
  // ~117 labels = ~132 delimited sections in one prompt. Even Sonnet
  // truncated the response — the cached Hindi row (2026-05-06) had 7
  // of 19 block labels translated, with the tail silently dropped via
  // mergeLabels English fallback. The fix is structural, not a bigger
  // model: split the work into parallel calls each with ≤30-40
  // sections, well within any model's reliable instruction-following
  // range, then merge results.
  //
  // Layout:
  //   - Call 1 (brief): MAIN = brief md, additional = founder note +
  //     brief subdocs (~15 sections total). Returns translatedBrief.
  //   - Calls 2..K (labels): no MAIN, additional = chunk of labels
  //     (≤30 each). Run in parallel via Promise.all.
  const briefAdditionalDocs: { id: string; tokenised: string }[] = [
    ...(noteTok ? [{ id: "founder_note", tokenised: noteTok.tokenised }] : []),
    ...flatBriefDocs.map(d => ({ id: d.id, tokenised: d.tokenised })),
  ];
 
  const LABEL_CHUNK_SIZE = 30;
  const labelChunks: Array<{ id: string; tokenised: string }[]> = [];
  for (let i = 0; i < flatLabelDocs.length; i += LABEL_CHUNK_SIZE) {
    labelChunks.push(
      flatLabelDocs.slice(i, i + LABEL_CHUNK_SIZE).map(d => ({ id: d.id, tokenised: d.text })),
    );
  }
 
  const agent = new NewsletterTranslatorAgent();
 
  // Fire all calls in parallel. The brief call carries MAIN + brief
  // subdocs; each label call is labels-only.
  const briefPromise = agent.translate({
    tokenisedMarkdown: briefTok.tokenised,
    targetLanguage:    language,
    additionalDocs:    briefAdditionalDocs.length > 0 ? briefAdditionalDocs : undefined,
  });
  const labelPromises = labelChunks.map(chunk =>
    agent.translate({
      tokenisedMarkdown: "",
      targetLanguage:    language,
      additionalDocs:    chunk,
    }),
  );
 
  const [briefResult, ...labelResults] = await Promise.all([briefPromise, ...labelPromises]);
 
  // Merge: brief.translated → translatedBrief; brief.additional has
  // founder_note + brief subdocs; labelResults[*].additional all have
  // label_<key> entries. unflattenLabels picks them out across the
  // merged map.
  const mergedAdditional: Record<string, string> = { ...briefResult.additional };
  for (const r of labelResults) {
    Object.assign(mergedAdditional, r.additional);
  }
 
  const translatedBrief = briefResult.translated;
  const translatedNote  = mergedAdditional.founder_note ?? null;
  // Sort the merged additional entries: founder_note + brief_<n>_<x> →
  // briefFields; label_<key> → uiLabels. Anything else is dropped.
  const translatedBriefFields: Record<string, string> = {};
  const translatedLabels = unflattenLabels(mergedAdditional);
  for (const [id, text] of Object.entries(mergedAdditional)) {
    if (id === "founder_note") continue;
    if (id.startsWith(LABEL_DOC_PREFIX)) continue;
    translatedBriefFields[id] = text;
  }
 
  // 3) Persist. translated_template_strings now carries valueMap (for
  //    re-substitution), briefFields (per-brief-content translations),
  //    and uiLabels (per-language static labels).
  await supabase.from("newsletter_translations").insert({
    market_brief_id:             marketBriefId,
    language_code:               language,
    translated_brief_md:         translatedBrief,
    translated_founder_note_md:  translatedNote,
    translated_template_strings: {
      valueMap:    combinedValueMap,
      briefFields: translatedBriefFields,
      uiLabels:    translatedLabels,
      // The English each label was translated FROM — freshCachedLabels drops a
      // translation once its English changes.
      uiLabelsSource: Object.fromEntries(
        Object.keys(translatedLabels).map((k) => [k, (EN_LABELS as unknown as Record<string, string>)[k]]),
      ),
    },
  });
 
  const subOpts = { currency: customerCurrency, fxRate, locale };
  const substitutedFields: Record<string, string> = {};
  for (const [k, v] of Object.entries(translatedBriefFields)) {
    substitutedFields[k] = substituteValues(v, combinedValueMap, subOpts);
  }
 
  return {
    briefMd:       substituteValues(translatedBrief, combinedValueMap, subOpts),
    founderNoteMd: translatedNote
      ? substituteValues(translatedNote, combinedValueMap, subOpts)
      : null,
    briefJson:     briefJson ? unflattenBrief(briefJson, substitutedFields) : null,
    labels:        translatedLabels,
    fromCache:     false,
  };
}
