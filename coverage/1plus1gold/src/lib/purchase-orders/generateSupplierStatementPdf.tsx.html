/**
 * Supplier statement PDF — one document covering several purchase orders, with
 * every line grouped under its PO (Mukesh, 2026-10-27: "a single statement of
 * all the PO lines (under each PO) ... the 2% as a separate col ... and how did
 * we arrive at total of each line"; then "add the supplier statement button").
 *
 * Same split as the PO PDF:
 *   - `buildSupplierStatementModel()` is PURE — it turns raw rows into what the
 *     sheet prints, so the per-line arithmetic and the serial assignment are
 *     unit-testable without rendering.
 *   - `generateSupplierStatementPdf()` renders it through the shared brand chrome.
 *
 * Processing fee — two cases, because both exist in the data:
 *   - Header `processing_fee_pct > 0`: the fee sits on top of the stored unit
 *     cost, so unit price = unit × (1 + p/100).
 *   - Header pct = 0 but the fee was folded INTO the unit cost
 *     (`foldProcessingFeeIntoLines`): the operator states the folded % and the
 *     sheet splits it back out, base = unit / (1 + f/100). With f = 0 the fee
 *     column simply reads 0.00 — nothing is invented.
 *
 * Serials are the MANUFACTURER bar serials (`inventory.serial_number`), never
 * the NFC serial — the supplier knows the bar by what is stamped on it.
 */
 
import {
  Document, Page, Text, View, StyleSheet, renderToBuffer,
  BrandHeader, BrandFooter, palette, fmtSgd,
} from "@/lib/pdf/branding";
import { poProcessingFeePct } from "@/lib/purchase-orders/totals";
 
// ── Raw input ─────────────────────────────────────────────────────────────
export interface StatementRawLine {
  product_id: string | null;
  quantity_ordered: number | string;
  unit_cost_sgd: number | string | null;
  unit_cost?: number | string | null;
  spot_sgd_per_g?: number | string | null;
  sort_order?: number | null;
  product?: { label: string | null; grams: number | string | null } | null;
}
 
/** A physical bar that came in against the PO (receipt or consignment). */
export interface StatementRawUnit {
  product_id: string | null;
  serial_number: string | null;
  cost_per_g_sgd: number | string | null;
  /** Consignment only: the supplier spot the line was confirmed at. */
  supplier_spot_sgd_per_g?: number | string | null;
}
 
export interface StatementRawPo {
  header: {
    po_number: string;
    supplier_name?: string | null;
    processing_fee_pct?: number | string | null;
    spot_sgd_at_creation?: number | string | null;
  };
  lines: StatementRawLine[];
  units: StatementRawUnit[];
}
 
// ── Model ─────────────────────────────────────────────────────────────────
export interface StatementRow {
  label: string;
  grams: number;
  serials: string;
  qty: number;
  /** Null when no spot is known — the gold/premium split is then not shown. */
  spot: number | null;
  gold: number | null;
  premium: number | null;
  base: number;
  /** Fee per bar. */
  fee: number;
  unitPrice: number;
  /** The multiplier applied to base (e.g. 1.02). */
  feeFactor: number;
  lineTotal: number;
  how: string;
}
 
export interface StatementSection {
  poNumber: string;
  supplierName: string | null;
  rows: StatementRow[];
  bars: number;
  feeTotal: number;
  total: number;
}
 
export interface StatementModel {
  sections: StatementSection[];
  supplierLine: string;
  feeLabel: string;
  bars: number;
  feeTotal: number;
  grandTotal: number;
  generatedAt: string;
}
 
const r2 = (n: number) => Math.round(n * 100) / 100;
const num = (v: unknown): number | null => {
  if (v === null || v === undefined || v === "") return null;
  const n = Number(v);
  return Number.isFinite(n) ? n : null;
};
const fmtPct = (p: number) => `${Number(p.toFixed(4))}%`;
const fmtFactor = (p: number) => String(Number((1 + p / 100).toFixed(6)));
 
export function buildSupplierStatementModel(input: {
  pos: StatementRawPo[];
  /** % already folded into unit costs on POs whose header pct is 0. */
  foldedFeePct: number;
  generatedAt?: string;
}): StatementModel {
  const folded = Number.isFinite(input.foldedFeePct) && input.foldedFeePct > 0 ? input.foldedFeePct : 0;
  const pcts = new Set<number>();
 
  const sections: StatementSection[] = input.pos.map((po) => {
    const headerPct = poProcessingFeePct(po.header);
    const headerSpot = num(po.header.spot_sgd_at_creation);
    // Each unit is assigned to at most one line.
    const pool = po.units.map((u) => ({ ...u, used: false }));
    const lines = [...po.lines].sort((a, b) => (a.sort_order ?? 0) - (b.sort_order ?? 0));
 
    const rows: StatementRow[] = lines.map((l) => {
      const qty = Number(l.quantity_ordered ?? 0);
      const grams = num(l.product?.grams) ?? 0;
      const unit = num(l.unit_cost_sgd) ?? num(l.unit_cost) ?? 0;
 
      let base: number, fee: number, unitPrice: number, pct: number;
      if (headerPct > 0) {
        pct = headerPct;
        base = r2(unit);
        fee = r2(unit * headerPct / 100);
        unitPrice = r2(base + fee);
      } else {
        pct = folded;
        unitPrice = r2(unit);
        base = r2(unit / (1 + folded / 100));
        fee = r2(unitPrice - base);
      }
      pcts.add(pct);
 
      const consignSpot = pool.find((u) => u.product_id === l.product_id && num(u.supplier_spot_sgd_per_g) !== null);
      const spot = num(l.spot_sgd_per_g) ?? num(consignSpot?.supplier_spot_sgd_per_g) ?? headerSpot;
      const gold = spot !== null && grams > 0 ? r2(spot * grams) : null;
      const premium = gold !== null ? r2(base - gold) : null;
 
      // Serials: same product, closest unit cost first, serial as tie-break.
      const picked = pool
        .filter((u) => !u.used && u.product_id === l.product_id && u.serial_number)
        .sort((a, b) =>
          Math.abs((num(a.cost_per_g_sgd) ?? 0) * grams - unit) - Math.abs((num(b.cost_per_g_sgd) ?? 0) * grams - unit)
          || String(a.serial_number).localeCompare(String(b.serial_number)))
        .slice(0, Math.max(0, qty));
      picked.forEach((u) => { u.used = true; });
      const serials = picked.map((u) => String(u.serial_number)).sort().join(", ") || "—";
 
      const factor = fmtFactor(pct);
      const how = gold !== null && premium !== null
        ? `(${fmtSgd(gold)} + ${fmtSgd(premium)}) × ${factor} × ${qty}`
        : `${fmtSgd(base)} × ${factor} × ${qty}`;
 
      return {
        label: l.product?.label ?? "—",
        grams, serials, qty, spot, gold, premium, base, fee, unitPrice,
        feeFactor: 1 + pct / 100,
        lineTotal: r2(unitPrice * qty),
        how,
      };
    });
 
    return {
      poNumber: po.header.po_number,
      supplierName: po.header.supplier_name ?? null,
      rows,
      bars: rows.reduce((s, r) => s + r.qty, 0),
      feeTotal: r2(rows.reduce((s, r) => s + r.fee * r.qty, 0)),
      total: r2(rows.reduce((s, r) => s + r.lineTotal, 0)),
    };
  });
 
  const suppliers = [...new Set(sections.map((s) => s.supplierName).filter(Boolean))] as string[];
  const feeLabel = pcts.size === 1 ? `${fmtPct([...pcts][0])} FEE` : "FEE";
  return {
    sections,
    supplierLine: suppliers.length ? suppliers.join(", ") : "Supplier",
    feeLabel,
    bars: sections.reduce((s, x) => s + x.bars, 0),
    feeTotal: r2(sections.reduce((s, x) => s + x.feeTotal, 0)),
    grandTotal: r2(sections.reduce((s, x) => s + x.total, 0)),
    generatedAt: input.generatedAt ?? new Date().toISOString(),
  };
}

const s = StyleSheet.create({
  page: { paddingBottom: 90, backgroundColor: "#FFFFFF" },
  body: { paddingHorizontal: 30, paddingTop: 10 },
  note: { fontSize: 8, color: palette.muted, marginBottom: 8, lineHeight: 1.4, fontFamily: "Montserrat" },
  poHead: { fontSize: 10, fontWeight: 700, color: palette.dark, marginTop: 10, marginBottom: 4, fontFamily: "Montserrat" },
  tr: { flexDirection: "row", borderBottomWidth: 0.5, borderBottomColor: palette.rule, paddingVertical: 3 },
  th: { fontSize: 7, fontWeight: 700, color: palette.gold, fontFamily: "Montserrat" },
  td: { fontSize: 7.5, color: palette.dark, fontFamily: "Montserrat" },
  sub: { flexDirection: "row", paddingVertical: 3, backgroundColor: palette.cream },
  grand: { flexDirection: "row", paddingVertical: 5, marginTop: 10, borderTopWidth: 1, borderTopColor: palette.gold },
});
 
const gramsLabel = (g: number) => `${Number(g.toFixed(4))}g`;
 
export async function generateSupplierStatementPdf(m: StatementModel): Promise<Buffer> {
  const cols: { h: string; w: number; right?: boolean; v: (r: StatementRow) => string }[] = [
    { h: "BAR", w: 15, v: (r) => `${r.label} (${gramsLabel(r.grams)})` },
    { h: "BAR SERIAL(S)", w: 15, v: (r) => r.serials },
    { h: "QTY", w: 4, right: true, v: (r) => String(r.qty) },
    { h: "SPOT / G", w: 7, right: true, v: (r) => (r.spot !== null ? r.spot.toFixed(4) : "—") },
    { h: "GOLD VALUE", w: 8, right: true, v: (r) => (r.gold !== null ? fmtSgd(r.gold) : "—") },
    { h: "PREMIUM", w: 7, right: true, v: (r) => (r.premium !== null ? fmtSgd(r.premium) : "—") },
    { h: "BASE", w: 8, right: true, v: (r) => fmtSgd(r.base) },
    { h: m.feeLabel, w: 6, right: true, v: (r) => fmtSgd(r.fee) },
    { h: "UNIT PRICE", w: 8, right: true, v: (r) => fmtSgd(r.unitPrice) },
    { h: "HOW THE LINE TOTAL IS MADE", w: 14, v: (r) => r.how },
    { h: "LINE TOTAL", w: 8, right: true, v: (r) => fmtSgd(r.lineTotal) },
  ];
  const cellStyle = (w: number, right?: boolean) => ({ width: `${w}%`, textAlign: right ? ("right" as const) : ("left" as const), paddingRight: 3 });
  const bold = { fontWeight: 700 as const };
 
  const doc = (
    <Document title={`Supplier statement — ${m.supplierLine}`} author="1Plus1Gold">
      <Page size="A4" orientation="landscape" style={s.page}>
        <BrandHeader
          title="SUPPLIER STATEMENT"
          subtitle={`${m.supplierLine} · ${m.sections.map((x) => x.poNumber).join(", ")}`}
          generatedAt={m.generatedAt}
          logoSize={20}
        />
        <View style={s.body}>
          <Text style={s.note}>
            Each line: Gold value = spot per gram × grams. Base = gold value + supplier premium. Unit price = base × the fee
            factor (the processing fee is shown in its own column). Line total = unit price × quantity. Amounts in SGD.
            Bar serials are the manufacturer serials stamped on each bar.
          </Text>
          {m.sections.map((sec) => (
            <View key={sec.poNumber}>
              <Text style={s.poHead}>{sec.poNumber}</Text>
              <View style={s.tr} wrap={false}>
                {cols.map((c) => <Text key={c.h} style={[s.th, cellStyle(c.w, c.right)]}>{c.h}</Text>)}
              </View>
              {sec.rows.map((r, i) => (
                <View key={i} style={s.tr} wrap={false}>
                  {cols.map((c) => <Text key={c.h} style={[s.td, cellStyle(c.w, c.right)]}>{c.v(r)}</Text>)}
                </View>
              ))}
              <View style={s.sub} wrap={false}>
                <Text style={[s.td, bold, { width: "34%" }]}>{sec.poNumber} subtotal · {sec.bars} bars</Text>
                <Text style={[s.td, { width: "23%" }]} />
                <Text style={[s.td, bold, cellStyle(6, true)]}>{fmtSgd(sec.feeTotal)}</Text>
                <Text style={[s.td, { width: "29%" }]} />
                <Text style={[s.td, bold, cellStyle(8, true)]}>{fmtSgd(sec.total)}</Text>
              </View>
            </View>
          ))}
          <View style={s.grand} wrap={false}>
            <Text style={[s.td, bold, { width: "34%", fontSize: 9 }]}>
              GRAND TOTAL · {m.sections.length} PO{m.sections.length === 1 ? "" : "s"} · {m.bars} bars
            </Text>
            <Text style={[s.td, { width: "23%", fontSize: 9, textAlign: "right", paddingRight: 3 }]}>Total fee included:</Text>
            <Text style={[s.td, bold, cellStyle(6, true), { fontSize: 9 }]}>{fmtSgd(m.feeTotal)}</Text>
            <Text style={[s.td, { width: "29%" }]} />
            <Text style={[s.td, bold, cellStyle(8, true), { fontSize: 9 }]}>{fmtSgd(m.grandTotal)}</Text>
          </View>
        </View>
        <View style={{ position: "absolute", bottom: 0, left: 0, right: 0 }} fixed>
          <BrandFooter brandLine="1PLUS1GOLD — SUPPLIER STATEMENT" trailing="1plus1gold.com" />
        </View>
      </Page>
    </Document>
  );
  return renderToBuffer(doc);
}
