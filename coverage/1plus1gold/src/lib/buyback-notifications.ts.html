/**
 * Buyback WhatsApp notification helpers — thin wrappers over the shared
 * customer-returns primitives so the buyback-specific WhatsApp template SIDs
 * + variable shape stay co-located with the buyback domain.
 *
 * Used by both the Lalamove webhook (production) and the refresh-pickup
 * polling path (admin/localhost fallback).
 */
 
import type { SupabaseClient } from "@supabase/supabase-js";
import { BUYBACK_PICKED_UP, sendAdminAlert } from "@/lib/twilio";
import { getAdminPhonesCsv } from "@/lib/admin-team";
import { notifyPickedUpIfTransitioned as sharedNotifyPickedUp } from "@/lib/customer-returns";
 
/**
 * Fire the "Your gold has been picked up" WhatsApp if the lalamove_status
 * has just transitioned to "picked_up". No-op if already at that state or
 * if profile has no phone.
 */
export async function notifyPickedUpIfTransitioned(
  // eslint-disable-next-line @typescript-eslint/no-explicit-any
  supabase:   SupabaseClient<any, any, any>,
  buybackId:  string,
  oldStatus:  string | null,
  newStatus:  string,
): Promise<void> {
  await sharedNotifyPickedUp({
    supabase,
    table:       "buyback_requests",
    linesTable:  "buyback_request_lines",
    templateSid: BUYBACK_PICKED_UP,
    recordId:    buybackId,
    oldStatus,
    newStatus,
    buildVars:   ({ ref, totalGrams }) => ({
      "1": totalGrams.toFixed(2),
      "2": ref,
    }),
  });
}
 
// ── Admin alert: a customer has asked to sell to us ──────────────────────────
//
// The admin WhatsApp for a new buyback was silenced on 2026-05-08 on the belief
// that the `buyback_reviewer` Control Tower card replaced it. It never did: that
// card is filed at `approve`, and only `escalate` cards send a WhatsApp, so from
// then on no admin was told on their phone (PROD `admin_alert_dispatches` held
// no buyback alert at all on 2026-11-24). Every path a customer sells through
// calls this ONE helper: bars at home (`POST /api/buybacks`), a vaulted bar
// (`POST /api/vault/sell/[id]`) and MAG grams (`POST /api/mag/sell`).
 
const COLLECTION_LABEL: Record<string, string> = {
  lalamove:     "Lalamove pickup",
  self_dropoff: "self drop-off",
  vault:        "already in our care",
};
 
const fmtSgd = (n: number) =>
  `S$${new Intl.NumberFormat("en-SG", { minimumFractionDigits: 2, maximumFractionDigits: 2 }).format(n)}`;
 
type AlertLine = { grams: number | string; quantity?: number | string | null; quoted_price_sgd: number | string | null };
 
/**
 * What we owe on a buyback, for an admin glance: the agreed total once an admin
 * has set it (`total_payout_sgd`), else the sum of the customer's line quotes.
 * `total_payout_sgd` is NULL until approval, so reading it alone showed every
 * new request as "S$0.00" on the Control Tower card (Mukesh, 2026-11-24).
 * Null when any line has no quote (spot was unavailable at submission).
 */
export function buybackQuoteSgd(
  totalPayoutSgd: number | string | null | undefined,
  lines: Array<{ quoted_price_sgd: number | string | null }>,
): number | null {
  if (totalPayoutSgd != null) return Number(totalPayoutSgd);
  if (lines.length === 0 || lines.some((l) => l.quoted_price_sgd == null)) return null;
  return Math.round(lines.reduce((s, l) => s + Number(l.quoted_price_sgd), 0) * 100) / 100;
}
 
/** The alert's detail line. Pure, so the wording is tested directly. */
export function buildBuybackAdminAlertDetail(args: {
  requestId:    string;
  customerName: string | null;
  shippingType: string | null;
  lines:        AlertLine[];
}): string {
  const pieces = args.lines.reduce((s, l) => s + (Number(l.quantity ?? 1) || 1), 0);
  const grams  = args.lines.reduce((s, l) => s + Number(l.grams), 0);
  const quoteSgd = buybackQuoteSgd(null, args.lines);
  const quote  = quoteSgd != null ? fmtSgd(quoteSgd) : "quote to confirm";
  const how    = COLLECTION_LABEL[args.shippingType ?? ""] ?? (args.shippingType || "collection not set");
  const who    = args.customerName?.trim() || "A customer";
  return `New buyback request ${args.requestId.slice(0, 8).toUpperCase()}: ${who}, ` +
    `${pieces} piece${pieces === 1 ? "" : "s"}, ${Number(grams.toFixed(3))} g, ${quote}, ${how}. ` +
    `Review it in Admin → Buybacks.`;
}
 
/**
 * WhatsApp every admin that a buyback request was just created. Reads the
 * request, its lines and the customer's name itself, so callers pass only the
 * id. Never throws: a failed send is logged (and audited by sendAdminAlert) and
 * must never fail or slow the customer's request — call it from `after()`.
 */
export async function alertAdminsNewBuyback(
  // eslint-disable-next-line @typescript-eslint/no-explicit-any
  supabase:  SupabaseClient<any, any, any>,
  requestId: string,
): Promise<void> {
  try {
    const phones = getAdminPhonesCsv();
    if (!phones) {
      console.warn("[buyback-admin-alert] ADMIN_TEAM is empty — no admin to alert");
      return;
    }
    const { data: req, error } = await supabase
      .from("buyback_requests")
      .select("id, customer_id, shipping_type, buyback_request_lines(grams, quantity, quoted_price_sgd)")
      .eq("id", requestId)
      .single();
    if (error || !req) {
      console.error("[buyback-admin-alert] request not found:", requestId, error?.message);
      return;
    }
    const { data: profile } = await supabase
      .from("profiles")
      .select("full_name")
      .eq("id", req.customer_id)
      .maybeSingle();
    const detail = buildBuybackAdminAlertDetail({
      requestId:    req.id,
      customerName: profile?.full_name ?? null,
      shippingType: req.shipping_type,
      lines:        (req.buyback_request_lines ?? []) as AlertLine[],
    });
    await sendAdminAlert(phones, "high", "Buyback request", detail);
  } catch (err) {
    console.error("[buyback-admin-alert] failed:", requestId, err);
  }
}
