/**
 * Gold Yield projections — pure math, no I/O.
 *
 * ▶ IT NO LONGER OWNS THE ARITHMETIC. Every number below comes from
 * `projectSale` in src/lib/gold-yield/settlement.ts, the same function that
 * computes a real payout. The formula that used to live here was right, and the
 * page feeding it was not: it derived the participant's premium against TODAY's
 * spot instead of the spot their bar was bought at, which silently collapses the
 * hedged model into the frozen-dollar one. Routing both the projection and the
 * settlement through one function is what makes that class of divergence
 * impossible rather than merely unlikely.
 *
 * ▶ WHAT A PARTICIPANT GETS BACK, in the two parts they are owed:
 *
 *   earnings   — their share of the premium SPREAD between what a buyer pays
 *                and what their own bar is carried at. Nearly spot-neutral,
 *                which is the whole point of the hedge.
 *   principal  — their capital, returned as grams sell. Hedged: what they paid.
 *                Un-hedged: their bar revalued at the spot on the day it sells,
 *                which may be less.
 *
 * Buyer premium depends on where the buyer is in their own 50g cycle:
 *   - Position 1g   → ~22% (top of curve, new customer first gram)
 *   - Position 25g  → ~7%
 *   - Position 50g  → ~4% (bottom of curve, full cycle completed)
 *
 * The calculator models three tier buckets so participants can scenario-plan.
 */
 
import { getPremiumForWeight } from "./gold";
import {
  projectSale, PARTICIPANT_SHARE, summariseBarSettlement,
  type GyBarSaleRow, type GyBarSettlement,
} from "./gold-yield/settlement";
 
// ── Tier definitions (buyer's position in their own 50g cycle) ───────────────
 
export const TIER_RANGES = {
  early: { from: 1,  to: 10, label: "Early-cycle (positions 1–10g)" },
  mid:   { from: 11, to: 30, label: "Mid-cycle (positions 11–30g)" },
  late:  { from: 31, to: 50, label: "Late-cycle (positions 31–50g)" },
} as const;
 
export type TierKey = keyof typeof TIER_RANGES;
 
export interface TierDistribution {
  earlyPct: number;
  midPct:   number;
  latePct:  number;
}
 
export const SCENARIO_PRESETS: Record<"optimistic" | "realistic" | "late", TierDistribution> = {
  // Every buyer is a fresh customer (position 1–10g) — best case for participant
  optimistic: { earlyPct: 100, midPct: 0,  latePct: 0  },
  // Mixed audience — defaults biased towards new customers (most likely in practice)
  realistic:  { earlyPct: 50,  midPct: 30, latePct: 20 },
  // Worst case — most buyers are late in their cycle (low premium)
  late:       { earlyPct: 10,  midPct: 30, latePct: 60 },
};
 
// ── Helpers ──────────────────────────────────────────────────────────────────
 
/**
 * Average premium decimal across an integer-gram range, using the same
 * log-space curve the live pricing uses (getPremiumForWeight).
 */
export function avgPremiumOverRange(
  fromG: number,
  toG:   number,
  premiumMap: Map<number, number>,
): number {
  if (toG < fromG || premiumMap.size === 0) return 0;
  let sum = 0;
  let count = 0;
  for (let g = fromG; g <= toG; g++) {
    sum += getPremiumForWeight(g, premiumMap);
    count++;
  }
  return count > 0 ? sum / count : 0;
}
 
export interface PerGramProjection {
  earningsPerG:  number;
  principalPerG: number;
  investedPerG:  number;
  /** The platform's half of net profit, per gram. */
  platformPerG:  number;
  /** The participant's bar revalued at spot-at-sale — what the sale settles against. */
  basisPerG:     number;
  /** What the buyer pays, per gram. */
  salePricePerG: number;
}
 
/**
 * Per-gram projection for one buyer premium, through the settlement rule.
 *
 * Returns nulls rather than zeros when the inputs cannot be settled: a zero
 * here renders as "you earn nothing", which is a claim, where the truth is
 * "this cannot be projected yet" (spot still loading, bar missing its stamps).
 */
export function perGramProjection(params: {
  spotAtSale:         number;
  spotAtEnrol:        number;
  buyerPremium:       number;  // decimal, e.g. 0.22 for 22%
  participantPremium: number;  // decimal, e.g. 0.04 for 4% (50g bar)
  hedged:             boolean;
  sharePct?:          number;  // default 0.5 — participant's share of net
}): PerGramProjection | null {
  const r = projectSale({
    grams:           1,
    spotAtSale:      params.spotAtSale,
    spotAtEnrol:     params.spotAtEnrol,
    buyerPremiumPct: params.buyerPremium * 100,
    premiumPct:      params.participantPremium * 100,
    hedged:          params.hedged,
    sharePct:        params.sharePct ?? PARTICIPANT_SHARE,
  });
  if (!r.ok) return null;
  return {
    earningsPerG:  r.settlement.participantSharePerG,
    principalPerG: r.settlement.principalPerG,
    investedPerG:  r.settlement.investedPerG,
    // At grams = 1 the kernel's dollar totals ARE the per-gram rates. Taken
    // from it rather than re-derived here, so the projection and the payout
    // cannot disagree about the platform's half either.
    platformPerG:  r.settlement.platformShare,
    basisPerG:     r.settlement.revaluedBasisPerG,
    salePricePerG: params.spotAtSale * (1 + params.buyerPremium),
  };
}
 
// ── Scenario simulation ──────────────────────────────────────────────────────
 
export interface ScenarioInputs {
  enrolledGrams:      number;
  /** Spot their premium was struck over. For a hypothetical bar, today's spot. */
  spotAtEnrol:        number;
  /** Modelled spot when the grams sell. Equal to spotAtEnrol = no spot move. */
  spotAtSale:         number;
  participantPremium: number;  // decimal
  /** false = the participant carries the spot move on their principal. */
  hedged:             boolean;
  gramsPerWeek:       number;
  distribution:       TierDistribution;
  premiumMap:         Map<number, number>;
  sharePct?:          number;
  /**
   * How the bar earns — `resolveEarningPath` on the product it was bought as.
   *
   * ▶ IT IS NOT A PRESENTATION CHOICE. A SPLIT bar is scored and sells as 1 g
   * cards, so it is bought by many buyers at many positions on the Premium
   * Curve and the tier distribution is the whole model. A WHOLE bar is sold
   * intact, in one transaction, to ONE buyer — so a tier MIX is meaningless on
   * it, and pacing it out gram-by-gram would draw a curve for a sale that never
   * happens that way. Defaults to `"split"`, which is what every currently
   * eligible product is.
   */
  earningPath?:       "split" | "whole";
  /**
   * The premium the single buyer of a WHOLE bar pays, as a decimal. Ignored on
   * the split path.
   *
   * ▶ WHY IT IS AN INPUT RATHER THAN READ OFF THE CURVE. The Premium Curve
   * prices a buyer's position in their own 50 g accumulation cycle, which is a
   * 1 g-card concept; a whole 50 g bar is not sold at "position 1 g". What the
   * buyer actually pays is `netPaidPerGram` on their order line, i.e. our
   * retail premium for THAT bar — a number the operator sets, not one the curve
   * derives. Left to the caller so the page can default it to today's retail
   * premium for the weight and still let a participant model a different one.
   */
  wholeSalePremium?:  number;
}
 
export interface TierBreakdown {
  key:            TierKey;
  label:          string;
  pctOfGrams:     number;   // normalised 0–100
  grams:          number;
  buyerPremium:   number;   // decimal
  earningsPerG:   number;   // SGD
  totalEarnings:  number;   // SGD
}
 
export interface WeeklyPoint {
  week:                number;
  gramsSold:           number;
  cumulativeGramsSold: number;
  weeklyEarnings:      number;
  cumulativeEarnings:  number;
}
 
export interface ScenarioResult {
  totalEarnings:          number;
  weeksToFullSale:        number;
  truncated:              boolean;  // true when the sim hit MAX_WEEKS cap
  avgBuyerPremium:        number;   // decimal, weighted
  avgEarningsPerG:        number;   // SGD
  effectiveYieldPct:      number;   // earnings ÷ participant cost × 100
  participantCostSgd:     number;
  /** Capital returned across the whole sale. Hedged: equals the cost. */
  principalSgd:           number;
  /** principal + earnings — everything that comes back. */
  totalReturnedSgd:       number;
  /**
   * (returned − invested) ÷ invested × 100.
   *
   * ▶ MAY BE NEGATIVE, and that is the point of having it. `effectiveYieldPct`
   * divides CLAMPED earnings by a cost, so it structurally cannot go below
   * zero — on an un-hedged bar sold into a falling spot it reads +0.71% while
   * the participant is really −0.40%. That is the most misleading number the
   * simulator could show on the one screen where someone chooses to take spot
   * risk, so both are reported and this is the headline.
   */
  returnPct:              number;
  tiers:                  TierBreakdown[];
  weekly:                 WeeklyPoint[];
  /** Echoed back so a renderer never has to re-derive which model it is reading. */
  earningPath:            "split" | "whole";
  /**
   * The projection expressed in the EXACT shape the Gold Yield dashboard
   * renders a real settlement in.
   *
   * ▶ WHY IT IS BUILT BY `summariseBarSettlement` RATHER THAN ASSEMBLED HERE.
   * The dashboard's per-bar breakdown (src/components/gold-yield/BarCard.tsx)
   * is that function's output, and the operator asked for the simulator to
   * "show the earnings potential the same way as we show it on the user
   * dashboard". Composing the same figures a second time would put two
   * independent implementations of one presentation on two screens a
   * participant compares directly — the drift this whole module was rebuilt to
   * make impossible. So the projection is turned into synthetic SALE ROWS and
   * run through the dashboard's own summariser: one implementation, two
   * surfaces, and a change to the settlement presentation reaches both.
   */
  settlement:             GyBarSettlement;
}
 
export function simulate(inputs: ScenarioInputs): ScenarioResult {
  const { premiumMap, distribution: dist, spotAtEnrol, spotAtSale, participantPremium, enrolledGrams, hedged } = inputs;
  const sharePct    = inputs.sharePct    ?? PARTICIPANT_SHARE;
 
  // Normalise distribution so it always sums to 100
  const rawSum = dist.earlyPct + dist.midPct + dist.latePct;
  const norm   = rawSum > 0 ? 100 / rawSum : 0;
  const nEarly = (dist.earlyPct * norm) / 100;
  const nMid   = (dist.midPct   * norm) / 100;
  const nLate  = (dist.latePct  * norm) / 100;
 
  // Tier averages
  const earlyAvg = avgPremiumOverRange(TIER_RANGES.early.from, TIER_RANGES.early.to, premiumMap);
  const midAvg   = avgPremiumOverRange(TIER_RANGES.mid.from,   TIER_RANGES.mid.to,   premiumMap);
  const lateAvg  = avgPremiumOverRange(TIER_RANGES.late.from,  TIER_RANGES.late.to,  premiumMap);
 
  const mkTier = (key: TierKey, pct: number, buyerPremium: number): TierBreakdown => {
    const grams = enrolledGrams * pct;
    const proj  = perGramProjection({
      spotAtSale, spotAtEnrol, buyerPremium, participantPremium, hedged, sharePct,
    });
    const earningsPerG = proj?.earningsPerG ?? 0;
    return {
      key,
      label:         TIER_RANGES[key].label,
      pctOfGrams:    pct * 100,
      grams,
      buyerPremium,
      earningsPerG,
      totalEarnings: grams * earningsPerG,
    };
  };
 
  // ▶ THE WHOLE PATH IS ONE SALE TO ONE BUYER, so the tier mix does not apply
  // to it. Modelled as a single 100%-weight "tier" at the operator's retail
  // premium for that bar: same arithmetic, one bucket, and the weekly timeline
  // below collapses to the single transaction it really is.
  const isWhole = inputs.earningPath === "whole";
  const wholePremium = inputs.wholeSalePremium ?? 0;
 
  const tiers: TierBreakdown[] = isWhole
    ? [{ ...mkTier("early", 1, wholePremium), label: "Sold whole — one buyer" }]
    : [
        mkTier("early", nEarly, earlyAvg),
        mkTier("mid",   nMid,   midAvg),
        mkTier("late",  nLate,  lateAvg),
      ];
 
  const totalEarnings     = tiers.reduce((s, t) => s + t.totalEarnings, 0);
  const avgBuyerPremium   = isWhole
    ? wholePremium
    : earlyAvg * nEarly + midAvg * nMid + lateAvg * nLate;
  const avgEarningsPerG   = enrolledGrams > 0 ? totalEarnings / enrolledGrams : 0;
 
  // Principal does not depend on the buyer's premium — only on whether the bar
  // is hedged and where spot lands — so it is taken once rather than per tier.
  const anchor = perGramProjection({
    spotAtSale, spotAtEnrol, buyerPremium: avgBuyerPremium, participantPremium, hedged, sharePct,
  });
  const participantCostSgd = (anchor?.investedPerG ?? spotAtEnrol * (1 + participantPremium)) * enrolledGrams;
  const principalSgd       = (anchor?.principalPerG ?? 0) * enrolledGrams;
  const totalReturnedSgd   = principalSgd + totalEarnings;
  const effectiveYieldPct  = participantCostSgd > 0 ? (totalEarnings / participantCostSgd) * 100 : 0;
  const returnPct          = participantCostSgd > 0
    ? ((totalReturnedSgd - participantCostSgd) / participantCostSgd) * 100
    : 0;
 
  // Weekly timeline: uniform sale rate, earnings/g = blended average.
  // Hard-capped at 5200 weeks (~100 years). Beyond that the projection is
  // meaningless and the 100k+ element array hangs the chart + page navigation.
  const MAX_WEEKS = 5200;
  // A whole bar is one transaction, so pacing it out over weeks would draw a
  // ramp for a sale that happens in a single step. One week, all the grams.
  const rawWeeks  = isWhole
    ? (enrolledGrams > 0 ? 1 : 0)
    : inputs.gramsPerWeek > 0 ? Math.ceil(enrolledGrams / inputs.gramsPerWeek) : 0;
  const weeksToFullSale = Math.min(rawWeeks, MAX_WEEKS);
  const weekly: WeeklyPoint[] = [];
  let cumul       = 0;
  let cumulGrams  = 0;
  let gramsLeft   = enrolledGrams;
  for (let w = 1; w <= weeksToFullSale; w++) {
    const sold       = isWhole ? gramsLeft : Math.min(inputs.gramsPerWeek, gramsLeft);
    const weeklyEarn = sold * avgEarningsPerG;
    cumul           += weeklyEarn;
    cumulGrams      += sold;
    weekly.push({
      week:                w,
      gramsSold:           sold,
      cumulativeGramsSold: cumulGrams,
      weeklyEarnings:      weeklyEarn,
      cumulativeEarnings:  cumul,
    });
    gramsLeft -= sold;
  }
 
  // ── The dashboard's own settlement shape ───────────────────────────────────
  //
  // Each tier becomes one synthetic sale row, because that is exactly what a
  // tier IS: a quantity of grams sold at one buyer premium. Feeding them to
  // `summariseBarSettlement` — the function the live per-bar breakdown is built
  // from — means the projected block and the real one are the same code.
  //
  // ⚠️ `bar_cost_per_g` here is the REVALUED basis (their premium over
  // spot-at-sale), NOT what they paid. That distinction is the whole reason
  // `invested_sgd` is a separate field fed by `barPaidPerG`; conflating them
  // was a real bug on the dashboard, and passing the paid figure in both places
  // would silently reintroduce it here.
  const settlementRows: GyBarSaleRow[] = tiers
    .filter((t) => t.grams > 0)
    .map((t) => {
      const proj = perGramProjection({
        spotAtSale, spotAtEnrol, buyerPremium: t.buyerPremium,
        participantPremium, hedged, sharePct,
      });
      return {
        grams_sold:        t.grams,
        sale_price_per_g:  proj?.salePricePerG ?? null,
        bar_cost_per_g:    proj?.basisPerG ?? null,
        principal_per_g:   proj?.principalPerG ?? null,
        participant_share: (proj?.earningsPerG ?? 0) * t.grams,
        platform_share:    (proj?.platformPerG ?? 0) * t.grams,
      };
    });
 
  const settlement = summariseBarSettlement(
    settlementRows,
    anchor?.investedPerG ?? (spotAtEnrol * (1 + participantPremium)),
  );
 
  return {
    totalEarnings,
    weeksToFullSale,
    truncated: rawWeeks > MAX_WEEKS,
    earningPath: isWhole ? "whole" : "split",
    settlement,
    avgBuyerPremium,
    avgEarningsPerG,
    effectiveYieldPct,
    participantCostSgd,
    principalSgd,
    totalReturnedSgd,
    returnPct,
    tiers,
    weekly,
  };
}
