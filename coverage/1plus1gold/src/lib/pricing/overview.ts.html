/**
 * The Our Pricing overview payload (premium range, MAG bars, the live summary
 * for the offering cards and Premium-or-Spread), shared by GET
 * /api/pricing/overview AND the server-rendered pages, so a page can render its
 * numbers in the first HTML instead of fetching them after hydration
 * (2026-11-12, perf: the Premium Transparency spinner).
 *
 * One in-process cache keyed by the pricing version (2026-11-15), in-flight dedupe. See the route's header for the 2026-10-07 history.
 */
const TTL_MS = 60_000;
/** The competitor medians are a rolling 7-day window, which moves without a write. */
const WINDOW_BUCKET_MS = 60 * 60_000;
 
export type OverviewPayload = {
  answerValue: string;
  magBars: ReturnType<typeof magPricingBars>;
  hasMagBars: boolean;
  summary: Awaited<ReturnType<typeof fetchPricingSummary>> | null;
};
 
/** The payload and the key it was built for — the ETag of /api/pricing/overview. */
export type VersionedOverview = { key: string | null; payload: OverviewPayload };
 
let cache: { at: number; key: string | null; payload: OverviewPayload } | null = null;
let inflight: { key: string | null; promise: Promise<OverviewPayload> } | null = null;
 
async function buildPayload(): Promise<OverviewPayload> {
  const supabaseAdmin = createClient(process.env.NEXT_PUBLIC_SUPABASE_URL!, process.env.SUPABASE_SERVICE_KEY!);
  const magP = fetchPublicMagListings(supabaseAdmin).catch(() => []);
  const summaryP = magP.then((m) => fetchPricingSummary(supabaseAdmin, m)).catch(() => null);
  const [premiums, magListings, summary] = await Promise.all([fetchPremiums(), magP, summaryP]);
  const { answerValue } = derivePricingDisplay(premiums);
  const magBars = magPricingBars(magListings);
  return { answerValue, magBars, hasMagBars: magBars.length > 0, summary };
}
 
/**
 * The payload for the current pricing version (lib/pricing/pricing-version): rebuilt
 * as soon as a price changes, never served across a version. When the version cannot
 * be read it falls back to the old 60 s refresh, with no key (no ETag).
 * Null only when nothing has ever loaded and the database is unreachable.

/** The current payload (server pages and the iOS spec). */
export async function getPricingOverview(): Promise<OverviewPayload | null> {
  return (await getVersionedPricingOverview())?.payload ?? null;
}
