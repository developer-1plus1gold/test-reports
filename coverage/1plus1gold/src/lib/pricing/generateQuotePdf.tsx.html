/**
 * Wholesale quote PDF — a branded, downloadable quote the operator sends to a
 * buyer (especially wholesale buyers). Built on the shared @react-pdf brand
 * chrome (src/lib/pdf/branding.tsx).
 *
 * THE POINT (2026-08-01, Mukesh): the operator chooses WHICH price columns go on
 * the sheet — a buyer copy shows the sell price (retail and/or wholesale) and the
 * all-in landed premium, and hides our cost + margin; an internal copy can show
 * everything. The numbers come from the SAME `assembleQuote` the on-screen Quote
 * Builder uses (by-qty freight spread), so the PDF can never drift from the app.
 *
 * Landscape A4 — a quote line can carry up to ~10 columns.
 */

/** Which columns the operator chose to include. Product + Qty are always shown. */
export interface QuotePdfColumns {
  retail: boolean;     // retail unit + line total (buyer-safe)
  wholesale: boolean;  // wholesale unit + line total (buyer-safe)
  landed: boolean;     // our landed cost unit + line total (INTERNAL)
  premium: boolean;    // effective landed premium over spot, % (all-in)
  margin: boolean;     // gross margin %, against the shown sell tier (INTERNAL)
}
 
/** The default buyer copy: wholesale price + all-in premium, cost/margin hidden. */
export const DEFAULT_QUOTE_COLUMNS: QuotePdfColumns = {
  retail: false,
  wholesale: true,
  landed: false,
  premium: true,
  margin: false,
};
 
/** Coerce arbitrary input (query/body) into a valid column set. Every flag is a
 *  strict boolean; unknown keys are ignored. If NOTHING sell-facing is chosen we
 *  fall back to the wholesale column so the sheet is never priceless. */
export function normalizeQuoteColumns(raw: unknown): QuotePdfColumns {
  // Omitted entirely → the buyer-safe default (wholesale + premium). Only an
  // explicit object narrows to exactly the flags the operator ticked.
  if (raw == null || typeof raw !== "object") return { ...DEFAULT_QUOTE_COLUMNS };
  const r = raw as Record<string, unknown>;
  const cols: QuotePdfColumns = {
    retail: r.retail === true,
    wholesale: r.wholesale === true,
    landed: r.landed === true,
    premium: r.premium === true,
    margin: r.margin === true,
  };
  if (!cols.retail && !cols.wholesale && !cols.landed) cols.wholesale = true;
  // Margin needs a sell tier to measure against; drop it if there's none shown.
  if (cols.margin && !cols.retail && !cols.wholesale) cols.margin = false;
  return cols;
}

function QuoteDoc({ quote, columns, meta }: { quote: AssembledQuote; columns: QuotePdfColumns; meta: QuotePdfMeta }) {
  const spot = quote.spotSgdPerGram;
  const cols = buildColumns(columns, spot);
  const t = quote.totals;
 
  return (
    <Document>
      <Page size="A4" orientation="landscape" style={s.page}>
        <BrandHeader
          title="WHOLESALE QUOTE"
          subtitle={meta.customerName || meta.quoteName || null}
          generatedAt={meta.generatedAt}
          logoSize={22}
        />
        <View style={s.body}>
          {/* Meta strip */}
          <View style={s.metaRow}>
            {meta.quoteName ? (
              <View style={s.metaCell}><Text style={s.metaLbl}>QUOTE</Text><Text style={s.metaVal}>{meta.quoteName}</Text></View>
            ) : null}
            <View style={s.metaCell}><Text style={s.metaLbl}>DATE</Text><Text style={s.metaVal}>{fmtDate(meta.generatedAt)}</Text></View>
            <View style={s.metaCell}><Text style={s.metaLbl}>LINES</Text><Text style={s.metaVal}>{t.lines}</Text></View>
            <View style={s.metaCell}><Text style={s.metaLbl}>PIECES</Text><Text style={s.metaVal}>{t.pieces}</Text></View>
            <View style={s.metaCell}><Text style={s.metaLbl}>WEIGHT</Text><Text style={s.metaVal}>{t.grams.toLocaleString("en-SG")} g</Text></View>
            {spot != null ? (
              <View style={s.metaCell}><Text style={s.metaLbl}>SPOT S$/g</Text><Text style={s.metaVal}>{spot.toFixed(2)}{quote.spotStale ? " (stale)" : ""}</Text></View>
            ) : null}
          </View>
 
          {meta.note ? <Text style={s.note}>{meta.note}</Text> : null}
 
          {/* Header row */}
          <View style={s.headRow}>
            {cols.map((c) => (
              <Text key={c.key} style={{ ...s.headCell, flex: c.flex, textAlign: c.align }}>{c.header}</Text>
            ))}
          </View>
 
          {/* Line rows */}
          {quote.lines.map((l, i) => (
            <View key={`${l.productId}-${i}`} style={s.row} wrap={false}>
              {cols.map((c) => (
                <Text key={c.key} style={{ ...s.cell, flex: c.flex, textAlign: c.align, color: l.missingPrice && c.key === "product" ? palette.red : palette.dark }}>
                  {c.cell(l)}
                </Text>
              ))}
            </View>
          ))}
 
          {/* Totals row */}
          <View style={s.totRow}>
            {cols.map((c) => (
              <Text key={c.key} style={{ ...s.totCell, flex: c.flex, textAlign: c.align }}>{c.total(quote)}</Text>
            ))}
          </View>
 
          {t.wholesaleCoverage === "some" && columns.wholesale ? (
            <Text style={s.warn}>Some lines have no wholesale price (curve products) — their wholesale total is excluded.</Text>
          ) : null}
          {t.missingPrice ? (
            <Text style={s.warn}>Lines shown in red could not be priced by the selected supplier and are excluded from the totals.</Text>
          ) : null}
 
          <Text style={s.summary}>
            Freight is absorbed and spread across the quote by piece. Prices are indicative and move with the live gold spot; confirm on order.
          </Text>
        </View>
 
        <BrandFooter
          brandLine="1PLUS1GOLD — WHOLESALE QUOTE"
          notice="Indicative quote — prices move with the live gold spot and are confirmed at order."
          trailing="1plus1gold.com"
        />
      </Page>
    </Document>
  );
}
 
export async function generateQuotePdf(
  quote: AssembledQuote,
  columns: QuotePdfColumns,
  meta: QuotePdfMeta,
): Promise<Buffer> {
  return renderToBuffer(<QuoteDoc quote={quote} columns={columns} meta={meta} />);
}
