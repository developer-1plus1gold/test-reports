/** Median of a non-empty list. */
export function median(values: number[]): number {
  const s = [...values].sort((a, b) => a - b);
  const mid = Math.floor(s.length / 2);
  return s.length % 2 ? s[mid] : (s[mid - 1] + s[mid]) / 2;
}

/**
 * Premium vs spread on one purchase (percent in, percent out).
 *   buy      = spot × (1 + p)
 *   buy-back = spot × (1 + b)
 *   premium  = (buy − spot) / spot          = p           — divided by spot
 *   spread   = (buy − buy-back) / buy       = (p − b)/(1+p) — divided by the buy price
 * Spot cancels out of both, so the lesson needs no price at all.
 */
export function premiumAndSpread(premiumPct: number, buybackPct: number): { premiumPct: number; spreadPct: number } {
  const p = premiumPct / 100;
  const b = buybackPct / 100;
  return { premiumPct, spreadPct: ((p - b) / (1 + p)) * 100 };
}
