/**
 * Margin-divergence acceptances ("snoozes").
 *
 * An operator can accept a known margin divergence (e.g. a loss-leader, or a
 * razor-thin combibar margin) as a deliberate business decision for a fixed
 * window. While active, the margin_monitor downgrades that size's loss
 * escalation to a notify (recorded, no page). When the window passes, the
 * escalation resumes automatically — a stale acceptance can't be forgotten.
 *
 * Keyed by `grams` to match the monitor's per-size dedup.
 */
 
import type { SupabaseClient } from "@supabase/supabase-js";
import { createHash } from "crypto";
 
/**
 * Deterministic uuid per (prefix, grams) — the key the automation-monitor dedups
 * on for margin decisions. Shared by the margin-monitor cron AND the acceptances
 * API so accepting a divergence can immediately close the matching open decision
 * (rather than waiting for the next daily run).
 */
export function marginSizeUuid(prefix: string, grams: number): string {
  const h = createHash("sha256").update(`${prefix}:${grams}`).digest("hex");
  return `${h.slice(0, 8)}-${h.slice(8, 12)}-${h.slice(12, 16)}-${h.slice(16, 20)}-${h.slice(20, 32)}`;
}
 
/**
 * Deterministic uuid per (prefix, productKey) — the key margin decisions dedup on.
 *
 * Replaces the per-GRAMS key for the monitor itself. Weight is not unique: prod
 * carries FIVE products at 31.1035g (PAMP Fortuna CR, Maple Random Year, PAMP
 * Random Design, 2026 Maple, PAMP Fortuna), five at 1g, three at 50g. Keying
 * decisions on grams gave all of them ONE slot, so a second product at the same
 * weight overwrote the first's alert and a single product returning to green
 * closed an alert that belonged to a different product.
 *
 * Acceptances stay keyed by grams — `margin_acceptances.grams` is the schema and
 * "accept this size" is the operator's intent — so accepting a size snoozes every
 * product at that size. That is deliberate, not the collision above.
 */
export function marginProductUuid(prefix: string, productKey: string): string {
  const h = createHash("sha256").update(`${prefix}:product:${productKey}`).digest("hex");
  return `${h.slice(0, 8)}-${h.slice(8, 12)}-${h.slice(12, 16)}-${h.slice(16, 20)}-${h.slice(20, 32)}`;
}
 
export interface ActiveAcceptance {
  grams:         number;
  acceptedUntil: string;       // ISO timestamp
  reason:        string | null;
}
 
/**
 * grams → active acceptance (accepted_until in the future). Latest-expiring
 * wins per grams. Returns an empty Map on error (fail-open: the monitor keeps
 * escalating rather than silently swallowing a real loss).
 */
export async function getActiveMarginAcceptances(
  db: SupabaseClient,
): Promise<Map<number, ActiveAcceptance>> {
  const m = new Map<number, ActiveAcceptance>();
  const { data, error } = await db
    .from("margin_acceptances")
    .select("grams, accepted_until, reason")
    .gt("accepted_until", new Date().toISOString())
    .order("accepted_until", { ascending: false });
  if (error || !data) return m;
  for (const r of data as { grams: number; accepted_until: string; reason: string | null }[]) {
    const g = Number(r.grams);
    if (!m.has(g)) m.set(g, { grams: g, acceptedUntil: r.accepted_until, reason: r.reason ?? null });
  }
  return m;
}
