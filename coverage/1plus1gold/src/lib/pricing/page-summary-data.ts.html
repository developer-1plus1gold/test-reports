/**
 * I/O half of the Our Pricing summary — reads the live tables and hands plain
 * rows to the pure `buildPricingSummary`. Server-only (service-role client).
 * Every read is fail-soft: a failure blanks that one figure, never the page.
 */
import type { SupabaseClient } from "@supabase/supabase-js";
import { fetchPremiums } from "@/lib/premiums";
import { fetchShopCatalogRaw } from "@/lib/shop-catalog";
import type { MagListingPublic } from "@/lib/mag/listings";
import { buildPricingSummary, type PricingSummary } from "@/lib/pricing/page-summary";
 
/** Competitor sizes the live marker reports on (match the survey + our range). */
const MARKER_GRAMS = [1, 2.5, 5, 10, 20, 31.1035, 50, 100];
 
/** A computed competitor premium outside this band is a scrape error, not a price. */
export const COMPETITOR_PREMIUM_PLAUSIBLE = { minPct: -5, maxPct: 200 } as const;
 
export async function fetchPricingSummary(
  db: SupabaseClient,
  magListings: MagListingPublic[],
): Promise<PricingSummary> {
  const since = new Date(Date.now() - 7 * 86_400_000).toISOString().slice(0, 10);
 
  const [premiums, catalog, buyback, comp] = await Promise.all([
    fetchPremiums().catch(() => []),
    fetchShopCatalogRaw().then((r) => r.products).catch(() => []),
    db.from("products").select("buyback_premium_pct").eq("buyback_eligible", true).not("buyback_premium_pct", "is", null)
      .then((r) => r.data ?? [], () => []),
    db.from("competitor_prices")
      .select("competitor_id, weight_grams, price_sgd, competitor_price_lists!inner(spot_sgd_per_gram)")
      // Only sizes the marker reports on (largest is 100g) — keeps the 7-day
      // read small; the trans-Pacific transfer, not the query, is the cost.
      .gte("list_date", since).gt("price_sgd", 0).lte("weight_grams", 101)
      .then((r) => r.data ?? [], () => []),
  ]);
 
  const standardProducts = catalog
    .filter((p) => p.product_type === "standard" && Number(p.grams) > 0)
    .map((p) => {
      const tiers = [...(p.standard_premium_tiers ?? [])].sort((a, b) => a.min_qty - b.min_qty);
      return { grams: Number(p.grams), label: p.label, tier1Pct: tiers.length ? Number(tiers[0].premium_pct) : NaN };
    })
    .filter((p) => Number.isFinite(p.tier1Pct));
 
  const competitorRows = (comp as unknown as {
    competitor_id: string; weight_grams: number | string; price_sgd: number | string;
    competitor_price_lists: { spot_sgd_per_gram: number | string | null } | null;
  }[]).flatMap((r) => {
    const g = Number(r.weight_grams);
    const spot = Number(r.competitor_price_lists?.spot_sgd_per_gram ?? 0);
    const price = Number(r.price_sgd);
    if (!g || !spot || !price) return [];
    const marker = MARKER_GRAMS.find((m) => Math.abs(m - g) < 0.01);
    if (marker === undefined) return [];
    const premiumPct = (price / g / spot - 1) * 100;
    if (premiumPct < COMPETITOR_PREMIUM_PLAUSIBLE.minPct || premiumPct > COMPETITOR_PREMIUM_PLAUSIBLE.maxPct) return [];
    return [{ competitorId: String(r.competitor_id), grams: marker, premiumPct }];
  });
 
  return buildPricingSummary({
    premiums: premiums.map((p) => ({ g: p.g, label: p.label, premium_pct: p.premium_pct })),
    standardProducts,
    magListings,
    buybackPcts: (buyback as { buyback_premium_pct: number | string }[]).map((r) => Number(r.buyback_premium_pct)),
    competitorRows,
  });
}
