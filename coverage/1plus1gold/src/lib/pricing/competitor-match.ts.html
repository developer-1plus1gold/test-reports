/**
 * Competitor↔product matching for the Pricing Heat Map (2026-06-19).
 *
 * Deliberately simple — it must mirror how a customer actually compares us, not
 * an over-fitted catalogue join. Weight is the hard gate; the ladder picks the
 * CHEAPEST candidate within the tightest level that has any (Mukesh, 2026-08-02):
 *   • EXACT    — same normalised product name
 *   • SERIES   — our product IS a special edition → the competitor's same series
 *   • REFINERY — same refinery/brand, special editions EXCLUDED, form-aware
 *   • WEIGHT   — same weight only, special editions excluded (the market floor)
 *   • none     — no competitor at that weight
 *
 * SPECIAL EDITIONS (2026-08-02 fix): a competitor lists collectibles (Lunar,
 * Buddha, Lucky Cat, Ayat Al Kursi, …) alongside the plain investment bar at the
 * same weight, at a much higher premium. A STANDARD product must NEVER benchmark
 * against a collectible — and a collectible must match only its own series. The
 * prior FORM grade did the opposite: the plain bar's name has no "minted" keyword
 * so it graded UNKNOWN form and was dropped, leaving only the (minted-labelled)
 * special edition — so RF systematically matched our standard bars to the most
 * expensive collectible. Now special editions are detected heuristically
 * (`seriesTokens` — whatever descriptor survives after the standard vocabulary is
 * stripped; no hard-coded series list) and excluded from the standard tiers. Form
 * (minted/cast) still refines the REFINERY tier but only ever rejects the OPPOSITE
 * known form — a standard/unknown-form bar is never dropped.
 *
 * The competitor feed has only a free-text `product_name` (no structured brand),
 * so the brand + form are PARSED from the name; our side has a clean `refinery`
 * field (form parsed from refinery + label).
 *
 * Weight tolerance: a troy ounce is 31.1034768g but our catalogue stores 31.1035
 * and UOB stores 31.1034768 — literal equality misses it (and ½oz/2oz later).
 * canonicalGrams() snaps any near-troy-oz weight to the exact multiple so all oz
 * weights match regardless of float rounding. Safe because the nearest DISTINCT
 * catalogue weights differ by ≥20% at small sizes.
 */

// ── Form (minted vs cast) ────────────────────────────────────────────────────
 
export type Form = "MINTED" | "CAST" | "UNKNOWN";
 
/**
 * Parse minted/cast form from a free-text name or label. Conservative — only the
 * signals that reliably indicate form; anything ambiguous stays UNKNOWN so it
 * never over-rides a good same-brand match. Minted signals include the common
 * minted product lines (Fortuna, Kinebar) + PAMP's minted assurance marks
 * (Veriscan / CertiPAMP), so a competitor's "PAMP 50g Fortuna Veriscan" grades
 * minted even without the literal word "minted".
 */
export function parseForm(text: string): Form {
  const s = (text ?? "").toLowerCase();
  if (/\bcast\b|cast[\s-]?bar|castbar/.test(s)) return "CAST";
  if (/\bminted\b|fortuna|kinebar|certicard|certipamp|veriscan|rosa\b/.test(s)) return "MINTED";
  return "UNKNOWN";
}
 
/** Our product's form (refinery + label together — form words can sit in either). */
export function productForm(p: { refinery?: string | null; label?: string | null }): Form {
  return parseForm(`${p.refinery ?? ""} ${p.label ?? ""}`);
}

/**

 * after every STANDARD word (weight, unit, brand, form, assay mark, and the

export function seriesTokens(text: string): string[] {
  const toks = (text ?? "").toLowerCase().replace(/[^a-z0-9]+/g, " ").split(/\s+/).filter(Boolean);
  const out: string[] = [];
  for (const t of toks) {
    if (/^\d/.test(t)) continue;      // pure numbers / "50g" / "2" — never a series word
    if (t.length < 3) continue;       // drop "al", "hk", "in", initials
    if (STANDARD_VOCAB.has(t)) continue;
    out.push(t);
  }
  return Array.from(new Set(out)).sort();
}
 
/** A collectible / special edition (Lunar, Buddha, …) rather than a plain bar. */
export function isSpecialEdition(text: string): boolean {
  return seriesTokens(text).length > 0;
}
 
/** Normalise a product name for exact comparison (lowercased, alnum-collapsed). */
function normName(text: string): string {
  return (text ?? "").toLowerCase().replace(/[^a-z0-9]+/g, " ").trim().replace(/\s+/g, " ");
}

/**
 * Is this grade a LIKE-FOR-LIKE comparison (same product / series / brand), as
 * opposed to merely the cheapest thing at the same weight?
 *
 * This is the distinction that matters for any AUTOMATED decision. A WEIGHT-grade
 * benchmark is the market floor at that weight for a DIFFERENT product (a bank's
 * house-brand bar, another refinery's bar) — legitimate context to show an
 * operator, but never a target to price against. See `docs/pricing-benchmark-redesign.md`.
 */
export function isComparableGrade(g: Grade): boolean {
  return g !== "WEIGHT";
}

/**
 * Best match for OUR product among ONE competitor's candidates. Weight is the
 * hard gate (oz-tolerant). Ladder — the CHEAPEST candidate within the tightest
 * level that has any (Mukesh, 2026-08-02):
 *   1. EXACT    — same normalised product name.
 *   2. SERIES   — only when OUR product is itself a special/collectible edition:
 *                 a competitor's SAME series (shared descriptor token, same brand
 *                 when ours is known).
 *   3. REFINERY — same brand, SPECIAL EDITIONS EXCLUDED, form-aware (never drop a
 *                 standard/unknown-form bar; only reject the opposite known form).
 *   4. WEIGHT   — cheapest at the weight, special editions excluded.
 *
 * The exclusion of special editions is the 2026-08-02 fix: a standard bar was
 * being benchmarked against a pricier collectible because the collectible was the
 * only same-*minted-form* candidate (the plain bar's name carries no "minted"
 * keyword, so it graded UNKNOWN form and was dropped). A standard product now
 * never matches a special edition; a special edition matches only its own series.
 * If a weight has ONLY special editions, we fall back to them rather than return
 * nothing. Returns null when the competitor lists nothing at this weight.
 */
export function pickBestMatch(

/**
 * Plausibility band for a competitor's retail price, as a multiple of live spot.
 *
 * Retail gold is never sold below spot, and a legitimate premium tops out well
 * under 2× even on small collectibles. Measured against the real PROD feed
 * (2026-08-16, 91 rows across four competitors) the observed range is
 * **1.01× – 1.79×**, so this band has wide headroom in both directions and has
 * never rejected a genuine row.
 *
 * The realistic threat here is not an attacker — `competitor_prices` is
 * service-role-only (RLS `USING(false)`) — it is a **broken scrape**. A parser
 * regression that reads the wrong column or a volume-tier price (see the
 * 2026-08-02 BullionStar qty-tier bug) can silently emit a far-too-low per-gram
 * figure, which `resolveCompetitorBenchmark` would then treat as the cheapest
 * like-for-like match and hand to the pricing agents as a target.
 */
export const COMPETITOR_MIN_SPOT_MULTIPLE = 0.9;
export const COMPETITOR_MAX_SPOT_MULTIPLE = 3.0;
 
/**
 * Absolute fallback band (SGD per gram), used when live spot is unavailable so the
 * guard degrades instead of switching off. Deliberately very wide: gold traded
 * around S$50/g in 2015 and about S$180/g in 2026, so these bounds cannot plausibly
 * reject a real quote in either direction for the foreseeable future — they exist
 * only to stop an obviously corrupt row (S$0.50/g, S$50,000/g) being used as a
 * benchmark by ANY caller, including a future one that lacks the crons'
 * spot-unavailable bail-out.
 */
export const COMPETITOR_ABS_MIN_SGD_PER_G = 10;
export const COMPETITOR_ABS_MAX_SGD_PER_G = 2000;
 
/** Is this per-gram price a believable retail offer at the given spot? */
export function isPlausibleCompetitorPrice(perG: number, spotSgdPerGram: number | null | undefined): boolean {
  if (!Number.isFinite(perG) || perG <= 0) return false;
  const haveSpot = spotSgdPerGram != null && Number.isFinite(spotSgdPerGram) && spotSgdPerGram > 0;
  if (!haveSpot) {
    // No spot to judge against — fall back to the absolute sanity band rather
    // than accepting everything (security_agent_pr, 2026-08-16).
    return perG >= COMPETITOR_ABS_MIN_SGD_PER_G && perG <= COMPETITOR_ABS_MAX_SGD_PER_G;
  }
  return perG >= spotSgdPerGram! * COMPETITOR_MIN_SPOT_MULTIPLE
      && perG <= spotSgdPerGram! * COMPETITOR_MAX_SPOT_MULTIPLE;
}
 
export interface CompetitorBenchmark {
  perG: number;
  grade: Grade;
  /** True for EXACT/SERIES/REFINERY — see `isComparableGrade`. */
  comparable: boolean;
  competitorName: string;
  productName: string;
}
 
/**
 * THE benchmark for one of our products across ALL competitors — the single
 * source of truth for "what is the competing price", used by the Heat Map AND by
 * `strategy-data` (which feeds both pricing agents).
 *
 * Rule (2026-08-16): run `pickBestMatch` per competitor, keep the **TIGHTEST
 * grade any competitor achieved**, then the **cheapest within that grade**.
 *
 * The tightest-grade-first step is load-bearing and is the whole point of this
 * function. A plain min-across-competitors lets a WEIGHT-grade row from one
 * competitor beat a REFINERY-grade row from another — which is exactly the
 * cross-brand comparison the E/S/R/W ladder exists to prevent, and exactly the
 * bug that produced the 2026-08-16 false "lower premium" proposals (our 10g PAMP
 * was benchmarked against a Perth Mint bar at S$188.84/g while the like-for-like
 * PAMP sat at S$191.80/g — we were already the cheapest PAMP).
 *
 * Returns null when no competitor lists anything at this weight.
 * Full rationale: `docs/pricing-benchmark-redesign.md`.
 */
export function resolveCompetitorBenchmark(
  our: { refinery?: string | null; label?: string | null; grams: number },
  byCompetitor: CompetitorCandidates[],
  opts?: { spotSgdPerGram?: number | null },
): CompetitorBenchmark | null {
  // Drop implausible rows BEFORE matching. Because the ladder picks the cheapest
  // candidate within the tightest grade, ONE bad row at REFINERY grade is enough
  // to drag the benchmark down on its own — so the guard has to run here, not as
  // a post-hoc sanity check on the winner.
  //
  // The guard runs ALWAYS, not only when spot is supplied: without spot it falls
  // back to the absolute sanity band, so no caller can silently opt out of it by
  // omitting spot. (The Heat Map's own per-cell display is a separate path that
  // deliberately still shows raw rows, so an operator can SEE a bad scrape rather
  // than have it hidden — only this benchmark, which feeds automation, is filtered.)
  const spot = opts?.spotSgdPerGram ?? null;
  const matched: Array<{ m: MatchResult; competitorName: string }> = [];
  for (const c of byCompetitor) {
    const candidates = c.candidates.filter((x) => isPlausibleCompetitorPrice(x.perG, spot));
    const m = pickBestMatch(our, candidates);
    if (m) matched.push({ m, competitorName: c.competitorName });
  }
  if (matched.length === 0) return null;
 
  const tightest = Math.min(...matched.map((x) => GRADE_RANK[x.m.grade]));
  const best = matched
    .filter((x) => GRADE_RANK[x.m.grade] === tightest)
    .sort((a, b) => a.m.candidate.perG - b.m.candidate.perG)[0];
 
  return {
    perG: best.m.candidate.perG,
    grade: best.m.grade,
    comparable: isComparableGrade(best.m.grade),
    competitorName: best.competitorName,
    productName: best.m.candidate.productName,
