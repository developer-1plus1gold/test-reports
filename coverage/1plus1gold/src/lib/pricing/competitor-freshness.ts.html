/**
 * Competitor price freshness — one rule, used everywhere a competitor price feeds
 * a number someone acts on.
 *
 * ▶ WHY (2026-11-04). The pricing strategist benchmarked against each competitor's
 * LATEST list with no age limit. BullionPoint's prices are entered by hand (its site
 * sits behind a Cloudflare bot challenge on every path, robots.txt included, so it
 * cannot be fetched) and the last entry was 12 Oct. Twenty-three days later those
 * prices were still the cheapest like-for-like PAMP benchmark at 5g (7.76% vs 9.36%
 * live) and 10g, so the agent was steering toward a premium nobody in the market was
 * charging any more. The public Our Pricing page already used a 7-day window; this
 * makes that the shared rule, and adds a reminder so a manual source is re-entered
 * before it drops out.
 *
 * The heat map deliberately keeps showing older lists — with their date — because an
 * operator can judge age; an agent cannot.
 */
import { createHash } from "crypto";
import type { MonitorMatch } from "@/lib/control-tower/automation-monitor";
import { isProd } from "@/lib/env";
 
/** A competitor list older than this is context, never a benchmark. */
export const COMPETITOR_LIST_MAX_AGE_DAYS = 7;
 
const DAY_MS = 86_400_000;
 
/** Earliest `list_date` (YYYY-MM-DD, inclusive) still fresh enough to benchmark on. */
export function competitorListCutoff(now: Date = new Date()): string {
  return new Date(now.getTime() - COMPETITOR_LIST_MAX_AGE_DAYS * DAY_MS).toISOString().slice(0, 10);
}
 
export interface ManualCompetitor {
  id: string;
  name: string;
  slug: string;
  website_url?: string | null;
  /** Latest `competitor_price_lists.list_date`, or null when nothing was ever entered. */
  latestListDate: string | null;
}
 
function reminderUuid(competitorId: string): string {
  const h = createHash("sha256").update(`competitor_prices_stale:${competitorId}`).digest("hex");
  return `${h.slice(0, 8)}-${h.slice(8, 12)}-${h.slice(12, 16)}-${h.slice(16, 20)}-${h.slice(20, 32)}`;
}
 
/**
 * Stale-price reminders for hand-entered competitors are raised on PROD only.
 *
 * ▶ WHY (Mukesh, 2026-11-08). Hand-entered prices (BullionPoint) are only ever typed
 * into PROD, so DEV's list is always empty and the DEV cron opened a "never entered"
 * card every day that had to be denied every morning. A reminder nobody can act on
 * is noise. Off PROD the monitor still runs with no matches, so any card already
 * open there is closed instead of left behind.
 */
export function manualCompetitorRemindersEnabled(prod: boolean = isProd()): boolean {
  return prod;
}
 
/**
 * One reminder per hand-entered competitor whose latest list has aged out of the
 * benchmark window. The id is stable per competitor, so `runMonitor` keeps ONE open
 * card while the prices are stale and closes it on the first tick after a fresh
 * list is entered — which makes the cadence weekly without a weekly schedule.
 */
export function staleManualCompetitorReminders(competitors: ManualCompetitor[], now: Date = new Date()): MonitorMatch[] {
  const cutoff = competitorListCutoff(now);
  const out: MonitorMatch[] = [];
  for (const c of competitors) {
    if (c.latestListDate && c.latestListDate >= cutoff) continue;
    const age = c.latestListDate
      ? `last entered ${c.latestListDate} (${Math.floor((now.getTime() - Date.parse(`${c.latestListDate}T00:00:00Z`)) / DAY_MS)} days ago)`
      : "never entered";
    const site = c.website_url ? ` from ${c.website_url}` : "";
    out.push({
      id: reminderUuid(c.id),
      summary: `${c.name} prices ${age}, so they are no longer used as a pricing benchmark. Enter current prices${site} on Competitors → ${c.name}.`,
      payload: { competitor: c.slug, latestListDate: c.latestListDate, maxAgeDays: COMPETITOR_LIST_MAX_AGE_DAYS, page: "/admin/pricing/competitors" },
    });
  }
  return out;
}
