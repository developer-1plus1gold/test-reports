/**
 * Promo code utilities — generation and validation helpers.
 *
 * LOW-5 (security audit 2026-05-03): generation uses `crypto.randomInt`
 * (CSPRNG) instead of `Math.random`. Promo codes are admin-issued but
 * customer-redeemable; predictable codes would let an attacker enumerate
 * issued codes and burn them ahead of legitimate recipients.
 */
 
import { randomInt } from "crypto";
 
// Unambiguous uppercase chars (no 0/O, 1/I/L)
const SAFE_CHARS = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
 
function randomChars(len: number): string {
  return Array.from({ length: len }, () =>
    SAFE_CHARS[randomInt(0, SAFE_CHARS.length)]
  ).join("");
}
 
/**
 * Generate a single promo code with "1+1" embedded at a random position.
 * Output examples: 1+1KGF7BX · KG1+1F7BX · KGFBX1+1
 */
export function generatePromoCode(): string {
  const pos = randomInt(0, 3); // 0=start, 1=middle, 2=end
  if (pos === 0) {
    return `1+1${randomChars(6)}`;
  } else if (pos === 2) {
    return `${randomChars(6)}1+1`;
  } else {
    const before = 2 + randomInt(0, 3); // 2–4 chars
    const after  = 6 - before;
    return `${randomChars(before)}1+1${randomChars(after)}`;
  }
}
 
/**
 * Generate N unique promo codes.
 */
export function generatePromoCodes(n: number): string[] {
  const codes = new Set<string>();
  while (codes.size < n) {
    codes.add(generatePromoCode());
  }
  return [...codes];
}
 
/** Normalize code to uppercase for storage and comparison */
export const normalizeCode = (code: string) => code.toUpperCase().trim();
 
// ── Custom / campaign codes ──────────────────────────────────────────────────
// The `code` column is VARCHAR(20). Codes are entered by customers, so we keep
// a tight, unambiguous alphabet + the brand "+" token. Two shapes:
//   • a fully custom single code (e.g. "DIWALI", "CNY2026") for shared campaigns
//   • an optional PREFIX for anonymous batches → "DIWALI-A2B3", "DIWALI-C4D5"
 
export const PROMO_CODE_MIN = 3;
export const PROMO_CODE_MAX = 20;
export const PROMO_PREFIX_MAX = 10;
 
/** Uppercase + strip to the allowed code alphabet (A–Z 0–9 + -). */
export function cleanPromoCode(input: string): string {
  return (input ?? "").toUpperCase().trim().replace(/[^A-Z0-9+-]/g, "");
}
 
/** True when a cleaned custom code is a legal length. */
export function isValidCustomCode(cleaned: string): boolean {
  return cleaned.length >= PROMO_CODE_MIN && cleaned.length <= PROMO_CODE_MAX;
}
 
/** Uppercase + strip a prefix to A–Z 0–9 (no + / - so the "-" separator stays
 *  unambiguous), capped so prefix + "-" + 4 random ≤ 20. */
export function cleanPromoPrefix(input: string): string {
  return (input ?? "").toUpperCase().trim().replace(/[^A-Z0-9]/g, "").slice(0, PROMO_PREFIX_MAX);
}
 
/** Generate N unique "<PREFIX>-<4 CSPRNG chars>" codes (e.g. DIWALI-A2B3). */
export function generatePrefixedCodes(prefix: string, n: number): string[] {
  const codes = new Set<string>();
  let guard = 0;
  while (codes.size < n && guard < n * 50) {
    codes.add(`${prefix}-${randomChars(4)}`);
    guard++;
  }
  return [...codes];
}
 
/**
 * Generate a referral reward code: "REF" prefix + 8 CSPRNG chars from
 * the unambiguous SAFE_CHARS alphabet.
 *
 * LOW-4 + LOW-29 (security audit 2026-05-03): replaces two ad-hoc
 * `"REF" + Math.random().toString(36).substring(2, 9).toUpperCase()`
 * call sites in orders/create + order-post-payment. Math.random was
 * predictable (PRNG state recoverable) AND lower-cardinality (base36
 * 7 chars ≈ 78B vs base32-safe 8 chars ≈ 1T), so collisions could
 * also throw on the unique constraint. CSPRNG + 8 chars from
 * 32-char alphabet = 32⁸ ≈ 1.1T candidates.
 */
export function generateReferralRewardCode(): string {
  return `REF${randomChars(8)}`;
}
