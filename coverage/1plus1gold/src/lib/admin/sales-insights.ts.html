/**
 * Sales tiles for Admin → Orders and Admin → External Sales (Mukesh 2026-11-24: "analytics on
 * the Customer order, external sales … screens, just like we did for Customer and COD").
 *
 * ONE calculation, two views, so the two screens can never disagree about a sale:
 *   scope "all"      — every order (our website / app + every marketplace and in-person sale);
 *   scope "external" — only the external channels (lib/channels registry: Shopee, Lazada,
 *                      TikTok Shop, Carousell, In person).
 *
 * Pure: the routes load the rows, this aggregates. Counts, sums and grams only — no names.
 *
 * Population (as every sales figure since 11-23, see lib/admin/cod-insights): orders placed on
 * or after the first trading day (BUSINESS_CONFIG.tradingStarted) and not by an internal or
 * supplier account (nonSalesCustomerIds). A SALE is an order in PAID_STATUSES — an order is
 * created paid (COD collected later is still a sale); a cancelled one is not.
 *
 * Per-CUSTOMER figures (customers, first-time, repeat) also leave out the marketplace "house"
 * accounts (nonPersonCustomerIds): every Shopee sale is booked to one house customer, which
 * would otherwise read as one very loyal buyer.
 *
 * Calendar months are Singapore months (lib/finance/sgt-window).
 */
import { BUSINESS_CONFIG } from "@/lib/business-config";
import { EXTERNAL_CHANNELS } from "@/lib/channels";
import { expectedCashSgd } from "@/lib/finance/reconciliation";
import { marketplaceChannelFeeSgd } from "@/lib/finance/marketplace-costs";
import { lastSgtMonths, sgtDateOf, sgtToday } from "@/lib/finance/sgt-window";
import { PAID_STATUSES } from "@/lib/order-aggregations";
import { nonPersonCustomerIds, nonSalesCustomerIds, type SalesExclusions } from "@/lib/sales/sales-exclusions";
 
export const SALES_CHART_MONTHS = 6;
 
export interface SalesInsightOrder {
  id: string;
  customer_id: string | null;
  created_at: string;
  status: string;
  payment_method: string;
  payment_status: string | null;
  grand_total_sgd: number | string | null;
  source_channel: string | null;
  mag_redeemed_sgd?: number | string | null;
  external_channel_fee_sgd?: number | string | null;
  shipper_actual_fee_sgd?: number | string | null;
}
 
/** Gold grams per order (quantity × grams each; accessories carry 0). */
export type GramsByOrder = Map<string, number>;
 
export type SalesScope = "all" | "external";
 
interface Period { count: number; salesSgd: number; grams: number; avgSgd: number | null }
 
export interface SalesInsights {
  scope: SalesScope;
  since: string;
  thisMonth: Period;
  lastMonth: Period;
  customersThisMonth: number;
  firstTimeThisMonth: number;
  /** Share of this month's customers who had bought before this month, 0–1 (null if none). */
  repeatShareThisMonth: number | null;
  cancelledThisMonth: { count: number; valueSgd: number };
  sinceLaunch: { count: number; salesSgd: number; grams: number };
  /** Marketplace channel fees on this month's sales (external view). */
  feesThisMonthSgd: number;
  /** Marketplace payouts not yet received (all time, not cancelled) — what we expect to bank. */
  payoutsAwaited: { count: number; netSgd: number };
  /** External share of ALL sales this month, 0–1 (null if no sales) — both views carry it. */
  externalShareThisMonth: number | null;
  /** This month's sales by channel, largest first. */
  byChannel: Array<{ channel: string; label: string; count: number; salesSgd: number }>;
  byMonth: Array<{ month: string; count: number; salesSgd: number; grams: number }>;
  /** The orders behind the tiles that filter the list below (tap to list them). */
  ids: { thisMonth: string[]; cancelledThisMonth: string[]; payoutsAwaited: string[] };
}
 
const EXTERNAL = new Map<string, string>(EXTERNAL_CHANNELS.map((c) => [c.channel, c.label]));
const num = (v: unknown) => (v == null || v === "" ? 0 : Number(v)) || 0;
const r2 = (n: number) => Math.round(n * 100) / 100;
const r4 = (n: number) => Math.round(n * 10000) / 10000;
 
/** The money fields as numbers (PostgREST returns numeric as strings) for the finance helpers. */
const money = (o: SalesInsightOrder) => ({
  payment_method: o.payment_method,
  grand_total_sgd: num(o.grand_total_sgd),
  mag_redeemed_sgd: num(o.mag_redeemed_sgd),
  external_channel_fee_sgd: o.external_channel_fee_sgd == null ? null : num(o.external_channel_fee_sgd),
  shipper_actual_fee_sgd: o.shipper_actual_fee_sgd == null ? null : num(o.shipper_actual_fee_sgd),
});
 
export function isExternalChannel(channel: string | null | undefined): boolean {
  return !!channel && EXTERNAL.has(channel);
}
 
export function channelLabel(channel: string | null | undefined): string {
  if (!channel || channel === "web") return "Website & app";
  return EXTERNAL.get(channel) ?? channel;
}
 
export function computeSalesInsights(
  orders: SalesInsightOrder[],
  grams: GramsByOrder,
  exclusions: SalesExclusions,
  scope: SalesScope,
  now: Date,
): SalesInsights {
  const since = BUSINESS_CONFIG.tradingStarted;
  const today = sgtToday(now);
  const months = lastSgtMonths(today, SALES_CHART_MONTHS);
  const thisKey = months[months.length - 1].key;
  const lastKey = months[months.length - 2].key;
  const monthOf = (iso: string) => sgtDateOf(iso).slice(0, 7);
 
  const nonSales = nonSalesCustomerIds(exclusions);
  const nonPerson = nonPersonCustomerIds(exclusions);
  const real = orders.filter((o) => sgtDateOf(o.created_at) >= since && !(o.customer_id && nonSales.has(o.customer_id)));
  const inScope = (o: SalesInsightOrder) => scope === "all" || isExternalChannel(o.source_channel);
  const isSale = (o: SalesInsightOrder) => PAID_STATUSES.includes(o.status);
 
  const sales = real.filter(isSale);
  const scoped = sales.filter(inScope);
 
  const period = (key: string): Period => {
    const rows = scoped.filter((o) => monthOf(o.created_at) === key);
    const salesSgd = r2(rows.reduce((s, o) => s + num(o.grand_total_sgd), 0));
    return {
      count: rows.length, salesSgd,
      grams: r4(rows.reduce((s, o) => s + (grams.get(o.id) ?? 0), 0)),
      avgSgd: rows.length ? r2(salesSgd / rows.length) : null,
    };
  };
 
  // Customers: real people only (house accounts out), first purchase = their earliest sale.
  const person = (o: SalesInsightOrder) => !!o.customer_id && !nonPerson.has(o.customer_id);
  const firstMonth = new Map<string, string>();
  for (const o of sales.filter(person)) {
    const m = monthOf(o.created_at);
    const prev = firstMonth.get(o.customer_id!);
    if (!prev || m < prev) firstMonth.set(o.customer_id!, m);
  }
  const customersNow = new Set(scoped.filter((o) => person(o) && monthOf(o.created_at) === thisKey).map((o) => o.customer_id!));
  const firstTime = [...customersNow].filter((c) => firstMonth.get(c) === thisKey).length;
 
  const cancelled = real.filter((o) => inScope(o) && o.status === "cancelled" && monthOf(o.created_at) === thisKey);
  const thisMonthScoped = scoped.filter((o) => monthOf(o.created_at) === thisKey);
 
  const thisMonthAll = sales.filter((o) => monthOf(o.created_at) === thisKey);
  const allValue = thisMonthAll.reduce((s, o) => s + num(o.grand_total_sgd), 0);
  const extValue = thisMonthAll.filter((o) => isExternalChannel(o.source_channel)).reduce((s, o) => s + num(o.grand_total_sgd), 0);
 
  // Payouts awaited: every marketplace order not yet paid out (not only since launch — what is
  // owed is owed), in scope for both views.
  const awaited = orders.filter((o) => o.payment_method === "external" && o.payment_status === "unpaid" && o.status !== "cancelled")
    .sort((a, b) => a.created_at.localeCompare(b.created_at));
 
  const fees = scoped.filter((o) => monthOf(o.created_at) === thisKey)
    .reduce((s, o) => s + marketplaceChannelFeeSgd(money(o)), 0);
 
  const chan = new Map<string, { count: number; salesSgd: number }>();
  for (const o of scoped.filter((x) => monthOf(x.created_at) === thisKey)) {
    const k = o.source_channel && o.source_channel !== "" ? o.source_channel : "web";
    const e = chan.get(k) ?? { count: 0, salesSgd: 0 };
    e.count += 1; e.salesSgd += num(o.grand_total_sgd);
    chan.set(k, e);
  }
 
  return {
    scope,
    since,
    thisMonth: period(thisKey),
    lastMonth: period(lastKey),
    customersThisMonth: customersNow.size,
    firstTimeThisMonth: firstTime,
    repeatShareThisMonth: customersNow.size ? r4((customersNow.size - firstTime) / customersNow.size) : null,
    cancelledThisMonth: { count: cancelled.length, valueSgd: r2(cancelled.reduce((s, o) => s + num(o.grand_total_sgd), 0)) },
    sinceLaunch: {
      count: scoped.length,
      salesSgd: r2(scoped.reduce((s, o) => s + num(o.grand_total_sgd), 0)),
      grams: r4(scoped.reduce((s, o) => s + (grams.get(o.id) ?? 0), 0)),
    },
    feesThisMonthSgd: r2(fees),
    payoutsAwaited: {
      count: awaited.length,
      netSgd: r2(awaited.reduce((s, o) => s + expectedCashSgd(money(o)), 0)),
    },
    externalShareThisMonth: allValue > 0 ? r4(extValue / allValue) : null,
    byChannel: [...chan.entries()]
      .map(([channel, v]) => ({ channel, label: channelLabel(channel), count: v.count, salesSgd: r2(v.salesSgd) }))
      .sort((a, b) => b.salesSgd - a.salesSgd),
    byMonth: months.map((m) => {
      const p = period(m.key);
      return { month: m.key, count: p.count, salesSgd: p.salesSgd, grams: p.grams };
    }),
    ids: {
      thisMonth: thisMonthScoped.map((o) => o.id),
      cancelledThisMonth: cancelled.map((o) => o.id),
      payoutsAwaited: awaited.map((o) => o.id),
    },
  };
}
