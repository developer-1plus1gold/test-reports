/**
 * Brand icons for the admin Dispatch and Shippers screens (Mukesh, 2026-11-20).
 * Each brand's own mark (from its own site, the simple-icons set, or our app
 * logo), stored as 64 px WebP (ours 128 px, from the vector logo Mukesh supplied 2026-11-20 —
 * the plain stacked "1+1 / GOLD", NOT the archived app icon with the bar) in public/images/brands/.
 *
 * An order shows TWO things:
 *   channel — where the order came from: our website/app, Lazada, TikTok Shop, Shopee
 *   courier — who delivers it: Lalamove, SPX, Grab, SingPost
 * A marketplace order's carrier is the marketplace's own (e.g. "SpeedPost"); it has
 * no courier icon of ours, so only its text label shows.
 *
 * Pure + tested; components/admin/BrandIcon.tsx only renders.
 */
export type BrandKey = "website" | "lazada" | "tiktok" | "shopee" | "lalamove" | "spx" | "grab" | "singpost";
 
/**
 * `fullBleed` = the mark is a complete tile on its own ground (our dark logo, SPX's
 * orange square, Lazada's gradient). Those render DIRECTLY, rounded, with no white
 * chip (Mukesh 2026-11-22: the white added nothing and flashed behind them while
 * scrolling). Marks on a transparent ground keep the white chip, because some would
 * vanish on the admin's dark background (TikTok's black note, SingPost's dark-blue
 * script) and the chip matches the payment-method logos.
 */
export const BRAND_ICONS: Record<BrandKey, { src: string; alt: string; fullBleed?: boolean }> = {
  website:  { src: "/images/brands/website.webp",  alt: "1Plus1Gold website / app", fullBleed: true },
  lazada:   { src: "/images/brands/lazada.webp",   alt: "Lazada", fullBleed: true },
  tiktok:   { src: "/images/brands/tiktok.webp",   alt: "TikTok Shop" },
  shopee:   { src: "/images/brands/shopee.webp",   alt: "Shopee" },
  lalamove: { src: "/images/brands/lalamove.webp", alt: "Lalamove" },
  spx:      { src: "/images/brands/spx.webp",      alt: "SPX Express", fullBleed: true },
  grab:     { src: "/images/brands/grab.webp",     alt: "Grab" },
  singpost: { src: "/images/brands/singpost.webp", alt: "SingPost" },
};
 
const MARKETPLACES: Record<string, BrandKey> = { lazada: "lazada", tiktok_shop: "tiktok", shopee: "shopee" };
 
/**
 * Where the order came from. A marketplace → its mark; no external channel →
 * our website/app. Other external channels (in-person, Carousell) → no icon.
 */
export function channelIconKey(o: { source_channel?: string | null }): BrandKey | null {
  const ch = String(o.source_channel ?? "").trim().toLowerCase();
  if (!ch || ch === "website" || ch === "web" || ch === "app") return "website";
  return MARKETPLACES[ch] ?? null;
}
 
/** Who delivers it, by our courier key (orders.shipper / shipper_settings.shipper). */
const COURIERS: Record<string, BrandKey> = { lalamove: "lalamove", evening: "lalamove", spx: "spx", grab: "grab", singpost: "singpost" };
 
export function courierIconKey(shipper: string | null | undefined): BrandKey | null {
  return COURIERS[String(shipper ?? "").trim().toLowerCase()] ?? null;
}
 
/**
 * Each courier's colour, for tile BORDERS only (Mukesh 2026-11-22: "colours on
 * borders is what we generally do" — never a filled tile). Inside a tile the usual
 * meaning colours stay: red = discount, green = premium. Picked from each brand
 * and checked against the dark admin background; SingPost uses its lighter blue
 * because its navy disappears on #1A1810.
 */
const COURIER_BORDER: Partial<Record<BrandKey, string>> = {
  lalamove: "#F5822A",
  spx:      "#EE4D2D",
  grab:     "#00B14F",
  singpost: "#4A7BD0",
};
 
/** Border colour for a courier's tile; the neutral gold rule for anything else. */
export function courierBorderColor(shipper: string | null | undefined): string {
  const key = courierIconKey(shipper);
  return (key && COURIER_BORDER[key]) || "var(--color-gold-dim)";
}
