/**
 * Which dispatch bucket an order belongs in.
 *
 * ▶ THE BUG THESE EXIST FOR. Each carrier had its own inline pair of
 * predicates, and both excluded `failed`:
 *
 *     unbooked = !shipper_order_id && status not in (delivered, failed)
 *     active   =  shipper_order_id && status not in (delivered, failed)
 *
 * An order that is `failed` WHILE STILL HOLDING a trip id matches NEITHER — so
 * it vanished from the Dispatch screen entirely, with no way to re-book it from
 * the app. That is exactly where a carrier-side cancellation lands, and it
 * happened live on 2026-09-11: Lalamove cancelled a booked trip and the
 * operator had to book on Lalamove's own portal.
 *
 * ▶ IT ALSO HID A SECOND, OLDER CASE. `book-delivery` writes `failed` when OUR
 * booking call errors (PR #136) — no trip id, status failed — which the old
 * predicate excluded too. That route's own comment claims "the dispatch tile
 * shows it… the operator taps Book again to retry". It did not and they could
 * not. Silent since #136, because a failed booking is rare and neither the
 * operator nor the customer saw anything.
 *
 * ▶ `failed` IS RE-BOOKABLE, NOT TERMINAL. It means "this order still needs a
 * courier", which is the same thing the operator wants to do about it as a
 * never-booked order — so it belongs in the same bucket rather than in a state
 * of its own. `/api/spx/book` already drew this line correctly
 * (`shipper_order_id && shipper_status !== "failed"`); this brings the rest
 * into line rather than inventing a rule.
 */
 
import { LANE_CARRIERS } from "@/lib/shipping/lanes";
 
/** Only the fields the bucketing reads. Keeps this callable from loose admin rows. */
export interface DispatchBucketOrder {
  shipper_order_id?: string | null;
  shipper_status?:   string | null;
}
 
/** Terminal for bucketing: delivered orders leave the dispatch screen for good. */
const DELIVERED = "delivered";
/** Not terminal — a retryable state. See the header. */
const FAILED = "failed";
 
/**
 * Still needs a courier booked: never booked, OR booked and then failed (ours
 * or the carrier's). A re-book overwrites the stale trip id on success.
 */
export function needsCourierBooking(o: DispatchBucketOrder): boolean {
  const status = o.shipper_status ?? "";
  if (status === DELIVERED) return false;
  return !o.shipper_order_id || status === FAILED;
}
 
/**
 * Booked and in flight. Deliberately still excludes `failed`, so a failed order
 * appears in exactly ONE bucket (the one with the Book button) rather than
 * being listed twice.
 */
export function isCourierDispatched(o: DispatchBucketOrder): boolean {
  const status = o.shipper_status ?? "";
  return !!o.shipper_order_id && status !== DELIVERED && status !== FAILED;
}
 
/** True when a Book action is re-booking over a dead trip id, not booking fresh. */
export function isRebook(o: DispatchBucketOrder): boolean {
  return !!o.shipper_order_id && (o.shipper_status ?? "") === FAILED;
}
 
/** Fields the external-sale bucket reads. */
export interface ExternalHandoverOrder {
  source_channel?: string | null;
  shipper?:        string | null;
  shipper_status?: string | null;
}
 
/**
 * An external sale (marketplace or in-person channel) still to be handed over —
 * the "<channel> · marketplace ships" section on Dispatch.
 *
 * ▶ A SELF-COLLECT IS NOT A HAND-OVER (2026-11-22). An in-person sale the
 * customer collects (shipper 'self_pickup') is already in "Self pickup · awaiting
 * collection", where Mark Collected closes it. Matching it here too listed the
 * same order twice in To pack, with two different closing buttons (Sneh Makhija
 * EC823F30). Vault orders are excluded by the caller (isAlreadyInVault).
 */
export function isExternalHandover(o: ExternalHandoverOrder, externalChannels: readonly string[]): boolean {
  return externalChannels.includes(o.source_channel ?? "")
    && o.shipper !== "self_pickup"
    && !shipsWithOurCourier(o)
    && o.shipper_status !== DELIVERED && o.shipper_status !== FAILED;
}
 
/** Couriers WE book from Dispatch: the lane carriers plus NinjaVan. */
export const OUR_COURIERS: readonly string[] = [...LANE_CARRIERS, "ninjavan"];
 
/**
 * ▶ AN IN-PERSON SALE WE SHIP OURSELVES IS NOT A HAND-OVER EITHER (2026-11-23).
 * Moving an in-person order onto SPX (Manoj Kewalramani 0B35EB6C) put it in
 * "SPX · awaiting booking" AND in "In person · marketplace ships", the same
 * double listing as the self-collect above. Only `in_person` is checked: a
 * marketplace order's `shipper` names the marketplace's own carrier, which may
 * be the same slug (a TikTok parcel on `spx`), and that order IS a hand-over.
 */
function shipsWithOurCourier(o: ExternalHandoverOrder): boolean {
  return o.source_channel === "in_person" && OUR_COURIERS.includes(o.shipper ?? "");
}
