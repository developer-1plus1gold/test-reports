/**
 * Per-order "buyer" statistics for the admin dashboard.
 *
 * For each order we report how many orders share the same *buyer* and this
 * order's 1-based chronological position among them (order #k of N).
 *
 * The subtlety is EXTERNAL / marketplace orders: they're all owned by one
 * synthetic per-channel "house customer" (getOrCreateChannelHouseCustomer), so
 * grouping by `customer_id` would lump every Shopee buyer together and report a
 * nonsense count. The real buyer lives on `external_buyer_label`, so external
 * orders are keyed by (channel, buyer label) instead. A generic/empty buyer
 * label ("Shopee buyer") carries no identity — those orders are left ungrouped
 * (`grouped:false`, count 1) so the UI shows no misleading repeat signal.
 *
 * GOLD vs ALL (added 2026-08-27). The count answers a different question on each
 * screen, so we report both rather than picking one:
 *   • Dispatch asks "how many parcels for this person right now?" — an ALL-orders
 *     count is what surfaces a consolidation opportunity. On 2026-08-27 a customer
 *     bought a 1g bar and, 7 minutes later, a S$5 storage box to the same address;
 *     a gold-only count would have shown "First order" on both and hinted nothing.
 *   • Customers asks "how good a gold customer is this?" — accessories are noise.
 * Measured on prod the same day: 262 gold orders / 34 customers vs 6
 * accessory-only orders / 2 customers (~2%). The average understates it, because
 * the error is INVERSELY proportional to customer size — 5 stray accessory orders
 * against 207 gold is 2% noise, but 1 against 1 is a 100% error on a brand-new
 * customer, which is exactly when the signal matters most.
 *
 * Pure + total: the caller passes the full order set; no DB access here.
 */
 
import { isExternalChannel } from "@/lib/channels";
import { isGenericBuyerLabel } from "@/lib/channels/external-order-claim";
 
export interface OrderStatInput {
  id:                    string;
  customer_id:           string | null;
  created_at:            string;
  source_channel?:       string | null;
  external_buyer_label?: string | null;
  /** Gold weight on the order. PostgREST returns numerics as strings ("1.0000"),
   *  so this is deliberately widened rather than typed `number`. */
  total_grams?:          number | string | null;
}
 
export interface OrderStat {
  /** How many orders share this order's effective buyer (gold AND accessory). */
  count:     number;
  /** 1-based chronological position among that buyer's orders (oldest = #1). */
  seq:       number;
  /** How many of that buyer's orders contain gold. <= count. */
  goldCount: number;
  /** 1-based position among that buyer's GOLD orders, or null when THIS order
   *  carries no gold (an accessory-only order has no place in the gold sequence). */
  goldSeq:   number | null;
  /** True when grouped by a real identity (a first-party customer, or a named
   *  marketplace buyer). False for an anonymous marketplace buyer / no customer —
   *  count/seq are then a meaningless 1/1 and the UI should show no repeat badge. */
  grouped: boolean;
}
 
/**
 * The key an order is counted under:
 *  - external channel + a real (non-generic) buyer label → that marketplace buyer
 *  - external channel + generic/empty buyer label        → null (anonymous)
 *  - first-party order                                   → its customer_id
 *  - no customer_id                                      → null
 */
export function effectiveBuyerKey(o: OrderStatInput): string | null {
  const channel = (o.source_channel ?? "").trim();
  if (channel && isExternalChannel(channel)) {
    if (isGenericBuyerLabel(o.external_buyer_label)) return null; // anonymous marketplace buyer
    return `ext:${channel.toLowerCase()}:${(o.external_buyer_label ?? "").trim().toLowerCase()}`;
  }
  return o.customer_id ? `cust:${o.customer_id}` : null;
}
 
/**
 * Does this order contain gold?
 *
 * `total_grams > 0` is the discriminator, and it was verified against prod rather
 * than assumed: all 6 zero-gram orders in the register are accessory-only, and no
 * gold-yield or conversion order has zero grams.
 *
 * Typed on the ONE field it reads rather than on OrderStatInput, so callers that
 * hold a leaner row (the Customers route selects no `id`) can use the same
 * predicate instead of casting — one definition of "is a gold order", repo-wide.
 *
 * ⚠️ This is a PROXY — it means "has gold" only while gold is the only
 * gram-bearing product we sell. `products.metal_type` already exists; if silver
 * (or any other metal) ever ships, this predicate must move to metal_type rather
 * than quietly counting silver as gold.
 */
export function hasGold(o: { total_grams?: number | string | null }): boolean {
  return Number(o.total_grams ?? 0) > 0;
}
 
/** order id → {count, seq, goldCount, goldSeq, grouped}. */
export function computeCustomerOrderStats(orders: OrderStatInput[]): Map<string, OrderStat> {
  const byKey = new Map<string, OrderStatInput[]>();
  for (const o of orders) {
    const key = effectiveBuyerKey(o);
    if (!key) continue; // ungroupable → filled as a singleton below
    const arr = byKey.get(key) ?? [];
    arr.push(o);
    byKey.set(key, arr);
  }
 
  const result = new Map<string, OrderStat>();
  for (const arr of byKey.values()) {
    // Oldest-first so the earliest order is #1; stable tie-break on id.
    const sorted = [...arr].sort((a, b) =>
      a.created_at < b.created_at ? -1 : a.created_at > b.created_at ? 1 : a.id < b.id ? -1 : 1,
    );
    const goldCount = arr.filter(hasGold).length;
    let goldSeen = 0;
    sorted.forEach((x, idx) => {
      // goldSeq advances only on gold orders, so an accessory-only order does not
      // consume a position in the gold sequence (and reports null, not a number).
      const goldSeq = hasGold(x) ? ++goldSeen : null;
      result.set(x.id, { count: arr.length, seq: idx + 1, goldCount, goldSeq, grouped: true });
    });
  }
  // Ungroupable orders (anonymous external buyer / null customer) — singletons.
  for (const o of orders) {
    if (!result.has(o.id)) {
      const gold = hasGold(o);
      result.set(o.id, { count: 1, seq: 1, goldCount: gold ? 1 : 0, goldSeq: gold ? 1 : null, grouped: false });
    }
  }
  return result;
}
 
/**
 * The dispatch-tile repeat badge — pure, so the rule can be tested. It was
 * inline JSX in a 5,000-line page, which is how the bug below survived.
 *
 * ▶ "Repeat" is about THIS order's position, not the buyer's lifetime total.
 * The original test was `count > 1`, so a customer's FIRST order was badged
 * "Repeat · order #1 of 2" the moment a second order existed. Seen live on
 * 2026-08-27: a first gold purchase read as a repeat because a S$5 storage box
 * followed 7 minutes later. `seq > 1` is the honest test.
 *
 * ALL orders stay the headline here — that is the parcel count, and it is what
 * surfaces a same-address consolidation. Gold rides alongside so an accessory
 * order is never mistaken for gold custom.
 */
export function orderRepeatBadge(
  stat: Pick<OrderStat, "count" | "seq" | "goldCount" | "goldSeq">,
  who = "this customer",
): { isRepeat: boolean; label: string; title: string } {
  const { count, seq, goldCount, goldSeq } = stat;
  const isRepeat = seq > 1;
  // Suppress the gold note when it would just repeat the total.
  const goldNote = goldCount > 0 && goldCount !== count ? ` · ${goldCount} gold` : "";
  const goldTitle = goldSeq === null
    ? ` No gold on this order; ${goldCount} of their orders carry gold.`
    : ` This is their gold order #${goldSeq} of ${goldCount}.`;
 
  return isRepeat
    ? {
        isRepeat: true,
        label: `Repeat · order #${seq} of ${count}${goldNote}`,
        title: `${who} has placed ${count} orders — this is order #${seq}.${goldTitle}`,
      }
    : {
        isRepeat: false,
        label: `First order${count > 1 ? ` · ${count} total${goldNote}` : ""}`,
        title: `First order from ${who}${count > 1 ? ` — they have ${count} orders in total.` : "."}${goldTitle}`,
      };
}
