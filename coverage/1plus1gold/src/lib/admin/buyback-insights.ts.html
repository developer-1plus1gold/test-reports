/**
 * Tiles for Admin → Sell-to-Us Requests (Mukesh 2026-11-24, "just like Customer and COD").
 *
 * Pure: GET /api/admin/buybacks/insights loads the rows, this aggregates. Counts, sums, grams and
 * days — no names.
 *
 * WHAT COUNTS AS "SELLING TO US": a request a customer made to be paid for their gold — bars at
 * home (self drop-off / Lalamove pickup), a vaulted bar, or MAG grams. Two kinds of
 * buyback_requests row are NOT, and are left out:
 *   • payout_method "store_credit" — the one-off 2026-10-25 vault → MAG conversion (45 rows),
 *     settled as credit on a new order (lib/finance/consignment-payments);
 *   • MAG redeem rows (mag_redeem_ref set) — grams spent at checkout, not sold back.
 * Internal / supplier accounts are left out like every sales figure (nonSalesCustomerIds).
 *
 * "Paid out" is dated by paid_at (when the payout was marked); a request is counted in the month
 * it was submitted. Calendar months are Singapore months.
 */
import { lastSgtMonths, sgtDateOf, sgtToday } from "@/lib/finance/sgt-window";
import { nonSalesCustomerIds, type SalesExclusions } from "@/lib/sales/sales-exclusions";
 
export const BUYBACK_CHART_MONTHS = 6;
/** Statuses where we still owe the customer something (an answer or the money). */
export const BUYBACK_OPEN_STATUSES = ["submitted", "received", "approved"] as const;
 
export interface BuybackInsightRow {
  id: string;
  customer_id: string | null;
  status: string;
  shipping_type: string | null;
  payout_method: string | null;
  mag_listing_id: string | null;
  vault_holding_id: string | null;
  mag_redeem_ref: string | null;
  total_payout_sgd: number | string | null;
  submitted_at: string | null;
  paid_at: string | null;
  created_at: string;
  /** Sum of the request's line grams and quotes (the route adds them). */
  grams: number;
  quotedSgd: number | null;
}
 
export type BuybackRoute = "home_dropoff" | "home_pickup" | "vault" | "mag";
 
export interface BuybackInsights {
  requestsThisMonth: number;
  requestsLastMonth: number;
  paidThisMonth: { count: number; amountSgd: number; grams: number };
  paidLastMonth: { count: number; amountSgd: number; grams: number };
  open: { count: number; quotedSgd: number; oldestDays: number | null };
  /** Days from submitted to paid, over paid requests (null when none). */
  timeToPay: { medianDays: number | null; sample: number };
  avgPayoutSgd: number | null;
  /** Rejected or cancelled (by us or the customer), all time. */
  notCompleted: number;
  sinceStart: { count: number; amountSgd: number; grams: number };
  byRoute: Array<{ route: BuybackRoute; label: string; count: number }>;
  byMonth: Array<{ month: string; count: number; amountSgd: number }>;
  /** The requests behind the tiles that filter the list below (tap to list them). */
  ids: { open: string[]; thisMonth: string[]; notCompleted: string[] };
}
 
const ROUTE_LABEL: Record<BuybackRoute, string> = {
  home_dropoff: "Bars at home · drop-off",
  home_pickup: "Bars at home · pickup",
  vault: "From the vault",
  mag: "MAG grams",
};
 
const num = (v: unknown) => (v == null || v === "" ? 0 : Number(v)) || 0;
const r2 = (n: number) => Math.round(n * 100) / 100;
const r4 = (n: number) => Math.round(n * 10000) / 10000;
const DAY = 86_400_000;
 
export function isCustomerSellBack(r: Pick<BuybackInsightRow, "payout_method" | "mag_redeem_ref" | "status">): boolean {
  return r.payout_method !== "store_credit" && !r.mag_redeem_ref && r.status !== "draft";
}
 
export function buybackRoute(r: Pick<BuybackInsightRow, "mag_listing_id" | "vault_holding_id" | "shipping_type">): BuybackRoute {
  if (r.mag_listing_id) return "mag";
  if (r.vault_holding_id || r.shipping_type === "vault") return "vault";
  return r.shipping_type === "lalamove" ? "home_pickup" : "home_dropoff";
}
 
export function computeBuybackInsights(rows: BuybackInsightRow[], exclusions: SalesExclusions, now: Date): BuybackInsights {
  const months = lastSgtMonths(sgtToday(now), BUYBACK_CHART_MONTHS);
  const thisKey = months[months.length - 1].key;
  const lastKey = months[months.length - 2].key;
  const monthOf = (iso: string) => sgtDateOf(iso).slice(0, 7);
  const nonSales = nonSalesCustomerIds(exclusions);
 
  const real = rows.filter((r) => isCustomerSellBack(r) && !(r.customer_id && nonSales.has(r.customer_id)));
  const submittedAt = (r: BuybackInsightRow) => r.submitted_at ?? r.created_at;
  const paid = real.filter((r) => !!r.paid_at);
 
  const paidIn = (key: string) => {
    const p = paid.filter((r) => monthOf(r.paid_at!) === key);
    return { count: p.length, amountSgd: r2(p.reduce((s, r) => s + num(r.total_payout_sgd), 0)), grams: r4(p.reduce((s, r) => s + r.grams, 0)) };
  };
 
  const open = real.filter((r) => (BUYBACK_OPEN_STATUSES as readonly string[]).includes(r.status) && !r.paid_at);
  const oldest = open.reduce<number | null>((m, r) => {
    const d = Math.floor((now.getTime() - new Date(submittedAt(r)).getTime()) / DAY);
    return m == null || d > m ? d : m;
  }, null);
 
  const waits = paid.map((r) => (new Date(r.paid_at!).getTime() - new Date(submittedAt(r)).getTime()) / DAY).filter((d) => d >= 0).sort((a, b) => a - b);
  const median = waits.length
    ? waits.length % 2 ? waits[(waits.length - 1) / 2] : (waits[waits.length / 2 - 1] + waits[waits.length / 2]) / 2
    : null;
 
  const routes = new Map<BuybackRoute, number>();
  for (const r of real) routes.set(buybackRoute(r), (routes.get(buybackRoute(r)) ?? 0) + 1);
 
  const totalPaid = paid.reduce((s, r) => s + num(r.total_payout_sgd), 0);
  const thisMonth = real.filter((r) => monthOf(submittedAt(r)) === thisKey);
  const notCompleted = real.filter((r) => r.status === "rejected" || r.status === "cancelled");
  return {
    requestsThisMonth: thisMonth.length,
    requestsLastMonth: real.filter((r) => monthOf(submittedAt(r)) === lastKey).length,
    paidThisMonth: paidIn(thisKey),
    paidLastMonth: paidIn(lastKey),
    open: { count: open.length, quotedSgd: r2(open.reduce((s, r) => s + (r.quotedSgd ?? 0), 0)), oldestDays: oldest },
    timeToPay: { medianDays: median == null ? null : Math.round(median * 10) / 10, sample: waits.length },
    avgPayoutSgd: paid.length ? r2(totalPaid / paid.length) : null,
    notCompleted: notCompleted.length,
    sinceStart: { count: paid.length, amountSgd: r2(totalPaid), grams: r4(paid.reduce((s, r) => s + r.grams, 0)) },
    byRoute: (Object.keys(ROUTE_LABEL) as BuybackRoute[]).map((route) => ({ route, label: ROUTE_LABEL[route], count: routes.get(route) ?? 0 })),
    byMonth: months.map((m) => {
      const p = paidIn(m.key);
      return { month: m.key, count: p.count, amountSgd: p.amountSgd };
    }),
    ids: {
      open: [...open].sort((a, b) => submittedAt(a).localeCompare(submittedAt(b))).map((r) => r.id),
      thisMonth: thisMonth.map((r) => r.id),
      notCompleted: notCompleted.map((r) => r.id),
    },
  };
}
