/**
 * Admin Customers list — display + sort helpers.
 *
 * OAuth (Apple/Google) and phone-OTP customers often have no profile full_name
 * or phone, so the email + sign-in provider are how the admin identifies them.
 * These pure helpers drive the name line, the muted contact sub-line, and the
 * sortable columns. Kept here (not inline in the admin page) so they're unit-
 * tested independently.
 */
 
export type CustomerSortable = {
  id:                        string;
  full_name:                 string | null;
  phone:                     string | null;
  email:                     string | null;
  provider:                  string | null;
  kyc_status:                string | null;
  signed_up_at:              string | null;
  last_sign_in_at:           string | null;
  last_order_at:             string | null;
  self_pickup_enabled:       boolean;
  self_dropoff_enabled:      boolean;
  is_gold_yield_participant: boolean;
  wholesale_enabled:         boolean;
  vault_enabled:             boolean;
  /** Post-take-over SHELL: a synthetic WhatsApp-OTP account whose phone has
   *  since been transferred to a real account by link-phone's HIGH-C2 flow.
   *  Such a row is legitimately blank (phone moved away, synthetic email nulled
   *  for display) and used to read as a data bug. Optional so every existing
   *  caller and test compiles unchanged. See lib/admin/shell-accounts. */
  is_phone_shell?:      boolean;
  shell_moved_to_name?: string | null;
  shell_moved_phone?:   string | null;
};
 
export type CustSortKey =
  | "name" | "signed_up_at" | "last_sign_in_at" | "last_order_at" | "kyc_status"
  | "self_pickup_enabled" | "self_dropoff_enabled" | "is_gold_yield_participant"
  | "wholesale_enabled" | "vault_enabled";
 
export type CustSort = { key: CustSortKey; dir: "asc" | "desc" };
 
const PROVIDER_LABEL: Record<string, string> = { apple: "Apple", google: "Google", email: "Email", phone: "WhatsApp" };
export const providerLabel = (p: string | null | undefined) => (p ? (PROVIDER_LABEL[p] ?? p) : "");
 
/** Best display name: real name → email → phone → short id. Never a bare dash. */
export const customerDisplayName = (u: CustomerSortable) =>
  u.full_name?.trim() || u.email || u.phone || `Customer ${u.id.slice(0, 8)}`;
 
/** Muted contact sub-line: phone + email + provider, skipping whatever the name
 *  line already shows (incl. the email/phone when it's the display-name fallback). */
export const customerContactLine = (u: CustomerSortable) => {
  const shown = customerDisplayName(u);
  const bits = [u.phone, u.email].filter((v): v is string => !!v && v !== shown);
  // A post-take-over shell has no phone and no displayable email, so without
  // this the sub-line is empty and the row looks like a data bug. Say what it
  // is, and where the phone went when we can identify the new holder.
  if (u.is_phone_shell) {
    const dest = u.shell_moved_to_name?.trim() || "another account";
    bits.push(u.shell_moved_phone ? `↪ phone ${u.shell_moved_phone} moved to ${dest}` : `↪ phone moved to ${dest}`);
  }
  const prov = providerLabel(u.provider);
  if (prov) bits.push(prov);
  return bits.join("  ·  ");
};

const CUST_DATE_KEYS = new Set<CustSortKey>(["signed_up_at", "last_sign_in_at", "last_order_at"]);
 
/** Comparable value for a column: name/kyc → lowercased string, dates → epoch ms
 *  (null = 0, sorts oldest/empty last when desc), flags → 1/0. */
export function custSortValue(u: CustomerSortable, key: CustSortKey): string | number {
  if (key === "name") return customerDisplayName(u).toLowerCase();
  Iif (key === "kyc_status") return (u.kyc_status ?? "").toLowerCase();
  if (CUST_DATE_KEYS.has(key)) { const v = u[key] as string | null; return v ? new Date(v).getTime() : 0; }
  return (u[key] as boolean) ? 1 : 0;

/** Default direction when first clicking a column: text → A→Z, dates/flags → high-first. */
export const defaultDirForKey = (key: CustSortKey): "asc" | "desc" =>
  (key === "name" || key === "kyc_status") ? "asc" : "desc";
 
/** Stable, non-mutating sort of customers by the active column + direction. */
export function sortCustomers<T extends CustomerSortable>(list: T[], sort: CustSort): T[] {
  return [...list].sort((a, b) => {
    const av = custSortValue(a, sort.key), bv = custSortValue(b, sort.key);
    const cmp = av < bv ? -1 : av > bv ? 1 : 0;
    return sort.dir === "asc" ? cmp : -cmp;
  });
}
 
/**
 * The order-count chip on a Customers row — ONE definition, used by both the
 * mobile card and the desktop table so the two can never drift.
 *
 * Leads with GOLD orders (2026-08-27). The Customers screen's question is "how
 * good a gold customer is this?", and an accessory-only order is noise against
 * it: on 2026-08-27 a customer's 1g bar plus a S$5 storage box read as "2 orders",
 * making a first-time gold buyer look like a repeat one. The all-orders total is
 * kept alongside — never dropped — because it is still the honest count of what
 * we shipped them; it is simply not the headline here. (Dispatch is the mirror
 * image: ALL is the headline there, because that is the parcel count.)
 *
 * Marketplace house accounts are unchanged: they own many buyers' orders under
 * one synthetic customer, so the meaningful figure is distinct buyers.
 */
export function customerOrderChip(u: {
  is_marketplace_house?:    boolean | null;
  marketplace_buyer_count?: number | null;
  order_count?:             number | null;
  gold_order_count?:        number | null;
}): string | null {
  if (u.is_marketplace_house) {
    const n = u.marketplace_buyer_count ?? 0;
    return n > 0 ? `${n} buyer${n === 1 ? "" : "s"}` : null;
  }
  const total = u.order_count ?? 0;
  if (total <= 0) return null;
  const gold = u.gold_order_count ?? 0;
  // No gold at all — say so plainly rather than showing a bare order count that
  // reads as gold custom.
  if (gold === 0) return `${total} order${total === 1 ? "" : "s"} · no gold`;
  const head = `${gold} gold order${gold === 1 ? "" : "s"}`;
  return gold === total ? head : `${head} · ${total} total`;
}
