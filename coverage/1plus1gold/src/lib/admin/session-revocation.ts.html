/**
 * Admin session revocation — the "force re-2FA" mechanism (2026-09-07).
 *
 * The admin session is a stateless HMAC cookie ({issuedAt}.{hmac}); there is no
 * server-side session record to invalidate. So revocation is a single
 * timestamp in admin_security_controls: any session issued strictly before it
 * is rejected by checkAdminAuth, forcing a fresh login — which re-runs the
 * WhatsApp OTP, because verify-2fa is the only path that mints a session cookie.
 *
 * Read path is CACHED (short TTL) and FAIL-OPEN: this sits on the universal
 * admin-auth hot path, so a DB blip must never lock every admin out — the HMAC
 * check is still the primary gate, and revocation is defence-in-depth on top.
 */

/** Extract the issuedAt (ms) from a `{issuedAt}.{hmac}` admin session cookie. */
export function parseSessionIssuedAtMs(cookieValue: string): number | null {
  const dot = cookieValue.indexOf(".");
  if (dot < 1) return null;
  const n = Number(cookieValue.slice(0, dot));
  return Number.isFinite(n) && n > 0 ? n : null;
}
 
/** Pure: is a session issued at `issuedAtMs` revoked by a `revokeBeforeMs`
 *  cut-off? 0 (no cut-off) never revokes anything. */
export function isRevoked(issuedAtMs: number, revokeBeforeMs: number): boolean {
  return revokeBeforeMs > 0 && issuedAtMs < revokeBeforeMs;
}

/** Is this admin session cookie's issue time before the current revocation

}
 
/**
 * Force re-2FA for ALL admins: set the revocation cut-off to `tsMs` so every
 * session issued before it is invalid. For a two-person admin team this
 * deliberately re-logs-in everyone — the correct response to a suspected
 * mass-export, and the OTP goes to the operator's phone, so an attacker without
 * that phone cannot get back in.
 */
export async function revokeAdminSessionsBefore(
  sb: SupabaseClient,
  tsMs: number,
  reason: string,
): Promise<void> {
  try {
    await sb.from("admin_security_controls").upsert(
      { id: 1, revoke_sessions_before: new Date(tsMs).toISOString(), reason, updated_at: new Date().toISOString() },
      { onConflict: "id" },
    );
    cache = null; // bust so the next auth check sees the new cut-off immediately
  } catch (err) {
    console.error("[session-revocation] revoke write failed:", (err as Error)?.message ?? err);
  }
}
