/**
 * Customer structure tiles for Admin → Customers (Mukesh 2026-11-21: "some information at the
 * top in tiles … to understand our customer structure").
 *
 * Pure: the route (GET /api/admin/users/insights) loads the rows, this turns them into
 * aggregates. Counts, sums and shares — plus, for the tiles that filter the list below them,
 * the customer ids in that segment (`segments`). Never a name, phone or email.
 *
 * Population ("customers") = real people with an account: every profile EXCEPT the accounts
 * lib/sales/sales-exclusions.ts keeps out of per-customer metrics (internal / demo / test,
 * supplier accounts, marketplace + in-person house accounts — one house row is many buyers)
 * and post-take-over shell accounts (a husk of a person who also has a real account —
 * counting both double-counts them; lib/admin/shell-accounts).
 *
 * A "buyer" has at least one PAID first-party order carrying gold (accessory-only orders do not
 * make a gold customer — the same rule as the list's gold-order chip, hasGold). Marketplace
 * buyers are counted separately, by their marketplace identity (effectiveBuyerKey).
 *
 * Every calendar boundary is a Singapore day (lib/finance/sgt-window).
 */
import { isExternalChannel } from "@/lib/channels";
import { effectiveBuyerKey, hasGold } from "@/lib/admin/order-stats";
import { isShellAccount } from "@/lib/admin/shell-accounts";
import { sgtDayStart, sgtToday } from "@/lib/finance/sgt-window";
import { PAID_STATUSES } from "@/lib/order-aggregations";
import { nonPersonCustomerIds, nonSalesCustomerIds, type SalesExclusions } from "@/lib/sales/sales-exclusions";
 
export interface InsightAccount {
  id: string;
  phone: string | null;
  /** Raw auth email (synthetic OTP address included) — used only to classify the account. */
  rawEmail: string | null;
  /** auth.users.created_at. */
  signedUpAt: string | null;
  /** auth app_metadata.provider: "phone", "google", "apple", "email", … */
  provider: string | null;
}
 
export interface InsightOrder {
  customer_id: string | null;
  created_at: string;
  status: string;
  grand_total_sgd: number | string | null;
  total_grams: number | string | null;
  source_channel: string | null;
  external_buyer_label: string | null;
}
 
export interface InsightHolding { customer_id: string | null; grams: number | string | null }
 
/** The tiles that filter the Customers list. */
export const CUSTOMER_SEGMENT_KEYS = ["newThisMonth", "buyers", "repeat", "lapsed", "top10", "holding"] as const;
export type CustomerSegmentKey = typeof CUSTOMER_SEGMENT_KEYS[number];
 
export type SignupMethod = "whatsapp" | "google" | "apple" | "email" | "other";
 
export interface CustomerInsights {
  customers: number;
  newThisMonth: number;
  newLastMonth: number;
  buyers: number;
  /** buyers / customers, 0–1. */
  conversion: number;
  repeatBuyers: number;
  /** repeatBuyers / buyers, 0–1. */
  repeatRate: number;
  /** Buyers by number of paid gold orders. */
  orderBands: { one: number; two: number; threeToFive: number; sixPlus: number };
  active90: number;
  lapsed: number;
  avgLifetimeSgd: number;
  medianLifetimeSgd: number;
  /** Share of first-party revenue from the top 10 % of buyers (at least one buyer), 0–1. */
  top10Share: number;
  holdingCustomers: number;
  holdingGrams: number;
  signupMix: Record<SignupMethod, number>;
  marketplaceBuyers: number;
  marketplaceOrders: number;
  /** Customer ids behind each tile, so tapping it filters the list to exactly the customers it
   *  counted (Mukesh 11-21). Buyer segments are most recent gold order first; `top10` is highest
   *  spend first; `newThisMonth` and `holding` are newest sign-up / most grams first. */
  segments: Record<CustomerSegmentKey, string[]>;
  asOf: string;
}
 
const DAY_MS = 86_400_000;
const num = (v: unknown) => { const n = Number(v); return Number.isFinite(n) ? n : 0; };
const r2 = (n: number) => Math.round(n * 100) / 100;
 
export function signupMethod(provider: string | null, rawEmail: string | null): SignupMethod {
  const p = (provider ?? "").toLowerCase();
  if (p === "google") return "google";
  if (p === "apple") return "apple";
  if (p === "phone" || /^phone_\d+@auth\.1plus1gold\.com$/i.test(rawEmail ?? "")) return "whatsapp";
  if (p === "email") return "email";
  return "other";
}
 
/** The account counts as a real customer. `excluded` = nonPersonCustomerIds(...). Pure. */
export function isRealCustomer(a: InsightAccount, excluded: ReadonlySet<string>): boolean {
  if (excluded.has(a.id)) return false;
  return !isShellAccount({ phone: a.phone, rawEmail: a.rawEmail });
}
 
function median(xs: number[]): number {
  if (!xs.length) return 0;
  const s = [...xs].sort((a, b) => a - b);
  const m = Math.floor(s.length / 2);
  return s.length % 2 ? s[m] : (s[m - 1] + s[m]) / 2;
}
 
/** First instant of the Singapore calendar month `offset` months before the one `today` is in. */
function sgtMonthStart(today: string, offset: number): string {
  const d = new Date(Date.UTC(Number(today.slice(0, 4)), Number(today.slice(5, 7)) - 1 - offset, 1));
  return sgtDayStart(d.toISOString().slice(0, 10));
}
 
export function computeCustomerInsights(
  accounts: InsightAccount[],
  exclusions: SalesExclusions,
  orders: InsightOrder[],
  holdings: InsightHolding[],
  now: Date,
): CustomerInsights {
  const notPerson = nonPersonCustomerIds(exclusions);
  const notSales = nonSalesCustomerIds(exclusions);
  const real = accounts.filter((a) => isRealCustomer(a, notPerson));
  const realIds = new Set(real.map((a) => a.id));
 
  const today = sgtToday(now);
  const thisMonth = sgtMonthStart(today, 0);
  const lastMonth = sgtMonthStart(today, 1);
  const signedAt = (a: InsightAccount) => (a.signedUpAt ? new Date(a.signedUpAt).toISOString() : "");
  const newThisMonthIds = real.filter((a) => signedAt(a) >= thisMonth)
    .sort((a, b) => (signedAt(a) < signedAt(b) ? 1 : signedAt(a) > signedAt(b) ? -1 : 0))
    .map((a) => a.id);
  const newThisMonth = newThisMonthIds.length;
  const newLastMonth = real.filter((a) => { const t = signedAt(a); return t >= lastMonth && t < thisMonth; }).length;
 
  const signupMix: Record<SignupMethod, number> = { whatsapp: 0, google: 0, apple: 0, email: 0, other: 0 };
  for (const a of real) signupMix[signupMethod(a.provider, a.rawEmail)]++;
 
  const paid = orders.filter((o) => PAID_STATUSES.includes(o.status));
 
  // First-party buyers: revenue from every paid order, gold-order count and last gold order.
  const per = new Map<string, { revenue: number; gold: number; lastGold: string }>();
  for (const o of paid) {
    if (!o.customer_id || !realIds.has(o.customer_id)) continue;
    if (o.source_channel && isExternalChannel(o.source_channel)) continue;
    const e = per.get(o.customer_id) ?? { revenue: 0, gold: 0, lastGold: "" };
    e.revenue += num(o.grand_total_sgd);
    if (hasGold(o)) {
      e.gold++;
      if (o.created_at > e.lastGold) e.lastGold = o.created_at;
    }
    per.set(o.customer_id, e);
  }
  const buyerRows = [...per.entries()].filter(([, e]) => e.gold > 0).map(([id, e]) => ({ id, ...e }));
  const buyers = buyerRows.length;
 
  const orderBands = { one: 0, two: 0, threeToFive: 0, sixPlus: 0 };
  for (const b of buyerRows) {
    if (b.gold === 1) orderBands.one++;
    else if (b.gold === 2) orderBands.two++;
    else if (b.gold <= 5) orderBands.threeToFive++;
    else orderBands.sixPlus++;
  }
  const repeatBuyers = buyers - orderBands.one;
 
  const cutoff90 = new Date(now.getTime() - 90 * DAY_MS).toISOString();
  const isActive = (b: { lastGold: string }) => new Date(b.lastGold).toISOString() >= cutoff90;
  const active90 = buyerRows.filter(isActive).length;
  const byRecentGold = [...buyerRows].sort((a, b) => (a.lastGold < b.lastGold ? 1 : a.lastGold > b.lastGold ? -1 : 0));
  const lapsedIds = byRecentGold.filter((b) => !isActive(b)).map((b) => b.id);
 
  const revenues = buyerRows.map((b) => b.revenue);
  const totalRevenue = revenues.reduce((s, v) => s + v, 0);
  const topN = Math.max(1, Math.ceil(buyers * 0.1));
  const topBuyers = [...buyerRows].sort((a, b) => b.revenue - a.revenue).slice(0, topN);
  const topRevenue = topBuyers.reduce((s, b) => s + b.revenue, 0);
 
  // Gold we hold for them: vault bars in storage + MAG grams allocated.
  const holders = new Map<string, number>();
  let holdingGrams = 0;
  for (const h of holdings) {
    if (!h.customer_id || !realIds.has(h.customer_id)) continue;
    const g = num(h.grams);
    if (g <= 0) continue;
    holders.set(h.customer_id, (holders.get(h.customer_id) ?? 0) + g);
    holdingGrams += g;
  }
 
  // Marketplace buyers, by their marketplace identity (anonymous buyers are not counted).
  const mpKeys = new Set<string>();
  let marketplaceOrders = 0;
  for (const o of paid) {
    if (!o.source_channel || !isExternalChannel(o.source_channel)) continue;
    if (o.customer_id && notSales.has(o.customer_id)) continue; // test / supplier orders are not sales
    marketplaceOrders++;
    const key = effectiveBuyerKey({ id: "", customer_id: o.customer_id, created_at: o.created_at, source_channel: o.source_channel, external_buyer_label: o.external_buyer_label });
    if (key) mpKeys.add(key);
  }
 
  return {
    customers: real.length,
    newThisMonth,
    newLastMonth,
    buyers,
    conversion: real.length ? buyers / real.length : 0,
    repeatBuyers,
    repeatRate: buyers ? repeatBuyers / buyers : 0,
    orderBands,
    active90,
    lapsed: buyers - active90,
    avgLifetimeSgd: buyers ? r2(totalRevenue / buyers) : 0,
    medianLifetimeSgd: r2(median(revenues)),
    top10Share: totalRevenue > 0 ? topRevenue / totalRevenue : 0,
    holdingCustomers: holders.size,
    holdingGrams: Math.round(holdingGrams * 10000) / 10000,
    signupMix,
    marketplaceBuyers: mpKeys.size,
    marketplaceOrders,
    segments: {
      newThisMonth: newThisMonthIds,
      buyers: byRecentGold.map((b) => b.id),
      repeat: byRecentGold.filter((b) => b.gold >= 2).map((b) => b.id),
      lapsed: lapsedIds,
      top10: buyers ? topBuyers.map((b) => b.id) : [],
      holding: [...holders.entries()].sort((a, b) => b[1] - a[1]).map(([id]) => id),
    },
    asOf: now.toISOString(),
  };
}
