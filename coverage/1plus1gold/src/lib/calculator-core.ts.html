/**
 * calculator-core.ts — the PURE calculator simulation (Phase 6.5).
 *
 * This is the single orchestration of the curve-vs-standard simulation, the
 * summary stats, the comparison line, progress, and the preset chips — factored
 * BY MOVE out of `/api/calculator/simulate/route.ts`. It is dependency-free
 * (imports only the pure pricing engine + pure helpers; NO supabase / next /
 * fetch / process.env), so it runs in BOTH:
 *   • the Next.js route (server), and
 *   • the on-device JS engine (esbuild → JavaScriptCore) for the iOS Buy tab,
 *     where it computes locally on every keystroke/slide = INSTANT.
 *
 * The impure inputs (live spot, ownedGrams from auth, the product catalog +
 * premium rows) are gathered ONCE by the caller and passed in as `boot`. The
 * live inputs (mode/budget/grams/currency) are passed in as `input`. All
 * returned money is SGD — the client formats to the display currency (Rule #1).
 *
 * See contract §22 + memory `feedback_on_device_js_engine_pricing_kernel`.
 * NEVER re-implement this in Swift ([[no-parallel-implementations]]).
 */
 
import {
  executeSimulateCart,
  pickStandardOption,
  type AiCartSimulation,
  type CartSimItem,
  type PaymentMethodRow,
  type StandardProductForAI,
  type AccessoryForAI,
  type CurveComparison,
} from "./ai-calculator";
import { getComparisons } from "./comparisons";
import { toPremiumMap, type PremiumRow } from "./premiums";
import { formatPremium, roundToNice, formatDateTime } from "./format-pure";
import { PRODUCT_CATALOG } from "./products";
 
const TROY_OZ = 31.1035;
 
const PRESET_BUDGETS_SGD = [500, 1000, 2500, 5000, 10000, 25000];
const PRESET_GRAMS = [1, 5, 10, 50, 100, 200];
 
/** key → product_id + image, for native add-to-cart + OptionCard thumbnails.
 *  Keys: `curve:<grams>`, `standard:<grams>`, `accessory:<lowercased label>`.
 *  (Array form — a Map is not JSON-serialisable across the bootstrap boundary.) */
export interface CalcMetaEntry {
  key: string;
  productId: string | null;
  imageUrl: string | null;
}
 
/** Everything the pure core needs that it can't compute locally — gathered ONCE
 *  per Calculator open (server route or iOS `/api/calculator/bootstrap`). */
export interface CalcBootstrap {
  spotSgd: number;                        // SGD gram price
  spotTs: number;
  currencyGram: Record<string, number>;   // per-currency gram price (incl. SGD)
  premiums: PremiumRow[];
  standardProducts: StandardProductForAI[];
  meta: CalcMetaEntry[];
  ownedGrams: number;                      // resolved from auth (0 = anon)
  goalGrams: number;                       // default 50
}
 
/** Live inputs — change as the user types / drags / switches currency. */
export interface CalcInput {
  mode?: string;
  budget?: number | string;
  grams?: number | string;
  value?: number | string;   // legacy single-value fallback (AI / tests)
  currency?: string;
}
 
interface PresetChip { value: string; label: string; active: boolean; }
interface OptionItem {
  type: CartSimItem["type"];
  grams: number; label: string; count: number; isAssay: boolean;
  priceEach: number; productId: string | null; imageUrl: string | null;
}
interface OptionPayload {
  items: OptionItem[];
  totalGrams: number; totalSgd: number; perGramSgd: number;
  displayPremium: number; displayPremiumLabel: string;
  curveComparison: CurveComparison | null;
}
export interface SimulateResponse {
  spotSgd: number | null;
  /** spot per troy-oz in SGD (= spotSgd × 31.1035) — for the spot line "/oz". */
  spotOzSgd: number | null;
  spotTs: number | null;
  /** Preformatted "3 Jun 2026, 14:35:22" for the spot line "Updated …". */
  spotUpdatedLabel: string | null;
  hasResult: boolean;
  tooLow: boolean;
  summary: {
    totalGrams: number; blendedPremium: number; premiumLabel: string;
    perGramSgd: number; totalSgd: number; unspentSgd: number;
  } | null;
  comparisonLine: { spend: string; punchline: string } | null;
  /** `accumulatedLabel` = "8.50g accumulated + 12.00g from this purchase"
   *  (null when ownedGrams 0 — the PWA only shows it for signed-in holders). */
  progress: { currentG: number; targetG: number; ownedGrams: number; accumulatedLabel: string | null };
  options: { curve: OptionPayload | null; standard: OptionPayload | null };
  standardNote: string | null;
  /** SGD value for the "My budget" line on the Accumulation chart embed:
   *  budget mode → the monthly budget; grams mode → the order total (PWA
   *  calculator/page.tsx:683). null when there's no result. (#14) */
  accumulationBudgetSgd: number | null;
  sliderMin: number;
  sliderMax: number;
  presetsGrams: PresetChip[];
  presetsBudget: PresetChip[];
}
 
/** Preset chips for the SDUI repeater — value (set into state) + label +
 *  `active` (highlights the matching chip). Budget presets are currency-scaled. */
function buildPresets(gramsNum: number, budgetNum: number, rate: number) {
  return {
    presetsGrams: PRESET_GRAMS.map((g) => ({
      value: String(g), label: `${g}g`, active: gramsNum === g,
    })),
    presetsBudget: PRESET_BUDGETS_SGD.map((sgd) => {
      const inCur = roundToNice(sgd * rate);
      return {
        value: String(inCur),
        label: inCur.toLocaleString("en-US"),
        active: budgetNum > 0 && Math.abs(budgetNum - inCur) < inCur * 0.01,
      };
    }),
  };
}
 
function decorateItems(items: CartSimItem[], meta: Map<string, CalcMetaEntry>): OptionItem[] {
  return items.map((i) => {
    const key =
      i.type === "curve_split" ? `curve:${i.grams}`
      : i.type === "standard_bar" ? `standard:${i.grams}`
      : `accessory:${i.label.toLowerCase()}`;
    const m = meta.get(key);
    // Image: curve splits resolve from the static PRODUCT_CATALOG (the PWA's
    // OptionCard does this — `/images/1g.png` etc.), standard/accessory from the
    // DB product_media meta. (OptionCard.tsx:122-134.)
    const imageUrl = i.type === "curve_split"
      ? (PRODUCT_CATALOG.find((p) => p.g === i.grams)?.imageUrl ?? m?.imageUrl ?? null)
      : (m?.imageUrl ?? null);
    return {
      type: i.type, grams: i.grams, label: i.label, count: i.count,
      isAssay: i.isAssay, priceEach: i.priceEach,
      productId: m?.productId ?? null, imageUrl,
    };
  });
}
 
/** OptionCard's effective premium — gold-only, derived from prices vs spot
 *  (sim.blendedPremium is curve-only → wrong for standard/mixed). Mirrors
 *  OptionCard.tsx:94-102. */
function displayPremium(sim: AiCartSimulation): number {
  const goldGrams = sim.items
    .filter((i) => i.type === "curve_split" || i.type === "standard_bar")
    .reduce((s, i) => s + i.grams * i.count, 0);
  const goldOnly = sim.items
    .filter((i) => i.type === "curve_split" || i.type === "standard_bar")
    .reduce((s, i) => s + i.priceEach * i.count, 0);
  return goldGrams > 0 && sim.spotSgd > 0
    ? goldOnly / (sim.spotSgd * goldGrams) - 1
    : sim.blendedPremium;
}
 
function optionPayload(sim: AiCartSimulation, meta: Map<string, CalcMetaEntry>): OptionPayload {
  const totalSgd   = sim.goldSubtotalSgd + sim.accessorySubtotalSgd;
  const perGramSgd = sim.totalGrams > 0 ? sim.goldSubtotalSgd / sim.totalGrams : 0;
  const prem = displayPremium(sim);
  return {
    items:               decorateItems(sim.items, meta),
    totalGrams:          sim.totalGrams,
    totalSgd,
    perGramSgd,
    displayPremium:      prem,
    displayPremiumLabel: formatPremium(prem),   // formatted percent string (formatPremium)
    curveComparison:     sim.curveComparison ?? null,
  };
}
 
/**
 * Run the calculator simulation. Pure: same inputs → same output, no I/O.
 */
export function simulateCalculator(input: CalcInput, boot: CalcBootstrap): SimulateResponse {
  const mode = input.mode === "budget" ? "budget" : "grams";
  // The spec keeps separate `budget` + `grams` state (PWA preserves both across
  // a mode switch); `value` is a legacy single-field fallback.
  const rawValue = mode === "budget"
    ? Number(input.budget ?? input.value)
    : Number(input.grams ?? input.value);
  const currency = (input.currency ?? "SGD").toUpperCase();
 
  const spotSgd = boot.spotSgd;
  const spotTs  = boot.spotTs;
  const spotOzSgd = spotSgd * TROY_OZ;
  const spotUpdatedLabel = formatDateTime(spotTs);
  const ownedGrams = boot.ownedGrams > 0 ? boot.ownedGrams : 0;
  const goalGrams  = boot.goalGrams > 0 ? boot.goalGrams : 50;
 
  // Per-currency gram price → SGD↔currency rate (rate = SGD→currency).
  const curGram = boot.currencyGram[currency];
  const rate = curGram && curGram > 0 ? curGram / spotSgd : 1;
  const sliderMin = Math.round(100 * rate);
  const sliderMax = Math.round(100000 * rate);
  const presets = buildPresets(Number(input.grams), Number(input.budget), rate);
 
  // Not enough input to simulate — well-formed empty result so the surface can
  // show its input chrome (+ the live spot line) without an error state.
  if (!rawValue || isNaN(rawValue) || rawValue <= 0 || (mode === "grams" && rawValue < 1)) {
    return {
      spotSgd, spotOzSgd, spotTs, spotUpdatedLabel,
      hasResult: false, tooLow: false, summary: null, comparisonLine: null,
      progress: { currentG: ownedGrams, targetG: goalGrams, ownedGrams, accumulatedLabel: null },
      options: { curve: null, standard: null }, standardNote: null,
      accumulationBudgetSgd: null,
      sliderMin, sliderMax, ...presets,
    };
  }
 
  // budget mode: rawValue is in the selected currency → SGD via the gram-price ratio.
  const valueSgd = mode === "budget"
    ? (curGram && curGram > 0 ? rawValue * (spotSgd / curGram) : rawValue)
    : rawValue;
 
  const premiumMap = toPremiumMap(boot.premiums);
  const meta = new Map<string, CalcMetaEntry>(boot.meta.map((m) => [m.key, m]));
  const ctxBase = {
    ninjaVanFeeSgd:   0,
    paymentMethods:   [] as PaymentMethodRow[],
    standardProducts: boot.standardProducts,
    accessories:      [] as AccessoryForAI[],
  };
 
  // Curve option — at the customer's actual cycle position (ownedGrams).
  const curveSim = executeSimulateCart(
    mode, valueSgd, spotSgd, premiumMap,
    { ...ctxBase, accumulatedCurveGrams: ownedGrams },
  );
  const curve = curveSim.items.length > 0 ? curveSim : null;
 
  // Standard option — best-matching standard bar(s).
  const std = pickStandardOption(mode, valueSgd, spotSgd, boot.standardProducts);
  const standardSim = std
    ? executeSimulateCart("items", 0, spotSgd, premiumMap, ctxBase, std.items)
    : null;
 
  // Summary stats track the CURVE option (page.tsx:219-239 `result`).
  const summary = curve
    ? {
        totalGrams:     curve.totalGrams,
        blendedPremium: curve.blendedPremium,
        premiumLabel:   formatPremium(curve.blendedPremium),   // formatted percent string (formatPremium)
        perGramSgd:     curve.totalGrams > 0 ? curve.goldSubtotalSgd / curve.totalGrams : 0,
        totalSgd:       curve.goldSubtotalSgd,
        unspentSgd:     mode === "budget" ? Math.max(0, valueSgd - curve.goldSubtotalSgd) : 0,
      }
    : null;
 
  // Curiosity comparison, keyed on the curve total. Picked DETERMINISTICALLY
  // (first option) — a reactive recompute must not flicker per keystroke.
  const comp = curve ? getComparisons(curve.goldSubtotalSgd) : null;
  const comparisonLine = comp && comp.length > 0 ? { spend: comp[0].spend, punchline: comp[0].punchline } : null;
 
  const purchaseGrams = curve?.totalGrams ?? 0;
  const accumulatedLabel = ownedGrams > 0
    ? `${ownedGrams.toFixed(2)}g accumulated + ${purchaseGrams.toFixed(2)}g from this purchase`
    : null;
 
  return {
    spotSgd,
    spotOzSgd,
    spotTs,
    spotUpdatedLabel,
    hasResult: !!(curve || standardSim),
    tooLow:    !curve && !standardSim,
    summary,
    comparisonLine,
    progress: {
      currentG: ownedGrams + purchaseGrams,
      targetG:  goalGrams,
      ownedGrams,
      accumulatedLabel,
    },
    options: {
      curve:    curve ? optionPayload(curve, meta) : null,
      standard: standardSim ? optionPayload(standardSim, meta) : null,
    },
    standardNote: std?.note ?? null,
    accumulationBudgetSgd: mode === "budget" ? valueSgd : (summary?.totalSgd ?? null),
    sliderMin,
    sliderMax,
    ...presets,
  };
}
