/**
 * The home page's "Popular bars" strip — which products customers actually buy.
 *
 * Standard bars only, ranked by REVENUE on paid, non-cancelled orders over the
 * last 90 days, so the strip follows demand without anyone curating it
 * (Mukesh 2026-10-30, home refresh step 4). Standard is ~72% of revenue and its
 * price is the same for every customer, so the strip's Add to cart is exact.
 * Revenue, not units: by units a 1g piece or a gift box outranks a 1oz bar.
 *
 * Read server-side through the same L1 TTL primitive as the carousel — a warm
 * instance pays nothing. Fails OPEN to an empty list: the strip then renders
 * nothing, and the home page never fails because of it.
 */
import { createClient } from "@supabase/supabase-js";
import { createTtlCache } from "@/lib/ttl-cache";
 
export const POPULAR_WINDOW_DAYS = 90;
export const POPULAR_MAX = 4;
 
export interface SoldRow {
  product_id: string | null;
  quantity: number | string | null;
  price_sgd: number | string | null;
  status: string | null;
}
 
/** Product ids by revenue (qty × unit price), most first; ties keep first-seen order. Pure. */
export function rankPopular(rows: readonly SoldRow[], max = POPULAR_MAX): string[] {
  const revenue = new Map<string, number>();
  for (const r of rows) {
    if (!r.product_id || r.status === "cancelled") continue;
    const line = Number(r.quantity) * Number(r.price_sgd);
    if (!Number.isFinite(line) || line <= 0) continue;
    revenue.set(r.product_id, (revenue.get(r.product_id) ?? 0) + line);
  }
  return [...revenue.entries()].sort((a, b) => b[1] - a[1]).slice(0, max).map(([id]) => id);
}
 
const cache = createTtlCache<string[]>(3_600_000); // hourly — demand moves slowly
