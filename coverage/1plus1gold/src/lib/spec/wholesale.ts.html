/**
 * SDUI spec for the Wholesale surface (Phase 6.x, contract §24).
 *
 * Wholesale is the Shop's Standard section with three deltas (everything else
 * REUSES the Shop substrate — `goldCard`/`productRow`/`productList`/`filterChips`/
 * `checkbox`/`viewToggle`, all exported from ./shop, and the same on-device
 * `catalog_local` pricing kernel):
 *   1. data_source = /api/wholesale/products (role-gated: the entry only shows
 *      when bootstrap.wholesaleEnabled; the route 403s otherwise). The route
 *      normalises each product to product_type:"standard" + standard_premium_tiers
 *      (= wholesale_premium_tiers) so engine.catalogView prices it unchanged.
 *   2. MIN-QTY enforcement — the cards/rows pass `enforce_min_qty:true` so
 *      product_qty_line pre-fills to the product's minQty and blocks Add below it.
 *   3. Wholesale header + footer copy; no curve/accessory/calculator/disclaimer.
 *
 * Port-diff: see the §24 table in docs/sdui-pwa-port-contract.md. Source:
 * src/app/wholesale/page.tsx. Contract §23 (shared substrate) + §24.
 */
 
import type { SDUISpec, SDUINode } from "./types";
import { text, goldCard, productRow, productList, filterChips, checkbox, viewToggle } from "./shop";
 
export function wholesaleSpec(): SDUISpec {
  return {
    surface: "wholesale",
    spec_version: 1,
    data_source: "/api/wholesale/products",
    data_source_method: "catalog_local",
    data_source_body_keys: ["gramsFilter", "refineryFilter", "sort", "groupByRefinery", "showSoldOut"],
    state_schema: {
      view:            { type: "string",  default: "row" },
      sort:            { type: "string",  default: "weight-asc" },
      groupByRefinery: { type: "boolean", default: false },
      showSoldOut:     { type: "boolean", default: false },
      gramsFilter:     { type: "array",   default: [] },
      refineryFilter:  { type: "array",   default: [] },
    },
    root: {
      type: "scroll",
      children: [{
        type: "vstack", props: { spacing: 24, padding: 16, alignment: "leading" },
        children: [
 
          // ── Header ─────────────────────────────────────────────────────
          { type: "vstack", props: { spacing: 8, alignment: "leading" }, children: [
            text("Wholesale Catalogue", "cinzel:12", "gold_mid", { tracking: 4, text_case: "upper" }),
            text("Wholesale", "cinzel.bold:30", "cream"),
            text("Volume-tier pricing on standard products. Each line has its own minimum order quantity; price drops at each tier breakpoint. Checkout flow is the same as retail.",
              "montserrat:14", "muted"),
          ] },
 
          // ── Empty state (no wholesale products) ────────────────────────
          { type: "vstack", visible_when: "data.gramsOptions.length == 0",
            props: { spacing: 12, alignment: "center", padding: 24, full_width: true }, children: [
              { type: "icon", props: { name: "shippingbox", size: 28, color: "gold_mid" } },
              text("No wholesale products available yet.", "montserrat:14", "offwhite", { alignment: "center" }),
            ] },
 
          // ── Controls + product list (when products exist) ──────────────
          { type: "vstack", visible_when: "data.gramsOptions.length > 0",
            props: { spacing: 16, alignment: "leading" }, children: [
              // Weight + refinery filter chips (only when >1 option).
              text("Weight", "montserrat:12", "muted", { visible_when: "data.gramsOptions.length > 1" }),
              filterChips("data.gramsOptions", "gramsFilter"),
              text("Refinery", "montserrat:12", "muted", { visible_when: "data.refineryOptions.length > 1" }),
              filterChips("data.refineryOptions", "refineryFilter"),
              // Sort row.
              { type: "hstack", props: { spacing: 8, alignment: "center" }, children: [
                { type: "icon", props: { name: "arrow.up.arrow.down", size: 12, color: "muted" } },
                text("Sort", "montserrat:12", "muted"),
                { type: "option_select", props: { bind: "sort", style: "menu", options: [
                  { value: "default", label: "Default (curated)" },
                  { value: "weight-asc", label: "Weight (light → heavy)" },
                  { value: "weight-desc", label: "Weight (heavy → light)" },
                  { value: "refinery", label: "Refinery (A–Z)" },
                ] } },
              ] },
              // Group + sold-out checkboxes, view toggle pushed right (PWA ml-auto).
              { type: "hstack", props: { spacing: 16, alignment: "center" }, children: [
                checkbox("groupByRefinery", "Group by Refinery"),
                checkbox("showSoldOut", "Show sold-out"),
                { type: "spacer", props: {} },
                viewToggle(),
              ] },
              // Grouped products — sub-header per group ("" = no header). All
              // wholesale items are standard; enforce_min_qty on the cards/rows.
              { type: "repeater", props: { data_path: "data.standardGroups", item_id_field: "key", spacing: 24 }, children: [
                { type: "vstack", props: { spacing: 12, alignment: "leading" }, children: [
                  text("{item.key}", "cinzel.semibold:16", "gold", { visible_when: "item.key != ''" }),
                  productList("item.products", goldCard(256, true), productRow(false, true)),
                ] },
              ] },
            ] },
 
          // ── Footer note ────────────────────────────────────────────────
          { type: "vstack", props: { spacing: 8, alignment: "center", full_width: true }, children: [
            text("Wholesale pricing is per-order, locked at the moment you click Add. The cart locks the spot price; final invoice is in SGD.",
              "montserrat:12", "muted", { alignment: "center" }),
            { type: "button", props: { label_text: "Need help? Reach us.", label_font: "montserrat:12", label_color: "gold_mid",
              bg: "transparent", padding_v: 0, action: { kind: "navigate", url: "/contact" } } },
          ] },
        ],
      }],
    },
  };
}
