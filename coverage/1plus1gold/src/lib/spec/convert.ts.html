function text(content: string, font: string, color: string, extra: Record<string, unknown> = {}): SDUINode {
  const { visible_when, ...rest } = extra as { visible_when?: string };
  return { type: "text", ...(visible_when ? { visible_when } : {}), props: { content, font, color, ...rest } };
}
 
/** Section header — PWA `font-cinzel text-sm semibold uppercase gold`. */

  return text(t, "cinzel.semibold:14", "gold", { tracking: 1.5, text_case: "upper" });
}

/** One Combibar-denomination stepper TILE (label-on-top, −count＋ row, max-N
 *  subtitle). `key` = state counter; `maxPath` = the live ceiling token. */
function stepperTile(label: string, key: string, maxPath: string): SDUINode {
  const btn = (glyph: string, step: number, bound: Record<string, unknown>): SDUINode => ({
    type: "button",
    props: {
      leading_icon: glyph, label_color: "offwhite", bg: "rule",
      corner_radius: 6, padding_h: 10, padding_v: 8, full_width: false,
      action: { kind: "state_step", key, step, ...bound } as unknown as SDUIAction,
    },
  });
  return {
    type: "vstack",
    props: { bg: "bg", corner_radius: 8, padding: 12, spacing: 8, alignment: "center", max_width: "infinity" },
    children: [
      text(label, "montserrat:12", "muted", { alignment: "center" }),
      {
        type: "hstack",
        props: { justify: "between", alignment: "center" },
        children: [
          btn("minus", -1, { min: 0 }),
          text(`{state.${key}}`, "cinzel.bold:20", "gold"),
          btn("plus", 1, { max: maxPath }),
        ],
      },
      text(`max ${maxPath}`, "montserrat:10", "muted", { alignment: "center" }),
    ],
  };
}

export function convertSpec(): SDUISpec {
  return {
    surface: "convert",
    spec_version: 1,
    data_source: "/api/conversions/form",
    data_source_method: "POST",
    data_source_body_keys: ["target50g", "target100g", "sortBy"],
    // Reactive-POST surface, but the payment_method_picker needs the on-device
    // engine to compute the HitPay processing fee (engine.paymentFee). Without
    // this the picker shows "No processing fee" while the server charges it
    // (device 2026-06-06). *_local surfaces get the engine implicitly; this opts
    // a POST surface in. See SDUISpec.needsCalcEngine.
    needs_calc_engine: true,
    state_schema: {
      target50g:      { type: "number", default: 0 },
      target100g:     { type: "number", default: 0 },
      sortBy:         { type: "string", default: "date_asc" },
      // M3 — payment + submit (payment_method_picker writes processingFee/
      // grandTotal; address_picker writes selectedAddrId; the native button
      // sets convConfirmed on HitPay success).
      paymentMethod:  { type: "string", default: "" },
      processingFee:  { type: "number", default: 0 },
      grandTotal:     { type: "number", default: 0 },
      selectedAddrId: { type: "string", default: "" },
      convConfirmed:  { type: "boolean", default: false },
      confirmedConvRef: { type: "string", default: "" },   // PWA conversion ref (place_conversion_button stamps first-8-upper)
      // Native HitPay-return result (place_conversion_button sets these from pollHitpay
      // after the SafariView returns) — iOS port of the web /payment/confirming failed
      // + timeout states, so a non-success return is NATIVE not the web page. 2026-06-07.
      convFailed:     { type: "boolean", default: false },
