/**
 * Shared SDUI spec composites for auth surfaces.
 *
 * Pattern policy:
 *   * If the same SDUI sub-tree appears in 2+ specs (sign-in's WhatsApp
 *     OTP + email OTP + 2fa TOTP), it lives here.
 *   * Each helper has a fixed shape — callers pass only the small set
 *     of varying props (the `bind` key, label text). This keeps the
 *     visual + interaction language identical across surfaces so a
 *     fix in one place fixes all.
 *
 * Contract row references:
 *   * otpField — see PWA→SDUI port contract row C-OTP1 (6-digit code
 *     entry field). Padding / font / tracking / text_align are exact
 *     and should NOT be diverged per-call-site.
 */
 
/**
 * 6-digit OTP / TOTP entry field used by:
 *   * /auth/sign-in  — WhatsApp OTP stage
 *   * /auth/sign-in  — email OTP stage
 *   * /auth/2fa      — TOTP code
 *
 * Visual contract: large monospace-ish glyphs, center-aligned with
 * generous tracking so each digit reads as its own slot. Mirrors PWA's
 * `text-2xl tracking-[0.5em] text-center` (rough numeric equivalents
 * in SDUI props: font 22pt, tracking 8pt, padding_v 16pt).
 *
 * @param bind  the state key the field writes back to (typically "otp").
 *              The spec's state_schema MUST declare this key with
 *              type "string", default "".
 */
export function otpField(bind: string = "otp") {
  return {
    type: "text_field",
    props: {
      bind,
      placeholder: "••••••",
      keyboard: "number",
      // PWA: text-2xl (24px) tracking-[0.5em] (=0.5×24=12px) text-center.
      // iOS renders this via CenteredUITextField (full-width, .kern=tracking)
      // so the caret centres + digits spread exactly like PWA. (2026-05-31.)
      font: "montserrat:24",
      color: "white",
      text_align: "center",
      tracking: 12,
      padding_v: 16,
      padding_h: 12,
      submit_label: "done",
      max_length: 6,
      // KB1 — surfaces the SMS code from Messages above the keyboard (and the
      // QuickType autofill chip). The single highest-value autofill hint.
      content_type: "one_time_code",
      autofocus: true,
    },
  };
}
 
/**
 * Red error band that surfaces when the spec's `error` state key is
 * a non-empty string. Visual contract: light-red bg, red text,
 * 8pt rounded corners, 12pt horizontal / 8pt vertical padding.
 *
 * Mirrors PWA's `bg-[rgba(224,80,80,0.08)] text-[#E05050] px-3 py-2 rounded`.
 *
 * Used by sign-in (WhatsApp + email stages) + 2fa.
 */
export function errorBand() {
  return {
    type: "text",
    visible_when: 'state.error != ""',
    props: {
      content: "{state.error}",
      font: "montserrat:12",
      color: "hex:E05050",
      bg: "hex:E05050@8",
      corner_radius: 6,
      padding_h: 12,
      padding_v: 8,
      max_width: "infinity",
    },
  };
}
 
/**
 * The stacked 1+1/GOLD logo. Used by every unauthenticated surface
 * (sign-in + 2fa + future password-reset).
 *
 * Emits a single `logo_mark` node so iOS renders the CANONICAL `LogoMarkView`
 * (the one used in nav / account hero / etc.) — NOT a second hand-rolled copy
 * via text nodes. The prior text-node version (cinzel.bold 36 "1" + cormorant
 * 28 "+" + GOLD) rendered invisibly on device and was the same drift-prone
 * parallel-implementation anti-pattern the contract warns against. Size 40
 * matches the PWA auth pages' `<LogoMark size={40}>`. (Changed 2026-05-31.)
 */
export function stackedLogo(id?: string) {
  return {
    type: "logo_mark",
    props: { size: 40, ...(id ? { id } : {}) },
  };
}
