/**
 * SDUI spec for the Account → Security surface (Q-Security).
 *
 * Anchored to the PWA renderSecurity() (src/app/account/page.tsx:2649-2929) +
 * SignInMethodsCard / DeleteAccountSection. SIX cards in order:
 *   1. Sign-in methods   — pure SDUI (data_source repeater + unlink + link)
 *   2. Identity (KYC)    — SDUI status/messaging + kyc_verify WebView node (Path B)
 *   3. Transaction Limit — pure SDUI (bool_toggle SW1)
 *   4. Two-Factor Auth   — native escape-hatch node (totp_card) — CoreImage QR
 *   5. Passkeys          — native escape-hatch node (passkey_card) — ASAuthorization
 *   6. Delete account    — pure SDUI (confirm text_field + delete_account action)
 *
 * Cards 4+5 stay native (no shared web impl — web uses supabase.auth.mfa /
 * @simplewebauthn; iOS uses CoreImage / ASAuthorization). The push_opt_in
 * precedent: inherently-native capability → native card embedded via a node.
 *
 * data_source = /api/account/security/overview (identities + linked booleans +
 * kyc + self_limit). The 2FA + passkey nodes self-fetch (mfa.listFactors /
 * /api/account/passkeys), so they're not in the aggregator.
 *
 * PORT-DIFF + Contract column: see docs/sdui-pwa-port-contract.md §17. Native
 * 2FA/passkey copy is NOT byte-identical to the PWA (different SDKs, no shared
 * source) — that's the sanctioned native-escape-hatch carve-out, not a drift.
 *
 * Accepted divergences (documented): (a) unlink confirm is a BANNER above the
 * list (state.confirm_id), not in-row — SDUI predicates can't match a per-row
 * id against state. (b) passkey ROW shows ✓ only (no in-row "Manage →"); the
 * Add-Passkey affordance lives in "Add another" + scrolls to the passkey card
 * (scroll_anchors). (c) KYC status label/color computed server-side (overview).
 * "Change WhatsApp"/"Add WhatsApp/Email" route in-app to /account/profile
 * (InternalLinkRouter → Profile sheet), not the browser.
 */
 
import type { SDUISpec, SDUINode, SDUIAction } from "./types";
import { KYC_THRESHOLD_SGD } from "@/lib/kyc/constants";
 
/** Formatted KYC gate amount, e.g. "20,000" — server-driven copy. */
const KYC_AMT = KYC_THRESHOLD_SGD.toLocaleString();
 
const OVERVIEW_EP   = "/api/account/security/overview";

const text = (content: string, font: string, color: string, extra: Record<string, unknown> = {}): SDUINode => ({
  type: "text", props: { content, font, color, ...extra },
});
 
/** Standard card: cinzel gold title + optional muted desc + body. */
function card(title: string, desc: string | null, children: SDUINode[]): SDUINode {
  const head: SDUINode[] = [text(title, "cinzel.semibold:16", "gold", { padding_b: desc ? 4 : 8 })];
  if (desc) head.push(text(desc, "montserrat:12", "muted", { padding_b: 8 }));
  return {
    type: "card",
    props: { padding: 24, corner_radius: 12, bg: "bg_card", spacing: 8, alignment: "leading" },
    children: [...head, ...children],
  };
}

            type: "hstack",
            visible_when: "item.removable == true",
            props: { spacing: 8, alignment: "center" },

                    { kind: "state_set", key: "confirm_label", value: "{item.label}" },
                    { kind: "state_set", key: "error", value: "" },
                  ] },
                },
              },
            ],
          },
          // Phone → Change (routes to Profile).
          {
            type: "hstack",
            visible_when: 'item.removable == false && item.kind == "phone"',
            props: { spacing: 8, alignment: "center" },
            children: [
              check,
              { type: "button", props: {
                label_text: "Change", label_font: "montserrat:11", label_color: "gold", trailing_icon: "chevron.right",
                bg: "transparent", full_width: false, padding_h: 8, padding_v: 4,
                action: { kind: "navigate", url: "/account/profile" } } },
            ],
          },
          // Everything else not removable + not phone → ✓ only.
          {
            type: "hstack",
            visible_when: 'item.removable == false && item.kind != "phone"',
            props: { spacing: 8 },
            children: [check],
          },
        ],
      },
    ],
  };
 
  const addBtn = (provider: string, label: string, asset: string | null, sf: string | null, action: SDUIAction): SDUINode => ({
    type: "button",
    visible_when: `data.linked.${provider} == false`,

    kind: "native", name: "link_oauth_identity", params: { provider },

  return card(
    "Sign-in methods",

    [
      // Unlink confirm banner (divergence: banner, not in-row).
      {
        type: "card",
        visible_when: 'state.confirm_id != ""',
        props: { padding: 12, corner_radius: 8, bg: "hex:E05050@8", alignment: "leading", spacing: 8 },
        children: [
          { type: "rich_text", props: { font: "montserrat:12", color: "offwhite", segments: [
            { kind: "text", content: "Remove " },
            { kind: "text", content: "{state.confirm_label}", color: "offwhite" },
            { kind: "text", content: " as a sign-in method? You can add it back later." },
          ] } },
          {
            type: "hstack", props: { spacing: 12 },
            children: [
              { type: "button", props: {
                label_text: "Cancel", label_font: "montserrat:12", label_color: "muted",

                label_text: "Yes, remove", label_font: "montserrat.semibold:12", label_color: "white",

                action: { kind: "api_call", method: "DELETE", url: `${IDENTITIES_EP}?identity_id={state.confirm_id}`,

                  on_failure: [{ kind: "state_set", key: "confirm_id", value: "" }] } } },
            ],
          },
        ],
      },
      { type: "text", visible_when: 'state.error != ""', props: {
        content: "{state.error}", font: "montserrat:12", color: "hex:E05050",
        bg: "hex:E05050@8", corner_radius: 6, padding_h: 12, padding_v: 8, max_width: "infinity" } },

      {
        type: "hstack", props: { spacing: 8 },
        children: [
          addBtn("apple", "Apple", null, "applelogo", linkAction("apple")),
          addBtn("google", "Google", "GoogleLogoColor", null, linkAction("google")),
          addBtn("phone", "WhatsApp", null, "phone.fill", { kind: "navigate", url: "/account/profile" }),
          addBtn("email", "Email", null, "envelope.fill", { kind: "navigate", url: "/account/profile" }),

        content: "All sign-in methods are linked. Your account is as portable as it gets.",

// ── Card 2: Identity Verification (KYC) ──────────────────────────────────────
function kycCard(): SDUINode {
  return card("Identity Verification", null, [
    { type: "rich_text", props: { font: "montserrat:12", color: "muted", segments: [
      { kind: "text", content: "Required for orders above " },
      { kind: "text", content: `SGD ${KYC_AMT}`, color: "gold" },
      { kind: "text", content: ". One-time — valid for 2+ years." },
    ] } },
    // Status label (server-computed label + color_token).
    text("{data.kyc.label}", "montserrat.semibold:14", "{data.kyc.color_token}"),
    // Valid-until (approved only).
    { type: "rich_text", visible_when: 'data.kyc.status == "approved" && data.kyc.expires_at != null',
      props: { font: "montserrat:12", color: "muted", segments: [
        { kind: "text", content: "Valid until " },
        { kind: "date", value: "{data.kyc.expires_at}", style: "d_mon_y" },
      ] } },
 
    // Fatal rejection — contact us, no retry.
    {
      type: "vstack", visible_when: "data.kyc.fatal == true",
      props: { spacing: 12, alignment: "leading", padding_t: 8 },
      children: [
        text("Your verification could not be completed and additional documents will not resolve it. Please contact us so we can review your account.",
          "montserrat:12", "hex:e57373"),
        { type: "button", props: {
          label_text: "Contact us", label_font: "montserrat.semibold:12", label_color: "bg",
          bg: "gold", full_width: false, padding_h: 16, padding_v: 8,
          action: { kind: "navigate", url: "/contact" } } },
      ],
    },
 
    // Retryable states — message + verify button. Hidden when approved or fatal.
    {
      type: "vstack",
      visible_when: 'data.kyc.fatal == false && data.kyc.status != "approved"',
      props: { spacing: 8, alignment: "leading", padding_t: 8 },
      children: [
        { type: "text", visible_when: 'data.kyc.status == "rejected"', props: {
          content: "Your previous verification was unsuccessful. Please try again with a clear, valid ID document.",
          font: "montserrat:12", color: "hex:e57373" } },
        { type: "text", visible_when: 'data.kyc.status == "on_hold"', props: {
          content: "We need additional information to complete your verification.", font: "montserrat:12", color: "gold" } },
        { type: "text", visible_when: 'data.kyc.status == "on_hold" && data.kyc.hold_reason != null', props: {
          content: "Reviewer note: {data.kyc.hold_reason}", font: "montserrat:12", color: "muted" } },
        { type: "text", visible_when: 'data.kyc.status == "expired"', props: {
          content: `Your verification has expired. Please complete a new identity check to continue placing orders above S$${KYC_AMT}.`,
          font: "montserrat:12", color: "muted" } },
        { type: "text", visible_when: 'data.kyc.status == "pending"', props: {
          content: "Your verification is under review. If you did not finish submitting your documents, you can continue below.",
          font: "montserrat:12", color: "muted" } },
        // Verify launch — WebView node. resume for pending/on_hold; label varies.
        { type: "kyc_verify", visible_when: 'data.kyc.status == "not_required" || data.kyc.status == "expired" || data.kyc.status == "rejected"',
          props: { label: "Start verification", resume: false } },
        { type: "kyc_verify", visible_when: 'data.kyc.status == "pending" || data.kyc.status == "on_hold"',
          props: { label: "Continue verification", resume: true } },
      ],
    },
 
    // Help footer (always).
    { type: "rich_text", props: { font: "montserrat:11", color: "muted", padding_t: 8, segments: [
      { kind: "text", content: "Stuck or need help with verification? " },
      { kind: "text", content: "Contact us", color: "gold" },
      { kind: "text", content: "." },
    ], action: { kind: "navigate", url: "/contact" } } },
  ]);
}
 
// ── Card 3: Transaction Limit (self-limit) ───────────────────────────────────
function selfLimitCard(): SDUINode {
  return card(
    "Transaction Limit",
    null,
    [
      { type: "rich_text", props: { font: "montserrat:12", color: "muted", segments: [
        { kind: "text", content: "Restrict your account to orders below " },

      ] } },
      {
        type: "hstack",
        props: { spacing: 12, alignment: "center", padding_t: 4 },
        children: [
          {
            type: "vstack", props: { spacing: 0, alignment: "leading", max_width: "infinity" },
            children: [
              { type: "text", visible_when: "data.self_limit_below_5k == true", props: {
                content: `Active — limit: SGD ${KYC_AMT}`, font: "montserrat.semibold:12", color: "success" } },
              { type: "text", visible_when: "data.self_limit_below_5k == false", props: {
                content: "Off — no limit set", font: "montserrat.semibold:12", color: "muted" } },
            ],
          },
          {
            type: "bool_toggle",
            props: {
              bind: "self_limit", on_when: "data.self_limit_below_5k == true",
              endpoint: SELF_LIMIT_EP, body_on: { enabled: true }, body_off: { enabled: false },
            },
          },
        ],
      },
    ],
  );
}
