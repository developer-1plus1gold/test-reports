 * data_source = /api/pricing/overview (answerValue, shared derivePricingDisplay).
 *
 * PORT-DIFF (element-level, every string) + Contract: docs/sdui-pwa-port-contract.md
 * §20. Anchored to our-pricing/page.tsx (verbatim copy). No element dropped
 * (no-deferrals rule, Mukesh 2026-06-03).
 *
 * Device-polish pass 2026-06-03 (6 findings): line-height (line_spacing on body
 * text — rich_text already defaults 4); section rhythm (card spacing 18 + tight
 * header groups so h2→subtext stays close while body blocks breathe like the
 * PWA mb-6/mb-8); single-column tiles on phone (PWA grid-cols-1 sm:grid-cols-2
 * → vstack, NOT a 2-col grid); §3 chart→math-note divider (PWA mt-5 pt-5
 * border-top); lazy chart panels (#6, native SDUIWebPanel). Embeds lost their
 * px-4 so the chart fills the card width (#3).
 *
 * Round 2 (2026-06-03 device): the pill ↘ is forced to EMOJI presentation
 * (U+FE0F) to MATCH the PWA, which renders the bare glyph as the greyish-blue

 * ↓/→ stay text-presentation (not flagged). §3 math disclosure → chevron:"leading"

 * scroll node opts into `scroll_anchors`, and a `state_set _scroll_target =
 * "premium-vs-median"` scrolls to the §9 card (device-verified for Security's

const EP = "/api/pricing/overview";

// Body text carries leading-relaxed line-height (PWA). rich_text already
// defaults line_spacing:4; plain `text` defaults 0 → it read tight on device
// (finding #2). Default 4 here; single-line headers/values are unaffected
// (line spacing only changes multi-line wrapping). Override via `extra`.
const text = (content: string, font: string, color: string, extra: Record<string, unknown> = {}): SDUINode => ({
  type: "text", props: { content, font, color, line_spacing: 4, ...extra },
});
 
const h2 = (content: string, color = "gold"): SDUINode =>
  text(content, "cinzel.semibold:20", color);
 
/** A 1px hairline rule (PWA `border-top`/`border-b`). */
const rule = (padding_v = 4): SDUINode => ({
  type: "divider", props: { color: "rule", padding_v },
});
 
/**
 * A `.card rounded-xl p-8` section wrapper. spacing:18 gives the PWA body-block
 * rhythm (mb-6/mb-8 between elements); tight header pairs are grouped in an
 * inner vstack so h2→subtext stays close (finding #2 — uniform spacing alone
 * over-loosened the headers).
 */
const sectionCard = (children: SDUINode[]): SDUINode => ({
  type: "card",
  props: { stroke: "rule", bg: "bg_card", corner_radius: 12, padding: 24, alignment: "leading", spacing: 18 },
  children,
});
 
/** A tight header cluster (h2 + subtext) — stays close while the card spacing
 *  handles the larger gap down to the content (matches PWA mb-1/mb-2 + mb-6). */
const headerGroup = (children: SDUINode[], spacing = 6): SDUINode => ({
  type: "vstack", props: { spacing, alignment: "leading", full_width: true }, children,
});
 
/** A web_panel chart embed (height = fallback; embed reports real surfaceHeight).
 *  `lazy` defers the WebView mount until the panel scrolls into the native
 *  viewport, so the reveal animation auto-plays on-scroll like the PWA's
 *  IntersectionObserver instead of off-screen on first paint (finding #6). */
const chart = (slug: string, fallbackHeight: number, lazy = false): SDUINode => ({
  type: "web_panel",
  props: { path: `/embed/our-pricing/${slug}`, height: fallbackHeight, ...(lazy ? { lazy: true } : {}) },
});
 
/** A "label … value" row (Repacked / Standard comparison cards). */
const compRow = (label: string, value: string, valueColor: string): SDUINode => ({
  type: "hstack",
  props: { justify: "between", spacing: 16, alignment: "top" },
  children: [
    text(label, "montserrat:12", "muted", { fixed_size_h: true }),
    text(value, "montserrat:12", valueColor, { alignment: "trailing" }),
  ],
});
 
/** A bullet row: "· item" (dot in accent colour, copy in offwhite). */
const bullet = (item: string, dot: string): SDUINode => ({
  type: "hstack",
  props: { spacing: 8, alignment: "top" },
  children: [
    text("·", "montserrat:12", dot),
    text(item, "montserrat:12", "offwhite"),
  ],
});

  const accent = highlight ? "gold" : ORANGE_TEXT;
  return {
    type: "card",
