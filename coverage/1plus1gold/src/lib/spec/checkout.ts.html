/** A numbered collapsible step. Native `step_accordion` renders the header
 *  (badge + title + collapsed summary + chevron, tappable→reopen when done) and
 *  its children as the body when active. `done`/`active`/`locked` derive from
 *  the numeric `state.step` (2|3|4) inside the node, given `step_index`. */
const stepAccordion = (index: number, title: string, body: SDUINode[], opts: { collapsed_summary?: string } = {}): SDUINode => ({
  type: "step_accordion",
  props: {
    step_index: index,
    title,
    ...(opts.collapsed_summary ? { collapsed_summary: opts.collapsed_summary } : {}),
    reopen_action: { kind: "native", name: "checkout_set_step", params: { step: String(index) } },
  },
  children: body,
});
