/**
 * SDUI spec for the public Certificate-of-Authenticity — `/verify/[serial]`.
 *
 * Replaces the native `VerifyView` (a triple-stub: data-load 404'd because
 * `/api/coa/verify` never existed; "Sign in & Accept" was an empty closure; the
 * My-Account / Continue-Shopping CTAs were NEVER rendered). This surface is the
 * faithful SDUI port of the WEB benchmark (the native view had drifted — its
 * market-value strip was a single "spot × weight × premium" line, not the web's
 * Shop-price / We-buy-back two-tile strip).
 *
 * Data: GET /api/coa/verify?serial={state.serial} — the JSON twin of the web
 * server page, sharing `lib/verify-core` (one impl, no drift). The serial is
 * threaded in via SDUIView's `seed` (App.swift route → seed serial+auto_claim);
 * the GET source is interpolated with `{state.serial}` before the fetch. Gift
 * PII is auth-gated server-side (giver name/last-4/message are null for an
 * anonymous viewer); `gift_claim_card` renders the PII-free teaser then.
 *
 * FULL PORT-DIFF vs the web (src/app/verify/[serial]/page.tsx + VerifyPriceStrip
 * + GiftClaimBlock). Contract column cites the existing row or NEW.
 *
 * | Element                | PWA (page.tsx / strip / gift)                         | iOS SDUI                                                     | Contract |
 * |------------------------|-------------------------------------------------------|-------------------------------------------------------------|----------|
 * | Outer card             | max-w-lg rounded-2xl bg-card border gold-dim          | card corner 16 stroke gold_dim bg bg_card padding 0          | C-card   |
 * | Header LogoMark        | <LogoMark size={28}> centered                         | logo_mark size 28 (centered; not inline)                    | I/LogoMark |
 * | Header eyebrow         | "Certificate of Authenticity" cinzel 10 0.4em upper   | text cinzel.bold:10 muted tracking 4 upper center           | T        |
 * | Header divider         | border-bottom rule                                    | divider rule (full width)                                   | C2/L     |
 * | Verified pill          | ShieldCheck 18 success + cinzel 14 semibold success   | hstack center [icon checkmark.shield.fill 18 success + text cinzel.semibold:14 success] | I2/T |
 * | Detail row label       | montserrat 10 tracking-widest upper muted shrink-0    | kv_row label_caps:true (10 upper tracked muted)             | P4 NOTE UPDATE (label_caps) |
 * | Detail row value       | montserrat 14 offwhite text-right (mono for serial)   | kv_row value (offwhite) value_mono for serial/parent        | P4 NOTE UPDATE (value_mono) |
 * | Rows                   | Serial(mono)/Packed by/Content/Purity/[Dim]/Source/[Parent(mono)]/Date | kv_row × (server `purity_label`/`assigned_at_label`; conditional dims/parent via visible_when) | P4 / server-fmt |
 * | Strip title            | "Today's market value" cinzel 10 0.4em upper muted     | text cinzel.bold:10 muted tracking 4 upper center           | T        |
 * | Strip 2-col grid       | grid grid-cols-2 gap-4                                 | grid columns 2 spacing 16                                    | Grid     |
 * | Price tile             | rounded-lg px-4 py-3 center bg=bg border rule         | card bg bg stroke rule corner 8 center padding_h 16 padding_v 12 | C-card |
 * | Tile label             | "Shop price"/"We buy back" montserrat 10 widest upper  | text montserrat:10 muted upper tracking 1 center            | T        |
 * | Tile value             | cinzel 16 semibold gold/gold-mid Flag+fmt OR "Loading…"| currency_amount cinzel.semibold:16 (visible_when !=null) + "Loading…" text when null | Currency |
 * | Tile subline           | "spot +X.X%" / "at spot" montserrat 9 muted           | text {data.shop_premium_label}/{buyback_premium_label} (server-fmt) montserrat:9 muted | server-fmt |
 * | Strip disclosure       | 4-variant montserrat 9 muted (spot-null/stale/curve/std)| text {data.price_disclosure} (server 4-branch) montserrat:9 muted center | server-fmt |
 * | Not-found pill         | ShieldAlert 18 #E05050 + cinzel 14 semibold #E05050    | hstack center [icon exclamationmark.shield.fill 18 hex:E05050 + text cinzel.semibold:14 hex:E05050] | I2/T |
 * | Not-found body         | montserrat 12 offwhite                                 | text montserrat:12 offwhite center line_spacing 3           | T        |
 * | Not-found chip         | rounded-lg E05050@06 bg / @25 border, info + mono 11   | hstack bg hex:E05050@10 stroke hex:E05050@40 [info.circle 12 + text {state.serial} 11] (mono→regular: no mono font token) | L (chip) |
 * | Gift block (authed)    | message box + "A gift from X" + Accept Gift             | gift_claim_card native — full block (message/giver/Accept)  | NEW gift_claim_card |
 * | Gift block (anon)      | "A gift from X…" + Sign in & Accept (web shows PII)     | gift_claim_card teaser: "This card was sent as a gift" + Sign in & Accept, NO PII (auth-gate) | NEW gift_claim_card |
 * | CTA My Account         | <Link/account> bg-card border gold-dim LogoMark+text   | button composed[tabswitch account, dismiss] children[logo_mark 18 inline + text gold] | N/L |
 * | CTA Continue Shopping  | <Link/shop> transparent border gold-dim ShoppingBag    | button composed[tabswitch shop, dismiss] children[icon bag 15 + text gold] | N/L |
 * | Footer note            | montserrat 10 muted center (contact link)             | text montserrat:10 muted center (plain — link inlined as text) | T |
 *
 * Deferred/divergent (surfaced): not-found chip is regular-weight (no mono font
 * token); the footer "1plus1gold.com/contact" renders as plain text (the web
 * <Link> is a minor affordance). Print mode is web/admin-only — N/A on iOS.
 */
 
import type { SDUISpec, SDUINode, SDUIAction } from "./types";
import { text } from "./shop";
 
const SECTION_PAD_H = 20;
 
/** Full-width 1pt rule between sections (the web's border-top/border-bottom). */
const sectionDivider: SDUINode = { type: "divider", props: { color: "rule", thickness: 1 } };
 
/** A detail row — kv_row with the verify label style (10 upper tracked muted)
 *  + optional monospaced value, divider under every row (web border-b on all). */
function row(
  label: string,
  value: string,
  opts: { mono?: boolean; visible_when?: string } = {},
): SDUINode {
  return {
    type: "kv_row",
    ...(opts.visible_when ? { visible_when: opts.visible_when } : {}),
    props: { label, value, label_caps: true, value_mono: opts.mono ?? false, divider_below: true },
  };
}
 
/** One market-value tile (Shop price / We buy back). */
function priceTile(opts: {
  label: string;
  pricePath: string;
  premiumPath: string;
  valueColor: string;
}): SDUINode {
  return {
    type: "card",
    // top_accent:"none" — a rule-stroked card auto-adds a gold top bar
    // (SDUICard), but the web price tile is a plain bordered box.
    props: { bg: "bg", stroke: "rule", top_accent: "none", corner_radius: 8, padding_h: 16, padding_v: 12, alignment: "center", spacing: 4 },
    children: [
      text(opts.label, "montserrat:10", "muted", { text_case: "upper", tracking: 1, alignment: "center" }),
      // Value when spot resolved; "Loading…" otherwise (web `loading || price==null`).
      {
        type: "currency_amount",
        visible_when: `${opts.pricePath} != null`,
        props: { sgd_value: `{${opts.pricePath}}`, font: "cinzel.semibold:16", color: opts.valueColor, flag_height: 13 },
      },
      text("Loading…", "montserrat:12", "muted", { visible_when: `${opts.pricePath} == null`, alignment: "center" }),
      text(`{${opts.premiumPath}}`, "montserrat:9", "muted", { alignment: "center" }),
    ],
  };
}
 
/** A CTA button (My Account / Continue Shopping) — switches tab + dismisses. */
function ctaButton(opts: {
  tab: "account" | "shop";
  leading: SDUINode;
  label: string;
  transparent?: boolean;
}): SDUINode {
  const action: SDUIAction = {
    kind: "composed",
    actions: [{ kind: "tabswitch", target: opts.tab }, { kind: "dismiss" }],
  };
  return {
    type: "button",
    props: {
      action,
      bg: opts.transparent ? "transparent" : "bg_card",
      stroke: "gold_dim",
      stroke_width: 1,
      corner_radius: 6,
      full_width: true,
      padding_v: 12,
    },
    children: [
      {
        type: "hstack",
        props: { spacing: 6, alignment: "center", justify: "center" },
        children: [opts.leading, text(opts.label, "montserrat.semibold:14", "gold")],
      },
    ],
  };
}
 
export function verifySpec(): SDUISpec {
  const root: SDUINode = {
    type: "scroll",
    props: { padding_h: 16, padding_v: 24, spacing: 24 },
    children: [
      // ── The certificate card ──────────────────────────────────────────────
      {
        type: "card",
        props: { stroke: "gold_dim", bg: "bg_card", corner_radius: 16, padding: 0, spacing: 0, alignment: "leading" },
        children: [
          // Header strip
          {
            type: "vstack",
            props: { alignment: "center", spacing: 12, padding_h: SECTION_PAD_H, padding_v: 22, max_width: "infinity" },
            children: [
              { type: "logo_mark", props: { size: 28 } },
              text("Certificate of Authenticity / Provenance", "cinzel.bold:10", "muted", { tracking: 4, text_case: "upper", alignment: "center" }),
            ],
          },
          sectionDivider,
 
          // ── VERIFIED body (found == true) ──
          {
            type: "vstack",
            visible_when: "data.found == true",
            props: { spacing: 0, alignment: "leading" },
            children: [
              // CoA photo (§31) — the real bar + NFC card photo (nfc_serials.image_url),
              // shown when present. Mirrors the web certificate's inline photo
              // (verify/[serial]/page.tsx:243-251). Data: data.image_url via verify-core.
              {
                type: "vstack",
                visible_when: "data.image_url",
                props: { alignment: "center", spacing: 0, padding_h: SECTION_PAD_H, padding_v: 16, max_width: "infinity" },
                children: [
                  { type: "image", props: { url: "{data.image_url}", max_height: 220, fit: "fit", corner_radius: 12, bg: "bg" } },
                ],
              },
              // CoA authenticity video (§67 Step 2) — the PAMP Veriscan clip
              // (nfc_serials.video_url), shown below the photo when present.
              // Mirrors the web certificate's inline <video> (verify/[serial]/
              // page.tsx:302-320). The native SDUI renderer has no video node, so
              // the clip plays through the registered `web_panel` node pointed at
              // a chrome-free /embed page that renders the HTML5 <video>; the embed
              // reports its content height so the panel sizes to the clip. `lazy`
              // defers the WebView mount until the panel scrolls into view (no
              // buffering a clip the viewer may never scroll to). Data:
              // data.video_url via verify-core; serial threaded from state.
              {
                type: "vstack",
                visible_when: "data.video_url",
                props: { alignment: "center", spacing: 0, padding_h: SECTION_PAD_H, padding_v: 16, max_width: "infinity" },
                children: [
                  { type: "web_panel", props: { path: "/embed/verify/coa-video?serial={state.serial}", height: 240, lazy: true } },
                ],
              },
              // Pill + detail rows
              {
                type: "vstack",
                props: { spacing: 16, alignment: "leading", padding_h: SECTION_PAD_H, padding_v: 22, max_width: "infinity" },
                children: [
                  {
                    type: "hstack",
                    props: { spacing: 6, alignment: "center", justify: "center" },
                    children: [
                      { type: "icon", props: { name: "checkmark.shield.fill", size: 18, color: "success" } },
                      text("Verified · in our records", "cinzel.semibold:14", "success"),
                    ],
                  },
                  {
                    type: "vstack",
                    props: { spacing: 0, alignment: "leading", max_width: "infinity" },
                    children: [
                      row("Serial Number", "{data.serial}", { mono: true }),
                      row("Packed by", "1Plus1Gold"),
                      row("Content", "{data.product_label}"),
                      row("Purity", "{data.purity_label}"),
                      row("Dimensions", "{data.product_dimensions}", { visible_when: "data.product_dimensions" }),
                      // Source folds in the parent combibar's serial (· Combibar XXXX) — no separate Parent Bar row.
                      row("Source", "{data.source_label}"),
                      row("Date Packed", "{data.assigned_at_label}"),
                    ],
                  },
                ],
              },
              sectionDivider,
              // Today's market value strip
              {
                type: "vstack",
                props: { spacing: 14, alignment: "leading", padding_h: SECTION_PAD_H, padding_v: 22, max_width: "infinity" },
                children: [
                  text("Today's market value", "cinzel.bold:10", "muted", { tracking: 4, text_case: "upper", alignment: "center" }),
                  {
                    type: "grid",
                    props: { columns: 2, spacing: 16 },
                    children: [
                      priceTile({ label: "Shop price", pricePath: "data.shop_price_sgd", premiumPath: "data.shop_premium_label", valueColor: "gold" }),
                      priceTile({ label: "We buy back", pricePath: "data.buyback_price_sgd", premiumPath: "data.buyback_premium_label", valueColor: "gold_mid" }),
                    ],
                  },
                  text("{data.price_disclosure}", "montserrat:9", "muted", { alignment: "center", line_spacing: 2 }),
                ],
              },
            ],
          },
 
          // ── NOT-FOUND body (found == false) ──
          {
            type: "vstack",
            visible_when: "data.found == false",
            props: { spacing: 14, alignment: "center", padding_h: SECTION_PAD_H, padding_v: 28, max_width: "infinity" },
            children: [
              {
                type: "hstack",
                props: { spacing: 6, alignment: "center", justify: "center" },
                children: [
                  { type: "icon", props: { name: "exclamationmark.shield.fill", size: 18, color: "hex:E05050" } },
                  text("Not in our records", "cinzel.semibold:14", "hex:E05050"),
                ],
              },
              text(
                "We couldn't verify a card with this serial number. Please re-check the serial engraved on your card — or contact us if you believe this is an error.",
                "montserrat:12",
                "offwhite",
                { alignment: "center", line_spacing: 3 },
              ),
              {
                type: "hstack",
                visible_when: "state.serial",
                props: { spacing: 6, alignment: "center", justify: "center", bg: "hex:E05050@10", stroke: "hex:E05050@40", stroke_width: 1, corner_radius: 8, padding_h: 12, padding_v: 8 },
                children: [
                  { type: "icon", props: { name: "info.circle", size: 12, color: "hex:E05050" } },
                  text("{state.serial}", "montserrat:11", "offwhite"),
                ],
              },
            ],
          },
 
          // ── Gift claim block (has owner & viewer ≠ owner) ──
          { type: "divider", visible_when: "data.found == true && data.can_show_gift_cta == true", props: { color: "rule", thickness: 1 } },
          { type: "gift_claim_card", visible_when: "data.found == true && data.can_show_gift_cta == true" },
 
          // ── Giver's own outstanding gift — read-only keepsake note (no CTA) ──
          // The owner reviewing their OWN card sees the note they wrote, gated on
          // can_show_owner_note (server: is_pending_gift && viewer == owner &&
          // gift_message present). Mutually exclusive with gift_claim_card
          // (can_show_gift_cta requires viewer ≠ owner). Built from GENERIC
          // card+text nodes — no native gift_claim_card state — so it ships OTA;
          // pre-this-spec binaries simply skip it (server sends false → EmptyView).
          { type: "divider", visible_when: "data.can_show_owner_note == true", props: { color: "rule", thickness: 1 } },
          {
            type: "vstack",
            visible_when: "data.can_show_owner_note == true",
            props: { spacing: 12, alignment: "center", justify: "center", padding_h: SECTION_PAD_H, padding_v: 22, max_width: "infinity" },
            children: [
              {
                type: "card",
                props: { bg: "gold@6", stroke: "gold_dim", top_accent: "none", corner_radius: 8, padding_h: 20, padding_v: 16, alignment: "center", spacing: 8 },
                children: [
                  text("Your gift note", "montserrat:10", "gold_mid", { text_case: "upper", tracking: 2, alignment: "center" }),
                  text("“{data.gift_message}”", "cormorant.italic:14", "offwhite", { alignment: "center", multiline: true }),
                ],
              },
              {
                type: "hstack",
                props: { spacing: 6, alignment: "center", justify: "center" },
                children: [
                  { type: "icon", props: { name: "gift", size: 14, color: "gold" } },
                  text("Marked as a gift · pending acceptance", "montserrat:11", "gold", { text_case: "upper", tracking: 1, alignment: "center" }),
                ],
              },
              text("The recipient can accept this card by scanning it and signing in.", "cormorant:14", "offwhite", { alignment: "center", multiline: true }),
            ],
          },
 
          // ── External-purchase claim block (§49 — marketplace order, unclaimed) ──
          // Mutually exclusive with the gift block: an unclaimed external card has
          // no current_owner, so can_show_gift_cta is false. Renders EmptyView on
          // pre-§49 binaries (renderer default case), so it's safe to ship server-
          // side ahead of the batched iOS build.
          { type: "divider", visible_when: "data.external_claim != null && data.external_claim.claimed == false", props: { color: "rule", thickness: 1 } },
          { type: "external_claim_card", visible_when: "data.external_claim != null && data.external_claim.claimed == false" },
 
          // ── CTAs ──
          sectionDivider,
          {
            type: "vstack",
            props: { spacing: 8, alignment: "center", padding_h: SECTION_PAD_H, padding_v: 22, max_width: "infinity" },
            children: [
              ctaButton({
                tab: "account",
                label: "My Account",
                leading: { type: "logo_mark", props: { size: 18, inline: true } },
              }),
              ctaButton({
                tab: "shop",
                label: "Continue Shopping",
                transparent: true,
                leading: { type: "icon", props: { name: "bag", size: 15, color: "gold" } },
              }),
            ],
          },
        ],
      },
 
      // ── Footer note (outside the card) ──
      text(
        "This page confirms the serial on your engraved card matches our records. For sales support, visit 1plus1gold.com/contact.",
        "montserrat:10",
        "muted",
        { alignment: "center", line_spacing: 3 },
      ),
    ],
  };
 
  return {
    surface: "verify",
    spec_version: 1,
    data_source: "/api/coa/verify?serial={state.serial}",
    // `auto_claim` is deliberately NOT declared here. It must come ONLY from the
    // trusted local deep-link seed (App.swift, from `?claim=auto`), never from
    // this server spec — otherwise a tampered spec could default it true and
    // force every viewer to auto-claim a gift (transfer ownership) on open. The
    // seed sets it regardless of the schema; leaving it out means the spec can't.
    // (security_agent server-forced-auto-claim.)
    state_schema: {
      serial: { type: "string", default: "" },
    },
    root,
  };
}
