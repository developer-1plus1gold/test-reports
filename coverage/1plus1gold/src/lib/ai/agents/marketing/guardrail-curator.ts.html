/**
 * MarketingGuardrailCurator — tends the Brand Guardrails learning loop.
 *
 * The guardrails set fills up over time (reviewer proposes one per flag; the
 * operator adds/rejects). Left untended it rots: duplicates, over-specific rules,
 * contradictions, and ineffective rules that keep getting flagged anyway — and
 * every active negative_exclusion is concatenated into EVERY generation's
 * negative prompt, so a noisy set makes output worse, not better.
 *
 * This agent reconciles the set daily and PROPOSES curation actions (it never
 * mutates the table — the operator approves each via agent_decisions, then the
 * decisions PATCH route applies the change deterministically in code).
 *
 * Tier: reasoning (multi-rule reconciliation + effectiveness judgement).
 * Autonomy: notify / propose-only. Mirrors playbook_curator for security.
 */
 
import { BaseAgent } from "../../base-agent";
import type { ModelTier } from "../../registry";
import {
  isCurationType,
  isNoOpCuration,
  resolveCurationWrite,
  type CurationActionType,
  type CuratedRule,
  type GuardrailKindValue,
  type GuardrailScopeValue,
} from "@/lib/marketing/curation-action";
 
export type { CurationActionType };
 
export interface CurationAction {
  type:          CurationActionType;
  /** Existing guardrail ids this action targets. */
  guardrailIds:  string[];
  /** Canonical text for merge/generalize/rewrite (the rule that survives/updates). */
  newRuleText?:  string;
  newScope?:     GuardrailScopeValue;
  newKind?:      GuardrailKindValue;
  /** Surface a contradiction between two ACTIVE rules at higher severity. */
  isContradiction?: boolean;
  rationale:     string;
}
 
export interface CuratorInput {
  guardrails: Array<{ id: string; scope: string; kind: string; rule_text: string; status: string }>;
  /** Recent review flag descriptions — the effectiveness signal (what keeps failing). */
  recentFlags: string[];
}
 
export interface CuratorOutput {
  actions: CurationAction[];
  summary: string;
}
 
export class MarketingGuardrailCuratorAgent extends BaseAgent {
  readonly name = "marketing_guardrail_curator";
  readonly tier: ModelTier = "reasoning";
 
  getRole(): string {
    return [
      "I am the Marketing Guardrail Curator for 1Plus1Gold.",
      "I keep the Brand Guardrails ruleset tight, non-contradictory, and effective:",
      "  I dedupe, merge, generalise, retire, rewrite, and promote rules.",
      "Every active negative_exclusion is added to every generation's negative",
      "  prompt, so a lean, sharp set matters more than a big one.",
      "I PROPOSE changes only — the operator approves each before it takes effect.",
    ].join("\n");
  }
 
  async curate(input: CuratorInput): Promise<CuratorOutput> {
    if (input.guardrails.length === 0) return { actions: [], summary: "No guardrails to curate." };
 
    const response = await this.runInference({
      runtimeContext: this.composeRuntimeContext(input),
      messages: [{ role: "user", content: this.composeUserPrompt(input) }],
      maxTokens: 3000,
    });
 
    const text = response.text;
    const block = text.match(/```json\s*([\s\S]*?)\s*```/);
    if (!block) {
      // No proposals is a valid outcome (a clean set) — don't throw.
      Eif (/no\s+(curation|changes|actions)/i.test(text)) return { actions: [], summary: text.slice(0, 300) };
      throw new Error("Curator output missing JSON block. Raw: " + text.slice(0, 300));
    }
    let parsed: CuratorOutput;
    try { parsed = JSON.parse(block[1]) as CuratorOutput; }
    catch (e) { throw new Error("Curator JSON parse failed: " + (e instanceof Error ? e.message : String(e))); }
 
    // Validate each action references real ids + a known type. Drop invalid ones
    // (the LLM proposes; code keeps it honest).
    const ruleById = new Map<string, CuratedRule>(input.guardrails.map((g) => [g.id, g]));
    const structurallyValid = (parsed.actions ?? []).filter((a) =>
      isCurationType(a.type) &&
      Array.isArray(a.guardrailIds) && a.guardrailIds.length > 0 &&
      a.guardrailIds.every((id) => ruleById.has(id)) &&
      // Text-bearing actions must still carry text AFTER sanitisation — text made
      // only of stripped characters would reach the apply step and write nothing.
      (["retire", "promote"].includes(a.type) || resolveCurationWrite(a).text.length > 0),
    );
 
    // Drop actions that would change nothing. Every filed action costs an operator
    // a Control Tower tap, so a proposal that writes the row back byte-identical is
    // worse than no proposal at all. See isNoOpCuration for the 13-occurrence
    // history that made this a code check rather than a prompt rule.
    const actions = structurallyValid.filter((a) => {
      const targets = a.guardrailIds.map((id) => ruleById.get(id)).filter((r): r is CuratedRule => Boolean(r));
      if (!isNoOpCuration(a, targets)) return true;
      // Named, never silent — a curator quietly discarding its own output should
      // be visible in the cron log rather than look like a clean pass.
      console.warn(
        `[guardrail_curator] dropped no-op ${a.type} on ${a.guardrailIds.join(",")} ` +
        `(the write is identical to the current rule)`,
      );
      return false;
    });
 
    return { actions, summary: (parsed.summary ?? "").slice(0, 500) };
  }
 
  private composeRuntimeContext(input: CuratorInput): string {
    return [
      `Today: ${new Date().toISOString().slice(0, 10)}.`,
      `Guardrail kinds: negative_exclusion (added to the negative prompt), positive_directive (added to the prompt), review_check (added to the reviewer's checklist).`,
      `Scopes: all | generate | edit | animate | caption | response | brief.`,
      `  'all' is the BROADEST scope — it is folded into EVERY surface, so widening a`,
      `  scoped rule to 'all' is a real behaviour change, not a tidy-up.`,
      `  caption = post copy the Writer honours; response = reply copy; brief = brief specs.`,
      `The set currently has ${input.guardrails.length} rules.`,
    ].join("\n");
  }
 
  private composeUserPrompt(input: CuratorInput): string {
    const lines: string[] = [];
    lines.push("Reconcile this Brand Guardrails ruleset. Propose curation actions ONLY where they clearly help — an already-tight set warrants zero actions.");
    lines.push("");
    lines.push("CURRENT GUARDRAILS (id · status · scope/kind · text):");
    for (const g of input.guardrails) {
      lines.push(`  ${g.id} · ${g.status} · ${g.scope}/${g.kind} · ${g.rule_text}`);
    }
    lines.push("");
    if (input.recentFlags.length) {
      lines.push("RECENT REVIEW FAILURES (effectiveness signal — if a failure recurs despite an active rule, that rule needs a stronger rewrite):");
      for (const f of input.recentFlags.slice(0, 20)) lines.push(`  • ${f}`);
      lines.push("");
    }
    lines.push("Action types: merge (fold duplicates into one canonical), generalize (replace specifics with a broader rule), retire (drop stale/contradictory/superseded), rewrite (strengthen an ineffective rule OR fix a MISCATEGORISED one), promote (a strong proposed rule -> active).");
    lines.push("MISCATEGORISATION — rules now arrive auto-activated straight from operator rejections, heuristically typed, so some kinds/scopes will be wrong. Fix them with a rewrite that sets newKind/newScope: a rule that asks for something to be PRESENT but is typed negative_exclusion must become positive_directive, because an avoid-typed 'always show the hallmark' pushes the model AWAY from the hallmark — the exact opposite of its intent. Also retire one-off observations that aren't reusable rules.");
    lines.push("▶ A rule that is ALREADY correctly typed and worded needs NO action. Do NOT file a rewrite to confirm a rule is fine, to restate it unchanged, or to explain why it is not miscategorised — an action whose resulting text+kind+scope equals the current rule is DISCARDED in code and only costs the operator a wasted approval tap. Say it in the summary instead, or return no actions at all. Likewise do not 'promote' a rule that is already active.");
    lines.push("Set isContradiction=true only when two ACTIVE rules genuinely conflict.");
    lines.push("");
    lines.push("OUTPUT — wrap in ```json … ``` exactly. Schema:");
    lines.push("```json");
    lines.push(JSON.stringify({
      actions: [{
        type: "merge|generalize|retire|rewrite|promote",
        guardrailIds: ["<existing id>", "..."],
        newRuleText: "canonical text (omit for retire/promote)",
        newScope: "all|generate|edit|animate|caption|response|brief",
        newKind: "negative_exclusion|positive_directive|review_check",
        isContradiction: false,
        rationale: "1 sentence why",
      }],
      summary: "1-2 sentence overview of the curation pass",
    }, null, 2));
    lines.push("```");
    lines.push("If the set is already clean, return an empty actions array with a short summary.");
    return lines.join("\n");
  }
}
