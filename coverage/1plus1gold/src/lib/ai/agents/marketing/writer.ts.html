/**
 * MarketingWriter — produces all marketing copy.
 *
 * Generates briefs, captions (3 variants per channel), hashtags,
 * image-generation prompts, and response drafts. The most-used agent
 * in the marketing pipeline; replaces the manual "Mukesh in chat with
 * Claude" loop for content production.
 *
 * Tier: reasoning. Captions and briefs need brand-voice judgement
 * + chakra-arc constraints + compliance-aware vocabulary — fast tier
 * misses nuance, hard tier is overkill. Reasoning hits the sweet spot.
 *
 * Output always lands in `marketing_drafts` at status=proposed. The
 * /admin/agents Awaiting Implementation queue surfaces each draft for
 * Mukesh's approve/edit/reject. Approval transitions status=approved
 * + triggers downstream agents (Generator on briefs, Reviewer on
 * captions).
 */

/** What kind of draft to generate. */
export type DraftKind = "brief" | "caption" | "hashtags" | "response" | "image_prompt";
 
/** Input shape passed to the agent's main entry. The route handler
 *  validates this at the boundary. */
export interface WriterInput {
  kind:         DraftKind;
  /** The content piece this draft is for. Always present except for
   *  free-form responses where it's optional (a DM not yet tied to a
   *  post). */
  piece?:       ContentPieceRow;
  /** For captions: which channel? */
  channel?:     Channel;
  /** For responses: the inbound comment/DM body + classification. */
  inbound?: {
    text:           string;
    platform:       string;
    handle:         string | null;
    intent:         string | null;
    sentiment:      string | null;
    isKnownCustomer: boolean;
  };
  /** For re-drafts: prior approved drafts on the same piece (Writer
   *  uses these for coherence across captions). */
  priorDrafts?: Array<{ kind: DraftKind; channel: string | null; body: unknown }>;
  /** For re-drafts after a flagged Reviewer pass: the flag rationale +
   *  recommended adjustments to incorporate. */
  reviewerFeedback?: {
    flags:                   Array<{ flag_type: string; description: string; severity: string }>;
    recommended_adjustments: string | null;
  };
  /** Operator's free-text revision request on a regenerate ("what should change").
   *  Steers a re-draft with an explicit reason instead of a blind re-roll. */
  operatorInstructions?: string;
  /** For captions: active brand guardrails for COPY (the caption learning loop).
   *  `avoid` = rules learned from rejected captions ("don't do X"); `directives`
   *  = standing copy directives to always honour. The route resolves these. */
  captionGuardrails?: { avoid: string[]; directives: string[] };
  /** For captions: the operator's rejection reasons from THIS piece's already-
   *  rejected captions (across ALL channels — copy rules aren't channel-specific).
   *  Fed in verbatim as a "you MUST fix every one of these" block so a regenerate
   *  honours them immediately, with no guardrail-approval step. Unlike
   *  captionGuardrails (pre-classified avoid/directive), these are raw — the
   *  model decides per reason whether to ADD what's missing or DROP what's banned. */
  priorRejections?: string[];
  /** For captions + briefs: what the operator ACTUALLY SHIPPED after editing a
   *  previous draft, paired with what the agent originally wrote.
   *
   *  This is the highest-value feedback the system has and it used to be thrown
   *  away: the operator's real workflow is edit-and-approve, but the learning
   *  loop only read `rejection_reason`, and the edit path overwrote the agent's
   *  output in place. Each pair is "you wrote X, the operator shipped Y" —
   *  the model is asked to infer the standing preference and pre-apply it, not
   *  to copy Y verbatim. */
  priorEdits?: Array<{ generated: string; shipped: string; note?: string }>;
  /** Number of variants to produce. Captions default to 3, others to 1. */
  variantCount?: number;
  /** For captions: the actual post media so copy matches the real visuals.
   *  Images go to the model as vision input; videos can't (vision takes images
   *  only) so they're summarised in mediaNote. The route gathers + caps these. */
  media?:     Array<{ mediaType: string; base64: string }>;
  mediaNote?: string;
}
 
/** Output shape — what gets persisted into marketing_drafts.content. */
export interface WriterOutput {
  /** Echo of the input.kind for downstream routing. */
  kind:        DraftKind;
  /** The drafts themselves. */
  variants:    Array<{
    label:     string;
    body:      unknown;        // shape varies by draft_kind — see card.purpose
    rationale: string;         // why Writer chose this — surfaces in Control Tower
  }>;
  /** Self-reported confidence (0..1). Used by Listener to short-circuit
   *  low-confidence response drafts that should escalate to operator. */
  confidence:  number;
  /** Any compliance / brand-voice concerns Writer self-flagged. */
  selfFlags:   string[];
}
