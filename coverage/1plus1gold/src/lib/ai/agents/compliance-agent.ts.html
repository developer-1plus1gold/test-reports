/**
 * Compliance Agent — onboarding business-context risk assessment.
 *
 * Migrated onto the Phase 0 BaseAgent foundation. Behaviour parity with the
 * pre-migration `risk-rating/route.ts` Anthropic call:
 *   • Same model tier (cheap / Haiku 4.5)
 *   • Same JSON output schema
 *   • Same input categories (profile, orders, applicant, jurisdiction, AML)
 *
 * Key change vs pre-migration: AML screening status is now READ from the
 * the provider's real screening result instead of being hardcoded to "PASSED". If the
 * upstream review hasn't completed, the prompt says so plainly rather than
 * asserting a clean status that may not be true.
 *
 * The route handler (`/api/compliance/risk-rating`) does the data fetching,
 * the JSON parsing, and the post-processing (compliance_cases insert, STR
 * draft, WhatsApp alert). The agent class owns the prompt composition + the
 * inference call.
 */
 
import { BaseAgent } from "../base-agent";
import type { ModelTier } from "../registry";
import { KYC_THRESHOLD_SGD } from "@/lib/kyc/constants";
 
export interface ComplianceContext {
  customerId:           string;
  /**
   * ⚠️ DELIBERATELY NARROW — do NOT widen this to the whole `profiles` row.
   *
   * We tell customers, in privacy §5 and on the sub-processor register, that
   * Anthropic "does not receive your name or NRIC/passport number". Until
   * 2026-09-08 that was true only because `buildRuntimeContext` happened not to
   * render `full_name` — the field WAS on this type and WAS populated by the
   * caller, so one added line would have broken a published claim silently.
   *
   * The caller now passes an object LITERAL of exactly these two fields, so
   * TypeScript's excess-property check rejects handing over the whole row, and
   * the name cannot reach this agent at all. See
   * `src/__tests__/compliance-prompt-no-identity.test.ts`.
   */
  profile: {
    kyc_verified_at?:    string | null;
    self_limit_below_5k?: boolean | null;
  };
  applicantData: {
    nationality?:    string;
    country?:        string;
    idDocTypes?:     (string | undefined)[] | null;
  };
  orderHistory: {
    count:        number;
    totalSgd:     number;
    maxSingleSgd: number;
  };
  highRiskJurisdiction: boolean;
  /** Real AML status from the verification provider (Didit). One of:
   *    - a CLEARED screen (no sanctions, no PEP, no adverse media)
   *    - a HIT (sanctions / PEP / watchlist)
   *    - PENDING — screening not yet completed
   *    - UNAVAILABLE — no screening on the session, or the fetch failed
   *  Replaces the pre-migration hardcode that always said "PASSED". */
  amlStatus: string;
}
 
export interface ComplianceAgentOutput {
  riskLevel:         "low" | "medium" | "high" | "critical";
  rationale:         string;
  flags:             string[];
  recommendedAction: "monitor_only" | "enhanced_due_diligence" | "escalate_immediately";
  nextReviewMonths:  number;
}
 
export class ComplianceAgent extends BaseAgent {
  readonly name = "compliance";
  readonly tier: ModelTier = "cheap";
 
  getRole(): string {
    return `You assess BUSINESS-CONTEXT compliance risk for a newly-KYC-approved customer of 1Plus1Gold. \
The verification provider has already run sanctions and PEP screening upstream — your job is the residual risk that identity \
checks cannot detect: structuring, transaction velocity, jurisdiction risk, and behavioural patterns inconsistent \
with a normal retail gold buyer.
 
Apply MAS PSPM Act / ACIP / FATF standards for a precious metals dealer. Focus on:
1. Jurisdiction risk — is the customer from a FATF high-risk or MAS-monitored jurisdiction?
2. Structuring — multiple orders just below the S$${KYC_THRESHOLD_SGD.toLocaleString()} KYC/CDD threshold suggests deliberate avoidance
3. Velocity — total pre-KYC spend unusually high for a new retail customer?
4. Consistency — does spend pattern match a normal retail buyer or suggest commercial scale?
5. Self-limit — voluntarily capping below the S$${KYC_THRESHOLD_SGD.toLocaleString()} KYC threshold is a genuine low-risk indicator
 
Sanctions and PEP risk are already cleared upstream. Do NOT flag these unless the jurisdiction itself is \
high-risk regardless of individual screening.
 
OUTPUT CONTRACT — respond ONLY with valid JSON, no markdown, no preamble:
{
  "riskLevel": "low" | "medium" | "high" | "critical",
  "rationale": "2-4 sentence plain-English explanation focused on business patterns",
  "flags": ["specific", "concern", "flags"],
  "recommendedAction": "monitor_only" | "enhanced_due_diligence" | "escalate_immediately",
  "nextReviewMonths": 12 | 6 | 3 | 1
}
 
AML screening handling:
- A FAILED screen (a sanctions/PEP/watchlist hit) is a genuine hit → CRITICAL, escalate_immediately, regardless of other signals.
- "Pending" or "Status not available" means the screen has NOT completed — this is the ABSENCE of a result, NOT a hit. Do NOT treat it as critical on its own. For a normal retail buyer below the S$${KYC_THRESHOLD_SGD.toLocaleString()} CDD threshold, an outstanding screen is at most a low/medium note (recommend monitor_only and mention that screening is pending). Only when the customer is transacting AT OR ABOVE S$${KYC_THRESHOLD_SGD.toLocaleString()} without a completed screen does the unscreened state itself become critical.`;
  }
 
  /**
   * Compose the per-request runtime context block. This is the only volatile
   * part of the prompt — buildAgentPrompt above stamps in persona + role +
   * capability + APP FACTS as the cache-friendly prefix.
   */
  buildRuntimeContext(ctx: ComplianceContext): string {
    return `Customer ID: ${ctx.customerId}
Self-limit (below S$${KYC_THRESHOLD_SGD.toLocaleString()}): ${ctx.profile.self_limit_below_5k ? "Yes" : "No"}
KYC verified at: ${ctx.profile.kyc_verified_at ?? "Just now"}
 
Identity (from the verification provider — sanctions/PEP screened upstream):
  Nationality: ${ctx.applicantData.nationality ?? "Not available"}
  Country of document: ${ctx.applicantData.country ?? "Not available"}
  ID document types: ${JSON.stringify(ctx.applicantData.idDocTypes ?? [])}
  High-risk jurisdiction (FATF/MAS): ${ctx.highRiskJurisdiction ? "YES" : "No"}
 
AML Screening Status:
  ${ctx.amlStatus}
 
Order History (pre-KYC):
  Total completed orders: ${ctx.orderHistory.count}
  Total spend (SGD): ${ctx.orderHistory.totalSgd.toFixed(2)}
  Largest single order (SGD): ${ctx.orderHistory.maxSingleSgd.toFixed(2)}`;
  }
 
  /**
   * Run the assessment and parse the JSON output.
   * Throws if the model produced unparseable JSON or an invalid riskLevel —
   * the caller should treat this as a 500-equivalent failure and surface it.
   */
  async assess(ctx: ComplianceContext): Promise<{
    output: ComplianceAgentOutput;
    runLogId: string;
    costUsd: number;
  }> {
    const runtimeContext = this.buildRuntimeContext(ctx);
    const result = await this.runInference({
      runtimeContext,
      messages: [{
        role: "user",
        content: "Assess the customer's risk per the OUTPUT CONTRACT. Respond with JSON only.",
      }],
      customerId: ctx.customerId,
      maxTokens:  512,
    });
 
    const jsonMatch = result.text.match(/\{[\s\S]*\}/);
    if (!jsonMatch) {
      throw new Error("compliance_agent_invalid_json: no JSON object in response");
    }
 
    let parsed: ComplianceAgentOutput;
    try {
      parsed = JSON.parse(jsonMatch[0]) as ComplianceAgentOutput;
    } catch (err) {
      throw new Error(`compliance_agent_invalid_json: ${err instanceof Error ? err.message : String(err)}`);
    }
 
    const riskLevel = (parsed.riskLevel ?? "").toLowerCase() as ComplianceAgentOutput["riskLevel"];
    if (!["low", "medium", "high", "critical"].includes(riskLevel)) {
      throw new Error(`compliance_agent_invalid_risk_level: ${parsed.riskLevel}`);
    }
    parsed.riskLevel = riskLevel;
 
    return {
      output:   parsed,
      runLogId: result.runLogId,
      costUsd:  result.costUsd,
    };
  }
}
