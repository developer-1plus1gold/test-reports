/**
 * playbook-index — deterministic facts about docs/attacker-playbook.md that the
 * playbook_curator agent must NOT be left to infer.
 *
 * ── Why this module exists ────────────────────────────────────────────────────
 * The curator was handed `current_playbook.slice(0, 25_000)`. The playbook is
 * ~193 KB, so the agent saw the first ~13% of the file: two of its twelve `## `
 * sections and none of its running-count notes. Every chronic curator failure in
 * the record traces to that one blind prefix:
 *
 *   • The A5 running count was proposed wrong five mornings running. The three
 *     "running count → N" notes sit at byte offsets ~29k/36k/40k — all past the
 *     cut. The last ordinal inside the agent's window was "Fourteenth", so
 *     "fifteenth" was correct reasoning on the input it was given.
 *   • Colliding `## F. Frontend / Headers`, `## P. Payment Integrity`,
 *     `## H. Harness Defects`, duplicate F5 / I10 rows — it could not see
 *     sections B..J to collide with them.
 *   • "No matching playbook scenario existed" — from its window, true.
 *
 * A prompt rule added 2026-08-25 told the agent it was "given the COMPLETE
 * current file". That claim was false, which is worse than saying nothing.
 *
 * The fix is to stop asking the model for things the code can determine:
 *   • buildPlaybookIndex()     — the complete section + row inventory, always
 *                                sent whole. Bounded (~4 KB for 61 rows), so it
 *                                survives however large the prose grows.
 *   • buildAtoCountDirective() — the A5 count and whether THIS run increments
 *                                it, computed rather than inferred.
 *
 * The playbook file stays the single source of truth for the count; this module
 * only reads it reliably. Counting from `pentest_reports` instead was considered
 * and rejected: a pass is one logical run recorded in whichever database the
 * cron targeted, so neither DB alone sees the union (verified 2026-08-26 — DEV
 * alone yields 15, PROD alone 17, the union 18), and the curator has no
 * cross-database access at runtime.
 */

/** English cardinal for 0-99 (e.g. 18 -> "eighteen"). Falls back to digits. */
export function cardinalWord(n: number): string {
  if (!Number.isInteger(n) || n < 0 || n > 99) return String(n);
  if (n < 20) return CARDINALS_0_19[n];
  const tens = Math.floor(n / 10);
  const ones = n % 10;
  return ones === 0 ? TENS[tens] : `${TENS[tens]}-${CARDINALS_0_19[ones]}`;
}
 
/** English ordinal for 0-99 (e.g. 18 -> "eighteenth"). Falls back to "Nth". */
export function ordinalWord(n: number): string {
  if (!Number.isInteger(n) || n < 0 || n > 99) return `${n}th`;
  if (n < 20) return ORDINALS_0_19[n];
  const tens = Math.floor(n / 10);
  const ones = n % 10;
  return ones === 0 ? TENS_ORDINAL[tens] : `${TENS[tens]}-${ORDINALS_0_19[ones]}`;
}
 
/** Inverse of cardinalWord for 0-99, plus bare digits. Returns null if unparseable. */
export function parseCountWord(raw: string): number | null {
  const s = raw.trim().toLowerCase().replace(/\s+/g, "-");
  if (/^\d{1,3}$/.test(s)) return Number(s);
  const flat = CARDINALS_0_19.indexOf(s as (typeof CARDINALS_0_19)[number]);
  if (flat >= 0) return flat;
  const tensOnly = TENS.indexOf(s as (typeof TENS)[number]);
  if (tensOnly >= 2) return tensOnly * 10;
  const m = s.match(/^([a-z]+)-([a-z]+)$/);
  if (m) {
    const t = TENS.indexOf(m[1] as (typeof TENS)[number]);
    const o = CARDINALS_0_19.indexOf(m[2] as (typeof CARDINALS_0_19)[number]);

  }
  return null;

/**
 * The current A5 running count, read from the LAST "running count → <N>" note in
 * the file. Deliberately takes the last occurrence in document order: the notes
 * are appended chronologically, so the newest is the live value.
 */
export function extractAtoPassCount(playbook: string): number | null {
  // ── Authoritative marker wins, whatever its position ────────────────────────
  // Document order stopped being a safe proxy for "current" on 2026-09-06. A5's
  // hand-authored current-status digest leads the scenario, but the 14-day
  // retention window keeps recent run notes BELOW it — and those notes quote the
  // counts that were current when they were written. Last-match-wins would then
  // read a superseded number out of history sitting under the live statement.
  // (It returned the right answer on the full file only by luck, because the
  // newest retained note happened to agree; truncate to 25k and it returned 18
  // against a live 24.) So the digest marks its count AUTHORITATIVE and that is
  // preferred outright. The positional read stays as the fallback, because it is
  // what every pre-digest revision of this file relies on.
  const authoritative = [
    ...playbook.matchAll(/AUTHORITATIVE running count\s*(?:→|->)\s*([A-Za-z-]+|\d{1,3})/gi),
  ];
  for (const m of authoritative) {
    const parsed = parseCountWord(m[1]);
    Eif (parsed !== null) return parsed;
  }
 
  // Matches "running count → seventeen", "running count -> SIXTEEN", "running count → 18".
  const re = /running count\s*(?:→|->)\s*([A-Za-z-]+|\d{1,3})/gi;
  // Take the last PARSEABLE match, not simply the last match. This file
  // discusses its own conventions in prose, so a sentence such as
  // 'the "running count → <N>" notes' can otherwise shadow the real value and
  // silently blank the count. Unparseable captures are skipped, not fatal.
  let last: number | null = null;
  for (const m of playbook.matchAll(re)) {
    const parsed = parseCountWord(m[1]);
    if (parsed !== null) last = parsed;
  }
  return last;
}
 
/**
 * Whether a run counts toward the A5 prod-target counter.
 *
 * "Prod-target" means the harness probed the production host. A run recorded in
 * the DEV database can still be prod-target (and usually is) — the database it
 * landed in is not the target. `is_dev_target` is honoured as a veto so an
 * explicitly dev-flagged run never increments.
 */
export function isProdTarget(targetBaseUrl?: string, isDevTarget?: boolean): boolean {
  if (isDevTarget === true) return false;
  if (!targetBaseUrl) return false;
  try {
    const u = new URL(targetBaseUrl);
    return u.protocol === "https:" && u.hostname.toLowerCase() === "1plus1gold.com";
  } catch {
    return false;
  }
}
 
export interface AtoCountFacts {
  /** Current recorded count, or null when the file carries no parseable note. */
  current:    number | null;
  /** Whether THIS run increments the counter. */
  increments: boolean;
  /** The value the curator should write. Null when `current` is unknown. */
  next:       number | null;
}
 
/** Pure decision: does this run move the counter, and to what. */
export function computeAtoCountFacts(args: {
  playbook:        string;
  objective?:      string;
  targetBaseUrl?:  string;
  isDevTarget?:    boolean;
}): AtoCountFacts {
  const current = extractAtoPassCount(args.playbook);
  const increments =
    (args.objective ?? "").trim().toLowerCase() === "account_takeover" &&
    isProdTarget(args.targetBaseUrl, args.isDevTarget);
  const next = current === null ? null : increments ? current + 1 : current;
  return { current, increments, next };
}
 
/**
 * The A5 directive handed to the agent. States the number outright so the model
 * never infers an ordinal — the single most repeated error in curator output.
 */
export function buildAtoCountDirective(args: {
  playbook:        string;
  objective?:      string;
  targetBaseUrl?:  string;
  isDevTarget?:    boolean;
}): string {
  const { current, increments, next } = computeAtoCountFacts(args);
  const head = "A5 RUNNING-COUNT (computed from the playbook — do NOT infer or recount):";
 
  if (current === null) {
    return [
      head,
      "  The current count could not be read from the playbook. Write \"count unchanged\" and",
      "  do NOT guess an ordinal.",
    ].join("\n");
  }
 
  if (!increments) {
    const why = isProdTarget(args.targetBaseUrl, args.isDevTarget)
      ? `this run's objective is '${args.objective ?? "unknown"}', not 'account_takeover'`
      : "this run is not prod-target";
    return [
      head,
      `  Current count: ${current} (${cardinalWord(current).toUpperCase()}).`,
      `  This run does NOT increment it — ${why}.`,
      `  If you mention the count, write exactly: "ATO count UNCHANGED at ${cardinalWord(current).toUpperCase()}".`,
      "  Do NOT propose any other ordinal.",
    ].join("\n");
  }
 
  return [
    head,
    `  Current count: ${current} (${cardinalWord(current).toUpperCase()}).`,
    "  This run IS a prod-target 'account_takeover' pass, so it increments by exactly one.",
    `  Write the new count as: "running count → ${cardinalWord(next!).toUpperCase()}" and describe`,
    `  this pass as the "${ordinalWord(next!).replace(/^./, (c) => c.toUpperCase())}" consecutive prod-target pass.`,
    `  Do NOT propose any ordinal other than ${next}.`,
    "  If another run of the same objective and target landed on this same date, it is still",
    "  ONE pass and ONE increment — the operator folds both proposals into a single note.",
  ].join("\n");
}
 
export interface PlaybookIndex {
  sections: string[];
  rows:     string[];
  text:     string;
}
 
/**
 * The COMPLETE structural inventory of the playbook: every `## ` section and
 * every `### ` row heading. Always sent to the agent in full, even when the
 * prose body is truncated — this is precisely what the collision rule needs, and
 * it is bounded by the number of headings rather than the size of the prose.
 */
export function buildPlaybookIndex(playbook: string): PlaybookIndex {
  const sections: string[] = [];
  const rows:     string[] = [];
  for (const line of playbook.split("\n")) {
    if (line.startsWith("### "))     rows.push(line.slice(4).trim());
    else if (line.startsWith("## ")) sections.push(line.slice(3).trim());
  }
  const text = [
    "COMPLETE PLAYBOOK INDEX — every section and row that already exists.",
    "This inventory is COMPLETE even if the playbook body below is truncated.",
    "Do NOT propose a '## ' letter or a '### ' identifier that appears here; amend the",
    "existing entry instead, and never claim a scenario is absent without checking this list.",
    "",
    `SECTIONS (${sections.length}):`,
    ...sections.map((s) => `  ## ${s}`),
    "",
    `ROWS (${rows.length}):`,
    ...rows.map((r) => `  ### ${r}`),
  ].join("\n");
  return { sections, rows, text };
}
