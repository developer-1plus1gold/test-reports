/**
 * computeConversionTotals — the fee + combibar-stock + live-spot reads for a
 * conversion of `totalGrams`. Shared by /api/conversions/preview (web) and
 * /api/conversions/form (the iOS `convert` surface) so the fee/stock/spot logic
 * is one implementation, no drift. Pure-ish (DB reads only, no writes).
 *
 * Security (MED-016, 2026-05-03): returns only `combibarStockOk` (boolean) —
 * never the exact shortfall — so an authed session can't map inventory levels.
 */
 
import type { SupabaseClient } from "@supabase/supabase-js";
import { greedyCombibarSplit, isValidConversionTotal } from "@/lib/conversions";
import { resolveConversionTargets, targetsHeldNow } from "@/lib/conversions-target";
import { getSpotPayload } from "@/lib/spot/fetch-spot";
 
export interface ConversionTotals {
  valid:            boolean;
  enabled:          boolean;
  totalGrams:       number;
  count50g:         number;
  count100g:        number;
  conversionFeeSgd: number;
  spotSgdPerG:      number | null;
  /** True when we hold enough bars to ship today. Informational — never a
   *  submit gate; a false simply means approval will source the bar. */
  combibarStockOk:  boolean;
}
 
async function liveSpotSgd(): Promise<number | null> {
  try {
    const data = await getSpotPayload();
    if ("error" in data) return null;
    const gram = Number(data.currencies?.SGD?.gram ?? null);
    return isNaN(gram) || gram <= 0 ? null : gram;
  } catch {
    return null;
  }
}
 
/**
 * Do we hold, right now, enough bars to meet this conversion from stock?
 *
 * INFORMATIONAL. It used to gate submission, and it resolved its products with
 * `product_type='curve'` at 50g/100g — a filter that matches nothing, since
 * every such product is `standard`. So the gate was permanently closed and no
 * conversion could be submitted. See lib/conversions-target.ts.
 *
 * Sourcing is now a back-office concern: false here does not refuse the
 * request, it means approval raises a consignment obligation instead — the same
 * route the shop already uses to sell these bars. Still a boolean, never a
 * count (MED-016).
 */
async function combibarStockOk(need50g: number, need100g: number, db: SupabaseClient): Promise<boolean> {
  const [t50, t100] = await Promise.all([
    need50g  > 0 ? resolveConversionTargets(50,  db) : Promise.resolve([]),
    need100g > 0 ? resolveConversionTargets(100, db) : Promise.resolve([]),
  ]);
  const ok50  = need50g  <= 0 || t50.reduce((n, c) => n + c.ownedAvailable, 0)  >= need50g;
  const ok100 = need100g <= 0 || t100.reduce((n, c) => n + c.ownedAvailable, 0) >= need100g;
  return ok50 && ok100 && (need50g > 0 ? targetsHeldNow(t50) : true);
}
 
/** Compute the fee/stock/spot for a `totalGrams` conversion. `valid` is false
 *  (and the rest zero/false) when totalGrams isn't a positive multiple of 50. */
export async function computeConversionTotals(totalGrams: number, db: SupabaseClient): Promise<ConversionTotals> {
  if (!isValidConversionTotal(totalGrams)) {
    return { valid: false, enabled: true, totalGrams, count50g: 0, count100g: 0, conversionFeeSgd: 0, spotSgdPerG: null, combibarStockOk: false };
  }
  const split = greedyCombibarSplit(totalGrams);
  const [feeRow, stockOk, spot] = await Promise.all([
    db.from("shipper_settings").select("flat_fee_sgd, enabled").eq("shipper", "conversion").single().then((r) => r.data),
    combibarStockOk(split.count50g, split.count100g, db),
    liveSpotSgd(),
  ]);
  return {
    valid:            true,
    enabled:          !!feeRow?.enabled,
    totalGrams,
    count50g:         split.count50g,
    count100g:        split.count100g,
    conversionFeeSgd: Number(feeRow?.flat_fee_sgd ?? 0),
    spotSgdPerG:      spot,
    combibarStockOk:  stockOk,
  };
}
