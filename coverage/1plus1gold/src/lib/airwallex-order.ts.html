/**
 * Airwallex side of the shared order pipeline (2026-11-07).
 *
 * processCompletedHitPayPayment (src/lib/hitpay-order.ts) is gateway-agnostic:
 * it creates the order, consumes stock, settles MAG / vault / promo / referral
 * and sends the receipt. Airwallex only differs in its pending table, the id
 * columns it stamps on the order, and what it records after completion — the
 * same split as razorpay-order.ts.
 */
 
import type { SupabaseClient } from "@supabase/supabase-js";
import {
  processCompletedHitPayPayment,
  type CompletedPaymentFields,
  type OrderGatewayConfig,
  type PendingPaymentRow,
} from "@/lib/hitpay-order";
import { AIRWALLEX_METHOD_MAP, paymentMethodMismatch } from "@/lib/airwallex";
import { sendAdminAlert } from "@/lib/twilio";
import { getAdminPhonesCsv } from "@/lib/admin-team";
 
export const AIRWALLEX_ORDER_GATEWAY: OrderGatewayConfig = {
  label:        "Airwallex",
  pendingTable: "airwallex_pending_payments",
  // paymentRequestId carries the Checkout Session id; hitpayPaymentId the
  // PaymentIntent id (the field names predate the second gateway).
  orderIdFields: (f) => ({
    airwallex_checkout_session_id: f.paymentRequestId ?? null,
    airwallex_payment_intent_id:   f.hitpayPaymentId  ?? null,
  }),
  pendingDoneFields: (f) => ({ airwallex_payment_intent_id: f.hitpayPaymentId ?? null }),
};
 
export interface AirwallexCompletion {
  checkoutSessionId: string;
  paymentIntentId:   string;
  /** `latest_payment_attempt.payment_method.type` (pay_now, card, …). */
  paidType:          string | null;
  amountPaidSgd:     number;
  feeSgd:            number | null;
}
 
/**
 * Complete a paid Airwallex order. Throws DuplicateOrderError (from the shared
 * pipeline) when another path already claimed it — callers treat that as success.
 */
export async function processCompletedAirwallexPayment(
  pending:         PendingPaymentRow & { reference_number?: string },
  referenceNumber: string,
  c:               AirwallexCompletion,
  supabaseAdmin:   SupabaseClient,
  internalBaseUrl: string,
  awaitReceipt = true,
): Promise<{ orderId: string }> {
  const fields: CompletedPaymentFields = {
    referenceNumber,
    hitpayPaymentId:  c.paymentIntentId,
    paymentRequestId: c.checkoutSessionId,
    paymentMethod:    c.paidType ?? "unknown",
    amountPaid:       c.amountPaidSgd,
  };
  const { orderId } = await processCompletedHitPayPayment(
    pending, fields, supabaseAdmin, internalBaseUrl, awaitReceipt, AIRWALLEX_ORDER_GATEWAY,
  );
 
  if (c.feeSgd != null) await recordAirwallexFee(supabaseAdmin, orderId, c.feeSgd);
 
  const chosen = String((pending.order_data as { paymentMethod?: unknown }).paymentMethod ?? "");
  if (paymentMethodMismatch(chosen, c.paidType)) {
    const phones = getAdminPhonesCsv();
    if (phones) {
      await sendAdminAlert(phones, "high", "Airwallex",
        `Order ${orderId.slice(0, 8)} was priced for ${AIRWALLEX_METHOD_MAP[chosen]} but paid by ${c.paidType}. ` +
        `Check the processing fee.`).catch(() => {});
    }
  }
  return { orderId };
}
 
/** Gateway fee — best-effort, separate UPDATE, never via the RPC, never overwrites. */
export async function recordAirwallexFee(supabaseAdmin: SupabaseClient, orderId: string, feeSgd: number): Promise<boolean> {
  const { error } = await supabaseAdmin
    .from("orders")
    .update({ payment_gateway_fee_sgd: feeSgd })
    .eq("id", orderId)
    .is("payment_gateway_fee_sgd", null);
  if (error) console.warn(`[airwallex-order] fee capture failed order=${orderId}: ${error.message}`);
  return !error;
}
