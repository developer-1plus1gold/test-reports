/**
 * Single source of truth for the 1+1 GOLD typographic logo spec.
 *
 * Mukesh's standing rule (memory: feedback_logo_use_receipt_pdf_spec):
 * NEVER recreate the logo. Always derive from this spec. Recreating
 * from memory has been one of his biggest recurring time-wasters across
 * the project — the "+" font, the muted-gold colour for GOLD, the
 * letter-spacing on GOLD all get mis-remembered when rebuilt by hand.
 *
 * The three logo contexts:
 *   1. React component (web UI)         → src/components/LogoMark.tsx
 *   2. HTML string  (slips, emails)     → src/lib/branding/logo-html.ts  (getLogoHtml)
 *   3. React-PDF    (receipts, agreements) → src/lib/branding/logo-pdf.tsx (Logo)
 *
 * All three import their proportions/colors from THIS file. Change the
 * spec here and every surface updates together.
 *
 * Brand spec verified against the canonical source (receipt PDF approved
 * by Mukesh + LogoMark.tsx on the home page). DO NOT change ratios or
 * colors here without checking those two anchor references first.
 */
 
export const LOGO_COLORS = {
  /** Primary brand gold — used for "1", "+", "1" numerals + most accent UI. */
  numeral: "#C9A84C",
  /** Muted brand gold — used ONLY for the "GOLD" wordmark in the stacked
   *  logo (per brand packaging spec). Using primary gold here is the
   *  most common mis-render. */
  wordmark: "#A07830",
} as const;
 
/**
 * Size proportions, derived from `size` (the "1" numeral fontSize).
 * - plus    = size × 0.77 — "+" sits visually balanced next to the numerals
 * - word    = size × 0.50 — "GOLD" wordmark legibility floor for screens.
 *             The receipt-PDF (A4) historically used 0.31, but at smaller
 *             surfaces 0.50 keeps the wordmark readable. 0.50 is the
 *             default; PDF callers can override if they need 0.31.
 * - spacing = size × 0.065 — letter-spacing on "GOLD"
 * - plusMargin = size × 0.045 — horizontal margin around "+" (HTML only;
 *             React-PDF uses inline text-run so no margin needed).
 */
export const LOGO_RATIOS = {
  plus:       0.77,
  word:       0.50,
  wordPrint:  0.31,   // tighter ratio used by full-page A4 PDFs
  spacing:    0.065,
  plusMargin: 0.045,
} as const;
 
/**
 * Font weights — same across all surfaces.
 *   numeral 700 (Cinzel Bold), plus 300 (Cormorant Light), GOLD 600/400.
 *   LogoMark uses 600 for GOLD (screen weight); receipt PDF uses 400.
 *   600 has slightly better legibility at small sizes; 400 is the brand
 *   packaging spec. Default to 600 to match the home-page logo.
 */
export const LOGO_WEIGHTS = {
  numeral: 700,
  plus:    300,
  word:    600,
} as const;
 
/**
 * Font families. NOTE: family-name string here MUST match the family
 * registered in the consuming surface:
 *   - HTML (Google Fonts CDN):  "Cinzel" / "Cormorant Garamond"
 *   - React-PDF (Font.register): "Cinzel" / "Cormorant"
 * The HTML emitter and PDF emitter each handle the family-name mapping.
 */
export interface LogoFamilies {
  cinzel:    string;
  cormorant: string;
}
 
export const LOGO_FAMILIES_HTML: LogoFamilies = {
  cinzel:    "'Cinzel', serif",
  cormorant: "'Cormorant Garamond', 'Cormorant', serif",
};
 
export const LOGO_FAMILIES_PDF: LogoFamilies = {
  cinzel:    "Cinzel",
  cormorant: "Cormorant",
};
 
export interface LogoDerivedSizes {
  numeral:   number;
  plus:      number;
  word:      number;
  spacing:   number;
  plusMargin: number;
  lineH:     number;
  marginTop: number;
}
 
export function deriveLogoSizes(size: number, opts?: { print?: boolean }): LogoDerivedSizes {
  const wordRatio = opts?.print ? LOGO_RATIOS.wordPrint : LOGO_RATIOS.word;
  return {
    numeral:    size,
    plus:       size * LOGO_RATIOS.plus,
    word:       size * wordRatio,
    spacing:    size * LOGO_RATIOS.spacing,
    plusMargin: size * LOGO_RATIOS.plusMargin,
    lineH:      size >= 20 ? 1 : 0.85,
    marginTop:  size >= 20 ? 3 : 1,
  };
}
