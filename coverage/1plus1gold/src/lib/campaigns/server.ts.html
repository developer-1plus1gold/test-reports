/**
 * Everything the campaign panels need (optionally for one placement), for an
 * optional signed-in customer. Anonymous callers (the server-rendered `/`,
 * `/shop` and the app embed) are served entirely from the L1 snapshot; only the
 * signed-in `myRegistration` read goes to the database.
 */
export async function loadShopCampaigns(sb: SupabaseClient, customerId: string | null, now: Date = new Date(), placement?: CampaignPlacement): Promise<PublicCampaign[]> {
  const snapshot = await fetchLiveCampaignSnapshot(sb);
  const campaigns = snapshot.rows.filter((c) => campaignVisible(c, now, placement));
  if (campaigns.length === 0) return [];
  const mine = customerId
    ? await fetchMyRegistrations(sb, customerId, campaigns.map((c) => c.id))
    : ({} as Record<string, MyRegistration>);
  const byId = new Map(snapshot.products.map((p) => [p.id, p]));
  return campaigns.map((c) => toPublicCampaign(
    c,
    c.product_ids.map((id) => byId.get(id)).filter((p): p is CampaignProduct => !!p),
    mine[c.id] ?? null,
    now,
  ));
}
