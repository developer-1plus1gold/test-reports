const ms = (iso: string | null | undefined) => (iso ? new Date(iso).getTime() : NaN);
 
/**
 * Where a campaign is in its life at `now`.
 *  - registering: the Register button is live
 *  - redeeming:   registration has closed but issued codes still work
 *  - upcoming / ended: nothing actionable
 * `archived` reads as ended; `draft` is never shown to customers.
 */
export function campaignPhase(c: Pick<CampaignRow, "status" | "register_starts_at" | "register_ends_at" | "redeem_ends_at">, now: Date = new Date()): CampaignPhase {
  if (c.status === "draft") return "draft";
  if (c.status === "archived") return "ended";
  const t = now.getTime();
  if (t < ms(c.register_starts_at)) return "upcoming";
  if (t < ms(c.register_ends_at)) return "registering";
  if (c.redeem_ends_at && t < ms(c.redeem_ends_at)) return "redeeming";
  return "ended";
}
 
/**
 * Whether the campaign should render at `now` — anywhere, or on one placement
 * (the full panel on Home, the compact banner on Shop).
 */
export function campaignVisible(c: CampaignRow, now: Date = new Date(), placement?: CampaignPlacement): boolean {
  const p = campaignPhase(c, now);
  if (p !== "registering" && p !== "redeeming") return false;
  if (placement === "home") return c.show_on_home;
  if (placement === "shop") return c.show_on_shop;
  return c.show_on_home || c.show_on_shop;
}
 
/** The instant after which the panel disappears (and a linked carousel slide should end). */
export function campaignEndsAt(c: Pick<CampaignRow, "register_ends_at" | "redeem_ends_at">): string {
  if (c.redeem_ends_at && ms(c.redeem_ends_at) > ms(c.register_ends_at)) return c.redeem_ends_at;
  return c.register_ends_at;
}

        (c.offer_max_discount_sgd != null ? ` (up to ${sgd(Number(c.offer_max_discount_sgd))})` : "");

/** Add whole days to a `YYYY-MM-DD` string. */
export function addDays(ymd: string, days: number): string {
  const t = Date.parse(`${ymd}T00:00:00Z`) + days * 86_400_000;
  return new Date(t).toISOString().slice(0, 10);
}

/** Per-date demand: how many registrants want delivery on each day (+ slot). */
export function deliveryDemand(regs: Array<{ preferred_delivery_date: string | null; preferred_delivery_slot: string | null }>): Array<{ date: string; daytime: number; evening: number; unspecified: number; total: number }> {
  const by = new Map<string, { daytime: number; evening: number; unspecified: number }>();
  for (const r of regs) {
    if (!r.preferred_delivery_date) continue;
    const e = by.get(r.preferred_delivery_date) ?? { daytime: 0, evening: 0, unspecified: 0 };
    if (r.preferred_delivery_slot === "daytime") e.daytime++;
    else if (r.preferred_delivery_slot === "evening") e.evening++;
    else e.unspecified++;
    by.set(r.preferred_delivery_date, e);
  }
  return [...by.entries()].sort(([a], [b]) => a.localeCompare(b))
    .map(([date, e]) => ({ date, ...e, total: e.daytime + e.evening + e.unspecified }));
}
 
/** Per-product demand: registrants and total pieces wanted, per product. */
export function productInterest(regs: Array<{ product_ids: string[] | null; quantities?: number[] | null }>): Map<string, { registrants: number; pieces: number }> {
  const m = new Map<string, { registrants: number; pieces: number }>();
  for (const r of regs) {
    (r.product_ids ?? []).forEach((p, i) => {
      const e = m.get(p) ?? { registrants: 0, pieces: 0 };
      e.registrants++;
      e.pieces += Math.max(1, Number(r.quantities?.[i] ?? 1));
      m.set(p, e);
    });
  }
  return m;
}
