/**
 * Didit integration for KYC identity verification.
 *
 * Replaced Sumsub, which cost US$325/month flat against a
 * volume of two real verifications in four months. Didit bills per check —
 * ID Verification, Face Match and Device & IP are free under 500/month, so a
 * KYC + AML session costs US$0.42 (Liveness $0.15 + AML screening $0.27).
 *
 * Environment variables (set per Vercel project — DEV holds the SANDBOX
 * application's credentials, PROD holds the LIVE application's):
 *   DIDIT_API_KEY         — `x-api-key` for every verification.didit.me call
 *   DIDIT_WORKFLOW_ID     — the workflow a session runs (must include AML Screening)
 *   DIDIT_WEBHOOK_SECRET  — per-destination shared key for inbound webhook HMAC
 *
 * ── Two things about this API that are easy to get wrong ────────────────────
 *
 * 1. SANDBOX IS NOT A DIFFERENT HOST. Live and sandbox are separate Didit
 *    *Applications*, Stripe-test-mode style, distinguished only by which API
 *    key you send. Both talk to https://verification.didit.me. There is
 *    deliberately no DIDIT_API_URL env var: a base-URL switch is an invitation
 *    to point production at sandbox one day. The key is the only selector, and
 *    the webhook envelope's `environment` field is what we assert against.
 *
 * 2. STATUS IS SPELLED TWO WAYS. Webhook envelopes and decision payloads use
 *    Title Case ("In Review"); the list-sessions `status` query filter uses
 *    UPPER_SNAKE ("IN_REVIEW"). `normaliseSessionStatus` accepts both so a
 *    caller never has to remember which surface it came from.
 */

  IN_PROGRESS: "In Progress",
  IN_REVIEW: "In Review",
  APPROVED: "Approved",
  DECLINED: "Declined",

/**
 * Accept either spelling of a session status and return the canonical Title
 * Case form, or null if it is a value we do not know.
 *
 * Returning null rather than a default is deliberate: an unrecognised status
 * used as a map key is a crash waiting for its first unknown input, and a
 * silent fallback to a benign value would let a new terminal state be treated
 * as "nothing happened". Callers must handle null explicitly.
 */
export function normaliseSessionStatus(raw: string | null | undefined): DiditSessionStatus | null {
  if (!raw) return null;
  const key = raw.trim().toUpperCase().replace(/[\s-]+/g, "_");
  return STATUS_BY_NORMALISED[key] ?? null;
}

async function diditFetch<T>(
  path: string,
  init: { method: "GET" | "POST"; body?: unknown } = { method: "GET" },
): Promise<T> {
  const res = await fetch(BASE_URL + path, {
    method: init.method,
    headers: {
      "x-api-key": apiKey(),
      "Accept": "application/json",
      ...(init.body !== undefined ? { "Content-Type": "application/json" } : {}),
    },
    ...(init.body !== undefined ? { body: JSON.stringify(init.body) } : {}),
  });
 
  if (!res.ok) {
    // Body is echoed for diagnosis but the API key never appears in it — we send
    // it as a header, and Didit does not reflect request headers in errors.
    const text = await res.text().catch(() => "");
    throw new Error(`Didit ${init.method} ${path} failed: ${res.status} ${text.slice(0, 500)}`);
  }
  return res.json() as Promise<T>;
}

/**
 * True only for an https URL on Didit's own domain.
 *
 * ▶ WHY THIS EXISTS AT ALL. `createDiditSession` returns a URL that we then
 * send a signed-in customer's browser to, and that URL arrives in a THIRD
 * PARTY's response body. If that response were ever tampered with — a
 * compromised upstream, a hijacked DNS answer, a proxy in the middle — we
 * would be the ones handing our own authenticated customer to an arbitrary
 * site, from our own button, with our brand's trust behind it, to photograph
 * their passport and take a selfie. That is the exact shape of attack a
 * phishing page wants and cannot normally obtain.
 *
 * Pinned at the library boundary rather than at a route so that every caller
 * inherits it and a future second caller cannot forget.
 */
export function isDiditVerificationUrl(raw: string): boolean {
  let u: URL;
  try {
    u = new URL(raw);
  } catch {
    return false;
  }
  if (u.protocol !== "https:") return false;
  const host = u.hostname.toLowerCase();
  // Exact domain, or a subdomain of it. Never a suffix match on the whole
  // string — "evildidit.me" ends with "didit.me" but is a different domain.
  return host === "didit.me" || host.endsWith(".didit.me");
}
 
export async function createDiditSession(opts: {
  vendorData: string;
  callbackUrl: string;
  workflowId?: string;

  const raw = await diditFetch<RawSession>("/v3/session/", {
    method: "POST",
    body: {
      workflow_id: workflowId,
      vendor_data: opts.vendorData,
      callback: opts.callbackUrl,
    },
  });
 
  if (!raw.session_id || !raw.url) {
    throw new Error("Didit create session returned no session_id/url");

/**
 * Find the session Didit holds for one of our customers.
 *
 * The recovery path: it answers "does Didit already hold an approval for this
 * person?" when our own row says otherwise. Returns null when they have never
 * started one.
 *
 * ▶ `expectedSessionId` IS REQUIRED, AND THAT IS THE POINT. Callers use this
 * result to stamp `kyc_status: approved` with a fresh 2.5-year
 * `kyc_expires_at`, so "whatever session came back first" is not a safe
 * default — Didit retention is 5 YEARS (operator decision 2026-09-11; unlimited
 * before that), which still outlives our 2.5-year KYC validity by 2.5 years, so
 * a years-old approved session is still sitting there to be found. Bounding the
 * window did not remove this hazard; it only made it finite. Passing the
 * `kyc_reference` we are actually waiting on makes the answer specific to the
 * session WE opened, which is the only one a dropped webhook can be about.
 * Making it required rather than optional means a future caller has to make
 * that choice deliberately instead of inheriting the loose behaviour.
 *
 * Pass `null` only where "any latest session" is genuinely what you mean.
 */
export async function getLatestSessionByVendorData(
  vendorData: string,
  expectedSessionId: string | null,
): Promise<DiditDecision | null> {
  const res = await diditFetch<{ results?: DiditDecision[]; sessions?: DiditDecision[] }>(
    `/v3/sessions?vendor_data=${encodeURIComponent(vendorData)}`,
  );
  // Their list endpoint has used both keys across versions; accept either
  // rather than silently returning null if the envelope is renamed.
  const rows = res.results ?? res.sessions ?? [];
  if (expectedSessionId) {
    // Scan rather than trust position: the list is not documented as ordered,
    // so "the newest" is an assumption and the session id is a fact.
    return rows.find((r) => r.session_id === expectedSessionId) ?? null;
  }
  return rows.length > 0 ? rows[0] : null;
}

/**
 * Reproduce Python's
 *   json.dumps(obj, sort_keys=True, separators=(",", ":"), ensure_ascii=False)
 * which is what Didit signs for `X-Signature-V2`.
 *
 * This is a re-implementation of another language's serialiser, so it is the
 * FALLBACK, not the primary: number formatting and Unicode escaping are exactly
 * where the two runtimes can disagree. `X-Signature` over the raw bytes is tried
 * first and needs no re-encoding at all, which is why it is preferred here even
 * though Didit's docs recommend V2 — their recommendation targets frameworks
 * that re-encode the body before you see it. Next.js route handlers hand us the
 * original bytes via `req.text()`, so we keep them.
 */
export function canonicalJson(value: unknown): string {
  if (value === null || typeof value !== "object") return JSON.stringify(value) ?? "null";
  if (Array.isArray(value)) return `[${value.map(canonicalJson).join(",")}]`;
  const obj = value as Record<string, unknown>;
  const keys = Object.keys(obj).sort();
  return `{${keys.map((k) => `${JSON.stringify(k)}:${canonicalJson(obj[k])}`).join(",")}}`;
}

/**

 *
 * A sandbox event reaching production would write simulated verification data
 * onto a real customer, so this is a hard reject rather than a warning. An
 * envelope with no `environment` field is accepted — older senders omit it, and
 * refusing those would drop genuine traffic on a field we cannot require.
 */
export function environmentMatches(
  envelope: Pick<DiditWebhookEnvelope, "environment">,
  isProdRuntime: boolean,
): boolean {
  if (!envelope.environment) return true;
  return envelope.environment === (isProdRuntime ? "live" : "sandbox");
