/**
 * Order value at which customer-facing KYC (identity verification) is required (SGD).
 *
 * Set to S$20,000 — the CDD trigger under Singapore's PSPM Act (the regulatory
 * line for precious-metals dealers). Below this, no identity check is forced, so
 * mid-size orders (e.g. 20g+) are not gated.
 *
 * NOTE: this is the *customer-facing friction* threshold (blocks checkout +
 * self-limit cap). It is intentionally SEPARATE from AML_MONITORING_THRESHOLD_SGD
 * below — invisible backend risk-scoring runs from a LOWER value so we keep
 * vigilance on orders that no longer hit KYC.
 */
export const KYC_THRESHOLD_SGD = 20000;
 
/**
 * Order value at/above which the INVISIBLE backend AML monitoring runs (SGD).
 *
 * This restricts NO orders — it only triggers risk-scoring (`runComplianceCheck`
 * → /api/compliance/risk-rating) and transaction monitoring (`monitorTransaction`,
 * which writes aml_checks + compliance_cases). Deliberately kept BELOW the KYC
 * gate (S$5,000) so orders in the S$5k–S$20k band — which no longer require
 * identity verification — are still watched for structuring / velocity. Regulators
 * never object to more monitoring; this preserves the caution we want without the
 * customer friction. (The structuring/CTR bands inside monitorTransaction are
 * separately anchored to the S$20,000 CTR_THRESHOLD_SGD.)
 */
export const AML_MONITORING_THRESHOLD_SGD = 5000;
 
/** How long a KYC approval stays valid (milliseconds) */
export const KYC_VALIDITY_MS = 2.5 * 365 * 24 * 60 * 60 * 1000; // 2.5 years
 
/** KYC statuses that count as "verified and current" */
export const KYC_APPROVED_STATUSES = ["approved"] as const;
 
export type KycStatus = "not_required" | "pending" | "on_hold" | "approved" | "rejected" | "expired";
 
/**
 * Sumsub rejectLabels that mean "do not let this customer retry."
 *
 * These map to regulated AML / sanctions / fraud signals — letting a
 * customer with a sanctions hit re-submit and (potentially) sneak through
 * a less-strict moderator is a compliance breach. The /account page hides
 * the retry CTA and shows "Contact us" instead when any of these appears
 * in `profiles.kyc_reject_labels`.
 *
 * Recoverable labels (BLURRY_PHOTO, DOCUMENT_DAMAGED, INCOMPLETE_DOCUMENT,
 * WRONG_USER_REGION, etc.) are NOT in this list — those customers can
 * retry and Sumsub may approve them on resubmission.
 *
 * Reference: Sumsub rejectLabels catalogue.
 */
export const KYC_FATAL_REJECT_LABELS = [
  "BLOCKLIST",
  "BLACKLIST",
  "FRAUD_LIST",
  "CRIMINAL",
  "SANCTIONS",
  "WATCH_LIST",
  "PEP",
  "ADVERSE_MEDIA",
  "FRAUDULENT_PATTERNS",
  "PERMANENT_REJECTION",
  "REGULATIONS_VIOLATIONS",
] as const;
 
/**
 * Returns true if the rejection reasons mark this case as non-retryable.
 * Either: reviewRejectType is explicitly FINAL, OR any of the rejectLabels
 * matches a fatal label.
 */
export function isKycRejectionFatal(
  rejectType: string | null | undefined,
  rejectLabels: string[] | null | undefined,
): boolean {
  if (rejectType === "FINAL") return true;
  if (!rejectLabels?.length) return false;
  const fatalSet: Set<string> = new Set(KYC_FATAL_REJECT_LABELS);
  return rejectLabels.some((l) => fatalSet.has(l));
}
