/**
 * The side effects that must follow a KYC status change, in ONE place.
 *
 * ▶ WHY THIS EXISTS. A verification outcome reaches us by two independent
 * routes: the Didit webhook (primary) and the kyc-reconcile cron (backstop).
 * Both compute the same `ProfilePatch`, but only the webhook was acting on
 * `notify` and `triggerRiskRating` — so a customer whose webhook was dropped
 * would be approved by the cron SILENTLY: no WhatsApp telling them they are
 * verified, and no onboarding_risk compliance case ever opened.
 *
 * That asymmetry was survivable under Sumsub, which retries a failed webhook
 * for a long time, so the cron almost never had to act. It is not survivable
 * under Didit, which retries TWICE (~1 min, ~4 min) and then drops the
 * delivery permanently — for a Didit customer the cron is not a safety net for
 * an unlucky day, it is the only thing that will ever notice. A missing
 * compliance case is a regulatory gap, not a cosmetic one.
 *
 * Both callers now go through here, so the two paths cannot drift again.
 */
 
import type { SupabaseClient } from "@supabase/supabase-js";
import type { ComplianceReview, ProfilePatch } from "@/lib/kyc/didit-profile";
import { notifyKycApproved, notifyKycRejected } from "@/lib/notifications/notify-customer";
import { emitComplianceDecision } from "@/lib/compliance/emit-decision";
import { getAdminPhonesCsv } from "@/lib/admin-team";
import { sendAdminAlert } from "@/lib/twilio";
import { redactId } from "@/lib/log-redact";
 
/**
 * How long we wait on the risk-rating dispatch before returning.
 *
 * ▶ THIS BOUND IS THE POINT, and it is a correctness bound rather than a
 * performance one. `/api/compliance/risk-rating` runs an LLM assessment plus
 * several database round-trips and a WhatsApp send — tens of seconds on a bad
 * day — and the caller that reaches here most often is the DIDIT WEBHOOK, whose
 * own function has a far shorter budget. Awaiting the rating to completion
 * therefore risked the webhook exceeding its own timeout and answering Didit
 * with a 5xx on a delivery that had, in fact, succeeded: the profile write is
 * already committed by the time we get here. Didit retries a failed delivery
 * exactly twice and then drops it forever, so burning both retries on a
 * successful write is the expensive mistake.
 *
 * We still AWAIT rather than firing and forgetting, because an un-awaited fetch
 * in a serverless invocation can be frozen before the request is ever sent —
 * and that would mean no compliance case, silently. Waiting a few seconds is
 * enough for the request to be delivered and accepted; the rating then runs to
 * completion in ITS OWN invocation and writes its own result, so nothing
 * downstream depends on us reading the response.
 */
const RISK_RATING_DISPATCH_TIMEOUT_MS = 5_000;
 
/** Five-year AML retention, matching every other compliance case we open. */
const RETAIN_MS = 5 * 365 * 24 * 60 * 60 * 1000;
 
/**
 * How each confidential decline is filed.
 *
 * An ongoing-monitoring hit outranks an onboarding one: that customer was
 * already approved and may have transacted, so it is escalated rather than
 * merely opened.
 */
const REVIEW_SHAPE: Record<
  ComplianceReview["kind"],
  { caseType: "onboarding_risk" | "ongoing_monitoring"; riskLevel: "high" | "critical"; caseStatus: "open" | "escalated"; headline: string }
> = {
  aml_decline: {
    caseType: "onboarding_risk",
    riskLevel: "high",
    caseStatus: "open",
    headline: "AML screening declined a verification",
  },
  ongoing_monitoring: {
    caseType: "ongoing_monitoring",
    riskLevel: "critical",
    caseStatus: "escalated",
    headline: "Ongoing AML monitoring flagged an APPROVED customer",
  },
};
 
/**
 * Fire the customer notification and the onboarding risk rating implied by a
 * patch. Never throws: these are follow-ups to a status change that has
 * already been committed, and failing them must not turn a successful
 * reconciliation into an error. Failures are logged loudly instead.
 */
export async function applyKycSideEffects(opts: {
  customerId: string;
  /** Didit session id — recorded on the compliance case as the applicant ref. */
  sessionId: string;
  patch: ProfilePatch;
  /** Log prefix of the calling route, e.g. "[didit/webhook]". */
  logPrefix: string;
  /**
   * Service-role client, for opening a compliance case. Optional only so the
   * callers that can never produce a `complianceReview` patch need not supply
   * one; when a review IS required and no client was given, that is logged
   * loudly rather than dropped.
   */
  client?: SupabaseClient;
}): Promise<void> {
  const { customerId, sessionId, patch, logPrefix, client } = opts;
 
  if (patch.complianceReview) {
    await openComplianceReview({
      client,
      customerId,
      sessionId,
      review: patch.complianceReview,
      logPrefix,
    });
  }
 
  if (patch.notify === "approved") {
    notifyKycApproved(customerId).catch((err) =>
      console.error(`${logPrefix} approved notify failed:`, err),
    );
  } else if (patch.notify === "rejected") {
    const labels = patch.updates.kyc_reject_labels as string[] | null;
    notifyKycRejected(customerId, labels?.length ? labels.join(", ") : undefined).catch((err) =>
      console.error(`${logPrefix} rejected notify failed:`, err),
    );
  }
 
  if (!patch.triggerRiskRating) return;
 
  if (!process.env.INTERNAL_API_KEY) {
    console.error(
      `${logPrefix} INTERNAL_API_KEY not set — skipping risk rating for customer=${redactId(customerId)}`,
    );
    return;
  }
 
  const siteUrl = process.env.INTERNAL_BASE_URL ?? "http://localhost:3000";
  try {
    await fetch(`${siteUrl}/api/compliance/risk-rating`, {
      method: "POST",
      headers: { "Content-Type": "application/json", "x-internal-key": process.env.INTERNAL_API_KEY },
      body: JSON.stringify({ customerId, applicantId: sessionId, triggerSource: "kyc_approved" }),
      signal: AbortSignal.timeout(RISK_RATING_DISPATCH_TIMEOUT_MS),
    });
  } catch (err) {
    // A timeout here is NOT a failure of the rating — see the constant's note.
    // It is logged distinctly from a real error so a run of timeouts is
    // visible as "we stopped waiting", not as "the rating did not happen".
    const timedOut = err instanceof Error && (err.name === "TimeoutError" || err.name === "AbortError");
    console.error(
      timedOut
        ? `${logPrefix} risk rating dispatched; stopped waiting after ${RISK_RATING_DISPATCH_TIMEOUT_MS}ms (it continues in its own invocation)`
        : `${logPrefix} risk rating trigger failed:`,
      timedOut ? "" : err,
    );
  }
}
 
/**
 * File a confidential decline: a compliance case, its Control Tower mirror, and
 * a WhatsApp to the Compliance Officer.
 *
 * ▶ NO AGENT RUNS HERE, deliberately. The materiality triage IPPC 6.1 describes
 * ("false positive -> auto-cleared; genuine concern -> case opened") is now
 * performed UPSTREAM by Didit's two-score model: a match scoring below the
 * match-score threshold is classified False Positive and excluded from the risk

 * would add latency and a second opinion nobody should act on — that decision
 * belongs to a human.
 *

 * insert hiccup must not turn a committed status change into a webhook retry
