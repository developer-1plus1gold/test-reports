/**
 * ONE definition of "a Didit session outcome landed on a profile".
 *
 * Four paths can learn a verification's result — the webhook, the nightly
 * kyc-reconcile backstop, the recovery inside /api/kyc/initiate, and the
 * return-path reconcile that runs when a customer comes back from the hosted
 * flow. Before this module each wrote its own update, and they had already
 * drifted: the webhook guarded its write and fired side effects only on a real
 * change, while the cron wrote unconditionally and notified every time. Two
 * paths landing the same outcome seconds apart therefore messaged the customer
 * twice.
 *
 * ▶ THE GUARD IS THE WHOLE POINT, AND IT IS NOT A TIDINESS CONCERN. Whichever
 * path arrives first does the work; the others match no row, write nothing and
 * return `changed: false`. Side effects hang off that flag, so exactly one
 * notification is sent no matter how many paths observe the same outcome — and
 * a duplicate can never re-stamp kyc_verified_at / kyc_expires_at, which would
 * silently extend a customer's verification validity.
 *
 * Extracted from the webhook rather than written fresh: its 25 tests are the
 * control, and they pass unchanged against this.
 */

export type ApplyOutcome =
  /** The write failed. The caller decides whether that is retryable. */
  | { ok: false; reason: "update_failed"; error: unknown }
  /** Nothing to write — an unrecognised status, or a state carrying no change. */
  | { ok: true; changed: false; reason: "no_state_change"; status: null }
  /** A row matched this state already. Not an error; the work was already done. */
  | { ok: true; changed: false; reason: "duplicate"; status: string }
  /** This call is the one that landed the outcome. Side effects have fired. */
  | { ok: true; changed: true; reason: "applied"; status: string };
