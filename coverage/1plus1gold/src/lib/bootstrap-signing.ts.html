/**
 * Ed25519 signing for the iOS on-device calculator bootstrap payload
 * (`/api/calculator/bootstrap`) — security defense-in-depth (phase row 13,
 * finding `bootstrap-payload-no-integrity-check-v1`).
 *
 * The bootstrap feeds the JavaScriptCore pricing kernel that computes the
 * customer's DISPLAY prices. The effective risk is LOW (display-only; the
 * actual charge is server-authoritative via priceOrder/deriveCart; TLS already
 * protects the wire), so this is a tamper-detection layer, not a live hole.
 *
 * Design:
 *   - Ed25519 (asymmetric), NOT HMAC: a bundled symmetric key would be
 *     extractable from the IPA and thus forgeable. The app embeds only the
 *     PUBLIC key; the private key lives server-side in BOOTSTRAP_SIGNING_KEY_B64.
 *   - We sign the EXACT response body bytes (the caller passes the serialized
 *     string) and return the signature + key id in headers. The app verifies
 *     over the raw bytes it received — so there is NO JSON-canonicalization
 *     contract to keep in sync (the #1 source of signature bugs).
 *   - Fail-SAFE on the server: if the key isn't configured (local dev without
 *     the secret) signing returns null and the route ships the body unsigned;
 *     the app fail-OPENs on a missing signature. A signing exception likewise
 *     degrades to unsigned rather than 500-ing the calculator.
 *
 * Key rotation: bump KEY_ID + ship the new public key in the app; the server
 * env swaps to the new private key. The app can carry multiple public keys
 * keyed by id during a rollover (not needed for v1).
 */

/**
 * Sign the exact response body string. Returns the base64 Ed25519 signature +
 * the key id, or null when no key is configured / signing fails (ship unsigned;
 * the client fail-opens). For Ed25519 the algorithm argument MUST be null — the
 * key type selects the scheme (RFC 8032).
 */
export function signBootstrapBody(body: string): { signature: string; keyId: string } | null {
  const key = getPrivateKey();
  if (!key) return null;
  try {
    const sig = edSign(null, Buffer.from(body, "utf8"), key);
    return { signature: sig.toString("base64"), keyId: BOOTSTRAP_KEY_ID };
  } catch {
    return null;
  }
}
 
/** Test-only: reset the memoised key so a test can swap the env between cases. */
export function __resetBootstrapKeyCache(): void {
  cachedB64 = undefined;
  cachedKey = null;
  warnedMissing = false;
}
