/**
 * Fit the printed Certificate of Authenticity onto ONE sheet (Mukesh 2026-11-27:
 * "COA — I always have to change settings here: page 1 only and 60% size").
 *
 * The certificate is laid out for a screen: a 512 px card that, in print mode, is
 * about 1.6× the height of an A4 page — so the browser split it over two sheets and
 * the operator shrank it by hand every time. We measure the card and scale it to the
 * printable area instead, so it prints at "Actual size" with no setting changed.
 *
 * The printable area is the SMALLER of A4 and US Letter (the one tray holds either),
 * less the 10 mm page margin set in globals.css, at the CSS 96 px per inch.
 */
 
const MM_TO_PX = 96 / 25.4;
const MARGIN_MM = 10;
const A4_MM     = { w: 210, h: 297 };
const LETTER_MM = { w: 215.9, h: 279.4 };
 
export const COA_PRINT_AREA_PX = {
  w: Math.floor((Math.min(A4_MM.w, LETTER_MM.w) - 2 * MARGIN_MM) * MM_TO_PX),
  h: Math.floor((Math.min(A4_MM.h, LETTER_MM.h) - 2 * MARGIN_MM) * MM_TO_PX),
};
 
/** Headroom for the browser's own rounding of a zoomed box (measured 2026-11-27:
 *  an exact fit still pushed the last line onto a second US Letter sheet). */
export const COA_FIT_HEADROOM = 0.96;
 
/** The zoom that fits a card of this size on one sheet: never enlarges, never 0. */
export function coaPrintZoom(cardWidthPx: number, cardHeightPx: number, area = COA_PRINT_AREA_PX): number {
  if (!(cardWidthPx > 0) || !(cardHeightPx > 0)) return 1;
  const z = Math.min(1, area.w / cardWidthPx, (area.h * COA_FIT_HEADROOM) / cardHeightPx);
  // Floor to 2 decimals so rounding can never push the card a pixel over the page.
  return Math.max(0.1, Math.floor(z * 100) / 100);
}
