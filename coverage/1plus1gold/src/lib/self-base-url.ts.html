/**
 * The base URL a cron uses to call one of our own routes.
 *
 * NEVER `req.nextUrl.origin` inside a cron: Vercel invokes crons on the
 * deployment's own *.vercel.app host, which Vercel Authentication protects — the
 * call is answered "401 Protected deployment" before it reaches our code. The
 * first TikTok backfill post (2026-11-17 11:30 UTC) failed exactly that way.
 * The production domain is not protected, so self-calls go there (the rule the
 * marketing-publisher cron has followed since it was written).
 */
export function selfBaseUrl(): string {
  return process.env.NEXT_PUBLIC_SUPABASE_URL?.includes("supabase.co")
    ? "https://1plus1gold.com"
    : "http://localhost:3000";
}
