/**
 * User-agent signals for the analytics bot filter (Signal A,
 * docs/analytics-bot-filter-design.md §3.2).
 *
 * Pure, no I/O. The raw User-Agent is NEVER stored — only the short reason
 * code returned by `classifyUserAgent` and the coarse family from `uaFamily`.
 *
 * Signal A only catches honest crawlers and headless tools. A crawler that
 * runs JavaScript with a normal browser UA (the 2026-10-19 one) is caught by
 * the burst classifier (Signal B, `classify_analytics_bursts`) instead.
 */
import { IOS_APP_USER_AGENT_MARKER } from "@/lib/ios-client-constants";
 
const IOS_UA = new RegExp(`^${IOS_APP_USER_AGENT_MARKER}/\\d+\\.\\d+`);
 
/**
 * Ordered: the first match becomes the reason token. Specific tool names come
 * before the generic `bot` so a reason reads `ua:headlesschrome`, not `ua:bot`.
 */
const BOT_TOKENS: ReadonlyArray<[token: string, re: RegExp]> = [
  ["headlesschrome", /headlesschrome/i],
  ["phantomjs", /phantomjs/i],
  ["puppeteer", /puppeteer/i],
  ["playwright", /playwright/i],
  ["lighthouse", /lighthouse/i],
  ["python-requests", /python-requests/i],
  ["curl", /\bcurl\//i],
  ["wget", /\bwget\//i],
  ["go-http-client", /go-http-client/i],
  ["axios", /\baxios\//i],
  ["facebookexternalhit", /facebookexternalhit/i],
  ["slurp", /slurp/i],
  ["spider", /spider/i],
  ["crawl", /crawl/i],
  // "bot" anywhere in a word (Googlebot, AhrefsBot, GPTBot, bingbot) — but not
  // the CUBOT phone brand, which real Android browsers carry in their UA.
  ["bot", /(?<!cu)bot/i],
];
 
/** Matched reason code (`ua:<token>`), or null for an ordinary client. */
export function classifyUserAgent(ua: string | null | undefined): string | null {
  const u = (ua ?? "").trim();
  if (!u) return "ua:empty";
  if (IOS_UA.test(u)) return null; // our own native app is never a bot
  for (const [token, re] of BOT_TOKENS) {
    if (re.test(u)) return `ua:${token}`;
  }
  return null;
}
 
export type UaFamily = "ios-app" | "bot" | "edge" | "samsung" | "chrome" | "firefox" | "safari" | "other";
 
/** Coarse client family — stored on every row so future rules can work retroactively. */
export function uaFamily(ua: string | null | undefined): UaFamily {
  const u = (ua ?? "").trim();
  if (IOS_UA.test(u)) return "ios-app";
  if (classifyUserAgent(u) !== null) return "bot";
  if (/\bEdg(?:e|A|iOS)?\//.test(u)) return "edge";
  if (/SamsungBrowser\//.test(u)) return "samsung";
  if (/\b(?:Firefox|FxiOS)\//.test(u)) return "firefox";
  if (/\b(?:Chrome|CriOS|Chromium)\//.test(u)) return "chrome";
  if (/\bSafari\//.test(u) || /AppleWebKit\//.test(u)) return "safari";
  return "other";
}
