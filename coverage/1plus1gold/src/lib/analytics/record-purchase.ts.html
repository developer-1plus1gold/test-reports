/**
 * Server-side purchase event + order attribution (Measurement spec §07/§08).
 *
 * Called at order creation. Emits the authoritative `purchase` analytics event
 * (server-side so revenue always reconciles with `orders` — no ad-blocker loss)
 * and stamps first-/last-touch channel + surface + landing page onto the order,
 * joining acquisition to revenue.
 *
 * ENTIRELY best-effort: every path is wrapped so analytics can NEVER break or
 * slow-fail order creation. If the visitor has no first-party id (opted out /
 * fresh) we simply skip — the order is still created normally.
 */
import { randomUUID } from "node:crypto";
import type { NextRequest } from "next/server";
import type { SupabaseClient } from "@supabase/supabase-js";
import { deriveSurface } from "./surface";
 
const UUID_RE = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;
 
interface SessionRow {
  session_id:       string;
  entry_channel:    string | null;
  entry_sub_source: string | null;
  surface:          string | null;
  landing_path:     string | null;
  started_at:       string;
}
