/**
 * Shared analytics contract — event taxonomy, funnel stages, and the wire
 * schema for POST /api/track. One source of truth for the client tracker
 * (src/lib/analytics/track.ts) and the ingest route.
 *
 * See the Measurement & Attribution spec §04 (events) + §08 (funnel).
 * First-party only: no PII in props — identity is the customer_id join.
 */
import { z } from "zod";
 
// ── Event taxonomy (§04) ─────────────────────────────────────────────────────
export const EVENT_NAMES = [
  "session_start",
  "page_view",
  "view_item",
  "add_to_cart",
  "remove_from_cart",
  "begin_checkout",
  "add_shipping_info",
  "add_payment_info",
  "purchase",
  "identify",
] as const;
export type EventName = (typeof EVENT_NAMES)[number];
 
// ── Funnel stage high-water mark (§08). The drop-off stage of a session is
//    MAX(stage) across its events. Non-funnel events map to 0. ────────────────
export const STAGE_BY_EVENT: Partial<Record<EventName, number>> = {
  view_item:         1,
  add_to_cart:       2,
  begin_checkout:    3,
  add_shipping_info: 4,
  add_payment_info:  5,
  purchase:          6,
};
export const MAX_FUNNEL_STAGE = 6;
 
/** 30-minute inactivity window defines a session (industry standard, §13). */
export const SESSION_WINDOW_MS = 30 * 60 * 1000;
 
export type Surface = "web" | "pwa" | "twa" | "ios";
export type Channel = "direct" | "organic" | "social" | "ai" | "email" | "referral" | "paid";
 
// ── Wire schema: what the client batches to /api/track ───────────────────────
// The client sends WHAT happened + minimal context. The server decides surface,
// channel and country from request headers (unforgeable) — never trusted here.
const utmSchema = z
  .object({
    source:   z.string().trim().max(120).optional(),
    medium:   z.string().trim().max(120).optional(),
    campaign: z.string().trim().max(200).optional(),
    term:     z.string().trim().max(200).optional(),
    content:  z.string().trim().max(200).optional(),
  })
  .strip();
 
export const trackEventSchema = z.object({
  event: z.enum(EVENT_NAMES),
  path:  z.string().trim().max(512).optional(),
  // Bounded, PII-free bag. Values are primitives only; the route re-validates size.
  props: z.record(z.string().max(60), z.union([z.string().max(512), z.number(), z.boolean(), z.null()])).optional(),
});
export type TrackEventInput = z.infer<typeof trackEventSchema>;
 
export const trackBodySchema = z.object({
  anonymousId: z.string().uuid(),
  sessionId:   z.string().uuid(),
  standalone:  z.boolean().optional(),          // PWA display-mode hint (§06)
  context: z
    .object({
      referrer:    z.string().trim().max(2048).optional(),
      landingPath: z.string().trim().max(512).optional(),
      utm:         utmSchema.optional(),
      // Reliable paid-ad platform, derived client-side from an ad-click id
      // (gclid/msclkid/ttclid → slug only, never the raw id). See channel.ts.
      adPlatform:  z.enum(["google_ads", "microsoft_ads", "tiktok_ads"]).optional(),
    })
    .optional(),
  // customerId is NOT accepted from the body — it's read from the auth session
  // server-side so it can't be spoofed to poison another customer's funnel.
  events: z.array(trackEventSchema).min(1).max(30),
});
export type TrackBody = z.infer<typeof trackBodySchema>;
