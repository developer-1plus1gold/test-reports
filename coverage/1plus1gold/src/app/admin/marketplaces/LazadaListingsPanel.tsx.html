"use client";
 
/**
 * Lazada products ↔ our products (step 2a), shown inside the Lazada connection card.
 * "Load Lazada products" reads every listing from Lazada; each row shows our proposed
 * product and a one-tap Confirm, or a picker to choose / change. Only rows that are
 * exact-SKU or Confirmed are ever used by the stock/price push and order import.
 */
import { useCallback, useEffect, useMemo, useState } from "react";
 
const BG       = "var(--color-bg)";
const GOLD     = "var(--color-gold)";
const OFFWHITE = "var(--color-offwhite)";
const MUTED    = "var(--color-muted)";
const RULE     = "var(--color-rule)";
const SUCCESS  = "var(--color-success)";
const fieldStyle: React.CSSProperties = { background: BG, color: OFFWHITE, border: `1px solid ${RULE}` };
 
interface Listing {
  seller_sku: string; name: string; marketplace_status: string | null;
  marketplace_qty: number | null; marketplace_price: number | null; marketplace_special_price: number | null;
  product_id: string | null; match_source: "sku" | "suggested" | "confirmed" | "none"; match_reason: string | null;
  last_seen_at: string;
  /** Not linked to a product still on our shop — may be deleted from Lazada. */
  removable: boolean;
}
interface Product { id: string; label: string }
 
const SOURCE_TEXT: Record<Listing["match_source"], { text: string; colour: string }> = {
  sku:       { text: "Matched by SKU", colour: SUCCESS },
  confirmed: { text: "Confirmed", colour: SUCCESS },
  suggested: { text: "Suggested — confirm", colour: GOLD },
  none:      { text: "Not mapped", colour: "#E88" },
};
 
export default function LazadaListingsPanel() {
  const [rows, setRows] = useState<Listing[] | null>(null);
  const [products, setProducts] = useState<Product[]>([]);
  const [busy, setBusy] = useState<string | null>(null);
  const [msg, setMsg] = useState<string | null>(null);
  /** The last Remove result, shown on its own row (the panel message is out of view on a long list). */
  const [rowMsg, setRowMsg] = useState<{ sku: string; ok: boolean; text: string } | null>(null);
 
  const load = useCallback(async (signal?: AbortSignal) => {
    try {
      const r = await fetch("/api/admin/marketplace/lazada/listings", { cache: "no-store", signal });
      Iif (!r.ok) return;
      const j = await r.json();
      setRows(j.listings); setProducts(j.products);
    } catch (e) {
      if ((e as Error).name !== "AbortError") setMsg("Could not load the Lazada products");
    }
  }, []);
  useEffect(() => { const c = new AbortController(); load(c.signal); return () => c.abort(); }, [load]);
 
  const label = useMemo(() => new Map(products.map((p) => [p.id, p.label])), [products]);
  const counts = useMemo(() => {
    const c = { ready: 0, suggested: 0, none: 0 };
    for (const r of rows ?? []) {
      if (r.match_source === "sku" || r.match_source === "confirmed") c.ready++;
      else if (r.match_source === "suggested") c.suggested++;
      else c.none++;
    }
    return c;
  }, [rows]);
 
  const sync = async () => {
    setBusy("sync"); setMsg(null);
    const r = await fetch("/api/admin/marketplace/lazada/listings", {
      method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ action: "sync" }),
    });
    const j = await r.json().catch(() => ({}));
    setBusy(null);
    setMsg(r.ok
      ? `Loaded ${j.result.listings} Lazada SKU${j.result.listings === 1 ? "" : "s"}: ${j.result.autoMatched} matched by SKU, ${j.result.suggested} suggested, ${j.result.unmapped} to pick.`
      : (j.error ?? "Could not load Lazada products"));
    load();
  };
 
  const map = async (sellerSku: string, productId: string | null) => {
    setBusy(sellerSku); setMsg(null);
    const r = await fetch("/api/admin/marketplace/lazada/listings", {
      method: "PUT", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ sellerSku, productId }),
    });
    setBusy(null);
    if (!r.ok) setMsg("Could not save that mapping");
    load();
  };
 
  const remove = async (r: Listing) => {
    if (!confirm(`Delete "${r.name || r.seller_sku}" from Lazada? This cannot be undone on Lazada.`)) return;
    setBusy(r.seller_sku); setMsg(null); setRowMsg({ sku: r.seller_sku, ok: true, text: "Removing…" });
    try {

        method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ action: "remove", sellerSku: r.seller_sku }),
      });
      const j = await res.json().catch(() => ({}));
      const text = res.ok ? `Removed ${r.seller_sku} from Lazada.` : `${j.error ?? "Could not remove that listing"}${j.correlationId ? ` (ref ${j.correlationId})` : ""}`;

      setMsg(text);
    } catch {
      setRowMsg({ sku: r.seller_sku, ok: false, text: "Network error — nothing was removed" });
    } finally {
      setBusy(null);
      load();
    }
  };
 
  return (
    <div className="rounded-lg p-3 space-y-3" style={{ background: BG }}>
      <div className="flex flex-wrap items-center justify-between gap-2">
        <div>
          <div className="font-medium" style={{ color: GOLD }}>Lazada products</div>

            <div className="text-xs" style={{ color: MUTED }}>

          style={{ background: "transparent", color: OFFWHITE, border: `1px solid ${RULE}` }}>

                <div className="flex flex-col md:flex-row md:items-center gap-2 md:gap-4">
                  <div className="min-w-0 md:w-1/2">
                    <div className="truncate" title={r.name}>{r.name || "(no title)"}</div>
                    <div className="text-xs" style={{ color: MUTED }}>
                      SKU {r.seller_sku}
                      {r.marketplace_qty != null && ` · ${r.marketplace_qty} on Lazada`}
                      {r.marketplace_price != null && ` · S$${Number(r.marketplace_price).toFixed(2)}`}
                    </div>
                  </div>
                  <div className="md:w-1/2 space-y-1">
                    <div className="text-xs" style={{ color: s.colour }}>
                      {s.text}{r.match_reason && r.match_source !== "confirmed" ? ` · ${r.match_reason}` : ""}
                    </div>
                    <div className="flex gap-2">
                      <select value={r.product_id ?? ""} disabled={busy !== null}
                        onChange={(e) => map(r.seller_sku, e.target.value || null)}
                        className="min-w-0 flex-1 rounded-lg px-2 py-1.5" style={fieldStyle} aria-label={`Our product for ${r.seller_sku}`}>
                        <option value="">— Not mapped —</option>
                        {products.map((p) => <option key={p.id} value={p.id}>{p.label}</option>)}
                      </select>
                      {r.match_source === "suggested" && r.product_id && (
                        <button onClick={() => map(r.seller_sku, r.product_id)} disabled={busy !== null}
                          className="px-3 py-1.5 rounded-lg text-sm font-medium disabled:opacity-50" style={{ background: GOLD, color: BG }}>
                          Confirm
                        </button>
                      )}
                    </div>
                    {r.removable && (
                      <button onClick={() => remove(r)} disabled={busy !== null}
                        className="px-3 py-1.5 rounded-lg text-xs disabled:opacity-50" style={{ background: "transparent", color: "#E88", border: `1px solid ${RULE}` }}>
                        {busy === r.seller_sku ? "Removing…" : "Remove from Lazada"}
                      </button>
                    )}
                    {rowMsg?.sku === r.seller_sku && (
                      <div className="text-xs" style={{ color: rowMsg.ok ? SUCCESS : "#E88" }}>{rowMsg.text}</div>
                    )}
                    {r.product_id && r.match_source === "suggested" && (
                      <div className="text-xs" style={{ color: MUTED }}>Proposed: {label.get(r.product_id) ?? "—"}</div>
                    )}
                  </div>
                </div>
              </li>
            );
          })}
        </ul>
      )}
    </div>
  );
}
