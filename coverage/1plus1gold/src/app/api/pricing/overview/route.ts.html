/**
 * GET /api/pricing/overview — read-API (data_source) for the iOS SDUI
 * our-pricing spec. Returns the dynamic "answer value" (premium range, e.g.
 * "+22% → +3%") used by the §2 "Our Answer" tile + the §3 curve summary card.
 * The charts are Path-B embeds that compute their own data, so this stays
 * minimal. Shares derivePricingDisplay with the page + the market-range embed.
 *
 * `summary` (2026-10-06) carries the live numbers for the three offering cards
 * and the Premium-or-Spread explainer — one payload for web AND iOS, so the two
 * cannot quote different figures. Percentages only.
 *
 * ▶ PERFORMANCE (2026-10-07, Mukesh: "the loading of the data is very bad").
 * The payload is several Supabase reads, and the lambda was in iad1 (functions are in sin1 now) while the
 * database is in ap-southeast-1 — every read is a trans-Pacific round trip,
 * and the summary used to be awaited only AFTER premiums + MAG had finished,
 * so a cold request paid two serial round-trip chains. Now:
 *   1. the summary starts as soon as the MAG rows arrive, concurrently with
 *      the premiums read (no second serial chain);
 *   2. the whole payload is cached in-process for 60s with in-flight dedupe,
 *      so only one request per minute per instance touches the database;
 *   3. a failed refresh serves the last good payload rather than blanking.
 *
 * ▶ NEVER OLD FIGURES (2026-11-15, Mukesh). The payload is keyed by the pricing
 * version (DB triggers bump it at commit on any price change, migration
 * 20261115g), which is the ETag. `Cache-Control: no-cache` makes the browser and
 * WebView revalidate on EVERY visit — a 304 with no body until a price changes —
 * and there is no CDN copy to go stale. (Was s-maxage=60 + swr=300, on top of
 * 5-minute premiums / catalog caches: up to ~6 minutes of old prices.)
 */

/** `If-None-Match` may list several tags, weak or strong. */
function matches(ifNoneMatch: string | null, etag: string): boolean {
  if (!ifNoneMatch) return false;
  return ifNoneMatch.split(",").some((t) => t.trim().replace(/^W\//, "") === etag || t.trim() === "*");
}
