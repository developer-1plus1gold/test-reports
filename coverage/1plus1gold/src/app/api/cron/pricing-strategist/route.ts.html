function sizeUuid(grams: number): string {
  const h = createHash("sha256").update(`premium_proposal:${grams}`).digest("hex");
  return `${h.slice(0, 8)}-${h.slice(8, 12)}-${h.slice(12, 16)}-${h.slice(16, 20)}-${h.slice(20, 32)}`;
}
