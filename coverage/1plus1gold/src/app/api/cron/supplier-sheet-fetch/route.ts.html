/**
 * supplier_sheet_fetcher — daily supplier-sheet freshness monitor (Pricing module).
 *
 * GET /api/cron/supplier-sheet-fetch  (Bearer CRON_SECRET)
 *
 * Reads each supplier's `fetch_mode` (set in admin):
 *   • MANUAL — if today's sheet hasn't been uploaded, OPEN a Control Tower reminder
 *     (approve): "download from <url> and upload on /admin/pricing/suppliers". The
 *     operator uploads manually (which runs the parser); the reminder auto-closes
 *     once today's sheet exists.
 *   • AUTO   — the day's PDF is expected to have been fetched by the browser-automation
 *     runtime (a scheduled GitHub Action using the stored credentials; not on Vercel).
 *     If today's sheet is still missing at cron time, the auto-fetch did NOT deliver →
 *     ESCALATE (pages) so the operator uploads manually as a fallback. When the
 *     auto-fetch runtime is provisioned + MTB creds are set, the sheet will be present
 *     and no decision is filed.
 *
 * ⏰ SCHEDULE — `0 9 * * *` UTC = **17:00 SGT, the same day it is about.**
 * It used to run `0 22 * * *` UTC, which is 06:00 SGT the FOLLOWING morning: the
 * reminder for day D arrived after day D had already closed in Singapore, so it
 * was never actionable and was hand-dismissed ten times in a row. Moved
 * 2026-08-23 (Mukesh's call) so it fires during the working day it refers to.
 * `today` is the UTC date, and at 09:00 UTC the UTC and SGT dates agree — that
 * alignment is what makes this hour correct; do not move it past 16:00 UTC or
 * the two dates diverge again.
 * ▶ Expect it to fire on days the sheet is uploaded LATE (the usual upload is
 * ~13:00–14:30 SGT, but ~6 of the last 20 landed after 17:00). That is harmless:
 * `runMonitor`'s backward pass auto-closes the reminder on the next tick once
 * the day no longer matches. Do NOT hand-dismiss these — they self-heal.
 *
 * This route never logs into the supplier itself (Vercel can't run a headless
 * browser) — it monitors freshness. The actual auto-login/download is the GitHub
 * Action, which POSTs the PDF to /api/admin/pricing/suppliers/sheets.
 */
 
import { NextRequest, NextResponse } from "next/server";
import { createClient } from "@supabase/supabase-js";
import { createHash } from "crypto";
import { verifyCronAuth } from "@/lib/cron-auth";
import { isProd } from "@/lib/env";
import { runMonitor, type MonitorMatch } from "@/lib/control-tower/automation-monitor";
 
const supabaseAdmin = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_KEY!,
);
 
function dayUuid(supplierId: string, dayIso: string): string {
  const h = createHash("sha256").update(`sheet_missing:${supplierId}:${dayIso}`).digest("hex");
  return `${h.slice(0, 8)}-${h.slice(8, 12)}-${h.slice(12, 16)}-${h.slice(16, 20)}-${h.slice(20, 32)}`;
}
 
export async function GET(req: NextRequest) {

  // Sheet-freshness monitoring is PROD-ONLY. DEV prices are intentionally kept
  // stale (we edit them freely while testing) and nobody uploads a daily DEV
  // sheet, so running here only manufactures "sheet not uploaded" reminders that
  // clutter the Control Tower queue — the same reasoning that gated the sibling
  // pricing crons (margin-monitor, pricing-strategist). PROD is unaffected.
  if (!isProd()) {
    return NextResponse.json({ ok: true, skipped: "non-prod" });
  }
 
  const today = new Date().toISOString().slice(0, 10);
 
  const { data: suppliers } = await supabaseAdmin
    .from("suppliers")
    .select("id, name, slug, fetch_mode, fetch_url")
    .eq("is_active", true);
 
  // Split matches by severity so manual reminders (approve) and auto failures
  // (escalate) are independent open/close monitors.
  const manualMissing: MonitorMatch[] = [];
  const autoMissing: MonitorMatch[] = [];
 
  for (const s of suppliers ?? []) {
    // 'none' = we don't expect a daily sheet from this supplier at all. Our
    // consignment supplier (BullionKing) is priced off consignment_terms at sale
    // time, not off a sheet, so reminding about a "missing" sheet is pure queue
    // noise — and a queue that cries wolf daily stops being read.
    if ((s.fetch_mode ?? "manual") === "none") continue;
 
    const { data: sheet } = await supabaseAdmin
      .from("supplier_price_sheets")
      .select("id").eq("supplier_id", s.id).eq("sheet_date", today).limit(1).maybeSingle();
    if (sheet) continue; // today's sheet present → nothing to remind/escalate (auto-close handles old ones)
 
    const where = s.fetch_url ? ` from ${s.fetch_url}` : "";
    if ((s.fetch_mode ?? "manual") === "auto") {
      autoMissing.push({
        id: dayUuid(s.id, today),
        summary: `${s.name}: auto-fetch did NOT deliver today's (${today}) price sheet — download${where} and upload on Supplier Sheets.`,
        payload: { supplier: s.slug, date: today, mode: "auto", page: "/admin/pricing/suppliers" },
      });
    } else {
      manualMissing.push({
        id: dayUuid(s.id, today),
        summary: `${s.name}: today's (${today}) price sheet not uploaded yet — download${where} and upload on Supplier Sheets, then it parses automatically.`,
        payload: { supplier: s.slug, date: today, mode: "manual", page: "/admin/pricing/suppliers" },
      });
    }
  }
 
  const manualRes = await runMonitor({ sb: supabaseAdmin, agentName: "supplier_sheet_fetcher", severity: "approve", entityType: "sheet_reminder", matches: manualMissing });
  const autoRes = await runMonitor({ sb: supabaseAdmin, agentName: "supplier_sheet_fetcher", severity: "escalate", entityType: "sheet_autofetch_failed", matches: autoMissing });
 
  return NextResponse.json({ ok: true, today, manual: manualRes, auto: autoRes });
}
