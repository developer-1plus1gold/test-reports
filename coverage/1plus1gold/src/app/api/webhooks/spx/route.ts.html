/**
 * POST /api/webhooks/spx
 * Receives SPX (Shopee Xpress) status updates. Contract: docs/spx-api-reference.md §Webhooks.
 *
 * SPX posts JSON and signs the body with HMAC-SHA256 in the `check-sign` header
 * (same method as outbound). We MUST return HTTP 200 within 3 s or SPX retries
 * then stops sending — so we ack fast and let side-effects run best-effort.
 *
 * Tracking Web Hook (4.1) payload (subset we use):
 *   { status, status_code, message, tracking_no, tracking_link, order_id, id, timestamp }
 *
 * ⚠️ Inbound trust model: ENFORCE-BY-DEFAULT (no feature flag). The exact inbound
 * signing input set is not fully documented (see spx-api-reference.md §Gaps), so
 * verifyWebhookSignature tries the two plausible constructions (app_id_ts_rnd_body
 * then body-only). An event that does NOT verify never mutates an order — interim
 * statuses are dropped outright; a terminal (delivered/failed) is applied only if
 * SPX's OWN authenticated trackOrder() independently confirms it (unforgeable). We
 * always ack 200 so a forger learns nothing and SPX doesn't mark our endpoint
 * failing. Once a real webhook confirms SPX's signing format every event verifies
 * and the trackOrder fallback stops being exercised. (There is deliberately no
 * "log-only" bypass env — the trust gate below is unconditional.)
 */

// Cap what the capture ledger persists per inbound webhook. The endpoint is public
// and records the raw body before signature verification (by design), so the body
// is attacker-controllable. Real SPX tracking payloads are a few hundred bytes;
// 64 KB comfortably covers any legitimate body (incl. a batch update) so a genuine
// webhook is NEVER truncated (truncation would break offline HMAC reconstruction —
// the whole point of the ledger), while still bounding an oversized malicious POST.
const MAX_CAPTURE_RAW_BODY = 65_536;
 
interface SpxWebhookPayload {
  status?: string;
  status_code?: string | number;
  // SPX sends TWO webhook shapes to the same URL (both signature-verified):
  //  (1) the Tracking Web Hook (4.1) — carries `status_code` + `status`.
  //  (2) an Order-Detail / ePOD webhook — carries the SAME numeric code under
  //      `status_code_name` (e.g. "4001"), an internal `tracking_code` +
  //      `tracking_code_name` (F980=Delivered, F100=SOC_Pickup_Done,
  //      F001=SOC_Pickup_Failed), `customer_order_id` (our order UUID,
  //      hyphenless) and the proof-of-delivery/pickup image lists.
  // Family (2) was 43% of all captured webhooks and — because it has no
  // `status_code` — was silently mis-mapped to `in_transit` (the mapSpxEvent
  // null-fallback), including 45 real `4001` Delivered events. Only the family-1
  // webhook + the 5-min poll kept those orders converging; a missed family-1
  // Delivered would have stranded the order at in_transit (no inventory-sold, no
  // WhatsApp) until the next poll tick. We now read the code from either field.
  status_code_name?: string | number;
  tracking_code?: string;
  tracking_code_name?: string;
  customer_order_id?: string;
  epod_list?: string[] | null; // proof-of-delivery images (family 2, on Delivered)
  message?: string;
  tracking_no?: string;
  tracking_link?: string;
  order_id?: string;
  id?: string;
  timestamp?: number;
}
 
/**
 * SPX's numeric status code, read from EITHER webhook family: the Tracking Web
 * Hook carries it as `status_code`, the Order-Detail/ePOD webhook as
 * `status_code_name`. Both are the same "1001"/"2001"/"4001" vocabulary that
 * SPX_STATUS_MAP keys on. Returns undefined only for envelopes carrying neither
 * (e.g. Order-Create-Feedback 4.3), which have no status to apply.
 */
function spxStatusCode(p: SpxWebhookPayload): string | number | undefined {
  return p.status_code ?? p.status_code_name;
}
