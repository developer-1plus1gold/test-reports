/**
 * POST /api/webhooks/razorpay
 *
 * Razorpay requires a 2xx within 5 seconds, retries for 24 h, then DISABLES
 * the webhook. Order creation can take longer than that, so this route:
 *   1. verifies X-Razorpay-Signature over the raw body (fails closed, 401),
 *   2. finds the pending row from the Payment Link in the payload,
 *   3. acks 200 immediately and reconciles in the background (waitUntil).
 * Reconcile re-reads the link + payment from Razorpay's API and completes via
 * the shared atomic claim, so a replayed or duplicated event (Razorpay is
 * at-least-once) is harmless. The status poll and the reconcile cron are the
 * fallbacks if this never arrives.
 */
import { NextRequest, NextResponse } from "next/server";
import { createClient } from "@supabase/supabase-js";
import { waitUntil } from "@vercel/functions";
import { verifyRazorpayWebhook } from "@/lib/razorpay";
import { RAZORPAY_PENDING_SELECT, reconcilePendingRazorpayPayment, type ReconcilableRazorpayPayment } from "@/lib/razorpay-reconcile";
 
export const runtime     = "nodejs";
export const maxDuration = 60;
 
const supabaseAdmin = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_KEY!,
);
 
const HANDLED = new Set(["payment_link.paid", "payment_link.expired", "payment_link.cancelled"]);
 
export async function POST(req: NextRequest) {
  const rawBody = await req.text();
  if (!verifyRazorpayWebhook(rawBody, req.headers.get("x-razorpay-signature"))) {
    console.warn("[razorpay/webhook] signature check failed");
    return NextResponse.json({ error: "Invalid signature" }, { status: 401 });
  }
 
  // eslint-disable-next-line @typescript-eslint/no-explicit-any
  let body: any;
  try { body = JSON.parse(rawBody); } catch {
    return NextResponse.json({ received: true });   // signed but unreadable — nothing to act on
  }
 
  const event = String(body?.event ?? "");
  const eventId = req.headers.get("x-razorpay-event-id") ?? "";
  if (!HANDLED.has(event)) {
    console.log(`[razorpay/webhook] event=${event} id=${eventId} — no action`);
    return NextResponse.json({ received: true });
  }
 
  const linkId = body?.payload?.payment_link?.entity?.id as string | undefined;

  const { data: pending } = await supabaseAdmin
    .from("razorpay_pending_payments")
    .select(RAZORPAY_PENDING_SELECT)
    .eq("razorpay_payment_link_id", linkId)
    .maybeSingle();
 
  if (!pending) {
    // Not ours (or a link created outside checkout). 200 so Razorpay does not retry it for a day.
    console.warn(`[razorpay/webhook] no pending row for link=${linkId} event=${event}`);
    return NextResponse.json({ received: true });
  }
 
  const internalBaseUrl = process.env.INTERNAL_BASE_URL ?? "http://localhost:3000";
  waitUntil(
    reconcilePendingRazorpayPayment(pending as ReconcilableRazorpayPayment, supabaseAdmin, internalBaseUrl)
      .then((r) => console.log(`[razorpay/webhook] event=${event} id=${eventId} ref=${pending.reference_number} → ${r.outcome}`))
      .catch((err) => console.error(`[razorpay/webhook] reconcile failed ref=${pending.reference_number}:`, err)),
  );
  return NextResponse.json({ received: true });
}
