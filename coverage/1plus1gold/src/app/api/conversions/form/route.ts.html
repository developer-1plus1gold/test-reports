/**
 * POST /api/conversions/form — the reactive data source for the iOS `convert`
 * SDUI surface (the Combibar conversion submission form). Given the target mix
 * (50g/100g counts) + sort order, returns the WHOLE form view model in one shot:
 * eligible splits, the greedy selection ("splits we'll use"), the fee/spot/stock
 * summary, and the saved pickup addresses.
 *
 * Auth: Bearer+cookies (getUserFromRequest). Reuses the SAME shared kernels as
 * the web /convert-to-combibar page — loadEligibleBars + selectSplits +
 * computeConversionTotals — so the form can't drift between platforms. The
 * CHARGE stays server-authoritative (/api/conversions/submit re-validates).
 */
 
import { NextResponse, type NextRequest } from "next/server";
import { createClient } from "@supabase/supabase-js";
import { getUserFromRequest } from "@/lib/auth/get-user";
import { loadEligibleBars } from "@/lib/conversions-eligible";
import { computeConversionTotals } from "@/lib/conversions-totals";
import {
  selectSplits, totalAvailableGrams, targetGramsFor,
  type ConversionSortBy,
} from "@/lib/conversion-select";
import { conversionMixLabel } from "@/lib/conversions";
 
const supabaseAdmin = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_KEY!,
);
 
const SORTS: ConversionSortBy[] = ["date_asc", "date_desc", "cost_desc", "cost_asc"];
 
export async function POST(req: NextRequest) {
  const auth = await getUserFromRequest(req);
  if (!auth) return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
  const userId = auth.user.id;
 
  let body: { target50g?: number; target100g?: number; sortBy?: string } = {};
  try { body = await req.json(); } catch { /* empty body = initial load */ }
  const count50g  = Math.max(0, Math.floor(Number(body.target50g  ?? 0)) || 0);
  const count100g = Math.max(0, Math.floor(Number(body.target100g ?? 0)) || 0);
  const sortBy: ConversionSortBy = SORTS.includes(body.sortBy as ConversionSortBy) ? (body.sortBy as ConversionSortBy) : "date_asc";
 
  const [bars, addrRes] = await Promise.all([
    loadEligibleBars(userId, supabaseAdmin).catch((e) => { console.error("[conversions/form] bars", e); return []; }),
    supabaseAdmin.from("addresses")
      .select("id, label, full_name, phone, line1, line2, postal_code, is_default")
      .eq("customer_id", userId)
      .order("is_default", { ascending: false }),
  ]);
 
  const available   = totalAvailableGrams(bars);
  const targetGrams = targetGramsFor(count50g, count100g);
  const selection   = selectSplits(bars, count50g, count100g, sortBy);
 
  // "Splits we'll use" — flatten the picks for the SDUI repeater (mirrors the
  // web row: label / "Order #id · date [· cost] [· N in active sales/conversions]"
  // / "{qty} × {g}g" + "= {tot}g"). page.tsx:336-360.
  const picks = selection.picks.map((p) => {
    const s = p.split;
    const date = new Date(s.order_date).toLocaleDateString("en-SG", { day: "numeric", month: "short", year: "numeric" });
    const ann: string[] = [];
    Eif ((s.quantity_in_buybacks ?? 0) > 0) ann.push(`${s.quantity_in_buybacks} in active sales`);
    Eif (s.quantity_converted > 0) ann.push(`${s.quantity_converted} in active conversions`);
    return {
      id:              s.order_item_id,
      label:           s.product_label || `${s.grams_each}g split`,
      orderMeta:       `Order #${s.order_id.slice(0, 8).toUpperCase()} · ${date}`,
      costSgd:         s.cost_per_g,            // number | null → currency_amount
      hasCost:         s.cost_per_g != null,
      annotations:     ann.length ? ` · ${ann.join(" · ")}` : "",
      hasAnnotations:  ann.length > 0,
      qtyLabel:        `${p.quantity} × ${s.grams_each}g`,
      totalGramsLabel: `= ${p.quantity * s.grams_each}g`,
    };
  });
 
  // Totals only when the target is reachable (matches the web preview gate).
  const ready = targetGrams > 0 && selection.shortfall === 0;
  const totals = ready ? await computeConversionTotals(targetGrams, supabaseAdmin) : null;
 
  const items = selection.picks.map((p) => ({ order_item_id: p.split.order_item_id, quantity: p.quantity }));
 
  return NextResponse.json(
    {
      eligibleEmpty:        bars.length === 0,
      totalAvailableLabel:  `${available}g available in splits`,
      maxCount50g:          Math.floor(available / 50),
      maxCount100g:         Math.floor(available / 100),
      targetGrams,
      hasShortfall:         selection.shortfall > 0,
      // page.tsx:306 — appended after "Total needed: {g}g".
      shortfallLabel:       selection.shortfall > 0 ? ` — short by ${selection.shortfall}g (reduce target or buy more splits first)` : "",
      picks,
      hasPicks:             picks.length > 0,
      summary: {
        ready,
        valid:            !!totals?.valid && (totals?.enabled ?? false),
        splitsInLabel:    targetGrams > 0 ? `${targetGrams}g` : "—",
        mixLabel:         conversionMixLabel(count100g, count50g),
        conversionFeeSgd: totals?.conversionFeeSgd ?? 0,
        spotPerG:         totals?.spotSgdPerG ?? null,
        stockOk:          totals?.combibarStockOk ?? true,
        disabled:         totals ? !totals.enabled : false,
      },
      savedAddresses: addrRes.data ?? [],
      // The exact selection the submit must POST (also recomputed server-side
      // on submit, so this is a convenience echo, not the source of truth).
      items,
    },
    { headers: { "Cache-Control": "private, no-store" } },
  );
}
