import { getAdminPhonesCsv } from "@/lib/admin-team";
import { barOwedToParticipant, owedBreakdown } from "@/lib/gold-yield/settlement";
 
const supabaseAdmin = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_KEY!

/**
 * POST /api/gold-yield/request-payout
 *
 * Records a payout REQUEST and alerts an operator. It does not move money —
 * the operator records the payout itself through the admin panel.
 *
 * ▶ IT USED TO QUOTE THE WRONG FIGURE, AND THE CAUSE IS A KNOWN SHAPE
 * (operator, 2026-09-22: "the whatsapp message for payout is also wrong with
 * incorrect $ amounts"). It summed `participant_share` and nothing else:
 *
 *     const totalEarned = sales.reduce((s, x) => s + Number(x.participant_share), 0);
 *
 * That is their PROFIT. It omits the returned PRINCIPAL, which on a hedged bar
 * is the larger of the two by an order of magnitude — so a participant asked
 * for a fraction of what they were owed while the admin screen, which has used
 * `barOwedToParticipant` since it was written, showed the real number. Exactly
 * the divergence `settlement.ts` records for the exit gate and the
 * exit-complete message: "the gate summed both, the message summed the profit
 * share alone". This route was the last copy still doing it.
 *
 * ▶ IT ALSO LEFT NO TRACE, WHICH IS WHY THE BUTTON NEVER DISARMED. The previous
 * header said so outright — "Does NOT create a payout row" — so a request was a
 * WhatsApp and nothing else. Nothing on the next page load could know one had
 * been made. It now writes `gold_yield_payout_requests`, which the dashboard
 * reads back to disable the button and say why.
 *
 * ▶ THE SECOND REQUEST IS REFUSED BY THE DATABASE, NOT BY A CHECK HERE. A
 * check-then-insert loses a double-tap race on a slow connection; the partial
 * unique index cannot. A 23505 is therefore an expected outcome and is mapped
 * to 409, not to a 500.
 */
export async function POST(req: NextRequest) {
  // ── Auth ──────────────────────────────────────────────────────────────────
  const cookieStore = await cookies();
  const supabase = createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    {
      cookies: {
        getAll: () => cookieStore.getAll(),
        setAll: (cs) => cs.forEach(({ name, value, options }) => cookieStore.set(name, value, options)),
      },
    }
  );
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
 
  // ── Participant check ─────────────────────────────────────────────────────
  const { data: profile } = await supabaseAdmin
    .from("profiles")
    .select("is_gold_yield_participant, full_name")
    .eq("id", user.id)
    .single();
 
  if (!profile?.is_gold_yield_participant) {
    return NextResponse.json({ error: "Not enrolled in Gold Yield Programme" }, { status: 403 });
  }
 
  // ── What is owed ──────────────────────────────────────────────────────────
  // Read through this participant's OWN bars, so the sum is scoped by the join
  // rather than by anything a caller could supply — the same shape the admin
  // payout route uses, deliberately, so the two screens cannot disagree.
  const [barsRes, payoutsRes] = await Promise.all([
    supabaseAdmin
      .from("gold_yield_bars")
      .select("gold_yield_sales(participant_share, principal_per_g, grams_sold)")
      .eq("user_id", user.id),
    supabaseAdmin
      .from("gold_yield_payouts")
      .select("amount")
      .eq("user_id", user.id),
  ]);
 
  // ▶ A FAILED READ IS LOUD, NOT AN EMPTY BALANCE. Falling back to [] here
  // would tell a participant they are owed nothing on precisely the reads that
  // broke — and `principal_per_g` does not exist until this branch's migrations
  // land, so the likeliest cause of a failure is the state where the number
  // would be wrong.
  if (barsRes.error || payoutsRes.error) {
    console.error("[gold-yield/request-payout] read failed:", barsRes.error ?? payoutsRes.error);
    return NextResponse.json(
      { error: "We could not read your balance just now. Please try again shortly." },
      { status: 500 },
    );
  }

    ((b as unknown as { gold_yield_sales: unknown[] }).gold_yield_sales ?? [])
  ) as Parameters<typeof barOwedToParticipant>[0];
 
  const owed  = barOwedToParticipant(sales);
  const split = owedBreakdown(sales);
  const paid  = (payoutsRes.data ?? []).reduce((s, p) => s + Number(p.amount), 0);
 
  // ▶ REFUSES RATHER THAN GUESSING when a sale carries no principal. The
  // computable floor is real money and is shown on the dashboard, but a REQUEST
  // names a figure an operator will act on, and naming a floor as if it were
  // the balance is how an underpayment gets proposed by the system itself.
  if (!Number.isFinite(owed)) {
    return NextResponse.json({
      error: "We cannot total your balance yet — one of your sales is missing a "
           + "recorded principal. We have been alerted; please contact us and we "
           + "will settle it manually.",
      code: "owed_uncomputable",
    }, { status: 409 });
  }
 
  const pending = Number((owed - paid).toFixed(2));
  if (pending <= 0) {
    return NextResponse.json({ error: "No pending payout to request" }, { status: 400 });
  }
 
  // ── Record the request ────────────────────────────────────────────────────
  // The split stored is of the TOTAL OWED, not of `pending` — see
  // `owedBreakdown`, which explains why allocating a part payment across the
  // two halves would be an invention rather than a fact.

    .from("gold_yield_payout_requests")
    .insert({
      user_id:              user.id,
      amount_at_request:    pending,
      profit_at_request:    split.profit_sgd,
      principal_at_request: split.principal_sgd,
    })
    .select("id, requested_at")
    .single();
 
  if (insertErr) {

      return NextResponse.json({
        error: "You already have a payout request open. We will be in touch — "
             + "anything that sells in the meantime is added to the same payout.",
        code: "request_already_open",
      }, { status: 409 });
    }
    console.error("[gold-yield/request-payout] insert failed:", insertErr);
    return NextResponse.json({ error: "Could not record your request. Please try again." }, { status: 500 });
  }

  // ▶ The template body is Meta-approved and fixed at {1: name, 2: amount}, so
  // the profit/principal split cannot go in the message without a new template
  // and a fresh approval round. It lives on the admin screen instead; what is
  // fixed here is that the AMOUNT is now the real one.

    ).catch((e) => console.error("[gold-yield/request-payout] admin WhatsApp failed:", e));
  }

    request_id:    request.id,
    requested_at:  request.requested_at,
    pending_sgd:   pending,
    profit_sgd:    split.profit_sgd,
    principal_sgd: split.principal_sgd,
  });
}
