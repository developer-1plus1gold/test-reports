/**
 * Admin — sub-processor / cross-border transfer register (PDPA G5).
 *
 *   GET    /api/admin/governance/transfers        → list, by sort_order then vendor
 *   POST   /api/admin/governance/transfers        → add a vendor
 *   PATCH  /api/admin/governance/transfers?id=…     → edit a vendor
 *   DELETE /api/admin/governance/transfers?id=…     → remove a vendor
 *
 * Admin-only + rate-limited. Service-role client (RLS deny-all).
 */

export async function GET(req: NextRequest) {
  const blocked = await gate(req, "gov-transfer-list", 60);
  if (blocked) return blocked;
  const { data, error } = await admin
    .from("data_transfer_register")
    .select("*")
    .order("sort_order", { ascending: true })
    .order("vendor", { ascending: true });
  if (error) return NextResponse.json({ error: "query_failed" }, { status: 500 });
  return NextResponse.json({ entries: data ?? [] });
}
 
export async function POST(req: NextRequest) {
  const blocked = await gate(req, "gov-transfer-write", 30);
  if (blocked) return blocked;
  const body = (await req.json().catch(() => ({}))) as Record<string, unknown>;
  const vendor = typeof body.vendor === "string" ? body.vendor.trim() : "";
  if (!vendor) return NextResponse.json({ error: "vendor_required" }, { status: 400 });
  const row = pickAllowed(body, TRANSFER_WRITE_FIELDS);
  if ("dpa_url" in row) {
    const dpa = normaliseDpaUrl(row.dpa_url);
    if (dpa === undefined) return NextResponse.json({ error: "bad_dpa_url" }, { status: 400 });
    row.dpa_url = dpa;
  }
  row.vendor = vendor;
  const { data, error } = await admin.from("data_transfer_register").insert(row).select("*").single();
  if (error) return NextResponse.json({ error: "insert_failed" }, { status: 500 });
  auditAdminAction(admin, req, { action: "governance.transfer.create", resourceType: "data_transfer_register", resourceId: data.id, details: { vendor } });
  return NextResponse.json({ entry: data }, { status: 201 });
}
