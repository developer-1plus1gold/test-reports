/**
 * POST /api/admin/ledger/receipt
 *
 * Multipart form upload — single file per request. Saves to the PRIVATE
 * ledger-receipts Supabase Storage bucket and returns the object's stored
 * reference (its URL form, kept for backward compatibility with existing
 * rows — it is an identifier, not a renderable link; see the GET below).
 * The caller (admin UI) then sets receipt_url on the entry via the
 * normal POST/PATCH flow.
 *
 * Two-step on purpose so admin can upload before they've decided to save
 * the entry, and so the same upload route can be re-used for editing.
 */
 
import { NextRequest, NextResponse } from "next/server";
import { createClient } from "@supabase/supabase-js";
import { checkAdminAuth } from "@/lib/admin-auth";
import { safeError } from "@/lib/safe-error";
import { detectKind } from "@/lib/magic-bytes";
import { LEDGER_RECEIPTS_BUCKET, ledgerReceiptPath } from "@/lib/ledger/receipt-url";
 
const supabase = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_KEY!,
);
 
const MAX_BYTES = 10 * 1024 * 1024; // 10 MB — enough for any receipt PDF / photo
 
const ALLOWED_TYPES = new Set([
  "image/jpeg", "image/png", "image/webp", "image/gif", "image/heic",
  "application/pdf",
]);
 
function safeFilename(name: string): string {
  return name.normalize("NFKD").replace(/[^\w.-]+/g, "_").slice(0, 80);
}
 
export async function POST(req: NextRequest) {
  const auth = await checkAdminAuth(req);
  if (!auth.ok) return NextResponse.json({ error: auth.error }, { status: auth.status });
 
  let form: FormData;
  try { form = await req.formData(); }
  catch { return NextResponse.json({ error: "Invalid multipart body" }, { status: 400 }); }
 
  const file = form.get("file");
  if (!(file instanceof File))     return NextResponse.json({ error: "Missing file" },  { status: 400 });
  if (file.size <= 0)              return NextResponse.json({ error: "Empty file" },    { status: 400 });
  if (file.size > MAX_BYTES) {
    return NextResponse.json({ error: `File too large — max ${MAX_BYTES / 1024 / 1024}MB` }, { status: 413 });
  }
  if (!ALLOWED_TYPES.has(file.type)) {
    return NextResponse.json(
      { error: `Unsupported type: ${file.type || "unknown"}. Use PDF or image (JPG/PNG/WEBP/HEIC).` },
      { status: 415 },
    );
  }
 
  // Year-month folder so the bucket stays browsable as it grows.
  const now = new Date();
  const yyyymm = `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, "0")}`;
  const path = `${yyyymm}/${Date.now()}-${safeFilename(file.name || "receipt")}`;
  const bytes = new Uint8Array(await file.arrayBuffer());
 
  // MED-011 (security audit 2026-05-03): magic-byte verification — the
  // claimed Content-Type alone is not enough; admin uploads land in a
  // bucket that's served back to admin browsers with whatever Content-Type
  // a sniffer decides on, so a renamed binary could otherwise be served as
  // PDF/image and executed by the browser.
  const detected = detectKind(bytes);
  const acceptableDetected = new Set([
    "image/jpeg", "image/png", "image/webp", "image/gif", "image/heic",
    "application/pdf",
  ]);
  if (!acceptableDetected.has(detected)) {
    console.warn(
      `[admin/ledger/receipt] rejected upload — claimed=${file.type} detected=${detected}`,
    );
    return NextResponse.json(
      { error: "File contents do not match the claimed type" },
      { status: 415 },
    );
  }
 
  const { error: uploadErr } = await supabase.storage
    .from(LEDGER_RECEIPTS_BUCKET)
    .upload(path, bytes, {
      contentType: file.type,
      upsert:      false,
    });
  if (uploadErr) {
    return NextResponse.json(
      safeError(uploadErr, "admin/ledger/receipt", "Upload failed"),
      { status: 500 },
    );
  }
 
  const { data: pub } = supabase.storage.from(LEDGER_RECEIPTS_BUCKET).getPublicUrl(path);
 
  return NextResponse.json({
    url:  pub.publicUrl,
    path,
    size: file.size,
    type: file.type,
  });
}
 
/**
 * GET /api/admin/ledger/receipt?u=<stored receipt_url or object path>
 *
 * The bucket is private (2026-10-15), so a stored receipt_url no longer
 * resolves on its own. Admin-gated: resolves the stored reference to an
 * object path inside ledger-receipts and 302s to a 5-minute signed URL.
 */
const VIEW_TTL_SECONDS = 300;
 
export async function GET(req: NextRequest) {
  const auth = await checkAdminAuth(req);
  if (!auth.ok) return NextResponse.json({ error: auth.error }, { status: auth.status });
 
  const path = ledgerReceiptPath(req.nextUrl.searchParams.get("u"));
  if (!path) return NextResponse.json({ error: "Invalid receipt reference" }, { status: 400 });
 
  const { data, error } = await supabase.storage
    .from(LEDGER_RECEIPTS_BUCKET)
    .createSignedUrl(path, VIEW_TTL_SECONDS);
  if (error || !data?.signedUrl) {
    return NextResponse.json({ error: "Receipt not found" }, { status: 404 });
  }
  const res = NextResponse.redirect(data.signedUrl, 302);
  res.headers.set("Cache-Control", "private, no-store");
  return res;
}
