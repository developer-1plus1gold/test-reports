import { NextRequest, NextResponse } from "next/server";
import { createClient } from "@supabase/supabase-js";
import { checkAdminAuth } from "@/lib/admin-auth";
import { cardsRequired } from "@/lib/inventory/split-pools";
import { computeCustomerOrderStats } from "@/lib/admin/order-stats";
 
const supabaseAdmin = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_KEY!
);
 
/**
 * GET /api/admin/orders
 * Header: x-admin-secret
 *
 * Returns all orders with customer info and line items, newest first.
 */
export async function GET(req: NextRequest) {
  const auth = await checkAdminAuth(req);
  if (!auth.ok) return NextResponse.json({ error: auth.error }, { status: auth.status });
 
  const { data: orders, error } = await supabaseAdmin
    .from("orders")
    .select(`
      id, created_at, status, status_changed_at, lifecycle, lifecycle_completed_at,
      total_sgd, total_grams, customer_id,
      grand_total_sgd, shipping_fee_sgd, processing_fee_sgd, shipper_actual_fee_sgd,
      shipper, shipper_service, shipper_status, shipper_order_id, shipper_share_link,
      delivery_address, packing_slip_note, payment_method, payment_status, spot_sgd,
      source_channel, external_order_id, external_channel_fee_sgd, external_buyer_label,
      order_items (
        id, product_id, quantity, price_sgd, grams_each, premium_applied_pct, gift_messages,
        products ( id, label, is_assay, product_type )
      )
    `)
    .order("created_at", { ascending: false });
 
  if (error) {
    return NextResponse.json({ error: "Operation failed" }, { status: 500 });
  }
 
  // Fetch profiles + internal-customer flags separately — avoids PostgREST FK
  // hint issues. is_internal lets the admin UI hide demo/test orders by default.
  const customerIds = [...new Set((orders ?? []).map((o) => o.customer_id).filter(Boolean))];
  const [{ data: profiles }, { data: internalRows }] = await Promise.all([
    customerIds.length
      ? supabaseAdmin.from("profiles").select("id, full_name, phone").in("id", customerIds)
      : Promise.resolve({ data: [] as { id: string; full_name: string | null; phone: string | null }[] }),
    customerIds.length
      ? supabaseAdmin.from("customers").select("id").eq("is_internal", true).in("id", customerIds)
      : Promise.resolve({ data: [] as { id: string }[] }),
  ]);
 
  const profileMap = Object.fromEntries((profiles ?? []).map((p) => [p.id, p]));
  const internalSet = new Set((internalRows ?? []).map((r) => r.id));
 
  // Pack multiplier for pool `bundle` products ("1g x 5", "1g x 10"). One
  // catalogue row, N physical 1g pieces, N NFC cards. Loaded once for the whole
  // page rather than per order_item — split_pool_members is a handful of rows,
  // and a per-line fetch here is exactly the burst that trips the admin rate
  // limiter. The UI must not recompute this: `cards_required` is authoritative.
  const { data: bundleRows } = await supabaseAdmin
    .from("split_pool_members")
    .select("product_id, grams")
    .eq("role", "bundle");
  const packGramsByProduct = new Map(
    (bundleRows ?? []).map((r) => [r.product_id as string, Number(r.grams)]),
  );
 
  // Per-order buyer stats — computed once over the full set (all orders are
  // already in memory). `customer_order_count` = how many orders share this
  // order's BUYER; `customer_order_seq` = this order's 1-based chronological
  // position among them (order #k of N). External/marketplace orders are keyed by
  // (channel, external_buyer_label) rather than the shared house customer_id, so
  // the count reflects the real buyer, not every buyer on the channel;
  // `customer_order_grouped=false` marks an anonymous buyer / no customer, where
  // the UI shows no repeat signal. Drives the dispatch-tile badge + the
  // per-customer order count in the Customers section.
  // Also reports the GOLD-only count/seq (2026-08-27) — see order-stats.ts for why
  // both are needed rather than one.
  const orderStats = computeCustomerOrderStats(
    (orders ?? []).map((o) => ({
      id: o.id,
      customer_id: o.customer_id,
      created_at: o.created_at,
      source_channel: o.source_channel,
      external_buyer_label: (o as { external_buyer_label?: string | null }).external_buyer_label ?? null,
      total_grams: o.total_grams,
    })),
  );
 
  const enriched = (orders ?? []).map((o) => ({
    ...o,
    profiles: profileMap[o.customer_id] ?? null,
    is_internal: internalSet.has(o.customer_id),
    customer_order_count:   orderStats.get(o.id)?.count ?? 1,
    customer_order_seq:     orderStats.get(o.id)?.seq ?? 1,
    customer_order_grouped: orderStats.get(o.id)?.grouped ?? false,
    // Gold-only view of the same buyer. Dispatch keeps ALL as its headline (that
    // is the parcel/consolidation signal) and shows gold alongside it, so a
    // customer whose second "order" was a S$5 storage box is not read as a repeat
    // gold buyer. goldSeq is null on an accessory-only order.
    customer_gold_order_count: orderStats.get(o.id)?.goldCount ?? 0,
    customer_gold_order_seq:   orderStats.get(o.id)?.goldSeq ?? null,
    order_items: (o.order_items ?? []).map((i) => ({
      ...i,
      cards_required: cardsRequired(
        i.quantity,
        i.product_id ? packGramsByProduct.get(i.product_id) : null,
      ),
    })),
  }));
 
  return NextResponse.json({ orders: enriched });
}
