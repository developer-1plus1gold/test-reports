/**
 * Admin — MAG pool offers (owner 2026-11-14; lib/mag/pool-offers.ts).
 *
 * GET    → { offers, products, spotSgdPerG }
 *          offers:   every row, with its product and two reference figures —
 *                    `currentPremiumPct` (what the product charges today: the
 *                    curve rate at its weight, or its tier 1) and
 *                    `costPremiumPct` (restock cost over spot, card fee out —
 *                    the 11-14 comparison rule), so a quote below cost shows red.
 *          products: the catalog gold bars an offer can be made on (curve +
 *                    standard), with the same two figures, for the picker.
 * POST   → { productId, durationValue, durationUnit, premiumPct, active? }
 * PATCH  → { id, premiumPct?, active? }
 * DELETE → ?id=<uuid>  (a request already quoted keeps its snapshot; offer_id
 *          on it is set null by the FK)
 */

}
 
export async function GET(req: NextRequest) {
  const auth = await checkAdminAuth(req);
  if (!auth.ok) return NextResponse.json({ error: auth.error }, { status: auth.status });
  try {
    const [offersRes, productsRes, premiums, strategy] = await Promise.all([
      supabaseAdmin.from("mag_pool_offers")
        .select("id, product_id, duration_value, duration_unit, premium_pct, active").limit(2000),
      supabaseAdmin.from("products")
        .select("id, label, grams, product_type, refinery, status, shop_visible, standard_premium_tiers(min_qty, premium_pct)")
        .in("product_type", ["curve", "standard"]).gt("grams", 0).limit(2000),
      fetchPremiums().catch(() => []),
      // Restock cost per product; a failure only blanks the cost column.
      assembleStrategyData(supabaseAdmin).catch((e) => { console.warn(`[${LOG}] strategy unavailable:`, e); return null; }),
    ]);
    if (offersRes.error) throw new Error(offersRes.error.message);
    if (productsRes.error) throw new Error(productsRes.error.message);
 
    const spot = strategy?.spot?.spotSgdPerGram ?? null;
    const costByProduct = new Map<string, number>();
    for (const r of strategy?.rows ?? []) {
      if (r.productId && r.effectiveCostSgd != null && !costByProduct.has(r.productId)) costByProduct.set(r.productId, r.effectiveCostSgd);
    }
    const products = ((productsRes.data ?? []) as ProductRow[]).map((p) => {
      const grams = Number(p.grams);
      const tier1 = [...(p.standard_premium_tiers ?? [])].sort((a, b) => a.min_qty - b.min_qty)[0];
      const currentPremiumPct = p.product_type === "curve"
        ? (premiums.length ? curvePremiumPctForGrams(grams, premiums) : null)
        : (tier1 ? Number(tier1.premium_pct) : null);
      const cost = costByProduct.get(p.id);
      return {
        id: p.id, label: p.label, grams, productType: p.product_type, refinery: p.refinery,
        status: p.status, shopVisible: !!p.shop_visible,
        currentPremiumPct: currentPremiumPct != null ? Math.round(currentPremiumPct * 100) / 100 : null,
        costPremiumPct: cost != null && spot ? Math.round((cost / spot - 1) * 10000) / 100 : null,
      };
    }).sort((a, b) => a.grams - b.grams || a.label.localeCompare(b.label));
 
    return NextResponse.json({
      offers: (offersRes.data ?? []).map(offerFromRow),
      products,
      spotSgdPerG: spot,
    }, { headers: NO_STORE });
  } catch (e) {
    return NextResponse.json(safeError(e, LOG, "Could not load pool offers"), { status: 500 });
  }
}

export async function PATCH(req: NextRequest) {
  const auth = await checkAdminAuth(req);
  if (!auth.ok) return NextResponse.json({ error: auth.error }, { status: auth.status });
  const body = await req.json().catch(() => null) as Record<string, unknown> | null;
  const id = typeof body?.id === "string" ? body.id : "";
  if (!UUID_RE.test(id)) return NextResponse.json({ error: "id (uuid) is required" }, { status: 400 });
  const v = validateOfferInput(body, true);
  if (!v.ok) return NextResponse.json({ error: v.error }, { status: 400 });
  const patch: Record<string, unknown> = { updated_at: new Date().toISOString() };
  if (v.value.premiumPct !== undefined) patch.premium_pct = v.value.premiumPct;
  if (v.value.active !== undefined) patch.active = v.value.active;
  const { data, error } = await supabaseAdmin.from("mag_pool_offers").update(patch).eq("id", id)
    .select("id, product_id, duration_value, duration_unit, premium_pct, active");
  if (error) return NextResponse.json(safeError(error, LOG, "Could not update the offer"), { status: 500 });
  if (!data?.length) return NextResponse.json({ error: "Offer not found" }, { status: 404 });
  auditAdminAction(supabaseAdmin, req, { action: "mag_pool_offer_update", resourceType: "mag_pool_offer", resourceId: id });
  return NextResponse.json({ offer: offerFromRow(data[0]) });
}
 
export async function DELETE(req: NextRequest) {
  const auth = await checkAdminAuth(req);
  if (!auth.ok) return NextResponse.json({ error: auth.error }, { status: auth.status });
  const id = req.nextUrl.searchParams.get("id") ?? "";
  if (!UUID_RE.test(id)) return NextResponse.json({ error: "id (uuid) is required" }, { status: 400 });
  const { data, error } = await supabaseAdmin.from("mag_pool_offers").delete().eq("id", id).select("id");
  if (error) return NextResponse.json(safeError(error, LOG, "Could not delete the offer"), { status: 500 });
  if (!data?.length) return NextResponse.json({ error: "Offer not found" }, { status: 404 });
  auditAdminAction(supabaseAdmin, req, { action: "mag_pool_offer_delete", resourceType: "mag_pool_offer", resourceId: id });
  return NextResponse.json({ ok: true });
}
