/**
 * TikTok backfill queue (Admin → Drumbeat → Channels → TikTok backfill).
 *
 * GET  → { paused, items: [...queue, oldest first], eligible: [...published reels not yet queued/posted] }
 * POST { action: "pause" | "resume" }
 *      { action: "remove" | "retry", pieceId }   — retry puts a failed piece back in the queue
 *      { action: "add", pieceIds: [...] }        — appended in post-date order
 *      { action: "mark_posted", pieceId, permalink? } — a FAILED piece that is in fact live
 *        on TikTok (a lost status reply after TikTok accepted it, 2026-11-26): recorded as
 *        posted, never re-posted; the TikTok link fills the piece's post id + link.
 *
 * Admin-only; service-role tables (RLS USING false). docs/tiktok-organic-integration.md step 3b.
 */

export async function GET(req: NextRequest) {
  const auth = await checkAdminAuth(req);
  if (!auth.ok) return NextResponse.json({ error: auth.error }, { status: auth.status });
  try {
    const db = admin();
    const [{ data: state }, { data: queue }, { data: pieces }, { data: videos }] = await Promise.all([
      db.from("tiktok_backfill_state").select("paused").eq("id", true).maybeSingle(),
      db.from("tiktok_backfill_queue").select("content_piece_id, position, status, last_error, published_at")
        .neq("status", "removed").order("position", { ascending: true }),
      db.from("marketing_content_pieces").select("id, theme, post_date, platform_post_ids, platform_permalinks")
        .eq("status", "published").order("post_date", { ascending: true }),
      db.from("marketing_assets").select("content_piece_id").like("mime_type", "video/%")
        .in("status", [...OPERATOR_APPROVED_ASSET_STATUSES]),
    ]);
    const byId = new Map((pieces ?? []).map((p) => [p.id as string, p]));
    const queued = new Set((queue ?? []).map((q) => q.content_piece_id as string));
    const withVideo = new Set((videos ?? []).map((v) => v.content_piece_id as string));
    const items = (queue ?? []).map((q) => {
      const p = byId.get(q.content_piece_id as string);
      return {
        pieceId: q.content_piece_id, position: q.position, status: q.status, lastError: q.last_error, publishedAt: q.published_at,
        theme: (p?.theme as string | undefined) ?? "(piece no longer published)", postDate: p?.post_date ?? null,
        permalink: (p?.platform_permalinks as Record<string, string> | null)?.tiktok ?? null,
      };
    });
    const eligible = (pieces ?? [])
      .filter((p) => withVideo.has(p.id as string) && !queued.has(p.id as string)
        && !(p.platform_post_ids as Record<string, string> | null)?.tiktok)
      .map((p) => ({ pieceId: p.id, theme: p.theme, postDate: p.post_date }));
    return NextResponse.json({ paused: !!state?.paused, items, eligible });
  } catch (err) {
    return NextResponse.json(safeError(err, "tiktok/backfill/GET", "Failed to load the TikTok backfill"), { status: 500 });
  }
}
