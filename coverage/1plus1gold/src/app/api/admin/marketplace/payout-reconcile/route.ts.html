/**
 * Marketplace payout reconciliation — admin (Mukesh 2026-11-24). Rules: lib/marketplace/payout-reconcile.
 *
 *   GET                       — what Admin → Payments shows: recent payout batches, the orders
 *                               flagged as differences, and each channel's mode + last run.
 *   POST ?mode=dry_run        — run now, writing nothing (the default): what WOULD be closed,
 *                               corrected and flagged, with raw samples of the marketplace's
 *                               answers so field names can be checked. `?channel=` narrows it.
 *   POST ?mode=configured     — run now in each channel's configured mode (live writes + WhatsApp).
 *   PATCH { channel, mode }   — set a channel's mode: off | dry_run | live.
 * Admin-only (cookie, or x-admin-secret from the "Marketplace action (manual)" workflow).
 */

import { PAYOUT_CHANNELS, compactRun, runPayoutReconcile, type ReconcileMode } from "@/lib/marketplace/payout-run";
import type { PayoutChannel } from "@/lib/marketplace/payout-reconcile";
 
export const dynamic = "force-dynamic";

const MODES: readonly ReconcileMode[] = ["off", "dry_run", "live"];
const isChannel = (v: unknown): v is PayoutChannel => typeof v === "string" && (PAYOUT_CHANNELS as readonly string[]).includes(v);
 
export async function GET(req: NextRequest) {
  const auth = await checkAdminAuth(req);
  if (!auth.ok) return NextResponse.json({ error: auth.error }, { status: auth.status });
  try {
    const db = admin();
    const [batches, differences, conns] = await Promise.all([
      db.from("marketplace_payouts")
        .select("channel, payout_ref, status, amount_sgd, paid_at, period_label, order_count, orders_sgd, other_deductions_sgd, closed_count, difference_count, fee_correction_sgd")
        .order("paid_at", { ascending: false, nullsFirst: true }).limit(30),
      db.from("marketplace_payout_orders")
        .select("channel, payout_ref, external_order_id, order_id, revenue_sgd, fee_sgd, shipping_sgd, adjustment_sgd, settlement_sgd, note, reconciled_at")
        .eq("state", "difference").order("reconciled_at", { ascending: false }).limit(100),
      db.from("marketplace_connections").select("channel, payout_reconcile_mode, last_payout_reconcile").in("channel", [...PAYOUT_CHANNELS]),
    ]);
    if (batches.error || differences.error || conns.error) throw new Error("payout_reconcile_read_failed");
    return NextResponse.json(
      { batches: batches.data ?? [], differences: differences.data ?? [], channels: conns.data ?? [] },
      { headers: { "Cache-Control": "private, no-store" } },
    );
  } catch (err) {
    return NextResponse.json(safeError(err, "admin/marketplace/payout-reconcile GET", "Could not load payouts"), { status: 500 });
  }
}
 
export async function POST(req: NextRequest) {
  const auth = await checkAdminAuth(req);
  if (!auth.ok) return NextResponse.json({ error: auth.error }, { status: auth.status });
  const url = new URL(req.url);
  const mode = url.searchParams.get("mode") ?? "dry_run";
  if (mode !== "dry_run" && mode !== "configured") return NextResponse.json({ error: "mode must be dry_run or configured" }, { status: 400 });
  const ch = url.searchParams.get("channel");
  if (ch && ch !== "all" && !isChannel(ch)) return NextResponse.json({ error: "channel must be lazada, tiktok_shop or all" }, { status: 400 });
  try {
    const runs = await runPayoutReconcile(admin(), new Date(), {
      channels: ch && ch !== "all" ? [ch as PayoutChannel] : undefined,
      forceDryRun: mode === "dry_run",
    });
    return NextResponse.json({ runs: runs.map(compactRun) }, { headers: { "Cache-Control": "private, no-store" } });
  } catch (err) {
    return NextResponse.json(safeError(err, "admin/marketplace/payout-reconcile POST", "Payout reconcile failed"), { status: 500 });
  }
}
