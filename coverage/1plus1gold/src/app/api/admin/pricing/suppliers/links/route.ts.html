/**
 * Per-supplier product links — which line on a supplier's sheet is OUR product.
 *
 * GET    /api/admin/pricing/suppliers/links?supplier=stonex
 *        → { supplier, products: [{ id, label, grams, refinery, linkedKey,
 *            resolved: { name, match } | null, options: [{ key, name }] }] }
 *        `resolved` is what pricing uses TODAY (the link if set, else the
 *        automatic exact → same-refinery → any-brand ladder); `options` are the
 *        supplier's priced lines at the product's weight on its latest sheet.
 * PUT    { supplier, productId, supplierProductKey }  → pin a product to a line.
 * DELETE { supplier, productId }                      → back to automatic.
 *
 * Why it exists: a product carries ONE `supplier_product_key`, which is MTB's
 * naming. StoneX names the same bar differently, so the exact match never hit
 * and the quote fell back to another brand's bar (2026-10-08). The matcher now
 * prefers the same refinery; this is the operator's override when that guess is
 * still wrong. Admin-only.
 */

export async function DELETE(req: NextRequest) {

  if (!supplier) return NextResponse.json({ error: "Supplier not found" }, { status: 404 });

  if (typeof productId !== "string" || !UUID_RE.test(productId)) {
    return NextResponse.json({ error: "productId must be a uuid" }, { status: 400 });
  }
  const { error } = await db.from("supplier_product_links").delete()
    .eq("supplier_id", supplier.id).eq("product_id", productId);

  return NextResponse.json({ ok: true });
