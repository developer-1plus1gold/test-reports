/**
 * GET /api/config
 *
 * Runtime configuration consumed by the iOS native app (and future
 * Android equivalent) so the binary doesn't need to be re-released
 * every time a threshold, banner, or feature flag changes.
 *
 * Things that DO NOT belong here: currency / country lists (those are
 * codegen'd into the binary because they're essentially static and we
 * don't want to ship a network round-trip to render a price label),
 * brand colours (codegen), product catalogue (separate /api/products).
 *
 * Things that DO belong here: KYC threshold, KYC validity window,
 * pre-launch banner state + copy, support contact strings, anything
 * else that changes by editing an env var on Vercel rather than
 * editing source.
 *
 * Response shape:
 *   {
 *     kyc: { threshold_sgd: 20000, validity_days: 912 },
 *     prelaunch: { active: true, banner_text: "Pre-launch …" },
 *     support: { whatsapp_url: "https://wa.me/…", phone: "+1…" },
 *     server_time: "2026-05-22T07:00:00.000Z"
 *   }
 *
 * Caching: short 30s public cache; iOS should poll once at app launch
 * and on foreground.
 */
import { NextResponse } from "next/server";
import { KYC_THRESHOLD_SGD, KYC_VALIDITY_MS } from "@/lib/kyc/constants";
import { isPrelaunchActive } from "@/lib/launch/state";
 
export const runtime = "nodejs";
export const dynamic = "force-dynamic";
 
const PRELAUNCH_BANNER_TEXT = "Pre-launch preview · public launch coming soon";
 
// WhatsApp business number — same one displayed on /contact and used by
// every customer-visible "message us" CTA. Keep in sync with web copy.
const SUPPORT_WHATSAPP_NUMBER = "+16507194653";
const SUPPORT_WHATSAPP_URL = "https://wa.me/16507194653";
 
export async function GET() {
  // Effective pre-launch: env flag AND the launch ceremony hasn't completed.
  // Cutting all 4 ribbons flips this to false at runtime (no redeploy), so the
  // banner hides on web + iOS. Fail-closed (DB blip → stays pre-launch).
  const prelaunchActive = await isPrelaunchActive();
 
  const payload = {
    kyc: {
      threshold_sgd: KYC_THRESHOLD_SGD,
      validity_days: Math.round(KYC_VALIDITY_MS / (24 * 60 * 60 * 1000)),
    },
    prelaunch: {
      active: prelaunchActive,
      banner_text: prelaunchActive ? PRELAUNCH_BANNER_TEXT : null,
    },
    support: {
      whatsapp_url: SUPPORT_WHATSAPP_URL,
      whatsapp_number: SUPPORT_WHATSAPP_NUMBER,
    },
    // Canonical Supabase endpoint for the native app. The URL + anon key are
    // baked into the iOS binary (Info.plist); a project move (e.g. a future
    // DEV→PROD cutover) would otherwise force an App Store rebuild. The app
    // adopts these on next launch (RuntimeConfig.adopt) so the endpoint can
    // move with NO binary release. The anon key is public (it already ships
    // in the binary) — serving it here is not a secret leak.
    supabase: {
      url: process.env.NEXT_PUBLIC_SUPABASE_URL ?? null,
      anon_key: process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY ?? null,
    },
    server_time: new Date().toISOString(),
  };
 
  return NextResponse.json(payload, {
    headers: {
      // Short cache so the banner clears promptly once the ceremony completes.
      "Cache-Control": "public, max-age=10, s-maxage=10",
    },
  });
}
