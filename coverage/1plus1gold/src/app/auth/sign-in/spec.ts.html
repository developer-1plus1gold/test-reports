/**
 * SDUI spec for /auth/sign-in.
 *
 * Mirrors the PWA's 5-method sign-in card with three stages:
 *   input     — entry: Apple / Google / Email / Passkey / WhatsApp buttons
 *   wa_otp    — WhatsApp 6-digit OTP entry
 *   email_otp — Email 6-digit OTP entry
 *
 * Native escape hatches via `{kind: "native"}` actions:
 *   apple_signin, passkey_signin, google_signin (stub),
 *   send_wa_otp, verify_wa_otp,
 *   send_email_otp, verify_email_otp,
 *   mfa_route_after_signin
 * See apps/ios-customer/Sources/SDUI/SDUINativeActionRegistry.swift for
 * the handler registry.
 *
 * Root is `scroll` so SwiftUI keyboard avoidance kicks in for the OTP
 * field — non-negotiable per the Contact-page lesson.
 *
 * Button prop names — SDUIButton expects `label_text`, `label_font`,
 * `label_color` (NOT bare `label`/`font`/`color`). Icon names are
 * SF Symbols (no `system:` prefix) or `asset:<Name>` for bundled images.
 */
 
import type { SDUISpec } from "@/lib/spec/types";

export function signInPageSpec(): SDUISpec {
  // Env-gated demonstration mode (App Store / Play reviewer access). Non-null
  // ONLY when STORE_REVIEW_DEMO_ENABLED + the demo envs are set, so the button
  // is entirely ABSENT from the spec in normal operation (server-side gate).
  const demoEmail = getStoreReviewDemoEmail();
  return {
    surface: "sign-in",
    spec_version: 1,
    min_renderer_version: 1,
    data: {},
    state_schema: {
      stage:         { type: "string",  default: "input" },     // input | wa_otp | email_otp
      country_iso:   { type: "string",  default: "sg" },
      country_dial:  { type: "string",  default: "+65" },
      phone:         { type: "string",  default: "" },
      email_input:   { type: "string",  default: "" },
      email_sent_to: { type: "string",  default: "" },
      otp:           { type: "string",  default: "" },
      show_email_form: { type: "boolean", default: false },
      loading:       { type: "boolean", default: false },
      error:         { type: "string",  default: "" },
      mfa_route:     { type: "string",  default: "" },          // "none" | "aal2"
    },
    root: {
      type: "scroll",
      // padding_v: 24 is INTENTIONAL — gives the logo breathing room
      // from the SpotPriceTicker chrome pinned above the tab content
      // via MainTabView's safeAreaInset. Without this padding the
      // logo visually runs into the ticker (Mukesh 2026-05-25, lesson
      // originally surfaced on Home). The natural .defaultScrollAnchor(.top)
      // lands content at y=0 which is correct. NO initial_scroll_to —
      // earlier attempts to skip the padding or jump to the card both
      // hid the logo unnecessarily.
      // scroll_anchors (2026-06-12): the OTP "← change number/email" buttons
      // set _scroll_target="signin-top" so going back from a wrong-code OTP
      // screen reliably RETURNS to the top sign-in options on iPad (the Apple
      // 2.1(a) "couldn't get back to the Login page" bug — the stage flipped
      // but the scroll stayed at the OTP field). Keeps the y=0 initial anchor.
      props: { padding_h: 16, padding_v: 24, spacing: 0, scroll_anchors: true },
      children: [
        // Invisible top anchor for the back-to-sign-in scroll (above the logo).
        { type: "spacer", props: { height: 1, id: "signin-top" } },
        // ── Logo (stacked 1+1 over GOLD) — shared composite ───────────
        stackedLogo(),
 
        { type: "spacer", props: { height: 28 } },
 
        // ── Card ─────────────────────────────────────────────────────
        {
          type: "card",
          props: { padding: 24, corner_radius: 12, bg: "bg_card", max_width: "infinity" },
          children: [
            // ── Stage: input ──────────────────────────────────────────
            inputStageBlock(),
            // Demonstration mode (reviewer access) — env-gated, input stage only.
            ...(demoEmail ? [demoAccessBlock(demoEmail)] : []),
            // ── Stage: wa_otp ─────────────────────────────────────────
            waOtpStageBlock(),
            // ── Stage: email_otp ──────────────────────────────────────
            emailOtpStageBlock(),
          ],
        },
 
        { type: "spacer", props: { height: 16 } },
 
        // ── Anti-phishing notice (always visible) ────────────────────
        {
          type: "card",
          props: {
            padding: 14, corner_radius: 12,
            bg: "gold@6", stroke: "gold_dim", max_width: "infinity",
          },
          children: [
            {
              type: "hstack",
              props: { spacing: 10, alignment: "top" },
              children: [
                { type: "icon", props: { name: "shield.checkered", size: 15, color: "gold_mid" } },
                {
                  type: "text",
                  props: {
                    content:
                      "Security notice: 1Plus1Gold sends order updates only via WhatsApp or the email address on your account. We will never contact you by SMS or phone call.",

                    color: "muted",

                    max_width: "infinity",
                  },
                },
              ],
            },
          ],
        },
 
        { type: "spacer", props: { height: 16 } },
 
        // ── Back to home ────────────────────────────────────────────
        {
          type: "button",
          props: {
            label_text: "← Back",
            label_font: "montserrat:12",
            label_color: "muted",
            bg: "transparent",
            padding_v: 8,
            corner_radius: 6,
            action: { kind: "tabswitch", target: "home" },
          },
        },
 
        { type: "spacer", props: { height: 32 } },
      ],
    },
  };
}

function verifyOtpAction(channel: "wa" | "email") {
  if (channel === "wa") {
    return {
      kind: "native",
      name: "verify_wa_otp",
      params: { phone: "{state.country_dial}{state.phone}", code: "{state.otp}" },
      set_state_on_request: { loading: true, error: "" },
      on_success: [
        { kind: "state_set", key: "loading", value: false },
        nativeWithMFARoute("noop"),
      ],
      on_failure: [{ kind: "state_set", key: "loading", value: false }],
    };
  }
  return {
    kind: "native",
    name: "verify_email_otp",
    params: { email: "{state.email_sent_to}", code: "{state.otp}" },
    set_state_on_request: { loading: true, error: "" },
    on_success: [
      { kind: "state_set", key: "loading", value: false },
      nativeWithMFARoute("noop"),
    ],
    on_failure: [{ kind: "state_set", key: "loading", value: false }],
  };
}
 
function nativeWithMFARoute(name: string) {
  const mfaThenRoute = {
    kind: "native" as const,
    name: "mfa_route_after_signin",
    on_success: [
      { kind: "tabswitch" as const, target: "account" as const },
    ],
  };
 
  if (name === "noop") return mfaThenRoute;
 
  return {
    kind: "native" as const,
    name,
    set_state_on_request: { loading: true, error: "" },
    on_success: [
      { kind: "state_set", key: "loading", value: false },
      mfaThenRoute,
    ],
    on_failure: [{ kind: "state_set", key: "loading", value: false }],
  };
}
