"use client";
 
/**
 * Slim status strip shown above the AI input when continuous voice
 * mode is on. Conveys the loop states with colour + animation so the
 * customer always knows whether the AI is hearing, thinking, or
 * speaking — without the dramatic full-screen sphere of ChatGPT
 * (which would hide our cart cards / graphs).
 *
 * Visible states:
 *   connecting             — cream spin, "Connecting to <Guide>…"
 *                            (between session-open and greeting start;
 *                             mic is gated, bars wouldn't move yet)
 *   listening + recording  — teal-cyan pulse + live mic-level bars,
 *                            "Speak now — I'm listening". Gold was
 *                            indistinguishable from the rest of the
 *                            brand surface; teal makes it pop as a
 *                            distinct "I'm receiving" state.
 *   thinking               — white spin, "Thinking…"
 *   speaking               — green pulse, "<Guide> is speaking — tap
 *                            to interrupt"
 *   idle                   — hidden (voice mode off, or transitioning)
 */
 
import { useEffect, useState } from "react";
import { Loader2, Volume2 } from "lucide-react";
import AudioMeter from "./AudioMeter";
 
export type VoiceState =
  | "idle"
  | "connecting"
  | "listening"
  | "recording"
  | "thinking"
  | "speaking"
  | "paused";
 
interface Props {
  state:        VoiceState;
  /** Called when the customer taps the badge — used to interrupt the AI
   *  while it's speaking and immediately start listening again. Only
   *  meaningful in the "speaking" state; on other states the badge is
   *  informational. */
  onInterrupt?: () => void;
  /** Called when the customer taps a tappable LISTENING pill — the
   *  expected behaviour is to mute the mic and flip to "paused".
   *  Optional; provide to make listening tappable. */
  onPause?: () => void;
  /** Called when the customer taps a "paused" pill to resume — the
   *  expected behaviour is to unmute the mic and flip to "listening".
   *  Required if state can be "paused". */
  onResume?: () => void;
  /** Local mic stream — drives the live level meter shown in the
   *  listening / recording state so the customer can SEE the mic is
   *  hot. Provided by the parent that owns the RealtimeSession. */
  micStream?: MediaStream | null;
  /** Guide's display name (e.g. "Aurel"). Surfaces in the labels:
   *  "Connecting to Aurel…", "Aurel is speaking — tap to interrupt".
   *  Optional — without it labels fall back to generic copy. */
  guideName?: string;
}
 
// Listening uses a teal-cyan that's distinct from gold (brand) and
// green (speaking). Visually telegraphs "I'm receiving you" as its
// own state, not just the same gold mood as the rest of the panel.
const LISTEN_COLOR  = "#5FB3C8";
const LISTEN_BG     = "rgba(95,179,200,0.12)";
const LISTEN_BORDER = "rgba(95,179,200,0.45)";
 
const COLORS: Record<Exclude<VoiceState, "idle">, { bg: string; border: string; text: string; iconColor: string }> = {
  connecting: {
    bg:        "rgba(232,213,163,0.06)",
    border:    "rgba(232,213,163,0.35)",
    text:      "var(--color-offwhite)",
    iconColor: "var(--color-offwhite)",
  },
  paused: {
    bg:        "rgba(232,213,163,0.04)",
    border:    "rgba(232,213,163,0.25)",
    text:      "var(--color-muted)",
    iconColor: "var(--color-muted)",
  },
  listening: {
    bg:        LISTEN_BG,
    border:    LISTEN_BORDER,
    text:      LISTEN_COLOR,
    iconColor: LISTEN_COLOR,
  },
  // recording shares the listening visuals on purpose — see header doc.
  recording: {
    bg:        LISTEN_BG,
    border:    LISTEN_BORDER,
    text:      LISTEN_COLOR,
    iconColor: LISTEN_COLOR,
  },
  thinking: {
    bg:        "rgba(232,213,163,0.06)",
    border:    "rgba(232,213,163,0.35)",
    text:      "var(--color-offwhite)",
    iconColor: "var(--color-offwhite)",
  },
  speaking: {
    bg:        "rgba(29,158,117,0.08)",
    border:    "rgba(29,158,117,0.45)",
    text:      "var(--color-success)",
    iconColor: "var(--color-success)",
  },
};
 
function makeLabel(state: Exclude<VoiceState, "idle">, guideName?: string, longThinking = false): string {
  switch (state) {
    case "connecting":
      return guideName ? `Connecting to ${guideName}…` : "Connecting…";
    case "listening":
    case "recording":
      return "Speak now — I'm listening · tap to pause";
    case "thinking":
      return longThinking ? "Still thinking — tap to restart" : "Thinking…";
    case "speaking":
      return guideName ? `${guideName} is speaking — tap to interrupt` : "Tap to interrupt";
    case "paused":
      return "Paused — tap to resume";
  }
}
 
export default function VoiceStateIndicator({ state, onInterrupt, onPause, onResume, micStream, guideName }: Props) {
  // Escape hatch: if "thinking" persists past 10s the model is hung
  // (often after a mid-conversation language switch where Whisper or
  // the model stalls). Hint at the recovery by changing the copy and
  // letting the customer tap to cancel + return to listening.
  const [longThinking, setLongThinking] = useState(false);
  useEffect(() => {
    if (state !== "thinking") { setLongThinking(false); return; }
    const t = window.setTimeout(() => setLongThinking(true), 10_000);
    return () => clearTimeout(t);
  }, [state]);
 
  if (state === "idle") return null;
  const c = COLORS[state];
  const isListening = state === "listening" || state === "recording";
  const isSpinning  = state === "thinking" || state === "connecting";
  // Tap targets:
  //   speaking → interrupt the AI
  //   thinking → cancel a hung response
  //   listening/recording → pause the mic
  //   paused → resume the mic
  const tapHandler =
      state === "speaking" || state === "thinking"   ? onInterrupt
    : isListening                                    ? onPause
    : state === "paused"                             ? onResume
    :                                                  undefined;
  const interactive = !!tapHandler;
 
  // Larger visuals than v1 — testers reported the indicator was too small
  // to read at a glance, especially during the connecting wait when it's
  // the only thing on screen. Doubled the icon, bumped the label one tier,
  // gave the pill more breathing room.
  const content = (
    <>
      {isListening ? (
        <AudioMeter stream={micStream ?? null} active color={c.iconColor} />
      ) : state === "speaking" ? (
        <Volume2 size={16} style={{ color: c.iconColor }} />
      ) : (
        <Loader2
          size={16}
          className={isSpinning ? "animate-spin" : ""}
          style={{ color: c.iconColor }}
        />
      )}
      <span className="font-montserrat text-sm font-semibold tracking-wide" style={{ color: c.text }}>
        {makeLabel(state, guideName, longThinking)}
      </span>
    </>
  );
 
  return (
    <>
      <style>{`
        @keyframes voice-state-pulse {
          0%, 100% { opacity: 0.85; }
          50%      { opacity: 1;    }
        }
        .voice-state-pulse { animation: voice-state-pulse 1.4s ease-in-out infinite; }
      `}</style>
      {interactive ? (
        <button
          type="button"
          onClick={tapHandler}
          className="flex items-center gap-2.5 px-4 py-2.5 rounded-full mx-3 mb-2 voice-state-pulse transition-opacity hover:opacity-100 focus:opacity-100"
          style={{
            backgroundColor: c.bg,
            border:          `1px solid ${c.border}`,
            width:           "fit-content",
            alignSelf:       "center",
            cursor:          "pointer",
          }}
          aria-label={
              state === "speaking" ? "Interrupt and speak"
            : state === "thinking" ? "Cancel and restart"
            : isListening          ? "Pause listening"
            : state === "paused"   ? "Resume listening"
            :                        "Voice control"
          }
        >
          {content}
        </button>
      ) : (
        <div
          className="flex items-center gap-2.5 px-4 py-2.5 rounded-full mx-3 mb-2 voice-state-pulse"
          style={{
            backgroundColor: c.bg,
            border:          `1px solid ${c.border}`,
            width:           "fit-content",
            alignSelf:       "center",
          }}
          role="status"
          aria-live="polite"
        >
          {content}
        </div>
      )}
    </>
  );
}
