/**
 * DispatchPackPanel — the single section inside every dispatch tile that
 * handles the entire packing workflow. Mukesh's redesign 2026-05-13:
 *
 *   - All gold lines (curve + standard) with per-unit NFC / serial entry
 *   - Accessories ordered — read-only, already decremented at checkout
 *   - Suggested consumables / fulfilment-consumables (pre-ticked from
 *     consumable_links)
 *   - Other back-of-house items (un-ticked)
 *   - The action row is the packer's sequence, in order (Mukesh 2026-11-28):
 *       1. Confirm packing — saves the pack only (prints nothing: Mukesh 2026-11-28)
 *       2. Print documents — receipt, packing list and every certificate as ONE
 *          A4 PDF (lib/documents/order-documents.tsx). Only after Confirm: the
 *          receipt carries the serials, which are saved by Confirm.
 *       3. Print shipping label — whichever label the courier needs
 *          (lib/admin/label-step.ts), passed in by the tile as `labelStep`
 *     Reprints read "Reprint …" in the same places, so the order never changes.
 *   - "Print Insert Card" — the A6 thank-you / sell-back / review card that
 *     goes in the box. Unlike every other artefact here it is a CONSTANT (no
 *     order number, no serial, no name), so it is not built from pick state —
 *     see lib/insert-card.ts.
 *   - "Confirm Packing" — commits, decrements consumable inventory, flips
 *     the gate so the shipper-side action (Book Lalamove / NV select /
 *     Mark Collected) becomes enabled
 *
 * Inventory semantics:
 *   Gold (curve / standard)            — decremented at CHECKOUT via FIFO
 *                                        inventory reservation. Skip here.
 *   Accessories                        — decremented at CHECKOUT via
 *                                        consumeStockForOrderItem(). Skip here.
 *   Consumables / fulfilment_consumables — NOT in order_items. Decremented
 *                                        HERE via pack_order RPC.
 *
 * So pack_order items[] only carries the operator-picked consumables /
 * fulfilment_consumables / back-of-house extras. Accessories appear in
 * the UI for the packer's reference + on the printed slip, but never go
 * through pack_order.
 */

interface Suggestion {
  product_id:   string;
  label:        string;
  product_type: string;
  qty:          number;
  stock_qty:    number | null;
  ticked:       boolean;

   *  all_consumables = the un-ticked catalogue for exceptions. */
  source:       "pack_rule" | "consumable_link" | "all_consumables";
  /** Why the rule put it there ("one per piece (2 pieces)") — shown on the line. */
  reason?:      string;
}
interface SuggestionsResponse {
  order:          { id: string; items: Array<{ product_id: string; label: string; product_type: string; quantity: number }> };
  packed_already: { id: string; packed_at: string } | null;

   *  so "Add to pack" only ever submits genuinely new picks. */

  suggestions:    Suggestion[];
}

            <label key={s.product_id} className="flex items-center gap-2 font-montserrat text-xs"
              style={{
                color:  p?.ticked ? "var(--color-cream)" : "var(--color-muted)",
                cursor: disabled  ? "not-allowed"        : "pointer",
                opacity: disabled ? 0.6 : 1,
              }}>
              <input
                type="checkbox"
                checked={p?.ticked ?? false}
                onChange={() => onToggle(s.product_id)}
                disabled={disabled}
                style={{ accentColor: "var(--color-gold)", cursor: disabled ? "not-allowed" : "pointer" }}
              />
              <span className="flex-1">
                {s.label}
                {s.reason && (
                  <span className="block text-[10px]" style={{ color: "var(--color-muted)" }}>{s.reason}</span>
                )}
              </span>
              <span className="text-[10px]" style={{ color: "var(--color-muted)" }}>{s.product_type.replace(/_/g, " ")}</span>
              <input
                type="number"
                min={1}
                value={p?.qty ?? s.qty}
                onChange={(e) => onQtyChange(s.product_id, Number(e.target.value))}
                disabled={!p?.ticked || disabled}
                className="w-12 px-1 py-0.5 rounded border text-center font-montserrat text-xs outline-none disabled:opacity-50"
                style={{ backgroundColor: "var(--color-bg)", borderColor: "var(--color-rule)", color: "var(--color-cream)" }}
              />
            </label>
          );
        })}
      </div>
    </div>
  );
