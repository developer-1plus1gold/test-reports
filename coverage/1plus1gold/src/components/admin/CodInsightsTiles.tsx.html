"use client";
 
/**
 * Summary tiles at the top of Admin → COD / Bank Transfer (Mukesh 2026-11-23, "just like the
 * Customer screen"). Reads GET /api/admin/cod/insights (lib/admin/cod-insights holds every
 * definition) and renders eight tiles + a "COD collected per month" chart. `reloadKey` changes
 * when the screen's Refresh is tapped.
 *
 * Overdue, and each line of "Who owes us" (a marketplace or a COD customer, Mukesh 11-23), is a
 * button: it hands the page the exact order ids it counted, so the lists below show only those;
 * tapping it again clears the filter. Marketplace payouts scrolls to its list.
 */
import { useEffect, useState } from "react";
import Flag from "@/components/Flag";
import { useFlag, useFmt } from "@/store/currencyStore";
import type { CodInsights } from "@/lib/admin/cod-insights";
 
import { InsightTile as Tile, MonthlyBars, filterTap, pct, type InsightFilter } from "@/components/admin/InsightTile";
 
/** The lists below are narrowed to these orders (the shared tile filter). */
export type CodFilter = InsightFilter;
 
const DATE = new Intl.DateTimeFormat("en-GB", { day: "numeric", month: "short", year: "numeric", timeZone: "UTC" });
const dateLabel = (ymd: string) => DATE.format(new Date(`${ymd}T00:00:00Z`));
const days = (d: number | null) => (d == null ? "—" : d === 1 ? "1 day" : `${d} days`);
 
export default function CodInsightsTiles({ reloadKey, filterKey, onFilter, onPayouts }: {
  reloadKey: number;
  /** The key of the filter the lists are narrowed to, if any. */
  filterKey?: string | null;
  /** Narrow the lists to these orders, or clear the filter (null). */
  onFilter?: (f: CodFilter | null) => void;
  /** Scroll to the marketplace payouts list. */
  onPayouts?: () => void;
}) {
  const fmt = useFmt();
  const flagCode = useFlag();
  const [data, setData] = useState<CodInsights | null>(null);
  const [error, setError] = useState<string | null>(null);
 
  useEffect(() => {
    const ctrl = new AbortController();
    setError(null);
    fetch("/api/admin/cod/insights", { cache: "no-store", signal: ctrl.signal })
      .then(async (r) => {
        const j = await r.json().catch(() => ({}));
        if (!r.ok) throw new Error(j.error ?? "Could not load COD insights");
        setData(j.insights as CodInsights);
      })
      .catch((e: Error) => { Eif (e.name !== "AbortError") setError(e.message); });
    return () => ctrl.abort();
  }, [reloadKey]);
 
  if (error) return <p className="font-montserrat text-xs" style={{ color: "#e57373" }}>{error}</p>;
  if (!data) return <div className="h-24 rounded-xl animate-pulse" style={{ backgroundColor: "rgba(201,168,76,0.04)" }} aria-label="Loading COD insights" />;
 
  const money = (v: number) => <span className="inline-flex items-center gap-1"><Flag code={flagCode} />{fmt(v)}</span>;
  const { waiting, overdue, payouts, collectedThisMonth: now, collectedLastMonth: last, timeToPay, customers, neverPaid } = data;
 
  const overdueTap = filterTap(filterKey, onFilter,
    { key: "overdue", label: `${overdue.count} overdue ${overdue.count === 1 ? "order" : "orders"}`, ids: data.overdueIds });
  const owedBy = data.owedBy ?? [];
  const payoutsTap = onPayouts && payouts.count > 0 ? { actionHint: "Tap to see them", onClick: onPayouts } : {};
 
  return (
    <div className="flex flex-col gap-3" data-testid="cod-insights">
      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
        <Tile label="Waiting for payment" value={waiting.count} alert={overdue.count > 0}
          sub={waiting.count
            ? <span className="inline-flex flex-wrap items-center gap-1">{money(waiting.amountSgd)} · oldest {days(waiting.oldestDays)}</span>
            : "Nothing to collect"} />
        <Tile label={`Overdue · over ${overdue.days} days`} value={overdue.count} alert={overdue.count > 0}
          sub={overdue.count ? money(overdue.amountSgd) : "None overdue"} {...overdueTap} />
        <Tile label="Marketplace payouts due" value={payouts.count}
          sub={payouts.count ? <span className="inline-flex items-center gap-1">net {money(payouts.netSgd)}</span> : "None due"}
          {...payoutsTap} />
        <Tile label="Collected this month" value={money(now.amountSgd)}
          sub={<span className="inline-flex flex-wrap items-center gap-1">{now.count} orders · last month {money(last.amountSgd)}</span>} />
        <Tile label="COD share of sales" value={data.codShareThisMonth == null ? "—" : pct(data.codShareThisMonth)}
          sub="of this month's paid sales (excl. marketplaces)" />
        <Tile label="Time to pay" value={days(timeToPay.medianDays)}
          sub={timeToPay.sample ? `median · average ${days(timeToPay.avgDays)} · ${timeToPay.sample} orders` : "No paid orders with a paid date yet"} />
        <Tile label="COD customers" value={customers.allowed}
          sub={`allowed · ${customers.used} have used it · ${customers.repeat} more than once`} />
        <Tile label="Never paid" value={neverPaid.count}
          sub={neverPaid.count ? <span className="inline-flex items-center gap-1">{money(neverPaid.amountSgd)} cancelled unpaid</span> : "None cancelled unpaid"} />
      </div>
      <div className="rounded-xl p-4 flex flex-col gap-2" data-testid="cod-owed-by"
        style={{ backgroundColor: "rgba(201,168,76,0.04)", border: "1px solid var(--color-gold-dim)" }}>
        <span className="font-montserrat text-[11px] uppercase tracking-widest" style={{ color: "var(--color-muted)" }}>
          Who owes us
        </span>
        {owedBy.length === 0 ? (
          <span className="font-montserrat text-xs" style={{ color: "var(--color-muted)" }}>Nobody — nothing is waiting to be paid</span>
        ) : (
          <div className="flex flex-col">
            {owedBy.map((d) => {
              const on = filterKey === d.key;
              const row = (
                <>
                  <span className="min-w-0 truncate" style={{ color: "var(--color-offwhite)" }}>
                    {d.label}
                    <span className="ml-2 text-[11px]" style={{ color: "var(--color-muted)" }}>
                      {d.kind === "marketplace" ? "payout" : "COD"} · {d.count} {d.count === 1 ? "order" : "orders"}
                    </span>
                  </span>
                  <span className="shrink-0 font-semibold tabular-nums" style={{ color: "var(--color-gold)" }}>{money(d.amountSgd)}</span>
                </>
              );
              const cls = "flex items-center justify-between gap-3 py-2 font-montserrat text-sm text-left w-full";
              const style = { borderTop: "1px solid var(--color-rule)", backgroundColor: on ? "rgba(201,168,76,0.10)" : "transparent" };
              return onFilter ? (
                <button key={d.key} type="button" aria-pressed={on} className={`${cls} px-1 rounded`} style={style}
                  onClick={() => onFilter(on ? null : { key: d.key, label: `${d.label} (${d.count} ${d.count === 1 ? "order" : "orders"})`, ids: d.orderIds })}>
                  {row}
                </button>
              ) : <div key={d.key} className={cls} style={style}>{row}</div>;
            })}
            {onFilter && <span className="font-montserrat text-[11px] pt-2" style={{ color: "var(--color-muted)" }}>Tap a line to list its orders below.</span>}
          </div>
        )}
      </div>
      <MonthlyBars title="COD collected per month"
        months={data.byMonth.map((m) => ({ month: m.month, value: m.amountSgd, count: m.count }))}
        label={(m) => fmt(m.value)}
        note={`Real sales since we opened (${dateLabel(data.since)}); excludes internal, test and supplier accounts. Dated by when the payment was marked.`} />
    </div>
  );
}
