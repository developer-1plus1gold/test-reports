"use client";
 
/**
 * The summary tile and composition bar shared by the admin insight panels (Customers since
 * 2026-11-21, COD / Bank Transfer since 2026-11-23). One component, so every panel looks and
 * behaves the same: a tile with a filter is a button with aria-pressed and a "Tap to …" hint.
 */
import type { ReactNode } from "react";
 
export const pct = (x: number) => `${Math.round(x * 100)}%`;
const BAR_SHADES = ["var(--color-gold)", "rgba(201,168,76,0.7)", "rgba(201,168,76,0.45)", "rgba(201,168,76,0.25)", "var(--color-rule)"];
const ALERT = "#e57373";
 
export function InsightTile({ label, value, sub, onClick, active, actionHint, alert }: {
  label: string; value: ReactNode; sub: ReactNode;
  onClick?: () => void; active?: boolean; actionHint?: string;
  /** Needs attention (e.g. an overdue payment): the value and border turn red. */
  alert?: boolean;
}) {
  const body = (
    <>
      <span className="font-montserrat text-[11px] uppercase tracking-widest" style={{ color: "var(--color-muted)" }}>{label}</span>
      <div className="font-cinzel text-xl sm:text-2xl font-bold tabular-nums" style={{ color: alert ? ALERT : "var(--color-gold)" }}>{value}</div>
      <span className="font-montserrat text-xs" style={{ color: "var(--color-muted)" }}>{sub}</span>
      {actionHint && (
        <span className="font-montserrat text-[11px] font-semibold" style={{ color: "var(--color-gold)" }}>{actionHint}</span>
      )}
    </>
  );
  const style = {
    backgroundColor: active ? "rgba(201,168,76,0.14)" : "rgba(201,168,76,0.04)",
    border: `1px solid ${active ? "var(--color-gold)" : alert ? ALERT : "var(--color-gold-dim)"}`,
  };
  return onClick ? (
    <button type="button" onClick={onClick} aria-pressed={!!active}
      className="rounded-xl p-4 flex flex-col gap-1.5 text-left transition-colors hover:opacity-90" style={style}>
      {body}
    </button>
  ) : (
    <div className="rounded-xl p-4 flex flex-col gap-1.5" style={style}>{body}</div>
  );
}
 
/** A labelled 100 % bar with a legend; segments with 0 are left out. */
export function CompositionBar({ title, parts }: { title: string; parts: Array<{ label: string; value: number }> }) {
  const total = parts.reduce((s, p) => s + p.value, 0);
  const shown = parts.map((p, i) => ({ ...p, shade: BAR_SHADES[i % BAR_SHADES.length] })).filter((p) => p.value > 0);
  return (
    <div className="rounded-xl p-4 flex flex-col gap-2"
      style={{ backgroundColor: "rgba(201,168,76,0.04)", border: "1px solid var(--color-gold-dim)" }}>
      <span className="font-montserrat text-[11px] uppercase tracking-widest" style={{ color: "var(--color-muted)" }}>{title}</span>
      {total === 0 ? (
        <span className="font-montserrat text-xs" style={{ color: "var(--color-muted)" }}>None yet</span>
      ) : (
        <>
          <div className="flex h-3 rounded-full overflow-hidden" role="img" aria-label={title}>
            {shown.map((p) => <div key={p.label} style={{ width: `${(p.value / total) * 100}%`, backgroundColor: p.shade }} />)}
          </div>
          <div className="flex flex-wrap gap-x-4 gap-y-1">
            {shown.map((p) => (
              <span key={p.label} className="font-montserrat text-xs inline-flex items-center gap-1.5" style={{ color: "var(--color-offwhite)" }}>
                <span className="inline-block w-2.5 h-2.5 rounded-sm" style={{ backgroundColor: p.shade }} />
                {p.label} <span style={{ color: "var(--color-muted)" }}>{p.value} · {pct(p.value / total)}</span>
              </span>
            ))}
          </div>
        </>
      )}
    </div>
  );
}
 
const MONTH_FMT = new Intl.DateTimeFormat("en-GB", { month: "short", timeZone: "UTC" });
 
/**
 * A per-month bar chart (oldest month first): the value above each bar, the month under it.
 * Shared by every insight panel's "… per month" chart (COD collected, sales, buybacks, MAG).
 */
export function MonthlyBars({ title, months, label, note }: {
  title: string;
  /** `month` is "YYYY-MM"; `value` sizes the bar; `count` is the tooltip. */
  months: Array<{ month: string; value: number; count?: number }>;
  /** The text above a bar (e.g. a formatted amount); blank for an empty month. */
  label: (m: { month: string; value: number; count?: number }) => string;
  note?: ReactNode;
}) {
  const peak = Math.max(0, ...months.map((m) => m.value));
  const cols = { gridTemplateColumns: `repeat(${months.length}, minmax(0, 1fr))` };
  return (
    <div className="rounded-xl p-4 flex flex-col gap-2"
      style={{ backgroundColor: "rgba(201,168,76,0.04)", border: "1px solid var(--color-gold-dim)" }}>
      <span className="font-montserrat text-[11px] uppercase tracking-widest" style={{ color: "var(--color-muted)" }}>{title}</span>
      <div className="grid gap-2 items-end h-28" style={cols} role="img" aria-label={title}>
        {months.map((m) => (
          <div key={m.month} className="h-full flex flex-col justify-end items-center gap-1">
            <span className="font-montserrat text-[10px] tabular-nums" style={{ color: "var(--color-muted)" }}>
              {m.value > 0 ? label(m) : ""}
            </span>
            <div className="w-full rounded-t" title={m.count != null ? `${m.month}: ${m.count}` : m.month}
              style={{ height: `${peak > 0 ? Math.max(m.value > 0 ? 4 : 0, (m.value / peak) * 70) : 0}%`, backgroundColor: "var(--color-gold)" }} />
          </div>
        ))}
      </div>
      <div className="grid gap-2" style={cols}>
        {months.map((m) => (
          <span key={m.month} className="font-montserrat text-[11px] text-center" style={{ color: "var(--color-muted)" }}>
            {MONTH_FMT.format(new Date(`${m.month}-01T00:00:00Z`))}
          </span>
        ))}
      </div>
      {note && <span className="font-montserrat text-[11px]" style={{ color: "var(--color-muted)" }}>{note}</span>}
    </div>
  );
}
 
/** A tile's filter: the exact rows it counted, handed to the list below it. */
export interface InsightFilter { key: string; label: string; ids: string[] }
 
/**
 * Tap props for a tile that filters the list below it (COD's Overdue pattern): tapping lists
 * exactly the rows it counted; tapping again clears. Nothing when there is nothing to list.
 */
export function filterTap(
  filterKey: string | null | undefined,
  onFilter: ((f: InsightFilter | null) => void) | undefined,
  f: InsightFilter,
): { active?: boolean; actionHint?: string; onClick?: () => void } {
  if (!onFilter || f.ids.length === 0) return {};
  const on = filterKey === f.key;
  return {
    active: on,
    actionHint: on ? "Showing · tap to clear" : "Tap to list them",
    onClick: () => onFilter(on ? null : f),
  };
}
 
/** "Showing … · Clear" above a list a tile has narrowed. */
export function FilterChip({ filter, onClear }: { filter: InsightFilter | null; onClear: () => void }) {
  if (!filter) return null;
  return (
    <div className="mb-3 flex items-center justify-between gap-3 rounded-lg px-3 py-2 font-montserrat text-xs"
      style={{ border: "1px solid var(--color-gold)", backgroundColor: "rgba(201,168,76,0.08)", color: "var(--color-offwhite)" }}>
      <span>Showing {filter.label}</span>
      <button type="button" onClick={onClear} className="font-semibold shrink-0" style={{ color: "var(--color-gold)" }}>
        Clear
      </button>
    </div>
  );
}
