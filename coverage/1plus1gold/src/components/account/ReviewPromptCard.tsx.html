"use client";
 
/**
 * "A word from you" — the My Account review ask, web side.
 *
 * A faithful 1:1 port of the native SDUI card (src/lib/spec/dashboard.ts
 * `reviewCard()`), in the dashboard's own accordion rhythm so it reads as a
 * sibling of the sections around it rather than a banner bolted on. Port-diff:
 * SDUI contract §153.
 *
 * ▶ THE ASK IS QUIET BY DESIGN and that shape was the operator's call. It is an
 * inline card, never a modal or a popup, and it appears only after an order has
 * actually reached the customer — see src/lib/account/review-prompt.ts for the
 * reasoning and the quotes behind it. "Not now" closes it for good.
 *
 * ▶ BOTH LINKS ARE SHOWN TO EVERYONE. There is no "were you happy?" step and no
 * branch on anything we know about the customer: the review link and the "tell
 * us first" link sit side by side for every eligible person. Sorting customers
 * by expected sentiment is review gating, which Trustpilot forbids and which
 * would make the resulting score worth nothing. Do not add a sentiment step.
 *
 * ▶ NO RATING WIDGET. The account carries no reviews yet; a star component with
 * nothing in it reads as broken or as a bad score. Text until the number is
 * real, at which point Trustpilot's own widget is the honest upgrade because
 * the number then comes from them.
 */
import { useEffect, useState } from "react";
import Link from "next/link";
import { ChevronDown } from "lucide-react";
import { BUSINESS_CONFIG } from "@/lib/business-config";
import { TRUSTPILOT_LOCKUP_SVG, TRUSTPILOT_LOCKUP_W, TRUSTPILOT_LOCKUP_H } from "@/lib/trustpilot";
import { REVIEW_PROMPT_COPY, type ReviewPromptState } from "@/lib/account/review-prompt";
 
const GOLD = "var(--color-gold)";
const MUTED = "var(--color-muted)";
 
export default function ReviewPromptCard() {
  const [state, setState] = useState<ReviewPromptState | null>(null);
 
  useEffect(() => {
    const ctrl = new AbortController();
    fetch("/api/account/review-prompt", { signal: ctrl.signal, credentials: "same-origin" })
      .then((r) => (r.ok ? r.json() : null))
      .then((d) => setState((d?.state as ReviewPromptState) ?? "none"))
      .catch((e) => { Eif ((e as Error).name !== "AbortError") setState("none"); });
    return () => ctrl.abort();
  }, []);
 
  // Hidden while unknown AND when the answer is "don't ask" — a card that
  // flashes in and then vanishes is worse than one that never appeared.
  if (state !== "ask") return null;
 
  // Optimistic: the card goes immediately. A failed dismissal is not worth
  // making the customer look at the thing they just closed — it comes back on
  // the next page load, which is the honest recovery.
  const dismiss = () => {
    setState("none");
    void fetch("/api/account/review-prompt", { method: "POST", credentials: "same-origin" })
      .catch(() => {});
  };
 
  return (
    <details className="account-accordion" open>
      <summary className="cursor-pointer list-none flex items-center justify-between pb-2 mb-3"
        style={{ borderBottom: "2px solid var(--color-gold-dim, rgba(201,168,76,0.35))" }}>
        <span className="font-cinzel text-sm md:text-base font-semibold tracking-widest uppercase" style={{ color: GOLD }}>
          {REVIEW_PROMPT_COPY.title}
        </span>
        <ChevronDown size={14} className="account-accordion-chevron" style={{ color: "var(--color-gold-mid)", transition: "transform 0.2s" }} />
      </summary>
 
      <div className="flex flex-col gap-3">
        <p className="font-montserrat text-[13px]" style={{ color: MUTED, lineHeight: 1.5 }}>
          {REVIEW_PROMPT_COPY.body}
        </p>
 
        {/* ▶ `items-start`, NOT `items-center`, AND THE REASON IS THE NATIVE
            TWIN. Once the primary link became two lines tall (lockup over
            label), centring left "Something not right?" floating at the middle
            of a block it has no relationship to — visible the moment it was
            rendered rather than reasoned about. Top-aligning reads as two
            columns starting together.
 
            ▶ It is also the only value that AGREES WITH NATIVE. That row is an
            SDUI hstack with `axis: "wrap"`, and FlowLayout places every child
            at `anchor: .topLeading` unconditionally (SDUIRepeater.swift:171) —
            it ignores `alignment` outright. So `items-end` would have looked
            right on web and been silently impossible on iOS. */}
        <div className="flex flex-wrap items-start gap-x-5 gap-y-3">
          {/* ▶ TRUSTPILOT'S OWN LOCKUP, AT THE FOOTER'S SIZE. Operator,
              2026-09-18: *"I was just looking at the inline trustpilot in
              Account. I think we should make it look same as its in the footer
              with correct logo and Text."*
 
              ⚠️ THIS REPLACES A RECONSTRUCTED MARK, and the file it replaces
              warned about exactly this. Until now the card drew
              `<TrustpilotStar size={16} />` — the flat single-green
              simple-icons path — beside "Trustpilot" set in OUR Montserrat.
              `Footer.tsx` records the tell: *"their star is TWO-TONE (#00b67a
              over #005128), not the flat green a single-path icon set gives
              you"*. So the card was showing an approximation of somebody's
              trademark next to our own typeface, three metres from a footer
              showing the real thing.
 
              ▶ AND THE PRIOR REASONING IS NOW WRONG, not merely superseded.
              `lib/trustpilot.ts` argued the star belongs here because the
              lockup *"would be a banner in the middle of a paragraph"*. That
              holds for a mark sitting INLINE inside a sentence; it does not
              hold once the mark is given its own line above the link, which is
              how the footer composes it too. Corrected there rather than left
              standing.
 
              ▶ SIZE IS THE FOOTER'S 110x27, NOT A SCALED-DOWN ONE. Trustpilot's
              digital guidance sets roughly a 100px floor for the horizontal
              lockup (Footer.tsx §SIZE), so shrinking it to sit beside 12px type
              would breach the one rule we can actually read. Both surfaces
              import the same constants, so they cannot drift apart. */}
          <a
            href={BUSINESS_CONFIG.reviews.write}
            target="_blank"
            rel="noopener noreferrer"
            className="inline-flex flex-col items-start gap-1.5 transition-opacity hover:opacity-80"
          >
            {/* eslint-disable-next-line @next/next/no-img-element -- a static
                brand SVG, same call as the footer: next/image would re-encode
                and proxy someone else's trademark through our optimiser for no
                gain, and an SVG has no raster sizes to pick. */}
            <img
              src={TRUSTPILOT_LOCKUP_SVG}
              alt="Trustpilot"
              width={TRUSTPILOT_LOCKUP_W}
              height={TRUSTPILOT_LOCKUP_H}
              loading="lazy"
              decoding="async"
            />
            <span className="font-montserrat text-[12px] font-semibold" style={{ color: GOLD }}>
              {REVIEW_PROMPT_COPY.ctaWithMark}
            </span>
          </a>
          {/* Always present, never conditional — see the review-gating note. */}
          <Link href="/contact" className="font-montserrat text-[12px]" style={{ color: MUTED }}>
            {REVIEW_PROMPT_COPY.fallback}
          </Link>
        </div>
 
        <div className="flex flex-wrap items-center justify-between gap-x-4 gap-y-1">
          <span className="font-montserrat text-[11px]" style={{ color: MUTED }}>
            {REVIEW_PROMPT_COPY.disclosure}
          </span>
          <button
            onClick={dismiss}
            className="font-montserrat text-[11px] underline underline-offset-2"
            style={{ color: MUTED }}
          >
            {REVIEW_PROMPT_COPY.dismiss}
          </button>
        </div>
      </div>
    </details>
  );
}
