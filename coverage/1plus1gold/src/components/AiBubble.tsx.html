"use client";
 
/**
 * Persistent AI affordance — a small gold FAB pinned bottom-right that opens
 * the existing AiCalculator in a slide-in side panel (right on desktop,
 * bottom-aligned full-width sheet on mobile).
 *
 * Mounted from SiteShell so it follows the user across non-bare pages.
 * Hidden on routes that need full focus: /admin*, /checkout, /payment/confirming.
 * On mobile it floats above the BottomNav so the two don't collide.
 */
 
import { useState, useEffect, useCallback } from "react";
import { usePathname } from "next/navigation";
import { Sparkles, X, Headphones, MessageCircle, ArrowLeft } from "lucide-react";
import AiCalculator from "@/components/AiCalculator";
import LogoMark from "@/components/LogoMark";
import VoicePrefsMenu from "@/components/voice/VoicePrefsMenu";
import PersonaList from "@/components/voice/PersonaList";
import { readPersona, writePersona, readMuted, writeMuted } from "@/lib/voice/preferences";
import { PERSONAS, type PersonaKey } from "@/lib/voice/personas";
import { useOwnedGramsStore } from "@/store/ownedGramsStore";
import { useGoalStore } from "@/store/goalStore";
import { signalSurfaceReady } from "@/lib/embed-ready";
 
const HIDE_PREFIXES = ["/admin", "/checkout", "/payment/confirming", "/coming-soon"];
const FIRST_SEEN_KEY    = "ai-bubble-pulsed";
const SESSION_MODE_KEY  = "ai-bubble-mode";  // "type" | "talk"
 
type Mode = "chooser" | "voice-picker" | "type" | "talk";
 
// `embedded` = hosted inside the iOS app's native sheet (the WebView at
// /embed/ai). In that mode the panel opens immediately (the native FAB is the
// entry — no web FAB), and closing the panel tells native to dismiss the sheet
// via the `aiClose` message bridge. Default false = normal web overlay (FAB).
export default function AiBubble({ embedded = false }: { embedded?: boolean } = {}) {
  const pathname   = usePathname();
  const [open,     setOpen]     = useState(embedded);
  const [hovering, setHovering] = useState(false);
  const [pulse,    setPulse]    = useState(false);
 
  const fetchOwnedGrams = useOwnedGramsStore((s) => s.fetch);
  const ownedGrams      = useOwnedGramsStore((s) => s.ownedGrams);
  const goalGrams       = useGoalStore((s) => s.goalGrams);
 
  // Voice preferences — persisted to localStorage. The AiCalculator
  // reads these via props; the AI bubble owns the UI surface so the
  // toggle persists outside the chat scroll.
  const [persona, setPersona] = useState<PersonaKey>("voice_onyx");
  const [muted,   setMuted]   = useState(false);
  // Voice availability today — driven by /api/ai/voice-quota. When the
  // daily cap is exhausted, the Talk button + speaker icon + Talk chip
  // grey out so the customer doesn't tap into a path that'll fail.
  const [voiceExhausted, setVoiceExhausted] = useState(false);
 
  // Mode chooser: customer picks Type or Talk on first panel-open in a
  // tab session. Subsequent opens skip the chooser and use the saved
  // choice. Mid-conversation switching happens via the header chip.
  // Persisted to sessionStorage (per-tab) — fresh tab gets a new
  // chooser, but reopening the bubble within the same tab doesn't
  // re-prompt. Continuous voice mode mirrors `mode === "talk"`.
  const [mode, setMode] = useState<Mode>("chooser");
  const voiceModeContinuous = mode === "talk";
 
  useEffect(() => { setPersona(readPersona()); setMuted(readMuted()); }, []);
  const handleSetPersona = (k: PersonaKey) => {
    setPersona(k);
    writePersona(k);
    // For logged-in customers, mirror to the DB so the choice travels
    // across devices. 401 (anonymous) is the expected no-op path. We
    // don't await — UI shouldn't wait on telemetry.
    void fetch("/api/account/guide", {
      method:  "PATCH",
      headers: { "Content-Type": "application/json" },
      body:    JSON.stringify({ guide_key: k }),
    }).catch(() => { /* anonymous or transient — fine */ });
  };
  const handleSetMuted   = (m: boolean)    => { setMuted(m);   writeMuted(m);   };
 
  // Refresh voice quota state on bubble open. Drives the grey-out of
  // voice affordances when today's cap is hit (or comes back after a
  // 429 from a session start).
  const refreshVoiceQuota = useCallback(async (signal?: AbortSignal) => {
    try {
      const res  = await fetch("/api/ai/voice-quota", { signal });
      if (!res.ok) return;
      const data = await res.json() as { exhausted: boolean };
      setVoiceExhausted(!!data.exhausted);
    } catch (err) {
      if ((err as Error).name !== "AbortError") {
        /* network glitch — assume voice is available */
      }
    }
  }, []);
 
  useEffect(() => {
    if (!open) return;
    const ctrl = new AbortController();
    void refreshVoiceQuota(ctrl.signal);
    return () => ctrl.abort();
  }, [open, refreshVoiceQuota]);
 
  // Reconcile localStorage with the server when the bubble opens for a
  // logged-in customer. If the server has a guide, it wins (the customer
  // may have picked one on a different device). If only localStorage
  // has a value, push it up. Anonymous callers get 401 — no-op.
  useEffect(() => {
    if (!open) return;
    const ctrl = new AbortController();
    void (async () => {
      try {
        const res = await fetch("/api/account/guide", { signal: ctrl.signal });
        if (!res.ok) return;  // 401 = anonymous, fall through to localStorage
        const data = await res.json() as { guide_key: PersonaKey | null };
        if (data.guide_key) {
          // Server is authoritative — adopt it locally.
          if (data.guide_key !== persona) {
            setPersona(data.guide_key);
            writePersona(data.guide_key);
          }
        } else {
          // Server has nothing yet — push local pick up so the customer's
          // device-1 choice survives logging in on device-2.
          await fetch("/api/account/guide", {
            method:  "PATCH",
            headers: { "Content-Type": "application/json" },
            body:    JSON.stringify({ guide_key: persona }),
            signal:  ctrl.signal,
          });
        }
      } catch (err) {
        if ((err as Error).name !== "AbortError") {
          /* network glitch — localStorage is fine for now */
        }
      }
    })();
    return () => ctrl.abort();
    // Intentionally only re-runs on `open` — persona is read inside.
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [open]);
 
  // On every panel-open, restore the saved mode (or show chooser). On
  // close we don't clear the saved choice — only the in-component mode
  // state resets to "chooser" so the next tab session starts clean
  // *only* if sessionStorage was also wiped (new tab).
  useEffect(() => {
    if (!open) return;
    if (typeof window === "undefined") return;
    const saved = sessionStorage.getItem(SESSION_MODE_KEY);
    if (saved === "talk" || saved === "type") setMode(saved);
    else                                       setMode("chooser");
  }, [open]);
 
  const handleChoose = (choice: "type" | "talk") => {
    setMode(choice);
    if (typeof window !== "undefined") {
      try { sessionStorage.setItem(SESSION_MODE_KEY, choice); } catch { /* ignore */ }
    }
  };
  // Picking Talk on the chooser routes through the voice picker so the
  // customer can audition + confirm a persona before the session opens.
  // The header "Talk" chip (mid-conversation) skips the picker — they
  // already chose, just flip back into voice.
  const handleChoosePicker = () => setMode("voice-picker");
  const handleSwitchMode   = () => handleChoose(mode === "talk" ? "type" : "talk");
 
  // Continuous-mode change events bubble up from AiCalculator (e.g.
  // when the customer manually exits voice). Keep our mode state in
  // sync so the header chip flips correctly.
  const handleContinuousChange = (on: boolean) => {
    if (mode === "chooser") return;
    handleChoose(on ? "talk" : "type");
  };
 
  // First-session pulse: draws attention once, then never again.
  useEffect(() => {
    if (typeof window === "undefined") return;
    if (sessionStorage.getItem(FIRST_SEEN_KEY)) return;
    setPulse(true);
    const t = setTimeout(() => {
      setPulse(false);
      sessionStorage.setItem(FIRST_SEEN_KEY, "1");
    }, 3000);
    return () => clearTimeout(t);
  }, []);
 
  // Lazy-load owned grams the first time the customer opens the bubble.
  useEffect(() => {
    if (open) fetchOwnedGrams();
  }, [open, fetchOwnedGrams]);
 
  // FX rates are preloaded + polled app-wide by SiteShell — no fetch
  // needed here. Cart cards inside the bubble render with fresh rates
  // from first paint.
 
  // Lock body scroll while the panel is open.
  useEffect(() => {
    if (!open) return;
    const prev = document.body.style.overflow;
    document.body.style.overflow = "hidden";
    return () => { document.body.style.overflow = prev; };
  }, [open]);
 
  // Embedded (iOS): the panel renders immediately (open from mount), so signal
  // real paint right away → the native host drops its opaque loading cover at
  // the chooser instead of the 2.5s fallback (no white flash). No-op on web.
  useEffect(() => { if (embedded) signalSurfaceReady(); }, [embedded]);
 
  // Embedded (iOS native sheet): when the panel closes, tell native to dismiss
  // the hosting sheet. Fires only on the open→closed transition (not on the
  // initial open). No-op outside the native shell.
  useEffect(() => {
    if (embedded && !open) {
      // eslint-disable-next-line @typescript-eslint/no-explicit-any
      (window as any).webkit?.messageHandlers?.aiClose?.postMessage(1);
    }
  }, [embedded, open]);
 
  const hidden = HIDE_PREFIXES.some((r) => pathname === r || pathname.startsWith(r + "/"));
  if (hidden) return null;
 
  return (
    <>
      {/* Pulse + label-reveal animations — scoped to this component */}
      <style>{`
        @keyframes ai-bubble-pulse {
          0%   { box-shadow: 0 0 0 0   rgba(201,168,76,0.55); }
          70%  { box-shadow: 0 0 0 18px rgba(201,168,76,0);   }
          100% { box-shadow: 0 0 0 0   rgba(201,168,76,0);   }
        }
        .ai-bubble-pulse { animation: ai-bubble-pulse 1.6s ease-out infinite; }
        @keyframes ai-bubble-panel-in-desktop {
          from { transform: translateX(100%); }
          to   { transform: translateX(0); }
        }
        @keyframes ai-bubble-panel-in-mobile {
          from { transform: translateY(8%); opacity: 0; }
          to   { transform: translateY(0);  opacity: 1; }
        }
        .ai-bubble-panel       { animation: ai-bubble-panel-in-mobile 200ms ease-out; }
        @media (min-width: 640px) {
          .ai-bubble-panel     { animation: ai-bubble-panel-in-desktop 220ms ease-out; }
        }
      `}</style>
 
      {/* ── FAB ─────────────────────────────────────────────────────────────── */}
      {/* Hidden when embedded — the iOS app's NATIVE bubble is the entry. */}
      {!embedded && (
      <button
        data-ai-bubble
        onClick={() => setOpen(true)}
        onMouseEnter={() => setHovering(true)}
        onMouseLeave={() => setHovering(false)}
        aria-label="1+1 GOLD AI"
        className="fixed right-4 z-40 flex items-center rounded-full transition-transform duration-150 hover:scale-105 active:scale-95"
        style={{
          // Above BottomNav on mobile (~70px tall + safe area), 24px from bottom on desktop
          bottom: "calc(env(safe-area-inset-bottom, 0px) + 92px)",
          background: "linear-gradient(135deg, var(--color-gold) 0%, #B8923E 100%)",
          color: "var(--color-bg)",
          boxShadow: "0 4px 16px rgba(201,168,76,0.35), 0 1px 4px rgba(0,0,0,0.3)",
        }}
      >
        <span className={`flex items-center justify-center w-14 h-14 rounded-full ${pulse ? "ai-bubble-pulse" : ""}`}>
          <Sparkles size={22} strokeWidth={2.2} />
        </span>
        {/* Hover expand — uses the stacked LogoMark + 'AI' (matches the panel
            header). Brand spec is "logo always stacked", so we render the
            real LogoMark instead of plain text. */}
        <span
          className="flex items-center gap-2 whitespace-nowrap overflow-hidden transition-all duration-200"
          style={{
            maxWidth:     hovering ? "200px" : "0px",
            opacity:      hovering ? 1 : 0,
            paddingLeft:  hovering ? "4px" : "0px",
            paddingRight: hovering ? "16px" : "0px",
          }}
        >
          <LogoMark size={18} numeralColor="var(--color-bg)" wordColor="var(--color-bg)" />
          <span className="w-px h-7 shrink-0" style={{ backgroundColor: "rgba(10,10,8,0.35)" }} />
          <span className="font-cinzel text-base font-bold leading-none">AI</span>
        </span>
      </button>
      )}
 
      {/* ── Panel ──────────────────────────────────────────────────────────── */}
      {open && (
        <>
          <div
            className="fixed inset-0 z-[55]"
            style={{ backgroundColor: "rgba(0,0,0,0.55)" }}
            onClick={() => setOpen(false)}
            aria-hidden
          />
          <aside
            className="ai-bubble-panel fixed z-[60] flex flex-col top-0 right-0 left-0 bottom-0 sm:left-auto sm:w-[440px] sm:max-w-[92vw]"
            style={{
              backgroundColor: "var(--color-bg)",
              borderLeft:      "1px solid var(--color-rule)",
              // Honour the iOS / Android status-bar safe area so the
              // header logo + guide name don't sit under the notch.
              paddingTop:    "env(safe-area-inset-top, 0px)",
              paddingBottom: "env(safe-area-inset-bottom, 0px)",
            }}
            role="dialog"
            aria-label="1+1 GOLD AI"
          >
            <header className="flex items-center justify-between px-4 py-4 border-b shrink-0" style={{ borderColor: "var(--color-rule)" }}>
              <div className="flex items-center gap-3 min-w-0">
                <LogoMark size={26} />
                <div className="w-px h-8 shrink-0" style={{ backgroundColor: "var(--color-gold-dim)" }} />
                {mode === "talk" ? (
                  <div className="flex flex-col min-w-0">
                    <span className="font-montserrat text-[9px] tracking-widest uppercase" style={{ color: "var(--color-muted)" }}>
                      Your guide
                    </span>
                    <span className="font-cinzel text-2xl font-bold leading-none truncate" style={{ color: "var(--color-gold)" }}>
                      {PERSONAS[persona].name}
                    </span>
                  </div>
                ) : (
                  <span className="font-cinzel text-2xl font-bold leading-none" style={{ color: "var(--color-cream)" }}>AI</span>
                )}
              </div>
              <div className="flex items-center gap-1">
                {/* Mode switch chip — only after the customer has chosen.
                    Replaces the old standalone headphones icon: same
                    position, but contextual label ("Talk" while in Type,
                    "Type" while in Talk) so the next action is unambiguous. */}
                {(mode === "type" || mode === "talk") && (
                  <button
                    type="button"
                    onClick={handleSwitchMode}
                    disabled={mode === "type" && voiceExhausted}
                    title={
                      mode === "talk"           ? "Switch to typing"
                      : voiceExhausted          ? "Voice rests until tomorrow — type works now"
                      :                           "Switch to voice"
                    }
                    aria-label={mode === "talk" ? "Switch to typing" : "Switch to voice"}
                    className="flex items-center gap-1.5 px-2.5 py-1 rounded-full text-[11px] font-montserrat transition-opacity hover:opacity-80 disabled:cursor-not-allowed disabled:hover:opacity-100"
                    style={{
                      color:           mode === "talk" ? "var(--color-gold)" : "var(--color-offwhite)",
                      backgroundColor: mode === "talk" ? "rgba(201,168,76,0.15)" : "rgba(232,213,163,0.06)",
                      border:          `1px solid ${mode === "talk" ? "rgba(201,168,76,0.45)" : "var(--color-rule)"}`,
                      opacity:         (mode === "type" && voiceExhausted) ? 0.4 : 1,
                    }}
                  >
                    {mode === "talk" ? <MessageCircle size={12} /> : <Headphones size={12} />}
                    {mode === "talk" ? "Type"          : "Talk"}
                  </button>
                )}
                <VoicePrefsMenu
                  persona={persona}
                  muted={muted}
                  onPersona={handleSetPersona}
                  onMute={handleSetMuted}
                  disabled={voiceExhausted}
                />
                {/* Close / End — in Talk mode this is the primary "I'm done"
                    affordance, so it gets a label and a slightly more emphatic
                    border-tinted style. In Type/chooser modes it stays a quiet
                    icon-only X. Testers couldn't find a way out of Talk mode
                    quickly, hence the explicit label. */}
                {mode === "talk" ? (
                  <button
                    onClick={() => setOpen(false)}
                    className="flex items-center gap-1.5 px-2.5 py-1 rounded-full text-[11px] font-montserrat font-semibold transition-opacity hover:opacity-80"
                    style={{
                      color:           "#E05050",
                      backgroundColor: "rgba(224,80,80,0.10)",
                      border:          "1px solid rgba(224,80,80,0.45)",
                    }}
                    aria-label="End conversation"
                  >
                    <X size={12} strokeWidth={2.5} />
                    End
                  </button>
                ) : (
                  <button onClick={() => setOpen(false)} className="p-1.5 rounded-full transition-opacity hover:opacity-70" style={{ color: "var(--color-muted)" }} aria-label="Close">
                    <X size={20} />
                  </button>
                )}
              </div>
            </header>
            <div className="flex-1 min-h-0 overflow-y-auto">
              {mode === "chooser" ? (
                <ModeChooser
                  onChooseType={() => handleChoose("type")}
                  onChooseTalk={handleChoosePicker}
                  voiceExhausted={voiceExhausted}
                />
              ) : mode === "voice-picker" ? (
                <VoicePickerStep
                  persona={persona}
                  onPersona={handleSetPersona}
                  onStart={() => handleChoose("talk")}
                  onBack={() => setMode("chooser")}
                />
              ) : (
                <AiCalculator
                  ownedGrams={ownedGrams}
                  goalGrams={goalGrams}
                  voicePersona={persona}
                  voiceMuted={muted}
                  continuousVoiceMode={voiceModeContinuous}
                  onContinuousVoiceModeChange={handleContinuousChange}
                  onVoiceQuotaExhausted={() => { void refreshVoiceQuota(); }}
                />
              )}
            </div>
          </aside>
        </>
      )}
    </>
  );
}
