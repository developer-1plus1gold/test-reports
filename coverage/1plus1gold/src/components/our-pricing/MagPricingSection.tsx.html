"use client";
 
/**
 * Our Pricing — 1+1Gold MAG section (web only; the iOS Our Pricing surface is
 * a server-authored spec and gains MAG with the native build).
 *
 * The per-bar listing card was removed 2026-10-07 (Mukesh): the MAG offering
 * card at the top of the page already shows the premium range. This section
 * keeps the explanation of how MAG is priced; listings are still read (from the
 * same public route the shop uses) only to decide whether to link to the shop.
 * Wording follows src/lib/mag/home-copy.ts: MAG is never described by
 * reference to standard bars.
 */
import { useEffect, useState } from "react";
import Link from "next/link";
import { MAG_ACCENT, MAG_HOME_CARD } from "@/lib/mag/home-copy";
import type { MagListingPublic } from "@/lib/mag/listings";
import { magTierLabel, magPricingBars } from "@/lib/mag/pricing-rows";
 
export { magTierLabel };
 
export default function MagPricingSection() {
  const [listings, setListings] = useState<MagListingPublic[] | null>(null);
 
  useEffect(() => {
    const ctrl = new AbortController();
    fetch("/api/mag/listings", { signal: ctrl.signal })
      .then((r) => (r.ok ? r.json() : { listings: [] }))
      .then((d) => setListings(d.listings ?? []))
      .catch((err) => { if ((err as Error).name !== "AbortError") setListings([]); });
    return () => ctrl.abort();
  }, []);
 
  const shown = magPricingBars(listings ?? []);
 
  return (
    <div id="mag" className="card rounded-xl p-8 mb-10 scroll-mt-24">
      <h2 className="font-cinzel text-xl font-semibold mb-1" style={{ color: MAG_ACCENT }}>
        1+1Gold MAG — My Allocated Gold
      </h2>
      <p className="font-montserrat text-sm mb-6" style={{ color: "var(--color-muted)" }}>
        {MAG_HOME_CARD.body}
      </p>
 
      <div className="rounded-lg px-5 py-4" style={{ backgroundColor: "rgba(217,160,138,0.06)", border: "1px solid rgba(217,160,138,0.2)" }}>
        <p className="font-montserrat text-sm leading-relaxed" style={{ color: "var(--color-offwhite)" }}>
          <span style={{ color: MAG_ACCENT, fontWeight: 600 }}>How MAG is priced.</span>
          {" "}Live spot plus the premium for the number of grams in that purchase, locked when you confirm. No storage fee while
          we hold it. When you sell back, each gram is valued at spot plus that bar&apos;s buyback premium; when you apply MAG towards a physical bar at checkout, it counts at its full MAG value — what the same grams cost to buy that day.
          {" "}<Link href="/mag-agreement" className="underline" style={{ color: "var(--color-gold)" }}>MAG Agreement</Link>
          {shown.length > 0 && <>{" · "}<Link href="/shop#mag" className="underline" style={{ color: "var(--color-gold)" }}>See MAG bars →</Link></>}
        </p>
      </div>
    </div>
  );
}
