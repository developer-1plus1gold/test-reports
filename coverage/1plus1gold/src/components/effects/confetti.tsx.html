"use client";
 
/**
 * Confetti — zero-dependency, CSS-keyframe confetti shared by the launch
 * ceremony (ribbon cutting) and the occasion / sale campaign panel.
 *
 * `host` must be `position: relative; overflow: hidden` — pieces are absolutely
 * positioned inside it and remove themselves when their fall ends. Render
 * <ConfettiStyles/> once on the page. Honours prefers-reduced-motion (pieces
 * are hidden). Brand colours only.
 */
 
export const CONFETTI_COLORS = ["#C9A84C", "#E8D5A3", "#FFFFFF", "#C9783A", "#1D9E75"];
 
/** Spawn N confetti pieces inside `host`, each removed when its fall ends. The
 *  pieces fall the full height of the section with a little horizontal drift. */
export function fireConfetti(host: HTMLElement | null, count: number) {
  if (!host || typeof document === "undefined") return;
  const W = host.clientWidth || 320;
  const H = Math.max(360, host.clientHeight || 0);
  for (let i = 0; i < count; i++) {
    const el = document.createElement("span");
    const color = CONFETTI_COLORS[i % CONFETTI_COLORS.length];
    const left = Math.round((i / count) * W + (((i * 53) % 60) - 30));
    const drift = (((i * 47) % 160) - 80); // px sideways over the fall
    const dur = 1900 + ((i * 37) % 1600);
    const delay = (i * 13) % 700;
    const size = 6 + ((i * 7) % 9);
    el.className = "lc-confetti";
    el.style.left = `${left}px`;
    el.style.width = `${size}px`;
    el.style.height = `${size * 0.5}px`;
    el.style.background = color;
    el.style.setProperty("--lc-x", `${drift}px`);
    el.style.setProperty("--lc-fall", `${H + 40}px`);
    el.style.animationDuration = `${dur}ms`;
    el.style.animationDelay = `${delay}ms`;
    host.appendChild(el);
    window.setTimeout(() => el.remove(), dur + delay + 200);
  }
}
 
/** Several confetti bursts in sequence — a sustained celebratory shower. */
export function fireConfettiWaves(host: HTMLElement | null, perWave: number, waves: number, gapMs: number) {
  for (let w = 0; w < waves; w++) {
    window.setTimeout(() => fireConfetti(host, perWave), w * gapMs);
  }
}
 
export function ConfettiStyles() {
  return (
    <style>{`
      .lc-confetti {
        position: absolute; top: -16px; border-radius: 1px;
        opacity: 0; will-change: transform, opacity;
        animation-name: lc-fall; animation-timing-function: ease-in; animation-fill-mode: forwards;
      }
      @keyframes lc-fall {
        0%   { transform: translate(0, 0) rotate(0deg); opacity: 1; }
        100% { transform: translate(var(--lc-x, 0px), var(--lc-fall, 460px)) rotate(720deg); opacity: 0; }
      }
      @media (prefers-reduced-motion: reduce) { .lc-confetti { display: none; } }
    `}</style>
  );
}
