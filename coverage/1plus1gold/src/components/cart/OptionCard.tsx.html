"use client";
 
/**
 * OptionCard — shared "Choose your gold" card used by the calculator
 * page and the AI calculator (text + voice).
 *
 * Purpose: gold-only summary with the curve USP gauge when relevant.
 * Shipping + payment-fee detail is intentionally NOT shown here — the
 * AI's role is gold-pricing first; logistics are follow-up territory.
 * Customer taps Add-to-Cart and picks shipping/payment in the actual
 * cart, which is the source of truth for those.
 *
 * The card adapts to its content:
 *  • Curve sim with a curveComparison → renders the "Where you are
 *    on the curve" gauge with first-time vs closing-cycle endpoints
 *  • Standard sim → no gauge, just items + totals
 *  • Mixed sim → gauge if there's curve content, header copy adapts
 */
 
import { useState } from "react";
import { Check, ShoppingCart } from "lucide-react";
import Flag from "@/components/Flag";
import { PRODUCT_CATALOG } from "@/lib/products";
import { formatPremium } from "@/lib/utils";
import { type CatalogProduct, getThumb } from "@/store/catalogStore";
import type { AiCartSimulation } from "@/lib/ai-calculator";
import { RegMarks } from "@/components/brand/RegMarks";
 
interface Props {
  /** Header label, e.g. "On the Premium Curve" or "As a Standard Bar". */
  title: string;
  sim:   AiCartSimulation;
  /** Small note shown when the standard option doesn't exactly match
   *  the customer's grams target — e.g. "3g more than asked". */
  note?: string | null;
  /** One-line "why this option" hook shown above the totals. */
  hook:  string;
  /** Whether the viewer is logged-in with at least 1g accumulated.
   *  Adapts the gauge's footer copy from "save by accumulating" to
   *  "you've moved this far along the curve already". */
  isLoggedInWithGrams?: boolean;
  /** Catalog rows for image lookup (standard bars + accessories).
   *  Curve splits use the hardcoded PRODUCT_CATALOG. */
  catalogProducts: CatalogProduct[];
  /** SGD-to-display-currency formatter (already converts internally;
   *  pass raw SGD numbers). */
  fmt: (sgd: number) => string;
  /** Active currency config — used for the flag glyph. */
  cfg: { flag: string };
  /** Add-this-option to cart and navigate / route as appropriate. Omit for an
   *  EXAMPLE card (a what-if price for a later order): checkout prices from the
   *  customer's real curve position, so that price cannot be added as is.
   *  `exampleNote` is shown in place of the button. */
  onAdd?: () => void;
  exampleNote?: string;
  /** Two presentation modes:
   *    "page"    — rendered standalone on the calculator page (full
   *                styling, large totals, large CTA)
   *    "compact" — rendered inline inside an AI bubble message; uses
   *                tighter spacing so it doesn't dominate the chat
   *  Default is "page". */
  variant?: "page" | "compact";
  /** Cart-state cue — when truthy and the user clicks Add-to-Cart,
   *  the button briefly shows a green-check confirmation in place
   *  of immediate navigation. Compact (AI) variant uses this for
   *  inline feedback when adding to an existing cart. */
  inlineConfirmOnAdd?: boolean;
}
 
export default function OptionCard({
  title,
  sim,
  note,
  hook,
  isLoggedInWithGrams,
  catalogProducts,
  fmt,
  cfg,
  onAdd,
  exampleNote,
  variant            = "page",
  inlineConfirmOnAdd = false,
}: Props) {
  const [added, setAdded] = useState(false);
  const handleAdd = () => {
    if (!onAdd) return;
    onAdd();
    if (inlineConfirmOnAdd) {
      setAdded(true);
      setTimeout(() => setAdded(false), 2500);
    }
  };
 
  const totalSgd   = sim.goldSubtotalSgd + sim.accessorySubtotalSgd;
  const perGramSgd = sim.totalGrams > 0 ? sim.goldSubtotalSgd / sim.totalGrams : 0;
 
  // Effective premium: derived from gold prices vs spot grams across
  // all gold lines (curve + standard). Replaces sim.blendedPremium for
  // display because that field is curve-only — for a standard-only or
  // mixed sim it would render as 0% which is wrong.
  const goldGrams = sim.items
    .filter((i) => i.type === "curve_split" || i.type === "standard_bar")
    .reduce((s, i) => s + i.grams * i.count, 0);
  const goldOnlySubtotal = sim.items
    .filter((i) => i.type === "curve_split" || i.type === "standard_bar")
    .reduce((s, i) => s + i.priceEach * i.count, 0);
  const displayPremium = goldGrams > 0 && sim.spotSgd > 0
    ? goldOnlySubtotal / (sim.spotSgd * goldGrams) - 1
    : sim.blendedPremium;
 
  const comparison = sim.curveComparison ?? null;
 
  // Density tokens — pulled out so the JSX below stays readable.
  const padding   = variant === "compact" ? "p-3.5" : "p-5";
  const titleSize = variant === "compact" ? "text-[9px]" : "text-[10px]";
  const totalSize = variant === "compact" ? "text-lg"   : "text-xl";
  const ctaPad    = variant === "compact" ? "py-2.5"    : "py-3";
 
  return (
    <div
      className={`flex flex-col ${padding} rounded-xl border`}
      style={{ borderColor: "var(--color-rule)", backgroundColor: "var(--color-bg)" }}
    >
      <p className={`font-montserrat ${titleSize} tracking-widest uppercase mb-3`} style={{ color: "var(--color-gold-mid)" }}>
        <RegMarks text={title} />
      </p>
 
      {/* Item rows — usually one per item type */}
      <div className="flex flex-col gap-2 mb-4">
        {sim.items.map((item, idx) => {
          let imageUrl: string | null = null;
          // The product's FIRST media image (admin sort order) — the same
          // picture the shop shows. `image_url` is not in the /api/products
          // payload, so reading it rendered no image at all.
          if (item.type === "curve_split") {
            const curve = catalogProducts.find((p) => p.product_type === "curve" && p.grams === item.grams);
            imageUrl = (curve ? getThumb(curve) : null)
              ?? PRODUCT_CATALOG.find((p) => p.g === item.grams)?.imageUrl ?? null;
          } else Eif (item.type === "standard_bar") {
            const std = catalogProducts.find((p) => p.product_type === "standard" && p.grams === item.grams);
            imageUrl = std ? getThumb(std) : null;
          } else if (item.type === "accessory") {
            const acc = catalogProducts.find(
              (p) => p.product_type === "accessory" && p.label.toLowerCase() === item.label.toLowerCase(),
            );
            imageUrl = acc ? getThumb(acc) : null;
          }
          return (
            <div key={`${item.type}-${item.label}-${idx}`} className="flex items-center gap-3">
              {imageUrl ? (
                // eslint-disable-next-line @next/next/no-img-element
                <img
                  src={imageUrl} alt={item.label}
                  className="w-12 h-12 rounded shrink-0"
                  style={{ objectFit: "contain" }}
                />
              ) : (
                <div
                  className="w-12 h-12 rounded flex items-center justify-center font-cinzel text-[10px] font-bold shrink-0"
                  style={{ backgroundColor: "var(--color-bg-card)", color: "var(--color-gold)" }}
                >
                  {item.type === "accessory" ? "ACC" : `${item.grams}g`}
                </div>
              )}
              <div className="flex-1 min-w-0">
                <p className="font-cinzel text-sm font-semibold truncate" style={{ color: "var(--color-gold)" }}>
                  ×{item.count} {item.label}
                </p>
                <p className="font-montserrat text-[11px] flex items-center gap-1 mt-0.5" style={{ color: "var(--color-muted)" }}>
                  <Flag code={cfg.flag} />{fmt(item.priceEach)} each
                </p>
              </div>
            </div>
          );
        })}
      </div>
 
      {note && (
        <p className="font-montserrat text-[11px] mb-3" style={{ color: "var(--color-muted)" }}>
          {note}
        </p>
      )}
 
      {/* Curve USP gauge — only when this option is a curve simulation
          with a non-trivial cycle spread. Shows the customer's price
          dot between the "starting" and "closing" endpoints of a 50g
          cycle so they can SEE where they are without mental math. */}
      {comparison && (() => {
        const span = comparison.firstTimePriceSgd - comparison.regularPriceSgd;
        const rawProgress = span > 0
          ? (comparison.firstTimePriceSgd - totalSgd) / span
          : 1;
        const progress = Math.min(Math.max(rawProgress, 0.04), 0.96);
        return (
          <div
            className="rounded-lg p-3.5 mb-4"
            style={{
              backgroundColor: "rgba(201,168,76,0.06)",
              border:          "1px solid rgba(201,168,76,0.20)",
            }}
          >
            <p className="font-montserrat text-[10px] tracking-widest uppercase mb-3" style={{ color: "var(--color-gold)" }}>
              Where you are on the curve
            </p>
 
            <div className="flex items-baseline justify-center gap-1.5 mb-2.5">
              <span className="font-montserrat text-[10px] tracking-widest uppercase" style={{ color: "var(--color-muted)" }}>
                Your rate
              </span>
              <span className="font-montserrat text-sm font-bold flex items-center gap-1" style={{ color: "var(--color-gold)" }}>
                <Flag code={cfg.flag} />{fmt(totalSgd)}
              </span>
              <span className="font-montserrat text-xs" style={{ color: "var(--color-muted)" }}>·</span>
              <span className="font-montserrat text-sm font-bold" style={{ color: "var(--color-gold)" }}>
                {formatPremium(displayPremium)}
              </span>
            </div>
 
            <div className="relative h-2 mb-2">
              <div
                className="absolute inset-0 rounded-full"
                style={{ background: "linear-gradient(to right, rgba(201,168,76,0.15), rgba(201,168,76,0.55))" }}
              />
              <div className="absolute w-2 h-2 rounded-full" style={{ left: 0, top: 0, backgroundColor: "var(--color-muted)" }} />
              <div className="absolute w-2 h-2 rounded-full" style={{ right: 0, top: 0, backgroundColor: "var(--color-gold)" }} />
              <div
                className="absolute w-3 h-3 rounded-full -translate-x-1/2"
                style={{
                  left:            `${progress * 100}%`,
                  top:             "-2px",
                  backgroundColor: "var(--color-gold)",
                  boxShadow:       "0 0 0 3px var(--color-bg-card), 0 0 0 4px rgba(201,168,76,0.4)",
                }}
              />
            </div>
 
            <div className="flex items-start justify-between gap-2 mb-2">
              <div className="text-left">
                <p className="font-montserrat text-[11px] font-semibold flex items-center gap-1" style={{ color: "var(--color-offwhite)" }}>
                  <Flag code={cfg.flag} />{fmt(comparison.firstTimePriceSgd)}
                </p>
                <p className="font-montserrat text-[9px] uppercase tracking-wider mt-0.5" style={{ color: "var(--color-muted)" }}>
                  Starting a 50g cycle
                </p>
              </div>
              <div className="text-right">
                <p className="font-montserrat text-[11px] font-semibold flex items-center gap-1 justify-end" style={{ color: "var(--color-gold)" }}>
                  <Flag code={cfg.flag} />{fmt(comparison.regularPriceSgd)}
                </p>
                <p className="font-montserrat text-[9px] uppercase tracking-wider mt-0.5" style={{ color: "var(--color-muted)" }}>
                  Closing a 50g cycle
                </p>
              </div>
            </div>
 
            <p className="font-montserrat text-[10px] mt-2 pt-2 border-t" style={{ color: "var(--color-muted)", borderColor: "var(--color-rule)", lineHeight: 1.4 }}>
              {isLoggedInWithGrams
                ? <>You&apos;ve moved <Flag code={cfg.flag} className="mx-0.5" />{fmt(comparison.firstTimePriceSgd - totalSgd)} along the curve already. Reach the 50g milestone to land at the cheapest rate.</>
                : <>The curve runs in 50g cycles. Every gram you add slides this dot right toward the cheaper rate — up to <Flag code={cfg.flag} className="mx-0.5" />{fmt(comparison.savingsSgd)} saved per cycle.</>
              }
            </p>
          </div>
        );
      })()}
 
      {/* Hook + totals + button anchored to bottom of the card so they
          line up across multiple cards in a grid. */}
      <p className="font-montserrat text-[11px] leading-relaxed mb-4 mt-auto" style={{ color: "var(--color-offwhite)", opacity: 0.85 }}>
        {hook}
      </p>
 
      <div className="flex items-end justify-between gap-3 pb-4 mb-4 border-b" style={{ borderColor: "var(--color-rule)" }}>
        <div>
          <p className={`font-cinzel ${totalSize} font-bold flex items-center gap-1`} style={{ color: "var(--color-gold)" }}>
            <Flag code={cfg.flag} />{fmt(totalSgd)}
          </p>
          <p className="font-montserrat text-[11px] mt-0.5" style={{ color: "var(--color-muted)" }}>
            {sim.totalGrams.toFixed(2)}g · <Flag code={cfg.flag} className="mx-0.5" />{fmt(perGramSgd)}/g
          </p>
        </div>
        <p className="font-montserrat text-xs font-semibold text-right" style={{ color: "var(--color-offwhite)" }}>
          {formatPremium(displayPremium)}
          <br />
          <span className="font-normal text-[10px]" style={{ color: "var(--color-muted)" }}>above spot</span>
        </p>
      </div>
 
      {!onAdd ? (
        <p className="font-montserrat text-[11px] leading-relaxed text-center" style={{ color: "var(--color-muted)" }}>
          {exampleNote ?? "Example price — not available to add."}
        </p>
      ) : (
      <button
        onClick={handleAdd}
        className={`w-full font-montserrat font-semibold text-sm ${ctaPad} rounded-lg transition-colors flex items-center justify-center gap-2`}
        style={{
          backgroundColor: added ? "rgba(201,168,76,0.15)" : "var(--color-gold)",
          color:           added ? "var(--color-gold)" : "var(--color-bg)",
          border:          added ? "1px solid var(--color-gold-dim)" : "none",
        }}
      >
        {added ? (
          <><Check size={14} /> Added to Cart</>
        ) : (
          <><ShoppingCart size={14} /> Add to Cart — <Flag code={cfg.flag} className="mx-0.5" />{fmt(totalSgd)}</>
        )}
      </button>
      )}
    </div>
  );
}
