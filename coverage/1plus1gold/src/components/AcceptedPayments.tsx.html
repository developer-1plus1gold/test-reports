"use client";
 
/**
 * AcceptedPayments — a "secure online payments" strip showing the brand marks
 * of every CURRENTLY-ENABLED online payment method.
 *
 * Single source of truth: it reads the SAME `/api/payments/methods` endpoint
 * the checkout uses (backed by `payment_gateway_fees`) and renders the SAME
 * `PaymentMethodLogos` marks. So enabling/disabling a method in one place
 * (e.g. flipping `is_active` for GrabPay/Atome) updates the checkout AND this
 * strip together — no parallel list to keep in sync.
 *
 * Shows only `is_active` gateway methods (not COD/offline, not coming-soon),
 * each payment brand ONCE across gateways (lib/payments/accepted-strip.ts).
 */
 
import { fetchMethodsCached } from "@/lib/payments/methods-client";
import { useEffect, useState } from "react";
import { PaymentMethodLogos } from "@/components/PaymentMethodLogos";
import { acceptedStripEntries } from "@/lib/payments/accepted-strip";
 
interface Method { method_key: string; is_active: boolean; sort_order: number; card_networks?: string[] | null }
 
// Method keys that have a renderable brand mark in PaymentMethodLogos. Other
// active rows (stripe_*, uob_paynow) render nothing, so we skip them to avoid an
// empty gap.
const RENDERABLE = new Set(["hitpay_paynow", "hitpay_card", "hitpay_grabpay", "hitpay_atome", "hitpay_grab_paylater", "hitpay_shopeepay", "hitpay_shopback", "hitpay_upi", "razorpay_paynow", "razorpay_card", "razorpay_paylater", "airwallex_paynow", "airwallex_card", "airwallex_atome", "airwallex_wechatpay", "airwallex_alipaycn", "airwallex_spaylater", "airwallex_pay"]);
 
export default function AcceptedPayments({ align = "center", label = "Secure online payments", showLabel = true }: { align?: "center" | "start"; label?: string; showLabel?: boolean }) {
  const [methods, setMethods] = useState<Method[] | null>(null);
 
  useEffect(() => {
    // Shared, one request per page load (lib/payments/methods-client). The
    // `live` flag replaces the old AbortController: aborting a SHARED request
    // would cancel it for every other caller.
    let live = true;
    fetchMethodsCached().then((rows) => { if (live) setMethods(rows); });
    return () => { live = false; };
  }, []);
 
  const active = (methods ?? [])
    .filter((m) => m.is_active && RENDERABLE.has(m.method_key))
    .sort((a, b) => a.sort_order - b.sort_order);
 
  // Reserve height so the strip doesn't shift layout while loading; render
  // nothing if there are genuinely no enabled methods.
  if (methods !== null && active.length === 0) return null;
 
  return (
    <div className={`flex flex-col gap-2 ${align === "center" ? "items-center" : "items-start"}`}>
      {showLabel && (
        <span className="font-montserrat text-[10px] uppercase tracking-[0.18em]" style={{ color: "var(--color-muted)" }}>
          {label}
        </span>
      )}
      <div
        className={`flex flex-wrap items-center gap-1.5 ${align === "center" ? "justify-center" : "justify-start"}`}
        style={{ minHeight: 28 }}
        aria-label="Accepted payment methods"
      >
        {/* Each brand once: PayNow / Visa / Mastercard are offered by both gateways. */}
        {acceptedStripEntries(active).map((e) => <PaymentMethodLogos key={e.methodKey} methodKey={e.methodKey} networks={e.networks} />)}
      </div>
    </div>
  );
}
