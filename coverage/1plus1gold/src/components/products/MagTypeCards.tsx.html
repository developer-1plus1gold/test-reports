"use client";
 
/**
 * The three ways to hold MAG, always shown on the Shop (operator 2026-10-25),
 * even when no bar of a type is listed. Each card explains the type in a line
 * or two and carries a request: notify me / my own bar / a group pool. The
 * request lands in Admin → MAG, where the bar and its members are set up.
 * Copy lives in lib/mag/pool-requests.ts so web and native cannot drift.
 * While an open-pool bar has grams left, the Open pool card is an explanation
 * with no action (requestType "info", Mukesh 2026-11-13).
 */
import { useEffect, useState } from "react";
import Link from "next/link";
import { MAG_ACCENT } from "@/lib/mag/home-copy";
import { magTypeCards, type MagTypeCard } from "@/lib/mag/pool-requests";
import { MAG_SHOP_COPY, MAG_OFFER_COPY } from "@/lib/mag/shop-copy";
 
const COPY = { ...MAG_SHOP_COPY, ...MAG_OFFER_COPY };
import { durationLabel, premiumLabel, type DurationUnit, type OfferProduct } from "@/lib/mag/pool-offers";
 
/** The public offers payload (GET /api/mag/pool-offers). */
export interface OfferMenu {
  products: OfferProduct[];
  quoteSizes: { grams: number; label: string }[];
  quoteDurations: { durationValue: number; durationUnit: DurationUnit }[];
}
const EMPTY_MENU: OfferMenu = { products: [], quoteSizes: [], quoteDurations: [] };
const QUOTE_PREFIX = "quote:";
 
type SendState = "idle" | "sending" | "sent" | "already";
 
const inputCls = "w-full rounded px-3 py-2 font-montserrat text-sm bg-transparent";
const inputStyle = { border: "1px solid var(--color-rule)", color: "var(--color-offwhite)" };
 
function TypeCard({ card, signedIn, menu }: { card: MagTypeCard; signedIn: boolean; menu: OfferMenu }) {
  const [open, setOpen] = useState(false);
  // Pool offers (owner 2026-11-14): with offers set up, own bar / group pool is
  // ONE line — bar ▾, duration ▾, the premium, Request. The server re-reads the
  // offer and snapshots the price; the client never sends one.
  // A customer may also name a size we do not list (20 g / 1 oz / 50 g / 100 g)
  // with any duration on the list; we quote the premium ("quote:<grams>").
  const [choice, setChoice] = useState("");
  const [durationKey, setDurationKey] = useState("");
  const offers = menu.products;
  const useOffers = (offers.length > 0 || menu.quoteSizes.length > 0)
    && (card.requestType === "own_bar" || card.requestType === "group_pool");
  const quoteGrams = choice.startsWith(QUOTE_PREFIX) ? Number(choice.slice(QUOTE_PREFIX.length)) : null;
  const quoteSize = quoteGrams != null ? menu.quoteSizes.find(s => s.grams === quoteGrams) ?? null : null;
  const product = quoteGrams == null ? offers.find(p => p.productId === choice) ?? null : null;
  const offer = product?.durations.find(d => d.offerId === durationKey) ?? null;
  const quoteDuration = quoteSize ? menu.quoteDurations.find(d => `${d.durationValue}-${d.durationUnit}` === durationKey) ?? null : null;
  const ready = !!offer || !!quoteDuration;
  // Only for an offered bar that is itself on the shop (Mukesh 11-14).
  const outrightPct = product?.outrightPct ?? null;
  const [grams, setGrams] = useState("");
  const [groupName, setGroupName] = useState("");
  const [members, setMembers] = useState("");
  const [state, setState] = useState<SendState>("idle");
  const [error, setError] = useState<string | null>(null);
  const needsForm = card.requestType !== "notify";
 
  async function send() {
    setState("sending"); setError(null);
    try {
      const res = await fetch("/api/mag/pool-requests", {
        method: "POST", credentials: "include", headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          requestType: card.requestType,
          offerId: useOffers && offer ? offer.offerId : undefined,
          quoteGrams: useOffers && quoteDuration ? quoteSize!.grams : undefined,
          durationValue: useOffers && quoteDuration ? quoteDuration.durationValue : undefined,
          durationUnit: useOffers && quoteDuration ? quoteDuration.durationUnit : undefined,
          grams: useOffers || grams.trim() === "" ? null : grams.trim(),
          groupName: card.requestType === "group_pool" ? groupName : undefined,
          members: card.requestType === "group_pool" ? members : undefined,
        }),
      });
      const d = await res.json().catch(() => ({}));
      if (!res.ok) { setError(d.error ?? COPY.requestError); setState("idle"); return; }
      setState(d.alreadyOpen ? "already" : "sent");
    } catch {
      setError(COPY.requestError); setState("idle");
    }
  }
 
  const done = state === "sent" || state === "already";
  const infoOnly = card.requestType === "info";
  return (
    <div className="rounded-lg px-4 py-4 flex flex-col gap-3"
      style={{ border: "1px solid var(--color-rule)", backgroundColor: "var(--color-bg-card)" }}>
      <div>
        <p className="font-cinzel text-sm font-semibold" style={{ color: "var(--color-cream)" }}>{card.title}</p>
        <p className="font-montserrat text-xs mt-1" style={{ color: "var(--color-muted)" }}>{card.copy}</p>
      </div>
      {infoOnly ? null : done ? (
        <p className="font-montserrat text-xs" style={{ color: MAG_ACCENT }}>
          {state === "already"
            ? COPY.requestAlreadyOpen
            : COPY.requestSent}
        </p>
      ) : !signedIn ? (
        <Link href="/auth/sign-in?next=/shop%23mag" prefetch className="font-montserrat text-xs font-semibold" style={{ color: MAG_ACCENT }}>
          {COPY.signInTo} {card.cta.toLowerCase()} →
        </Link>
      ) : useOffers ? (
        <div className="flex flex-col gap-2">
          <div className="flex flex-wrap items-center gap-2">
            <select className="flex-1 min-w-[8.5rem] rounded px-2 py-2 font-montserrat text-xs bg-transparent" style={inputStyle}
              aria-label={COPY.offerBar} value={choice}
              onChange={e => { setChoice(e.target.value); setDurationKey(""); }}>
              <option value="">{COPY.offerBar}</option>
              {offers.map(p => <option key={p.productId} value={p.productId}>{p.shortLabel}</option>)}
              {menu.quoteSizes.length > 0 && (
                <optgroup label={COPY.offerQuoteGroup}>
                  {menu.quoteSizes.map(q => <option key={q.grams} value={`${QUOTE_PREFIX}${q.grams}`}>{q.label}</option>)}
                </optgroup>
              )}
            </select>
            <select className="rounded px-2 py-2 font-montserrat text-xs bg-transparent disabled:opacity-50" style={inputStyle}
              aria-label={COPY.offerDuration} value={durationKey} disabled={!product && !quoteSize}
              onChange={e => setDurationKey(e.target.value)}>
              <option value="">{COPY.offerDuration}</option>
              {product?.durations.map(d => <option key={d.offerId} value={d.offerId}>{durationLabel(d.durationValue, d.durationUnit)}</option>)}
              {quoteSize && menu.quoteDurations.map(d => {
                const k = `${d.durationValue}-${d.durationUnit}`;
                return <option key={k} value={k}>{durationLabel(d.durationValue, d.durationUnit)}</option>;
              })}
            </select>
            {offer && (
              <span className="font-montserrat text-xs font-semibold whitespace-nowrap" style={{ color: MAG_ACCENT }}
                title={`${premiumLabel(offer.premiumPct)} ${COPY.offerPremium}`}>
                Spot +{premiumLabel(offer.premiumPct)}
              </span>
            )}
            {quoteDuration && (
              <span className="font-montserrat text-xs font-semibold whitespace-nowrap" style={{ color: MAG_ACCENT }}>{COPY.offerQuoted}</span>
            )}
            <button disabled={!ready || state === "sending" || (card.requestType === "group_pool" && !groupName.trim())} onClick={send}
              className="font-montserrat text-xs font-semibold px-3 py-2 rounded disabled:opacity-50"
              style={{ backgroundColor: MAG_ACCENT, color: "var(--color-bg)" }}
              title={ready ? undefined : COPY.offerPick}>
              {state === "sending" ? COPY.sending : COPY.offerRequest}
            </button>
          </div>
          {(product || quoteSize) && (
            <div className="flex items-center gap-3" data-testid="mag-offer-detail">
              {product?.imageUrl && (
                // eslint-disable-next-line @next/next/no-img-element
                <img src={product.imageUrl} alt={product.label} width={48} height={48} loading="lazy"
                  className="w-12 h-12 rounded object-cover shrink-0" style={{ border: "1px solid var(--color-rule)" }} />
              )}
              <div className="font-montserrat text-[11px] min-w-0" style={{ color: "var(--color-muted)" }}>
                {product && <p className="break-words">{product.label}</p>}
                {quoteSize && <p>{COPY.offerQuoteNote}</p>}
                {outrightPct != null && (
                  <p>{COPY.offerOutright}: <span style={{ color: "var(--color-offwhite)" }}>Spot +{premiumLabel(outrightPct)}</span></p>
                )}
              </div>
            </div>
          )}
          {card.requestType === "group_pool" && (
            <div className="flex flex-wrap gap-2">
              <input className="flex-1 min-w-[8rem] rounded px-2 py-2 font-montserrat text-xs bg-transparent" style={inputStyle}
                placeholder={COPY.groupNamePlaceholder} maxLength={80} value={groupName} onChange={e => setGroupName(e.target.value)} />
              <input className="flex-[2] min-w-[8rem] rounded px-2 py-2 font-montserrat text-xs bg-transparent" style={inputStyle}
                placeholder={COPY.membersPlaceholder} maxLength={300} value={members} onChange={e => setMembers(e.target.value)} />
            </div>
          )}
        </div>
      ) : needsForm && open ? (
        <div className="flex flex-col gap-2">
          {card.requestType === "group_pool" && (
            <>
              <input className={inputCls} style={inputStyle} placeholder={COPY.groupNamePlaceholder} maxLength={80}
                value={groupName} onChange={e => setGroupName(e.target.value)} />
              <input className={inputCls} style={inputStyle} placeholder={COPY.membersPlaceholder} maxLength={300}
                value={members} onChange={e => setMembers(e.target.value)} />
            </>
          )}
          <input className={inputCls} style={inputStyle} inputMode="numeric" placeholder={COPY.gramsPlaceholder}
            value={grams} onChange={e => setGrams(e.target.value.replace(/[^0-9]/g, ""))} />
          <div className="flex gap-2">
            <button disabled={state === "sending"} onClick={send}
              className="font-montserrat text-xs font-semibold px-3 py-2 rounded disabled:opacity-50"
              style={{ backgroundColor: MAG_ACCENT, color: "var(--color-bg)" }}>
              {state === "sending" ? COPY.sending : COPY.send}
            </button>
            <button onClick={() => setOpen(false)} className="font-montserrat text-xs px-3 py-2" style={{ color: "var(--color-muted)" }}>
              {COPY.cancel}
            </button>
          </div>
        </div>
      ) : (
        <button disabled={state === "sending"} onClick={() => (needsForm ? setOpen(true) : send())}
          className="self-start font-montserrat text-xs font-semibold px-3 py-2 rounded disabled:opacity-50"
          style={{ border: `1px solid ${MAG_ACCENT}`, color: MAG_ACCENT }}>
          {state === "sending" ? COPY.sending : card.cta}
        </button>
      )}
      {error && <p className="font-montserrat text-xs" style={{ color: "#e57373" }}>{error}</p>}
    </div>
  );
}
