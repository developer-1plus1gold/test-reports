export function useMagLine(l: MagListingPublic, spotSgd: number | null) {
  const mergeItems = useCartStore(s => s.mergeItems);
  const key = cartItemKey(l.productId, "mag", 1, l.listingId);
  const inCart = useCartStore(s => s.items.find(i => i.productType === "mag" && i.magListingId === l.listingId)?.count ?? 0);
  const added = useCartStore(s => !!s.recentlyAdded[key]);
  const available = Math.max(0, l.gramsRemaining - inCart);
  const [grams, setGramsRaw] = useState(1);
  const soldOut = l.gramsRemaining <= 0;
 
  const clamp = (n: number) => Math.max(1, Math.min(Number.isFinite(n) ? Math.floor(n) : 1, Math.max(1, available)));
  const setGrams = (n: number) => setGramsRaw(clamp(n));
  const priced = spotSgd && l.tiers.length > 0 ? computeStandardLineTotal(grams, 1, spotSgd, l.tiers) : null;
  const soldPct = l.gramsTotal > 0 ? Math.round(((l.gramsTotal - l.gramsRemaining) / l.gramsTotal) * 100) : 0;
  const atCap = available > 0 && grams >= available;
 
  const add = () => {
    if (!priced) return;
    mergeItems([{
      productType: "mag", productId: l.productId, label: `${l.label} · ${l.serial}`,
      g: 1, isAssay: false, count: grams, priceEach: priced.priceEach, imageUrl: l.imageUrl,
      tiers: l.tiers, magListingId: l.listingId, magSerial: l.serial, magGramsRemaining: l.gramsRemaining,
    }]);
    setGramsRaw(1);
  };
 
  return { inCart, added, available, grams, setGrams, soldOut, priced, soldPct, atCap, add };
}
