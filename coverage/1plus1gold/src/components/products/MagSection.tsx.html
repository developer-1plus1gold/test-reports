"use client";
 
/**
 * MAG on the shop — one item per physical serial listed for per-gram
 * sale. Honours the shop's tile/row toggle (operator feedback #3/#18,
 * 2026-10-01): MagProductTile mirrors GoldProductCard, MagProductRow mirrors
 * StandardProductRow. Each bar also carries its bar type's stock chip (D2,
 * lib/mag/stock.ts). Design: docs/allocated-gold-design.md (v3).
 */
import { useEffect, useMemo } from "react";
import { MAG_ACCENT, MAG_FILED_NAME } from "@/lib/mag/home-copy";
import Link from "next/link";
import { magWindowClosed, type MagListingPublic } from "@/lib/mag/listings";
import {
  groupMagListings, magBuyableGrams, magWindowLabel,
  MAG_PRIVATE_CONTACT_TITLE, MAG_PRIVATE_CONTACT_COPY,
} from "@/lib/mag/sections";
import { aggregateMagStock, magStockChip } from "@/lib/mag/stock";
import { useMagAccess } from "./useMagAccess";
import MagProductRow from "./MagProductRow";
import MagProductTile from "./MagProductTile";
 
/**
 * Listings are fetched by the Shop page (so its Refiner/Type filters can see
 * them) and passed in already filtered. Buying is gated during the gradual
 * roll-out: MAG stays visible to everyone, but a customer who is not enabled
 * sees "Request early access" instead of Add. The server enforces the same
 * gate on order creation.
 */
export default function MagSection({ listings, spotSgd, fmt, flagCode, view = "row" }: {
  listings: MagListingPublic[];
  spotSgd: number | null;
  fmt: (v: number) => string;
  flagCode: string;
  /** The shop's tile/row toggle (shared with Splits + Standard). */
  view?: "tile" | "row";
}) {
  const { access, requesting, requestError, requestAccess } = useMagAccess();
 
  // The section mounts once listings load, so a /shop#mag arrival (home tiles,
  // Our Pricing, llms.txt) finds no anchor at page load and the browser never
  // scrolls. Scroll once it exists.
  const hasListings = listings.length > 0;
  useEffect(() => {
    Eif (!hasListings || typeof window === "undefined" || window.location.hash !== "#mag") return;
    document.getElementById("mag")?.scrollIntoView({ block: "start" });
  }, [hasListings]);
 
  const gated = !!access && !access.canBuy;
  const sections = useMemo(() => groupMagListings(listings), [listings]);
  // Stock is per section: a reserved bar's grams must not make the public
  // pool's chip read healthier than it is (or the reverse).
  const stockBySection = useMemo(
    () => new Map(sections.map(sec => [sec.key, aggregateMagStock(sec.items)])),
    [sections],
  );
  // Headings only add noise when the caller sees one pool.
  const showHeadings = sections.length > 1 || sections.some(sec => sec.access !== "public");
 
  return (
    <section id="mag" className="mb-20 scroll-mt-24">
      <div className="h-px mb-14" style={{ backgroundColor: "var(--color-rule)" }} />
      <p className="font-cinzel text-xs tracking-[0.4em] uppercase mb-3" style={{ color: MAG_ACCENT }}>
        1+1Gold
      </p>
      <h2 className="font-cinzel text-3xl font-bold mb-2" style={{ color: "var(--color-cream)" }}>{MAG_FILED_NAME}</h2>
      <p className="font-cormorant italic text-lg mb-6" style={{ color: "var(--color-muted)" }}>
        Own gold in 1-gram steps as your share of a large, serial-numbered bar — held for you in our vault.
      </p>
      {gated && (
        <div className="rounded-lg px-4 py-3 mb-6 font-montserrat text-sm" style={{ border: `1px solid ${MAG_ACCENT}`, color: "var(--color-offwhite)" }}>
          {access!.requested
            ? "Thanks — your early-access request is in. We'll message you when your account is enabled."
            : "We're opening MAG gradually. Request early access and we'll let you know when your account is enabled."}
          {requestError && <p className="mt-2 text-xs" style={{ color: "#e57373" }}>{requestError}</p>}
        </div>
      )}
      {hasListings ? (
        <div className="flex flex-col gap-10">
          {sections.map(sec => {
            const isPrivate = sec.access !== "public";
            const first = sec.items[0];
            const left = sec.items.reduce((n, l) => n + magBuyableGrams(l), 0);
            const windowLabel = isPrivate ? magWindowLabel(first?.windowEndsAt) : null;
            const closed = isPrivate && !!first && magWindowClosed(first);
            return (
              <div key={sec.key}>
                {showHeadings && (
                  <div className="flex flex-wrap items-baseline justify-between gap-2 mb-4">
                    <h3 className="font-cinzel text-lg font-semibold" style={{ color: "var(--color-cream)" }}>{sec.title}</h3>
                    {isPrivate && (
                      <span className="font-montserrat text-xs" style={{ color: closed ? "#e57373" : MAG_ACCENT }}>
                        {closed ? "Buying window closed" : `${left} g left${windowLabel ? ` · ${windowLabel}` : ""}`}
                      </span>
                    )}
                  </div>
                )}
                <div className={view === "tile" ? "flex flex-wrap justify-center gap-6" : "flex flex-col gap-4"}>
                  {sec.items.map(l => {
                    const stock = stockBySection.get(sec.key)?.get(l.productId);
                    const props = {
                      l, spotSgd, fmt, flagCode, access, requesting, onRequest: requestAccess,
                      stockChip: magStockChip(stock), typeSoldOut: !!stock?.soldOut,
                    };
                    return view === "tile" ? (
                      <div key={l.listingId} className="w-full sm:w-[calc(50%-12px)] lg:w-[calc(33.333%-16px)]">
                        <MagProductTile {...props} />
                      </div>
                    ) : (
                      <MagProductRow key={l.listingId} {...props} />
                    );
                  })}
                </div>
              </div>
            );
          })}
        </div>
      ) : (
        <p className="font-montserrat text-sm" style={{ color: "var(--color-muted)" }}>
          No MAG bars match these filters.
        </p>
      )}
      <MagPrivateContactRow />
    </section>
  );
}

      style={{ border: "1px solid var(--color-rule)", backgroundColor: "var(--color-bg-card)" }}>
      <div>
        <p className="font-cinzel text-sm font-semibold" style={{ color: "var(--color-cream)" }}>{MAG_PRIVATE_CONTACT_TITLE}</p>
        <p className="font-montserrat text-xs mt-1" style={{ color: "var(--color-muted)" }}>{MAG_PRIVATE_CONTACT_COPY}</p>
      </div>
      <Link href="/contact" prefetch className="font-montserrat text-xs font-semibold whitespace-nowrap" style={{ color: MAG_ACCENT }}>
        Contact us →
      </Link>
    </div>
  );
}
