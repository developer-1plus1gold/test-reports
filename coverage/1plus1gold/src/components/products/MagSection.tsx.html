"use client";
 
/**
 * MAG on the shop — one item per physical serial listed for per-gram
 * sale. Honours the shop's tile/row toggle (operator feedback #3/#18,
 * 2026-10-01): MagProductTile mirrors GoldProductCard, MagProductRow mirrors
 * StandardProductRow. Each bar also carries its bar type's stock chip (D2,
 * lib/mag/stock.ts). Design: docs/allocated-gold-design.md (v3).
 */
import { useEffect, useMemo } from "react";
import { MAG_ACCENT } from "@/lib/mag/home-copy";
import { MAG_SHOP_COPY as COPY } from "@/lib/mag/shop-copy";
import { magWindowClosed, type MagListingPublic } from "@/lib/mag/listings";
import {
  groupMagListings, magBuyableGrams, magWindowLabel,
} from "@/lib/mag/sections";
import { fmtPieceGrams, unsoldFraction } from "@/lib/mag/piece-grams";
import { aggregateMagStock, magStockChip } from "@/lib/mag/stock";
import { accessForListing, magListingBuyable } from "@/lib/mag/buyable";
import { useMagAccess } from "./useMagAccess";
import type { MagAccessState } from "./MagProductParts";
import MagProductRow from "./MagProductRow";
import MagProductTile from "./MagProductTile";
import MagTypeCards from "./MagTypeCards";
import { isOpenPoolLive } from "@/lib/mag/pool-requests";
 
/**
 * Listings are fetched by the Shop page (so its Refiner/Type filters can see
 * them) and passed in already filtered. Buying is gated during the gradual
 * roll-out: MAG stays visible to everyone, but a customer who is not enabled
 * sees "Request early access" instead of Add — except on a reserved/group bar
 * they are a member of, which membership alone lets them buy
 * (lib/mag/buyable.ts). The server enforces the same gate on order creation.
 */
interface MagSectionProps {
  listings: MagListingPublic[];
  spotSgd: number | null;
  fmt: (v: number) => string;
  flagCode: string;
  /** The shop's tile/row toggle (shared with Splits + Standard). */
  view?: "tile" | "row";
}
 
export default function MagSection(props: MagSectionProps) {
  const { access, requesting, requestError, requestAccess } = useMagAccess();
 
  // The section is client-rendered, so a /shop#mag arrival (home tiles, Our
  // Pricing, llms.txt) finds no anchor at page load and the browser never
  // scrolls. It now always renders (the type cards show with no bars listed),
  // so scroll once on mount.
  useEffect(() => {
    Eif (typeof window === "undefined" || window.location.hash !== "#mag") return;
    document.getElementById("mag")?.scrollIntoView({ block: "start" });
  }, []);
 
  return (
    <MagSectionView {...props} access={access} requesting={requesting}
      requestError={requestError} onRequest={requestAccess} />
  );
}
 
/**
 * The section itself, given the caller's access. The Shop renders it through
 * MagSection (the signed-in customer's own access); Admin → MAG → Customer
 * view renders it with a chosen customer's access and `preview` on, so the
 * admin sees exactly the customer's shop with every buy action replaced by a
 * read-only note (no cart, no request, no agreement can be sent).
 */
export function MagSectionView({
  listings, spotSgd, fmt, flagCode, view = "row", access, requesting, requestError, onRequest, preview = false,
}: MagSectionProps & {
  access: MagAccessState | null;
  requesting: boolean;
  requestError: string | null;
  onRequest: () => void;
  preview?: boolean;
}) {
  const hasListings = listings.length > 0;
 
  // The roll-out note is for bars the caller cannot buy; a member who sees
  // only their own private bars has nothing gated.
  const gated = !!access && !access.canBuy
    && !(hasListings && listings.every(l => magListingBuyable(false, l)));
  const sections = useMemo(() => groupMagListings(listings), [listings]);
  // Stock is per section: a reserved bar's grams must not make the public
  // pool's chip read healthier than it is (or the reverse).
  const stockBySection = useMemo(
    () => new Map(sections.map(sec => [sec.key, aggregateMagStock(sec.items)])),
    [sections],
  );
  // Headings only add noise when the caller sees one pool.
  const showHeadings = sections.length > 1 || sections.some(sec => sec.access !== "public");
 
  return (
    <section id="mag" className="mb-20 scroll-mt-24">
      <div className="h-px mb-14" style={{ backgroundColor: "var(--color-rule)" }} />
      <p className="font-cinzel text-xs tracking-[0.4em] uppercase mb-3" style={{ color: MAG_ACCENT }}>
        {COPY.eyebrow}
      </p>
      <h2 className="font-cinzel text-3xl font-bold mb-2" style={{ color: "var(--color-cream)" }}>{COPY.title}</h2>
      <p className="font-cormorant italic text-lg mb-6" style={{ color: "var(--color-muted)" }}>
        {COPY.subtitle}
      </p>
      {gated && (
        <div className="rounded-lg px-4 py-3 mb-6 font-montserrat text-sm" style={{ border: `1px solid ${MAG_ACCENT}`, color: "var(--color-offwhite)" }}>
          {access!.requested
            ? COPY.gateRequested
            : COPY.gateOpen}
          {requestError && !preview && <p className="mt-2 text-xs" style={{ color: "#e57373" }}>{requestError}</p>}
        </div>
      )}
      {hasListings ? (
        <div className="flex flex-col gap-10">
          {sections.map(sec => {
            const isPrivate = sec.access !== "public";
            const first = sec.items[0];
            // What the caller can still buy, incl. a piece's unsold fraction when they
            // can buy the whole rest of it (lib/mag/piece-grams.ts).
            const left = sec.items.reduce((n, l) => {
              const b = magBuyableGrams(l);
              return n + b + (b === l.gramsRemaining ? unsoldFraction(l) : 0);
            }, 0);
            const windowLabel = isPrivate ? magWindowLabel(first?.windowEndsAt) : null;
            const closed = isPrivate && !!first && magWindowClosed(first);
            return (
              <div key={sec.key}>
                {showHeadings && (
                  <div className="flex flex-wrap items-baseline justify-between gap-2 mb-4">
                    <h3 className="font-cinzel text-lg font-semibold" style={{ color: "var(--color-cream)" }}>{sec.title}</h3>
                    {isPrivate && (
                      <span className="font-montserrat text-xs" style={{ color: closed ? "#e57373" : MAG_ACCENT }}>
                        {closed ? COPY.windowClosed : `${fmtPieceGrams(left)} g left${windowLabel ? ` · ${windowLabel}` : ""}`}
                      </span>
                    )}
                  </div>
                )}
                <div className={view === "tile" ? "flex flex-wrap justify-center gap-6" : "flex flex-col gap-4"}>
                  {sec.items.map(l => {
                    const stock = stockBySection.get(sec.key)?.get(l.productId);
                    const props = {
                      l, spotSgd, fmt, flagCode, access: accessForListing(access, l), requesting, onRequest, preview,
                      stockChip: magStockChip(stock), typeSoldOut: !!stock?.soldOut,
                    };
                    return view === "tile" ? (
                      <div key={l.listingId} className="w-full sm:w-[calc(50%-12px)] lg:w-[calc(33.333%-16px)]">
                        <MagProductTile {...props} />
                      </div>
                    ) : (
                      <MagProductRow key={l.listingId} {...props} />
                    );
                  })}
                </div>
              </div>
            );
          })}
        </div>
      ) : (
        <p className="font-montserrat text-sm" style={{ color: "var(--color-muted)" }}>
          {COPY.empty}
        </p>
      )}
      {/* The three ways to hold MAG, always shown, each with its request. This
          replaced the old "Reserved & group bars — contact us" row. */}
      {!preview && <MagTypeCards signedIn={!!access?.signedIn} openPoolLive={isOpenPoolLive(listings)} />}
    </section>
  );
}
