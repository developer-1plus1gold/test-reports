// Teardrop: tip pointing down, centered at (0,0)
function teardropPath(r: number): string {
  return `M 0,${r * 1.4} C ${r},${r} ${r},-${r * 0.3} 0,-${r} C -${r},-${r * 0.3} -${r},${r} 0,${r * 1.4} Z`;
}
 
// Petal: lens/leaf shape, centered at (0,0)
function petalPath(w: number, h: number): string {
  const hw = w / 2, hh = h / 2;
  return (
    `M 0,${hh} ` +
    `C ${hw},${hh * 0.5} ${hw},-${hh * 0.6} 0,-${hh} ` +
    `C -${hw},-${hh * 0.6} -${hw},${hh * 0.5} 0,${hh} Z`
  );
}

function cubicPt(t: number, a: number, b: number, c: number, d: number): number {
  const u = 1 - t;
  return u*u*u*a + 3*u*u*t*b + 3*u*t*t*c + t*t*t*d;
}
function trunkPt(t: number): [number, number] {
  return [
    cubicPt(t, TK.x0, TK.x1, TK.x2, TK.x3),
    cubicPt(t, TK.y0, TK.y1, TK.y2, TK.y3),
  ];
}

// Sub-branches emerge from intermediate points along the main branch bezier.
// ax/ay…mtx/mty are the main branch bezier parameters.
function subBranchGeom(
  ax: number, ay: number,
  mc1x: number, mc1y: number,
  mc2x: number, mc2y: number,
  mtx: number, mty: number,
  subIdx: number,
  grams: number,
): SubBranchGeom {
  // Where along the main branch each sub-branch splits off (t ∈ [0,1])
  const splitTs = [0.52, 0.68, 0.82];
  const st = splitTs[subIdx] ?? 0.65;
 
  // Sample the main branch bezier at that t
  const sx = cubicPt(st, ax, mc1x, mc2x, mtx);
  const sy = cubicPt(st, ay, mc1y, mc2y, mty);
 
  const len = 45 + Math.min(grams / GRAMS_PER_HALO, 1) * 28;
  const angles = [-38, 5, 42]; // degrees from vertical
  const rad = ((angles[subIdx] ?? 0) * Math.PI) / 180;
  const dx = Math.sin(rad) * len;
  const dy = -Math.cos(rad) * len;
 
  return {
    sx, sy,
    c1x: sx + dx * 0.30, c1y: sy - 13,
    c2x: sx + dx * 0.75, c2y: sy + dy * 0.55,
    tx: sx + dx, ty: sy + dy,
  };
}
 
// ── Halo / petal / teardrop layout ────────────────────────────────────────────
// Petals + teardrops scaled up ~33% from the original sizes (2026-05-01)
// because mobile users couldn't reliably tap individual elements. With
// MAX_GRAMS reduced to 50g (was 100g, so 2 halos per branch instead of 4),
// we have visual headroom to make each piece bigger without crowding.
// 5 petal angles (degrees clockwise from up) — evenly spaced at 72°
// intervals starting from the top so the petals form a clean flower
// without overlap. Previous layout [-40, 40, 145, 185, 220] crowded
// three petals at the bottom and let neighbours overlap.
const PETAL_ANGLES = [0, 72, 144, 216, 288];

const PETAL_W      = 24;   // was 18
const PETAL_H      = 16;   // was 12
const TD_R         = 3.0;  // was 2.2
 
// 5 teardrop positions within a petal (petal-local coords, before rotation).
// Scaled in proportion to the PETAL_W / PETAL_H bump above.
const TD_POS: [number, number][] = [

];
 
// ── Seeded jitter ─────────────────────────────────────────────────────────────
function sr(seed: number, i = 0): number {
  const x = Math.sin(seed * 9301 + i * 49297 + 233) * 10000;
  return x - Math.floor(x);
}
 
// ── Atomic SVG sub-components ─────────────────────────────────────────────────
 
function TearDrop({ fill, x, y, seed }: { fill: number; x: number; y: number; seed: number }) {
  const j = (sr(seed) - 0.5) * 0.6;
  const clipId = useId();
  const d = teardropPath(TD_R);
  const transform = `translate(${x + j},${y + j})`;
  if (fill <= 0 || fill >= 1) {
    const filled = fill >= 1;
    return (
      <path
        d={d}
        transform={transform}
        fill={filled ? TD_FILLED : TD_EMPTY}
        stroke={filled ? GOLD : "none"}
        strokeWidth={0.4}
        style={{ transition: "fill 0.25s" }}
      />
    );
  }
  // Partly held gram: empty drop, filled from the tip up to `fill` of its
  // height (drop spans y = -r … 1.4r, tip at the bottom), gold outline.
  const top = -TD_R, bottom = TD_R * 1.4;
  const level = bottom - fill * (bottom - top);
  return (
    <g transform={transform}>
      <defs>
        <clipPath id={clipId}>
          <rect x={-TD_R * 1.5} y={level} width={TD_R * 3} height={bottom - level} />
        </clipPath>
      </defs>
      <path d={d} fill={TD_EMPTY} />
      <path d={d} fill={TD_FILLED} clipPath={`url(#${clipId})`} />
      <path d={d} fill="none" stroke={GOLD} strokeWidth={0.4} />
    </g>
  );
}
 
function Petal({
  filledDrops, cx, cy, angleDeg, petalIdx, seed,
}: {
  filledDrops: number; cx: number; cy: number;
  angleDeg: number; petalIdx: number; seed: number;
}) {
  const active = dropFillFraction(filledDrops, 0) > 0;
  return (
    <g transform={`translate(${cx},${cy}) rotate(${angleDeg})`}>
      <path
        d={petalPath(PETAL_W, PETAL_H)}
        fill={active ? PETAL_FILL : PETAL_EMPTY}
        stroke={GOLD}
        strokeWidth={0.4}
        strokeOpacity={0.5}
        style={{ transition: "fill 0.3s" }}
      />
      {TD_POS.map(([dx, dy], i) => (
        <TearDrop key={i} fill={dropFillFraction(filledDrops, i)} x={dx} y={dy} seed={seed + petalIdx * 10 + i} />
      ))}
    </g>
  );
}
 
function HaloGroup({
  grams, cx, cy, seed, highlighted,
}: {
  grams: number; cx: number; cy: number; seed: number; highlighted: boolean;
}) {
  const clamped      = Math.min(grams, GRAMS_PER_HALO);
  const filledPetals = Math.floor(clamped / TEARDROPS_PER_PETAL);
  const partialDrops = clamped % TEARDROPS_PER_PETAL;
 
  return (
    <g>
      {/* Soft glow ring — radius scaled with petal size bump */}
      <circle cx={cx} cy={cy} r={28}
        fill="none" stroke={GOLD}
        strokeWidth={highlighted ? 1.2 : 0.6}
        strokeOpacity={clamped > 0 ? 0.30 : 0.10}
      />
      {clamped > 0 && (
        <circle cx={cx} cy={cy} r={9}
          fill={GOLD}
          fillOpacity={0.05 + (clamped / GRAMS_PER_HALO) * 0.09}
        />
      )}
      {/* 5 petals */}
      {PETAL_ANGLES.map((angleDeg, i) => {
        const rad = (angleDeg * Math.PI) / 180;
        const px  = cx + Math.sin(rad) * PETAL_DIST;
        const py  = cy - Math.cos(rad) * PETAL_DIST;
        const drops = i < filledPetals ? TEARDROPS_PER_PETAL : i === filledPetals ? partialDrops : 0;
        return (
          <Petal key={i} filledDrops={drops} cx={px} cy={py} angleDeg={angleDeg} petalIdx={i} seed={seed + i * 7} />
        );
      })}
    </g>
  );
}
 
// ── Branch group (main + sub-branches + halos + label) ────────────────────────
function BranchGroup({
  member, side, attachX, attachY, seed, highlighted, showLabel,
  onPointerEnter, onPointerLeave, onClick,

  attachX: number; attachY: number;
  seed: number; highlighted: boolean;
  /** Show this branch's label. Driven by parent's hover state OR the
   *  global "Show all labels" toggle. When false, label fades to 0. */
  showLabel: boolean;
  onPointerEnter: () => void;
  onPointerLeave: () => void;
  onClick: () => void;
}) {
  const grams       = Math.min(member.grams, MAX_GRAMS);
  const mainGrams   = Math.min(grams, GRAMS_PER_HALO);
  const excessGrams = Math.max(0, grams - GRAMS_PER_HALO);
  const numSub      = Math.min(Math.ceil(excessGrams / GRAMS_PER_HALO), MAX_SUB_BRANCHES);
 
  const strokeW    = 2.5 + (grams / MAX_GRAMS) * 2.5;
  const branchClr  = highlighted ? BROWN_L : BROWN_D;
 
  const { tx, ty, c1x, c1y, c2x, c2y } = mainBranchGeom(attachX, attachY, side, grams);
 
  // Halo + petal furthest extent ≈ 36px from halo centre. Push the label
  // out past that so text never overlaps a petal — 38px sideways, 42px
  // above. Side-anchored so left-side branches read right-aligned and
  // right-side branches read left-aligned, matching the bonsai's wing.
  const labelX      = tx + (side === -1 ? -38 : side === 1 ? 38 : 0);
  const labelAnchor = side === -1 ? "end" : side === 1 ? "start" : "middle";
  const labelY      = ty - 42;
 
  return (
    <g
      style={{ cursor: "pointer" }}
      onPointerEnter={onPointerEnter}
      onPointerLeave={onPointerLeave}
      onClick={(e) => { e.stopPropagation(); onClick(); }}
      onContextMenu={(e) => { e.preventDefault(); onClick(); }}
    >
      <title>{member.name}: {member.grams.toFixed(1)}g · tap to focus</title>
 
      {/* Main branch path */}
      <path
        d={`M ${attachX},${attachY} C ${c1x},${c1y} ${c2x},${c2y} ${tx},${ty}`}
        fill="none" stroke={branchClr} strokeWidth={strokeW} strokeLinecap="round"
        style={{ transition: "stroke 0.25s" }}
      />

// ── Static trunk ──────────────────────────────────────────────────────────────
// Trunk is rendered from t=TRUNK_T_START to t=1 along the bezier — the
// portion below the lowest branch (t=0..0.10) is cropped so the visible
// trunk sits closer to its leaves and looks less "leggy". Branches stay
// at the exact same screen positions because SLOTS.t values are unchanged.
const TRUNK_T_START = 0.10;
const TRUNK_SAMPLES = 40;
 
function TrunkSVG() {
  // Sample the bezier between TRUNK_T_START and 1.0 and connect with a
  // smooth polyline. 40 samples + strokeLinecap=round reads as the same
  // curve, just truncated at the base.
  let path = "";
  for (let i = 0; i <= TRUNK_SAMPLES; i++) {
    const t = TRUNK_T_START + (1 - TRUNK_T_START) * (i / TRUNK_SAMPLES);
    const [x, y] = trunkPt(t);
    path += (i === 0 ? `M ` : ` L `) + `${x.toFixed(1)},${y.toFixed(1)}`;
  }
  const [baseX, baseY] = trunkPt(TRUNK_T_START);
 
  return (
    <g>
      <path d={path} fill="none" stroke={BROWN_D} strokeWidth={22} strokeLinecap="round" />
      <path d={path} fill="none" stroke={BROWN_L} strokeWidth={7}  strokeLinecap="round" strokeOpacity={0.45} />
      {/* Soil ellipse at the new visible base of the trunk */}
      <ellipse cx={baseX} cy={baseY + 5} rx={24} ry={9} fill={BROWN_D} />
    </g>
  );
}
