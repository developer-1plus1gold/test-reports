/**
 * About page — structured content source.
 *
 * Authoritative copy for the /about surface. Consumed by:
 *   - Web — src/app/about/page.tsx imports ABOUT_DATA and renders.
 *   - iOS — src/app/about/spec.ts reads the same import at server-render
 *     time and emits an SDUI spec served by /api/spec/about.
 *
 * Why TypeScript and not Markdown: the about page has structured
 * shapes (founder cards with photos + LinkedIn, KV regulatory rows,
 * accent-coloured cells, paired CTA buttons) that don't fit a flat
 * markdown body. A typed object keeps both renderers honest about
 * the shape and lets the legal/regulatory rows be edited as data
 * rather than HTML.
 *
 * Photo files live at public/about/<filename>. The Web renderer
 * resolves them with an mtime cache-buster (see about/page.tsx);
 * the iOS SDUI spec emits the absolute URL via async_image.
 */

export type FounderCard = {
  name: string;
  identity: string;
  bio: string;
  /** File under public/about/, served as /about/<filename>. */
  photoFilename: string;
  /** Optional public LinkedIn URL — asymmetric across founders by design. */
  linkedin: string | null;
};
 
export type RegulatoryRow = {
  label: string;
  value: string;
  /**
   * Optional accent for the value cell. "gold" = brand gold colour,
   * "success" = compliance-green. Presentation hint shared by both
   * renderers; falsy = default body colour.
   */
  valueAccent?: "gold" | "success";
};
 
export type AboutSection = {
  heading: string;
  /** Optional italic tagline rendered between heading and paragraphs. */
  italicTagline?: string;
  paragraphs: string[];
  /** Optional inline link rendered below the paragraphs. */
  footerLink?: { label: string; href: string };
};
 
export type ButtonRef = {
  label: string;
  href: string;
};
 
export type PhilosophyInvitation = {
  lead: string;
  linkLabel: string;
  linkHref: string;
};
 
export type AboutData = {
  header: {
    eyebrow: string;
    /** Plain text — renderers stylise the "1Plus1Gold" word themselves. */
    title: string;
    tagline: string;
  };
  founders: {
    sectionHeading: string;
    cards: FounderCard[];
    prose: string[];
    philosophyInvitation: PhilosophyInvitation;
    trustStrip: string;
  };
  sections: AboutSection[];
  regulatory: {
    heading: string;
    rows: RegulatoryRow[];
    disclaimers: string[];
    legalLinks: ButtonRef[];
  };
  ctaFooter: {
    italicQuestion: string;
    primaryButton: ButtonRef;
    secondaryButton: ButtonRef;
  };
  quietPhilosophyLink: PhilosophyInvitation;
};
