/** SHA-256 hex of a string (Web Crypto — available in the Edge runtime). The
 *  cookie stores this hash, never the raw password. */
async function sha256hex(input: string): Promise<string> {
  const data = new TextEncoder().encode(input);
  const buf = await crypto.subtle.digest("SHA-256", data);
  return Array.from(new Uint8Array(buf))
    .map((b) => b.toString(16).padStart(2, "0"))
    .join("");
}
 
function htmlEscape(s: string): string {
  return s
    .replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;").replace(/'/g, "&#39;");
}

/** Force noindex on non-production responses. No-op on prod (nonProd=false), so
 *  wrapping every return point is provably inert in production. */
function withNoindex(res: NextResponse, nonProd: boolean): NextResponse {
  if (nonProd) res.headers.set("X-Robots-Tag", "noindex, nofollow");
  return res;
}
