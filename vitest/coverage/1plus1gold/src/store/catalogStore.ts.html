const CATALOG_TTL_MS = 60_000; // re-fetch after 60 s
 
export const useCatalogStore = create<CatalogState>((set, get) => ({
  products:  [],
  loaded:    false,
  fetchedAt: 0,
 
  fetch: async () => {
    const { loaded, fetchedAt } = get();
    if (loaded && Date.now() - fetchedAt < CATALOG_TTL_MS) return;
    try {
      const res  = await fetch("/api/products");
      const data = await res.json();
      if (Array.isArray(data.products)) {
        set({ products: data.products, loaded: true, fetchedAt: Date.now() });
      }
    } catch { /* keep stale data */ }
  },
 
  shopProducts: () =>
    get().products
      .filter((p) => p.shop_visible)
      .sort((a, b) => a.sort_order - b.sort_order),
 
  /** @deprecated 2026-05-19 — premium_pct is being dropped from products.
   *  Use `usePremiumStore.getMap()` for curve premiums and read tiers
   *  directly for standards. This shim returns an empty map. */
  getPremiumMap: () => new Map<number, number>(),
 
  getProduct: (grams) => get().products.find((p) => p.grams === grams),
 
  getCurveProduct: (grams) =>
    get().products.find((p) => p.grams === grams && p.product_type === "curve"),
 
  accessoryProducts: () =>
    get().products.filter((p) => p.product_type === "accessory" && p.shop_visible),
 
  availableStock: (grams) => {
    const p = get().products.find((q) => q.grams === grams);
    if (!p || p.stock_qty === null) return null;
    return Math.max(0, p.stock_qty);
  },
 
  isSoldOut: (grams) => {
    const p = get().products.find((q) => q.grams === grams);
    if (!p || p.stock_qty === null) return false;
    return p.stock_qty <= 0;
  },
 
  isPresale: (grams) => {
    const p = get().products.find((q) => q.grams === grams);
    return p?.status === "presale";
  },
}));
