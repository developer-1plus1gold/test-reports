/**
 * GET /api/coa/verify?serial=<serial>
 *
 * JSON twin of the `/verify/[serial]` server page — the read-API the iOS SDUI
 * "verify" surface calls. Shares `lib/verify-core` with the web page so the
 * certificate payload (and the now-correct curve premium) can never drift.
 *
 * Public + anti-enumeration: same uniform `{ found:false }` body for invalid /
 * unknown / unassigned / rate-limited, so a scraper can't tell which serials
 * exist (rate-limit uses HTTP 429 so clients can back off without leaking that
 * signal in the body). Owner UUID + the full tier table are never returned —
 * the response is an explicit hand-projected `VerifyApiResponse`.
 *
 * Gift PII (giver name / phone last-4 / message) is personal data: returned
 * ONLY to an authenticated, non-owner viewer. Anonymous viewers get
 * `can_show_gift_cta:true` (the "sign in to view & claim" teaser) but the PII
 * fields come back null — a harvest-at-scale mitigation, in parity with the web
 * `/verify/[serial]` page. Viewer auth via getUserFromRequest (Bearer/cookie;
 * anonymous is fine).
 *
 * Rate-limited per IP (30/min) via the hardened `getClientIp` (anti-XFF-spoof)
 * — same key as the page.
 */

/** "dd MMMM yyyy" — server-side so the SDUI spec needs no client date logic
 *  (mirrors the web page's `fmtDate`). */
function fmtAssignedDate(iso: string): string {
  return new Date(iso).toLocaleDateString("en-SG", {
    day: "2-digit",
    month: "long",
    year: "numeric",
  });
}
 
/**
 * Today's shop + buyback value for the card, SGD. Mirrors VerifyPriceStrip:
 * shop premium = curve row's premium (standard → smallest-min-qty tier);
 * buyback adjustment = products.buyback_premium_pct (null = at-spot).
 * Returns nulls when spot is unavailable so the strip can show "Loading…".
 */
function marketValue(
  cert: VerifiedSerial,
  spot: Awaited<ReturnType<typeof getSpotPayload>> | null,
): {
  spot_sgd: number | null;
  spot_stale: boolean;
  shop_price_sgd: number | null;
  buyback_price_sgd: number | null;
  shop_premium_pct: number;
  buyback_premium_pct_display: number;
  shop_premium_label: string;
  buyback_premium_label: string;
  price_disclosure: string;
} {
  const shopPremiumPct =
    cert.product_type === "standard" && cert.tiers.length > 0
      ? [...cert.tiers].sort((a, b) => a.min_qty - b.min_qty)[0].premium_pct
      : cert.premium_pct;
  const buybackPct = cert.buyback_premium_pct ?? 0;
 
  let spotSgd: number | null = null;
  let stale = false;
  if (spot && !("error" in spot)) {
    spotSgd = spot.currencies.SGD?.gram ?? null;
    stale = !!(spot.xauStale || spot.fxStale);
  }
 
  const g = cert.product_grams;
  // Disclosure mirrors VerifyPriceStrip's 4-way branch (spot-null → stale →
  // curve → standard); the sublines mirror its `.toFixed(1)` formatting.
  const price_disclosure =
    spotSgd == null
      ? "Live price unavailable — please retry shortly."
      : stale
        ? "Live spot temporarily delayed. Final prices confirm at checkout / submission."
        : cert.product_type === "curve"
          ? "Indicative. Your shop price may be lower based on your position on the Premium Curve. Final prices confirm at checkout / sell-to-us submission."
          : "Indicative. Final prices confirm at checkout / sell-to-us submission.";
 
  return {
    spot_sgd: spotSgd,
    spot_stale: stale,
    shop_price_sgd: spotSgd != null ? spotSgd * g * (1 + shopPremiumPct / 100) : null,
    buyback_price_sgd: spotSgd != null ? spotSgd * g * (1 + buybackPct / 100) : null,
    shop_premium_pct: shopPremiumPct,
    buyback_premium_pct_display: buybackPct,
    shop_premium_label: `spot +${shopPremiumPct.toFixed(1)}%`,
    buyback_premium_label:
      buybackPct === 0 ? "at spot" : `spot ${buybackPct >= 0 ? "+" : ""}${buybackPct.toFixed(1)}%`,
    price_disclosure,
  };
}
