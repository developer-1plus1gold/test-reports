/**
 * pricing_reviewer — the daily pricing review (Mukesh 2026-11-14, option 1).
 *
 * GET /api/cron/pricing-review  (Bearer CRON_SECRET) — 08:00 UTC = 16:00 SGT,
 * 45 minutes after the afternoon competitor fetch (15:15 SGT).
 *
 * Gathers every buyable price (curve + standard bars on the shop, MAG open-pool
 * bar types, active MAG pool offers) with our cost and the cheapest comparable
 * competitor, builds the table and the flags in code (lib/pricing/daily-review),
 * asks the reviewer agent for the words, and files ONE Control Tower decision
 * for the Singapore day:
 *   approve — something to decide (Approve = Claude applies the changes marked
 *             "act" in the next session; the payload is the brief);
 *   notify  — all clear.
 * The previous day's review, if still undecided, is closed as superseded. A
 * second run the same day refreshes that day's decision instead of adding one.
 * A failed model call still files the review, with the computed headline.
 *
 * PROD only (DEV prices are deliberately stale).
 */

function dayUuid(date: string): string {
  const h = createHash("sha256").update(`pricing_review:${date}`).digest("hex");
  return `${h.slice(0, 8)}-${h.slice(8, 12)}-${h.slice(12, 16)}-${h.slice(16, 20)}-${h.slice(20, 32)}`;
}

/** Open flags of a stored review payload (empty when the payload is not a review). */
function decidedOpenFlags(payload: unknown): ReviewFlag[] {
  const flags = (payload as { review?: { flags?: unknown } } | null)?.review?.flags;
  return Array.isArray(flags) ? (flags as ReviewFlag[]).filter(f => f && !f.accepted) : [];
}
 
/** What a decision was taken on: each open flag and its target, order-free. */
function openFlagSignature(flags: Pick<ReviewFlag, "id" | "targetPct">[]): string {
  return flags.map(f => `${f.id}@${f.targetPct ?? ""}`).sort().join("|");
}
