/**
 * GET /api/cron/linkedin-token-monitor
 *
 * Daily pre-expiry watch on the LinkedIn OAuth credentials.
 *
 * On 2026-08-21 the marketing drumbeat failed for a whole morning with an
 * opaque "video upload/processing failed" message whose real cause was an
 * EXPIRED ACCESS TOKEN. Rotation is deliberately MANUAL (operator decision,
 * 2026-08-21) — so without this cron, not expiring silently depends on
 * somebody remembering a calendar entry. This warns while the token is still
 * valid. It does NOT auto-refresh; that decision stands.
 *
 * How it knows the deadline: LinkedIn's token-introspection endpoint returns a
 * real `expires_at`, so we can warn at 7/3/1 days rather than only noticing
 * after the fact. If introspection itself is unreachable we fall back to an
 * authenticated probe, which can only distinguish alive-vs-dead.
 *
 * ⚠️ Two traps encoded here, both verified live 2026-08-23:
 *  - A HEALTHY org-scoped token returns **403** on /v2/me, not 200 — our
 *    scopes are organization-only. Only 401 means dead.
 *  - Introspection returns **HTTP 400** for an unusable token. That is a
 *    definitive answer, not a transient failure.
 *
 * Alerts are deduped on (token, band, expiry) so a 7-day warning fires once
 * per band per credential — and rotating the token changes the expiry, which
 * retires every old key automatically with no manual reset.
 */
 
import { NextRequest, NextResponse } from "next/server";
import { createClient } from "@supabase/supabase-js";
import { verifyCronAuth } from "@/lib/cron-auth";
import { isProd } from "@/lib/env";
import { recordCronRun } from "@/lib/cron-runs";
import { sendAdminAlert } from "@/lib/twilio";
import { getAdminPhonesCsv } from "@/lib/admin-team";
import { introspectLinkedInToken, probeLinkedInTokenAlive } from "@/lib/marketing/integrations/linkedin";
import {
  resolveTokenHealth, alertKey, severityForBand, alertSeverityForBand, buildTokenAlertMessage,
  type TokenHealth,
} from "@/lib/marketing/linkedin-token-health";
 
const supabaseAdmin = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_KEY!,
);
 
const AGENT = "linkedin_token_monitor";
 
type TokenKind = "access" | "refresh";
 
/** Introspect one credential, falling back to a liveness probe. */
async function checkToken(kind: TokenKind, token: string, nowMs: number): Promise<TokenHealth> {
  try {
    const info = await introspectLinkedInToken(token);
    return resolveTokenHealth(
      { active: info.active === true && info.status !== "expired", expiresAtSec: info.expires_at ?? null },
      nowMs,
    );
  } catch (err) {
    // Introspection unreachable — degrade to alive/dead, never to silence.
    console.error(`[linkedin-token-monitor] introspect ${kind} failed, falling back to probe:`, (err as Error).message);
    const alive = await probeLinkedInTokenAlive(token);
    return resolveTokenHealth({ active: alive, expiresAtSec: null }, nowMs);
  }
}
 
export async function POST(req: NextRequest) {
  const denied = verifyCronAuth(req);
  if (denied) return denied;
 
  // PROD-only: the DEV Vercel project also registers vercel.json crons, and a
  // DEV run would ping Mukesh about the same shared LinkedIn credential twice.
  Eif (!isProd()) return NextResponse.json({ ok: true, skipped: "non-prod" });
 
  const accessToken  = process.env.LINKEDIN_ACCESS_TOKEN;
  const refreshToken = process.env.LINKEDIN_REFRESH_TOKEN;
  if (!accessToken) {
    await recordCronRun(supabaseAdmin, "linkedin-token-monitor", "ok", "LINKEDIN_ACCESS_TOKEN not set");
    return NextResponse.json({ ok: true, skipped: "no_access_token" });
  }
 
  const nowMs = Date.now();
  const checks: Array<{ kind: TokenKind; health: TokenHealth }> = [
    { kind: "access", health: await checkToken("access", accessToken, nowMs) },
  ];
  // The refresh token is what makes rotation possible at all — if it dies we
  // lose the cheap path and need a full re-auth, so it is worth the same watch.
  if (refreshToken) {
    checks.push({ kind: "refresh", health: await checkToken("refresh", refreshToken, nowMs) });
  }
 
  const alerted: string[] = [];
  for (const { kind, health } of checks) {
    if (health.band === "ok") continue;
 
    const key = alertKey(kind, health);
 
    // Dedup: one alert per (token, band, expiry). Rotating the token changes
    // the expiry, which retires every previous key with no manual reset.
    const { data: existing } = await supabaseAdmin
      .from("agent_decisions")
      .select("id")
      .eq("agent_name", AGENT)
      .eq("action_payload->>alert_key", key)
      .limit(1);
    if (existing && existing.length > 0) continue;
 
    const severity = severityForBand(health.band);
    const message  = buildTokenAlertMessage(kind, health);
 
    const { error: dErr } = await supabaseAdmin.from("agent_decisions").insert({
      agent_name: AGENT,
      severity,
      status: "proposed",
      summary: message,
      action_payload: {
        alert_key:      key,
        token_kind:     kind,
        band:           health.band,
        expires_at:     health.expiresAtIso,
        days_remaining: health.daysRemaining,
      },
      customer_id: null,
    });
    if (dErr) console.error("[linkedin-token-monitor] decision insert failed:", dErr.message);
 
    // Best-effort — a WhatsApp failure must not 500 the cron.
    const adminPhone = getAdminPhonesCsv();
    if (adminPhone) {
      await sendAdminAlert(
        adminPhone,
        alertSeverityForBand(health.band),
        health.band === "expired" ? "LinkedIn token EXPIRED" : "LinkedIn token expiring",
        message,
      ).catch((e) => console.error("[linkedin-token-monitor] admin alert failed:", e));
    }
    alerted.push(key);
  }
 
  const summary = checks
    .map((c) => `${c.kind}=${c.health.band}${c.health.daysRemaining === null ? "" : `(${c.health.daysRemaining}d)`}`)
    .join(" ");
  await recordCronRun(supabaseAdmin, "linkedin-token-monitor", "ok", `${summary}; alerts=${alerted.length}`);
 
  return NextResponse.json({
    ok: true,
    checks: checks.map((c) => ({ token: c.kind, ...c.health })),
    alerted,
  });
}
