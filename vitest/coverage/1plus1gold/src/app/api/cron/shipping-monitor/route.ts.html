/**
 * GET /api/cron/shipping-monitor
 *
 * Phase C.2.2 — shipping_monitor automation agent. Runs every 10 minutes.
 *
 * Four conditions, all surfaced under the same agent (Logistics team — same
 * operator):
 *
 *   1. shipper_status='failed', not cancelled/refunded        → approve
 *   2. lalamove, shipper_order_id IS NULL, paid, past cutoff   → approve
 *   3. lalamove, stalled in assigning_driver past the          → ESCALATE
 *      carrier's configured threshold
 *   4. lalamove, still stalled AND previously rebooked         → ESCALATE
 *
 * The timings for 2-4 are OPERATOR SETTINGS, not constants: the quiet window
 * runs from `shipper_schedules.next_day_cutoff` to
 * `shipper_settings.alert_resume_time`, and the stall threshold is
 * `shipper_settings.stall_alert_minutes`. All three are edited in
 * Admin → Shippers, and each falls back to a shipped default when NULL.
 *
 * Self-healing close handles all four: when the condition stops matching, the
 * decision flips to executed automatically.
 *
 * ▶ 3 AND 4 ARE NEW (2026-09-14) AND THEY ARE THE POINT. Order `07a9dfca`
 * (S$2,103.02) sat in `assigning_driver` for two hours, Lalamove auto-cancelled
 * it, and it was rebooked seventeen seconds later — restarting the same wait.
 * This monitor raised NOTHING, because it watched the two conditions that did
 * not apply. Meanwhile it HAD fired *"Lalamove unbooked 35min"* at 21:10 the
 * previous evening for an order placed after the dispatch cutoff that was never
 * going out that night — so it was loud where nothing was wrong and silent where
 * something was. Both halves are fixed here.
 *
 * ▶ WHY 1 COULD NOT HAVE CAUGHT IT. Condition 1 reads the LIVE column. That
 * order was `failed` for seventeen seconds against a ten-minute poll. A poll
 * against a transient state can only catch failures the operator has not already
 * reacted to — the opposite of useful. Condition 4 reads the permanent event
 * array instead, which is why it sees what condition 1 structurally cannot.
 */

/**
 * The SGT wall-clock time an order was placed — more useful in a summary than a
 * big minute count ("placed 13 Sep, 20:35 SGT" beats "unbooked 715min").
 *
 * ▶ THE MONTH COMES FROM A FIXED TABLE, NOT `Intl`, and that is a house rule
 * this repo already paid for (2026-08-23): Node's ICU renders en-GB September as
 * "Sept" where browsers render "Sep", so an Intl month label differs between the
 * test runner and the screen. Only the numeric parts go through Intl, where
 * there is nothing to spell.
 */
const MONTHS = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
 
function sgtTime(iso: string): string {
  const parts = new Intl.DateTimeFormat("en-GB", {
    timeZone: "Asia/Singapore", day: "2-digit", month: "numeric", hour: "2-digit", minute: "2-digit", hour12: false,
  }).formatToParts(new Date(iso));
  const get = (t: string) => parts.find((p) => p.type === t)?.value ?? "";
  return `${get("day")} ${MONTHS[Number(get("month")) - 1]}, ${get("hour")}:${get("minute")}`;
}
