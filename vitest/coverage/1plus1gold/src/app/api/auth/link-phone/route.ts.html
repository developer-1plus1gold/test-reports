/**
 * Mask the middle of a phone number for display in WhatsApp / admin
 * messages. Keeps the country code + last 4 visible. e.g. +6591234567
 * → "+65 ••••3456" — enough for the recipient to recognise it but not
 * leak the whole number to a casual observer.
 */
function maskPhone(p: string): string {
  if (!p) return "—";
  const last4 = p.slice(-4);
  const cc    = p.startsWith("+") ? p.slice(0, p.length - 4 > 4 ? 3 : 2) : "";
  return `${cc} ••••${last4}`.trim();
}
