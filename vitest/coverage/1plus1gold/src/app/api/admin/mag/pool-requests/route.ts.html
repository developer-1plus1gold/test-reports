/**
 * Admin — MAG type requests (lib/mag/pool-requests.ts).
 *
 * GET   → { requests } — open first, then the 50 most recent closed
 * PATCH → { id: uuid, status: 'open'|'done'|'declined' }
 *
 * Setting up the bar itself stays in the access-pool panel (list a bar, set
 * reserved/group, add members). This list is the inbox that tells the admin
 * what to set up.
 */
import { NextRequest, NextResponse } from "next/server";
import { createClient } from "@supabase/supabase-js";
import { checkAdminAuth } from "@/lib/admin-auth";
import { resolveCustomerNames } from "@/lib/admin/customer-names";
import { auditAdminAction } from "@/lib/admin/audit";
 
export const dynamic = "force-dynamic";
 
const supabaseAdmin = createClient(process.env.NEXT_PUBLIC_SUPABASE_URL!, process.env.SUPABASE_SERVICE_KEY!);
const UUID_RE = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;
const STATUSES = ["open", "done", "declined"] as const;
const COLS = "id, customer_id, request_type, grams, group_name, members, note, status, created_at, resolved_at";
 
interface Row {
  id: string; customer_id: string; request_type: string; grams: number | null; group_name: string | null;
  members: string | null; note: string | null; status: string; created_at: string; resolved_at: string | null;
}
 
export async function GET(req: NextRequest) {
  const auth = await checkAdminAuth(req);
  if (!auth.ok) return NextResponse.json({ error: auth.error }, { status: auth.status });
 
  const [open, closed] = await Promise.all([
    supabaseAdmin.from("mag_pool_requests").select(COLS).eq("status", "open")
      .order("created_at", { ascending: true }).limit(500),
    supabaseAdmin.from("mag_pool_requests").select(COLS).neq("status", "open")
      .order("resolved_at", { ascending: false }).limit(50),
  ]);
  if (open.error || closed.error) {
    console.error("[admin/mag/pool-requests] read failed:", open.error?.message ?? closed.error?.message);
    return NextResponse.json({ error: "Could not load requests" }, { status: 500 });
  }
  const rows = [...(open.data ?? []), ...(closed.data ?? [])] as Row[];
  const names = await resolveCustomerNames(supabaseAdmin, rows.map(r => r.customer_id));
  return NextResponse.json({
    requests: rows.map(r => ({
      id: r.id, customerId: r.customer_id,
      label: names.get(r.customer_id) ?? `Customer ${r.customer_id.slice(0, 8)}`,
      requestType: r.request_type, grams: r.grams, groupName: r.group_name, members: r.members,
      note: r.note, status: r.status, createdAt: r.created_at, resolvedAt: r.resolved_at,
    })),
  }, { headers: { "Cache-Control": "private, no-store" } });
}
 
export async function PATCH(req: NextRequest) {
  const auth = await checkAdminAuth(req);
  if (!auth.ok) return NextResponse.json({ error: auth.error }, { status: auth.status });
 
  let body: { id?: unknown; status?: unknown };
  try { body = await req.json(); } catch { return NextResponse.json({ error: "Invalid JSON" }, { status: 400 }); }
  if (typeof body.id !== "string" || !UUID_RE.test(body.id)
    || !STATUSES.includes(body.status as (typeof STATUSES)[number])) {
    return NextResponse.json({ error: "id (uuid) and status (open|done|declined) are required" }, { status: 400 });
  }
  const status = body.status as (typeof STATUSES)[number];
  const { data, error } = await supabaseAdmin.from("mag_pool_requests")
    .update({ status, resolved_at: status === "open" ? null : new Date().toISOString() })
    .eq("id", body.id).select("id");
  if (error) {
    // 23505: reopening while the customer already has another open request of this type.
    Eif (error.code === "23505") return NextResponse.json({ error: "This customer already has an open request of this type" }, { status: 409 });
    console.error("[admin/mag/pool-requests] update failed:", error.message);
    return NextResponse.json({ error: "Operation failed" }, { status: 500 });
  }
  if (!data?.length) return NextResponse.json({ error: "Request not found" }, { status: 404 });
  auditAdminAction(supabaseAdmin, req, { action: `pool_request_${status}`, resourceType: "mag_pool_request", resourceId: body.id });
  return NextResponse.json({ ok: true, id: body.id, status });
}
