/**
 * Members of a reserved bar or group pool (docs/mag-reserved-pools-design.md D5:
 * the customer asks, the admin sets it up here). Admin-only.
 *
 * GET    ?q=<name or phone>  → customer search results to add (when q is given)
 * GET                        → members with name, gram cap, grams bought and
 *                               whether they accepted the delivery recipient;
 *                               plus the bar's recipientId
 * POST   {customer_id, max_grams?}  → add a member
 * PATCH  {customer_id, max_grams}   → change a member's gram cap (null clears)
 * DELETE ?customer_id=<uuid>        → remove a member
 *
 * A reserved bar holds exactly one member. Removing a member does not touch
 * grams they already bought; it only stops new purchases.
 */
import { NextRequest, NextResponse } from "next/server";
import { createClient } from "@supabase/supabase-js";
import { checkAdminAuth } from "@/lib/admin-auth";
import { resolveCustomerNames, searchCustomersByNameOrPhone } from "@/lib/admin/customer-names";
import { isMagAccessType } from "@/lib/mag/listings";
import { customerSearchTerm, memberGramsBought, memberLimitFor, parseMemberCap } from "@/lib/mag/pool-admin";
 
const supabaseAdmin = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_KEY!,
);
 
const UUID_RE = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;
const NO_STORE = { "Cache-Control": "private, no-store" };
const json = (body: unknown, status = 200) => NextResponse.json(body, { status, headers: NO_STORE });
 
type Ctx = { params: Promise<{ id: string }> };
 
async function listingId(ctx: Ctx): Promise<string | null> {
  const { id } = await ctx.params;
  return UUID_RE.test(id) ? id : null;
}
 
async function readBody(req: NextRequest): Promise<Record<string, unknown> | null> {
  try { return (await req.json()) as Record<string, unknown>; } catch { return null; }
}
 
export async function GET(req: NextRequest, ctx: Ctx) {
  const auth = await checkAdminAuth(req);
  if (!auth.ok) return json({ error: auth.error }, auth.status);
  const id = await listingId(ctx);
  if (!id) return json({ error: "Invalid id" }, 400);
 
  const q = req.nextUrl.searchParams.get("q");
  if (q !== null) {
    const term = customerSearchTerm(q);
    if (!term) return json({ results: [] });
    try {
      return json({ results: await searchCustomersByNameOrPhone(supabaseAdmin, term) });
    } catch {
      return json({ error: "Search failed" }, 500);
    }
  }
 
  const [members, allocs, listing] = await Promise.all([
    supabaseAdmin.from("mag_listing_members").select("customer_id, max_grams, added_at, recipient_terms_accepted_at").eq("listing_id", id).limit(1000),
    supabaseAdmin.from("mag_allocations").select("customer_id, grams, status, hold_expires_at").eq("listing_id", id).limit(5000),
    supabaseAdmin.from("mag_listings").select("recipient_customer_id").eq("id", id).maybeSingle(),
  ]);
  if (members.error || allocs.error || listing.error) return json({ error: "Operation failed" }, 500);
 
  const rows = (members.data ?? []) as { customer_id: string; max_grams: number | null; added_at: string; recipient_terms_accepted_at: string | null }[];
  const bought = memberGramsBought((allocs.data ?? []) as never[], Date.now());
  const names = await resolveCustomerNames(supabaseAdmin, rows.map((m) => m.customer_id));
  return json({
    recipientId: (listing.data as { recipient_customer_id: string | null } | null)?.recipient_customer_id ?? null,
    members: rows.map((m) => ({
      customerId: m.customer_id,
      name: names.get(m.customer_id) ?? null,
      maxGrams: m.max_grams,
      gramsBought: bought.get(m.customer_id) ?? 0,
      addedAt: m.added_at,
      recipientTermsAcceptedAt: m.recipient_terms_accepted_at,
    })),
  });
}
 
export async function POST(req: NextRequest, ctx: Ctx) {
  const auth = await checkAdminAuth(req);
  if (!auth.ok) return json({ error: auth.error }, auth.status);
  const id = await listingId(ctx);
  if (!id) return json({ error: "Invalid id" }, 400);
  const body = await readBody(req);
  if (!body) return json({ error: "Invalid JSON" }, 400);
 
  const customerId = String(body.customer_id ?? "");
  if (!UUID_RE.test(customerId)) return json({ error: "Invalid customer" }, 400);
  const cap = parseMemberCap(body.max_grams);
  if (!cap.ok) return json({ error: cap.error }, 400);
 
  const { data: listing, error: lErr } = await supabaseAdmin
    .from("mag_listings").select("access_type").eq("id", id).maybeSingle();
  if (lErr) return json({ error: "Operation failed" }, 500);
  if (!listing) return json({ error: "Listing not found" }, 404);
  const access = (listing as { access_type: string }).access_type;
  if (!isMagAccessType(access) || access === "public") {
    return json({ error: "Set the bar to reserved or group before adding members" }, 409);
  }
 
  const { data: profile } = await supabaseAdmin.from("profiles").select("id").eq("id", customerId).maybeSingle();
  if (!profile) return json({ error: "Customer not found" }, 404);
 
  const limit = memberLimitFor(access);
  if (limit !== null) {
    const { count, error: cErr } = await supabaseAdmin
      .from("mag_listing_members").select("customer_id", { count: "exact", head: true }).eq("listing_id", id);
    if (cErr) return json({ error: "Operation failed" }, 500);
    if ((count ?? 0) >= limit) return json({ error: "A reserved bar holds one member — remove them first" }, 409);
  }
 
  const { error } = await supabaseAdmin.from("mag_listing_members").insert({
    listing_id: id, customer_id: customerId, max_grams: cap.cap, added_by: "admin",
  });
  if (error) {
    if (error.code === "23505") return json({ error: "Already a member" }, 409);
    return json({ error: "Operation failed" }, 500);
  }
  return json({ ok: true }, 201);
}
 
export async function PATCH(req: NextRequest, ctx: Ctx) {
  const auth = await checkAdminAuth(req);
  if (!auth.ok) return json({ error: auth.error }, auth.status);
  const id = await listingId(ctx);
  if (!id) return json({ error: "Invalid id" }, 400);
  const body = await readBody(req);
  if (!body) return json({ error: "Invalid JSON" }, 400);
 
  const customerId = String(body.customer_id ?? "");
  if (!UUID_RE.test(customerId)) return json({ error: "Invalid customer" }, 400);
  const cap = parseMemberCap(body.max_grams);
  if (!cap.ok) return json({ error: cap.error }, 400);
 
  const { data, error } = await supabaseAdmin
    .from("mag_listing_members").update({ max_grams: cap.cap })
    .eq("listing_id", id).eq("customer_id", customerId).select("customer_id");
  if (error) return json({ error: "Operation failed" }, 500);
  if (!data || data.length === 0) return json({ error: "Not a member" }, 404);
  return json({ ok: true });
}
