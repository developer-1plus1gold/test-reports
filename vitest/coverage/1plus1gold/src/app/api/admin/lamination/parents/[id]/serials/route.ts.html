/**
 * GET /api/admin/lamination/parents/[id]/serials
 *
 * The NFC serials already laminated from a parent Combibar — shown when the
 * operator picks a parent so they can see what has already come off this bar
 * (and catch a mistaken re-lamination or a missing stamp).
 *
 * Provenance can live in two places, so we UNION both to avoid blind spots:
 *   • inventory children  — inventory.parent_inventory_id = <parent id>
 *   • denormalised stamp  — nfc_serials.parent_bar_serial = <parent serial_number>
 *
 * The two normally agree, but a split assigned WITHOUT a lamination inventory
 * row (only the stamp) or a lamination whose stamp missed (only the child row)
 * would each be invisible to a single-source query — exactly the VC290647 gap.
 */
 
import { NextRequest, NextResponse } from "next/server";
import { createClient } from "@supabase/supabase-js";
import { checkAdminAuth } from "@/lib/admin-auth";
 
const supabaseAdmin = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_KEY!,
);
 
export async function GET(
  req: NextRequest,
  { params }: { params: Promise<{ id: string }> },
) {
  const auth = await checkAdminAuth(req);
  if (!auth.ok) return NextResponse.json({ error: auth.error }, { status: auth.status });
 
  const { id } = await params;
 
  // Parent serial_number drives the stamp-based half of the union.
  const { data: parent, error: pErr } = await supabaseAdmin
    .from("inventory")
    .select("serial_number")
    .eq("id", id)
    .maybeSingle();
  if (pErr) {
    console.error("[lamination/parents/serials] parent lookup error:", pErr);
    return NextResponse.json({ error: "Operation failed" }, { status: 500 });
  }
  const parentSerial = (parent as { serial_number?: string | null } | null)?.serial_number ?? null;
 
  // Half 1: inventory children of this parent that carry an NFC serial.
  const { data: kids } = await supabaseAdmin
    .from("inventory")
    .select("nfc_serial")
    .eq("parent_inventory_id", id)
    .not("nfc_serial", "is", null);
  const fromInventory = (kids ?? [])
    .map((k) => (k as { nfc_serial?: string | null }).nfc_serial)
    .filter((s): s is string => !!s);
 
  // Half 2: nfc_serials stamped with this parent's serial.
  let fromStamp: string[] = [];
  if (parentSerial) {
    const { data: stamped } = await supabaseAdmin
      .from("nfc_serials")
      .select("serial")
      .eq("parent_bar_serial", parentSerial);
    fromStamp = (stamped ?? []).map((r) => (r as { serial: string }).serial);
  }
 
  const union = Array.from(new Set([...fromInventory, ...fromStamp]));
  if (union.length === 0) return NextResponse.json({ rows: [] });
 
  // Hydrate status + order linkage for each serial.
  const { data: rows, error: rowsErr } = await supabaseAdmin
    .from("nfc_serials")
    .select("serial, status, order_id, assigned_at")
    .in("serial", union)
    .order("assigned_at", { ascending: true });
  if (rowsErr) {
    console.error("[lamination/parents/serials] rows error:", rowsErr);
    return NextResponse.json({ error: "Operation failed" }, { status: 500 });
  }
 
  return NextResponse.json({ rows: rows ?? [] });
}
