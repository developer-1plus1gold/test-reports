// Settable statuses restricted to the meaningful ones (order-status rewrite,
// audit D3). The legacy pending/processing/ready/shipped/on_hold are NONE of the
// PAID_STATUSES, so setting one silently dropped an order out of revenue/P&L/
// analytics — removed. Shipping progress lives in shipper_status (courier
// webhooks), not here.
const ALL_STATUSES = ["paid", "completed", "cancelled", "refunded"] as const;
type OrderStatus = typeof ALL_STATUSES[number];
 
const STATUS_LABELS: Record<OrderStatus, string> = {
  paid:       "Payment Confirmed",
  completed:  "Completed",
  cancelled:  "Cancelled",
  refunded:   "Refunded",
};
 
 
/**
 * PATCH /api/admin/orders/[id]/status
 * Header: x-admin-secret
 * Body: { newStatus: string, reason: string }
 *
 * 1. Validates the new status value
 * 2. Fetches the current order status + customer phone
 * 3. Updates the order status
 * 4. Logs the change to order_status_logs
 * 5. Sends a WhatsApp notification to the customer
 */
export async function PATCH(
  req: NextRequest,
  { params }: { params: Promise<{ id: string }> }
) {
  try {
    const auth = await checkAdminAuth(req);
    if (!auth.ok) return NextResponse.json({ error: auth.error }, { status: auth.status });
 
    const { id } = await params;
    const body = await req.json() as { newStatus?: string; reason?: string };
    const { newStatus, reason } = body;
 
    if (!newStatus || !ALL_STATUSES.includes(newStatus as OrderStatus)) {
      return NextResponse.json({ error: "Invalid status" }, { status: 400 });

      .from("orders")
      .select("id, status, customer_id, shipper, shipper_order_id")
      .eq("id", id)
      .single();
 
    if (orderErr || !order) {
