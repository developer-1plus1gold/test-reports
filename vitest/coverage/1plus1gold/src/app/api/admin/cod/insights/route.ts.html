/**
 * GET /api/admin/cod/insights — the tiles at the top of Admin → COD / Bank Transfer
 * (lib/admin/cod-insights holds every definition). Admin-only. Returns aggregates — counts,
 * sums, days — the order ids behind the tiles that filter the list, and the debtor names of
 * "Who owes us" (the names the list below already shows). Never a phone or email.
 *
 * Three order reads, merged by id: every COD order (owed now + analytics), every unpaid
 * marketplace payout (owed now), and every paid order since the first trading day (the
 * denominator of the COD share). Each is read in full, past PostgREST's 1,000-row cap.
 */
import { NextRequest, NextResponse } from "next/server";
import { createClient } from "@supabase/supabase-js";
import { checkAdminAuth } from "@/lib/admin-auth";
import { safeError } from "@/lib/safe-error";
import { pageAll } from "@/lib/db/page-all";
import { fetchSalesExclusions } from "@/lib/sales/sales-exclusions";
import { PAID_STATUSES } from "@/lib/order-aggregations";
import { BUSINESS_CONFIG } from "@/lib/business-config";
import { sgtDayStart } from "@/lib/finance/sgt-window";
import { computeCodInsights, isOwed, type CodInsightOrder } from "@/lib/admin/cod-insights";
 
export const dynamic = "force-dynamic";
 
const admin = () => createClient(process.env.NEXT_PUBLIC_SUPABASE_URL!, process.env.SUPABASE_SERVICE_KEY!);
const FAIL = "cod_insights_fetch_failed";
const COLS = "id, customer_id, created_at, paid_at, status, payment_method, payment_status, grand_total_sgd, mag_redeemed_sgd, external_channel_fee_sgd, shipper_actual_fee_sgd, source_channel, external_buyer_label";
 
export async function GET(req: NextRequest) {
  const auth = await checkAdminAuth(req);
  if (!auth.ok) return NextResponse.json({ error: auth.error }, { status: auth.status });
 
  try {
    const db = admin();
    const [exclusions, cod, payouts, paidSinceLaunch, allowed] = await Promise.all([
      fetchSalesExclusions(db),
      pageAll<CodInsightOrder>((f, t) => db.from("orders").select(COLS).eq("payment_method", "cod").order("id").range(f, t), FAIL),
      pageAll<CodInsightOrder>((f, t) => db.from("orders").select(COLS)
        .eq("payment_method", "external").eq("payment_status", "unpaid").order("id").range(f, t), FAIL),
      pageAll<CodInsightOrder>((f, t) => db.from("orders").select(COLS)
        .in("status", [...PAID_STATUSES]).gte("created_at", sgtDayStart(BUSINESS_CONFIG.tradingStarted))
        .order("id").range(f, t), FAIL),
      pageAll<{ id: string }>((f, t) => db.from("profiles").select("id").eq("cod_enabled", true).order("id").range(f, t), FAIL),
    ]);
 
    const byId = new Map<string, CodInsightOrder>();
    for (const o of [...cod, ...payouts, ...paidSinceLaunch]) byId.set(o.id, o);
 
    // Names for "Who owes us": only the customers with a COD order still owed.
    const debtorIds = [...new Set(cod.filter((o) => isOwed(o) && o.customer_id).map((o) => o.customer_id as string))];
    if (debtorIds.length) {
      const { data, error } = await db.from("profiles").select("id, full_name").in("id", debtorIds);

      for (const o of byId.values()) if (o.customer_id && names.has(o.customer_id)) o.customer_name = names.get(o.customer_id) ?? null;
    }
 
    const insights = computeCodInsights([...byId.values()], exclusions, allowed.map((p) => p.id), new Date());
    return NextResponse.json({ insights }, { headers: { "Cache-Control": "private, no-store" } });
  } catch (err) {
    return NextResponse.json(safeError(err, "admin/cod/insights", "Could not load COD insights"), { status: 500 });
  }
}
