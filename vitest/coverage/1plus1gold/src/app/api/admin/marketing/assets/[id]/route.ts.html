/**
 * PATCH /api/admin/marketing/assets/[id]
 *
 * Approve / reject / archive an asset. Approval is the gate for
 * Publisher to use the asset. Status enum:
 *   generated → approved → published_to_channel → archived
 *   generated → failed (if Reviewer block stays unfixed)
 */
 
import { NextRequest, NextResponse } from "next/server";
import { createClient } from "@supabase/supabase-js";
import { checkAdminAuth } from "@/lib/admin-auth";
import { isUuid, sanitizeRuleText } from "@/lib/marketing/sanitize";
import { activateGuardrailFromRejection } from "@/lib/marketing/guardrail-ingest";
import { detachAssetFromChannelPosts } from "@/lib/marketing/channel-posts";
 
const supabaseAdmin = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_KEY!,
);
 
export async function PATCH(req: NextRequest, ctx: { params: Promise<{ id: string }> }) {
  const auth = await checkAdminAuth(req);
  if (!auth.ok) return NextResponse.json({ error: auth.error }, { status: auth.status });
 
  const { id } = await ctx.params;
  if (!isUuid(id)) return NextResponse.json({ error: "invalid_id" }, { status: 400 });
  let body: { action: "approve" | "reject" | "archive"; reason?: string };
  try { body = await req.json(); } catch { return NextResponse.json({ error: "invalid_json" }, { status: 400 }); }
 
  // 'rejected' (NOT 'failed') — operator rejection is distinct from a generation
  // failure; the UI filters/hides rejected separately.
  const statusMap: Record<string, string> = {
    approve: "approved",
    reject:  "rejected",
    archive: "archived",
  };
  const newStatus = statusMap[body.action];
  if (!newStatus) return NextResponse.json({ error: "invalid_action" }, { status: 400 });
 
  // On reject, persist the reason ON the asset too (visible under "Show rejected")
  // — not only as a proposed guardrail.
  const cleanReason = sanitizeRuleText(body.reason);
  const update: Record<string, unknown> = { status: newStatus, updated_at: new Date().toISOString() };
  if (body.action === "reject") update.rejection_reason = cleanReason || null;
 
  const { data, error } = await supabaseAdmin
    .from("marketing_assets")
    .update(update)
    .eq("id", id)
    .select("id, status, content_piece_id")
    .single();
  if (error || !data) return NextResponse.json({ error: "update_failed", detail: error?.message }, { status: 500 });
 
  // Reject-with-reason feeds the learning loop: the reason becomes an ACTIVE
  // brand guardrail — the classifier picks scope + kind and it applies to the
  // next generation immediately, no manual approval. Best-effort.
  let learnedGuardrail: { created: boolean; kind?: string; scope?: string; deduped?: boolean } = { created: false };
  if (body.action === "reject" && cleanReason) {
    learnedGuardrail = await activateGuardrailFromRejection(supabaseAdmin, { reason: cleanReason, source: "asset", sourceAssetId: id });
  }
 
  // A rejected asset has no business in a channel post — strip its id from any
  // per-channel media assignment so it can't publish (the picker already hides
  // it; this removes a previously-assigned id too). Best-effort. Shared with
  // DELETE so the two removal paths cannot drift apart again.
  if ((body.action === "reject" || body.action === "archive") && data.content_piece_id) {
    await detachAssetFromChannelPosts(supabaseAdmin, data.content_piece_id, id);
  }
 
  // Auto-transition: when an asset is approved AND there's a passing
  // (or override_granted) brand review for the piece, advance the
  // piece status from in_production → brand_reviewed. Publisher's
  // final scheduled-time gate enforces the full readiness check.
  if (body.action === "approve" && data.content_piece_id) {
    const { data: review } = await supabaseAdmin
      .from("marketing_brand_reviews")
      .select("status")
      .eq("content_piece_id", data.content_piece_id)
      .in("status", ["passed", "override_granted"])
      .limit(1)
      .maybeSingle();
    if (review) {
      await supabaseAdmin
        .from("marketing_content_pieces")
        .update({ status: "brand_reviewed", updated_at: new Date().toISOString() })
        .eq("id", data.content_piece_id)
        .eq("status", "in_production");
    }
  }
 
  return NextResponse.json({ ok: true, asset: data, learnedGuardrail });
}
