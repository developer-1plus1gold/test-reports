/**
 * POST /api/admin/vault/audits/upload-url
 *
 * Mints a signed upload URL so the BROWSER uploads audit evidence — a PHOTO or
 * a VIDEO — straight into `product-media`, the same shape the CoA-video and
 * serial photo-inbox uploads use.
 *
 * ▶ Why neither kind goes through the audits route body: the platform rejects a
 * request body over ~4.5MB with a plain-text FUNCTION_PAYLOAD_TOO_LARGE BEFORE
 * the handler runs, so the route cannot answer with a usable error, and a phone
 * photo routinely exceeds it (reported 2026-09-01). Downscaling in the browser
 * first was tried and is not enough: the decode can fail on Safari and the
 * original is uploaded anyway. Removing the body from the path is the fix.
 *
 * The audit row does not exist yet while the operator is filling the form, so
 * paths are keyed by CUSTOMER, and the audits POST re-validates whatever key the
 * browser reports (prefix + object existence) before pinning it.
 *
 * Body: { customerId, kind: "photo" | "video", name, type, size }
 *
 * Admin-gated.
 */

const UUID_RE = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;
 
// Photo extensions are enforced by EXTENSION, not the browser's claimed MIME —
// a signed PUT often carries an empty or odd Content-Type. This is a shape
// check only: the audits route proves the bytes are a real image by re-encoding
// them, and deletes the object if that fails.
//
// ▶ HEIC/HEIF is deliberately NOT accepted, unlike the serial photo-inbox.
// That inbox is a private staging area whose files are decoded out-of-band by a
// pipeline that handles HEIC; this photo is re-encoded by sharp, whose prebuilt
// binary registers HEIF input for AVIF only. Accepting a HEIC here would mean
// taking the upload and then rejecting it — worse than declining at the picker.
// It costs nothing in practice: with an image `accept` list iOS converts a
// library photo to JPEG on selection (the reported failure was an IMG_*.jpeg).
const PHOTO_EXT: Record<string, string> = {
  jpg: "jpg", jpeg: "jpg", png: "png", webp: "webp", gif: "gif", avif: "avif",
};
 
function photoExt(name: string): string | null {
  const ext = (name.split(".").pop() ?? "").toLowerCase();
  return PHOTO_EXT[ext] ?? null;
}
