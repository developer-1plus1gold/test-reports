/**
 * Did this field actually change?
 *
 * ▶ A POSTGRES `time` DOES NOT ROUND-TRIP AS A STRING. The column comes back
 * from PostgREST as "08:30:00", while the admin editor sends "08:30" — its
 * control is `<input type="time">`, which is HH:MM. String-comparing the two is
 * never equal, so `alert_resume_time` was stamped as changed on EVERY save and
 * the screen claimed an edit nobody had made.
 *
 * Found 2026-09-14 in the `field_updated_at` of a real operator save, one day
 * after the field was added. It is cosmetic — the stamp only drives a "last
 * changed" read-out — but a field asserting a change that did not happen is the
 * wrong direction for anything audit-shaped.
 *
 * Compared as clock times whenever BOTH sides parse as one, rather than
 * special-casing today's single time column: the next time-typed setting
 * inherits the fix instead of re-introducing the bug. Every other trackable
 * field is a number, boolean or free string, which already round-trip
 * identically — this is the only shape that does not.
 */
function sameTrackedValue(prev: unknown, next: unknown): boolean {
  const a = prev === null || prev === undefined ? "" : String(prev);
  const b = next === null || next === undefined ? "" : String(next);
  if (a === b) return true;
  const ca = parseClockMinutes(a);
  const cb = parseClockMinutes(b);
  return ca !== null && cb !== null && ca === cb;
}
