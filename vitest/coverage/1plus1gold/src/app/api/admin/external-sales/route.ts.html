/**
 * External-platform sales (Shopee → Lazada → Carousell).
 *
 * GET  — list external orders (source_channel != web/ios) for the admin screen.
 * POST — manually record a pre-paid marketplace sale so it flows into our inventory
 *        / fulfilment / books. The buyer paid on the marketplace, so we skip checkout:
 *        create the order under the channel's synthetic house customer, reserve stock,
 *        and mark it paid. Fulfilment (ship → delivered) is driven from the detail route.
 *
 * Reuses the exact paid-order machinery: create_order_atomic RPC + consumeStockForOrderItem
 * (identical to the Stripe/HitPay path in src/lib/hitpay-order.ts).
 */

  createClient(process.env.NEXT_PUBLIC_SUPABASE_URL!, process.env.SUPABASE_SERVICE_KEY!);
 
const bodySchema = z.object({
  channel:         z.enum(["shopee", "lazada", "carousell", "in_person", "tiktok_shop"]),
  externalOrderId: z.string().trim().min(1).max(120),
  buyerLabel:      z.string().trim().max(200).optional().default(""),
  /** Attach the sale to a REAL customer instead of the channel's synthetic
   *  house customer. Intended for in-person sales where you know who bought —
   *  they then get the order in their history, their tree, and a working
   *  buyback/CoA-claim path. Omit for a genuinely anonymous walk-in or a
   *  marketplace buyer with no account, which falls back to the house customer.
   *
   *  Channel revenue attribution is UNAFFECTED either way: the P&L splits key
   *  on orders.source_channel, not on who owns the row. */
  customerId:      z.string().uuid().optional(),
  /** BUSINESS date of the sale (yyyy-mm-dd, SGT), for a sale keyed in after the
   *  fact. Omitted → the RPC stamps now(), which is the behaviour every
   *  automated caller keeps.
   *
   *  Bounded on BOTH sides deliberately. A future date parks revenue in a period
   *  that has not happened; a typo'd year ("2062") parks it somewhere nobody will
   *  ever look. Neither fails loudly on its own, so the boundary is the only
   *  place to catch it — and this is an admin form, not the paid-customer path,
   *  so rejecting here costs nobody their order. */
  orderDate: z.string()
    .regex(/^\d{4}-\d{2}-\d{2}$/, "orderDate must be yyyy-mm-dd")
    .refine((d) => d <= todaySgt(), "orderDate cannot be in the future")
    .refine((d) => d >= "2020-01-01", "orderDate is implausibly old")
    .optional(),
  /** Marketplace commission / transaction fees charged to US. Not buyer-visible. */
  channelFeeSgd:   z.number().nonnegative().max(1_000_000).optional().default(0),
  /** Seller voucher we funded. The buyer SAW this and paid less, so it lands on
   *  their receipt as a discount line — unlike the other two absorbed costs. */
  sellerVoucherSgd: z.number().nonnegative().max(1_000_000).optional().default(0),
  /** Shipping we absorbed (buyer charged $0.00, marketplace bills us). Reduces
   *  our payout; never appears on the buyer's receipt. */
  absorbedShippingSgd: z.number().nonnegative().max(1_000_000).optional().default(0),
  shipTo: z.object({
    full_name:   z.string().trim().max(200).optional().nullable(),
    phone:       z.string().trim().max(40).optional().nullable(),
    line1:       z.string().trim().max(300).optional().nullable(),
    line2:       z.string().trim().max(300).optional().nullable(),
    postal_code: z.string().trim().max(20).optional().nullable(),
  }).default({}),
  items: z.array(z.object({
    productId: z.string().uuid(),
    quantity:  z.number().int().positive().max(9999),
    priceSgd:  z.number().nonnegative().max(1_000_000),  // per-unit GROSS marketplace price
    isPresale: z.boolean().optional(),
  })).min(1).max(200),
}).superRefine((data, ctx) => {
  // For MARKETPLACE channels, the order number + buyer id + ship-to postal code
  // are the keys a customer later uses to self-claim the order into their account
  // (order number = exact anchor, postal = reproducible secondary, buyer id =
  // reviewer signal). Require all three so every marketplace sale stays claimable.
  // `in_person` is exempt — a cash/meetup sale has no marketplace id/username and
  // often no delivery address, and it is not part of the claim funnel.
  if (data.channel === "in_person") return;
  if (!data.buyerLabel?.trim()) {
    ctx.addIssue({ code: z.ZodIssueCode.custom, path: ["buyerLabel"],
      message: "Buyer id is required for marketplace orders (so the buyer can claim it to their account)" });
  }
  if (!data.shipTo?.postal_code?.trim()) {
    ctx.addIssue({ code: z.ZodIssueCode.custom, path: ["shipTo", "postal_code"],
      message: "Delivery postal code is required for marketplace orders (used to verify account claims)" });
  }
});
