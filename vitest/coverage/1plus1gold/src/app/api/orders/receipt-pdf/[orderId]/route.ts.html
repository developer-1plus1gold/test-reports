import { NextRequest, NextResponse } from "next/server";
import { createClient } from "@supabase/supabase-js";
import { generateReceiptPdf } from "@/lib/receipt/generateReceiptPdf";
import type { ReceiptData, ReceiptTheme } from "@/lib/receipt/generateReceiptPdf";
import { verifyPdfToken } from "@/lib/pdf-token";
import type { PdfTokenPayload } from "@/lib/pdf-token";
import { checkAdminAuth } from "@/lib/admin-auth";
import { receiptPrintKey, receiptStorageKey } from "@/lib/receipt/receipt-cache-key";
import { RECEIPT_ORDER_SELECT, RECEIPT_STATUSES, assembleReceiptData, buildReceiptContext } from "@/lib/receipt/load-receipt-data";
 
/**
 * GET /api/orders/receipt-pdf/[orderId]
 *
 * Accepts orderId with or without trailing ".pdf" so the WhatsApp template
 * sample URL (which must end in .pdf for Meta approval) also works.
 *
 * The URL is fetched server-to-server by WhatsApp — it is never visible to
 * the customer. The UUID (128-bit) acts as an implicit access token.
 *
 * Flow:
 *   1. Strip optional .pdf suffix
 *   2. Special-case all-zeros UUID → sample receipt for Meta template approval
 *   3. Check Supabase storage for an already-generated receipt
 *   4. If not found, generate on-the-fly, upload, then stream
 *   5. Returns 404 for non-existent or non-paid orders
 */
 
const supabaseAdmin = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_KEY!
);
 
export async function GET(
  _req: NextRequest,
  { params }: { params: Promise<{ orderId: string }> }
) {
  const { orderId: rawId } = await params;
 
  // Strip optional .pdf suffix (template sample URL uses it to satisfy Meta's extension check)
  const orderId = rawId.replace(/\.pdf$/i, "");
 
  // ── Sample receipt for Meta template approval ─────────────────────────────
  // When Twilio submits the template, Meta downloads the sample media URL.
  // We serve a real PDF so the approval succeeds.
  if (orderId === "00000000-0000-0000-0000-000000000000") {
    const sampleData: ReceiptData = {
      orderId:         "00000000-0000-0000-0000-000000000000",
      createdAt:       new Date().toISOString(),
      totalGrams:      50,
      totalSgd:        4200,
      grandTotalSgd:   4265,
      shippingFeeSgd:  65,
      deliveryAddress: { full_name: "Sample Customer", line1: "1 Gold Street", postal_code: "018989" },
      serviceType:     "MOTORCYCLE",
      customerName:    "Sample Customer",
      customerPhone:   null,
      items: [{ label: "50g Valcambi CombiBar", grams: 50, quantity: 1, price_sgd: 4200, gstTreatment: "ipm_exempt" as const }],
    };
    const sampleBuf = await generateReceiptPdf(sampleData);
    return new NextResponse(new Uint8Array(sampleBuf), {
      headers: {
        "Content-Type":        "application/pdf",
        "Content-Disposition": 'attachment; filename="sample-receipt.pdf"',
        "Cache-Control":       "public, max-age=86400",
      },
    });
  }
 
  // Basic format guard — must look like a UUID
  if (!/^[0-9a-f-]{32,36}$/i.test(orderId)) {
    return new NextResponse("Not found", { status: 404 });
  }
 
  // ── Auth: admin cookie OR the customer's HMAC token ───────────────────────
  // The customer-facing path (WhatsApp/email) signs a per-order HMAC token. An
  // authenticated admin (e.g. the dispatch "Print Receipt" button) may fetch
  // ANY order's receipt directly — the admin session cookie is the gate, so no
  // token is needed and the ownership check below is skipped for admins.
  const adminAuth = await checkAdminAuth(_req);
  const isAdmin = adminAuth.ok;
  let tokenUserId: string | null = null;
  if (!isAdmin) {
    const rawToken = _req.nextUrl.searchParams.get("t");
    if (!rawToken) return new NextResponse("Unauthorized", { status: 401 });
    const tokenPayload: PdfTokenPayload | null = verifyPdfToken(rawToken);
    if (!tokenPayload) return new NextResponse("Unauthorized", { status: 401 });
    if (tokenPayload.resource !== "receipt" || tokenPayload.resourceId !== orderId) {
      return new NextResponse("Forbidden", { status: 403 });
    }
    tokenUserId = tokenPayload.userId;
  }
  // ── Fetch order to verify ownership and status ───────────────────────────
  const { data: order } = await supabaseAdmin
    .from("orders")
    .select(RECEIPT_ORDER_SELECT)
    .eq("id", orderId)
    .single();
 
  // Only return receipts for paid orders
  if (!order || !RECEIPT_STATUSES.has(order.status)) {
    return new NextResponse("Not found", { status: 404 });
  }
 
  // Verify token userId matches order owner (skipped for admins).
  if (!isAdmin && order.customer_id !== tokenUserId) {
    return new NextResponse("Forbidden", { status: 403 });
  }
 
  // ── Try to serve from storage first (already generated) ──────────────────
  // Keyed on the order's CONTENT, not just its id: a receipt keyed on id alone
  // froze at whatever the order looked like when first printed, so correcting an
  // order left the customer document permanently stale and unreachable by any
  // cache-bust. See receiptStorageKey.
  const ctx = await buildReceiptContext(supabaseAdmin, order, orderId);
  // ?print=1 (admin only — the Dispatch "Print receipt" button): the ink-light
  // PRINT theme for the copy that goes in the parcel. Stored beside the screen
  // copy under its own key, so neither ever serves the other's look.
  const printTheme: ReceiptTheme = isAdmin && _req.nextUrl.searchParams.get("print") === "1" ? "print" : "screen";
  const fileName = receiptPrintKey(receiptStorageKey(orderId, ctx.cacheInput), printTheme);
  const { data: existingFile } = await supabaseAdmin.storage
    .from("receipts")
    .download(fileName);
 
  if (existingFile) {
    const buf = await existingFile.arrayBuffer();
    return new NextResponse(buf, {
      headers: {
        "Content-Type":        "application/pdf",
        "Content-Disposition": `inline; filename="receipt-${orderId.slice(0, 8).toUpperCase()}.pdf"`,
        "Cache-Control":       "private, max-age=3600",
      },
    });
  }
 
  const { data: profile } = await supabaseAdmin
    .from("profiles")
    .select("phone, full_name")
    .eq("id", order.customer_id)
    .single();
 
  const receiptData: ReceiptData = assembleReceiptData(order, ctx, profile);
 
  let pdfBuffer: Buffer;
  try {
    pdfBuffer = await generateReceiptPdf(receiptData, printTheme);
  } catch (err) {
    console.error("[receipt-pdf] Generation failed:", err);
    return new NextResponse("PDF generation failed", { status: 500 });
  }
 
  // Upload so next request hits storage
  await supabaseAdmin.storage
    .from("receipts")
    .upload(fileName, pdfBuffer, { contentType: "application/pdf", upsert: true });
 
  return new NextResponse(new Uint8Array(pdfBuffer), {
    headers: {
      "Content-Type":        "application/pdf",
      "Content-Disposition": `inline; filename="receipt-${orderId.slice(0, 8).toUpperCase()}.pdf"`,
      "Cache-Control":       "private, max-age=3600",
    },
  });
}
