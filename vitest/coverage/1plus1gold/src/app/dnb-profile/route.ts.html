/**
 * GET /dnb-profile — the one-hop bridge that opens our D&B Registered profile.
 *
 * ▶ WHY A BRIDGE EXISTS. Their profile is POST-ONLY. MEASURED against the live
 * endpoint (2026-09-10): a GET to TPSING-COMP-004.aspx carrying the seal key
 * returns a 2.2 KB empty shell titled "D-U-N-S Registered Profile"; the same
 * request as a POST returns 16 KB naming 1PLUS1GOLD and DUNS 659512753. iOS can
 * only ever hand a URL to the system (`UIApplication.shared.open`), which is a
 * GET — so there is no link for the native seal to open. This page IS the link:
 * the app opens it with `open_url`, and the page performs the POST.
 *
 * ▶ WHY IT WORKS WHERE THE IN-APP EMBED CANNOT. `open_url` goes to the SYSTEM
 * browser, outside the app's WKWebView, so the form POST is a plain browser
 * navigation with nothing intercepting it. Inside the app it would be caught by
 * `decidePolicyFor` and re-issued to SFSafariViewController as a GET, dropping
 * the body. VERIFIED that the hand-off actually leaves the app: our AASA
 * (public/.well-known/apple-app-site-association) allow-lists only
 * /account/orders/*, /auth/native-callback*, /shop/* and /verify/*, so this
 * path is NOT a universal link and cannot bounce back into the app.
 *
 * ▶ WHY A ROUTE HANDLER AND NOT A PAGE. The first cut was a React page whose
 * auto-submit lived in a client component — so it could not fire until the whole
 * app bundle had downloaded and hydrated. Device-reported: **2-3 seconds** of
 * looking at the fallback button before the redirect, and a hard dependency on
 * that bundle loading at all (if it failed, the page never submitted). This
 * returns ~1 KB of hand-written HTML with no framework, no hydration and no
 * bundle; the submit runs the moment /dnb-bridge.js arrives.
 *
 * ▶ THE SCRIPT IS A FILE, NOT INLINE. `script-src 'self'` already permits it, so
 * there is no CSP change, no hash to keep in sync, and no dependence on
 * 'unsafe-inline'. It also keeps `docs/attacker-playbook.md` F5's standing claim
 * intact — that `src/` holds exactly ONE `dangerouslySetInnerHTML`, the
 * non-executable JSON-LD.
 *
 * ▶ THE KEY IS READ PER REQUEST, AND THIS PAGE MUST NEVER BE CACHED.
 * D&B issues the key and EXPIRES it. `fetchSealSource` has always read it live,
 * but until 2026-09-10 this route carried `revalidate = 21600` and
 * `s-maxage=21600, stale-while-revalidate=86400` — so the key was read live once
 * and then the RENDERED PAGE, key baked in, was served from the edge for six
 * hours and stale for up to twenty-four. Every visitor after the first few
 * minutes POSTed a dead key and got D&B's error page ("reload the profile").
 * Device-reported by the operator the same day it shipped.
 *
 * ⚠️ The previous header said "THE KEY IS READ LIVE" and that described
 * `fetchSealSource`, not what a visitor received. A true statement about one
 * layer, load-bearing for a claim about another.
 *
 * ▶ MEASURED against the live endpoint on 2026-09-10, not assumed:
 *   fresh key  → HTTP 200, 16,085 bytes, names 1PLUS1GOLD + DUNS 659512753
 *   stale key  → HTTP 200,  2,304 bytes, empty shell with an error div
 *   Referer is irrelevant — a fresh key resolves with either our domain or
 *   theirs as Referer, so that is NOT the discriminator.
 *   The key is reusable within its life (three POSTs over 75s all returned the
 *   full profile), so this is expiry, not single-use — which is exactly why a
 *   six-hour page cache breaks it while a same-minute one would not.
 *
 * ⚠️ The 2026-09-04 record concluded "old keys are not invalidated, so the cache
 * is safe" from two keys tested MINUTES apart. That measurement cannot support a
 * claim about a SIX-HOUR window, and it is now disproven. A validity window has
 * to be measured over the interval you intend to rely on.
 *
 * The seal ARTWORK (`/api/dnb-seal`) keeps its 6h cache and is unaffected: it
 * composites `layers` only and never touches `sealKey`.
 *
 * ▶ CSP: the global policy is `form-action 'self'`, which BLOCKS a cross-origin
 * form POST. next.config.ts allows profiles.dunsregistered.com explicitly —
 * without that entry this page submits nothing and shows no error.
 */
import { fetchSealSource } from "@/lib/dnb-seal-fetch";
 
export const runtime = "nodejs";
// The rendered HTML embeds a short-lived seal key, so it can never be cached
// at any layer. Both directives are required: `force-dynamic` stops Next
// serving a prerendered copy, `revalidate = 0` stops the route being treated
// as static. Guarded by src/__tests__/dnb-profile-no-cache.test.ts.
export const dynamic = "force-dynamic";
export const revalidate = 0;
 
const PROFILE_ACTION = "https://profiles.dunsregistered.com/TPSING-COMP-004.aspx";
 
/**
 * The key is already constrained to `^[A-Za-z0-9]{6,32}$` by `fetchSealSource`,
 * so nothing that reaches here can break out of the attribute today. It is
 * escaped anyway because that guard lives in a DIFFERENT module: a future edit
 * loosening it — to admit a hyphen, say — would turn this line into an
 * injection point with nothing in this file to say so, and the escape costs one
 * function call. Defence at the point of interpolation, not at a distance.
 */
function attr(v: string): string {
  return v
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&#39;");
}
 
const BG = "#16110A";
const MUTED = "#8A7440";
const GOLD = "#C9A84C";
const FONT = "-apple-system, BlinkMacSystemFont, 'Segoe UI', system-ui, sans-serif";
 
function shell(inner: string): Response {
  return new Response(
    `<!doctype html><html lang="en"><head><meta charset="utf-8">
<meta name="viewport" content="width=device-width,initial-scale=1,viewport-fit=cover">
<meta name="robots" content="noindex,nofollow">
<title>D&amp;B Registered Profile — 1Plus1Gold</title>
<style>
html,body{margin:0;height:100%;background:${BG};color:${MUTED};font-family:${FONT}}
main{min-height:100%;display:flex;flex-direction:column;align-items:center;justify-content:center;gap:18px;padding:24px;text-align:center;box-sizing:border-box}
p{font-size:13px;margin:0}
button{font:600 14px ${FONT};color:${BG};background:${GOLD};border:0;border-radius:4px;padding:14px 28px;cursor:pointer}
</style></head><body><main>${inner}</main></body></html>`,
    {
      headers: {
        "Content-Type": "text/html; charset=utf-8",
        // `no-store`, not `no-cache`: the embedded seal key expires, so a
        // revalidated copy is as useless as a fresh one served late. This also
        // covers the failure shell — an unavailable state cached for six hours
        // would outlive the D&B blip that caused it, and the visitor's retry
        // would be answered from cache with the same failure.
        "Cache-Control": "no-store, max-age=0",
      },
    },
  );
}
 
export async function GET() {
  let sealKey = "";
  try {
    ({ sealKey } = await fetchSealSource());
  } catch {
    /* fall through to the explanatory state */
  }
 
  if (!sealKey) {
    return shell(
      `<p>The Dun &amp; Bradstreet profile is temporarily unavailable.<br>Please try again shortly.</p>`,
    );
  }
 
  // The button is the no-JS path and the visible fallback if the submit is
  // slow — never a dead end. `hidden` on the wrapper is removed by nothing:
  // it stays visible, because a bridge that shows only a spinner and then
  // fails silently is worse than one that offers the tap.
  return shell(
    `<p>Opening our Dun &amp; Bradstreet Registered profile…</p>
<form id="dnb" method="post" action="${PROFILE_ACTION}">
<input type="hidden" name="iSealKey" value="${attr(sealKey)}">
<button type="submit">Open D&amp;B Registered Profile</button>
</form>
<script src="/dnb-bridge.js"></script>`,
  );
}
