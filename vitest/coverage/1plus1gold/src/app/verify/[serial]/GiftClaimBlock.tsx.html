 * /verify/[serial].

 * Renders inside the Certificate of Authenticity card when the viewer is NOT the
 * current owner of an owned card (bearer-asset model — claim is enforced

 *
 *   • GENUINE GIFT (Flow A gift_message + unclaimed, or Flow B pending_gifts):
 *     "A gift from {holder}", the giver's note, "Accept Gift", giver thanked.
 *   • PLAIN HAND-OVER (owned card just handed over, never marked a gift):
 *     NEUTRAL "add to your account" copy — no "gift"/"who it's from" language.
 *     The current holder's first name IS still shown to a signed-in viewer
 *     ("Currently registered to {holder}"), and the previous holder gets a
 *     SECURITY notice (not a gift thank-you) when the card is claimed.
 *
 * The card's weight is server-formatted (`weightPhrase`) — serials attach to 1g
 * Splits, standard bars AND Combibars, so the copy never hardcodes "1 gram".
 *
 * PII: the holder's name / message are personal data — passed ONLY to an
 * authenticated viewer. Anonymous viewers get a PII-free teaser; the CTA links
 * to sign-in with ?claim=auto so the claim auto-fires after sign-in.
 */

/** Capitalise the first letter (for sentence-initial use of a weight phrase). */
function capitalise(s: string): string {
  return s.length ? s.charAt(0).toUpperCase() + s.slice(1) : s;
}
 
/** Renders the channel suffix for "{name} has been thanked{suffix}.". */
function thankChannelSuffix(thankedVia: ThankChannel[]): string {
  const labels = thankedVia.map((c) => c === "whatsapp" ? "WhatsApp" : "email");
  if (labels.length === 0) return "";
  if (labels.length === 1) return ` via ${labels[0]}`;
  return ` via ${labels.slice(0, -1).join(", ")} and ${labels[labels.length - 1]}`;
}
