"use client";
 
/**
 * Statements PDF — the downloadable statement pack, with the three choices a
 * counterparty pack needs (2026-11-17, StoneX APAC onboarding):
 *
 *  - **Up to**: any date. A month-end gives the month's close; a date inside the
 *    month gives "For the period 1 to 17 November 2026". Defaults to the anchor
 *    month's end, or today while that month is still running.
 *  - **Version**: Internal (every schedule, memo and creditor class) or External
 *    (for a supplier / bank: no internal schedules or wording).
 *  - **Cash at bank**: the actual balance on that date. It replaces the implied
 *    (balancing) cash and the difference shows as partner drawings. Required for
 *    External, where 0 is valid (the business holds no account of its own).
 */
import { useState } from "react";
import { Download, FileText } from "lucide-react";
import { sgtToday } from "@/lib/finance/sgt-window";
 
const BG    = "var(--color-bg)";
const CARD  = "var(--color-bg-card)";
const GOLD  = "var(--color-gold)";
const TEXT  = "var(--color-offwhite)";
const MUTED = "var(--color-muted)";
const RULE  = "var(--color-rule)";
 
type Variant = "internal" | "external";
 
/** The anchor month's last day, capped at today (a running month stops today). */
export function defaultUpTo(anchorIso: string, todayIso: string): string {
  const [y, m] = anchorIso.slice(0, 7).split("-").map(Number);
  const monthEnd = new Date(Date.UTC(y, m, 0)).toISOString().slice(0, 10);
  return todayIso < monthEnd ? todayIso : monthEnd;
}

export default function StatementsPdfControl({ anchorIso, jurisdiction }: { anchorIso: string; jurisdiction: string }) {
  const [open, setOpen] = useState(false);
  const [upTo, setUpTo] = useState(() => defaultUpTo(anchorIso, sgtToday()));
  const [variant, setVariant] = useState<Variant>("internal");
  const [cash, setCash] = useState("");
 
 
  const { href, reason } = statementsPdfHref({ upTo, variant, cash, jurisdiction });
  const field = { background: BG, border: `1px solid ${RULE}`, color: TEXT } as const;
 
  return (
    <div className="relative">
      <button
        type="button"
        onClick={() => {
          // Each opening starts from the period selector's month.
          if (!open) setUpTo(defaultUpTo(anchorIso, sgtToday()));

        }}
        className="inline-flex items-center gap-2 px-3 py-1.5 rounded font-montserrat text-xs font-semibold uppercase tracking-wider hover:opacity-80"

        >
          <label className="block">
            <span style={{ color: MUTED }}>Up to</span>

                  {v === "internal" ? "Internal" : "External (for counterparties)"}
