            value={search}

      <div className="flex items-center flex-wrap gap-2 mb-5">
        <span className="inline-flex items-center gap-1.5 text-xs" style={{ color: MUTED }}>
          <Calendar size={13} /> Ordered
        </span>
        <input
          type="date"
          value={dateFrom}
          max={dateTo || undefined}
          onChange={e => setDateFrom(e.target.value)}

            onClick={() => { setDateFrom(""); setDateTo(""); }}
            className="inline-flex items-center gap-1 px-2.5 py-1.5 rounded-full text-xs cursor-pointer border"

            <X size={11} /> Clear dates

        )}
      </div>
 
      {/* Select all shown — ticks every PO the current filters show, for one statement */}
      {!loading && visible.length > 0 && (
        <label className="inline-flex items-center gap-2 mb-3 px-1 text-xs cursor-pointer" style={{ color: MUTED }}>
          <input type="checkbox" aria-label="Select all shown purchase orders for the supplier statement"

        </label>
      )}
 
      {loading ? (
        <div className="flex flex-col gap-2.5">
          {[1,2,3].map(i => (
            <div key={i} className="rounded-xl h-16 animate-pulse"
              style={{ background: CARD, border: `1px solid ${RULE}` }} />
          ))}
        </div>
      ) : pos.length === 0 ? (

          <ShoppingCart size={44} className="mx-auto mb-4 opacity-25" color={GOLD} />
          <p className="mb-6 text-sm" style={{ color: MUTED }}>No purchase orders yet.</p>
          <button onClick={() => setShowNewModal(true)}
            className="inline-flex items-center gap-2 px-6 py-3 rounded-xl text-sm font-bold cursor-pointer border-0"
            style={{ background: GOLD, color: BG }}>
            <Plus size={16} /> Create First PO
          </button>
        </div>
      ) : visible.length === 0 ? (
        <div className="text-center py-16 text-sm" style={{ color: MUTED }}>
          No purchase orders match the current filters.
        </div>
      ) : groupBy === "none" ? (
        <div className="flex flex-col gap-2.5">
          {visible.map(renderCard)}
        </div>
      ) : (
        <div className="flex flex-col gap-5">
          {groups.map(g => {
            const subtotal = g.rows.reduce((s, po) => s + grandTotalSgd(po), 0);
            return (
              <div key={g.key} className="flex flex-col gap-2.5">
                {/* Group header — label · count on the left, subtotal grand total on the right */}
                <div className="flex items-center justify-between gap-2 px-1">
                  <span className="font-cinzel text-sm font-bold" style={{ color: GOLD }}>
                    {g.label}
                    <span className="font-montserrat font-normal ml-1.5 text-xs" style={{ color: MUTED }}>· {g.rows.length}</span>
                  </span>
                  <span className="font-montserrat text-xs font-bold tabular-nums" style={{ color: TEXT }}>
                    <Flag code="sg" /> S${new Intl.NumberFormat("en-SG", { maximumFractionDigits: 0 }).format(subtotal)}
                  </span>
                </div>
                {g.rows.map(renderCard)}
              </div>
            );
          })}
        </div>
      )}
 
      {showNewModal && (
        <NewPoModal
          onClose={() => setShowNewModal(false)}
          onCreate={(id) => { setShowNewModal(false); router.push(`/admin/purchase-orders/${id}`); }}
        />
      )}
    </div>
  );
}

function NewPoModal({ onClose, onCreate }: {
  onClose: () => void;
  onCreate: (id: string) => void;
}) {
  const [form, setForm] = useState({
    supplier_name: "", supplier_contact: "",
    status: "draft",
    purchase_currency: "SGD", fx_rate_to_sgd: "1",
    shipping_fee: "0", payment_fee: "0", insurance_fee: "0", other_costs: "0",
    processing_fee_pct: "0",
    payment_status: "pending", payment_reference: "",
    spot_sgd_at_creation: "",
    order_date: "", estimated_delivery_date: "", notes: "",
  });
  const [saving, setSaving] = useState(false);
  const [err, setErr] = useState("");
  const set = (k: string, v: string) => setForm(f => ({ ...f, [k]: v }));
 
  const isNonSgd = form.purchase_currency !== "SGD";
  const fx       = parseFloat(form.fx_rate_to_sgd) || 1;
  // Header costs + spot are entered in the purchase currency; persist in SGD
  // (mirrors the edit modal so the two stay symmetric).
  const toSgd = (nativeStr: string) => {
    const n = parseFloat(nativeStr) || 0;
    return isNonSgd ? n * fx : n;
  };
 
  async function submit(e: React.FormEvent) {
    e.preventDefault();
    if (!form.supplier_name.trim()) { setErr("Supplier name is required"); return; }
    setSaving(true); setErr("");
    const res = await fetch("/api/admin/purchase-orders", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        supplier_name:           form.supplier_name.trim(),
        supplier_contact:        form.supplier_contact.trim() || undefined,
        status:                  form.status,
        purchase_currency:       form.purchase_currency,
        fx_rate_to_sgd:          fx,
        shipping_fee:            toSgd(form.shipping_fee),
        payment_fee:             toSgd(form.payment_fee),
        insurance_fee:           toSgd(form.insurance_fee),
        other_costs:             toSgd(form.other_costs),
        processing_fee_pct:      parseFloat(form.processing_fee_pct) || 0,
        payment_status:          form.payment_status,
        payment_reference:       form.payment_reference.trim() || undefined,
        spot_sgd_at_creation:    form.spot_sgd_at_creation ? toSgd(form.spot_sgd_at_creation) : undefined,
        order_date:              form.order_date || undefined,
        estimated_delivery_date: form.estimated_delivery_date || undefined,
        notes:                   form.notes.trim() || undefined,
      }),
    });
    const d = await res.json();
    setSaving(false);
    if (!res.ok) { setErr(d.error ?? "Failed to create PO"); return; }
    onCreate(d.purchaseOrder.id);
  }
 
  const STATUSES: PoStatus[] = ["draft","confirmed","partially_received","fully_received","cancelled"];
  const PAY_STATUSES: PaymentStatus[] = ["pending","sent","confirmed"];
 
  return (
    // Sheet on mobile (slides up from bottom), centered dialog on sm+
    <AdminModal
      open
      onClose={onClose}
      maxWidthClass="sm:max-w-xl"
      overlayClassName="sm:p-4"
      backdrop="rgba(0,0,0,0.8)"
      panelClassName="p-6 font-montserrat"
      panelStyle={{ background: CARD, border: `1px solid ${RULE}` }}
    >
      <div>
        <h2 className="font-cinzel text-lg m-0 mb-5 flex items-center gap-2" style={{ color: TEXT }}>
          <Plus size={18} color={GOLD} /> New Purchase Order
        </h2>
        <form onSubmit={submit} className="flex flex-col gap-4">
          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
            <Field label="Supplier Name *">
              <input className={inputCls} style={inputSty} value={form.supplier_name}
                autoFocus onChange={e => set("supplier_name", e.target.value)} required />
            </Field>
            <Field label="Supplier Contact">
              <input className={inputCls} style={inputSty} value={form.supplier_contact}
                onChange={e => set("supplier_contact", e.target.value)} placeholder="Email / phone / notes" />
            </Field>
            <Field label="Status">
              <select className={inputCls} style={inputSty} value={form.status} onChange={e => set("status", e.target.value)}>
                {STATUSES.map(s => <option key={s} value={s}>{PO_STATUS_LABELS[s]}</option>)}
              </select>
            </Field>
            <Field label="Payment Status">
              <select className={inputCls} style={inputSty} value={form.payment_status} onChange={e => set("payment_status", e.target.value)}>
                {PAY_STATUSES.map(s => <option key={s} value={s}>{PAYMENT_STATUS_LABELS[s]}</option>)}
              </select>
            </Field>
            <Field label="Currency">
              <CurrencyPicker value={form.purchase_currency} onChange={v => set("purchase_currency", v)} />
            </Field>
            <Field label="FX Rate to SGD">
              <input className={inputCls} style={inputSty} type="number" step="0.0001" min="0.0001"
                value={form.fx_rate_to_sgd} onChange={e => set("fx_rate_to_sgd", e.target.value)} />
            </Field>
            <Field label={`Shipping (${form.purchase_currency})`}>
              <input className={inputCls} style={inputSty} type="number" step="0.01" min="0"
                value={form.shipping_fee} onChange={e => set("shipping_fee", e.target.value)} />
              {isNonSgd && <span className="text-[11px]" style={{ color: MUTED }}>≈ <Flag code="sg" /> S${fmtSgd(toSgd(form.shipping_fee))}</span>}
            </Field>
            <Field label={`Payment Fee (${form.purchase_currency})`}>
              <input className={inputCls} style={inputSty} type="number" step="0.01" min="0"
                value={form.payment_fee} onChange={e => set("payment_fee", e.target.value)} />
              {isNonSgd && <span className="text-[11px]" style={{ color: MUTED }}>≈ <Flag code="sg" /> S${fmtSgd(toSgd(form.payment_fee))}</span>}
            </Field>
            <Field label={`Insurance (${form.purchase_currency})`}>
              <input className={inputCls} style={inputSty} type="number" step="0.01" min="0"
                value={form.insurance_fee} onChange={e => set("insurance_fee", e.target.value)} />
              {isNonSgd && <span className="text-[11px]" style={{ color: MUTED }}>≈ <Flag code="sg" /> S${fmtSgd(toSgd(form.insurance_fee))}</span>}
            </Field>
            <Field label={`Other Costs (${form.purchase_currency})`}>
              <input className={inputCls} style={inputSty} type="number" step="0.01" min="0"
                value={form.other_costs} onChange={e => set("other_costs", e.target.value)} />
              {isNonSgd && <span className="text-[11px]" style={{ color: MUTED }}>≈ <Flag code="sg" /> S${fmtSgd(toSgd(form.other_costs))}</span>}
            </Field>
            <Field label="Processing Fee (%)">
              <input className={inputCls} style={inputSty} type="number" step="0.001" min="0" max="20"
                value={form.processing_fee_pct} onChange={e => set("processing_fee_pct", e.target.value)} />
              <span className="text-[11px]" style={{ color: MUTED }}>% of the lines total</span>
            </Field>
            <Field label="Payment Ref">
              <input className={inputCls} style={inputSty} value={form.payment_reference}
                onChange={e => set("payment_reference", e.target.value)} placeholder="bank ref / UEN" />
            </Field>
            <Field label={`Spot Price (${form.purchase_currency}/g)`}>
              <input className={inputCls} style={inputSty} type="number" step="0.01" min="0"
                value={form.spot_sgd_at_creation} onChange={e => set("spot_sgd_at_creation", e.target.value)}
                placeholder="e.g. 95.40" />
              {isNonSgd && form.spot_sgd_at_creation && (
                <span className="text-[11px]" style={{ color: MUTED }}>
                  ≈ <Flag code="sg" /> S${fmtSgd(toSgd(form.spot_sgd_at_creation))}/g
                </span>
              )}
            </Field>
            <Field label="Order Date">
              <input className={inputCls} style={inputSty} type="date"
                value={form.order_date} onChange={e => set("order_date", e.target.value)} />
            </Field>
            <Field label="Est. Delivery Date">
              <input className={inputCls} style={inputSty} type="date"
                value={form.estimated_delivery_date} onChange={e => set("estimated_delivery_date", e.target.value)} />
            </Field>
          </div>
          <Field label="Notes">
            <textarea className={inputCls} style={{ ...inputSty, minHeight: 68, resize: "vertical" }}
              value={form.notes} onChange={e => set("notes", e.target.value)} />
          </Field>
          {err && (
            <p className="text-sm m-0 px-3 py-2 rounded-lg border"
              style={{ color: "#e05252", background: "rgba(224,82,82,0.1)", borderColor: "rgba(224,82,82,0.3)" }}>
              {err}
            </p>
          )}
          <div className="flex gap-3 mt-1">
            <button type="button" onClick={onClose}
              className="flex-1 py-3 rounded-xl text-sm cursor-pointer font-montserrat border"
              style={{ background: BG, color: MUTED, borderColor: RULE }}>
              Cancel
            </button>
            <button type="submit" disabled={saving}
              className="flex-[2] py-3 rounded-xl text-sm font-bold cursor-pointer border-0 font-montserrat"
              style={{ background: GOLD, color: BG }}>
              {saving ? "Creating…" : "Create PO"}
            </button>
          </div>
        </form>
      </div>
    </AdminModal>
  );
}
