"use client";
 
/**
 * "<marketplace> price gap" (Mukesh 2026-11-19) — how far below our own site buyers get our
 * bars on a marketplace, and who funds the gap: our vouchers, or the marketplace's.
 *
 * Reads the orders route's GET (`priceGap` + each import's `buyer_view`, from
 * lib/marketplace/buyer-view). The site price is our checkout engine at the order's spot,
 * for a NEW customer — the price a direct buyer would have paid that day. It measures the
 * price gap, not whether that buyer would have come direct.
 */
import { useCallback, useEffect, useState } from "react";
import type { BuyerView, PriceGapSummary } from "@/lib/marketplace/buyer-view";
 
const CARD    = "var(--color-bg-card)";
const GOLD    = "var(--color-gold)";
const TEXT    = "var(--color-offwhite)";
const MUTED   = "var(--color-muted)";
const RULE    = "var(--color-rule)";
const SUCCESS = "var(--color-success)";
const WARN    = "#C77D2E";
 
interface GapView {
  priceGap?: { last30: PriceGapSummary; all: PriceGapSummary };
  imports: Array<{ external_order_id: string; order_id: string | null; buyer_view: BuyerView | null }>;
}
 
const sgd = (n: number) => `S$${n.toLocaleString("en-SG", { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`;
 
/** "4.2% below our site", "1.0% above our site". */
export function gapWords(gapPct: number | null): string {
  if (gapPct == null) return "no site price";
  if (Math.abs(gapPct) < 0.05) return "same as our site";
  return gapPct > 0 ? `${gapPct.toFixed(1)}% below our site` : `${Math.abs(gapPct).toFixed(1)}% above our site`;
}
 
function Summary({ title, s }: { title: string; s: PriceGapSummary }) {
  return (
    <div className="rounded-lg p-3 text-xs" style={{ border: `1px solid ${RULE}` }}>
      <p className="uppercase tracking-wider text-[10px] mb-1" style={{ color: MUTED }}>{title}</p>
      {s.compared === 0 ? (
        <p style={{ color: MUTED }}>{s.orders ? `${s.orders} order(s), none with a site price yet` : "No orders yet"}</p>
      ) : (
        <>
          <p className="text-sm font-semibold" style={{ color: s.gapSgd > 0 ? GOLD : TEXT }}>{gapWords(s.gapPct)}</p>
          <p style={{ color: MUTED }}>
            {s.compared} order(s): our site {sgd(s.siteSgd)} · buyers paid {sgd(s.buyerPaidSgd)} · gap {sgd(s.gapSgd)}
          </p>
          <p style={{ color: MUTED }}>
            Vouchers — ours {sgd(s.sellerVoucherSgd)} · the marketplace&rsquo;s {sgd(s.platformVoucherSgd)}
          </p>
        </>
      )}
      {s.mismatches > 0 && <p className="mt-1" style={{ color: WARN }}>{s.mismatches} order(s) where the marketplace&rsquo;s &ldquo;paid&rdquo; does not add up — left out.</p>}
    </div>
  );
}
 
export default function PriceGapSection({ label, api }: { label: string; api: string }) {
  const [view, setView] = useState<GapView | null>(null);
  const [busy, setBusy] = useState(false);
  const [msg, setMsg] = useState<{ ok: boolean; text: string } | null>(null);
 
  const load = useCallback(async (signal?: AbortSignal) => {
    try {
      const r = await fetch(api, { cache: "no-store", signal });
      if (r.ok) setView(await r.json());
    } catch (e) {
      if ((e as Error).name !== "AbortError") setMsg({ ok: false, text: `Could not load the ${label} price gap` });
    }
  }, [api, label]);
  useEffect(() => { const c = new AbortController(); load(c.signal); return () => c.abort(); }, [load]);
 
  const refill = async () => {
    setBusy(true); setMsg(null);
    try {
      const r = await fetch(api, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ action: "refill" }) });
      const j = await r.json().catch(() => ({}));
      if (!r.ok) { setMsg({ ok: false, text: j.error ?? "Refill failed" }); return; }
      setMsg(j.result?.ran
        ? { ok: true, text: `${j.result.seen} order(s) re-read · ${j.result.buyerViews} buyer view(s) filled.` }
        : { ok: false, text: `Not refilled: ${j.result?.reason ?? "unknown"}.` });
      await load();
    } finally {
      setBusy(false);
    }
  };
 
  if (!view?.priceGap) return null;
  const rows = view.imports.filter((r) => r.order_id && r.buyer_view);
 
  return (
    <section className="rounded-2xl p-4 md:p-5" style={{ background: CARD, border: `1px solid ${RULE}` }}>
      <h2 className="text-sm font-semibold mb-1" style={{ color: TEXT }}>{label} price gap</h2>
      <p className="text-xs mb-3" style={{ color: MUTED }}>
        What {label} buyers actually paid, against what our own site would have charged a new customer for the same bars at the same spot.
        It shows how far below our site the marketplace sells, and whose vouchers pay for it. It does not tell you whether that buyer would have come to us direct.
      </p>
      <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
        <Summary title="Last 30 days" s={view.priceGap.last30} />
        <Summary title="Since import started" s={view.priceGap.all} />
      </div>
      {rows.length > 0 && (
        <ul className="space-y-2 mt-3">
          {rows.map((r) => {
            const v = r.buyer_view!;
            return (
              <li key={r.external_order_id} className="rounded-lg p-3 text-xs" style={{ border: `1px solid ${v.check === "mismatch" ? WARN : RULE}` }}>
                <div className="flex flex-wrap justify-between gap-2">
                  <span className="break-all" style={{ color: TEXT }}>{label} {r.external_order_id}</span>
                  <span style={{ color: v.check === "mismatch" ? WARN : v.gapSgd != null && v.gapSgd > 0 ? GOLD : MUTED }}>
                    {v.check === "mismatch" ? "Paid figure does not add up" : gapWords(v.gapPct)}
                  </span>
                </div>
                <div style={{ color: MUTED }}>
                  Listed {sgd(v.listSgd)}
                  {v.sellerVoucherSgd > 0 && ` · our voucher −${sgd(v.sellerVoucherSgd)}`}
                  {v.platformVoucherSgd > 0 && ` · ${label} voucher −${sgd(v.platformVoucherSgd)}`}
                  {` · shipping ${v.buyerShippingSgd > 0 ? sgd(v.buyerShippingSgd) : "free"}`}
                  {` · buyer paid ${sgd(v.buyerPaidSgd)}`}
                  {v.site && ` · our site ${sgd(v.site.totalSgd)} (spot ${sgd(v.site.spotSgd)}/g)`}
                </div>
                {v.check === "mismatch" && (
                  <div style={{ color: WARN }}>{label} says {sgd(v.reportedPaidSgd ?? 0)}; the parts add up to {sgd(v.expectedPaidSgd)}. The receipt prints our own total until this is checked.</div>
                )}
              </li>
            );
          })}
        </ul>
      )}
      <div className="flex flex-wrap items-center gap-3 mt-3">
        <button onClick={refill} disabled={busy} title={`Re-read every ${label} order since the import started and fill any missing buyer view`}
          className="px-4 py-2 rounded-lg text-sm font-medium disabled:opacity-50" style={{ border: `1px solid ${GOLD}`, color: GOLD }}>
          {busy ? "Re-reading…" : "Refill buyer views"}
        </button>
        {msg && <span className="text-xs" style={{ color: msg.ok ? SUCCESS : WARN }}>{msg.text}</span>}
      </div>
    </section>
  );
}
