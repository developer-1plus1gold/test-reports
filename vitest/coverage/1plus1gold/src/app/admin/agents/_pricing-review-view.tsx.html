"use client";
 
/**
 * Control Tower → the daily pricing review (pricing_reviewer, 16:00 SGT;
 * Mukesh 2026-11-14). Renders the decision's action_payload: the summary, one
 * line per open flag (act / hold, the computed target, where it is changed),
 * accepted exceptions, and the full price table — cost is the cheapest restock
 * (supplier named), "Paid" the premium we paid for held stock. Numbers come from code
 * (lib/pricing/daily-review.ts); the words from the agent.
 */
import { useState } from "react";
import type React from "react";
import type { PricingReview, ReviewFlag, ReviewPool } from "@/lib/pricing/daily-review";
import type { ReviewSuggestion } from "@/lib/ai/agents/pricing-reviewer";
 
const CARD = "var(--color-bg-card)";
const GOLD = "var(--color-gold)";
const TEXT = "var(--color-offwhite)";
const MUTED = "var(--color-muted)";
const RULE = "var(--color-rule)";
const RED = "#e57373";
const GREEN = "var(--color-success)";
 
export interface PricingReviewPayload {
  kind: "pricing_review";
  date: string;
  headline: string;
  summary: string;
  suggestions: ReviewSuggestion[];
  agentError: string | null;
  review: PricingReview;
  applies: string;
}
 
export function isPricingReviewPayload(p: unknown): p is PricingReviewPayload {
  const o = p as Partial<PricingReviewPayload> | null;
  return !!o && o.kind === "pricing_review" && !!o.review && Array.isArray(o.review.rows) && Array.isArray(o.review.flags);
}
 
const POOL_LABEL: Record<ReviewPool, string> = {
  curve: "Premium Curve", standard: "Standard", mag_open: "MAG open pool", mag_offer: "MAG pool offer",
};
const pct = (n: number | null | undefined) => (n == null ? "—" : `${Number(n.toFixed(2))}%`);
 
export default function PricingReviewView({ payload }: { payload: PricingReviewPayload }) {
  const [showTable, setShowTable] = useState(false);
  const { review } = payload;
  const open = review.flags.filter(f => !f.accepted);
  const accepted = review.flags.filter(f => f.accepted);
  const byFlag = new Map(payload.suggestions.map(s => [s.flagId, s]));
 
  return (
    <div className="flex flex-col gap-5" data-testid="pricing-review">
      <section>
        <p className="font-cinzel text-xs uppercase tracking-widest mb-2" style={{ color: GOLD }}>
          Pricing review · spot S$ {review.spotSgdPerG.toFixed(2)}/g
        </p>
        {payload.summary && <p className="font-montserrat text-sm" style={{ color: TEXT }}>{payload.summary}</p>}
        {payload.agentError && (
          <p className="font-montserrat text-xs mt-2" style={{ color: RED }}>
            The reviewer could not write today&apos;s notes ({payload.agentError}). The flags below are computed and complete.
          </p>
        )}
      </section>
 
      <section>
        <p className="font-cinzel text-xs uppercase tracking-widest mb-2" style={{ color: GOLD }}>
          To decide ({open.length})
        </p>
        {open.length === 0 ? (
          <p className="font-montserrat text-xs" style={{ color: MUTED }}>Nothing today — every price is within the review thresholds.</p>
        ) : (
          <ul className="flex flex-col gap-2">
            {open.map(f => <FlagLine key={f.id} f={f} s={byFlag.get(f.id)} />)}
          </ul>
        )}
        {open.length > 0 && <p className="font-montserrat text-[11px] mt-2" style={{ color: MUTED }}>{payload.applies}</p>}
      </section>
 
      {accepted.length > 0 && (
        <section>
          <p className="font-cinzel text-xs uppercase tracking-widest mb-2" style={{ color: GOLD }}>Accepted by standing decision ({accepted.length})</p>
          <ul className="flex flex-col gap-1">
            {accepted.map(f => (
              <li key={f.id} className="font-montserrat text-xs" style={{ color: MUTED }}>
                {f.message} <span style={{ color: TEXT }}>· {f.accepted}</span>
              </li>
            ))}
          </ul>
        </section>
      )}
 
      <section>
        <button className="font-montserrat text-xs" style={{ color: GOLD }} onClick={() => setShowTable(v => !v)}>
          {showTable ? "Hide" : "Show"} all {review.rows.length} prices
        </button>
        {showTable && (
          <>
            {/* Phones: one card per price (the 8-column table needs ~760 px). */}
            <ul className="md:hidden mt-2 flex flex-col gap-2" data-testid="pricing-review-cards">
              {review.rows.map(r => <PriceCard key={r.key} r={r} flagged={open.some(f => f.rowKey === r.key)} />)}
            </ul>
            <div className="hidden md:block mt-2 overflow-x-auto rounded-lg" style={{ border: `1px solid ${RULE}` }}>
              <table className="w-full font-montserrat text-[11px]" style={{ color: TEXT }}>
                <thead>
                  <tr style={{ backgroundColor: CARD, color: MUTED }}>
                    {["Pool", "Bar", "Ours", "Cost (restock)", "Paid", "Margin", "Competitor", "Outright"].map(h => (
                      <th key={h} className="text-left px-2 py-1.5 font-semibold whitespace-nowrap">{h}</th>
                    ))}
                  </tr>
                </thead>
                <tbody>
                  {review.rows.map(r => {
                    const flagged = open.some(f => f.rowKey === r.key);
                    return (
                      <tr key={r.key} style={{ borderTop: `1px solid ${RULE}` }}>
                        <td className="px-2 py-1.5 whitespace-nowrap" style={{ color: MUTED }}>{POOL_LABEL[r.pool]}</td>
                        <td className="px-2 py-1.5 min-w-[10rem]" style={{ color: flagged ? GOLD : TEXT }}>
                          {r.label}{r.duration ? ` · ${r.duration}` : ""}
                        </td>
                        <td className="px-2 py-1.5 whitespace-nowrap">{pct(r.ourPct)}</td>
                        <td className="px-2 py-1.5 whitespace-nowrap">
                          {pct(r.costPct)}
                          {r.costSource && <span className="block text-[10px]" style={{ color: MUTED }}>{r.costSource}</span>}
                        </td>
                        <td className="px-2 py-1.5 whitespace-nowrap" title={PAID_HINT}>{pct(r.paidPct)}</td>
                        <td className="px-2 py-1.5 whitespace-nowrap" style={{ color: marginColor(r.marginPts) }}>{marginLabel(r.marginPts)}</td>
                        <td className="px-2 py-1.5 min-w-[8rem]">
                          {r.competitor ? `${pct(r.competitor.pct)} ${r.competitor.name}` : "—"}
                        </td>
                        <td className="px-2 py-1.5 whitespace-nowrap">{r.pool === "mag_offer" ? pct(r.outrightPct) : ""}</td>
                      </tr>
                    );
                  })}
                </tbody>
              </table>
            </div>
          </>
        )}
      </section>
    </div>
  );
}
 
const PAID_HINT = "Premium we paid for the bars we hold, against the spot on the day we bought";
const marginLabel = (m: number | null) => (m == null ? "—" : `${Number(m.toFixed(2))} pts`);
const marginColor = (m: number | null) => (m != null && m < 0 ? RED : TEXT);
 
/** One price as a card (phones). Same fields as a table row. */
function PriceCard({ r, flagged }: { r: PricingReview["rows"][number]; flagged: boolean }) {
  const cell = (label: string, value: React.ReactNode, sub?: string | null, color?: string) => (
    <div className="min-w-0">
      <p className="text-[10px] uppercase tracking-wider" style={{ color: MUTED }}>{label}</p>
      <p className="text-xs break-words" style={{ color: color ?? TEXT }}>{value}</p>
      {sub && <p className="text-[10px] break-words" style={{ color: MUTED }}>{sub}</p>}
    </div>
  );
  return (
    <li className="rounded-lg px-3 py-2 font-montserrat" style={{ backgroundColor: CARD, border: `1px solid ${flagged ? GOLD : RULE}` }}>
      <p className="text-[10px] uppercase tracking-wider" style={{ color: MUTED }}>{POOL_LABEL[r.pool]}</p>
      <p className="text-xs font-semibold break-words mb-2" style={{ color: flagged ? GOLD : TEXT }}>
        {r.label}{r.duration ? ` · ${r.duration}` : ""}
      </p>
      <div className="grid grid-cols-3 gap-x-3 gap-y-2">
        {cell("Ours", pct(r.ourPct))}
        {cell("Cost", pct(r.costPct), r.costSource)}
        {cell("Margin", marginLabel(r.marginPts), null, marginColor(r.marginPts))}
        {cell("Paid", pct(r.paidPct))}
        {cell("Competitor", r.competitor ? pct(r.competitor.pct) : "—", r.competitor?.name)}
        {r.pool === "mag_offer" ? cell("Outright", pct(r.outrightPct)) : <div />}
      </div>
    </li>
  );
}
 
function FlagLine({ f, s }: { f: ReviewFlag; s: ReviewSuggestion | undefined }) {
  const act = s?.action === "act";
  return (
    <li className="rounded-lg px-3 py-2" style={{ backgroundColor: CARD, border: `1px solid ${RULE}` }}>
      <div className="flex flex-wrap items-baseline gap-2">
        <span className="font-montserrat text-[10px] font-semibold uppercase tracking-wider px-1.5 py-0.5 rounded"
          style={{ color: act ? "var(--color-bg)" : MUTED, backgroundColor: act ? GOLD : "transparent", border: act ? "none" : `1px solid ${RULE}` }}>
          {act ? "Act" : "Hold"}
        </span>
        <span className="font-montserrat text-xs" style={{ color: TEXT }}>{f.message}</span>
      </div>
      {f.targetPct != null && act && (
        <p className="font-montserrat text-xs mt-1" style={{ color: GREEN }}>Suggested: spot + {pct(f.targetPct)} · {f.appliesTo}</p>
      )}
      {f.targetPct != null && !act && (
        <p className="font-montserrat text-[11px] mt-1" style={{ color: MUTED }}>Target if acted on: spot + {pct(f.targetPct)} · {f.appliesTo}</p>
      )}
      {s?.why && <p className="font-montserrat text-[11px] mt-1" style={{ color: MUTED }}>{s.why}</p>}
    </li>
  );
}
