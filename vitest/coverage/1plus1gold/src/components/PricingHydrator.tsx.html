"use client";
 
/**
 * PricingHydrator — the single source of pricing data for the entire app.
 *
 * Mounted once in the root layout, just below <Suspense> chrome. Takes a
 * server-rendered snapshot as a prop and hydrates the three Zustand stores
 * (spot, currency, premium) BEFORE any consumer mounts — so first paint of
 * SpotPriceTicker, LiveStartingPrice, ProductCard, etc. always has values.
 *
 * On the client, polls /api/pricing-snapshot every POLL_MS to refresh all
 * three stores atomically. Replaces the three independent polls that used
 * to live in SiteShell (fx-rates 60s, premiums on-demand) and the 10s poll
 * inside SpotPriceTicker.
 *
 * The poll cadence matches today's fastest poll (SpotPriceTicker's 10s) so
 * the customer-visible refresh of the ticker is unchanged. FX + premiums
 * get faster refresh as a side benefit at zero extra network cost — they
 * ride the same response payload.
 *
 * Read the architecture rationale: <ARCHITECTURE_REVIEW_2026-05-24.md>
 * Read related: project_spot_resilience_2026-05-20.md, feedback_currency_display.md
 */
 
import { useEffect, useRef } from "react";
import type { PricingSnapshot } from "@/lib/pricing-snapshot";
import { useSpotStore } from "@/store/spotStore";
import { useCurrencyStore } from "@/store/currencyStore";
import { usePremiumStore } from "@/store/premiumStore";
import { isEdgeCachedPath } from "@/lib/embed/edge-cached";
 
const POLL_MS = 10_000;
/** How old (by its composedAt stamp) a server-baked snapshot may be and still seed. */
const SEED_FRESH_MS = 15_000;
 
/** Exported for the tests. A snapshot without composedAt (older server) is trusted as before. */
export function isFreshSnapshot(snap: PricingSnapshot | null, now = Date.now()): boolean {
  if (!snap) return false;
  Iif (typeof snap.composedAt !== "number") return true;
  return Math.abs(now - snap.composedAt) <= SEED_FRESH_MS;
}
 
interface Props {
  /** Server-fetched snapshot at request time; null if the SSR fetch failed
   *  (network blip, upstream outage, etc.) — the client mount still fires
   *  its own fetch immediately to recover. */
  initial: PricingSnapshot | null;
}
 
export default function PricingHydrator({ initial: serverInitial }: Props) {
  const seededRef = useRef(false);
  // An edge-cached page's HTML (lib/embed/edge-cached) carries the snapshot of
  // whichever request filled the cache — possibly hours old. Never seed from it:
  // the figures stay as placeholders until this client's own fetch lands
  // (Mukesh 2026-11-15, "never old figures"). The server render has empty stores
  // either way (zustand SSR), so skipping the seed keeps hydration consistent.
  // Any other HTML can be replayed old too (the PWA service worker serves
  // navigations stale-while-revalidate, bfcache), so a snapshot is trusted only
  // when it was composed moments ago; otherwise the same: fetch now, no seed.
  const initial = typeof window === "undefined" ? serverInitial
    : isEdgeCachedPath(window.location.pathname) || !isFreshSnapshot(serverInitial) ? null
    : serverInitial;
 
  // ── Seed on the very first render, BEFORE any consumer reads. ──────────
  // Done synchronously during render rather than in useEffect so that the
  // first paint of any pricing-bound component already sees the data —
  // useEffect runs AFTER paint, which would still produce a flash.
  if (!seededRef.current && initial) {
    applyPricingSnapshot(initial);
    seededRef.current = true;
  }
 
  // ── Refresh loop ───────────────────────────────────────────────────────
  useEffect(() => {
    let cancelled = false;
    const ctrl = new AbortController();
 
    const tick = async () => {
      try {
        const res = await fetch("/api/pricing-snapshot", {
          cache:  "no-store",
          signal: ctrl.signal,
        });
        if (!res.ok) return;
        const snapshot = (await res.json()) as PricingSnapshot;
        if (cancelled) return;
        applyPricingSnapshot(snapshot);
      } catch (err) {
        if ((err as Error).name === "AbortError") return;
        // Network blip — the next tick will recover. No console noise.
      }
    };
 
    // If SSR didn't seed (initial=null), fetch immediately to recover.
    // If SSR did seed, the snapshot is already fresh — skip the immediate
    // fetch and wait one interval. Cuts a redundant request per page load.
    if (!initial) void tick();
 
    const interval = setInterval(tick, POLL_MS);
    return () => {
      cancelled = true;
      ctrl.abort();
      clearInterval(interval);
    };
  }, [initial]);
 
  return null;
}
 
export function applyPricingSnapshot(snap: PricingSnapshot) {
  // Spot — same shape as SpotPriceTicker's setSpot call today; the
  // ticker is now pure-display and subscribes to this store.
  useSpotStore.getState().setSpot({
    spotSgd:   snap.spot.sgd,
    spotOzSgd: snap.spot.sgdOz,
    chpSgd:    snap.spot.chpSgd,
    ts:        snap.ts,
    stale:     snap.spot.stale,
    xauStale:  snap.spot.xauStale,
    fxStale:   snap.spot.fxStale,
    xauAgeMs:  snap.spot.xauAgeMs,
    fxAgeMs:   snap.spot.fxAgeMs,
    source:    snap.spot.source,
    // A snapshot from before 2026-11-22 (replayed HTML) has no `market`.
    marketNote: snap.market?.note ?? null,
    marketSpotSgd:   snap.spot.marketSgd ?? null,
    marketSpotOzSgd: snap.spot.marketSgdOz ?? null,
  });
 
  // FX rates — currencyStore's setRates also flips ratesFresh=true.
  useCurrencyStore.getState().setRates(snap.fx.rates);
 
  // Premiums — set directly (no public setter on the store; we own the
  // shape). `loaded=true` flips LiveStartingPrice + similar consumers out
  // of the loading branch.
  usePremiumStore.setState({
    premiums:            snap.premiums,
    standardBaselinePct: snap.standardBaselinePct,
    loaded:              true,
    fetchedAt:           Date.now(),
    fetching:            false,
  });
}
