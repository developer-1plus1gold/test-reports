"use client";
 
/**
 * "Notify me" on a sold-out product (Mukesh, 2026-11-20) — replaces the dead
 * "Sold Out" button on every shop surface (tile, row, accessory, product page).
 *
 *   tap, signed in   → POST /api/account/stock-alerts → "Alert set ✓"
 *   tap, signed out  → sign in, then back to this page with ?notify=<id>, where
 *                      useNotifyFromQuery() sets the alert without a second tap
 *   409 not_sold_out → the page is out of date: the product can be bought now
 *
 * Which products already have an alert is read ONCE per page load and shared by
 * every button (one request however many sold-out cards are on screen).
 * Flow + firing rules: lib/stock-alerts.ts.
 */
import { useEffect, useState, useSyncExternalStore } from "react";
import { Bell, Check } from "lucide-react";
import { safeRedirectPath } from "@/lib/safe-redirect";
 
const API = "/api/account/stock-alerts";
 
// ── Shared, per-page-load set of product ids with an open alert ─────────────
let alerted = new Set<string>();
let loaded: Promise<void> | null = null;
const listeners = new Set<() => void>();
const emit = () => { for (const l of listeners) l(); };
 
function loadOnce(): Promise<void> {
  loaded ??= fetch(API, { cache: "no-store", credentials: "same-origin" })
    .then((r) => (r.ok ? r.json() : null))
    .then((d: { productIds?: string[] } | null) => {
      if (d?.productIds) { alerted = new Set(d.productIds); emit(); }
    })
    .catch(() => { /* signed out or offline: no alerts shown as set */ });
  return loaded;
}
 
function useAlerted(productId: string): boolean {
  useEffect(() => { void loadOnce(); }, []);
  return useSyncExternalStore(
    (cb) => { listeners.add(cb); return () => { listeners.delete(cb); }; },
    () => alerted.has(productId),
    () => false,
  );
}
 
export type NotifyOutcome = "set" | "sign_in" | "available" | "error";
 
/** Set the alert. On 401 it sends the customer to sign in and back. */
export async function requestStockAlert(productId: string): Promise<NotifyOutcome> {
  try {
    const r = await fetch(API, {
      method: "POST",
      credentials: "same-origin",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ product_id: productId }),
    });
    if (r.status === 401) {
      const url = new URL(window.location.href);
      url.searchParams.set("notify", productId);
      const back = safeRedirectPath(url.pathname + url.search + url.hash);
      window.location.href = `/auth/sign-in?next=${encodeURIComponent(back)}`;
      return "sign_in";
    }
    if (r.status === 409) return "available";
    if (!r.ok) return "error";
    alerted = new Set(alerted).add(productId);
    emit();
    return "set";
  } catch {
    return "error";
  }
}
 
/**
 * Back from sign-in with ?notify=<id>: set that alert once, then drop the
 * parameter so a reload or a shared link does not set it again.
 */
export function useNotifyFromQuery(): void {
  useEffect(() => {
    const url = new URL(window.location.href);
    const id = url.searchParams.get("notify");
    if (!id) return;
    url.searchParams.delete("notify");
    window.history.replaceState(window.history.state, "", url.pathname + url.search + url.hash);
    void requestStockAlert(id);
  }, []);
}
 
export default function NotifyMeButton({ productId, className = "" }: { productId: string; className?: string }) {
  const isSet = useAlerted(productId);
  const [busy, setBusy] = useState(false);
  const [note, setNote] = useState<string | null>(null);
 
  if (isSet) {
    return (
      <span
        className={`inline-flex items-center justify-center gap-1.5 font-montserrat text-xs font-semibold px-3 py-2 rounded whitespace-nowrap ${className}`}
        style={{ border: "1px solid var(--color-success)", color: "var(--color-success)" }}
        title="We will message you once when it can be bought again"
      >
        <Check size={12} /> Alert set
      </span>
    );
  }
 
  return (
    <div className={`flex flex-col items-stretch gap-1 ${className}`}>
      <button
        type="button"
        disabled={busy}
        onClick={async () => {
          setBusy(true); setNote(null);
          const out = await requestStockAlert(productId);
          if (out === "available") setNote("It can be bought again: refresh the page");
          else if (out === "error") setNote("Could not set the alert. Try again");
          if (out !== "sign_in") setBusy(false);
        }}
        className="inline-flex items-center justify-center gap-1.5 font-montserrat text-xs font-semibold px-3 py-2 rounded whitespace-nowrap transition-opacity hover:opacity-80 disabled:opacity-50"
        style={{ border: "1px solid var(--color-gold)", color: "var(--color-gold)", backgroundColor: "transparent" }}
        title="We will message you once when it is back in stock or open for pre-order"
      >
        <Bell size={12} /> {busy ? "Setting…" : "Notify me"}
      </button>
      {note && <span className="font-montserrat text-[10px] text-right" style={{ color: "var(--color-muted)" }}>{note}</span>}
    </div>
  );
}
