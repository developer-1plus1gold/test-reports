/**
 * Live Our-Pricing numbers from /api/pricing/overview — the same payload the
 * iOS spec reads. `undefined` while loading, `null` if the summary failed
 * (each section then shows its static wording without the live figure).
 */
// One request per page load, shared by every caller and kept for a minute so
// a back-navigation renders instantly instead of re-fetching (2026-10-07).
const CLIENT_TTL_MS = 60_000;
let shared: { at: number; promise: Promise<PricingSummary | null> } | null = null;
 
function loadSummary(): Promise<PricingSummary | null> {
  if (!shared || Date.now() - shared.at > CLIENT_TTL_MS) {
    const promise = fetch("/api/pricing/overview")
      .then((r) => (r.ok ? r.json() : null))
      .then((j) => (j?.summary as PricingSummary | undefined) ?? null)
      .catch(() => null);
    shared = { at: Date.now(), promise };
    // A failure must not be pinned for a minute — let the next caller retry.
    promise.then((v) => { if (v === null && shared?.promise === promise) shared = null; });
  }
  return shared.promise;
}
