"use client";
 
/**
 * Occasion / sale campaign panels (Admin → Occasions & Sales).
 *
 * Placement (review round 2, Mukesh, 2026-11-01 — per-campaign settings):
 *  - "home": the FULL panel, above the best sellers — the campaign's stage.
 *  - "shop": a COMPACT banner at the top of Shop that opens the same form in
 *            place, so browsing (and buying now) is never pushed down the page.
 *
 * The panel:
 *  - Renders instantly from server data (`initial`); the client fetch only adds
 *    the signed-in customer's registration.
 *  - Hero image left ~30% / details + form right ~70% (stacked on a phone).
 *  - Festive, religion-neutral frame: a thick metallic gold border with the
 *    Home hero bar's gold-dust shimmer and a travelling glow, plus twinkles.
 *  - Products are picked from a dropdown (image · name · weight · CURRENT
 *    price), up to three, each with a quantity — then dates → reminder → Register.
 *  - "Browse & buy now" sits beside Register: interest never blocks a purchase.
 *  - Confetti: a light sprinkle on first view; the launch-ceremony shower on a
 *    new registration.
 *
 * Registering needs a signed-in customer; a signed-out tap keeps the answers in
 * sessionStorage and returns after sign-in (web). Inside the iOS web_panel
 * (`embed`) the WebView is never navigated to sign-in.
 */

interface Item { productId: string; quantity: number }
interface Answers {
  items: Item[];
  plannedBuyDate: string;
  preferredDeliveryDate: string;
  preferredDeliverySlot: "" | "daytime" | "evening";
  marketingConsent: boolean;
}
 
const answersFrom = (c: PublicCampaign): Answers => {
  const m = c.myRegistration;
  return {
    items: (m?.productIds ?? []).map((id, i) => ({ productId: id, quantity: m?.quantities?.[i] ?? 1 })),
    plannedBuyDate: m?.plannedBuyDate ?? "",
    preferredDeliveryDate: m?.preferredDeliveryDate ?? "",
    preferredDeliverySlot: (m?.preferredDeliverySlot as Answers["preferredDeliverySlot"]) ?? "",
    marketingConsent: m?.marketingConsent ?? false,
  };
};
 
function readDraft(slug: string): Answers | null {
  try {
    const raw = sessionStorage.getItem(DRAFT_KEY(slug));
    const d = raw ? (JSON.parse(raw) as Answers) : null;
    return d && Array.isArray(d.items) ? d : null;
  } catch { return null; }
}
function writeDraft(slug: string, a: Answers | null) {
  try {
    if (a) sessionStorage.setItem(DRAFT_KEY(slug), JSON.stringify(a));
    else sessionStorage.removeItem(DRAFT_KEY(slug));
  } catch { /* storage unavailable — the answers are simply not kept */ }
}
 
const fmtDay = (iso: string | null) =>
  iso ? new Date(iso).toLocaleDateString("en-SG", { day: "numeric", month: "short", timeZone: "Asia/Singapore" }) : "";

const fmtLastDay = (iso: string | null) =>
  iso ? fmtDay(new Date(new Date(iso).getTime() - 60_000).toISOString()) : "";
 
export default function CampaignPanels({ initial = [], placement = "shop", embed = false, onLoaded, className = "" }: {
  /** Server-rendered campaigns for this placement (no per-user state). */
  initial?: PublicCampaign[];
  placement?: Placement;
  embed?: boolean;
  onLoaded?: (count: number) => void;
  /** Applied to the wrapper only when something renders (no empty gap). */
  className?: string;
} = {}) {
  const [campaigns, setCampaigns] = useState<PublicCampaign[]>(initial);
  const [signedIn, setSignedIn] = useState(false);
 
  const load = useCallback(async (signal?: AbortSignal) => {
    try {
      const res = await fetch(`/api/campaigns?placement=${placement}`, { signal, cache: "no-store" });
      if (!res.ok) { onLoaded?.(0); return; }
      const d = await res.json();
      setCampaigns(d.campaigns ?? []);
      setSignedIn(!!d.signedIn);
      onLoaded?.((d.campaigns ?? []).length);
    } catch (e) {
      if ((e as Error).name !== "AbortError") onLoaded?.(0);
    }
  }, [onLoaded, placement]);
 
  useEffect(() => {
    const ctrl = new AbortController();
    load(ctrl.signal);
    return () => ctrl.abort();
  }, [load]);
 
  // The iOS embed has no page shell to seed the catalogue — load it here so the
  // product picker shows images and current prices. (On web the page's
  // CatalogHydrator already seeded it.)
  const fetchCatalog = useCatalogStore((s) => s.fetch);
  useEffect(() => { if (embed) fetchCatalog(); }, [embed, fetchCatalog]);
 
  if (campaigns.length === 0) return null;
  return (
    // grid-cols-1 (= minmax(0,1fr)), not a bare `grid`: an implicit track sizes
    // to the content's min-content, and the long one-line product names then
    // pushed the panel to ~665px on a 375px phone (Mukesh, 2026-11-01).
    <div className={`grid grid-cols-1 gap-6 ${className}`}>
      <ConfettiStyles />
      <FestiveStyles />
      {campaigns.map((c) => (
        <CampaignPanel key={c.id} c={c} signedIn={signedIn} embed={embed} placement={placement} onSaved={() => load()} />
      ))}
    </div>
  );
}
 
interface Row { id: string; label: string; weight: string; thumb: string | null; priceSgd: number | null; soldOut: boolean }
 
/** The campaign's products as priced rows — the SAME pricing kernel as the Shop grid. */
function useCampaignRows(c: PublicCampaign): Row[] {
  const catalog = useCatalogStore(useShallow((s) => s.products));
  const spotSgd = useSpotStore((s) => s.spotSgd);
  const premiums = usePremiumStore((s) => s.premiums);
  return useMemo(() => {
    const ids = new Set(c.products.map((p) => p.id));
    const priced = priceShopCatalog(catalog.filter((p) => ids.has(p.id)), spotSgd, premiums);
    const byId = new Map(priced.map((pp) => [pp.id, pp]));
    return c.products.map((p) => {
      const pp = byId.get(p.id);
      return {
        id: p.id,
        label: p.label,
        weight: pp?.weight_display || (p.grams > 0 ? `${Number(p.grams)}g` : ""),
        thumb: (pp ? getThumb(pp) : null) ?? p.imageUrl,
        priceSgd: pp?.pricing.priceSgd ?? null,
        soldOut: pp?.pricing.soldOut === true,
      };
    });
  }, [c.products, catalog, spotSgd, premiums]);
}

/**
 * Choose up to `max` products from a dropdown (image · name · weight · current
 * price); the chosen ones show as short rows with a quantity each.
 */
function ProductPicker({ rows, max, items, onChange }: {
  rows: Row[]; max: number; items: Item[]; onChange: (items: Item[]) => void;
}) {
  const [open, setOpen] = useState(false);
  const boxRef = useRef<HTMLDivElement>(null);
  const fmt = useFmt();
  const flagCode = useFlag();
  const byId = useMemo(() => new Map(rows.map((r) => [r.id, r])), [rows]);
  const chosen = new Set(items.map((i) => i.productId));
  const full = items.length >= max;
 
  useEffect(() => {
    if (!open) return;
    const onDoc = (e: MouseEvent) => { if (boxRef.current && !boxRef.current.contains(e.target as Node)) setOpen(false); };
    document.addEventListener("mousedown", onDoc);
    return () => document.removeEventListener("mousedown", onDoc);
  }, [open]);
 
  const price = (r: Row) => (r.priceSgd != null
    ? <span className="whitespace-nowrap"><span className="text-[10px] uppercase tracking-wider mr-1" style={{ color: "var(--color-muted)" }}>Current price</span><Flag code={flagCode} />{fmt(r.priceSgd)}</span>
    : <span style={{ color: "var(--color-muted)" }}>Loading…</span>);
 
  const thumb = (r: Row) => (
    <span className="shrink-0 w-9 h-9 rounded overflow-hidden" style={{ background: "var(--color-bg)" }}>
      {r.thumb && (
        // eslint-disable-next-line @next/next/no-img-element
        <img src={r.thumb} alt="" width={36} height={36} className="w-full h-full object-contain" loading="lazy" decoding="async" />
      )}
    </span>
  );
 
  const setQty = (id: string, q: number) => onChange(items.map((i) => (i.productId === id ? { ...i, quantity: Math.min(MAX_QTY, Math.max(1, q)) } : i)));
 
  return (
    <div className="grid grid-cols-1 gap-2 min-w-0">
      <span className="font-montserrat text-xs" style={{ color: "var(--color-offwhite)" }}>
        Which pieces are you thinking of? Choose up to {max}.
      </span>
      <div ref={boxRef} className="relative">
        <button type="button" onClick={() => !full && setOpen((o) => !o)} aria-expanded={open} aria-haspopup="listbox" disabled={full}
          className="w-full flex items-center justify-between px-3 py-2.5 rounded font-montserrat text-sm"
          style={{ background: "var(--color-bg)", border: "1px solid var(--color-gold)", color: "var(--color-offwhite)", opacity: full ? 0.6 : 1 }}>
          <span>{full ? `${max} chosen — remove one to change` : items.length === 0 ? "Choose a product" : "Add another product"}</span>
          <ChevronDown size={16} style={{ color: "var(--color-gold)" }} />
        </button>
        {open && (
          <div role="listbox" className="absolute z-20 mt-1 w-full rounded shadow-xl overflow-y-auto"
            style={{ background: "var(--color-bg-card)", border: "1px solid var(--color-gold)", maxHeight: 300 }}>
            {rows.filter((r) => !chosen.has(r.id)).map((r) => (
              <button key={r.id} type="button" role="option" aria-selected={false}
                onClick={() => { onChange([...items, { productId: r.id, quantity: 1 }]); setOpen(false); }}
                className="w-full flex items-center gap-3 px-3 py-2 text-left hover:bg-[rgba(201,168,76,0.10)]">
                {thumb(r)}
                <span className="flex-1 min-w-0 font-montserrat text-sm" style={{ color: "var(--color-offwhite)" }}>
                  <span className="line-clamp-2 sm:line-clamp-1">
                    {r.label}{r.weight && <span style={{ color: "var(--color-muted)" }}> · {r.weight}</span>}
                    {r.soldOut && <span className="ml-1 text-xs" style={{ color: "var(--color-muted)" }}>(sold out)</span>}
                  </span>
                  <span className="block sm:hidden text-xs" style={{ color: "var(--color-gold)" }}>{price(r)}</span>
                </span>
                <span className="hidden sm:inline font-montserrat text-sm" style={{ color: "var(--color-gold)" }}>{price(r)}</span>
              </button>
            ))}
          </div>
        )}
      </div>
      {items.map((it) => {
        const r = byId.get(it.productId);
        if (!r) return null;
        return (
          <div key={it.productId} className="flex items-center gap-3 px-3 py-2 rounded" style={{ border: "1px solid var(--color-gold)", background: "rgba(201,168,76,0.06)" }}>
            {thumb(r)}
            <span className="flex-1 min-w-0 font-montserrat text-sm" style={{ color: "var(--color-offwhite)" }}>
              <span className="line-clamp-2 sm:line-clamp-1">{r.label}{r.weight && <span style={{ color: "var(--color-muted)" }}> · {r.weight}</span>}</span>
              <span className="block sm:hidden font-montserrat text-xs" style={{ color: "var(--color-gold)" }}>{price(r)}</span>
            </span>
            <span className="hidden sm:inline font-montserrat text-sm" style={{ color: "var(--color-gold)" }}>{price(r)}</span>
            <span className="inline-flex items-center rounded" style={{ border: "1px solid var(--color-rule)" }}>
              <button type="button" aria-label="Fewer" className="px-1.5 py-1" style={{ color: "var(--color-offwhite)" }} onClick={() => setQty(it.productId, it.quantity - 1)}><Minus size={12} /></button>
              <span className="px-2 font-montserrat text-sm" style={{ color: "var(--color-offwhite)" }} aria-live="polite">{it.quantity}</span>
              <button type="button" aria-label="More" className="px-1.5 py-1" style={{ color: "var(--color-offwhite)" }} onClick={() => setQty(it.productId, it.quantity + 1)}><Plus size={12} /></button>
            </span>
            <button type="button" aria-label={`Remove ${r.label}`} onClick={() => onChange(items.filter((i) => i.productId !== it.productId))} style={{ color: "var(--color-muted)" }}>
              <X size={14} />
            </button>
          </div>
        );
      })}
    </div>
  );
}
