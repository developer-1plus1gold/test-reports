"use client";
 
import { useEffect } from "react";
import { useCurrencyStore, CURRENCIES } from "@/store/currencyStore";
import { useUserPrefsStore, type PriceUnit } from "@/store/userPrefsStore";
 
// One-way native→web bridge for ANY ?embed=true WKWebView surface.
// Mounted globally by SiteShell's embed branch (2026-05-31) so every page
// the iOS app embeds (charts, gold-yield, gold-yield/calculator, sell-to-us,
// convert-to-combibar, wholesale, newsletter) responds to the native
// header's currency/unit. iOS native owns the user's active currency + unit
// preference; when those change on the iOS header, native runs
//   webView.evaluateJavaScript("window.__setCurrency('INR')")
// and the embed updates its prices without a page reload.
//
// Why one-way: in embed mode the PWA's AppHeader is stripped (SiteShell
// returns the bare branch), so users can't change the global currency from
// inside the embed — the native header is the only source of truth.
//
// Outbound (web→native) sync isn't wired here; if a future feature inside
// an embed needs to push state back to native, add a
// window.webkit?.messageHandlers?.<name>?.postMessage(...) call from the
// relevant component and a handler on the Swift side.
export default function EmbedNativeBridge() {
  const setCurrency = useCurrencyStore((s) => s.setCurrency);
  const updatePrefs = useUserPrefsStore((s) => s.updatePrefs);
 
  useEffect(() => {
    const w = window as unknown as {
      __setCurrency?: (code: string) => void;
      __setUnit?: (unit: PriceUnit) => void;
      __setPreferredCurrencies?: (codes: string[]) => void;
    };
 
    w.__setCurrency = (code: string) => {
      const upper = (code ?? "").toUpperCase();
      if (CURRENCIES[upper]) setCurrency(upper);
    };
 
    w.__setUnit = (unit: PriceUnit) => {
      if (unit === "gram" || unit === "oz" || unit === "kg") {
        updatePrefs({ priceUnit: unit });
      }
    };
 
    // Native (iOS) pushes the Gold-Price currency SET here when it changes
    // in Preferences — a live push (no reload), mirroring __setUnit. Keep
    // only known codes; ignore an empty list (the chart needs ≥1).
    w.__setPreferredCurrencies = (codes: string[]) => {
      if (!Array.isArray(codes)) return;
      const valid = codes.filter((c) => typeof c === "string" && CURRENCIES[c.toUpperCase()]).map((c) => c.toUpperCase());
      if (valid.length > 0) updatePrefs({ preferredCurrencies: valid });
    };
 
    return () => {
      delete w.__setCurrency;
      delete w.__setUnit;
      delete w.__setPreferredCurrencies;
    };
  }, [setCurrency, updatePrefs]);
 
  return null;
}
