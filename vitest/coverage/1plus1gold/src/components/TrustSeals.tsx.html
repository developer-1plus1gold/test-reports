/**
 * The footer's credential-seal row. The ROW — the seals themselves are data.
 *
 * ▶ WHY A COMPONENT AND NOT INLINE MARKUP. Two callers (the desktop footer band
 * and the mobile one) that must not drift: a seal added to one and forgotten in
 * the other is precisely the failure that put this band here, and it hid for
 * three months with every test green. One component, both breakpoints.
 *
 * ▶ WHY THE SEALS ARE DATA. Which credentials we hold, what each links to and
 * what it looks like are all in @/lib/trust-seals — read by this component AND
 * by the native trust band (`trustBandSection()` in src/app/spec.ts). Adding a
 * credential is a row there plus its artwork; it is not an edit here. The
 * reasoning for the registry's shape — in particular why `image + href` is the
 * standard and needs no CSP change — lives in that file's header.
 *
 * ▶ THE NATIVE BAND CARRIES THESE SEALS TOO. It did not until 2026-09-10 and
 * this file used to say so; that note is now wrong and is removed rather than
 * left to mislead. What the app cannot host is D&B's live <iframe>, because
 * their profile is POST-only and WKWebView hands every external URL to
 * `UIApplication.shared.open`, a GET — a dead badge, which @/lib/dnb-seal
 * records twice. The app renders the seal from `image` + `href` instead
 * (/api/dnb-seal composited from their own layers, /dnb-profile performing the
 * POST from the system browser). Device-confirmed on App Store 1.0.5. So the
 * right rule is not "there is no iOS port" but "do NOT port the EMBED".
 */
import { activeTrustSeals } from "@/lib/trust-seals";
 
export default function TrustSeals() {
  const seals = activeTrustSeals();
  if (seals.length === 0) return null;
 
  return (
    /* gap-12 = 48px, and it is a LICENCE requirement rather than taste. ATC
       guideline 1.4 requires clear space around the SME500 mark into which
       nothing may intrude — no text, no rule, no border and specifically no
       other logo. The ratio the kit's own usage sets is ~0.30 x mark height,
       so at 140px that is >= 42px on all four sides. gap-6 (24px) put the D&B
       seal inside that space. `flex-wrap` means this is the ROW gap too, so a
       wrapped mobile layout inherits the same separation rather than losing it.
       Do not tighten this to balance the row visually. */
    <div className="flex flex-wrap items-center justify-center gap-12" aria-label="Business credentials">
      {seals.map((seal) =>
        seal.webEmbed ? (
          /* The issuer's own live embed — today only D&B, whose seal is
             animated, interactive and authenticates by DOMAIN (so it renders
             blank on localhost and previews: expected, not a bug). Its sandbox
             policy and the two regressions behind each token live in
             @/lib/dnb-seal, pinned in BOTH directions by
             src/__tests__/dnb-seal.test.ts — do NOT re-tighten it here.
 
             loading="lazy" is NOT a nicety. Both footers render this component,
             so the markup exists twice on every page and only one copy is ever
             visible; MEASURED in Chromium, an eagerly-loaded iframe inside a
             `display:none` ancestor still fetches — 2 copies, 2 fetches, at
             every breakpoint — so mirroring the band to mobile would otherwise
             have doubled third-party requests site-wide. It also defers the
             visible copy until the reader scrolls to the footer, which is below
             the fold by definition. Safari before 16.4 ignores the attribute
             and loads eagerly, i.e. degrades to the old behaviour, never worse.
             Guarded because losing it is INVISIBLE: the page renders
             identically and only the request count changes. */
          <iframe
            key={seal.id}
            title={seal.label}
            src={seal.webEmbed.src}
            width={seal.width}
            height={seal.height}
            loading="lazy"
            sandbox={seal.webEmbed.sandbox}
            referrerPolicy={seal.webEmbed.referrerPolicy}
            style={{ border: 0, overflow: "hidden" }}
          />
        ) : (
          /* The standard shape: our own hosted mark, linked to the issuer.
             rel="noopener" because target="_blank" otherwise hands the opened
             page a live `window.opener` reference back into ours. */
          <a
            key={seal.id}
            href={seal.href}
            target="_blank"
            rel="noopener noreferrer"
            className="inline-flex items-center"
          >
            {/* eslint-disable-next-line @next/next/no-img-element -- a fixed-size
                credential mark at its own intrinsic geometry; next/image would
                add a loader round-trip and layout machinery for no gain. */}
            <img
              src={seal.image}
              alt={seal.label}
              width={seal.width}
              height={seal.height}
              loading="lazy"
              style={{ display: "block" }}
            />
          </a>
        ),
      )}
    </div>
  );
}
