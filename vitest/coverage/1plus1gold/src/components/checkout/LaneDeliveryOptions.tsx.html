"use client";
 
/**
 * Checkout: the delivery lane offers (SDUI contract §253, §255) — one radio tile
 * per lane that is on, under a header per delivery day ("TODAY · SUN 22 NOV", bold
 * offwhite so it holds its own against the tiles, §258 H1), in
 * the server's order (day, then promised time), nothing pre-selected. Every tile
 * shows a price: a 100% discount reads struck price → Free; a lane that genuinely
 * costs nothing reads S$0.00 — never a bare "Free" (Mukesh 2026-11-22). Offers outside their (configured) minimum or maximum for this
 * order are not listed; one line says from what amount more options exist.
 *
 * Display only: the fee shown is the server's, and the order routes re-check
 * the lane, its date and its min/max, and force the fee (checkShippingChoice).
 */
import { Fragment } from "react";
import Flag from "@/components/Flag";
import { useFmt, useFlag } from "@/store/currencyStore";
import type { LaneOffer } from "@/lib/shipping/lanes";
 
/**
 * The lane's courier, as its own line under the title (§258 C1, Mukesh 2026-11-22:
 * a bare logo "is not adding any value") — only when the lane shows it at checkout.
 * The name carries the meaning; the logo just helps recognition.
 */
function CourierLine({ logo }: { logo: { src: string; alt: string } }) {
  return (
    <div className="flex items-center gap-1.5 mt-1" data-testid="lane-courier">
      <span className="inline-flex items-center justify-center shrink-0"
        style={{ width: 16, height: 16, backgroundColor: "#ffffff", border: "1px solid rgba(0,0,0,0.15)", borderRadius: 3, overflow: "hidden", padding: 1 }}>
        {/* eslint-disable-next-line @next/next/no-img-element -- tiny static brand mark; next/image adds nothing here */}
        <img src={logo.src} alt="" width={16} height={16} className="w-full h-full object-contain" loading="lazy" />
      </span>
      <span className="font-montserrat text-xs" style={{ color: "var(--color-muted)" }}>Delivered by {logo.alt}</span>
    </div>
  );
}
 
/** "Afternoon delivery · 3–6 pm" / "Next Day Best Effort - SPX Express". */
export function laneOfferTitle(o: Pick<LaneOffer, "displayName" | "timeLabel">): string {
  return o.timeLabel ? `${o.displayName} · ${o.timeLabel}` : o.displayName;
}
 
export default function LaneDeliveryOptions({
  offers, note, selectedId, onSelect,
}: {
  /** The offers this order qualifies for (visibleLaneOffers(...).shown). */
  offers: readonly LaneOffer[];
  /** visibleLaneOffers(...).note — shown once under the list. */
  note: string | null;
  selectedId: string | null;
  onSelect: (offer: LaneOffer) => void;
}) {
  const fmt = useFmt();
  const flagCode = useFlag();
  return (
    <>
      {offers.map((o, i) => {
        const sel = o.id === selectedId;
        const newDay = i === 0 || offers[i - 1].date !== o.date;
        return (
          <Fragment key={o.id}>
          {newDay && (
            <p data-testid={`lane-day-${o.date}`}
              className={`font-montserrat text-xs font-bold uppercase tracking-wider px-1 ${i === 0 ? "" : "pt-1"}`}
              style={{ color: "var(--color-offwhite)" }}>
              {o.dayTitle}
            </p>
          )}
          <button type="button" onClick={() => onSelect(o)} aria-pressed={sel}
            data-testid={`lane-offer-${o.id}`}
            className="flex items-start gap-3 w-full text-left rounded-lg px-4 py-3.5 border transition-colors"
            style={{ borderColor: sel ? "var(--color-gold)" : "var(--color-rule)", backgroundColor: sel ? "rgba(201,168,76,0.06)" : "transparent" }}>
            <span className="mt-0.5 w-4 h-4 rounded-full border-2 flex items-center justify-center shrink-0"
              style={{ borderColor: sel ? "var(--color-gold)" : "var(--color-rule)" }}>
              {sel && <span className="w-2 h-2 rounded-full" style={{ backgroundColor: "var(--color-gold)" }} />}
            </span>
            <div className="flex-1 min-w-0">
              <div className="flex items-center justify-between gap-2 flex-wrap">
                <p className="font-montserrat text-sm font-semibold min-w-0" style={{ color: "var(--color-offwhite)" }}>
                  {laneOfferTitle(o)}
                </p>
                <div className="flex items-center gap-1.5">
                  {o.grossFee > o.fee && (
                    <span className="font-montserrat text-xs line-through" style={{ color: "var(--color-muted)" }}>
                      <Flag code={flagCode} />{fmt(o.grossFee)}
                    </span>
                  )}
                  <span className="font-montserrat text-sm font-bold flex items-center gap-0.5" style={{ color: "var(--color-gold)" }}>
                    {o.fee <= 0 && o.grossFee > 0
                      ? <span style={{ color: "var(--color-success)" }}>Free</span>
                      : <><Flag code={flagCode} />{fmt(Math.max(0, o.fee))}</>}
                  </span>
                </div>
              </div>
              {o.carrierLogo && <CourierLine logo={o.carrierLogo} />}
              {o.notes && (
                <p className="font-montserrat text-xs mt-1" style={{ color: "var(--color-muted)" }}>{o.notes}</p>
              )}
              {o.discountPct > 0 && o.grossFee > o.fee && (
                <span className="inline-block mt-1 font-montserrat text-[10px] px-1.5 py-0.5 rounded"
                  style={{ backgroundColor: "rgba(29,158,117,0.1)", color: "var(--color-success)" }}>
                  {o.discountPct}% discount applied
                </span>
              )}
            </div>
          </button>
          </Fragment>
        );
      })}
      {note && (
        <p className="font-montserrat text-xs px-1" data-testid="lane-min-note" style={{ color: "var(--color-muted)" }}>{note}</p>
      )}
    </>
  );
}
