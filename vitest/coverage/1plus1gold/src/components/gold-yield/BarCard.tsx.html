/**
 * One enrolled Gold Yield bar, as a COLLAPSIBLE card.
 *
 * ▶ WHY IT COLLAPSES (operator, 2026-09-21: "for the details of each bar that
 * we are showing on dashboards, we should be able to collapse those else this
 * page will become too big and complicated to read"). Every bar renders a
 * donut, four metric tiles, a nine-line settlement breakdown, a cumulative
 * earnings chart, a sales table and the exit controls — roughly a screen each.
 * A participant with four bars had four screens of detail before they could
 * see their fourth bar's name. The admin panel has had exactly this treatment
 * since it was written; this brings the participant's own dashboard into line.
 *
 * ▶ EXTRACTED FROM THE PAGE so the collapse can be tested by BEHAVIOUR rather
 * than by matching source text. `src/app/gold-yield/page.tsx` is a page file,
 * and a page may not safely carry an arbitrary named export (the #259 lesson —
 * an invalid export fails only at `next build`, which neither `verify` nor CI
 * Tests runs). A component file can, so the guard renders the real card,
 * clicks the real header and asserts what is and is not in the document.
 *
 * ▶ WHAT STAYS VISIBLE WHEN COLLAPSED IS THE DESIGN DECISION. A card showing
 * only an id forces the reader to open everything, which is the problem again
 * with extra clicks. The header keeps: which bar, what it is, its status, the
 * terms it is on (hedged), whether an exit is in flight, how much has sold,
 * and the earnings — so the common question ("how is my gold doing?") is
 * answered without opening anything.
 */

function StatusBadge({ status }: { status: string }) {
  const map: Record<string, { label: string; color: string; bg: string }> = {
    pending_receipt: { label: "Pending Receipt",  color: "#9CA3AF", bg: "#1F2937" },
    received:        { label: "Received — verifying", color: "#F59E0B", bg: "#3B2F00" },
    active:          { label: "Active",           color: "#22C55E", bg: "#14532D" },
    partially_sold:  { label: "Partially Sold",   color: "#D4AF37", bg: "#3B2F00" },
    fully_sold:      { label: "Fully Sold",       color: "#D4AF37", bg: "#3B2F00" },
    rejected:        { label: "Rejected",         color: "#EF4444", bg: "#3B0000" },
    exited:          { label: "Exited",           color: "#9CA3AF", bg: "#1F2937" },
  };
  const s = map[status] ?? { label: status, color: "#9CA3AF", bg: "#1F2937" };
  return (
    <span
      className="inline-flex items-center px-2 py-0.5 rounded text-xs font-montserrat font-semibold"
      style={{ color: s.color, backgroundColor: s.bg }}
    >
      {s.label}
    </span>
  );
}
 
/**
 * True when a live bar carries no settlement basis. Only an explicit NULL
 * counts — `undefined` means the caller never loaded the field, and a chip
 * that fired on a missing prop would label every bar on any page that omits it.
 */
export function awaitingCostBasis(bar: Pick<YieldBar, "status" | "premium_pct" | "spot_at_enrol">): boolean {
  if (["rejected", "exited", "fully_sold"].includes(bar.status)) return false;
  return bar.premium_pct === null || bar.spot_at_enrol === null;
}
