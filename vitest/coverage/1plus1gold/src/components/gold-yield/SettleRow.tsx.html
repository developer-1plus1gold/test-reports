import Flag from "@/components/Flag";
 
/**
 * One line of a Gold Yield settlement breakdown.
 *
 * ▶ SHARED BY TWO SURFACES ON PURPOSE. The dashboard renders a REAL settlement
 * (BarCard.tsx) and the simulator renders a PROJECTED one
 * (app/gold-yield/calculator/page.tsx), and a participant compares them
 * directly — the operator's ask was that the simulator "show the earnings
 * potential the same way as we show it on the user dashboard". Two copies of
 * this row would let the two screens drift apart in wording, colour or
 * null-handling, which is precisely the divergence the shared settlement kernel
 * exists to prevent one layer down. One row component, one presentation.
 *
 * ▶ A NULL RENDERS "—", NEVER 0.00. Several of these figures are unknown on a
 * legacy sale that predates `principal_per_g`, and a zero there would read as
 * "your capital is not coming back" on the page a participant reconciles their
 * money against. The dash plus the note under the block says which it is.
 *
 * ▶ `signed` COLOURS BY SIGN AND KEEPS THE MINUS. It is used for the net
 * position, which is genuinely negative when a bar sells at the premium it was
 * bought at — the number this whole block exists to stop hiding.
 */
export default function SettleRow({
  label, sub, value, fmt, flagCode, indent, strong, gold, signed,
}: {
  label:    string;
  sub?:     string | null;
  value:    number | null;
  fmt:      (n: number) => string;
  flagCode: string;
  indent?:  boolean;
  strong?:  boolean;
  gold?:    boolean;
  signed?:  boolean;
}) {
  const negative = signed && value != null && value < 0;
  const color = value == null
    ? "var(--color-muted)"
    : negative
      ? "#C1785A"
      : signed && value > 0
        ? "var(--color-success)"
        : gold
          ? "var(--color-gold)"
          : "var(--color-offwhite)";
 
  return (
    <div className="flex items-start justify-between gap-3">
      <dt className={indent ? "pl-3" : ""} style={{ color: "var(--color-muted)" }}>
        {label}
        {sub && (
          <span className="block text-[10px]" style={{ color: "var(--color-muted)" }}>{sub}</span>
        )}
      </dt>
      <dd
        className={`flex items-center gap-1 shrink-0 ${strong ? "font-semibold" : ""}`}
        style={{ color }}
      >
        {value == null ? "—" : (
          <>
            {negative && <span>-</span>}
            <Flag code={flagCode} />
            <span>{fmt(Math.abs(value))}</span>
          </>
        )}
      </dd>
    </div>
  );
}
