"use client";
 
/**
 * NfcRowEditor — renders the NFC packing state for a single order_item.
 *
 * Card-count aware: one NFC card per physical PIECE, which is not always
 * `quantity`. Two ways a line needs more than one card:
 *   • quantity > 1 (three separate 1g splits)
 *   • a pool `bundle` product — "1g x 10" is ONE unit that ships as TEN
 *     separate 1g pieces, each with its own card, certificate and buyback
 *     path. A qty-1 line therefore needs 10 cards.
 * The count arrives as `cardsRequired` (server-computed in /api/admin/orders
 * via cardsRequired()); we never re-derive the pack multiplier here. The
 * editor lists linked serials and shows ONE input row for the next card
 * until the line is full.
 *
 * Capacity (linked ≤ cardsRequired) is also enforced server-side in
 * /api/admin/serials/assign — both lines of defence keep the count honest
 * if someone fat-fingers an extra scan.
 *
 * Parent-bar serial — collected at warehouse RECEIPT, not at dispatch.
 * Each NFC card is generated against a product, not a parent bar; the
 * parent linkage is the inventory-side audit trail and gets attached
 * later. So no parent-bar field here.
 */
 
import { useState } from "react";
import Link from "next/link";
import { serialCounterpart } from "@/lib/admin/serial-counterpart";
import { ShieldCheck, ShieldOff, ExternalLink, Loader2, Printer, Gift } from "lucide-react";
 
const GOLD  = "var(--color-gold)";
const TEXT  = "var(--color-offwhite)";
const MUTED = "var(--color-muted)";
const RULE  = "var(--color-rule)";
const BG    = "var(--color-bg)";
 
export interface AssignedSerial {
  serial:            string;
  parent_bar_serial: string | null;
  // The real serial on a standard bar (provenance). null for curve/splits.
  manufacturer_serial?: string | null;
  /**
   * What the parent bar actually IS — "Multigram" / "Combibar" / "Source bar" —
   * resolved SERVER-SIDE from the parent product's own label by
   * `splitParentTypeLabel`, the one function that owns this decision
   * (src/lib/admin/split-parent-labels.ts).
   *
   * ▶ This chip hardcoded "Combibar" until 2026-09-19, so a PAMP Multigram+25
   * split read "Combibar <serial>" on the screen the operator reads while
   * physically packing the card — contradicting its own certificate, which had
   * been fixed on 2026-07-17. The rule was right and this was a third copy of
   * it. Never re-derive the word here; render what the server resolved.
   *
   * Optional because a route that has not been updated, or a parent whose
   * product cannot be resolved, yields null — in which case the bare serial is
   * shown. A missing word beats a wrong one on a card being packed.
   */
  parent_type_label?: string | null;
}
 
export default function NfcRowEditor({
  orderItemId,
  quantity,
  cardsRequired,
  initial,
  productType = "curve",
  giftMessages = [],
  onChange,
}: {
  orderItemId: string;
  quantity:    number;
  /** Cards this line needs — quantity × pack grams for a pool `bundle`
   *  ("1g x 10" → 10 cards on a qty-1 line), else quantity. Server-computed
   *  in /api/admin/orders; falls back to `quantity` when absent. */
  cardsRequired?: number;
  initial:     AssignedSerial[];
  /** "curve" → operator enters OUR NFC serial; "standard" → operator enters the
   *  bar's MANUFACTURER serial and we resolve + show our serial read-only. */
  productType?: string;
  /** Per-card gift messages captured at checkout. The packing-time
   *  /api/admin/serials/assign route pops slot [linked.length] off this
   *  array and writes it onto nfc_serials.gift_message when the operator
   *  scans the next card. We mirror that index here so the operator can
   *  preview which message will bind to the next slot, and confirm the
   *  right physical card is going into the right gift. */
  giftMessages?: string[];
  onChange?:   (serials: AssignedSerial[]) => void;
}) {
  const [linked, setLinked] = useState<AssignedSerial[]>(initial);
  const [serial, setSerial] = useState("");
  const [busy,   setBusy]   = useState(false);
  const [error,  setError]  = useState<string | null>(null);
 
  const needed      = cardsRequired ?? quantity;
  const remaining   = Math.max(needed - linked.length, 0);
  const allLinked   = remaining === 0;
  const isGiftLine  = giftMessages.length > 0;
  const nextMessage = isGiftLine && !allLinked ? giftMessages[linked.length] : null;
 
  function applyChange(next: AssignedSerial[]) {
    setLinked(next);
    onChange?.(next);
  }
 
  async function handleAssign(e: React.FormEvent) {
    e.preventDefault();
    if (!serial.trim() || busy) return;
    setBusy(true);
    setError(null);
    try {
      const res = await fetch("/api/admin/serials/assign", {
        method:  "POST",
        headers: { "Content-Type": "application/json" },
        body:    JSON.stringify({
          serial:        serial.trim(),
          order_item_id: orderItemId,
        }),
      });
      const data = await res.json().catch(() => ({}));
      if (!res.ok) throw new Error(data?.error ?? `HTTP ${res.status}`);
      applyChange([
        ...linked,
        {
          serial:              data.serial.serial,
          parent_bar_serial:   data.serial.parent_bar_serial ?? null,
          manufacturer_serial: data.serial.manufacturer_serial ?? null,
        },
      ]);
      setSerial("");
    } catch (err) {
      setError((err as Error).message);
    } finally {
      setBusy(false);
    }
  }
 
  async function handleVoid(target: string) {
    if (busy) return;
    const reason = prompt(
      `Void NFC serial ${target}?\n\nReason (required, ≤200 chars):`,
    );
    if (!reason || !reason.trim()) return;
    setBusy(true);
    setError(null);
    try {
      const res = await fetch("/api/admin/serials/void", {
        method:  "POST",
        headers: { "Content-Type": "application/json" },
        body:    JSON.stringify({ serial: target, reason: reason.trim() }),
      });
      const data = await res.json().catch(() => ({}));
      if (!res.ok) throw new Error(data?.error ?? `HTTP ${res.status}`);
      applyChange(linked.filter((s) => s.serial !== target));
    } catch (err) {
      setError((err as Error).message);
    } finally {
      setBusy(false);
    }
  }
 
  return (
    <div className="flex flex-col gap-1.5 mt-2">
      {/* Section label + count */}
      <div className="flex items-center gap-2 flex-wrap">
        <span className="font-montserrat text-[10px] tracking-widest uppercase" style={{ color: MUTED }}>
          NFC cards
        </span>
        <span
          className="font-montserrat text-[10px] tracking-wider font-semibold"
          style={{ color: allLinked ? "var(--color-success)" : "var(--color-gold-mid)" }}
        >
          {linked.length}/{needed}
        </span>
        {allLinked && (
          <span className="inline-flex items-center gap-1 font-montserrat text-[10px] font-semibold" style={{ color: "var(--color-success)" }}>
            <ShieldCheck size={11} /> Ready to pack
          </span>
        )}
      </div>
 
      {/* Linked serials list — each row shows the identity chip, the regular
          actions (Print CoA + Open verify), then — set well apart as an
          EXCEPTION — the Void control. Print CoA is the primary action so it
          gets a larger icon + label; Void is destructive (one-way) so it's
          pushed to the far right behind a divider to avoid a mis-tap next to
          Print. For gift lines we render the bound message in italic below
          each row so the operator can verify the right physical card went
          into the right gift before sealing the box. */}
      {linked.length > 0 && (
        <div className="flex flex-col gap-1.5">
          {linked.map((s, idx) => {
            const msg = isGiftLine ? giftMessages[idx] : null;
            return (
              <div key={s.serial} className="flex flex-col gap-0.5">
                <div className="flex items-center gap-2 flex-wrap">
                  {/* Identity chip */}
                  <span
                    className="inline-flex items-center gap-1.5 font-mono text-xs px-2 py-0.5 rounded"
                    style={{
                      backgroundColor: "rgba(154,215,143,0.10)",
                      color:           "var(--color-success)",
                      border:          "1px solid rgba(154,215,143,0.3)",
                    }}
                  >
                    <ShieldCheck size={11} /> {s.serial}
                  </span>
 
                  {/* Counterpart, read-only: the serial the operator checks against
                      the bar in hand (serialCounterpart owns the rule). */}
                  {(() => {
                    const c = serialCounterpart(productType, s);
                    return c ? (
                      <span
                        className="font-mono text-[10px] px-1.5 py-0.5 rounded"
                        style={{ color: MUTED, border: `1px solid ${RULE}` }}
                        title={c.title}
                      >
                        {c.text}
                      </span>
                    ) : null;
                  })()}
 
                  {/* Regular actions — Print CoA is primary (larger icon +
                      label so it's unmistakable), Open verify is secondary. */}
                  <Link
                    href={`/verify/${s.serial}?print`}
                    target="_blank"
                    rel="noreferrer"
                    className="inline-flex items-center gap-1 font-montserrat text-[10px] font-semibold px-2 py-1 rounded border transition-opacity hover:opacity-80"
                    title="Print Certificate of Authenticity / Provenance (CoA)"
                    style={{ color: GOLD, borderColor: "var(--color-gold-dim)" }}
                  >
                    <Printer size={15} /> Print CoA
                  </Link>
                  <Link
                    href={`/verify/${s.serial}`}
                    target="_blank"
                    rel="noreferrer"
                    className="inline-flex items-center gap-1 opacity-70 hover:opacity-100 transition-opacity text-[10px]"
                    title="Open the verify page (what the recipient sees)"
                    style={{ color: MUTED }}
                  >
                    <ExternalLink size={13} /> Open
                  </Link>
 
                  {/* Exception action — set apart behind a divider, pushed to
                      the far right. Voiding marks the card unusable (one-way,
                      reason required) and frees this slot so a replacement card
                      can be scanned in (the "relink"). */}
                  <span className="ml-auto h-4 w-px self-center" style={{ backgroundColor: RULE }} aria-hidden />
                  <button
                    type="button"
                    onClick={() => handleVoid(s.serial)}
                    disabled={busy}
                    className="inline-flex items-center gap-1 opacity-60 hover:opacity-100 transition-opacity disabled:opacity-30 text-[10px] font-semibold px-1.5 py-1 rounded"
                    title="Void this card — marks it unusable (one-way, reason required) and frees the slot so you can scan a replacement card (relink)"
                    style={{ color: "#E05050", cursor: "pointer" }}
                  >
                    {busy ? <Loader2 size={13} className="animate-spin" /> : <ShieldOff size={13} />} Void
                  </button>
                </div>
                {msg && (
                  <p
                    className="font-cormorant italic text-[11px] leading-tight pl-2"
                    style={{ color: "var(--color-gold-mid)", maxWidth: 420 }}
                  >
                    <Gift size={9} className="inline mr-1" style={{ verticalAlign: "middle" }} />
                    &ldquo;{msg.trim()}&rdquo;
                  </p>
                )}
              </div>
            );
          })}
        </div>
      )}
 
      {/* Next-slot gift message preview — shows the operator which message
          will be bound when they scan the upcoming card. Helps catch a
          mis-sequenced batch before it's sealed. Only renders for gift
          lines that still have slots remaining. */}
      {nextMessage && (
        <div
          className="rounded px-2 py-1.5 mt-1"
          style={{
            backgroundColor: "rgba(201,168,76,0.06)",
            border:          "1px dashed var(--color-gold-dim)",
            maxWidth:        420,
          }}
        >
          <p className="font-montserrat text-[9px] uppercase tracking-[0.18em] mb-0.5" style={{ color: "var(--color-gold-mid)" }}>
            Next scan binds to message
          </p>
          <p className="font-cormorant italic text-[12px] leading-snug" style={{ color: TEXT }}>
            &ldquo;{nextMessage.trim()}&rdquo;
          </p>
        </div>
      )}
 
      {/* Input for the next card — only when slots remain */}
      {!allLinked && (
        <form onSubmit={handleAssign} className="flex flex-wrap items-center gap-2">
          <input
            type="text"
            value={serial}
            onChange={(e) => setSerial(e.target.value)}
            placeholder={productType === "standard" ? "Manufacturer serial" : "1G-XXXX-XXXX"}
            disabled={busy}
            autoComplete="off"
            spellCheck={false}
            className="rounded px-2 py-1 font-mono text-xs"
            style={{ backgroundColor: BG, border: `1px solid ${RULE}`, color: TEXT, width: productType === "standard" ? 180 : 140 }}
          />
          <button
            type="submit"
            disabled={busy || !serial.trim()}
            className="inline-flex items-center gap-1 font-montserrat text-[10px] font-semibold px-2 py-1 rounded transition-opacity hover:opacity-80 disabled:opacity-50"
            style={{ backgroundColor: GOLD, color: BG, cursor: "pointer" }}
          >
            {busy ? <Loader2 size={10} className="animate-spin" /> : <ShieldCheck size={10} />} Pack {linked.length + 1} of {needed}
          </button>
        </form>
      )}
 
      {error && (
        <span className="font-montserrat text-[10px]" style={{ color: "#E05050" }}>{error}</span>
      )}
    </div>
  );
}
