"use client";
 
/**
 * ComplianceDraftView — renders an STR / CTR draft (a jsonb object persisted on
 * compliance_cases and built by src/lib/compliance/reports.ts) as HUMAN-READABLE
 * labelled fields for the compliance officer to review, with a "Copy JSON" action
 * that copies the exact machine-readable draft for filing with STRO / the
 * Registrar of Regulated Dealers.
 *
 * This is what docs/compliance-triggers-redesign.md §5 specified ("the JSON
 * rendered as labelled fields") — the previous implementation dumped raw
 * JSON.stringify into a <pre>, which is not an operator-facing format.
 *
 * The renderer is field-agnostic: it walks the draft recursively so new fields
 * added to buildStrDraft / buildCtrDraft surface automatically, no schema pinning.
 */
 
const OFFWHITE = "var(--color-offwhite)";
const MUTED = "var(--color-muted)";
 
/** snake_case / camelCase key → "Title Case" label. */
export function humanizeKey(key: string): string {
  const spaced = key
    .replace(/_/g, " ")
    .replace(/([a-z0-9])([A-Z])/g, "$1 $2")
    .trim();
  // Uppercase known acronyms; Title Case the rest.
  return spaced
    .split(" ")
    .map((w) => {
      const up = w.toUpperCase();
      if (["STR", "CTR", "AML", "PEP", "KYC", "SGD", "ID"].includes(up)) return up;
      return w.charAt(0).toUpperCase() + w.slice(1);
    })
    .join(" ");
}
 
function isPlainObject(v: unknown): v is Record<string, unknown> {
  return typeof v === "object" && v !== null && !Array.isArray(v);
}
 
function ScalarValue({ value }: { value: unknown }) {
  const text = value === null || value === undefined || value === "" ? "—" : String(value);
  return (
    <span className="font-montserrat text-xs" style={{ color: OFFWHITE }}>
      {text}
    </span>
  );
}
 
/** Recursively render one draft node. `depth` drives left indentation. */
function DraftNode({ label, value, depth }: { label: string; value: unknown; depth: number }) {
  const indent = { paddingLeft: depth > 0 ? depth * 12 : 0 };
 
  // Array — bulleted list (of scalars) or nested nodes (of objects).
  if (Array.isArray(value)) {
    return (
      <div style={indent} className="py-0.5">
        <p className="font-montserrat text-[11px] font-semibold" style={{ color: MUTED }}>
          {label}
        </p>
        {value.length === 0 ? (
          <ScalarValue value="—" />
        ) : (
          <ul className="mt-0.5 space-y-0.5">
            {value.map((item, i) =>
              isPlainObject(item) ? (
                <li key={i} className="pl-3">
                  <DraftFields obj={item} depth={depth + 1} />
                </li>
              ) : (
                <li key={i} className="flex gap-1.5">
                  <span style={{ color: MUTED }}>·</span>
                  <ScalarValue value={item} />
                </li>
              ),
            )}
          </ul>
        )}
      </div>
    );
  }
 
  // Nested object — header + indented sub-fields.
  if (isPlainObject(value)) {
    return (
      <div style={indent} className="py-0.5">
        <p className="font-montserrat text-[11px] font-semibold" style={{ color: MUTED }}>
          {label}
        </p>
        <div className="mt-0.5">
          <DraftFields obj={value} depth={depth + 1} />
        </div>
      </div>
    );
  }
 
  // Scalar — "Label: value" on one line.
  return (
    <div style={indent} className="py-0.5 flex flex-wrap gap-x-2">
      <span className="font-montserrat text-[11px] font-semibold" style={{ color: MUTED }}>
        {label}:
      </span>
      <ScalarValue value={value} />
    </div>
  );
}
 
function DraftFields({ obj, depth }: { obj: Record<string, unknown>; depth: number }) {
  return (
    <div className="space-y-0.5">
      {Object.entries(obj).map(([k, v]) => (
        <DraftNode key={k} label={humanizeKey(k)} value={v} depth={depth} />
      ))}
    </div>
  );
}
 
export default function ComplianceDraftView({
  label,
  draft,
}: {
  label: string;
  draft: Record<string, unknown>;
}) {
  const json = JSON.stringify(draft, null, 2);
  return (
    <div
      className="rounded-lg p-3"
      style={{ backgroundColor: "rgba(180,40,40,0.06)", border: "1px solid rgba(229,115,115,0.25)" }}
    >
      <div className="flex items-center justify-between gap-2 mb-2">
        <p
          className="font-montserrat text-xs font-semibold tracking-wide uppercase"
          style={{ color: "#e57373" }}
        >
          {label}
        </p>
        <button
          onClick={() => {
            try {
              navigator.clipboard?.writeText(json);
            } catch {
              /* clipboard unavailable */
            }
          }}
          className="font-montserrat text-[11px] px-2.5 py-1 rounded transition-opacity hover:opacity-80 shrink-0"
          style={{ backgroundColor: "var(--color-rule)", color: OFFWHITE }}
        >
          Copy JSON
        </button>
      </div>
 
      <DraftFields obj={draft} depth={0} />
 
      <p className="font-montserrat text-[10px] mt-2" style={{ color: MUTED }}>
        Draft only — review, verify the grounds, and file with the relevant authority before submission.
        Use Copy JSON to copy the full structured draft for filing.
      </p>
    </div>
  );
}
