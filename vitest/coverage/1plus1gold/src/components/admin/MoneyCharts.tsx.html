/**
 * The two chart forms the money map uses, and nothing else.
 *
 * ── Why these two ────────────────────────────────────────────────────────────
 * A drill-down row set answers one of two questions: "which few items are most
 * of this" (Pareto) or "how did it move" (time). Anything else is decoration.
 *
 * ── Why they measure themselves ──────────────────────────────────────────────
 * An SVG viewBox scales its text with the box, so a fixed 1000-wide viewBox in a
 * 360px phone renders 10px labels at 3.6px and pushes ticks outside the frame.
 * These components measure their container and draw at 1:1 pixels, so a label is
 * the size it says it is on every screen and nothing can overflow.
 *
 * ── One axis, always ─────────────────────────────────────────────────────────
 * A textbook Pareto has two y-scales (value bars + a 0–100% line). Instead the
 * bars are drawn as SHARE of the total, so the cumulative curve rides the same
 * percentage axis. Same reading, one scale. Values stay in the labels/tooltip.
 *
 * Stacked vs grouped is not a style choice: series that PARTITION a total stack
 * (channels, vendors, products); series that NEST — sessions ⊃ viewed ⊃ carted,
 * sent ⊃ opened — must be grouped, or a subset is added to its own total.
 */

/** Container width, tracked live so a rotate or a sidebar toggle redraws. */
function useWidth<T extends HTMLElement>() {
  const ref = useRef<T | null>(null);
  const [w, setW] = useState(0);
  useLayoutEffect(() => {
    const el = ref.current;
    if (!el) return;
    const set = () => setW(el.clientWidth);
    set();
    if (typeof ResizeObserver === "undefined") {
      window.addEventListener("resize", set);
      return () => window.removeEventListener("resize", set);
    }
    const ro = new ResizeObserver(set);
    ro.observe(el);
    return () => ro.disconnect();
  }, []);
  return [ref, w] as const;
}

/** Rotated tick labels need room; how much depends on the longest one shown. */
function tickRoom(labels: string[], maxChars: number) {
  const longest = Math.min(maxChars, labels.reduce((a, s) => Math.max(a, s.length), 0));
  return Math.min(96, Math.round(longest * 4.6 * Math.sin(Math.PI / 4)) + 20);
}
 
/**
 * Pareto geometry — the PLOT BAND is the input and the frame is derived from it.
 *
 * ⚠️ It used to be the other way round (fixed 280px frame, minus ~14px of top
 * padding and up to 96px of rotated tick labels), which meant long category
 * names silently ATE the chart: the more there was to label, the less there was
 * to read, and everything past the top few bars collapsed into stubs you could
 * not rank by eye. Sizing the band means a bar's height is what it says it is,
 * whatever the labels do — the frame just grows to fit them.
 *
 * Pure so the guarantee is testable without a DOM.
 */
export function paretoGeometry(opts: {
  w: number; labels: string[]; maxChars: number; cats: number;
  band?: number; minBand?: number;
}) {
  const { w, labels, maxChars, cats, band, minBand = 170 } = opts;
  const PL = 34, PR = 8, PT = 14;
  const PB = tickRoom(labels, maxChars);
  const ih = Math.max(band ?? (w < 460 ? 240 : 310), minBand);
  const H = PT + ih + PB;
  const iw = Math.max(w - PL - PR, 10);
  return { PL, PR, PT, PB, ih, H, iw, bw: iw / Math.max(cats, 1) };
}

        {stack.map((s) => (
          <span key={s.n}>
            <i style={{ display: "inline-block", width: 9, height: 9, borderRadius: 2, background: s.c, marginRight: 5, verticalAlign: -1 }} />
            {s.n}
          </span>
        ))}
      </div>
    </div>
  );
}
 
 
/* ── the flow, drawn ────────────────────────────────────────────────────────
   The headline graphic from the 2026-09-02 mock: everything the business
   consumed on the left, everything it earned on the right, a spine between
   them, and ribbons whose thickness IS the amount. Both lenses use it — "which
   team spent it / which team earned it" and "what the money went on / where the
   money came from" are the same picture with different nodes.
 
   ⚠️ The net block STAYS in the income column here, unlike the stacked
   fallback. That is what makes the diagram balance: costs and income each sum
   to the same total precisely because the gap is carried as a block, so both
   ribbon stacks fill the spine exactly. Pulling it out — right for a pair of
   ranked lists, where "money in: Net loss" reads as nonsense — would leave the
   right-hand ribbons short of the spine by the size of the loss and the picture
   would simply be wrong.
 
   Drawn at MEASURED 1:1 pixels rather than in a fixed viewBox. A viewBox scales
   its text with the box, which is the bug that made the Paretos unreadable on a
   phone; the same trap applies here, and the admin content column is narrower
   than the window (there is a sidebar), so a window-width media query would
   also mis-fire. Below the readable floor the caller shows the stacked lists. */
export const SANKEY_MIN_WIDTH = 660;

  k: string;
  label: string;
  amount: number;
  kind?: string;
}
 
/** The height the drill has to fill, measured live so a rotate or a chart toggle
 *  redraws. Mirrors `useWidth` above; the drill needs the other dimension. */
/* ── the drill, as a Sankey ─────────────────────────────────────────────────
   Replaces the three-column drill (2026-09-03, Mukesh: "I thought I had given a
   go ahead for sankey visualization, but it shows me the column visualization
   for drill down"). Ported by reading the approved prototype, not recalling it.
 
   The shape he corrected the first cut toward: everything you came through STAYS
   ON SCREEN. Opening a node divides it in place and adds a level to the right;
   it does not replace the view or push the previous level into a breadcrumb.
 
   Bar height IS the amount, on one scale within each level. Labels live in the
   gutters and are pushed apart when they would collide, with a leader line back
   to the bar — that is what lets the height stay honest on skewed data instead
   of imposing a minimum bar height, which would break the value encoding. */
/* ── the money map itself, drillable in place ───────────────────────────────
   Costs left, income right, the business between. Clicking a node OPENS IT
   WHERE IT IS: a column appears growing away from the spine and everything you
   came through stays on screen. That is the prototype Mukesh approved, and the
   correction he gave the first cut of it — "instead of moving previous layer to
   the top breadcrumb, that layer is visible also".
 
   The earlier shipped version put the drill in a separate focus stage that
   REPLACED this diagram. That was a scope decision of mine, not a limitation,
   and it broke the one property he had asked for by name. */
export function SankeyMap({
  costs, income, headings, tint, money, pct,
  pathL, pathR, onDrill, activeSide,
}: {
  costs: MmBlock[]; income: MmBlock[];
  headings: { left: string; middle: string; right: string };
  tint: (kind: string | undefined, side: "L" | "R") => string;
  money: (n: number) => string;
  pct: (part: number, whole: number) => string;
  /** Open path per side, by node name. Level 0 is the block key. */
  pathL: string[]; pathR: string[];
  onDrill: (side: "L" | "R", depth: number, id: string) => void;
  activeSide: "L" | "R" | null;
}) {
  const [showAll, setShowAll] = useState<Record<string, boolean>>({});
  const [tip, setTip] = useState<Tip | null>(null);
  // ▶ Clear the tooltip whenever the diagram's SHAPE changes.
  //
  // Mukesh, 2026-09-04: *"when I shifted from one item to another, the labels
  // from the previous one stayed stuck and displayed."* A click re-renders the
  // marks under a pointer that never moved, so the element the tooltip belongs
  // to can be replaced without ever firing pointerleave — and the old node's
  // name sits over the new column until the reader happens to hover something
  // else. Nothing is wrong with the diagram; the label is a ghost of the last
  // one. Keyed on everything that changes which marks exist.
  useEffect(() => { setTip(null); },
    [pathL.join(">"), pathR.join(">"), JSON.stringify(showAll)]);
  const wrapRef = useRef<HTMLDivElement>(null);
 
  const toNode = (r: MmRow): DrillNode => ({ id: r.n, n: r.n, m: r.m, a: r.a, src: r, kids: (r.kids ?? []).map(toNode) });
  const blockNode = (b: MmBlock): DrillNode => ({
    id: b.k, n: b.label, m: b.note ? undefined : undefined, a: b.amount,
    kids: (b.kids ?? []).map(toNode), src: b,
    // The net/shortfall block is the balancing figure, not a revenue line, so it
    // holds the bottom of its column instead of ranking by size beside real
    // income. Both signs are pinned: whether the year nets to a profit or a
    // shortfall must not change where the reader looks for it.
    ...(b.kind === "loss" || b.kind === "profit" ? { pin: "last" as const } : {}),
  });
 
  const rootL = useMemo(() => costs.filter((b) => b.amount > 0).map(blockNode),
    [costs]);  // eslint-disable-line react-hooks/exhaustive-deps
  const rootR = useMemo(() => income.filter((b) => b.amount > 0).map(blockNode),
    [income]); // eslint-disable-line react-hooks/exhaustive-deps
 
  // Two passes: the first sizes the band from the busiest column, the second
  // lays the columns out INTO that band. Doing it in one pass would need the
  // band before knowing how many rows there are.
  const probe = (roots: DrillNode[], path: string[]) =>
    buildDrill({ rootRows: roots, path, showAll, band: 1000 }).levels;
  const band = mapHeight(probe(rootL, pathL), probe(rootR, pathR));
 
  const L = buildDrill({ rootRows: rootL, path: pathL, showAll, band });
  const R = buildDrill({ rootRows: rootR, path: pathR, showAll, band });
  const g = mapGeometry(L.levels.length, R.levels.length);
  const H = TOPP + band + BOTP;
 
  if (L.levels.length === 0 || R.levels.length === 0) return null;
 
  const ribbon = (x1: number, y1: number, x2: number, y2: number, h: number, fill: string, op: number, key: string) => (
    <path key={key} d={`M${x1},${y1} C${x1 + (x2 - x1) * .5},${y1} ${x2 - (x2 - x1) * .5},${y2} ${x2},${y2}`
                     + ` L${x2},${y2 + h} C${x2 - (x2 - x1) * .5},${y2 + h} ${x1 + (x2 - x1) * .5},${y1 + h} ${x1},${y1 + h} Z`}
          fill={fill} opacity={op} />
  );
 
  const ribbons: React.ReactNode[] = [];
  const marks: React.ReactNode[] = [];
 
  for (const side of ["L", "R"] as const) {
    const { levels } = side === "L" ? L : R;
    const path = side === "L" ? pathL : pathR;
    const xAt = side === "L" ? g.xL : g.xR;
 
    // level 0 → the spine
    let cur = TOPP;
    for (const b of levels[0]) {
      const fill = tint((b.nd.src as MmBlock | undefined)?.kind, side);
      const x1 = side === "L" ? xAt(0) + BARW : g.spineX + BARW;
      const x2 = side === "L" ? g.spineX : xAt(0);
      ribbons.push(side === "L"
        ? ribbon(x1, b.y, x2, cur, b.h, fill, 0.26, `${side}0-${b.nd.id}`)
        : ribbon(x1, cur, x2, b.y, b.h, fill, 0.26, `${side}0-${b.nd.id}`));
      cur += b.h;
    }
    // each deeper level → the node it opened from
    for (let d = 1; d < levels.length; d++) {
      const parent = levels[d - 1].find((b) => b.nd.id === (side === "L" ? L.openIds : R.openIds)[d - 1]);
      if (!parent) continue;
      // The children may now be TALLER than the parent bar (see openedBand): a
      // drilled level takes the room its rows need. So the parent-side ends of
      // the ribbons are scaled back into the parent bar — they converge into it
      // rather than overflowing past it, which is what makes it read as a
      // fan-out from that one flow instead of a broken diagram.
      const childTotal = levels[d].reduce((a, b) => a + b.h, 0) || 1;
      const k = parent.h / childTotal;
      let c = parent.y;
      const fill = side === "L" ? COST_C : GOOD_C;
      for (const b of levels[d]) {
        const x1 = side === "L" ? xAt(d) + BARW : xAt(d - 1) + BARW;
        const x2 = side === "L" ? xAt(d - 1) : xAt(d);
        ribbons.push(side === "L"
          ? ribbon(x1, b.y, x2, c, b.h * k, fill, 0.26, `${side}${d}-${b.nd.id}`)
          : ribbon(x1, c, x2, b.y, b.h * k, fill, 0.26, `${side}${d}-${b.nd.id}`));
        c += b.h * k;
      }
    }
 
    // bars + gutter labels
    levels.forEach((lvl, d) => {
      const x = xAt(d);
      const labelY = placeLabels(lvl, TOPP, band);
      const colTotal = lvl.reduce((a, b) => a + Math.abs(b.nd.a), 0) || 1;
      // Only the OUTERMOST column labels outward into the clean margin; the rest
      // label inward. Labelling every column outward put one column's text over
      // the next column's bars.
      const outermost = d === levels.length - 1;
      for (const b of lvl) {
        const nd = b.nd;
        const kind = (nd.src as MmBlock | undefined)?.kind;
        const fill = d === 0 ? tint(kind, side) : (side === "L" ? COST_C : GOOD_C);
        // Every open node is one the reader opened, so every open node closes.
        // Nothing opens itself any more (see buildDrill) — the pass-through rule
        // that did was exactly why "gold desk does not collapse".
        const canOpen = nd.more || !!nd.kids?.length;
        const on = path[d] === nd.id;
        const ly = labelY.get(b) ?? b.y + b.h / 2;
        const cy = b.y + b.h / 2;
        const moved = Math.abs(ly - cy) > 2.5;
        const left = side === "L" ? outermost : !outermost;
        const lx = left ? x - 10 : x + BARW + 10;
        marks.push(
          <g key={`m${side}${d}-${nd.id}`} role={canOpen ? "button" : undefined} tabIndex={canOpen ? 0 : undefined}
             aria-label={`${nd.n}, ${money(Math.abs(nd.a))}`}
             style={{ cursor: canOpen ? "pointer" : "default", outline: "none" }}
             onClick={() => { if (nd.more) setShowAll((s) => ({ ...s, [nd.moreKey!]: !s[nd.moreKey!] })); else onDrill(side, d, nd.id); }}
             onKeyDown={(e) => { if (e.key === "Enter" || e.key === " ") { e.preventDefault();
               if (nd.more) setShowAll((s) => ({ ...s, [nd.moreKey!]: !s[nd.moreKey!] })); else onDrill(side, d, nd.id); } }}
             onPointerEnter={(e) => {
               const r = wrapRef.current?.getBoundingClientRect();
               setTip({ x: e.clientX - (r?.left ?? 0) + (wrapRef.current?.scrollLeft ?? 0),
                        y: e.clientY - (r?.top ?? 0) + (wrapRef.current?.scrollTop ?? 0) - 12,
                        title: nd.n,
                        lines: [`${money(Math.abs(nd.a))}`, `${pct(Math.abs(nd.a), colTotal)} of this column`,
                                ...(nd.m ? [nd.m] : []), ...(canOpen ? ["click to open"] : [])] });
             }}
             onPointerLeave={() => setTip(null)}>
            <rect x={x} y={b.y} width={BARW} height={Math.max(b.h, 1.6)} rx={1.5}
                  fill={fill} opacity={canOpen ? 0.95 : 0.62}
                  stroke={on ? "#F3E9CE" : undefined} strokeWidth={on ? 1.5 : undefined} />
            {moved && <path d={`M${left ? x - 3 : x + BARW + 3},${cy} L${left ? x - 7 : x + BARW + 7},${ly}`}
                            stroke="#4A4028" strokeWidth={1} fill="none" />}
            {/* Sized to the room this label actually has — see labelHitBox. A
                fixed outer-gutter width here covered the NEIGHBOURING column's
                bars, and since marks paint level by level the deeper column won
                the click: drilled blocks would not collapse. */}
            {(() => { const hb = labelHitBox(x, { left, outermost }); return (
              <rect x={hb.x} y={Math.min(ly, cy) - 8} width={hb.w}
                    height={Math.abs(ly - cy) + 16} fill="transparent" />
            ); })()}
            {/* An OUTWARD label has the whole gutter, so it carries the amount.
                An INWARD one has only the channel between two columns — name
                only, clipped to fit, with the figure a hover away. Putting the
                amount on both ran "COGS — gold (landed) S$83,456.41" straight
                into the next column's bar. */}
            <text x={lx} y={ly + 3.8} textAnchor={left ? "end" : "start"}
                  style={{ font: `${on ? 600 : 400} 11.5px Montserrat, sans-serif`, fill: "#E8D5A3" }}>
              {clip(nd.n, outermost ? 28 : 15)}
              {outermost && (
                <tspan style={{ font: "400 10.5px Montserrat, sans-serif", fill: "#8A7440", fontVariantNumeric: "tabular-nums" }}>
                  {" "}{kind === "loss" ? "−" : ""}{money(Math.abs(nd.a))}
                </tspan>
              )}
            </text>
          </g>
        );
      }
      // Head each drilled column with the NAME of the node it opened from, taken
      // off the node itself. Reading it from the open path prints a block KEY at
      // depth 1 ("T_GOLD_DESK") — level 0's ids are keys, deeper ids are names.
      if (d > 0) {
        const from = levels[d - 1].find((bx) => bx.nd.id === (side === "L" ? L.openIds : R.openIds)[d - 1]);
        if (from) marks.push(
          <text key={`h${side}${d}`} x={x + BARW / 2} y={TOPP - 12} textAnchor="middle"
                style={{ font: "600 9.5px Montserrat, sans-serif", fill: "#8A7440", letterSpacing: ".13em" }}>
            {clip(from.nd.n, 22).toUpperCase()}
          </text>
        );
      }
    });
  }
 
  return (
    /* ── The frame scrolls, the page does not ────────────────────────────────
       Vertical scroll added 2026-09-04 alongside the taller rows (Mukesh:
       "vertical scroll and increase the section height"). Before this the SVG's
       height WAS the container height, so a deep drill stretched the page and
       pushed the detail rail far below the fold — the reason the canvas needed a
       1,200px cap in the first place, which then squeezed rows.
 
       Bounding the frame lets `drillHeight` grow honestly: a 47-row level now
       gets its full 40px per row and the surplus scrolls, instead of every row
       being shrunk to fit a ceiling. `maxHeight` (not `height`) so a shallow
       drill still collapses to its natural size rather than leaving dead space.
 
       `overscroll-contain` stops a scroll that reaches the end of the diagram
       from continuing into the page — the nested-scroll trap that makes a
       drilled map feel like it is fighting you. And the tooltip already offsets
       by `wrapRef.scrollTop`, so it was written for this. */
    <div ref={wrapRef} className="relative overflow-auto overscroll-contain"
         style={{ maxHeight: "min(72vh, 900px)" }} data-flow-shown="1">
      <svg width={g.W} height={H} role="img"
           aria-label="Flow of money out and money in, drillable on both sides">
        {/* Side captions get their own row above the per-column headings. At one
            shared row they overlapped, which is what the render showed. */}
        {/* Anchored to the canvas EDGES, not to the outermost column. Centring
            each caption over its own side puts the income one on top of "THE
            BUSINESS" whenever that side has a single column — which is the
            common case. The top of the label gutter is empty, so the edges are
            free real estate. */}
        <text x={PADX} y={15} textAnchor="start"
              style={{ font: "600 10px Montserrat, sans-serif", fill: "#8A7440", letterSpacing: ".16em" }}>
          {headings.left.toUpperCase()}
        </text>
        <text x={g.spineX + BARW / 2} y={15} textAnchor="middle"
              style={{ font: "600 10px Montserrat, sans-serif", fill: "#C9A84C", letterSpacing: ".16em" }}>
          {headings.middle.toUpperCase()}
        </text>
        <text x={g.W - PADX} y={15} textAnchor="end"
              style={{ font: "600 10px Montserrat, sans-serif", fill: "#8A7440", letterSpacing: ".16em" }}>
          {headings.right.toUpperCase()}
        </text>
        <rect x={g.spineX} y={TOPP} width={BARW} height={band} rx={3} fill="#C9A84C" opacity={0.16} />
        {ribbons}
        {marks}
      </svg>
      <Tooltip tip={tip} />
      {activeSide && null}
    </div>
  );
}
