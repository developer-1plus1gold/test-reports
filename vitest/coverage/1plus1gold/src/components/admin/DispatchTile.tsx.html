"use client";
 
/**
 * DispatchTile — shared shell for every dispatch-tab card.
 *
 * Before this component existed, each shipper section (Lalamove unbooked,
 * Lalamove active, NinjaVan pending, NinjaVan dispatched, Self-pickup)
 * re-implemented the same outer-card + expand-toggle + chevron + result
 * footer JSX inline. A change to the card chrome had to be made in
 * five places. Now: per-tile JSX boils down to "header summary,
 * shipper-specific actions, expanded body, optional result message".
 *
 * Packing-flow uniformity (Mukesh's 2026-05-13 principle):
 *   - Packing is identical across all shippers — single
 *     <PackingChecklistPanel> + <OrderDetail/> goes in `body`.
 *   - Shipping varies — per-shipper buttons go in `actions`.
 *
 * No data fetching here; this is pure layout. State for expand,
 * fetchOrderBars trigger, and onPacked still lives at the
 * admin/page.tsx level where it can be shared across tiles.
 */
 
import type { ReactNode } from "react";
import { MapPin } from "lucide-react";
 
interface CheckboxConfig {
  checked:   boolean;
  disabled:  boolean;
  title?:    string;
  onChange:  (checked: boolean) => void;
}
 
export interface DispatchTileProps {
  /** Order id — used as the React key by the caller; not stored here. */
  orderId:   string;
  /** Highlight the outer border with gold when the order is in a
   *  "selected" state (NinjaVan pending uses this for the checkbox
   *  selection visual). */
  selected?: boolean;
  /** Left-side checkbox — present only on NinjaVan pending today. */
  checkbox?: CheckboxConfig;
  /** Header content — typically OrderInfo + any inline status badges.
   *  Wrapped by the shell in the expand-toggle button + chevron. */
  header:    ReactNode;
  /** Right-side action buttons. Stacked vertically inside the header
   *  row when present (e.g. "Book with Lalamove", "Refresh status",
   *  "Mark Collected"). Pass null for tiles that have no per-row action
   *  (e.g. NinjaVan pending — actions live at the section level). */
  actions?:  ReactNode;
  /** Public carrier tracking URL for this order (stored share-link, or derived
   *  from the carrier + tracking number). When set, the tile shows a "Track"
   *  link in the action column — a single-window parcel lookup on every
   *  dispatch order (Lalamove / NinjaVan / SPX / marketplace), for when a
   *  customer inquires. Omitted/null → no Track link (order not yet booked). */
  trackUrl?: string | null;
  /** Whether the tile is expanded. Owned by the caller via
   *  expandedDispatch state so only one tile is open at a time. */
  isExp:     boolean;
  /** Fires when the user clicks the expand toggle. */
  onToggle:  () => void;
  /** Expanded body content — typically OrderDetail + the packing
   *  panel. Wrapped by the shell in padding + flex-col. Only rendered
   *  when isExp is true. */
  body?:     ReactNode;
  /** Optional result footer — booking/pickup status text. Rendered
   *  below the body in a consistent style across all tiles. */
  result?: { text: string; ok: boolean } | null;
}
 
export default function DispatchTile({
  selected,
  checkbox,
  header,
  actions,
  trackUrl,
  isExp,
  onToggle,
  body,
  result,
}: DispatchTileProps) {
  const borderColor = selected ? "var(--color-gold)" : "var(--color-rule)";
  const bgColor     = selected ? "rgba(201,168,76,0.04)" : "rgba(201,168,76,0.02)";
 
  // Track link + per-tile actions share one right-side column, so every dispatch
  // order gets a consistent "Track" affordance regardless of which buttons the
  // tile carries. Rendered only when there's something to show.
  const actionCol = (trackUrl || actions) ? (
    <div className="flex flex-col gap-2 shrink-0 w-full sm:w-auto">
      {trackUrl && (
        <a
          href={trackUrl}
          target="_blank"
          rel="noopener noreferrer"
          title="Track parcel"
          className="font-montserrat font-semibold text-xs px-3 py-2 rounded-lg inline-flex items-center justify-center gap-1.5 transition-opacity hover:opacity-80"
          style={{ border: "1px solid var(--color-gold)", color: "var(--color-gold)" }}
        >
          <MapPin size={13} /> Track
        </a>
      )}
      {actions}
    </div>
  ) : null;
 
  // When the checkbox is present, lay out the row as
  //   [checkbox] [expand-button + actions]
  // — same visual as the previous NV pending tile. Otherwise the row is
  //   [expand-button] [actions] with center alignment.
  const headerRow = checkbox ? (
    <div className="flex items-start gap-3 p-4">
      <input
        type="checkbox"
        checked={checkbox.checked}
        disabled={checkbox.disabled}
        title={checkbox.title}
        style={{
          accentColor: "var(--color-gold)",
          marginTop:   3,
          flexShrink:  0,
          cursor:      checkbox.disabled ? "not-allowed" : "pointer",
        }}
        onChange={(e) => checkbox.onChange(e.target.checked)}
      />
      <div className="flex-1 flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3 sm:gap-4 min-w-0">
        <ExpandToggle isExp={isExp} onToggle={onToggle} header={header} />
        {actionCol}
      </div>
    </div>
  ) : (
    // Stack header-above-actions on mobile (side-by-side overflows a phone once
    // a carrier's action column is wide); row + right-aligned actions on sm+.
    <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3 sm:gap-4 p-4">
      <ExpandToggle isExp={isExp} onToggle={onToggle} header={header} />
      {actionCol}
    </div>
  );
 
  return (
    <div className="rounded-xl overflow-hidden" style={{ border: `1px solid ${borderColor}`, backgroundColor: bgColor }}>
      {headerRow}
      {isExp && body && (
        <div className="px-4 pb-4 flex flex-col gap-3">{body}</div>
      )}
      {result?.text && (
        <p
          className="font-montserrat text-xs mx-4 mb-3 pt-2"
          style={{
            color:     result.ok ? "var(--color-success)" : "#E05050",
            borderTop: "1px solid var(--color-rule)",
          }}
        >
          {result.text}
        </p>
      )}
    </div>
  );
}
 
function ExpandToggle({ isExp, onToggle, header }: { isExp: boolean; onToggle: () => void; header: ReactNode }) {
  return (
    <button className="flex-1 min-w-0 text-left" onClick={onToggle}>
      <div className="flex items-start gap-2 min-w-0">
        {header}
        <svg
          width="12" height="12" viewBox="0 0 24 24"
          fill="none" stroke="currentColor" strokeWidth={2}
          className="shrink-0 mt-0.5"
          style={{
            color:      "var(--color-muted)",
            transform:  isExp ? "rotate(180deg)" : "rotate(0deg)",
            transition: "transform 0.2s",
          }}
        >
          <polyline points="6 9 12 15 18 9" />
        </svg>
      </div>
    </button>
  );
}
