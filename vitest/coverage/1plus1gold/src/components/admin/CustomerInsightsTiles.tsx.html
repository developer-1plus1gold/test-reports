"use client";
 
/**
 * Customer structure at the top of Admin → Customers (Mukesh 2026-11-21). Reads the
 * aggregates from GET /api/admin/users/insights (lib/admin/customer-insights holds every
 * definition) and renders them as tiles + two composition bars. `reloadKey` changes when the
 * list's Refresh is tapped, so both reload together.
 *
 * A tile with a segment is a button: tapping it hands the page the exact ids it counted, so the
 * list below shows those customers; tapping it again clears the filter. Customers → new this
 * month · Gold buyers · Repeat buyers · Active · 90 days → the lapsed · Top 10 % · Gold in our
 * care. (Avg spend has no list of its own; marketplace buyers are not accounts here.)
 */
import { useEffect, useState } from "react";
import Flag from "@/components/Flag";
import { useFlag, useFmt } from "@/store/currencyStore";
import type { CustomerInsights, CustomerSegmentKey, SignupMethod } from "@/lib/admin/customer-insights";
import { CompositionBar, InsightTile as Tile, pct } from "@/components/admin/InsightTile";
 
/** Re-exported: the bar is shared with the other insight panels (components/admin/InsightTile). */
export { CompositionBar };
 
const SIGNUP_LABEL: Record<SignupMethod, string> = { whatsapp: "WhatsApp", google: "Google", apple: "Apple", email: "Email", other: "Other" };
 
export interface CustomerSegment { key: CustomerSegmentKey; label: string; ids: string[] }
 
export default function CustomerInsightsTiles({ reloadKey, segment, onSegment }: {
  reloadKey: number;
  /** The segment the list is filtered to, if any. */
  segment?: CustomerSegment["key"] | null;
  onSegment?: (s: CustomerSegment | null) => void;
}) {
  const fmt = useFmt();
  const flagCode = useFlag();
  const [data, setData] = useState<CustomerInsights | null>(null);
  const [error, setError] = useState<string | null>(null);
 
  useEffect(() => {
    const ctrl = new AbortController();
    setError(null);
    fetch("/api/admin/users/insights", { cache: "no-store", signal: ctrl.signal })
      .then(async (r) => {
        const j = await r.json().catch(() => ({}));
        if (!r.ok) throw new Error(j.error ?? "Could not load customer insights");
        setData(j.insights as CustomerInsights);
      })
      .catch((e: Error) => { Eif (e.name !== "AbortError") setError(e.message); });
    return () => ctrl.abort();
  }, [reloadKey]);
 
  if (error) return <p className="font-montserrat text-xs" style={{ color: "#e57373" }}>{error}</p>;
  if (!data) return <div className="h-24 rounded-xl animate-pulse" style={{ backgroundColor: "rgba(201,168,76,0.04)" }} aria-label="Loading customer insights" />;
 
  const money = (v: number) => <span className="inline-flex items-center gap-1"><Flag code={flagCode} />{fmt(v)}</span>;
  const b = data.orderBands;
 
  /** Props that make a tile filter the list to its segment (nothing when it is empty). */
  const tap = (key: CustomerSegmentKey, label: string, verb: string) => {
    const ids = data.segments?.[key] ?? [];
    if (!onSegment || ids.length === 0) return {};
    const on = segment === key;
    return {
      active: on,
      actionHint: on ? "Showing · tap to clear" : `Tap to list ${verb}`,
      onClick: () => onSegment(on ? null : { key, label: `${ids.length} ${label}`, ids }),
    };
  };
 
  return (
    <div className="flex flex-col gap-3" data-testid="customer-insights">
      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
        <Tile label="Customers" value={data.customers}
          sub={`+${data.newThisMonth} this month · ${data.newLastMonth} last month`}
          {...tap("newThisMonth", "customers who signed up this month", "this month's sign-ups")} />
        <Tile label="Gold buyers" value={data.buyers}
          sub={`${pct(data.conversion)} of customers have bought gold`}
          {...tap("buyers", "gold buyers", "the buyers")} />
        <Tile label="Repeat buyers" value={data.repeatBuyers}
          sub={`${pct(data.repeatRate)} of buyers came back`}
          {...tap("repeat", "repeat buyers (2+ gold orders)", "repeat buyers")} />
        <Tile label="Active · 90 days" value={data.active90}
          sub={`${data.lapsed} lapsed (no gold order in 90 days)`}
          {...tap("lapsed", "lapsed customers (bought gold, none in 90 days)", "the lapsed")} />
        <Tile label="Avg spend per buyer" value={money(data.avgLifetimeSgd)}
          sub={<span className="inline-flex items-center gap-1">median {money(data.medianLifetimeSgd)}</span>} />
        <Tile label="Top 10 % of buyers" value={pct(data.top10Share)}
          sub="of our own (non-marketplace) revenue"
          {...tap("top10", "top buyers by spend (top 10 %)", "the top buyers")} />
        <Tile label="Gold in our care" value={data.holdingCustomers}
          sub={`customers · ${data.holdingGrams.toLocaleString("en-SG", { maximumFractionDigits: 2 })} g in vault + MAG`}
          {...tap("holding", "customers with gold in our care", "them")} />
        <Tile label="Marketplace buyers" value={data.marketplaceBuyers}
          sub={`${data.marketplaceOrders} paid orders via marketplaces & in person`} />
      </div>
      <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
        <CompositionBar title="Gold orders per buyer" parts={[
          { label: "1", value: b.one }, { label: "2", value: b.two },
          { label: "3–5", value: b.threeToFive }, { label: "6+", value: b.sixPlus },
        ]} />
        <CompositionBar title="How customers sign up" parts={
          (Object.keys(SIGNUP_LABEL) as SignupMethod[]).map((k) => ({ label: SIGNUP_LABEL[k], value: data.signupMix[k] }))
        } />
      </div>
    </div>
  );
}
