"use client";
 
/**
 * AdminModal — the one overlay+panel primitive for every admin dialog.
 *
 * WHY THIS EXISTS (2026-09-01, reported by Mukesh from daily PWA use):
 * the "Generate serial batch" dialog on /admin/serials could not be scrolled
 * to its Generate button on a phone — dragging scrolled the PAGE BEHIND
 * instead. An audit of every admin overlay found the same defect on 8 of them
 * and a partial version of it on all 22, because each one hand-rolled its own
 * `fixed inset-0` overlay. Four separate things were missing:
 *
 *   1. NO SCROLL ON THE PANEL. `fixed inset-0` clips the overlay to the
 *      viewport, but the panel itself had no `max-height` and no
 *      `overflow-y-auto` — so a panel taller than the screen simply ran off
 *      the bottom with no way to reach it. This is the reported bug.
 *   2. SCROLL CHAINING. Even the panels that DID scroll had no
 *      `overscroll-behavior`, so once the panel hit its end the gesture
 *      chained to the document and the page behind moved — the second half
 *      of the report ("it's the main screen that scrolls").
 *   3. `vh`, NOT `dvh`. `100vh` on iOS is the viewport with browser chrome
 *      COLLAPSED, so a `max-h-[90vh]` panel is taller than the space actually
 *      visible while the chrome is showing — the bottom of the panel sits
 *      under it. `dvh` tracks the live viewport. The `max-h-[90vh]` class is
 *      kept as the fallback: a browser without `dvh` support drops the inline
 *      declaration and the class applies.
 *   4. NO BODY LOCK. Nothing stopped the document scrolling underneath.
 *
 * A FIFTH cause was found on 2026-09-01, after the first four were fixed and
 * the vault audit dialog still would not scroll ON A PHONE IN PORTRAIT — while
 * scrolling fine in landscape. That split is the whole diagnosis:
 *
 *   5. THE KEYBOARD IS NOT PART OF `dvh`. When an input is focused, iOS shrinks
 *      the VISUAL viewport to the strip above the keyboard, but `vh` and `dvh`
 *      keep reporting the full layout viewport. So in portrait the panel is
 *      sized to a height that is mostly hidden behind the keyboard: its content
 *      fits inside `90dvh`, so it has NOTHING to scroll, and the submit button
 *      is simply behind the keyboard and unreachable. Rotate to landscape and
 *      the viewport is short enough that the content overflows for real, the
 *      panel scrolls, and the bug appears to vanish. Fixed by sizing the
 *      overlay from `window.visualViewport` — the one measurement that excludes
 *      the keyboard — so the panel's max-height is 90% of what is ACTUALLY
 *      visible and overflow (and therefore scrolling) engages when it should.
 *
 * The component owns POSITIONING, the BACKDROP, the SCROLL CONTRACT and
 * DISMISSAL. It deliberately does NOT own the panel's look — callers keep
 * their own width / padding / border / background, so migrating an existing
 * dialog is visually byte-identical apart from the scroll fix.
 *
 * Scope note: this is for DIALOGS. The marketing media lightbox and the
 * AdminSidebar nav drawer are also `fixed inset-0` but are neither dialogs
 * nor scroll containers, and are correctly not built on this.
 */
 
import { useEffect, useState, type CSSProperties, type ReactNode } from "react";
 
/** Panels taller than this scroll internally instead of overflowing the screen. */
const MAX_PANEL_HEIGHT = "90dvh";
 
export interface AdminModalProps {
  open: boolean;
  /**
   * Dismiss handler for backdrop-click and Escape. Omit (or pass undefined)
   * for a modal that must be resolved by an explicit button — that is how a
   * caller expresses "not dismissible", including the common
   * `onClose={busy ? undefined : close}` guard that stops a dialog being
   * dismissed mid-save.
   *
   * `undefined` suppresses BOTH dismissal paths — Escape and backdrop-click —
   * because both read this one prop. Spelled out because the two are wired in
   * different places (a keydown listener and the overlay's onClick), so it is
   * not obvious from either alone that guarding the prop covers both.
   */
  onClose?: () => void;
  /** Panel width. Default matches the prevailing admin dialog width. */
  maxWidthClass?: string;
  /**
   * "sheet" (default) — bottom sheet on mobile, centred card on sm+.
   * "center"          — always centred (short confirms, non-dismissible warnings).
   */
  align?: "sheet" | "center";
  /** Extra overlay classes — a few dialogs inset the sheet on sm+ (`sm:p-4`). */
  overlayClassName?: string;
  /**
   * Panel classes: padding, radius, shadow — the caller's own look.
   *
   * ⚠️ MUST NOT contain `overflow-*` or `max-h-*`. Those are the scroll
   * contract this component exists to guarantee, and a conflicting utility here
   * would not reliably lose: both classes end up in the class list and the
   * generated stylesheet's order decides, not the order of this string. That is
   * precisely the regression AdminModal was introduced to stop, so it is
   * enforced by `admin-modal-drift.test.ts` rather than left to this comment.
   */
  panelClassName?: string;
  /** Panel inline style: background, border — kept caller-side so looks don't change. */
  panelStyle?: CSSProperties;
  /** Backdrop colour. Default is the prevailing admin scrim. */
  backdrop?: string;
  /** Stacking order. Default 50; AdminDialog sits above other modals at 100. */
  zIndex?: number;
  children: ReactNode;
}
 
export default function AdminModal({
  open,
  onClose,
  maxWidthClass = "sm:max-w-md",
  align = "sheet",
  overlayClassName = "",
  panelClassName = "",
  panelStyle,
  backdrop = "rgba(0,0,0,0.6)",
  zIndex = 50,
  children,
}: AdminModalProps) {
  // Lock the document while a modal is open so a gesture that escapes the
  // panel cannot move the page behind it. Restores the caller's own overflow
  // rather than assuming "" — nested modals then unwind correctly.
  useEffect(() => {
    if (!open) return;
    const previous = document.body.style.overflow;
    document.body.style.overflow = "hidden";
    return () => { document.body.style.overflow = previous; };
  }, [open]);
 
  // Track the VISUAL viewport (see cause 5 above). On iOS this is the area
  // above the on-screen keyboard; `vh`/`dvh` are not. Absent (older browsers,
  // jsdom) we fall back to the `dvh` sizing below, which is correct whenever no
  // keyboard is up.
  const [visual, setVisual] = useState<{ height: number; offsetTop: number } | null>(null);
  useEffect(() => {
    if (!open) return;
    const vv = typeof window !== "undefined" ? window.visualViewport : null;
    if (!vv) return;
    const sync = () => setVisual({ height: vv.height, offsetTop: vv.offsetTop });
    sync();
    vv.addEventListener("resize", sync);
    vv.addEventListener("scroll", sync);
    return () => {
      vv.removeEventListener("resize", sync);
      vv.removeEventListener("scroll", sync);
      setVisual(null);
    };
  }, [open]);
 
  // Escape closes — only when the modal is actually dismissible.
  useEffect(() => {
    if (!open || !onClose) return;
    const onKey = (e: KeyboardEvent) => { if (e.key === "Escape") onClose(); };
    window.addEventListener("keydown", onKey);
    return () => window.removeEventListener("keydown", onKey);
  }, [open, onClose]);
 
  if (!open) return null;
 
  const alignClass =
    align === "center"
      ? "items-center justify-center p-4"
      : "items-end sm:items-center justify-center";
 
  return (
    <div
      className={`fixed inset-0 flex ${alignClass}${overlayClassName ? " " + overlayClassName : ""}`}
      style={{
        backgroundColor: backdrop,
        zIndex,
        // Safe-area inset lives on the OVERLAY, not the panel. Putting it on
        // the panel means an inline paddingBottom that overrides the caller's
        // own `p-5`/`p-6` class and silently strips the dialog's bottom
        // padding. Here it just floats the sheet above the iPhone home
        // indicator; the backdrop still paints the full screen because
        // padding does not reduce the painted background box.
        paddingBottom: align === "sheet" ? "env(safe-area-inset-bottom, 0px)" : undefined,
        // Pin the overlay to the visible strip rather than the layout viewport,
        // so a sheet sits above the keyboard instead of behind it and the panel
        // below can size itself against something real. `bottom: auto` is
        // required: `inset-0` sets both edges, and top+bottom+height together
        // would otherwise over-constrain and ignore the height.
        ...(visual ? { top: visual.offsetTop, bottom: "auto", height: visual.height } : null),
      }}
      onClick={onClose}
      role="presentation"
    >
      <div
        role="dialog"
        aria-modal="true"
        onClick={(e) => e.stopPropagation()}
        className={[
          "w-full",
          maxWidthClass,
          align === "center" ? "rounded-2xl" : "rounded-t-2xl sm:rounded-2xl",
          // The fix. max-h-[90vh] is the no-`dvh` fallback for the inline
          // maxHeight below; overscroll-contain stops the gesture chaining to
          // the document once the panel reaches its end.
          "max-h-[90vh] overflow-y-auto overscroll-contain",
          panelClassName,
        ].filter(Boolean).join(" ")}
        // 90% resolves against the overlay's measured height (the visible
        // strip). Without a measurement it falls back to dvh, and the
        // `max-h-[90vh]` class backs that up where dvh is unsupported.
        style={{ maxHeight: visual ? "90%" : MAX_PANEL_HEIGHT, ...panelStyle }}
      >
        {children}
      </div>
    </div>
  );
}
