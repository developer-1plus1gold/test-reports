/**
 * Which payment gateways a caller can actually complete a payment on.
 *
 * iOS ≤1.0.10 only allows `*.hit-pay.com` as a payment redirect host, so a
 * Razorpay method shown to it would open a page the app refuses to load.
 * From 1.0.11 the app sends `x-gateways: hitpay,razorpay`; until it does, a
 * native caller must NOT see `razorpay_*`.
 *
 * ⚠️ "Native" is NOT only `User-Agent: 1plus1gold-ios/…`. The ≤1.0.10 payment
 * picker and "we accept" strip build a plain URLRequest without our UA, so
 * they go out with URLSession's DEFAULT agent — `<App>/<build> CFNetwork/<v>
 * Darwin/<v>` (found by the 1.0.11 audit, 2026-10-29). Matching only our own
 * prefix would have shown Razorpay to every installed app the moment the
 * methods were activated, and every such payment would have failed. A web
 * browser never sends a CFNetwork agent, so web always sees every method.
 */
 
const NATIVE_UA = /^1plus1gold-ios\b|\bCFNetwork\/[\d.]+ Darwin\//;
 
/** True for requests from the native iOS app (our UA, or URLSession's default). */
export function isNativeAppUserAgent(ua: string | null | undefined): boolean {
  return !!ua && NATIVE_UA.test(ua);
}
 
export function hidesRazorpay(req?: Pick<Request, "headers">): boolean {
  if (!req) return false;
  if (!isNativeAppUserAgent(req.headers.get("user-agent"))) return false;
  const gateways = (req.headers.get("x-gateways") ?? "").toLowerCase();
  return !gateways.split(/[,\s]+/).includes("razorpay");
}
 
/** True for any method completed through a hosted gateway page (HitPay or Razorpay). */
export function isOnlineGatewayMethod(methodKey: string | null | undefined): boolean {
  return !!methodKey && (methodKey.startsWith("hitpay_") || methodKey.startsWith("razorpay_"));
}
 
/** True when a method's limit (null = none) is exceeded by the order total. */
export function overMethodLimit(maxTxnSgd: number | null | undefined, totalSgd: number): boolean {
  return maxTxnSgd != null && Number.isFinite(maxTxnSgd) && totalSgd > maxTxnSgd;
}
