/**
 * Card networks a card payment method can accept — the registry.
 *
 * WHICH networks a method accepts is configuration, not code: it is the
 * `payment_gateway_fees.card_networks` column, edited in Admin → Payment fees
 * (migration 20261029d). This file only knows how each network LOOKS and how a
 * gateway SPELLS it, so turning a network off (e.g. Amex on Razorpay, whose
 * Amex rate is too high) is an admin edit with no deploy.
 *
 * One setting drives three things, so they cannot disagree:
 *   1. the card marks shown in checkout and the "we accept" strip;
 *   2. the networks Razorpay's hosted page is told to hide (checkoutOptionsFor);
 *   3. the post-payment check: a payment made on a network the method does not
 *      accept alerts an admin (the hosted page is Razorpay's, not ours, so the
 *      page restriction is best-effort and the account-level switch at Razorpay
 *      is the real block).
 *
 * Adding a network = one row here + its web SVG + its iOS PNG (kept in step by
 * payment-logo-parity.test.ts) + the key in the DB CHECK.
 */
 
export interface CardNetwork {
  key: string;
  label: string;
  /** Web mark under /public/images/payment (Iconify `logos` set, CC0). */
  webFile: string;
  /** Chip height in px — the marks' aspect ratios differ wildly. */
  h: number;
  /** Bundled iOS asset name (Assets.xcassets). */
  iosAsset: string;
  /**
   * How Razorpay names this network — on `card.network` of a payment, and in
   * the checkout display config. Compared after normalising (lowercase, no
   * spaces/punctuation), and every known spelling is listed because Razorpay's
   * docs use more than one (e.g. "Amex" and "American Express").
   */
  razorpayNames: readonly string[];
}
 
export const CARD_NETWORKS: readonly CardNetwork[] = [
  { key: "visa",       label: "Visa",       webFile: "visa",       h: 14, iosAsset: "PayLogoVisa",       razorpayNames: ["Visa"] },
  { key: "mastercard", label: "Mastercard", webFile: "mastercard", h: 20, iosAsset: "PayLogoMastercard", razorpayNames: ["MasterCard"] },
  { key: "amex",       label: "Amex",       webFile: "amex",       h: 18, iosAsset: "PayLogoAmex",       razorpayNames: ["Amex", "American Express"] },
  { key: "unionpay",   label: "UnionPay",   webFile: "unionpay",   h: 18, iosAsset: "PayLogoUnionpay",   razorpayNames: ["UnionPay", "Union Pay"] },
];
 
export const CARD_NETWORK_KEYS: readonly string[] = CARD_NETWORKS.map((n) => n.key);
 
const BY_KEY = new Map(CARD_NETWORKS.map((n) => [n.key, n]));
 
/** A method that takes cards and so can carry a network list. */
export const isCardMethod = (methodKey: string) => methodKey.endsWith("_card");
 
/** Networks in registry order; unknown keys dropped. `null` stays `null` (= not configured). */
export function cardNetworksFrom(raw: unknown): CardNetwork[] | null {
  if (!Array.isArray(raw)) return null;
  const want = new Set(raw.map(String));
  return CARD_NETWORKS.filter((n) => want.has(n.key));
}
 
/** Validate an admin edit: an array of known keys (deduped, registry order), or null to clear. */
export function validateCardNetworks(input: unknown):
  | { ok: true; value: string[] | null }
  | { ok: false; error: string } {
  if (input === null) return { ok: true, value: null };
  if (!Array.isArray(input)) return { ok: false, error: "card_networks must be an array of network keys, or null" };
  for (const k of input) {
    if (typeof k !== "string" || !BY_KEY.has(k)) {
      return { ok: false, error: `unknown card network: ${String(k).slice(0, 20)} (known: ${CARD_NETWORK_KEYS.join(", ")})` };
    }
  }
  if (input.length === 0) return { ok: false, error: "card_networks cannot be empty — deactivate the method instead" };
  return { ok: true, value: CARD_NETWORKS.filter((n) => input.includes(n.key)).map((n) => n.key) };
}
 
const norm = (s: string) => s.toLowerCase().replace(/[^a-z0-9]/g, "");
 
/** Our key for a Razorpay `card.network` value; null when we don't recognise it. */
export function networkKeyFromRazorpay(name: string | null | undefined): string | null {
  if (!name) return null;
  const n = norm(name);
  for (const net of CARD_NETWORKS) if (net.razorpayNames.some((r) => norm(r) === n)) return net.key;
  return null;
}
 
/** Razorpay spellings of every known network the method does NOT accept (for the page's hide list). */
export function razorpayHiddenNetworks(accepted: readonly string[]): string[] {
  return CARD_NETWORKS.filter((n) => !accepted.includes(n.key)).flatMap((n) => [...n.razorpayNames]);
}
 
/**
 * Post-payment check: why a paid card network falls outside what the method
 * accepts, or null when it is fine. An unrecognised spelling is reported too,
 * so a new Razorpay network name surfaces once instead of slipping through.
 * No list configured, or no network reported, is never a finding.
 */
export function cardNetworkOutsideConfig(accepted: readonly string[] | null, paidNetwork: string | null | undefined): string | null {
  if (!accepted || !paidNetwork) return null;
  const key = networkKeyFromRazorpay(paidNetwork);
  if (!key) return `an unrecognised card network "${paidNetwork.slice(0, 40)}"`;
  if (accepted.includes(key)) return null;
  return `${BY_KEY.get(key)!.label}, which this method is set not to accept`;
}
