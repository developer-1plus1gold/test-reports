/**
 * Shared media MIME helpers for direct-to-storage uploads.
 *
 * iOS/Safari frequently hand us a File with an empty or odd `type` (especially
 * for `.mov`). When a signed PUT is sent with no content-type, the object is
 * stored as `text/plain` — which then won't play in a <video>/<img> tag, and a
 * bucket with `allowed_mime_types` set would reject the upload outright. So we
 * resolve a trustworthy MIME from the file extension.
 *
 * Extracted from the marketing uploader so the serial-CoA video uploader shares
 * one source of truth (no drift between the two flows).
 */
 
export const EXT_MIME: Record<string, string> = {
  mp4: "video/mp4", m4v: "video/mp4", mov: "video/quicktime", webm: "video/webm",
  jpg: "image/jpeg", jpeg: "image/jpeg", png: "image/png", gif: "image/gif", webp: "image/webp",
};
 
/** Video containers we accept for a CoA authenticity clip. */
export const VIDEO_MIMES = new Set<string>(["video/mp4", "video/quicktime", "video/webm"]);
 
/**
 * Best-effort MIME for a file: trust a recognised `type`, else derive from the
 * extension, else fall back to the raw type or a generic binary type.
 * Accepts anything with a `name` (+ optional `type`) so it works for both a
 * browser `File` and a `{ name, type, size }` upload descriptor.
 */
export function safeMime(file: { name: string; type?: string }): string {
  const allowed = new Set(Object.values(EXT_MIME));
  if (file.type && allowed.has(file.type)) return file.type;
  const ext = (file.name.split(".").pop() ?? "").toLowerCase();
  return EXT_MIME[ext] ?? (file.type || "application/octet-stream");
}
 
/** Canonical storage extension for a known video MIME. */
export function extForVideoMime(mime: string): string {
  return mime === "video/webm" ? "webm" : mime === "video/quicktime" ? "mov" : "mp4";
}
