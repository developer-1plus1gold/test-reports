/**
 * Home-page carousel (offers / specials / highlights) — shared read + link safety.
 *
 * Single source for both surfaces:
 *   • Web  — `src/app/page.tsx` (server component) calls fetchLiveCarouselSlides()
 *            and passes the slides to the client <HomeCarousel/>.
 *   • iOS  — `homePageSpec()` calls it and embeds the slides in the SDUI spec's
 *            `data.carouselSlides`; the native media_carousel renders them.
 *
 * A slide is LIVE when active AND now() is within [starts_at, ends_at] (either
 * bound may be null = unbounded). The window is filtered in JS — the table is a
 * handful of rows, so this is cheaper + more robust than PostgREST or-strings on
 * timestamps. Ordering: sort_order ASC, then created_at ASC.
 *
 * The table is service-role only (RLS USING(false)); this module uses the
 * service key, mirroring lib/finance and the payments-methods route.
 */
import { createClient } from "@supabase/supabase-js";
import {
  getCarouselCache,
  setCarouselCache,
  type CachedCarouselRow,
} from "@/lib/carousel-cache";
 
export interface CarouselSlide {
  id: string;
  image_url: string;
  alt_text: string;
  /** Sanitised destination: an internal "/path" or an absolute https URL. null = non-tappable. */
  link_target: string | null;
  /** Admin-facing label; the visible text lives in the banner art. */
  headline: string | null;
}
 
const supabaseAdmin = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_KEY!,
);
 
/** True if the string contains any ASCII control char (newlines, NUL, etc.). */
function hasControlChar(s: string): boolean {
  for (let i = 0; i < s.length; i++) {
    const c = s.charCodeAt(i);
    if (c < 0x20 || c === 0x7f) return true;
  }
  return false;
}
 
/**
 * Hosts that are OURS. An absolute link to one of these is INTERNAL, and must be
 * collapsed to a path — because the native carousel routes purely on the string:
 * `SDUIMediaCarousel.actionForLink` sends anything starting `https://` to
 * `.openURL`, with NO same-host check, which EJECTS the reader out to Safari.
 * A relative path instead routes `.navigate` -> InternalLinkRouter and opens in
 * place. Web is indifferent (both forms work), so this diverges silently and is
 * invisible on the surface most people test on.
 *
 * Observed live 2026-09-16: a home-carousel slide pointing at our own SME500
 * newsroom post was stored absolute and kicked native readers out of the app.
 *
 * NEXT_PUBLIC_SITE_URL is included so dev/preview hosts normalise too — on dev,
 * an operator pasting a prod URL means "our newsroom post", not "leave the app".
 */
const OWN_HOSTS: ReadonlySet<string> = new Set(
  [
    "1plus1gold.com",
    "www.1plus1gold.com",
    "dev.1plus1gold.com",
    (() => {
      try {
        return new URL(process.env.NEXT_PUBLIC_SITE_URL ?? "").hostname;
      } catch {
        return "";
      }
    })(),
  ]
    .filter(Boolean)
    .map((h) => h.toLowerCase()),
);
 
/**
 * Alt text for a slide, for a screen reader.
 *
 * A TAPPABLE slide with no alt text is an unlabelled control — the reader is
 * told there is a button but not where it goes. So a linked slide falls back to
 * the admin headline, then to a generic label; a slide with NO link is
 * decorative and correctly stays "" (an empty alt is how you mark decoration —
 * inventing prose for it would be noise, not help).
 */
export function carouselAltText(
  altText: string | null | undefined,
  headline: string | null | undefined,
  linkTarget: string | null,
): string {
  const alt = (altText ?? "").trim();
  if (alt) return alt;
  if (!linkTarget) return "";
  const head = (headline ?? "").trim();
  return head || "Promotional banner";
}
 
/**
 * Validate a slide's link destination. Returns a safe href or null.
 *  - Internal: must start with a single "/" (not "//", which is protocol-relative
 *    and escapes the origin) and contain no control chars.
 *  - External: must parse as an absolute https URL (http/js/data/etc. rejected).
 * Used at write time (admin API) AND at render time (defence in depth — a tampered
 * row can never emit a javascript:/http: link to either surface).
 */
export function sanitizeCarouselLink(raw: string | null | undefined): string | null {
  if (!raw) return null;
  const v = raw.trim();
  if (!v || hasControlChar(v)) return null;
  if (v.startsWith("/")) {
    // Reject protocol-relative ("//evil.com") and backslash tricks ("/\evil").
    if (v.startsWith("//") || v.startsWith("/\\")) return null;
    return v;
  }
  try {
    const u = new URL(v);
    if (u.protocol !== "https:") return null;
    if (OWN_HOSTS.has(u.hostname.toLowerCase())) {
      // Our own site written absolutely -> collapse to a path so the native
      // carousel opens it in place instead of ejecting to Safari.
      const path = `${u.pathname}${u.search}${u.hash}`;
      // A pathname of "//x" would be protocol-relative once emitted as an
      // internal link, which the branch above rejects on purpose. Refuse rather
      // than guess at what was meant.
      if (!path.startsWith("/") || path.startsWith("/\\") || path.startsWith("//")) return null;
      return path;
    }
    return u.toString();
  } catch {
    return null;
  }
}
 
function isLive(
  row: { active: boolean; starts_at: string | null; ends_at: string | null },
  nowMs: number,
): boolean {
  if (!row.active) return false;
  if (row.starts_at && Date.parse(row.starts_at) > nowMs) return false;
  if (row.ends_at && Date.parse(row.ends_at) < nowMs) return false;
  return true;
}
 
/**
 * Live slides for the home carousel, ordered. Returns [] on any error (the
 * carousel then renders nothing).
 *
 * ▶ CACHED at L1 since 2026-09-17 — this was the ONE uncached Supabase read on
 * the home page's critical path, and the home page is `ƒ` dynamic so it has no
 * CDN layer to hide behind. See `carousel-cache.ts` for the measurements and
 * for why the cache stores the RAW ROWS rather than the filtered output (the
 * [starts_at, ends_at] window must stay exact to the second, so it is still
 * evaluated on every call). Admin writes call `invalidateCarouselCache()`, so
 * the TTL is a self-heal backstop, not the freshness mechanism.
 *
 * A DB error is NOT cached — a transient failure would otherwise pin an empty
 * carousel for the whole TTL (the same reasoning as the catalogue's empty-guard,
 * #13 2026-06-12).
 */
export async function fetchLiveCarouselSlides(): Promise<CarouselSlide[]> {
  let rows = getCarouselCache();
 
  if (!rows) {
    const { data, error } = await supabaseAdmin
      .from("home_carousel_slides")
      .select("id, image_url, alt_text, link_target, headline, active, starts_at, ends_at, sort_order, created_at")
      .eq("active", true)
      .order("sort_order", { ascending: true })
      .order("created_at", { ascending: true });
 
    if (error) {
      console.error("[home-carousel] fetch failed:", error.message);
      return [];
    }
 
    rows = (data ?? []) as unknown as CachedCarouselRow[];
    setCarouselCache(rows);
  }
 
  const nowMs = Date.now();
  return rows
    .filter((r) => isLive(r, nowMs))
    .map((r) => ({
      id: r.id,
      image_url: r.image_url,
      alt_text: carouselAltText(r.alt_text, r.headline, sanitizeCarouselLink(r.link_target)),
      link_target: sanitizeCarouselLink(r.link_target),
      headline: r.headline ?? null,
    }));
}
