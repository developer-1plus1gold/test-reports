/**
 * Home-page carousel (offers / specials / highlights) — shared read + link safety.
 *
 * Single source for both surfaces:
 *   • Web  — `src/app/page.tsx` (server component) calls fetchLiveCarouselSlides()
 *            and passes the slides to the client <HomeCarousel/>.
 *   • iOS  — `homePageSpec()` calls it and embeds the slides in the SDUI spec's
 *            `data.carouselSlides`; the native media_carousel renders them.
 *
 * A slide is LIVE when active AND now() is within [starts_at, ends_at] (either
 * bound may be null = unbounded). The window is filtered in JS — the table is a
 * handful of rows, so this is cheaper + more robust than PostgREST or-strings on
 * timestamps. Ordering: sort_order ASC, then created_at ASC.
 *
 * The table is service-role only (RLS USING(false)); this module uses the
 * service key, mirroring lib/finance and the payments-methods route.
 */
import { createClient } from "@supabase/supabase-js";
 
export interface CarouselSlide {
  id: string;
  image_url: string;
  alt_text: string;
  /** Sanitised destination: an internal "/path" or an absolute https URL. null = non-tappable. */
  link_target: string | null;
  /** Admin-facing label; the visible text lives in the banner art. */
  headline: string | null;
}
 
const supabaseAdmin = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_KEY!,
);
 
/** True if the string contains any ASCII control char (newlines, NUL, etc.). */
function hasControlChar(s: string): boolean {
  for (let i = 0; i < s.length; i++) {
    const c = s.charCodeAt(i);
    if (c < 0x20 || c === 0x7f) return true;
  }
  return false;
}
 
/**
 * Validate a slide's link destination. Returns a safe href or null.
 *  - Internal: must start with a single "/" (not "//", which is protocol-relative
 *    and escapes the origin) and contain no control chars.
 *  - External: must parse as an absolute https URL (http/js/data/etc. rejected).
 * Used at write time (admin API) AND at render time (defence in depth — a tampered
 * row can never emit a javascript:/http: link to either surface).
 */
export function sanitizeCarouselLink(raw: string | null | undefined): string | null {
  if (!raw) return null;
  const v = raw.trim();
  if (!v || hasControlChar(v)) return null;
  if (v.startsWith("/")) {
    // Reject protocol-relative ("//evil.com") and backslash tricks ("/\evil").
    if (v.startsWith("//") || v.startsWith("/\\")) return null;
    return v;
  }
  try {
    const u = new URL(v);
    return u.protocol === "https:" ? u.toString() : null;
  } catch {
    return null;
  }
}
 
function isLive(
  row: { active: boolean; starts_at: string | null; ends_at: string | null },
  nowMs: number,
): boolean {
  if (!row.active) return false;
  if (row.starts_at && Date.parse(row.starts_at) > nowMs) return false;
  if (row.ends_at && Date.parse(row.ends_at) < nowMs) return false;
  return true;
}
 
/** Live slides for the home carousel, ordered. Returns [] on any error (the carousel then renders nothing). */
export async function fetchLiveCarouselSlides(): Promise<CarouselSlide[]> {
  const { data, error } = await supabaseAdmin
    .from("home_carousel_slides")
    .select("id, image_url, alt_text, link_target, headline, active, starts_at, ends_at, sort_order, created_at")
    .eq("active", true)
    .order("sort_order", { ascending: true })
    .order("created_at", { ascending: true });
 
  if (error) {
    console.error("[home-carousel] fetch failed:", error.message);
    return [];
  }
 
  const nowMs = Date.now();
  return (data ?? [])
    .filter((r) => isLive(r as never, nowMs))
    .map((r) => ({
      id: r.id as string,
      image_url: r.image_url as string,
      alt_text: (r.alt_text as string) ?? "",
      link_target: sanitizeCarouselLink(r.link_target as string | null),
      headline: (r.headline as string | null) ?? null,
    }));
}
