/**
 * Which product stands in which case — the one pure function the scene code
 * depends on (docs/showroom-design.md §6.1).
 *
 * Catalogue + cases in, resolved cases + unplaced list out. Rules resolve in
 * HERE, so the rule/hand-picked distinction never leaks into the Three code
 * and the whole thing is unit-testable against a fixture.
 *
 * ▶ R4 — NOTHING GOES MISSING. Every shop-visible product appears somewhere.
 * A product in no case lands in a default "New arrivals" case and is listed in
 * the admin's "Unplaced" banner. This is tested against a fixture that ADDS a
 * product, because "a new product silently vanishes from the showroom" is
 * precisely the kind of failure that looks like nothing at all.
 *
 * ▶ WHY A RULE-BACKED CASE IS THE SAFER DEFAULT. A hand-picked case goes stale
 * the moment a product is added — that product falls into Unplaced and someone
 * has to notice. A `refinery=PAMP` case picks up the next PAMP bar the day it
 * is created, with no tap. Both kinds stay supported because some cases are
 * genuinely curatorial, but the editor should offer the rule first.
 */
 
export const RULE_KINDS = ["refinery", "product_type", "curve", "all"] as const;
export type ShowroomRuleKind = (typeof RULE_KINDS)[number];
 
export function isRuleKind(v: unknown): v is ShowroomRuleKind {
  return typeof v === "string" && (RULE_KINDS as readonly string[]).includes(v);
}
 
/** A case row as stored. */
export interface ShowroomCaseRow {
  id: string;
  name: string;
  rule_kind: string | null;
  rule_value: string | null;
  pos_x_mm: number;
  pos_z_mm: number;
  rotation_deg: number;
  width_mm: number;
  enabled: boolean;
  sort_order: number;
}
 
/** The minimum a product must carry to be placed. Generic so callers can pass
 *  their own richer row through without a second mapping step. */
export interface PlaceableProduct {
  id: string;
  product_type: "curve" | "standard" | "accessory";
  refinery: string | null;
}
 
export interface ResolvedCase<T extends PlaceableProduct> {
  id: string;
  name: string;
  posXMm: number;
  posZMm: number;
  rotationDeg: number;
  widthMm: number;
  products: T[];
  /**
   * True when the case holds more pieces than read as countable.
   *
   * ▶ This is a LAYOUT CONSTRAINT, not a styling preference. The brandkit
   * measured it: "Bars in a row need a real gap. At 15 bars a fan fused into a
   * single gold slab at 132px and only read as fifteen objects at 88px." A
   * case laid out to fit the most pieces reads as one gold shape rather than
   * as N pieces — which is the opposite of what a showroom is for. A
   * rule-backed case that grows past this needs to reflow or spill to a second
   * case, NOT tighten.
   */
  crowded: boolean;
}
 
export interface ShowroomLayout<T extends PlaceableProduct> {
  cases: ResolvedCase<T>[];
  /** Shop-visible products that matched no case and were swept into the
   *  default case. Surfaced in admin as "Unplaced (n)". */
  unplaced: T[];
}
 
/** Above this a row of pieces stops reading as countable objects. */
export const MAX_PIECES_PER_CASE = 8;
 
export const DEFAULT_CASE_NAME = "New arrivals";
 
function matchesRule(p: PlaceableProduct, kind: string | null, value: string | null): boolean {
  switch (kind) {
    case "all":
      return true;
    case "curve":
      return p.product_type === "curve";
    case "product_type":
      return !!value && p.product_type === value;
    case "refinery":
      // Refinery is operator-entered free text ("PAMP", "Puregold-Singapore"),
      // so compare case-insensitively and trimmed rather than requiring the
      // rule to be typed byte-identically to the product row.
      return !!value && (p.refinery ?? "").trim().toLowerCase() === value.trim().toLowerCase();
    default:
      return false;
  }
}
 
/**
 * Resolve the room.
 *
 * Order of claim, and each step is deliberate:
 *   1. HAND-PICKED cases claim first. An operator who explicitly put a product
 *      in a curated case means it, and a rule elsewhere must not steal it.
 *   2. RULE cases claim the remainder, in `sort_order`. First match wins, so a
 *      product that satisfies two rules appears ONCE — the same physical piece
 *      standing in two cases would be a lie about the room.
 *   3. Whatever is left goes to the default case and is reported as unplaced.
 */

    rotationDeg: c.rotation_deg,

}
