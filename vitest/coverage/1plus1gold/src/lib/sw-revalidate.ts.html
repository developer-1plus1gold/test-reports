/**
 * Tiny stale-while-revalidate cache for client-side fetches.
 *
 * Two-tier cache:
 *   - L1: module-level Map (fastest, in-memory). Survives SPA navigation
 *     within a tab; resets on full page reload.
 *   - L2: sessionStorage (mirror of L1 entries, keyed by `swr:<url>`).
 *     Survives reloads within the same tab, so the first paint after a
 *     Capacitor / PWA cold-start can render cached data immediately
 *     instead of showing the spinner while the network round-trips.
 *
 * Used to keep heavy account dashboards (Tree, Gold Yield, Yield
 * Simulator, Referrals) from re-fetching every time the customer
 * switches tabs and comes back. Cleared on sign-out / customer-switch.
 *
 * Usage:
 *   const data = swrGetFresh<MyShape>(url);
 *   if (data) setState(data);                      // instant render
 *   const fresh = await fetch(url).then(r => r.json());
 *   swrSet(url, fresh); setState(fresh);
 */

/** Read the cached value if it exists, regardless of age. */
export function swrGet<T>(key: string): T | null {
  const inMemory = cache.get(key);
  if (inMemory) return inMemory.value as T;

  // faster within the same tab.

}

/**
 * Fetch helper that always returns cached if fresh, else makes a real
 * fetch and stores the result. Single round-trip, no background
 * revalidation — use the lower-level helpers when you want
 * "show stale + revalidate in background".
 */
export async function swrFetchOrCached<T>(
  url: string,
  opts: { ttlMs?: number; init?: RequestInit } = {},
): Promise<T> {
  const ttl = opts.ttlMs ?? 60_000;
  const cached = swrGetFresh<T>(url, ttl);
  if (cached) return cached;
  const res = await fetch(url, opts.init);
  if (!res.ok) throw new Error(`fetch ${url} failed: ${res.status}`);
  const value = (await res.json()) as T;
  swrSet(url, value);
  return value;
}
