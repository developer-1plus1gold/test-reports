/**
 * SDUI spec for the Account → Preferences surface (Q3).
 *
 * Anchored to the PWA renderPreferences() (src/app/account/page.tsx:3112-3231)
 * — six cards in order: Notification channel matrix, Newsletter, Push,
 * Timezone, Price Display Unit, Gold Price Currencies.
 *
 * First consumer of the §15 settings-form primitives (option_select /
 * bool_toggle / toggle_chip / push_opt_in). Reads come from ONE aggregator
 * (data_source = /api/account/preferences/overview); each control PATCHes its
 * own endpoint and chains `refetch` so the aggregator reloads.
 *
 * PORT-DIFF vs PWA (Aspect | PWA | iOS SDUI | Contract):
 * | Notif card title       | "Notification Preferences"            | text                                  | T (reuse)  |
 * | "Always sent" box      | bg + rule border, long verbatim copy  | card{bg,stroke} + text                | T/card     |
 * | Channel chip (Email)   | Mail icon + "Email", active gold      | toggle_chip{icon:envelope.fill}       | TG1 NEW    |
 * | Channel chip (WhatsApp)| Bell icon + "WhatsApp" (PWA uses Bell) | toggle_chip{icon:bell.fill}           | TG1 NEW (NOTE: PWA uses Bell glyph for WA) |
 * | chip disabled gating   | !hasEmail / !hasWhatsapp              | disabled_when data.has_email==false   | TG1        |
 * | Newsletter intro link  | <a href=/account/newsletter> inline   | rich_text gold segment (tap not wired)| P1 (NOTE: inline-link tap deferred) |
 * | WA ping toggle         | checkbox (whatsapp && !unsubscribed)  | bool_toggle on_when + body_on/off     | SW1 NEW    |
 * | paused blurb date      | toLocaleDateString en-SG              | rich_text date segment (d_mon_y)      | P6 + rich_text date NEW |
 * | Reading language       | <select> SUPPORTED_LANGUAGES          | option_select style:menu              | PK1 NEW    |
 * | Push                   | Web PushManager / iOS bridge          | push_opt_in → PushOptInCard           | EH-push NEW|
 * | Timezone               | <select> 14 zones                     | option_select style:menu              | PK1        |
 * | Price unit             | 3 buttons gram/oz/kg                   | option_select style:segmented         | SG1 NEW    |
 * | Gold Price Currencies  | 19 flex-wrap chips + Select All/Reset | toggle_chip grid (4-col) + buttons    | TG1 (NOTE: 4-col grid vs flex-wrap) |
 */
 
import type { SDUISpec, SDUINode, SDUIAction } from "./types";
import {
  PREF_TIMEZONES, PRICE_UNITS, ALL_PREF_CURRENCIES, CURRENCY_FLAGS, DEFAULT_RESET_CURRENCIES,
} from "@/lib/types/preferences";
import { SUPPORTED_LANGUAGES } from "@/lib/ai/agents/newsletter-translator";
 
const PREFS_EP   = "/api/account/preferences";
const NOTIF_EP   = "/api/account/notification-preferences";
const NL_EP      = "/api/account/newsletter/preferences";
 
// ── tiny builders ───────────────────────────────────────────────────────────
const text = (content: string, font: string, color: string, extra: Record<string, unknown> = {}): SDUINode => ({
  type: "text", props: { content, font, color, ...extra },
});
 
/** Standard preference card: cinzel title (16/base) + optional muted desc + body. */
function prefCard(title: string, desc: string | null, children: SDUINode[]): SDUINode {
  const head: SDUINode[] = [text(title, "cinzel.semibold:16", "gold", { padding_b: 4 })];
  if (desc) head.push(text(desc, "montserrat:12", "muted", { padding_b: 8 }));
  return {
    type: "card",
    props: { padding: 24, corner_radius: 12, bg: "bg_card", spacing: 8, alignment: "leading" },
    children: [...head, ...children],
  };
}
 
// ── Card 1: Notification channel matrix ──────────────────────────────────────
function channelChip(opts: {
  bind: string; category: string; channel: "email" | "whatsapp"; label: string; icon: string; disabledWhen: string;
}): SDUINode {
  return {
    type: "toggle_chip",
    props: {
      bind: opts.bind, value: opts.channel, label: opts.label, icon: opts.icon,
      value_path: `data.notifications.${opts.category}.channels`,
      endpoint: NOTIF_EP, array_field: "channels", base_body: { category: opts.category },
      disabled_when: opts.disabledWhen,
    },
  };
}
 
function categoryRow(opts: { bind: string; category: string; title: string; desc: string }): SDUINode {
  return {
    type: "hstack",
    props: { spacing: 12, alignment: "top" },
    children: [
      {
        type: "vstack",
        props: { spacing: 2, alignment: "leading", max_width: "infinity" },
        children: [
          text(opts.title, "montserrat.semibold:14", "offwhite"),
          text(opts.desc, "montserrat:11", "muted"),
        ],
      },
      {
        type: "hstack",
        props: { spacing: 6 },
        children: [
          channelChip({ bind: opts.bind, category: opts.category, channel: "email",
            label: "Email", icon: "envelope.fill", disabledWhen: "data.has_email == false" }),
          channelChip({ bind: opts.bind, category: opts.category, channel: "whatsapp",
            label: "WhatsApp", icon: "bell.fill", disabledWhen: "data.has_whatsapp == false" }),
        ],
      },
    ],
  };
}

    {
      type: "card",

      children: [
        text("Always sent", "montserrat.semibold:12", "gold_mid"),
        text(
          "Order confirmations, payment receipts, delivery updates, identity verification (KYC), security alerts (2FA changes, account changes), and self-limit changes. Sent to your WhatsApp and email (whichever you have on file).",
          "montserrat:11", "muted",
        ),
      ],
    },
    text("Optional updates", "montserrat.semibold:12", "gold_mid", { padding_t: 8 }),
    // No-channel notice — only when the user has neither email nor WhatsApp.
    {
      type: "text",
      visible_when: "data.has_email == false && data.has_whatsapp == false",
      props: {
        content: "Add an email or WhatsApp number to your profile to receive optional notifications.",

// ── Card 2: Newsletter ────────────────────────────────────────────────────────
function newsletterCard(): SDUINode {
  return prefCard("Newsletter Preferences", null, [
    // Intro — tappable, routes in-app to the newsletter (InternalLinkRouter
    // maps /account/newsletter → .webView(.newsletterList), N1). The PWA links
    // only the path text; SwiftUI rich_text concatenates to one Text so a
    // sub-range can't carry its own tap — we make the whole intro the tap
    // target (the gold path segment signals it). Styling stays 12/muted exact.
    {
      type: "button",
      props: {
        action: { kind: "navigate", url: "/account/newsletter" },
        bg: "transparent", full_width: false, padding_h: 0, padding_v: 0, corner_radius: 0,
      },
      children: [{
        type: "rich_text",
        props: {
          font: "montserrat:12", color: "muted",
          segments: [
            { kind: "text", content: "Your monthly companion at " },
            { kind: "text", content: "/account/newsletter", color: "gold" },
            { kind: "text", content: " — your accumulation, the gold market last month, and a short note from the founder." },
          ],
        },
      }],
    },
    // WhatsApp ping toggle row.
    {
      type: "hstack",
      props: { spacing: 12, alignment: "top", padding_t: 8 },
      children: [
        {
          type: "vstack",
          props: { spacing: 2, alignment: "leading", max_width: "infinity" },
          children: [
            text("Get a WhatsApp ping when a new issue lands", "montserrat.semibold:14", "offwhite"),
            // Subscribed blurb.
            {
              type: "text",
              visible_when: "data.newsletter.unsubscribed_at == null",
              props: {
                content: "One message per month with a deep link. Past issues stay readable on this page either way.",
                font: "montserrat:11", color: "muted",
              },
            },
            // Paused blurb with inline formatted date.
            {
              type: "rich_text",
              visible_when: "data.newsletter.unsubscribed_at != null",
              props: {
                font: "montserrat:11", color: "muted",
                segments: [
                  { kind: "text", content: "Currently off (paused " },
                  { kind: "date", value: "{data.newsletter.unsubscribed_at}", style: "d_mon_y" },
                  { kind: "text", content: "). Past issues stay readable on this page." },
                ],
              },
            },
          ],
        },
        {
          type: "bool_toggle",
          props: {
            bind: "nl_whatsapp",
            on_when: "data.newsletter.whatsapp_notification_enabled == true && data.newsletter.unsubscribed_at == null",
            endpoint: NL_EP,
            body_on:  { whatsapp_notification_enabled: true,  unsubscribed: false },
            body_off: { whatsapp_notification_enabled: false, unsubscribed: true },
          },
        },
      ],
    },
    { type: "divider", props: { color: "rule" } },
    // Default reading language.
    text("Default Reading language", "montserrat.semibold:14", "offwhite", { padding_t: 4 }),
    text("New issues open in this language. You can flip per-issue from the article header. Translation runs once per language and is cached.",
      "montserrat:11", "muted", { padding_b: 4 }),
    {
      type: "option_select",
      props: {
        style: "menu", bind: "pref_language", value_path: "data.newsletter.preferred_language",
        endpoint: NL_EP, field: "preferred_language",
        options: Object.entries(SUPPORTED_LANGUAGES).map(([value, label]) => ({ value, label })),
      },
    },
  ]);
}
 
// ── Card 6: Gold Price Currencies ─────────────────────────────────────────────
function chunk<T>(arr: readonly T[], size: number): T[][] {
  const out: T[][] = [];
  for (let i = 0; i < arr.length; i += size) out.push(arr.slice(i, i + size) as T[]);
  return out;
}
 
function currencyCard(): SDUINode {
  const rows: SDUINode[] = chunk(ALL_PREF_CURRENCIES, 4).map((row) => ({
    type: "hstack",
    props: { spacing: 6 },
    children: row.map((code): SDUINode => ({
      type: "toggle_chip",
      props: {
        bind: "pref_currencies", value: code, label: code, flag: CURRENCY_FLAGS[code],
        value_path: "data.preferences.preferred_currencies",
        endpoint: PREFS_EP, array_field: "preferred_currencies", min_one: true,
      },
    })),
  }));
 
  const currencyApi = (codes: readonly string[]): SDUIAction => ({
    kind: "composed",
    actions: [
      { kind: "state_set", key: "pref_currencies", value: [...codes] },
      {
        kind: "api_call", method: "PATCH", url: PREFS_EP,
        body: { preferred_currencies: [...codes] },
        on_success: [{ kind: "refetch" }],
        on_failure: [{ kind: "state_set", key: "pref_currencies", value: null }, { kind: "refetch" }],
      },
    ],
  });
 
  return prefCard(
    "Gold Price Currencies",
    "Choose which currencies are selected by default in the Gold Prices chart. Changes you make directly in the chart are saved here automatically.",
    [
      { type: "vstack", props: { spacing: 6, alignment: "leading" }, children: rows },
      {
        type: "hstack",
        props: { spacing: 12, padding_t: 8 },
        children: [
          { type: "button", props: {
            label_text: "Select All", label_font: "montserrat:12", label_color: "muted",
            bg: "transparent", stroke: "rule", full_width: false, padding_h: 12, padding_v: 6, corner_radius: 6,
            action: currencyApi(ALL_PREF_CURRENCIES) } },
          { type: "button", props: {
            label_text: "Reset to Defaults", label_font: "montserrat:12", label_color: "muted",
            bg: "transparent", stroke: "rule", full_width: false, padding_h: 12, padding_v: 6, corner_radius: 6,
            action: currencyApi(DEFAULT_RESET_CURRENCIES) } },
        ],
      },
    ],
  );
}
 
export function preferencesSpec(): SDUISpec {
  return {
    surface: "preferences",
    spec_version: 1,
    data_source: "/api/account/preferences/overview",
    // Optimistic bind keys (nc_marketing, nc_gifts, nl_whatsapp, pref_language,
    // pref_timezone, pref_price_unit, pref_currencies) are intentionally NOT
    // declared: undeclared keys read as null, so each control falls back to the
    // authoritative `data` value until the user changes it. See §15.
    state_schema: {},
    root: {
      type: "scroll",
      children: [
        {
          type: "vstack",
          props: { spacing: 16, padding: 16, alignment: "leading" },
          children: [
            loadErrorBox(),
            notificationCard(),
            newsletterCard(),
            { type: "push_opt_in", props: {} },
            // Timezone
            prefCard("Timezone", "Used for displaying dates and times throughout the app.", [{
              type: "option_select",
              props: {
                style: "menu", bind: "pref_timezone", value_path: "data.preferences.timezone",
                endpoint: PREFS_EP, field: "timezone",
                options: PREF_TIMEZONES.map(({ value, label }) => ({ value, label })),
              },
            }]),
            // Price Display Unit
            prefCard("Price Display Unit", "Sets the primary unit shown in the live ticker and gold prices table.", [{
              type: "option_select",
              props: {
                style: "segmented", bind: "pref_price_unit", value_path: "data.preferences.price_unit",
                endpoint: PREFS_EP, field: "price_unit",
                options: PRICE_UNITS.map(({ value, label }) => ({ value, label })),
              },
            }]),
            currencyCard(),
          ],
        },
      ],
    },
  };
}
 
/** Surface a load failure for the aggregator (D1 _error), mirroring addresses. */
function loadErrorBox(): SDUINode {
  return {
    type: "text",
    visible_when: 'data._error != null && data._error != ""',
    props: { content: "Couldn't load your preferences. Pull to retry.", font: "montserrat:12", color: "muted" },
  };
}
