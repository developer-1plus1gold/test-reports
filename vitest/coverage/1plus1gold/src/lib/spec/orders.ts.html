const text = (content: string, font: string, color: string, extra: Record<string, unknown> = {}): SDUINode => {
  // `visible_when` is a TOP-LEVEL node field, NOT a prop — pull it out of
  // `extra` so the renderer's gate sees it (else the node renders always; this
  // was the fee-label + "Sent!" double-render device bug 2026-06-02).
  const { visible_when, ...rest } = extra as { visible_when?: string } & Record<string, unknown>;
  const node: SDUINode = { type: "text", props: { content, font, color, ...rest } };
  if (visible_when) node.visible_when = visible_when as string;
  return node;
};

// ── Per-page limit selector (5 / 10 / 20 / All) ─────────────────────────────
function limitSelector(): SDUINode {
  const chip = (label: string, n: number, activeWhen: string): SDUINode => ({
    type: "button",
    props: {
      label_text: label, label_font: "montserrat:12", full_width: false,
      label_color_when: activeWhen, label_color: "bg", label_color_else: "muted",
      bg_when: activeWhen, bg: "gold", bg_else: "transparent",
      stroke_when: activeWhen, stroke: "gold", stroke_else: "rule",
      corner_radius: 4, padding_h: 8, padding_v: 2,
      action: { kind: "state_set", key: "orders_limit", value: n },
    },
  });
  return {
    type: "hstack",
    props: { spacing: 4 },
    children: [
      chip("5", 5, "state.orders_limit == 5"),
      chip("10", 10, "state.orders_limit == 10"),
      chip("20", 20, "state.orders_limit == 20"),
      // "All" — active when the limit covers the whole list.
      {
        type: "button",
        props: {
          label_text: "All", label_font: "montserrat:12", full_width: false,
          label_color_when: "state.orders_limit >= data.orders.length", label_color: "bg", label_color_else: "muted",
          bg_when: "state.orders_limit >= data.orders.length", bg: "gold", bg_else: "transparent",
          stroke_when: "state.orders_limit >= data.orders.length", stroke: "gold", stroke_else: "rule",
          corner_radius: 4, padding_h: 8, padding_v: 2,
          action: { kind: "state_set", key: "orders_limit", value: ALL_LIMIT },
        },
      },
    ],
  };
}
 
// ── One order item row (nested repeater over item.items) ────────────────────
function orderItemRow(): SDUINode {
  // Enrol the bar (eligible + unenrolled + order enrollable). Optimistic
  // confirm via a shared transient id; refetch reflects enrolled state.
  const enrolAction: SDUIAction = {
    kind: "api_call", method: "POST", url: "/api/gold-yield/enrol",
    body: { order_id: "{item.orderId}", order_item_id: "{item.id}" },
    on_success: [
      { kind: "state_set", key: "enrol_msg_id", value: "{item.id}", revert_after_ms: 2500, revert_to: "" },
      { kind: "refetch" },
    ],
  };
  return {
    type: "vstack",
    props: { spacing: 6, alignment: "leading", max_width: "infinity" },
    children: [
      {
        type: "hstack",
        props: { spacing: 12, alignment: "top" },
        children: [
          {
            type: "hstack",
            props: { spacing: 6, alignment: "first_text_baseline", max_width: "infinity" },
            children: [
              text("{item.label} × {item.quantity}", "montserrat:12", "offwhite"),
              text("— {item.branchName}", "montserrat:12", "muted", { visible_when: "item.branchName" }),
              badge("● Gold Yield", "hex:D4AF37", "hex:D4AF37@12", "item.gyEligible == true && item.gyEnrolled == true"),
            ],
          },
          // Price / Free + premium, right-aligned.
          {
            type: "vstack",
            props: { spacing: 2, alignment: "trailing" },
            children: [
              text("Free", "montserrat:12", "gold", { visible_when: "item.isFree == true", multiline: false }),
              { type: "currency_amount", visible_when: "item.isFree != true",
                props: { sgd_value: "{item.lineSgd}", font: "montserrat:12", color: "muted", show_sgd_base: true, flag_height: 10 } },

}

          // Message: given rows always; received rows only as history (once passed on).

// ── Gift-a-card actions (Flow A + B) — ONE block, used by per-order owned
//    cards AND still-owned received cards (mirrors the web GiftCardActions).
//    `item` = the card/gift row (item.serial, item.giftMessage, item.pending).
function giftCardActions(): SDUINode {
  const openWhen = "state.gift_open_serial == item.serial";
  const busyThis = "state.gift_busy_serial == item.serial";
  const link     = "https://1plus1gold.com/verify/{item.serial}";
 
  const openForm = (withMsg: boolean): SDUIAction => ({
    kind: "composed", actions: [
      { kind: "state_set", key: "gift_open_serial",     value: "{item.serial}" },
      { kind: "state_set", key: "gift_draft_message",   value: withMsg ? "{item.giftMessage}" : "" },
      { kind: "state_set", key: "gift_draft_recipient", value: "" },
    ],
  });
  const closeThenRefetch = [
    { kind: "state_set", key: "gift_open_serial", value: "" } as SDUIAction,
    { kind: "state_set", key: "gift_busy_serial", value: "" } as SDUIAction,
    { kind: "refetch" } as SDUIAction,
  ];
  const saveMessage: SDUIAction = {
    kind: "api_call", method: "POST", url: "/api/account/cards/gift-message",
    body: { serial: "{item.serial}", message: "{state.gift_draft_message}" },
    set_state_on_request: { gift_busy_serial: "{item.serial}" },
    on_success: closeThenRefetch,
    on_failure: [{ kind: "state_set", key: "gift_busy_serial", value: "" }],
  };
  const sendGift: SDUIAction = {
    kind: "api_call", method: "POST", url: "/api/account/cards/send-gift",
    body: { serial: "{item.serial}", recipient: "{state.gift_draft_recipient}", message: "{state.gift_draft_message}" },
    set_state_on_request: { gift_busy_serial: "{item.serial}" },
    on_success: closeThenRefetch,
    on_failure: [{ kind: "state_set", key: "gift_busy_serial", value: "" }],
  };
  const cancelGift: SDUIAction = {
    kind: "api_call", method: "POST", url: "/api/account/cards/revoke-gift",
    body: { serial: "{item.serial}" },
    set_state_on_request: { gift_busy_serial: "{item.serial}" },
    on_success: [{ kind: "state_set", key: "gift_busy_serial", value: "" }, { kind: "refetch" }],
    on_failure: [{ kind: "state_set", key: "gift_busy_serial", value: "" }],
  };
  const copyLink: SDUIAction = {
    kind: "composed", actions: [
      { kind: "copy_to_clipboard", value: link },
      { kind: "state_set", key: "gift_copied_serial", value: "{item.serial}", revert_after_ms: 1500, revert_to: "" },
    ],
  };
  const giftBtn = (label: string, action: SDUIAction, when?: string, extra: Record<string, unknown> = {}): SDUINode => ({
    type: "button", ...(when ? { visible_when: when } : {}),
    props: { label_text: label, label_font: "montserrat.semibold:11", label_color: "bg", bg: "gold",
      corner_radius: 4, padding_h: 12, padding_v: 4, full_width: false, ...extra, action },
  });
 
  return {
    type: "vstack",
    props: { spacing: 6, alignment: "leading", max_width: "infinity", padding_top: 6 },
    children: [
      // ── Pending remote gift — reserved, awaiting claim ──
      { type: "vstack", visible_when: "item.pending != null", props: { spacing: 6, alignment: "leading" }, children: [
        { type: "rich_text", props: { font: "montserrat:11", color: "gold_mid", segments: [
          { kind: "text", content: "⏳ Reserved for ", color: "gold_mid" },
          { kind: "text", content: "{item.pending.reservedDisplay}", font: "montserrat.semibold:11", color: "offwhite" },
          { kind: "text", content: " · awaiting claim", color: "gold_mid" },
        ] } },
        { type: "hstack", props: { spacing: 8, alignment: "center" }, children: [
          { type: "button", visible_when: "state.gift_copied_serial != item.serial",
            props: { label_text: "Copy link", label_font: "montserrat.semibold:11", label_color: "gold", bg: "transparent", stroke: "gold_dim", corner_radius: 4, padding_h: 8, padding_v: 4, full_width: false, action: copyLink } },
          text("Copied ✓", "montserrat.semibold:11", "success", { visible_when: "state.gift_copied_serial == item.serial", multiline: false }),
          { type: "button", props: { label_text: "Cancel gift", label_font: "montserrat:11", label_color: "hex:E05050", bg: "transparent", full_width: false, padding_h: 8, padding_v: 4, disabled_when: busyThis, action: cancelGift } },
        ] },
      ] },
 
      // ── Idle — form closed for this serial ──
      { type: "vstack", visible_when: `item.pending == null && !(${openWhen})`, props: { spacing: 6, alignment: "leading" }, children: [
        text("“{item.giftMessage}”", "cormorant.italic:13", "offwhite", { visible_when: "item.giftMessage != null" }),
        giftBtn("Gift this card",    openForm(false), "item.giftMessage == null", { leading_icon: "gift" }),
        giftBtn("Edit or send gift", openForm(true),  "item.giftMessage != null", { leading_icon: "gift" }),
      ] },
 
      // ── Open — the one GiftForm (message + optional send-to) ──
      { type: "vstack", visible_when: `item.pending == null && ${openWhen}`, props: { spacing: 8, alignment: "leading", max_width: "infinity" }, children: [
        { type: "text_field", props: { bind: "gift_draft_message", axis: "vertical", line_limit: "3-6", max_length: 240,
          placeholder: "Write a personal message — it's revealed to the recipient in their 1+1 Gold account when they accept the card, and printed on the Certificate in the box.", max_width: "infinity" } },
        { type: "text_field", props: { bind: "gift_draft_recipient", placeholder: "Send to a phone or email (optional)", keyboard: "email", content_type: "email", max_width: "infinity" } },
        text("We'll send them a claim link — only they can accept it, and you can cancel any time before they do.", "montserrat:10", "muted", { visible_when: 'state.gift_draft_recipient != ""' }),
        text("Leave blank to hand the card over in person. Add a phone or email to send it to them instead.", "montserrat:10", "muted", { visible_when: 'state.gift_draft_recipient == ""' }),
        { type: "hstack", props: { spacing: 8, justify: "between", alignment: "center" }, children: [
          { type: "rich_text", props: { segments: [{ text: "{state.gift_draft_message.length} / 240", font: "montserrat:10", color: "muted" }] } },
          { type: "hstack", props: { spacing: 8, alignment: "center" }, children: [
            { type: "button", props: { label_text: "Cancel", label_font: "montserrat:11", label_color: "muted", bg: "transparent", full_width: false, padding_h: 6, padding_v: 4,
              action: { kind: "state_set", key: "gift_open_serial", value: "" } } },
            giftBtn("Save gift", saveMessage, 'state.gift_draft_recipient == ""', { disabled_when: busyThis }),
            giftBtn("Send gift", sendGift,    'state.gift_draft_recipient != ""', { disabled_when: busyThis }),
          ] },
        ] },
      ] },
    ],
  };
}
