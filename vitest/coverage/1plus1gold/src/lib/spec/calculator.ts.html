/**
 * SDUI spec for the Calculator surface (Phase 6.5).
 *
 * Architecture (Mukesh-signed-off 2026-06-03, contract §22): the Buy-now tab is
 * native SDUI driven by the ON-DEVICE JS engine (`data_source_method: "local"`).
 * The SAME pure pricing core (`lib/calculator-core`) runs in JavaScriptCore, so
 * every keystroke/slide recomputes LOCALLY and INSTANTLY — no network round-trip
 * (the old reactive-POST approach was structurally laggy). `data_source_body_keys`
 * still names the inputs that trigger a recompute (currency ← CurrencyManager).
 *
 * Four tabs: Buy now (native SDUI, JS engine) · Savings simulator (Path-B embed)
 * · Daily Life (Path-B embed, added 2026-08-30) · Charts (Path-B embed). The
 * buy-tab charts (rate curve, accumulation) are also Path-B web_panels (chart
 * rule). OptionCard is a dedicated composite.
 *
 * Daily Life ("Daily life, weighed in gold") is a heavy interactive unit
 * (per-category currency-scaled inputs, whole-bar/1g-card image pictographs,
 * localStorage, live-pricing fetch, share) → Path-B, the SAME call the Savings
 * simulator makes. Adding it is a pure server-spec change (segmented option +
 * tab_switcher web_panel case + header conditionals + prewarm) that the installed
 * binary renders generically — NO native build. Embed: /embed/calculator/life.
 *
 * PORT-DIFF + Contract: docs/sdui-pwa-port-contract.md §21/§22 (+ §93 Daily Life
 * row). Anchored to calculator/page.tsx. No element dropped (no-deferrals rule).
 */
 
import type { SDUISpec, SDUINode } from "./types";
 
const text = (content: string, font: string, color: string, extra: Record<string, unknown> = {}): SDUINode => ({
  type: "text", props: { content, font, color, line_spacing: 4, ...extra },
});
 
/** A `.card rounded-xl p-8` section (matches the our-pricing rhythm: spacing 18). */
const card = (children: SDUINode[], spacing = 18): SDUINode => ({
  type: "card",
  props: { stroke: "rule", bg: "bg_card", corner_radius: 12, padding: 24, alignment: "leading", spacing },
  children,
});
 
/** A web_panel chart embed (chart rule).
 *  - `query` seeds the INITIAL highlight via the URL (correct first paint).
 *  - `bridge` pushes LIVE updates via `window.__calcSet*` as the order changes
 *    (the panel is `.id(path)`-stable, so a URL change alone wouldn't update it).
 *    Maps a JS fn → scalar template, or → {key: template} object. Contract §22. */
const chart = (
  slug: string,
  fallbackHeight: number,
  query?: Record<string, string>,
  bridge?: Record<string, string | Record<string, string>>,
): SDUINode => ({
  type: "web_panel",
  props: {
    path: `/embed/calculator/${slug}`, height: fallbackHeight, lazy: true,
    ...(query ? { query } : {}), ...(bridge ? { bridge } : {}),
  },
});
 
/** A segmented form-mode option_select (no endpoint → just sets state). */
const segmented = (bind: string, options: { value: string; label: string }[]): SDUINode => ({
  type: "option_select", props: { bind, style: "segmented", options },
});
 
/** A summary stat tile. `valueNode` is the big figure; `label` the caption;
 *  `subs` are extra sub-lines below the caption (premium "above spot" + per-gram;
 *  total-spent "budget unspent"). */
const statTile = (valueNode: SDUINode, label: string, subs: SDUINode[] = []): SDUINode => ({
  type: "card",
  props: { stroke: "rule", bg: "bg_card", corner_radius: 12, padding: 16, alignment: "center", spacing: 4, full_width: true, fill_height: true },
  children: [valueNode, text(label, "montserrat:12", "muted", { alignment: "center" }), ...subs],
});
 
/** Preset chip repeater — WRAPS to multiple rows (PWA flex-wrap). Budget chips
 *  show the selected currency flag + symbol (`currency_prefix`); both keep the
 *  label on one line (`label_nowrap`). `item.active` highlights the match. */
const presetChips = (dataPath: string, bindKey: string, currencyPrefix: boolean): SDUINode => ({
  type: "repeater",
  props: { data_path: dataPath, item_id_field: "value", axis: "wrap", spacing: 8 },
  children: [{
    type: "button",
    props: {
      label_text: "{item.label}", label_font: "montserrat:12",
      label_color: "gold", label_color_when: "item.active == true", label_color_else: "muted",
      bg: "gold@10", bg_when: "item.active == true", bg_else: "transparent",
      stroke: "gold", stroke_when: "item.active == true", stroke_else: "rule",
      corner_radius: 6, padding_h: 12, padding_v: 6, full_width: false,
      label_nowrap: true, ...(currencyPrefix ? { currency_prefix: true } : {}),
      action: { kind: "state_set", key: bindKey, value: "{item.value}" },
    },
  }],
});

    // by CalculatorEngine.

    data_source_body_keys: ["mode", "budget", "grams", "currency"],
    // On a currency switch, re-express `budget` in the new currency (preserve SGD
    // value) so it doesn't fall below 1g and blank the result (§22, device 2026-06-04).
    currency_amount_key: "budget",
    // Cache-warm the heavy embed tabs on open so a later tab tap paints fast (#15).
    prewarm_embeds: ["/embed/calculator/simulator", "/embed/calculator/life", "/charts"],
    state_schema: {
      tab:    { type: "string", default: "buy" },
      mode:   { type: "string", default: "grams" },
      grams:  { type: "string", default: "10" },
      budget: { type: "string", default: "1000" },
    },
    root: {
      type: "scroll",
      children: [{
        type: "vstack",
        props: { spacing: 32, padding: 16, alignment: "leading" },
        children: [
 
          // ── Header (copy swaps per tab + mode) ─────────────────────────
          {
            type: "vstack",
            props: { spacing: 12, alignment: "center", full_width: true },
            children: [
              { type: "conditional", props: {
                cases: [
                  { when: "state.tab == 'charts'",    then: text("Market Data", "cinzel:12", "gold_mid", { tracking: 4, text_case: "upper", alignment: "center" }) },
                  { when: "state.tab == 'simulator'", then: text("Savings Simulator", "cinzel:12", "gold_mid", { tracking: 4, text_case: "upper", alignment: "center" }) },

                ],

              } },
              { type: "conditional", props: {
                cases: [
                  { when: "state.tab == 'charts'",    then: text("Gold Charts", "cinzel.bold:36", "cream", { alignment: "center" }) },
                  { when: "state.tab == 'simulator'", then: text("How Far Can My Saving Take Me?", "cinzel.bold:36", "cream", { alignment: "center" }) },
                  { when: "state.tab == 'life'",      then: text("Daily Life, Weighed in Gold", "cinzel.bold:36", "cream", { alignment: "center" }) },
                  { when: "state.mode == 'budget'",   then: text("How Much Gold Can I Get?", "cinzel.bold:36", "cream", { alignment: "center" }) },
                ],
                else: text("How Much Will It Cost?", "cinzel.bold:36", "cream", { alignment: "center" }),

                  { when: "state.tab == 'life'",      then: text("A month of everyday life in gold — see what each part is worth, and the gold you keep on our Premium Curve.", "cormorant.italic:18", "muted", { alignment: "center" }) },

                ],
                else: text("Enter your target grams and we'll show you the total cost and denomination mix", "cormorant.italic:18", "muted", { alignment: "center" }),
              } },
 
              // Spot strip — buy tab only, once spot is loaded (#1). Line 1:
              // per-gram + per-oz (flag+symbol via currency_amount). Line 2:
              // "Updated …" (preformatted by the core).
              { type: "conditional", props: { cases: [{

          // ── Tab content ────────────────────────────────────────────────

          // tab loads on first visit, then stays resident. Contract §22.
          { type: "tab_switcher", props: {
            cases: [
              { when: "state.tab == 'simulator'", then: { type: "web_panel", props: { path: "/embed/calculator/simulator", height: 720, lazy: true } } },

            else: buyTab(),
          } },
        ],
      }],
    },
  };
}
 
/** The Buy-now tab — input card + (live) results. */
function buyTab(): SDUINode {
  return {
    type: "vstack",
    props: { spacing: 24, alignment: "leading", full_width: true },
    children: [
 
      // ── Input card ───────────────────────────────────────────────────
      card([
        segmented("mode", [
          { value: "grams",  label: "My Grams" },
          { value: "budget", label: "My Budget" },
        ]),
        { type: "conditional", props: {
          cases: [{
            when: "state.mode == 'budget'",
            then: {
              type: "vstack", props: { spacing: 16, alignment: "leading", full_width: true },
              children: [
                text("My Budget (selected currency)", "montserrat:12", "gold_mid", { tracking: 1.5, text_case: "upper" }),
                { type: "text_field", props: { bind: "budget", keyboard: "decimal", font: "montserrat.semibold:24", color: "cream", placeholder: "0", prefix_symbol: true } },
                { type: "slider", props: { bind: "budget", min: "{data.sliderMin}", max: "{data.sliderMax}", step: 1, accent: "gold" } },
                presetChips("data.presetsBudget", "budget", true),
              ],
            },
          }],
          else: {
            type: "vstack", props: { spacing: 16, alignment: "leading", full_width: true },
            children: [
              text("I Want (grams) — minimum 1g", "montserrat:12", "gold_mid", { tracking: 1.5, text_case: "upper" }),
              { type: "text_field", props: { bind: "grams", keyboard: "decimal", font: "montserrat.semibold:24", color: "cream", placeholder: "10", suffix_text: "g" } },
              { type: "slider", props: { bind: "grams", min: 1, max: 200, step: 1, accent: "gold" } },
              presetChips("data.presetsGrams", "grams", false),
            ],
          },
        } },
      ]),
 
      // ── Results (live) ─────────────────────────────────────────────────
      { type: "conditional", props: {
        cases: [{
          when: "data.hasResult == true",
          then: {
            type: "vstack", props: { spacing: 24, alignment: "leading", full_width: true },
            children: [
 
              // Summary stats — 3 tiles
              { type: "grid", props: { columns: 3, spacing: 12 }, children: [
                statTile({ type: "grams_text", props: { value: "{data.summary.totalGrams}", num_font: "cinzel.bold:24", unit_font: "montserrat:12", color: "gold", alignment: "center" } }, "Total Grams"),
                statTile(
                  text("{data.summary.premiumLabel}", "cinzel.bold:24", "gold", { alignment: "center", scale_to_fit: true }),
                  "Your Premium",
                  [
                    text("above spot · on gold", "montserrat:10", "muted", { alignment: "center" }),
                    { type: "currency_amount", props: { sgd_value: "{data.summary.perGramSgd}", suffix: "/g", font: "montserrat:12", color: "offwhite", flag_height: 10, alignment: "center" } },
                  ],
                ),
                statTile(
                  // wrap (not scale_to_fit): keep 24pt and wrap to a 2nd line if
                  // wide — matches the PWA tile + the 24pt sibling tiles (shrinking
                  // made it look tiny). PWA src/app/calculator/page.tsx:575-577.
                  { type: "currency_amount", props: { sgd_value: "{data.summary.totalSgd}", font: "cinzel.bold:24", color: "gold", flag_height: 12, alignment: "center", wrap: true } },
                  "Total Spent",
                  [{
                    type: "conditional", props: { cases: [{
                      when: "data.summary.unspentSgd > 0.5",
                      then: { type: "currency_amount", props: { sgd_value: "{data.summary.unspentSgd}", suffix: " budget unspent", font: "montserrat.semibold:12", color: "cream", flag_height: 10, alignment: "center", wrap: true, wrap_lines: 0 } },
                    }] },
                  }],
                ),
              ] },
 
              // Curiosity comparison (server-picked; only when present)
              { type: "conditional", props: { cases: [{
                when: "data.comparisonLine != null",
                then: { type: "rich_text", props: { font: "cormorant.italic:16", color: "muted", alignment: "center", runs: [
                  { text: "Curiously, {data.comparisonLine.spend}. ", color: "hex:8A7440" },
                  { text: "{data.comparisonLine.punchline}", color: "hex:A07830" },
                ] } },
              }] } },
 
              // Progress to next 50g bar
              card([
                text("Progress to your next 50g bar", "montserrat:12", "gold_mid", { tracking: 1.5, text_case: "upper" }),
                // "X accumulated + Y from this purchase" — signed-in holders only (#10)
                { type: "conditional", props: { cases: [{
                  when: "data.progress.accumulatedLabel != null",
                  then: text("{data.progress.accumulatedLabel}", "montserrat:12", "muted"),
                }] } },
                { type: "progress_bar", props: { current: "{data.progress.currentG}", target: "{data.progress.targetG}", show_labels: true, accumulated_label: "with this purchase" } },
              ], 12),
 
              // Rate for this order size (chart embed)
              card([
                text("Rate for this order size", "montserrat:12", "gold_mid", { tracking: 1.5, text_case: "upper" }),
                text("Based on the total grams entered above. Your actual rate depends on your lifetime accumulated grams — visible in your account.", "montserrat:12", "muted"),
                chart("premium-curve", 200,
          { grams: "{data.summary.totalGrams}" },
          { __calcSetGrams: "{data.summary.totalGrams}" }),
              ], 12),
 
              // Choose your gold — two OptionCard composites (single column on phone)
              { type: "conditional", props: { cases: [{
                when: "data.options.curve != null",
                then: { type: "option_card", props: { option_path: "options.curve", title: "On the Premium Curve", hook: "Premium drops as you accumulate. Counts toward your next 50g bar." } },
              }] } },
              { type: "conditional", props: { cases: [{
                when: "data.options.standard != null",
                then: { type: "option_card", props: { option_path: "options.standard", title: "As a Standard Bar", hook: "Sealed assay card, fixed-size piece. No curve discount, but cheaper at this size.", note_path: "standardNote" } },
              }] } },
            ],
          },
        }],
      } },
 
      // Budget too low
      { type: "conditional", props: { cases: [{
        when: "data.tooLow == true",
        then: card([ text("Budget too low for the smallest available product", "montserrat:14", "muted", { alignment: "center" }) ], 0),
      }] } },
 
      // ── Accumulation Projection (chart embed) ─────────────────────────
      card([
        text("Accumulation Projection", "cinzel.semibold:20", "gold"),
        text("Months needed to reach the 50g bar milestone at different monthly investment levels, using the 1g Split premium at current spot prices.", "montserrat:12", "muted"),
        chart("accumulation", 320,
          { budget: "{data.accumulationBudgetSgd}", premium: "{data.summary.blendedPremium}" },
          { __calcSetAccumulation: { budgetSgd: "{data.accumulationBudgetSgd}", premium: "{data.summary.blendedPremium}" } }),
      ]),
    ],
  };
}
