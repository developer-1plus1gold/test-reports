/**
 * SDUI spec for Account → Refer & Earn (Q2).
 *
 * Read-mostly surface (no CRUD): D1 data_source fetch + display + copy. Reuses
 * the CRUD engine's D1/D2 + P4/P5 + the four primitives approved 2026-05-31:
 *   - copy_to_clipboard action + state_set revert_after_ms ("Copied!" inline)
 *   - date_text node (history dates)
 *   - live_product_price / live_price-segment `multiplier_pct` (live "worth $Y")
 *   - rich_text `text` segment now honours per-segment font/color
 *
 * Data: GET /api/referrals/me (camelCase — the store passes keys verbatim, so
 * tokens are {data.shareUrl} / {item.rewardSgd}). Auth fixed to
 * getUserFromRequest (Bearer+cookies) for iOS.
 *
 * FULL PORT-DIFF vs PWA src/app/account/page.tsx:3419-3631 (Contract column):
 * | Element                         | PWA                                  | iOS SDUI                                         | Contract |
 * |---------------------------------|--------------------------------------|--------------------------------------------------|----------|
 * | Header card + Gift + title      | card p-6, Gift 18, cinzel 16 gold    | card + icon + text                               | L/T/I    |
 * | Explainer (bold + live $)       | <p> w/ <strong> + Flag{fmt(5g×pct)}  | rich_text segments (styled text + live_price)    | P5 (+font seg, +multiplier_pct) |
 * | Loading                         | referralLoading                      | loadingRow(data._loading)                        | D1/crud  |
 * | "YOUR REFERRAL CODE" + code     | montserrat-uppercase + cinzel 24     | text + text {data.code}                          | T        |
 * | Influencer badge                | purple pill + Star                   | hstack chip visible_when tier                    | NEW chip |
 * | Copy Code / Copy Link           | clipboard + 2s "Copied!"             | button copy_to_clipboard + state_set revert      | NEW (copy + transient) |
 * | shareUrl text                   | montserrat 10 break-all              | text {data.shareUrl}                             | T        |
 * | Stats grid (3)                  | grid-cols-3, icon+value+label        | repeater grid_columns:3 (+ currency_amount)      | D2 + currency_amount |
 * | Influencer programme (if tier)  | gold-dim card + 3 rows               | card + kv_row ×3 (live $ values)                 | P4 + P5  |
 * | Become-influencer (else)        | gold-dim card + 3 mini + WA button   | card + repeater grid:3 + open_url WA button      | D2 + open_url |
 * | How it works (4 steps, tier)    | numbered circles + rich text + live $| two step-lists visible_when tier; zstack circle + rich_text | P5 + conditional |
 * | Referral history (if len>0)     | history.map rows                     | repeater; status chips, currency_amount, date_text | D2/D7 + date_text |
 *
 * Nothing dropped: the tier-conditional copy is two visible_when variants (not
 * omitted); the history date uses the new date_text primitive; the live values
 * use the live_price segment multiplier (not server-computed, per the live-price
 * architecture). See contract §12/§13 + the P-rows.
 */
 
import type { SDUISpec, SDUINode, SDUIAction } from "./types";

const GOLD_MID = "gold_mid";
 
/** Section eyebrow ("YOUR REFERRAL CODE" etc.). */
function eyebrow(text: string): SDUINode {
  return { type: "text", props: { content: text, font: "montserrat:12", color: GOLD_MID, tracking: 2 } };
}
 
function txt(content: string, opts?: { bold?: boolean; color?: string }): Record<string, unknown> {
  return {
    kind: "text", content,
    ...(opts?.bold ? { font: "montserrat.semibold:12" } : { font: "montserrat:12" }),
    color: opts?.color ?? "muted",
  };
}
 
/** Inline live price = spot × 5g × (1+premium@5g) × multiplierPct/100. */
function livePrice(opts: { multiplierPct: string | number; color: string; bold?: boolean }): Record<string, unknown> {
  return {
    kind: "live_price", grams: 5, premium_pct: "auto", multiplier_pct: opts.multiplierPct, flag_size: 11,
    font: opts.bold ? "montserrat.semibold:12" : "montserrat:12", color: opts.color,
  };
}
 
/** "worth up to $X" — X% of the 5g all-in price. bold→offwhite (explainer/
 *  steps), else muted (influencer rows / step 3). */
function worth(pctToken: string, bold: boolean): Record<string, unknown> {
  return livePrice({ multiplierPct: pctToken, color: bold ? "offwhite" : "muted", bold });
}
 
/** PWA `priceNote` (account/page.tsx:3433-3435) — the muted clause appended
 *  after a worth value: " at S$<perGram>/g today (5g Combibar rate — exact
 *  value confirmed at checkout)". perGram = 5g all-in ÷ 5 = ×20%. */
function priceNote(): Record<string, unknown>[] {
  return [
    txt(" at ", { color: "muted" }),
    livePrice({ multiplierPct: 20, color: "muted" }),
    txt("/g today (5g Combibar rate — exact value confirmed at checkout)", { color: "muted" }),
  ];
}
 
/** The header explainer — one variant per tier (the tier clause differs). */
function explainer(isInfluencer: boolean): SDUINode {
  return {
    type: "rich_text",
    visible_when: isInfluencer ? 'data.tier == "influencer"' : 'data.tier != "influencer"',
    props: {
      font: "montserrat:12", color: "muted", line_spacing: 3,
      segments: [
        txt("Share your referral code with friends. They get "),
        txt("{data.refereeDiscountPct}% off up to 5g of gold", { bold: true, color: "offwhite" }),
        txt(" on their first order — worth up to "),
        worth("{data.refereeDiscountPct}", true),
        ...priceNote(),
        txt(". You earn "),
        txt("{data.referrerRewardPct}% of their eligible purchase", { bold: true, color: "offwhite" }),
        txt(isInfluencer ? " paid monthly via PayNow" : " as a discount on your next order"),
        txt(" — worth up to "),
        worth("{data.referrerRewardPct}", true),
        ...priceNote(),
        txt("."),
      ],
    },
  };
}

function codeCard(): SDUINode {
  // Copy + 2s "Copied!" feedback (PWA refCopied) via the transient state_set.
  const copyAction = (value: string): SDUIAction => ({
    kind: "composed",
    actions: [
      { kind: "copy_to_clipboard", value },
      { kind: "state_set", key: "copied", value: true, revert_after_ms: 2000, revert_to: false },
    ],
  } as SDUIAction);
  const copyBtn = (label: string, icon: string, value: string, filled: boolean, visibleWhen?: string): SDUINode => ({
    type: "button",
    ...(visibleWhen ? { visible_when: visibleWhen } : {}),
    props: {
      label_text: label, leading_icon: icon, label_font: "montserrat.semibold:12",
      bg: filled ? "gold" : "bg", label_color: filled ? "bg" : "gold",
      ...(filled ? {} : { stroke: "gold_dim" }),
      full_width: false, corner_radius: 8, padding_h: 12, padding_v: 8,
      action: copyAction(value),
    },
  });
  return {
    type: "card",
    props: { padding: 24, corner_radius: 12, bg: "bg_card", spacing: 12, alignment: "leading" },
    children: [
      eyebrow("YOUR REFERRAL CODE"),
      {
        type: "hstack",
        props: { spacing: 12 },
        children: [
          { type: "text", props: { content: "{data.code}", font: "cinzel.bold:24", color: "gold", tracking: 2 } },
          // Influencer badge.
          {
            type: "text",
            visible_when: 'data.tier == "influencer"',
            props: {
              content: "★ INFLUENCER", font: "montserrat.semibold:10", color: "hex:C084FC",
              bg: "hex:A855F7@18", corner_radius: 999, padding_h: 8, padding_v: 2, tracking: 1,
            },
          },
        ],
      },
      {
        type: "hstack",
        props: { spacing: 8 },
        children: [
          copyBtn("Copy Code", "doc.on.doc", "{data.code}", true, "state.copied != true"),
          copyBtn("Copied!", "checkmark.circle.fill", "{data.code}", true, "state.copied == true"),
          copyBtn("Copy Link", "doc.on.doc", "{data.shareUrl}", false),
        ],
      },
      { type: "text", props: { content: "{data.shareUrl}", font: "montserrat:10", color: "muted" } },
    ],
  };
}

/** Influencer programme card (tier == influencer). */
function influencerCard(): SDUINode {
  // PWA renders these as justify-between label:value rows with a bottom rule.
  // kv_row's value is string-only (P4's "or rich_text" is unimplemented), and
  // these values carry an inline live price — so render directly as
  // hstack(label + Spacer + rich_text) + a divider. Same visual as PWA.
  const row = (label: string, valueSegments: Record<string, unknown>[]): SDUINode => ({
    type: "vstack",
    props: { spacing: 0, alignment: "leading" },
    children: [

        children: [
          { type: "text", props: { content: label, font: "montserrat:12", color: "muted" } },

        ],
      },
      { type: "divider", props: { color: "rule" } },

  });
  return {
    type: "card",
    visible_when: 'data.tier == "influencer"',
    props: { padding: 24, corner_radius: 12, bg: "gold@5", stroke: "gold_dim", spacing: 8, alignment: "leading" },

  return {
    type: "card",

    children: [eyebrow("HOW IT WORKS"), standardSteps, influencerSteps],
  };
}
