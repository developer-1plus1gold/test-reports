/**
 * Buybacks (Sell to Us — Requests) history surface — Account → "Sell" tile.
 *
 * Read-only list mirroring the PWA account "Sell to Us — Requests" section
 * (src/app/account/page.tsx:3342-3548). This is the FIRST screen the Sell tile
 * opens; the "+ New Request" CTA navigates to /sell-to-us → InternalLinkRouter
 * resolves it to the native `sell-to-us` wizard surface (N1/N2). Same IA as the
 * `conversions` surface (its sibling under Sell-to-Us).
 *
 * Data: GET /api/buybacks/my-requests (Bearer+cookies via getUserFromRequest).
 * Each request is enriched SERVER-side with display fields via the shared
 * `buildBuybackRequestView` (lib/buybacks.ts) — refLabel/statusLabel/
 * statusColorToken/dateLabel/totalGramsLabel/netPayoutSgd/lines[]/... — so the
 * numbers match the web list and the spec stays declarative.
 *
 * PORT-DIFF (Aspect | PWA | iOS | Contract):
 * | Header + "+ New Request"  | h2 + gold Link → /sell-to-us  | text + button navigate /sell-to-us  | N1/N2 |
 * | Empty state               | ArrowLeftRight + copy + CTA   | icon + text + button                | —     |
 * | Status pill               | {color}22 bg / {color} text   | text {statusColorToken} @13/@27     | §26 status-pill |
 * | Card .card gold top       | bg-card tile                  | card stroke:"rule" → auto gold-top  | C-card |
 * | Summary grid (3 col)      | grid-cols-3                   | hstack 3× vstack                    | L     |
 * | Line details (expandable) | "Details" toggle             | shown INLINE (no per-row state in a repeater — same as conversions) | §26 |
 * | Driver-tracking link      | <a target=_blank>             | button open_url (visible_when)      | N1    |
 */
 
import type { SDUISpec, SDUINode } from "./types";
 
function text(content: string, font: string, color: string, extra: Record<string, unknown> = {}): SDUINode {
  const { visible_when, ...rest } = extra as { visible_when?: string };
  return { type: "text", ...(visible_when ? { visible_when } : {}), props: { content, font, color, ...rest } };
}
function money(sgdPath: string, font: string, color: string, extra: Record<string, unknown> = {}): SDUINode {
  const { visible_when, ...rest } = extra as { visible_when?: string };
  return { type: "currency_amount", ...(visible_when ? { visible_when } : {}),
    props: { sgd_value: sgdPath, font, color, show_sgd_base: true, flag_height: 10, placeholder: "—", ...rest } };
}
function eyebrow(content: string, extra: Record<string, unknown> = {}): SDUINode {
  return text(content, "montserrat.semibold:10", "muted", { tracking: 1.5, text_case: "upper", ...extra });
}
 
/** Status pill — colour driven entirely by the server token (§26 pattern). */
function statusPill(): SDUINode {
  return { type: "text", props: {
    content: "{item.statusLabel}", font: "montserrat.semibold:10", text_case: "upper", tracking: 0.5,
    color: "{item.statusColorToken}", bg: "{item.statusColorToken}@13", stroke: "{item.statusColorToken}@27",
    corner_radius: 999, padding_h: 8, padding_v: 3,
  } };
}
 
/** One bar line inside the request (nested repeater over item.lines). */
function lineRow(): SDUINode {
  return { type: "vstack", props: { spacing: 0, alignment: "leading", max_width: "infinity" }, children: [
    { type: "hstack", props: { spacing: 8, alignment: "center", max_width: "infinity", padding_v: 8 }, children: [
      text("{item.label}", "montserrat:12", "offwhite"),
      text("{item.qtyGramsLabel}", "montserrat:12", "muted"),
      { type: "spacer", props: {} },
      money("{item.priceSgd}", "montserrat.semibold:12", "gold", { visible_when: "item.hasPrice == true", show_sgd_base: false }),
      text("—", "montserrat.semibold:12", "gold", { visible_when: "item.hasPrice == false" }),
    ] },
    { type: "divider", props: { color: "rule" } },
  ] };
}

function header(): SDUINode {
  return { type: "hstack", props: { spacing: 12, alignment: "center", max_width: "infinity", axis: "wrap" }, children: [
    text("Sell to Us — Requests", "cinzel.bold:18", "offwhite"),
    { type: "spacer", props: {} },
    { type: "button", props: { label_text: "+ New Request", label_font: "montserrat.semibold:12", label_color: "bg", bg: "gold", corner_radius: 8, padding_h: 16, padding_v: 8, full_width: false,
      action: { kind: "open_sell" } } },
  ] };
}
 
function emptyState(): SDUINode {
  return { type: "vstack", visible_when: "data._loading == false && data.requests.length == 0",
    props: { spacing: 12, alignment: "center", max_width: "infinity", padding_v: 48 }, children: [
      { type: "icon", props: { name: "arrow.left.arrow.right", size: 36, color: "gold@30" } },
      text("No sell requests yet.", "montserrat:14", "muted", { alignment: "center" }),
      { type: "button", props: { label_text: "Sell your gold", label_font: "montserrat.semibold:14", label_color: "bg", bg: "gold", corner_radius: 8, padding_h: 24, padding_v: 10, full_width: false,
        action: { kind: "open_sell" } } },
    ] };
}
 
export function buybacksSpec(): SDUISpec {
  return {
    surface: "buybacks",
    spec_version: 1,
    data_source: "/api/buybacks/my-requests",
    root: {
      type: "scroll",
      children: [
        { type: "vstack", props: { spacing: 16, padding_h: 16, padding_v: 24, alignment: "leading", max_width: "infinity" }, children: [
          header(),
          text("Loading requests…", "montserrat:14", "muted", { visible_when: "data._loading == true", padding_v: 8 }),
          emptyState(),
          { type: "repeater", props: { data_path: "data.requests", item_id_field: "id", spacing: 16 }, children: [buybackRow()] },
        ] },
      ],
    },
  };
}
