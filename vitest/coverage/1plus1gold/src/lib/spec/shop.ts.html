/**
 * SDUI spec for the Shop surface (Phase 6, contract §23).
 *
 * Rebuilt 2026-06-04 (round 2) against a SOURCE-VERIFIED port-diff — the first
 * pass assumed prop names the renderers don't read. EVERY prop here is verified
 * against its Swift renderer (no assumptions); layout matches the PWA
 * (src/app/shop/page.tsx + components/products/GoldProductCard.tsx +
 * AccessoryCard). Verified APIs used:
 *   - async_image: `src` (NOT url); resolves relative "/images/x.png" → web origin.
 *   - media_carousel: reads ctx.item.product_media; resolves relative URLs.
 *   - icon: `name:"brand:curve_sparkline"` / `name:"brand:standard_bars"`.
 *   - card: accent_leading/accent_width (PWA border-left), stroke/bg/corner/padding.
 *   - tile grid: REPEATER `grid_columns` (grid node does NOT iterate).
 *   - text badge: container props (stroke/bg/corner_radius/padding_*); hex:/@opacity.
 *   - button conditional styling: bg_when/bg_else/stroke_when/label_color_when (chips).
 *   - state_set op:"toggle" = array-membership toggle (multi-filters); `contains` predicate.
 *   - option_select style:"menu" form-mode (sort); kv_row (label/value/*_color).
 *   - product_qty_line (NATIVE, §23): owns the price + qty editor + line value +
 *     Add CTA. Reactive per-row pricing via the on-device kernel (engine.shopLine)
 *     with NO whole-catalog recompute (keeps keyboard focus). variant tile|row.
 *
 * Data envelope (engine.js catalogView, on-device): { curveGroups[{key,products[]}],
 * standardGroups[{key,products[]}], accessories[], gramsOptions[{value,label}],
 * refineryOptions[{value,label}] }; curve + standard BOTH run the shared
 * filter→sort→group pipeline (the page-top controls govern both), options span
 * all gold. Each product carries its /api/products fields + `pricing`. §23.
 */
 
import type { SDUISpec, SDUINode } from "./types";
 
// `visible_when` is a NODE-LEVEL field (SDUIRenderer reads node.visibleWhen),
// NOT a prop — so these helpers hoist it out of props. `extra.visible_when` on
// text() is lifted automatically.
export const text = (content: string, font: string, color: string, extra: Record<string, unknown> = {}): SDUINode => {
  const { visible_when, ...rest } = extra;
  return { type: "text", ...(visible_when ? { visible_when: visible_when as string } : {}), props: { content, font, color, ...rest } };
};
 
/** Pill badge = text + bordered/tinted container (verified container props). */
const badge = (content: string, color: string, stroke: string, visible_when?: string): SDUINode => ({
  type: "text",
  ...(visible_when ? { visible_when } : {}),
  props: { content, font: "montserrat:12", color, stroke, stroke_width: 1, corner_radius: 4, padding_h: 8, padding_v: 2 },
});
 
const rule = (): SDUINode => ({ type: "divider", props: { color: "rule", thickness: 1 } });
 
/** label/value detail row (kv_row reads label/value/label_color/value_color). */
const kv = (label: string, value: string, valueColor = "offwhite", visible_when?: string): SDUINode => ({
  type: "kv_row",
  ...(visible_when ? { visible_when } : {}),
  props: { label, value, label_color: "muted", value_color: valueColor },
});
 
/** Section header: badge pill + title (+ reused brand icon) + subtitle. */
function sectionHeader(badgeText: string, title: string, subtitle: SDUINode, iconName?: string): SDUINode {
  return {
    type: "vstack", props: { spacing: 8, alignment: "leading" },
    children: [
      text(badgeText, "montserrat.semibold:12", "gold",
        { stroke: "gold_dim", stroke_width: 1, corner_radius: 999, padding_h: 12, padding_v: 4, bg: "gold@8" }),
      {
        type: "hstack", props: { spacing: 8, alignment: "center" },
        children: [
          text(title, "cinzel.bold:28", "cream"),
          ...(iconName ? [{ type: "icon", props: { name: iconName, size: 46 } } as SDUINode] : []),
        ],
      },
      subtitle,
    ],
  };
}

/** The native price + qty editor + line value cluster (§23). `enforceMinQty`
 *  (Wholesale) makes it pre-fill to the product's minQty + block Add below it. */
function qtyLine(variant: "tile" | "row", enforceMinQty = false): SDUINode {
  return { type: "product_qty_line", props: { variant, ...(enforceMinQty ? { enforce_min_qty: true } : {}) } };
}
 
/** Gold product card (curve + standard) — port of GoldProductCard. */
export function goldCard(mediaHeight: number, enforceMinQty = false): SDUINode {
  return {
    type: "card",
    // id → scroll anchor for Search-driven scroll-to-product (SEARCH-ANCHOR,
    // parity with the PWA `/shop#product-{id}`). Keyed on the UNIQUE product id
    // (a curve combibar and a standard bar can share a weight → grams collides);
    // SearchSheet sets the same `product-<id>` anchor. The root scroll's
    // scroll_anchors mode targets this via _scroll_target.
    props: { id: "product-{item.id}", stroke: "rule", bg: "bg_card", corner_radius: 12, padding: 0, alignment: "leading", spacing: 0 },
    children: [
      { type: "media_carousel", props: { media_path: "product_media", height: mediaHeight,
        dim_when: "item.pricing.soldOut == true && item.pricing.presale == false" } },
      {
        type: "vstack", props: { spacing: 16, padding: 24, alignment: "leading" },
        children: [
          { type: "hstack", props: { justify: "between", alignment: "top" }, children: [
            text("{item.label}", "cinzel.bold:24", "gold"),
            { type: "vstack", props: { spacing: 4, alignment: "trailing" }, children: statusBadges() },
          ] },
          rule(),
          { type: "vstack", props: { spacing: 8, alignment: "leading" }, children: goldDetails() },
          rule(),
          // Price + qty editor + line value (native — reactive per-row pricing
          // without a whole-catalog recompute; §23). Owns the price, the tile's
          // spot-breakdown subline, the Add CTA, the Qty field, and the line total.
          qtyLine("tile", enforceMinQty),
        ],
      },
    ],
  };
}

/** "Row" view — faithful port of the PWA mobile row (flex-col): image + label,
 *  full DetailChips (label+value, value coloured), standard tier table, then
 *  price + per-unit subline + CTA. Driven by the kernel's `detailChips` array
 *  (a repeater is the only node that can wrap). */
export function productRow(accessory: boolean, enforceMinQty = false): SDUINode {
  const children: SDUINode[] = [
    { type: "hstack", props: { spacing: 12, alignment: "center" }, children: [
      // content_mode:"fit" (PWA object-fit:contain) so a non-square product
      // photo (e.g. the portrait 1g Split bar) letterboxes in the dark bordered
      // box instead of being cropped/distorted by the default scaledToFill.
      { type: "async_image", props: { src: "{item.product_media.0.url}", width: 96, height: 96, corner_radius: 6,
        content_mode: "fit", bg: "bg", stroke: "rule", stroke_width: 1 } },
      { type: "vstack", props: { spacing: 2, alignment: "leading" }, children: [
        text("{item.label}", "cinzel.semibold:16", "gold"),
        ...(accessory ? [] : [text("{item.grams}g", "montserrat:12", "offwhite")]),
      ] },
    ] },
    // DetailChips — wrapped (repeater is the only node that wraps); item is
    // re-scoped to each chip {label,value,color}; color interpolates (propString).
    { type: "repeater", props: { data_path: "item.detailChips", item_id_field: "label", axis: "wrap", spacing: 8 }, children: [
      { type: "hstack", props: { spacing: 4, alignment: "center" }, children: [
        text("{item.label}", "montserrat:12", "muted"),
        text("{item.value}", "montserrat.semibold:12", "{item.color}"),
      ] },
    ] },
  ];
  if (!accessory) {
    // Standard volume-tier table (PWA StandardProductRow) — only when tiered.
    children.push({ type: "vstack",
      visible_when: "item.product_type == 'standard' && item.pricing.tierFirstPct != null",
      props: { spacing: 4, alignment: "leading" }, children: [
        text("Volume tiers", "montserrat:11", "muted", { tracking: 1, text_case: "upper" }),
        { type: "repeater", props: { data_path: "item.tierRows", item_id_field: "rangeLabel", spacing: 2 }, children: [
          { type: "hstack", props: { justify: "between" }, children: [
            text("{item.rangeLabel}", "montserrat:12", "muted"),
            text("Spot +{item.premiumLabel}", "montserrat:12", "standard_blue"),
          ] },
        ] },
      ] });
  }
  // Price + qty editor + line value (native — §23). Reads item.product_type to
  // pick the per-type subline (curve "{qty}× per unit…" / standard "per unit at
  // qty {qty}"), recomputes the tier-aware price on qty change, and the line total.
  // No divider here — the PWA mobile row has none between details and price.
  children.push(qtyLine("row", enforceMinQty));
  return {
    type: "card",
    // id → scroll anchor (SEARCH-ANCHOR), as goldCard. Accessories aren't search
    // targets, so they get no anchor.
    props: { ...(accessory ? {} : { id: "product-{item.id}" }), stroke: "rule", bg: "bg_card", corner_radius: 12, padding: 20, alignment: "leading", spacing: 16 },
    children,
  };
}
 
/** A product list — Tile (rich card) vs Row (compact) on `state.view`, matching
 *  the PWA toggle. Single column on a phone (PWA tile is `w-full` < sm). */
export function productList(dataPath: string, card: SDUINode, row: SDUINode): SDUINode {
  return {
    type: "conditional", props: {
      cases: [{ when: "state.view == 'tile'", then:
        { type: "repeater", props: { data_path: dataPath, item_id_field: "id", spacing: 16 }, children: [card] } }],
      else: { type: "repeater", props: { data_path: dataPath, item_id_field: "id", spacing: 12 }, children: [row] },
    },
  };
}
 
// ── Standard-section filter chips (local state; button + state_set toggle) ──
export function filterChips(optionsPath: string, stateKey: string): SDUINode {
  const chip = (labelTemplate: string, valueTemplate: string, activePredicate: string): SDUINode => ({
    type: "button",
    props: {
      label_text: labelTemplate, label_font: "montserrat:12",
      label_color: "bg", label_color_when: activePredicate, label_color_else: "offwhite",
      bg: "gold", bg_when: activePredicate, bg_else: "transparent",
      stroke: "gold", stroke_when: activePredicate, stroke_else: "rule",
      corner_radius: 999, padding_h: 10, padding_v: 4, label_nowrap: true, full_width: false,
      action: { kind: "state_set", key: stateKey, value: valueTemplate, op: "toggle" },
    },
  });
  return {
    type: "repeater", props: { data_path: optionsPath, item_id_field: "value", axis: "wrap", spacing: 6 },
    // The active predicate MUST prefix the state key — the iOS predicate evaluator
    // resolves a BARE token to null (only state./data./item. resolve), so
    // `gramsFilter contains …` never matched → chips never highlighted (latent;
    // surfaced 2026-06-14). `state.gramsFilter contains item.value` resolves.
    children: [chip("{item.label}", "{item.value}", `state.${stateKey} contains item.value`)],
  };
}
 
/** The "All" chip — active (gold) when no filter is selected; clears the filter
 *  array when tapped. Parity with the web FilterChipGroup's "All" affordance
 *  (the only way to clear a multi-select filter). Mukesh 2026-06-14. */
function allChip(stateKey: string): SDUINode {
  const active = `state.${stateKey}.length == 0`;
  return {
    type: "button",
    props: {
      label_text: "All", label_font: "montserrat:12",
      label_color: "bg", label_color_when: active, label_color_else: "muted",
      bg: "gold", bg_when: active, bg_else: "transparent",
      stroke: "gold", stroke_when: active, stroke_else: "rule",
      corner_radius: 999, padding_h: 10, padding_v: 4, label_nowrap: true, full_width: false,
      action: { kind: "state_set", key: stateKey, value: [] },
    },
  };
}
 
/** A local form checkbox (no endpoint): two conditional buttons (checked/unchecked). */
export function checkbox(stateKey: string, label: string): SDUINode {
  const btn = (glyph: string, setTo: boolean): SDUINode => ({
    type: "button",
    props: {
      label_text: `${glyph}  ${label}`, label_font: "montserrat:12", label_color: "muted",
      bg: "transparent", padding_h: 6, padding_v: 4,
