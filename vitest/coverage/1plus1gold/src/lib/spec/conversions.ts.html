/**
 * SDUI spec for Account → 50g Bar Conversions (history).
 *
 * Read-only list surface: D1 data_source GET + repeater (D2). The genuinely-
 * missing iOS surface — iOS previously had only the WebView submission form and
 * no conversion history at all. Mirrors the PWA account "50g Bar Conversions"
 * section (src/app/account/page.tsx:3579-3786) exactly.
 *
 * Data: GET /api/conversions/my-requests (auth switched to getUserFromRequest →
 * Bearer+cookies; was cookie-only → iOS 401). Each request is enriched server-
 * side with pre-formatted display fields (refLabel / statusLabel /
 * statusColorToken / summaryPrefix+summaryMix / submittedLabel / feeSgd /
 * flattened lines + dispatchedBars / active-tracking booleans) so this spec
 * stays declarative — display logic lives in the shared lib helpers, not
 * duplicated in templates (memory/feedback_sdui_renderer_prop_gotchas).
 *
 * The "+ New Conversion" CTA navigates to /convert-to-combibar, which
 * InternalLinkRouter resolves to the existing WebView submission form (N1/N2) —
 * the payment-bearing form stays WebView for now (a separate, larger SDUI row).
 *
 * FULL PORT-DIFF vs PWA src/app/account/page.tsx:3579-3786 (Contract column):
 * | Element                | PWA                                   | iOS SDUI                                        | Contract |
 * |------------------------|---------------------------------------|-------------------------------------------------|----------|
 * | Header title           | "50g Bar Conversions" cinzel 18 bold  | text cinzel.bold:18 offwhite                    | T        |
 * | "+ New Conversion"     | <Link> → /convert-to-combibar gold     | button → navigate /convert-to-combibar          | N1/N2    |
 * | Loading                | conversionsLoading                    | text visible_when data._loading == true         | §12 D1   |
 * | Empty state            | Recycle + "No conversions yet." + CTA | vstack visible_when _loading==false & len==0     | §12 D7   |
 * | Row card               | .card (gold-top) rounded p-4          | card stroke:rule (auto gold top) corner 12      | C-card   |
 * | Reference              | #{id[0:8].upper} cinzel 14 bold gold   | text {item.refLabel}                            | T        |
 * | Status pill            | {color}22 bg / {color} / {color}44 brd | text color/bg@13/stroke@27 {statusColorToken}    | NEW status-pill |
 * | Summary line           | "{g}g splits → **{mix} bar**"         | rich_text [muted prefix + offwhite mix]         | P5       |
 * | Submitted date         | "Submitted {d Mon yyyy}" muted        | text {item.submittedLabel}                       | server-fmt |
 * | Fee paid               | "Fee paid" + Flag+fmt(fee) cinzel 16   | text + currency_amount {item.feeSgd}             | Currency |
 * | Splits list            | "Splits being converted" + rows       | repeater item.lines (label + qtyGramsLabel)     | D2       |
 * | Tracking links         | "Track pickup →" / "...delivery →"    | button→open_url visible_when active*            | open_url |
 * | Dispatched bars        | S/N list + Print delivery note btn    | repeater item.dispatchedBars (print DEFERRED)   | D2 (defer PDF) |
 * | Rejection reason       | "Reason: {x}" red                     | text hex:E05252 visible_when item.isRejected    | T        |
 *
 * Deferred (surfaced, not silently dropped): "Print delivery note" generates a
 * client-side jsPDF (lib/packing-slip) — no native equivalent without a server
 * PDF route; same class as the WebView/PDF boundary. The S/N LIST itself is
 * ported; only the print button is held back. The ?ref deep-link gold highlight
 * is deferred (deep-link only; no iOS entry path yet).
 */
 
import type { SDUISpec, SDUINode, SDUIAction } from "./types";
import { loadErrorBanner } from "./crud";
 
/** Section eyebrow ("SPLITS BEING CONVERTED" etc.) — montserrat 10 uppercase. */
function eyebrow(text: string): SDUINode {
  return { type: "text", props: { content: text, font: "montserrat:10", color: "muted", tracking: 1, text_case: "upper" } };
}
 
const divider: SDUINode = { type: "divider", props: { color: "rule" } };
 
/** A conversion's status pill — colour driven entirely by the server-computed
 *  item.statusColorToken (success | hex:E05252 | gold_mid | gold). bg @13 and
 *  border @27 mirror the PWA's {color}22 / {color}44. */
function statusPill(): SDUINode {
  return {
    type: "text",
    props: {
      content: "{item.statusLabel}",
      font: "montserrat.semibold:10",
      color: "{item.statusColorToken}",
      bg: "{item.statusColorToken}@13",
      stroke: "{item.statusColorToken}@27",
      corner_radius: 999, padding_h: 8, padding_v: 2,
    },
  };
}
 
/** Tappable "Track … →" link (gold text, no chrome) → opens the Lalamove
 *  share link in the system browser. */
function trackLink(label: string, urlToken: string, visibleWhen: string): SDUINode {
  return {
    type: "button",
    visible_when: visibleWhen,
    props: {
      label_text: label, label_font: "montserrat.semibold:12", label_color: "gold",
      bg: "transparent", full_width: false, padding_h: 0, padding_v: 0,
      action: { kind: "open_url", url: urlToken } as SDUIAction,
    },
  };
}
 
/** One conversion card (repeater item template). */
function conversionRow(): SDUINode {
  // Splits being converted — nested repeater over item.lines.
  const splitsSection: SDUINode = {
    type: "vstack",
    visible_when: "item.hasLines",
    props: { spacing: 6, alignment: "leading", max_width: "infinity" },
    children: [
      divider,
      eyebrow("Splits being converted"),
      {
        type: "repeater",
        props: { data_path: "item.lines", item_id_field: "id", spacing: 4 },
        children: [{
          type: "hstack",
          props: { spacing: 8 },
          children: [
            { type: "text", props: { content: "{item.label}", font: "montserrat:12", color: "offwhite" } },
            { type: "spacer", props: {} },
            { type: "text", props: { content: "{item.qtyGramsLabel}", font: "montserrat:12", color: "muted" } },
          ],
        }],
      },
    ],
  };
 
  // Tracking links (active Lalamove legs only).
  const trackingSection: SDUINode = {
    type: "vstack",
    visible_when: "item.hasTracking",
    props: { spacing: 6, alignment: "leading" },
    children: [
      divider,
      {
        type: "hstack",
        props: { spacing: 16 },
        children: [
          trackLink("Track pickup →", "{item.pickup_lalamove_share_link}", "item.activePickup"),
          trackLink("Track Combibar delivery →", "{item.dispatch_lalamove_share_link}", "item.activeDispatch"),
        ],
      },
    ],
  };
 
  // Combibars dispatched — header (eyebrow + "Print delivery note") + S/N list.
  const dispatchedSection: SDUINode = {
    type: "vstack",
    visible_when: "item.hasDispatched",
    props: { spacing: 6, alignment: "leading", max_width: "infinity" },
    children: [
      divider,
      {
        type: "hstack",
        props: { justify: "between", alignment: "center" },
        children: [
          eyebrow("Combibars dispatched"),
          // Opens the branded delivery note (shared buildDeliveryNoteHtml) in
          // Safari via a token-signed URL — no app session needed.
          {
            type: "button",
            visible_when: "item.deliveryNoteUrl != ''",
            props: {
              label_text: "Print delivery note", label_font: "montserrat.semibold:10", label_color: "gold",
              bg: "transparent", stroke: "gold_dim", stroke_width: 1, corner_radius: 6,
              padding_h: 8, padding_v: 4, full_width: false,
              action: { kind: "open_url", url: "{item.deliveryNoteUrl}" } as SDUIAction,
            },
          },
        ],
      },
      {
        type: "repeater",
        props: { data_path: "item.dispatchedBars", item_id_field: "id", spacing: 4 },
        children: [{
          type: "hstack",
          props: { spacing: 8 },
          children: [
            { type: "text", props: { content: "{item.gramsLabel}", font: "montserrat.semibold:12", color: "offwhite" } },
            { type: "spacer", props: {} },
            { type: "text", props: { content: "{item.serialLabel}", font: "montserrat:12", color: "muted" } },
          ],
        }],
      },
    ],
  };
 
  const rejectionSection: SDUINode = {
    type: "vstack",
    visible_when: "item.isRejected",
    props: { spacing: 6, alignment: "leading", max_width: "infinity" },
    children: [
      divider,
      { type: "text", props: { content: "Reason: {item.rejection_reason}", font: "montserrat:12", color: "hex:E05252", max_width: "infinity" } },
    ],
  };
 
  return {
    type: "card",
    props: { padding: 16, corner_radius: 12, bg: "bg_card", stroke: "rule", spacing: 10, alignment: "leading" },
    children: [
      // Header row: ref + status pill + summary + date  |  fee.
      {
        type: "hstack",
        props: { spacing: 12, alignment: "top" },
        children: [
          {
            type: "vstack",
            props: { spacing: 3, alignment: "leading", max_width: "infinity" },
            children: [
              {
                type: "hstack",
                props: { spacing: 6 },
                children: [
                  { type: "text", props: { content: "{item.refLabel}", font: "cinzel.bold:14", color: "gold" } },
                  statusPill(),
                ],
              },
              {
                type: "rich_text",
                props: {
                  font: "montserrat:12", color: "muted",
                  segments: [
                    { kind: "text", content: "{item.summaryPrefix}", font: "montserrat:12", color: "muted" },
                    { kind: "text", content: "{item.summaryMix}", font: "montserrat:12", color: "offwhite" },
                  ],
                },
              },
              { type: "text", visible_when: "item.submittedLabel", props: { content: "{item.submittedLabel}", font: "montserrat:11", color: "muted" } },
            ],
          },
          {
            type: "vstack",
            props: { spacing: 2, alignment: "trailing" },
            children: [
              eyebrow("Fee paid"),
              { type: "currency_amount", props: { sgd_value: "{item.feeSgd}", font: "cinzel.bold:16", color: "offwhite", flag_height: 12 } },
            ],
          },
        ],
      },
      splitsSection,
      trackingSection,
      dispatchedSection,
      rejectionSection,
    ],
  };
}
 
function header(): SDUINode {
  return {
    type: "hstack",
    props: { spacing: 12, alignment: "center" },
    children: [
      { type: "text", props: { content: "50g Bar Conversions", font: "cinzel.bold:18", color: "offwhite" } },
      { type: "spacer", props: {} },
      {
        type: "button",
        props: {
          label_text: "+ New Conversion", label_font: "montserrat.semibold:12",
          bg: "gold", label_color: "bg", full_width: false, corner_radius: 8, padding_h: 16, padding_v: 8,
          // Push the convert form onto the active tab's stack (normal screen +
          // native back) — NOT a root modal sheet. (Device 2026-06-08.)
          action: { kind: "open_conversion" },
        },
      },
    ],
  };
}
 
function emptyState(): SDUINode {
  return {
    type: "vstack",
    visible_when: "data._loading == false && data.requests.length == 0",
    props: { spacing: 12, alignment: "center", padding: 24, max_width: "infinity" },
    children: [
      { type: "icon", props: { name: "recycle", size: 36, color: "gold@30" } },
      { type: "text", props: { content: "No conversions yet.", font: "montserrat:14", color: "muted", alignment: "center" } },
      {
        type: "button",
        props: {
          label_text: "Convert splits to a 50g bar", label_font: "montserrat.semibold:14",
          bg: "gold", label_color: "bg", full_width: false, corner_radius: 8, padding_h: 24, padding_v: 10,
          // Push the convert form (normal screen + native back), not a sheet.
          action: { kind: "open_conversion" },
        },
      },
    ],
  };
}
 
export function conversionsSpec(): SDUISpec {
  return {
    surface: "conversions",
    spec_version: 1,
    data_source: "/api/conversions/my-requests",
    root: {
      type: "scroll",
      children: [
        {
          type: "vstack",
          props: { spacing: 16, padding: 16, alignment: "leading" },
          children: [
            header(),
            loadErrorBanner(),
            {
              type: "text",
              visible_when: "data._loading == true",
              props: { content: "Loading conversions…", font: "montserrat:14", color: "muted" },
            },
            emptyState(),
            {
              type: "repeater",
              props: { data_path: "data.requests", item_id_field: "id", spacing: 12 },
              children: [conversionRow()],
            },
          ],
        },
      ],
    },
  };
}
