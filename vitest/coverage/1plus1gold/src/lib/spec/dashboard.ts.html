/**
 * A stat TILE — PWA `.card rounded-xl p-4 text-center`: value on top (cinzel
 * gold), then the muted text-xs label, then an optional 10px tertiary sub.
 * stroke:"rule" gives the 2px gold top border app-wide. Centered.
 */
function tile(label: string, value: SDUINode, sub?: string): SDUINode {
  // fixed_size_v on label + sub so a wrapped 2-line sub claims its full height
  // and isn't clipped by the card (issue: "Avg Premium Paid truncated"). The
  // card uses fill_height so all tiles in a grid row share the row height
  // (issue: "tiles resize based on content").
  const kids: SDUINode[] = [
    value,
    text(label, "montserrat:12", "muted", { alignment: "center", fixed_size_v: true }),
  ];
  if (sub) kids.push(text(sub, "montserrat:10", "muted@70", { alignment: "center", fixed_size_v: true }));
  return {
    type: "card",
    props: { stroke: "rule", bg: "bg_card", corner_radius: 12, padding: 16, alignment: "center", spacing: 4, fill_height: true },
    children: kids,
  };
}
 
/** A column grid of tiles (PWA grid-cols-N gap-3). */
const gridN = (columns: number, children: SDUINode[]): SDUINode => ({
  type: "grid", props: { columns, spacing: 12 }, children,
});
 
/** A transparent section (PWA <details>): gold-dim summary underline + body. */
function section(title: string, open: boolean, children: SDUINode[], visibleWhen?: string): SDUINode {
  return {
    type: "disclosure",
    ...(visibleWhen ? { visible_when: visibleWhen } : {}),
    props: { title, title_case: "upper", open },
    children: [{ type: "vstack", props: { spacing: 12, alignment: "leading" }, children }],
  };
}

/**
 * "Your cost of living, in gold" return-hook card. Fed by gl_* on
 * /api/account/dashboard. Composed only of nodes already in the binary
 * (card/text/grams_text/currency_amount/grid/button) so it renders NATIVE with
 * NO build. A 1:1 mirror of the web BudgetCompare (src/components/gold-life):
 * two gold-top-accent tiles, each self-contained (label · its OWN date · grams ·
 * SGD value), then a coloured gold-move chip + the delta line. OPEN by default,
 * in the dashboard's own disclosure rhythm. Wording is about "the same monthly
 * budget", never "your life". Edit → the resident Calculator's Daily Life tab.
 * Redesign 2026-08-31 (Mukesh #1-#7). Port-diff: SDUI contract §96.
 */
function goldLifeCard(): SDUINode {
  // Edit → TAB SWITCH to the resident Calculator tab on the Daily Life subtab
  // (NOT a modal navigate, which opened a new page). `subtab` is seeded on the
  // bus before the switch; the Calculator consumes it (SDUIView §4-consume).
  const editAction = { kind: "tabswitch", target: "calculator", subtab: "life" };
  const editBtn: SDUINode = {
    type: "button",
    props: {
      label_text: "Edit my budget →", label_font: "montserrat.semibold:12", label_color: "gold",
      bg: "transparent", full_width: false, padding_h: 0, padding_v: 2, action: editAction,
    },
  };
  const colGrams = (path: string): SDUINode =>
    ({ type: "grams_text", props: { value: `{${path}}`, num_font: "cinzel.bold:24", unit_font: "montserrat:11", color: "gold", alignment: "center" } });
  // Money value in the user's display currency + SGD base (dashboard standard).
  const value = (path: string): SDUINode =>
    ({ type: "currency_amount", props: { sgd_value: `{${path}}`, font: "montserrat.semibold:12", color: "offwhite", placeholder: "—", show_sgd_base: true, alignment: "center" } });
  // One tile: label · its OWN date · grams · optional SGD value. stroke:"rule"
  // gives the 2px gold top accent (matches web's borderTop). The date sits IN
  // the tile so "when saved" and "today" read as different points in time (#3).
  const compareCol = (label: string, datePath: string, gramsPath: string, valuePath?: string): SDUINode => ({
    type: "card",
    props: { stroke: "rule", bg: "bg_card", corner_radius: 12, padding: 14, alignment: "center", spacing: 2, fill_height: true },
    children: [
      text(label, "montserrat:11", "muted", { alignment: "center", fixed_size_v: true }),
      text(`{${datePath}}`, "montserrat:10", "muted", { alignment: "center", fixed_size_v: true }),
      colGrams(gramsPath),
      ...(valuePath ? [value(valuePath)] : []),
    ],
  });
  // Coloured gold-move chip — success green when gold rose (same budget costs
  // less gold), a warm hex when it fell. Two gated variants bound to the same
  // server string (colour can't be data-driven, so we gate on gl_up).
  const moveChip = (up: boolean): SDUINode => {
    const color = up ? "success" : "hex:C1785A";
    return {
      type: "text",
      visible_when: `data.gl_has_move == true && data.gl_up == ${up}`,
      props: {
        content: "{data.gl_move_badge}", font: "montserrat.semibold:11", color,
        bg: `${color}@15`, padding_h: 8, padding_v: 2, corner_radius: 10, multiline: false,
      },
    };
  };
  const hookLine = text("{data.gl_hook_text}", "montserrat:12", "muted", { multiline: true });
  // Duration made explicit — this is a MONTHLY budget (#2).
  const durationCaption = text("Your monthly budget", "montserrat:11", "muted");
 
  // OPEN by default (#5), renamed "Your cost of living, in gold" (#4).
  return section("Your cost of living, in gold", true, [
    // READY — SIDE-BY-SIDE comparison (when saved vs today) + coloured move chip.
    {
      type: "vstack", visible_when: 'data.gl_state == "ready"',
      props: { spacing: 10, alignment: "leading" },
      children: [
        durationCaption,
        gridN(2, [
          compareCol("When you saved", "data.gl_saved_date", "data.gl_snapshot_grams", "data.gl_snapshot_value_sgd"),
          compareCol("Today", "data.gl_today_date", "data.gl_today_grams", "data.gl_today_value_sgd"),
        ]),
        { type: "hstack", props: { spacing: 8, alignment: "center" }, children: [moveChip(true), moveChip(false), hookLine] },
        editBtn,
      ],
    },
    // PENDING — saved, but live pricing briefly unavailable (grams-then only).
    {
      type: "vstack", visible_when: 'data.gl_state == "pending"',
      props: { spacing: 10, alignment: "leading" },
      children: [
        durationCaption,
        compareCol("When you saved", "data.gl_saved_date", "data.gl_snapshot_grams", "data.gl_snapshot_value_sgd"),
        hookLine,
        editBtn,
      ],
    },
    // NONE — no saved budget: a CTA into the Daily Life tab.
    {
      type: "vstack", visible_when: 'data.gl_state == "none"',
      props: { spacing: 8, alignment: "leading" },
      children: [
        text("See a month of everyday spending weighed in gold — and how it changes as gold moves.", "montserrat:12", "muted", { multiline: true }),
        {
          type: "button",
          props: {
            label_text: "Try the Daily Life tool →", label_font: "montserrat.semibold:12", label_color: "gold",
            bg: "transparent", full_width: false, padding_h: 0, padding_v: 2, action: editAction,
          },
        },
      ],
    },
  ]);
}
 
/**
 * "A word from you" — the review ask. Fed by `rv_state` on
 * /api/account/dashboard, which is `"ask"` only when an order has actually
 * reached this customer and they have not dismissed the card. The whole
 * disclosure is gated on that, so a customer who should not be asked sees
 * nothing at all rather than an empty section.
 *
 * ▶ NO APP STORE BUILD, and that is measured rather than assumed. Enumerated
 * from the shipped binary on 2026-09-17: SDUIRenderer registers 82 node types
 * and SDUIAction 23 actions. This card uses disclosure / vstack / hstack / text
 * / button, and open_url / navigate / api_call / refetch — all present. It
 * reaches every installed build, App Store included, on the next spec fetch.
 *
 * ▶ PORT-DIFF (web src/components/account/ReviewPromptCard.tsx → SDUI):
 *
 *  | # | Web                                          | SDUI / native                                   | Basis |
 *  |---|----------------------------------------------|--------------------------------------------------|-------|
 *  | 1 | `<details class="account-accordion" open>` + cinzel summary + gold-dim underline | `section(title, open=true)` → `disclosure`, `title_case:"upper"` | §96 (goldLifeCard uses the identical shell) |
 *  | 2 | component returns null when state !== "ask"   | `visible_when: 'data.rv_state == "ask"'` on the disclosure | SDUIBinding `visible_when` |
 *  | 3 | `font-montserrat text-[13px]` muted, line-height 1.5 | `text(..., "montserrat:13", "muted", { multiline: true })` | T0 font-token grammar (arbitrary size) |
 *  | 4 | `<a target="_blank">` gold `text-[12px] font-semibold` | `button` + `action:{kind:"open_url"}`, `label_font:"montserrat.semibold:12"`, `label_color:"gold"`, `bg:"transparent"`, `padding_h:0`, `padding_v:2` | §96 editBtn shape; `.openURL` → Safari, same as SDUIFounderCard's LinkedIn link |
 *  | 5 | `<Link href="/contact">` muted 12px             | `button` + `action:{kind:"navigate", url:"/contact"}` | VERIFIED in-app: InternalLinkRouter:29 maps "/contact" → `.sduiSurface(name:"contact")`, so this does NOT eject to Safari |
 *  | 6 | two links on one wrapping row (`gap-x-5`)     | `hstack` spacing 16                              | H-row |
 *  | 7 | disclosure line + "Not now" on a `justify-between` row | `hstack` spacing 12 (leading), same two children | see divergence (b) |
 *  | 8 | dismiss → optimistic local hide, then POST     | `api_call` POST → `onSuccess:[{refetch}]`        | D5 (api_call carries the bearer token at fire time) + D4 (refetch reloads the data_source) |
 *
 * ▶ TWO DECLARED DIVERGENCES, both forced rather than chosen:
 *   (a) DISMISS IS SERVER-TRUTH ON NATIVE, OPTIMISTIC ON WEB. `refetch` re-pulls
 *       the dashboard and `rv_state` flips to "none", which hides the card —
 *       strictly better than the web's optimistic hide. The web card cannot do
 *       the same cheaply because it composes from its own small endpoint rather
 *       than from this payload, so it hides locally and lets the next page load
 *       be the source of truth. Same outcome, different mechanism.
 *   (b) NO UNDERLINE ON "Not now". The web styles it `underline
 *       underline-offset-2`; `SDUIButton` has no `label_underline` prop, and
 *       adding one is a native change, i.e. an App Store build for a decoration.
 *       A muted tappable label is the correct SwiftUI idiom regardless.
 *   Also (not a divergence, a consequence): the web row is `justify-between`,
 *   the native one is leading-aligned — `hstack` has no space-between, and the
 *   two items read fine adjacent.
 *
 * ▶ WHAT THIS MUST NOT BECOME: a sentiment funnel. The review link and the
 * "tell us first" link are shown to EVERY eligible customer, side by side, with
 * no branch on anything we know about them. Sorting customers by expected
 * happiness is review gating — against Trustpilot's guidelines and the thing
 * that makes a score worthless. See src/lib/account/review-prompt.ts.
 */
function reviewCard(): SDUINode {
  // ▶ `mark` puts Trustpilot's OWN LOCKUP above the label — the same raster the
  // Home trust band draws, from the same shared constants, never a second copy
  // of a trademark.
  //
  // ⚠️ CHANGED 2026-09-18, AND THE WEB CARD CHANGED IN THE SAME COMMIT. This
  // drew the rasterised STAR inline before the text, with "Trustpilot" carried
  // in OUR Montserrat inside the label. The operator put the two surfaces side
  // by side — *"I was just looking at the inline trustpilot in Account. I think
  // we should make it look same as its in the footer with correct logo and
  // Text"* — and he is right: the star raster is a flat single-green
  // reconstruction, while Trustpilot's real star is two-tone. See
  // lib/trustpilot.ts §WHAT USES WHICH for the corrected assignment.
  //
  // ▶ IT MUST DROP `label_text`, not sit beside it. SDUIButton takes the
  // convenience label OR children, and `label_text` WINS (`if let labelText …
  // else { ForEach(contentChildren()) }`), so leaving it set would render the
  // text alone and silently throw the mark away.
  //
  // ▶ The lockup is NOT tinted by `label_color`: that is applied to the Text
  // specifically, and children render through SDUIRenderer untouched — an
  // AsyncImage decodes as `.original`, not as a template. So the artwork
  // survives inside a gold-labelled button. (It is the WHITE lockup, which is
  // correct on this card's dark ground — same file the footer serves.)
  const linkBtn = (label: string, action: Record<string, unknown>, color: string, mark = false): SDUINode => {
    const font = color === "gold" ? "montserrat.semibold:12" : "montserrat:12";
    const base = { bg: "transparent", full_width: false, padding_h: 0, padding_v: 2, action };
    if (!mark) {
      return { type: "button", props: { ...base, label_text: label, label_font: font, label_color: color } };
    }
    return {
      type: "button",
      props: base,
      children: [
        {
          // ▶ STACKED, NOT INLINE — mirrors the web card's `flex-col
          // items-start gap-1.5`. `alignment: "leading"` is SDUIVStack's
          // default and is PWA's items-start; `spacing: 6` is gap-1.5 at L4's
          // 1:1 gap→spacing mapping.
          //
          // ▶ THE STACK IS WHAT MAKES THE FULL LOCKUP USABLE HERE. At 110×27 it
          // cannot sit inline beside 12px type without dwarfing it, which is
          // exactly why the earlier reasoning reached for the star. Given its
          // own line it is the footer's own composition, and the size stays the
          // footer's — Trustpilot's digital guidance puts a ~100px floor under
          // the horizontal lockup, so scaling it down to fit a line would break
          // the one published rule we can actually read.
          type: "vstack",
          props: { spacing: 6, alignment: "leading" },
          children: [
            // `bg` explicit: the image node defaults to bg_card, which would
            // draw a block behind the lockup on the card's own ground.
            //

/** A tile that itself only shows when a predicate holds (e.g. "In Progress"). */
function tileCond(label: string, value: SDUINode, sub: string, visibleWhen: string): SDUINode {
  const t = tile(label, value, sub) as SDUINode & { visible_when?: string };
  t.visible_when = visibleWhen;
  return t;
}
