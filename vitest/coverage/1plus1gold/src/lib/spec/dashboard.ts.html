/**
 * SDUI spec for the Account → Dashboard surface (Q-Dashboard).
 *
 * Per the locked SDUI-default rule: an SDUI screen with ONE embedded chart
 * PANEL (Path B) + SDUI stat sections (Path A).
 *   • Chart panel — web_panel → /embed/account/portfolio-charts (hero value +
 *     P/L + value-vs-cost area + composition donut + avg-cost sparkline; the 3
 *     Recharts charts, chart rule).
 *   • 7 disclosure sections of stat TILES — Premium Curve Journey (open) ·
 *     Standard Holdings · Accessories · Overall Portfolio · Sell-to-Us
 *     Activity · 50g Bar Conversions. data_source = /api/account/dashboard
 *     (shared derivePortfolio aggregate — same numbers as the PWA page).
 *
 * REBUILT 2026-06-02 after device review (Mukesh). Prior build flattened every
 * stat into a leading vstack ("accepted divergence") and dropped tiles +
 * sections. Now FAITHFUL to the PWA:
 *   • Each stat = a `.card` TILE (bg-card + rule border + 2px gold top, via
 *     SDUICard's stroke=="rule" auto-top-accent), centered, value-on-top —
 *     matching account/page.tsx:1183-1210 (issue i + vii).
 *   • `grid` node lays tiles in PWA's columns (3 for the curve-cycle row, 2
 *     elsewhere) instead of a flat stack.
 *   • Sections gated by `> 0` now actually appear (the evalPredicate
 *     relational-operator fix — issue iv/v).
 *   • §3 Overall Portfolio regains "Avg. Cost / Gram"; §5 Conversions regains
 *     "Bars Received" (issue v).
 *   • The green "milestone reached" box (PHYSICAL/SPLITS + Convert CTA) is
 *     ported (issue ii-b); progress bar uses the rich single-bar mode (ii-c).
 *
 * PORT-DIFF (element-level) + Contract: docs/sdui-pwa-port-contract.md §18.
 * Anchored to account/page.tsx:1140-1551 + GoldProgressBar.tsx (verbatim copy).
 */
 
import type { SDUISpec, SDUINode } from "./types";
 
const EP = "/api/account/dashboard";
 
const text = (content: string, font: string, color: string, extra: Record<string, unknown> = {}): SDUINode => ({
  type: "text", props: { content, font, color, ...extra },
});
 
// ── Value primitives (cinzel.bold:20 ≈ PWA text-xl, gold) ───────────────────
const grams = (path: string): SDUINode => ({ type: "grams_text", props: { value: `{${path}}`, num_font: "cinzel.bold:20", color: "gold", alignment: "center" } });
const money = (path: string): SDUINode => ({ type: "currency_amount", props: { sgd_value: `{${path}}`, font: "cinzel.bold:20", color: "gold", placeholder: "—", show_sgd_base: true } });
const pct   = (path: string): SDUINode => ({ type: "number_text", props: { value: `{${path}}`, suffix: "%", placeholder: "—", num_font: "cinzel.bold:20", color: "gold", alignment: "center" } });
const countText = (tmpl: string): SDUINode => text(tmpl, "cinzel.bold:20", "gold", { alignment: "center" });
 
/**
 * A stat TILE — PWA `.card rounded-xl p-4 text-center`: value on top (cinzel
 * gold), then the muted text-xs label, then an optional 10px tertiary sub.
 * stroke:"rule" gives the 2px gold top border app-wide. Centered.
 */
function tile(label: string, value: SDUINode, sub?: string): SDUINode {
  // fixed_size_v on label + sub so a wrapped 2-line sub claims its full height
  // and isn't clipped by the card (issue: "Avg Premium Paid truncated"). The
  // card uses fill_height so all tiles in a grid row share the row height
  // (issue: "tiles resize based on content").
  const kids: SDUINode[] = [
    value,
    text(label, "montserrat:12", "muted", { alignment: "center", fixed_size_v: true }),
  ];
  if (sub) kids.push(text(sub, "montserrat:10", "muted@70", { alignment: "center", fixed_size_v: true }));
  return {
    type: "card",
    props: { stroke: "rule", bg: "bg_card", corner_radius: 12, padding: 16, alignment: "center", spacing: 4, fill_height: true },
    children: kids,
  };
}
 
/** A column grid of tiles (PWA grid-cols-N gap-3). */
const gridN = (columns: number, children: SDUINode[]): SDUINode => ({
  type: "grid", props: { columns, spacing: 12 }, children,
});
 
/** A transparent section (PWA <details>): gold-dim summary underline + body. */
function section(title: string, open: boolean, children: SDUINode[], visibleWhen?: string): SDUINode {
  return {
    type: "disclosure",
    ...(visibleWhen ? { visible_when: visibleWhen } : {}),
    props: { title, title_case: "upper", open },
    children: [{ type: "vstack", props: { spacing: 12, alignment: "leading" }, children }],
  };
}
 
/** A small status badge (PWA inline pill). */
const badge = (label: string, color: string): SDUINode =>
  text(label, "montserrat.semibold:10", color, { bg: `${color}@15`, padding_h: 6, padding_v: 2, corner_radius: 4, multiline: false });
 
/**
 * "Your cost of living, in gold" return-hook card. Fed by gl_* on
 * /api/account/dashboard. Composed only of nodes already in the binary
 * (card/text/grams_text/currency_amount/grid/button) so it renders NATIVE with
 * NO build. A 1:1 mirror of the web BudgetCompare (src/components/gold-life):
 * two gold-top-accent tiles, each self-contained (label · its OWN date · grams ·
 * SGD value), then a coloured gold-move chip + the delta line. OPEN by default,
 * in the dashboard's own disclosure rhythm. Wording is about "the same monthly
 * budget", never "your life". Edit → the resident Calculator's Daily Life tab.
 * Redesign 2026-08-31 (Mukesh #1-#7). Port-diff: SDUI contract §96.
 */
function goldLifeCard(): SDUINode {
  // Edit → TAB SWITCH to the resident Calculator tab on the Daily Life subtab
  // (NOT a modal navigate, which opened a new page). `subtab` is seeded on the
  // bus before the switch; the Calculator consumes it (SDUIView §4-consume).
  const editAction = { kind: "tabswitch", target: "calculator", subtab: "life" };
  const editBtn: SDUINode = {
    type: "button",
    props: {
      label_text: "Edit my budget →", label_font: "montserrat.semibold:12", label_color: "gold",
      bg: "transparent", full_width: false, padding_h: 0, padding_v: 2, action: editAction,
    },
  };
  const colGrams = (path: string): SDUINode =>
    ({ type: "grams_text", props: { value: `{${path}}`, num_font: "cinzel.bold:24", unit_font: "montserrat:11", color: "gold", alignment: "center" } });
  // Money value in the user's display currency + SGD base (dashboard standard).
  const value = (path: string): SDUINode =>
    ({ type: "currency_amount", props: { sgd_value: `{${path}}`, font: "montserrat.semibold:12", color: "offwhite", placeholder: "—", show_sgd_base: true, alignment: "center" } });
  // One tile: label · its OWN date · grams · optional SGD value. stroke:"rule"
  // gives the 2px gold top accent (matches web's borderTop). The date sits IN
  // the tile so "when saved" and "today" read as different points in time (#3).
  const compareCol = (label: string, datePath: string, gramsPath: string, valuePath?: string): SDUINode => ({
    type: "card",
    props: { stroke: "rule", bg: "bg_card", corner_radius: 12, padding: 14, alignment: "center", spacing: 2, fill_height: true },
    children: [
      text(label, "montserrat:11", "muted", { alignment: "center", fixed_size_v: true }),
      text(`{${datePath}}`, "montserrat:10", "muted", { alignment: "center", fixed_size_v: true }),
      colGrams(gramsPath),
      ...(valuePath ? [value(valuePath)] : []),
    ],
  });
  // Coloured gold-move chip — success green when gold rose (same budget costs
  // less gold), a warm hex when it fell. Two gated variants bound to the same
  // server string (colour can't be data-driven, so we gate on gl_up).
  const moveChip = (up: boolean): SDUINode => {
    const color = up ? "success" : "hex:C1785A";
    return {
      type: "text",
      visible_when: `data.gl_has_move == true && data.gl_up == ${up}`,
      props: {
        content: "{data.gl_move_badge}", font: "montserrat.semibold:11", color,
        bg: `${color}@15`, padding_h: 8, padding_v: 2, corner_radius: 10, multiline: false,
      },
    };
  };
  const hookLine = text("{data.gl_hook_text}", "montserrat:12", "muted", { multiline: true });
  // Duration made explicit — this is a MONTHLY budget (#2).
  const durationCaption = text("Your monthly budget", "montserrat:11", "muted");
 
  // OPEN by default (#5), renamed "Your cost of living, in gold" (#4).
  return section("Your cost of living, in gold", true, [
    // READY — SIDE-BY-SIDE comparison (when saved vs today) + coloured move chip.
    {
      type: "vstack", visible_when: 'data.gl_state == "ready"',
      props: { spacing: 10, alignment: "leading" },
      children: [
        durationCaption,
        gridN(2, [
          compareCol("When you saved", "data.gl_saved_date", "data.gl_snapshot_grams", "data.gl_snapshot_value_sgd"),
          compareCol("Today", "data.gl_today_date", "data.gl_today_grams", "data.gl_today_value_sgd"),
        ]),
        { type: "hstack", props: { spacing: 8, alignment: "center" }, children: [moveChip(true), moveChip(false), hookLine] },
        editBtn,
      ],
    },
    // PENDING — saved, but live pricing briefly unavailable (grams-then only).
    {
      type: "vstack", visible_when: 'data.gl_state == "pending"',
      props: { spacing: 10, alignment: "leading" },
      children: [
        durationCaption,
        compareCol("When you saved", "data.gl_saved_date", "data.gl_snapshot_grams", "data.gl_snapshot_value_sgd"),
        hookLine,
        editBtn,
      ],
    },
    // NONE — no saved budget: a CTA into the Daily Life tab.
    {
      type: "vstack", visible_when: 'data.gl_state == "none"',
      props: { spacing: 8, alignment: "leading" },
      children: [
        text("See a month of everyday spending weighed in gold — and how it changes as gold moves.", "montserrat:12", "muted", { multiline: true }),
        {
          type: "button",
          props: {
            label_text: "Try the Daily Life tool →", label_font: "montserrat.semibold:12", label_color: "gold",
            bg: "transparent", full_width: false, padding_h: 0, padding_v: 2, action: editAction,
          },
        },
      ],
    },
  ]);
}

/** A tile that itself only shows when a predicate holds (e.g. "In Progress"). */
function tileCond(label: string, value: SDUINode, sub: string, visibleWhen: string): SDUINode {
  const t = tile(label, value, sub) as SDUINode & { visible_when?: string };
  t.visible_when = visibleWhen;
  return t;
}
