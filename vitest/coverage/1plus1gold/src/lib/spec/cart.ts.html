/**
 * SDUI spec for the Cart surface (Phase 6, contract §25).
 *
 * Source-verified port of src/app/cart/page.tsx. The cart LINES come from the
 * client CartStore; the renderer (data_source_method "cart_local") GETs
 * /api/cart/context once for the per-user inputs, then computes the data
 * envelope ON-DEVICE via engine.cartView(CartStore.lines, ctx) — re-running
 * reactively whenever the cart changes. The whole envelope re-derives on every
 * cart mutation, so qty/gift/milestone edits update line totals + summary + gate
 * together with NO per-row recompute plumbing.
 *
 * Data envelope = CartView (lib/cart-core): { lines[], summary{}, milestoneGate{},
 * fulfillment{}, accessoryUnlocks[], progress{}, kycNudge{}, priceBlocked,
 * checkout{} }. Money is SGD → currency_amount converts to the display currency
 * (Rule #1). Every grams/curve-position/progress/unlock string is preformatted
 * in the kernel (the "format in the kernel" gotcha).
 *
 * Native escape-hatches: gift_panel (1g-curve gift input → CartStore), and the
 * `cart_action` / `apply_promo` / `apply_referral` native actions. Charge stays
 * server-authoritative (orders/create recomputes + rejects mismatch > 1¢).
 *
 * VERIFIED node props (against the Swift renderers, no assumptions): text
 * container styling (stroke/bg/corner_radius/padding_*); currency_amount
 * (sgd_value template, suffix, font, color, placeholder); kv_row
 * (label/value/label_color/value_color); button (label_*, bg/stroke, action,
 * bg_when/label_color_when, leading_icon, full_width); conditional (cases/else);
 * repeater (data_path/item_id_field/axis/spacing); card (accent_leading/width);
 * async_image (src + content_mode "fit"); progress_bar; icon (name).
 */

/** Money (SGD value → display currency via Rule #1). `visible_when` is a
 *  NODE-LEVEL field (SDUIRenderer reads node.visibleWhen) — hoist it out of
 *  props or it's IGNORED (fail-open). */
const money = (sgdPath: string, font: string, color: string, extra: Record<string, unknown> = {}): SDUINode => {
  const { visible_when, ...rest } = extra;
  return { type: "currency_amount", ...(visible_when ? { visible_when: visible_when as string } : {}), props: { sgd_value: sgdPath, font, color, ...rest } };
};

// ── Cart item card (repeater over data.lines) ───────────────────────────────
// Redesigned 2026-06-05 (Mukesh: relook, better than the cramped PWA 3-column).
// A single 64px image + a full-width details column (label+trash row, wrapped
// badges, price-each, then a qty-stepper row with the line total trailing).
// Stepper uses SF minus/plus GLYPHS (reliable; the "−"/"+" text didn't render).
function itemCard(): SDUINode {
  return {
    type: "card",
    props: { stroke: "rule", bg: "bg_card", corner_radius: 12, padding: 16, alignment: "leading", spacing: 12 },
    children: [
      { type: "hstack", props: { spacing: 12, alignment: "top" }, children: [
        // Single product image (fallback {g}g chip when none). Gated on thumbs
        // length (a reliable array predicate; imageUrl can be null).
        { type: "async_image", visible_when: "item.thumbs.length > 0",
          props: { src: "{item.imageUrl}", width: 64, height: 64, corner_radius: 6, content_mode: "fit", bg: "bg", stroke: "rule", stroke_width: 1 } },
        text("{item.g}g", "cinzel.bold:14", "gold",
          { visible_when: "item.thumbs.length == 0 && item.g > 0", bg: "bg", stroke: "rule", stroke_width: 1, corner_radius: 6, padding_h: 18, padding_v: 22 }),
        // Details — full remaining width.
        { type: "vstack", props: { spacing: 6, alignment: "leading", max_width: "infinity" }, children: [

          text("{item.label}", "cinzel.semibold:15", "gold", { fixed_size_v: true }),

            badge("{item.premiumLabel}", "gold_mid", "gold_dim", "item.premiumLabel != null && item.isStandardPremium == false"),
            badge("{item.premiumLabel}", "standard_blue", "standard_blue", "item.premiumLabel != null && item.isStandardPremium == true"),
            { type: "hstack", visible_when: "item.accessory.hasFree == true", props: { spacing: 4, alignment: "center", bg: "hex:9AD78F@20", corner_radius: 4, padding_h: 6, padding_v: 2 }, children: [
              { type: "icon", props: { name: "gift", size: 10, color: "hex:9AD78F" } },
              text("{item.accessory.freeQty} free", "montserrat:10", "hex:9AD78F"),
            ] },
          ] },
          // Price-each row (per type)
          { type: "hstack", props: { spacing: 6, alignment: "center", axis: "wrap" }, children: [
            // Accessory free split: "{freeQty} free + {paidQty} × $each"
            { type: "hstack", visible_when: "item.accessory.hasFree == true", props: { spacing: 4, alignment: "center" }, children: [
              text("{item.accessory.freeQty} free", "montserrat:12", "hex:9AD78F"),
              text("+ {item.accessory.paidQty} ×", "montserrat:12", "offwhite", { visible_when: "item.accessory.paidQty > 0" }),
              money("{item.accessory.paidPriceEach}", "montserrat:12", "offwhite", { visible_when: "item.accessory.paidQty > 0" }),
            ] },
            // Accessory discounted (not free): "$each each  N% off"
            { type: "hstack", visible_when: "item.accessory.isDiscounted == true", props: { spacing: 4, alignment: "center" }, children: [
              money("{item.accessory.paidPriceEach}", "montserrat:12", "offwhite", { suffix: " each" }),
              text("{item.accessory.discountPct}% off", "montserrat:10", "hex:9AD78F", { bg: "hex:9AD78F@20", corner_radius: 4, padding_h: 6, padding_v: 2 }),
            ] },
            // Curve / standard / plain accessory: "$each each" (or "—")
            // Paren-free (SDUI evaluator has no parens — SDUIBinding.swift). `A || (B && C)`
            // works without parens since || is loosest: `A || B && C` groups as A || (B && C).
            money("{item.priceEach}", "montserrat:12", "offwhite",
              { suffix: " each", placeholder: "—", visible_when: "item.accessory == null || item.accessory.hasFree == false && item.accessory.isDiscounted == false" }),
          ] },
          text("Ships when in stock", "montserrat:10", "muted", { visible_when: "item.isPresale == true" }),
        ] },
        // Trailing column — PWA `flex-col items-end shrink-0` (cart/page.tsx:696-748):
        // stepper+trash, line total, per-gram — stacked + right-aligned. NO max_width,
        // so the column sizes to its content (shrink-0) and the DETAILS column shrinks
        // instead. THE line-total fix: previously the total sat in a justify:between
        // row BESIDE the stepper, so currency_amount's minScale 0.6 collapsed it to
        // ~10pt for curve/standard (long value + per-gram sub-line squeezing the
        // shared vstack); accessories (no sub-line) didn't. In a shrink-0 column the
        // total gets its intrinsic width → renders at full montserrat.semibold:14
        // (= PWA text-sm font-semibold, cart/page.tsx:740) for EVERY item type.
        { type: "vstack", props: { spacing: 8, alignment: "trailing" }, children: [
          { type: "hstack", props: { spacing: 4, alignment: "center" }, children: [ stepper(), trashBtn() ] },
          // Stock cap hint — parity with the web cart ("Max available — N in stock").
          text("Max available — {item.stockLeft} in stock", "montserrat:10", "muted", { visible_when: "item.atStockCap == true" }),
          money("{item.lineSgd}", "montserrat.semibold:14", "offwhite", { placeholder: "—" }),
          { type: "hstack", visible_when: "item.g > 0", props: { spacing: 6, alignment: "center" }, children: [

            money("{item.perGramSgd}", "montserrat:12", "muted", { suffix: "/g", visible_when: "item.perGramSgd != null" }),
          ] },
        ] },
      ] },
      // Gift panel — only for 1g curve (native input → CartStore).
      { type: "gift_panel", visible_when: "item.isGiftable == true", props: { item_key: "{item.itemKey}" } },
    ],
  };
}

// ── Expandable promo / referral panel (local state + native validate) ───────

          } },

      ] },
      // Validate error/success message (flat string key; hidden when empty).
      text(`{state.${msgKey}}`, "montserrat:10", "hex:e57373", { visible_when: `state.${msgKey} != '' && state.${okKey} == false` }),
