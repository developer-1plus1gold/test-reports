import type { SDUISpec, SDUINode } from "./types";
 
function text(content: string, font: string, color: string, extra: Record<string, unknown> = {}): SDUINode {
  const { visible_when, ...rest } = extra as { visible_when?: string };
  return { type: "text", ...(visible_when ? { visible_when } : {}), props: { content, font, color, ...rest } };
}
 
/** Flag+symbol+number money value, with the "≈ S$X" SGD base beneath (PWA's
 *  ubiquitous `{fmt}` + `flagCode!=="sg" && ≈ {sgdNum}` pattern). */
function money(sgdPath: string, font: string, color: string, extra: Record<string, unknown> = {}): SDUINode {
  const { visible_when, ...rest } = extra as { visible_when?: string };
  return { type: "currency_amount", ...(visible_when ? { visible_when } : {}),
    props: { sgd_value: sgdPath, font, color, show_sgd_base: true, flag_height: 10, placeholder: "—", ...rest } };
}
 
/** Single-select filter chip (PWA gain/product chips) — gold-tint when active,
 *  muted otherwise. Driven by `activePredicate`; tap sets `key` = `value`. */
function chip(label: string, key: string, value: string): SDUINode {
  // Guardrail: `key`/`value` are interpolated into an SDUI predicate string, so
  // they must be predicate-safe. Every current call site passes a hardcoded
  // literal ("gainFilter"/"all"/…), but assert it so a future dynamic call site
  // (e.g. a DB-sourced label) fails the build/test instead of silently building
  // an injectable or broken predicate. (security_agent 2026-06-08 LOW.)
  Iif (!/^[A-Za-z_][A-Za-z0-9_]*$/.test(key)) throw new Error(`chip(): unsafe state key "${key}"`);
  Iif (/['"\\]/.test(value)) throw new Error(`chip(): unsafe predicate value "${value}"`);
  const active = `state.${key} == '${value}'`;

/** A small SF-symbol radio (PWA's 16pt border circle + gold dot). Filled when
 *  `activePred` holds, hollow when `inactivePred` holds. The unselected ring is
 *  `muted` (NOT `rule` — `rule` is near-black on the dark card → invisible, the
 *  "no radio" device report; PWA's 2px border reads as a ring, a 1px SF stroke
 *  needs a brighter colour). */
function radio(activePred: string, inactivePred: string, size = 16): SDUINode[] {
  return [
    { type: "icon", props: { name: "circle", size, color: "muted" }, visible_when: inactivePred },
    { type: "icon", props: { name: "circle.inset.filled", size, color: "gold" }, visible_when: activePred },
  ];
}
 
/** Top + bottom summary panel (PWA SummaryPanel 259-310) — Selected / Total Gold
 *  / Est. Payout, with a Continue CTA. Visible only when something is selected. */
function summaryPanel(): SDUINode {
  // Eyebrow that never wraps ("SELECTE D" bug) — fixed_size_h keeps it 1 line.
  const eb = (s: string): SDUINode => text(s, "montserrat:10", "muted", { tracking: 1.5, fixed_size_h: true });
  // A stat column with an attached left divider (PWA borderLeft) so the divider
  // travels with the column when the group wraps.
  const divCol = (label: string, valueNode: SDUINode): SDUINode => ({
    type: "hstack", props: { spacing: 12, alignment: "center" }, children: [
      { type: "divider", props: { color: "gold@20", vertical: true, thickness: 1, padding_v: 2 } },
      { type: "vstack", props: { spacing: 0, alignment: "leading" }, children: [eb(label), valueNode] },
    ],
  });
  return {
    type: "card",   // stacks have no bg/stroke — the tinted panel must be a card
    visible_when: "data.selectedCount > 0",
    props: { bg: "gold@6", stroke: "gold@25", corner_radius: 12, padding_h: 16, padding_v: 12 },
    children: [
      // Outer: left stat group (wraps, PWA flex-wrap) + Continue (shrink-0,
      // nowrap — never collapse to vertical letters). max_width:infinity on the
      // group pushes Continue to the right (PWA justify-between).
      { type: "hstack", props: { spacing: 12, alignment: "center", max_width: "infinity" }, children: [
        { type: "hstack", props: { spacing: 14, alignment: "center", axis: "wrap", max_width: "infinity" }, children: [
          { type: "vstack", props: { spacing: 0, alignment: "leading" }, children: [
            eb("SELECTED"),
            text("{data.lineCountLabel}", "montserrat.bold:14", "offwhite", { fixed_size_h: true }),
          ] },
          divCol("TOTAL GOLD", { type: "grams_text", props: { value: "{data.totalGrams}", num_font: "montserrat.bold:14", color: "offwhite" } }),
          divCol("EST. PAYOUT", money("{data.totalPayout}", "montserrat.bold:14", "gold")),
        ] },
        { type: "button", props: {
          label_text: "Continue", label_font: "montserrat.bold:14", label_color: "bg", bg: "gold",
          corner_radius: 12, padding_h: 20, padding_v: 10, full_width: false, label_nowrap: true, trailing_icon: "chevron.right",
          action: { kind: "state_set", key: "step", value: "shipping" },
        } },
      ] },
    ],
  };
}
