/**
 * Product updates log — one customer-facing line per milestone, dated:
 *   product  — a customer-visible feature that shipped;
 *   business — a company milestone (registration, partnership, licence, press,
 *              award, payment partner).
 * Table: product_updates (migration 20261103_product_updates.sql, admin-only).
 */
import type { SupabaseClient } from "@supabase/supabase-js";
 
export const PRODUCT_UPDATE_MAX_CHARS = 200;
 
export const PRODUCT_UPDATE_KINDS = ["product", "business"] as const;
export type ProductUpdateKind = (typeof PRODUCT_UPDATE_KINDS)[number];
 
export interface ProductUpdate {
  id:         string;
  shipped_on: string; // YYYY-MM-DD
  summary:    string;
  kind:       ProductUpdateKind;
}
 
export type ProductUpdateInput = { shipped_on: string; summary: string; kind: ProductUpdateKind };
 
/** Validate an admin create/edit. Returns the clean row or an error message. */
export function validateProductUpdate(body: unknown): ProductUpdateInput | string {
  const b = (body ?? {}) as Record<string, unknown>;
  const shipped = typeof b.shipped_on === "string" ? b.shipped_on.trim() : "";
  if (!/^\d{4}-\d{2}-\d{2}$/.test(shipped) || Number.isNaN(Date.parse(`${shipped}T00:00:00Z`))) {
    return "shipped_on must be a date (YYYY-MM-DD)";
  }
  const summary = typeof b.summary === "string" ? b.summary.replace(/\s+/g, " ").trim() : "";
  if (!summary) return "summary is required";
  if (summary.length > PRODUCT_UPDATE_MAX_CHARS) return `summary is too long (max ${PRODUCT_UPDATE_MAX_CHARS} characters)`;
  const kind = b.kind === undefined ? "product" : b.kind;
  if (!PRODUCT_UPDATE_KINDS.includes(kind as ProductUpdateKind)) return "kind must be product or business";
  return { shipped_on: shipped, summary, kind: kind as ProductUpdateKind };
}
 
/** Updates shipped in a calendar month (dates are SGT calendar dates), oldest first. */
export async function fetchProductUpdatesForMonth(
  db: SupabaseClient, year: number, month: number,
): Promise<ProductUpdate[]> {
  const from = `${year}-${String(month).padStart(2, "0")}-01`;
  const next = month === 12 ? `${year + 1}-01-01` : `${year}-${String(month + 1).padStart(2, "0")}-01`;
  const { data, error } = await db
    .from("product_updates")
    .select("id, shipped_on, summary, kind")
    .gte("shipped_on", from)
    .lt("shipped_on", next)
    .order("shipped_on", { ascending: true });
  if (error) throw new Error("product_updates_fetch_failed");
  return (data ?? []) as ProductUpdate[];
}
