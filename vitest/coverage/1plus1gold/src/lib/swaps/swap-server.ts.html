 * Server-side swap helpers that touch other systems: sending the return label to
 * the customer, and reading the bar costs for the admin's cost line.
 */
 
import { fetchSwapPackedAt } from "./swap-packing";

import type { SupabaseClient } from "@supabase/supabase-js";
import { notifySwapReturnLabel } from "@/lib/notifications/notify-customer";
import { swapCost, type OrderSwap, type SwapCost } from "./order-swap";

/** Short customer-facing order ref, as on the receipt. */
export function orderRef(orderId: string): string {
  return orderId.slice(0, 8).toUpperCase();
}
 
/**
 * Send the return label to the customer (WhatsApp media template once approved;
 * email meanwhile) and stamp what was sent. Idempotent: a swap whose label was
 * already sent is skipped. Best-effort — a failed send leaves the stamp NULL and
 * the admin's WhatsApp button covers it.
 */
export async function sendReturnLabel(supabase: SupabaseClient, swap: OrderSwap): Promise<string | null> {
  if (swap.return_label_sent_at || swap.status !== "return_booked" || !swap.return_tracking_no || !swap.return_collect_type) return null;
  const token = returnLabelToken(swap);
  const labelUrl = returnLabelUrl(swap, token);

    const r = await notifySwapReturnLabel(swap.customer_id, {
      orderRef: orderRef(swap.order_id), labelUrl,
      collectType: swap.return_collect_type, trackingNo: swap.return_tracking_no,
    });
    const channels = [r.whatsapp?.ok ? "whatsapp" : null, r.email?.ok ? "email" : null].filter(Boolean).join(",");
    if (!channels) return null;
    await supabase.from("order_swaps")
      .update({ return_label_sent_at: new Date().toISOString(), return_label_sent_channels: channels, updated_at: new Date().toISOString() })
      .eq("id", swap.id).is("return_label_sent_at", null);
    return channels;
  } catch (err) {
    console.error(`[swaps] return label send failed for ${swap.id}:`, err instanceof Error ? err.message : err);
    return null;
  }
}
 
export interface SwapViewWithCost extends SwapView {
  cost: SwapCost;
  /** Dispatch (2026-11-11): when the replacement parcel was packed (swap_packing_events), else null. */
  packed_at: string | null;
  /** The replacement bar's refinery number (nfc_serials.manufacturer_serial), for the slip + tile. */
  replacement_bar_no: string | null;
  /** Grams of the swapped line (parcel weight, slip). */
  grams: number;
}
 
/** swapView() plus the cost line, reading the two bars' landed costs in one query. */
export async function swapViewsWithCost(supabase: SupabaseClient, swaps: OrderSwap[]): Promise<SwapViewWithCost[]> {
  const invIds = [...new Set(swaps.flatMap((s) => [s.returned_inventory_id, s.replacement_inventory_id]).filter((x): x is string => !!x))];
  const itemIds = [...new Set(swaps.map((s) => s.order_item_id))];
  const nfcs = [...new Set(swaps.map((s) => s.replacement_nfc_serial).filter((x): x is string => !!x))];
  const [{ data: inv }, { data: items }, packedAt, mfr] = await Promise.all([
    invIds.length ? supabase.from("inventory").select("id, cost_per_g_sgd").in("id", invIds) : Promise.resolve({ data: [] }),
    itemIds.length ? supabase.from("order_items").select("id, grams_each").in("id", itemIds) : Promise.resolve({ data: [] }),
    fetchSwapPackedAt(supabase, swaps.map((s) => s.id)),
    fetchManufacturerSerials(supabase, nfcs),
  ]);
  const cost = new Map(((inv ?? []) as { id: string; cost_per_g_sgd: number | null }[]).map((r) => [r.id, r.cost_per_g_sgd == null ? null : Number(r.cost_per_g_sgd)]));
  const grams = new Map(((items ?? []) as { id: string; grams_each: number }[]).map((r) => [r.id, Number(r.grams_each)]));
  return swaps.map((s) => ({
    ...swapView(s),
    packed_at: packedAt.get(s.id) ?? null,
    replacement_bar_no: s.replacement_nfc_serial ? realManufacturer(s.replacement_nfc_serial, mfr.get(s.replacement_nfc_serial)) : null,
    grams: grams.get(s.order_item_id) ?? 0,
    cost: swapCost({
      returnedCostPerG:    s.returned_inventory_id ? cost.get(s.returned_inventory_id) ?? null : null,
      // Before Received the replacement bar is not chosen yet — the cost line waits for it.
      replacementCostPerG: s.replacement_inventory_id ? cost.get(s.replacement_inventory_id) ?? null : null,
      grams:               grams.get(s.order_item_id) ?? 0,
      returnFeeSgd:        s.return_actual_fee_sgd == null ? null : Number(s.return_actual_fee_sgd),
      outboundFeeSgd:      s.outbound_actual_fee_sgd == null ? null : Number(s.outbound_actual_fee_sgd),
    }),
  }));
}
