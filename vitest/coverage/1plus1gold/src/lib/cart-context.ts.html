/**
 * buildCartContext — the per-user inputs the on-device cart kernel (`deriveCart`
 * / `engine.cartView`) can't derive locally: accumulated curve position, owned
 * grams, KYC status + self-limit, the accessory/product catalog, the gift-box
 * product id, and live price-staleness flags.
 *
 * Extracted from `/api/cart/context` so BOTH the cart context route AND the
 * checkout context route (`/api/checkout/context`, which returns a superset)
 * consume ONE implementation — no parallel builder, no drift (the cart and
 * checkout surfaces price off byte-identical inputs).
 *
 * Auth: optional. Anonymous → curve/owned 0, kyc not_required (guest carts are
 * allowed; the charge stays server-authoritative).
 */
 
import { createClient } from "@supabase/supabase-js";
import { getSpotPayload } from "@/lib/spot/fetch-spot";
import { getAccumulatedCurveGrams } from "@/lib/curve-accumulation";
import { GIFT_BOX_PRODUCT_ID, type CartCatalogProduct } from "@/lib/cart-core";
import { sanitizeTiers } from "@/lib/wholesale-tiers";
import { getRedeemableRewards, type RedeemableReward } from "@/lib/promos-mine";
 
const supabaseAdmin = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_KEY!,
);
 
export interface CartContextPayload {
  accumulatedCurveGrams: number;
  accumulatedCurveGramsLoaded: boolean;
  ownedGrams: number;
  catalog: CartCatalogProduct[];
  /** Mirrors the server priceOrder gate: when true the cart applies the
   *  better-of(standard, wholesale) for standard lines so the locked price
   *  equals what the server will charge (no "cart total mismatch" 422). */
  wholesaleEnabled: boolean;
  giftBoxProductId: string;
  kycStatus: string;
  selfLimit: boolean;
  xauStale: boolean;
  fxStale: boolean;
  /** The signed-in customer's redeemable codes — feeds the in-cart "Your codes"
   *  picker (§45 PROMO-SURF-2). Empty for guests. Display only. */
  myRewards: RedeemableReward[];
}
 
/** Wholesale premium tiers are commercially-sensitive pricing structure and must
 *  never reach retail clients. The better-of(standard, wholesale) gate only fires
 *  when `wholesaleEnabled`, and `hydrateStandardTiers`/`deriveCart` degrade
 *  gracefully on a null wholesale tier (cart-core.ts:565 — only used when length
 *  > 0) — so for retail users we strip the field from every product in the
 *  payload entirely. (security_agent_pr wholesale-tiers-leak-cart-context-2026-06-07.) */
export function stripWholesaleForRetail(
  catalog: CartCatalogProduct[],
  wholesaleEnabled: boolean,
): CartCatalogProduct[] {
  if (wholesaleEnabled) return catalog;
  return catalog.map((p) => ({ ...p, wholesale_premium_tiers: null }));
}
 
/** Build the shared cart-context payload. `userId` is the already-resolved
 *  caller (the route auths once and passes it in) — so the builder never
 *  re-authenticates. */
export async function buildCartContext(userId: string | null): Promise<CartContextPayload> {
  const [spot, accumulatedCurveGrams, catalog, kyc, wholesaleEnabled, myRewards] = await Promise.all([
    getSpotPayload().catch(() => null),
    userId ? getAccumulatedCurveGrams(userId, supabaseAdmin).catch(() => 0) : Promise.resolve(0),
    fetchCartCatalog(),
    userId ? fetchKyc(userId) : Promise.resolve({ kycStatus: "not_required", selfLimit: false }),
    userId ? fetchWholesaleEnabled(userId) : Promise.resolve(false),
    userId ? getRedeemableRewards(userId, supabaseAdmin).catch(() => []) : Promise.resolve([] as RedeemableReward[]),
  ]);
 
  // `ownedGrams` and `accumulatedCurveGrams` are the SAME value — both are
  // getAccumulatedCurveGrams(userId). Previously `ownedGrams` came via
  // resolveOwnedGrams(req), which RE-AUTHENTICATED the Bearer token (a 2nd
  // GoTrue round-trip on top of the route's) AND re-ran the identical RPC.
  // Compute once: no re-auth, no duplicate query. (PERF audit 2026-06-11.)
  const ownedGrams = accumulatedCurveGrams;
 
  return {
    accumulatedCurveGrams,
    accumulatedCurveGramsLoaded: true,
    ownedGrams,
    catalog: stripWholesaleForRetail(catalog, wholesaleEnabled),
    wholesaleEnabled,
    giftBoxProductId: GIFT_BOX_PRODUCT_ID,
    kycStatus: kyc.kycStatus,
    selfLimit: kyc.selfLimit,
    xauStale: spot && !("error" in spot) ? !!spot.xauStale : false,
    fxStale: spot && !("error" in spot) ? !!spot.fxStale : false,
    myRewards,
  };
}
 
// ── Shared cart-catalog cache (user-independent; per-user wholesale strip runs
//    AFTER, in buildCartContext) ────────────────────────────────────────────
// This shop-visible catalog with nested tiers + media was the heaviest query in
// the cart/checkout context and ran on EVERY open. It's identical across users
// (the wholesale tiers are sanitized here, then stripped per-user downstream by
// stripWholesaleForRetail), so cache the mapped array in-process with a short
// TTL. (PERF audit 2026-06-11.)
const CART_CATALOG_CACHE_TTL_MS = 60_000;
let _cartCatalogCache: { value: CartCatalogProduct[]; at: number } | null = null;
 
async function fetchCartCatalog(): Promise<CartCatalogProduct[]> {
  const now = Date.now();
  if (_cartCatalogCache && now - _cartCatalogCache.at < CART_CATALOG_CACHE_TTL_MS) {
    return _cartCatalogCache.value;
  }
 
  const { data, error } = await supabaseAdmin
    .from("products")
    .select("id, grams, label, product_type, price_sgd, discount_rules, status, stock_qty, gy_eligible, vault_eligible, standard_premium_tiers(min_qty, max_qty, premium_pct), wholesale_premium_tiers, product_media(url, sort_order)")
    .eq("shop_visible", true);
  if (error || !data) return [];   // don't cache a transient failure
  const mapped = data.map((p): CartCatalogProduct => {
    const media = [...(p.product_media ?? [])].sort((a, b) => (a.sort_order ?? 0) - (b.sort_order ?? 0));
    return {
      id: p.id,
      label: p.label,
      grams: Number(p.grams),   // numeric(10,4) → string from PostgREST; coerce
      product_type: p.product_type,
      status: p.status,
      stock_qty: p.stock_qty ?? null,
      price_sgd: p.price_sgd,
      discount_rules: p.discount_rules ?? null,
      image_url: media[0]?.url ?? null,
      product_media: media.map((m) => ({ url: m.url })),
      standard_premium_tiers: p.standard_premium_tiers ?? null,
      wholesale_premium_tiers: sanitizeTiers(p.wholesale_premium_tiers),
      premium_pct: null, // curve premiums come from the live bootstrap; tier fallback only
      gy_eligible: p.gy_eligible === true,
      vault_eligible: p.vault_eligible === true,
    };
  });
  if (mapped.length > 0) _cartCatalogCache = { value: mapped, at: now };
  return mapped;
}
 
/** Test-only: clear the in-process cart-catalog cache between scenarios. */
export const __cartContextTest = {
  resetCatalogCache(): void { _cartCatalogCache = null; },
};
 
/** The customer's wholesale eligibility. `wholesale_enabled` lives on the
 *  `customers` table (NOT `profiles` — see admin/users + wholesale/products),
 *  and the same flag must gate the cart's better-of so the locked price equals
 *  the server charge. (2026-06-07: reading it from `profiles` errored — column
 *  doesn't exist there — silently returning false, so wholesale never applied.) */
async function fetchWholesaleEnabled(userId: string): Promise<boolean> {
  const { data } = await supabaseAdmin
    .from("customers")
    .select("wholesale_enabled")
    .eq("id", userId)
    .single();
  return data?.wholesale_enabled === true;
}
 
async function fetchKyc(userId: string): Promise<{ kycStatus: string; selfLimit: boolean }> {
  const { data } = await supabaseAdmin
    .from("profiles")
    .select("kyc_status, kyc_expires_at, self_limit_below_5k")
    .eq("id", userId)
    .single();
  if (!data) return { kycStatus: "not_required", selfLimit: false };
  const kycStatus =
    data.kyc_status === "approved" && data.kyc_expires_at && new Date(data.kyc_expires_at) < new Date()
      ? "expired"
      : (data.kyc_status as string);
  return { kycStatus, selfLimit: data.self_limit_below_5k ?? false };
}
