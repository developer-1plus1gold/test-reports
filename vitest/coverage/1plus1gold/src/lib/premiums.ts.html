export interface PremiumRow {
  g: number;
  label: string;
  is_assay: boolean;
  premium_pct: number; // stored as percentage: 20.0 = 20%
  updated_at: string;
}
 
// Server-side in-memory cache (not shared with client bundle)
let cache: { data: PremiumRow[]; ts: number } | null = null;
const CACHE_TTL = 5 * 60_000; // 5 minutes — premiums change rarely
 
/**
 * Fetch the premium-curve anchors.
 *
 * Source: the `premiums` table (since migration 20260519d). This is the
 * canonical curve, decoupled from products — anchors persist even when
 * the matching curve product is removed or moved to standard. Previous
 * implementation read from products WHERE product_type='curve', which
 * silently broke the curve whenever curve products changed.
 *
 * Uses anon key — table has public read RLS. Falls back to cache on
 * network error.
 */
export async function fetchPremiums(): Promise<PremiumRow[]> {
  const now = Date.now();
  if (cache && now - cache.ts < CACHE_TTL) return cache.data;
 
  try {
    const url = `${process.env.NEXT_PUBLIC_SUPABASE_URL}/rest/v1/premiums?select=grams,label,is_assay,premium_pct,updated_at&order=grams.asc`;
    const res = await fetch(url, {
      headers: {
        apikey: process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
        Authorization: `Bearer ${process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!}`,
      },
      cache: "no-store",
    });
    if (!res.ok) throw new Error(`Supabase ${res.status}`);
    const rows: { grams: number; label: string; is_assay: boolean; premium_pct: number; updated_at: string }[] = await res.json();
    const data: PremiumRow[] = rows.map((r) => ({
      g:           Number(r.grams),
      label:       r.label,
      is_assay:    r.is_assay,
      premium_pct: Number(r.premium_pct),
      updated_at:  r.updated_at,
    }));
    cache = { data, ts: now };
    return data;
  } catch {
    return cache?.data ?? [];
  }
}

interface StandardBaselineRow { refinery: string | null; standard_premium_tiers: StandardTierRow[] | null }
 
/** The flagship standard-bar refinery the Savings Simulator benchmarks the
 *  premium curve against. The comparison should be premium-for-premium — a like
 *  PAMP 1g standard bar vs our premium 1g splits on the curve — NOT against a
 *  cheaper budget bar, so we anchor on this refinery's qty-1 premium rather than
 *  the global cheapest. (Operator decision 2026-08-12: a budget flat-22% bar was
 *  understating the standard cost and shrinking the visible curve advantage.) */
export const STANDARD_BASELINE_REFINERY = "PAMP";
 
/**
 * Pick the standard-baseline premium (percent) from the candidate 1g standard
 * products: the qty-1 premium of the flagship-refinery bar when present, else
 * the cheapest qty-1 across all candidates — a graceful fallback so the number
 * never vanishes just because the flagship is momentarily hidden / out of stock.
 * Returns null when no candidate has a usable tier.
 */
export function selectStandardBaselinePct(
  rows: StandardBaselineRow[],
  flagshipRefinery: string = STANDARD_BASELINE_REFINERY,
): number | null {
  const withPct = rows
    .map((r) => ({ refinery: (r.refinery ?? "").trim(), pct: qty1StandardPremiumPct(r.standard_premium_tiers ?? []) }))
    .filter((x): x is { refinery: string; pct: number } => x.pct != null);
  if (withPct.length === 0) return null;
  const flag = flagshipRefinery.trim().toLowerCase();
  const flagship = withPct.filter((x) => x.refinery.toLowerCase() === flag);
  const pool = flagship.length > 0 ? flagship : withPct;
  return Math.min(...pool.map((x) => x.pct));
}
 
/**
 * The qty-1 premium (percent) a single standard product charges — routed
 * through the SAME `getTierPremium` the shopping cart prices standard lines
 * with, so the baseline equals what a real 1-unit standard checkout pays
 * (including its "lowest tier also applies below its min_qty" rule). Returns
 * null when the product has no usable tier.
 */
export function qty1StandardPremiumPct(tiers: StandardTierRow[]): number | null {
  const clean = tiers
    .filter((t) => Number.isFinite(Number(t.premium_pct)) && Number(t.premium_pct) > 0)
    .map((t) => ({
      min_qty: Number(t.min_qty ?? 1),
      max_qty: t.max_qty == null ? null : Number(t.max_qty),
      premium_pct: Number(t.premium_pct),
    }));
  if (clean.length === 0) return null;
  const dec = getTierPremium(1, clean);   // cart's canonical tier lookup
  return dec > 0 ? dec * 100 : null;
}

/**

 * against: the qty-1 (single-unit) standard-tier premium for a `grams`-weight
 * product at quantity 1 (default 1g), anchored on the flagship refinery
 * (STANDARD_BASELINE_REFINERY) so the comparison is premium-for-premium — see
 * `selectStandardBaselinePct`. Falls back to the cheapest qty-1 across all

// The expense-to-gold simulator needs the full STANDARD ladder (one bar per
// size, cheapest shop-visible) and which 1g split refineries are shop-enabled
// (drives the Premium-Curve card set). Curve anchors come from fetchPremiums().
 
export interface StandardLadderTier { grams: number; premiumPct: number; label: string }
export interface GoldLifePricing {
  /** Cheapest shop-visible standard bar per size, ascending by grams. */
  standardLadder: StandardLadderTier[];
  /** Curve anchors (grams + premium percent), ascending by grams. */
  curveAnchors: { grams: number; premiumPct: number }[];
  /** Lowercased refineries of the shop-enabled 1g splits (e.g. ["valcambi","pamp"]). */
  splitRefineries: string[];
}
 
interface LadderProductRow {
  grams: number | null;
  label: string | null;
  refinery: string | null;
  standard_premium_tiers: StandardTierRow[] | null;
}
 
let goldLifeCache: { data: GoldLifePricing; ts: number } | null = null;
 
async function anonGet<T>(path: string): Promise<T> {
  const res = await fetch(`${process.env.NEXT_PUBLIC_SUPABASE_URL}/rest/v1/${path}`, {
    headers: {
      apikey: process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
      Authorization: `Bearer ${process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!}`,
    },
    cache: "no-store",
  });
  if (!res.ok) throw new Error(`Supabase ${res.status}`);
  return res.json() as Promise<T>;
}
 
/**
 * Assemble the pricing the gold-life simulator needs, cached like the curve.
 * - standardLadder: every shop-visible standard bar, collapsed to the CHEAPEST
 *   qty-1 premium per distinct size (so a size never appears twice; the buyer's
 *   best price wins). Priced through the cart's `getTierPremium(1,…)` so it
 *   equals a real 1-unit standard checkout.
 * - splitRefineries: the refineries of the shop-visible 1g split products, so
 *   the card set self-adjusts if one is disabled in the shop.
 * Anon key (public-read). Falls back to last-known cache on error; empty ladder
 * makes the caller (simulator) hide rather than invent numbers.
 */
export async function fetchGoldLifePricing(): Promise<GoldLifePricing> {
  const now = Date.now();
  if (goldLifeCache && now - goldLifeCache.ts < CACHE_TTL) return goldLifeCache.data;
 
  try {
    const [curveRows, ladderRows, splitRows] = await Promise.all([
      fetchPremiums(),
      anonGet<LadderProductRow[]>(
        "products?select=grams,label,refinery,standard_premium_tiers(min_qty,max_qty,premium_pct)" +
        "&product_type=eq.standard&shop_visible=eq.true",
      ),
      anonGet<{ refinery: string | null }[]>(
        "products?select=refinery&product_type=eq.curve&shop_visible=eq.true&grams=lte.1",
      ),
    ]);
 
    // Collapse standard products to the cheapest qty-1 premium per size.
    const bySize = new Map<number, StandardLadderTier>();
    for (const r of ladderRows) {
      const g = Number(r.grams);
      if (!Number.isFinite(g) || g <= 0) continue;
      const pct = qty1StandardPremiumPct(r.standard_premium_tiers ?? []);
      if (pct == null) continue;
      const existing = bySize.get(g);
      if (!existing || pct < existing.premiumPct) {
        bySize.set(g, { grams: g, premiumPct: pct, label: (r.label ?? `${g} g`).trim() });
      }
    }
    const standardLadder = [...bySize.values()].sort((a, b) => a.grams - b.grams);
 
    const splitRefineries = [
      ...new Set(splitRows.map(r => (r.refinery ?? "").trim().toLowerCase()).filter(Boolean)),
    ];
 
    const data: GoldLifePricing = {
      standardLadder,
      curveAnchors: curveRows.map(r => ({ grams: r.g, premiumPct: r.premium_pct })),
      splitRefineries,
    };
    goldLifeCache = { data, ts: now };
    return data;
  } catch {
    return goldLifeCache?.data ?? { standardLadder: [], curveAnchors: [], splitRefineries: [] };
  }
}
