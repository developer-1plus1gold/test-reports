/**
 * Reconcile a pending Airwallex order against Airwallex's API (2026-11-07).
 *
 * The single "ask Airwallex, complete if paid" path, used by the webhook (as a
 * nudge — the webhook is verified, but completion reads the API so every path
 * completes from the same fact, and Airwallex does not guarantee event order),
 * the client status poll, the reconcile cron and reconcile-before-expire
 * cleanup. Completion holds the shared atomic claim, so concurrent callers are
 * safe. Twin of razorpay-reconcile.ts.
 *
 *   session ACTIVE                 → still pending (no payment yet). Reported as
 *                                    `unattempted`: the status poll only asks
 *                                    once the customer has RETURNED, so then it
 *                                    means they left without paying.
 *   session EXPIRED / CANCELLED    → close the row (expired / cancelled).
 *   session COMPLETED → PaymentIntent:
 *     SUCCEEDED                    → complete the order
 *     CANCELLED                    → failed (declined)
 *     anything else                → still pending (3DS, async, risk review)
 */

/** Move a still-pending row to a terminal unpaid state and give MAG grams back. */
export async function closeAirwallexPending(
  supabaseAdmin: SupabaseClient,
  pending: { id: string; reference_number: string },
  status: "expired" | "cancelled" | "failed",
): Promise<boolean> {
  const { data: closed } = await supabaseAdmin
    .from("airwallex_pending_payments")
    .update({ status })
    .eq("id", pending.id)
    .eq("status", "pending")
    .select("id");
  const moved = (closed?.length ?? 0) > 0;
  if (moved) {
    await releaseMagHold(supabaseAdmin, pending.reference_number);
    await releaseMagRedeem(supabaseAdmin, pending.reference_number);
  }
  return moved;
}
