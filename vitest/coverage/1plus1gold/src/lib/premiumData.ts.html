/**
 * Market premium survey — the dated benchmark (March 2026, Singapore retail
 * dealers). ONE source for every chart and card on Our Pricing; nothing else
 * should hard-code these numbers.
 *
 * `ours` is refreshed from live tables by the caller (derivePricingDisplay /
 * the page summary); the values here are the March 2026 snapshot fallback.
 * `competitor` is the optional live "competitors this week" marker, filled by
 * the caller from competitor_prices — never stored here.
 */
export interface PremiumDatum {
  label: string;
  /** Market minimum — null for a row surveyed at the median only. */
  min: number | null;
  median: number;
  /** Market maximum — null for a row surveyed at the median only. */
  max: number | null;
  /** Our premium; null when we do not currently offer this row (e.g. MAG with no bar listed). */
  ours: number | null;
  /** Live median of competitor premiums this week (percent), when known. */
  competitor?: number | null;
  /** Non-physical delivery row — compared against MAG, not a physical bar. */
  npd?: boolean;
}
 
/** 1g non-physical-delivery market median (percent) — MAG's market comparator. */
export const NPD_1G_MEDIAN_PCT = 2.2;
 
export const NPD_ROW_LABEL = "1g NPD";
 
/** Plain-words definition shown wherever an NPD row is charted (Mukesh
 *  2026-10-07: "in charts we have to say what NPD means"). */
export const NPD_DEFINITION =
  "NPD = Non-Physical Delivery — also called tokenised or digital gold: gold you own but do not take home. Compared with 1+1Gold MAG.";
 
/** Physical-delivery rows, smallest first. */
export const PREMIUM_DATA: PremiumDatum[] = [
  { label: "1g",   min: 10.12, median: 27.62, max: 73.65, ours: 20.0 },
  { label: "5g",   min: 7.16,  median: 16.60, max: 26.48, ours: 12.0 },
  { label: "10g",  min: 6.32,  median: 12.25, max: 20.04, ours:  9.0 },
  { label: "50g",  min: 2.84,  median: 3.15,  max: 7.29,  ours:  2.5 },
  { label: "100g", min: 2.11,  median: 2.50,  max: 5.26,  ours:  2.0 },
];
 
/** The NPD row — median only; `ours` is the MAG 1g premium when a bar is listed. */
export function npdRow(magPct: number | null): PremiumDatum {
  return { label: NPD_ROW_LABEL, min: null, median: NPD_1G_MEDIAN_PCT, max: null, ours: magPct, npd: true };
}
