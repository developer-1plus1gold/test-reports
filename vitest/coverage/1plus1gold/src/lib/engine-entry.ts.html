/**
 * engine-entry.ts — the JavaScriptCore entry for the iOS on-device pricing
 * kernel (Phase 6.5, contract §22). esbuild bundles this (+ the pure
 * `calculator-core` and its deps) into `apps/ios-customer/Resources/CalcEngine/engine.js`
 * as an IIFE exposing a `CalcEngine` global. Swift loads the script once into a
 * `JSContext`, calls `setBootstrap` with the `/api/calculator/bootstrap` payload,
 * then calls `simulate` on every keystroke/slide — INSTANT, local, no network.
 *
 * Strings cross the bridge (JSON in / JSON out) — the simplest, most robust
 * marshaling across JSC (no JSValue object graphs). Build: `npm run build:calc-engine`.
 * The artifact is committed + CI-freshness-checked so it can never drift from
 * the source (same single-source guarantee as chart-web-v2).
 */
 
import { simulateCalculator, type CalcInput, type CalcBootstrap } from "./calculator-core";
import { roundToNice } from "./format-pure";
import { feeAmountSgd, feeRateLabel, type PaymentFeeMethod } from "./payment-fees";
import { priceShopCatalog, partitionShop, shopStandardView, shopFilterOptions, shopLineForProduct, type ShopPriceable, type ShopViewState } from "./shop-pricing";
import {
  deriveCart, applyMilestone, reconcileGiftBox, refreshCartPrices, mergeAdd,
  type CartLine, type CartContext, type MilestoneVariant,
} from "./cart-core";
import { buildBuybackView, type BuybackContext, type BuybackViewState } from "./buyback-core";
import { computeRefereeDiscountSgd } from "./referral-discount";
import { computeCheckoutDiscounts, type CheckoutDiscountArgs } from "./checkout-discount";
 
let BOOT: CalcBootstrap | null = null;
 
/** Called once after `/api/calculator/bootstrap` returns. `json` = the payload. */
export function setBootstrap(json: string): void {
  BOOT = JSON.parse(json) as CalcBootstrap;
}
 
/** True once `setBootstrap` has run — Swift gates `simulate` on this. */
export function isReady(): boolean {
  return BOOT !== null;
}
 
/** Run the simulation for the current live inputs. `inputJson` = CalcInput.
 *  Returns the SimulateResponse as a JSON string. */
export function simulate(inputJson: string): string {
  if (!BOOT) return JSON.stringify({ error: "not_bootstrapped" });
  const input = JSON.parse(inputJson) as CalcInput;
  return JSON.stringify(simulateCalculator(input, BOOT));
}
 
/**
 * Re-express a money `amount` from one currency to another, preserving the SGD
 * value (the Calculator budget, on a currency switch — mirrors the PWA's
 * convert-on-rate-change). Uses the bootstrapped per-currency gram price
 * (`currencyGram`) as the rate. Rounds to a clean number (roundToNice), like the
 * web. Returns `{ value }` (null if not convertible). Contract §22.
 */
export function convertAmount(amount: number | string, from: string, to: string): string {
  if (!BOOT) return JSON.stringify({ error: "not_bootstrapped" });
  const a = Number(amount);
  const f = BOOT.currencyGram[(from || "SGD").toUpperCase()];
  const t = BOOT.currencyGram[(to || "SGD").toUpperCase()];
  if (!a || !isFinite(a) || a <= 0 || !f || !t) return JSON.stringify({ value: null });
  return JSON.stringify({ value: roundToNice(a * (t / f)) });
}
 
/**
 * Shop (+ Wholesale) on-device pricing + view. `catalogJson` = the rich product
 * catalog the iOS app fetched from /api/products (card fields); `viewStateJson`
 * = the Standard-section controls (filters/sort/group). Prices every product
 * from the bootstrapped spot + premiums, partitions into sections, and applies
 * the standard-section view — INSTANT, local, no network. Returns the
 * render-ready sections as JSON. Contract §23. Spot/premium currency conversion
 * is done client-side by `currency_amount`; this stays SGD.
 */
export function catalogView(catalogJson: string, viewStateJson: string): string {
  if (!BOOT) return JSON.stringify({ error: "not_bootstrapped" });
  // Accept either the bare product array or the /api/products `{products:[…]}`
  // wrapper, so iOS can pass the response body verbatim.
  const parsed = JSON.parse(catalogJson) as ShopPriceable[] | { products?: ShopPriceable[] };
  const catalog = (Array.isArray(parsed) ? parsed : parsed.products ?? []) as ShopPriceable[];
  const viewState = JSON.parse(viewStateJson || "{}") as ShopViewState;
  const priced = priceShopCatalog(catalog, BOOT.spotSgd, BOOT.premiums);
  const parts = partitionShop(priced);
  // The filter/sort/group controls now govern BOTH the Premium Curve and Standard
  // sections (Mukesh 2026-06-14, web parity): curve runs through the SAME
  // shopStandardView pipeline → curveGroups, and the filter options span ALL gold
  // (curve + standard) so the curve's refineries + new split sizes appear as chips.
  const curve = [...parts.curveSplits, ...parts.curveCombibars];
  const { gramsOptions, refineryOptions } = shopFilterOptions([...curve, ...parts.standard]);
  return JSON.stringify({
    curveGroups:    shopStandardView(curve, viewState),
    standardGroups: shopStandardView(parts.standard, viewState),
    accessories:    parts.accessories,
    gramsOptions,
    refineryOptions,
  });
}
 
/**
 * Qty-aware line price for ONE product (the Shop qty editor). `productJson` =
 * a single catalog product (the repeater item the native node holds); `qty` =
 * the chosen quantity. Returns the `ShopLine` envelope as JSON — INSTANT, local,
 * from the bootstrapped spot + premiums, byte-identical to the web qty editor.
 * Contract §23.
 */
export function shopLine(productJson: string, qty: number): string {
  if (!BOOT) return JSON.stringify({ error: "not_bootstrapped" });
  const p = JSON.parse(productJson) as ShopPriceable;
  // BOOT.ownedGrams = the signed-in customer's accumulated curve position (0 anon)
  // → the curve line follows the curve AT the customer's position, matching the
  // cart + the web shop. Mukesh 2026-06-14 (#1 + pricing-based-on-logged-in-user).
  return JSON.stringify(shopLineForProduct(p, qty, BOOT.spotSgd, BOOT.premiums, BOOT.ownedGrams ?? 0));
}
 
// ── Cart (+ Checkout) on-device derivation — contract §25 ────────────────────
// Spot + premiums come from the live calc bootstrap (BOOT); the per-user inputs
// (curve position, owned grams, accessory catalog, KYC, staleness) ride in
// `ctxJson` (the iOS app's /api/cart/context fetch). deriveCart is the SAME impl
// the web cart runs — instant, local, byte-identical, no drift. Charge stays
// server-authoritative (orders/create recomputes + rejects mismatch > 1¢).
 
function buildCartCtx(ctxJson: string): CartContext {
  const c = JSON.parse(ctxJson || "{}") as Partial<CartContext>;
  return {
    spotSgd: BOOT!.spotSgd,
    premiums: BOOT!.premiums,
    accumulatedCurveGrams: c.accumulatedCurveGrams ?? 0,
    accumulatedCurveGramsLoaded: c.accumulatedCurveGramsLoaded ?? true,
    ownedGrams: c.ownedGrams ?? 0,
    catalog: c.catalog ?? [],
    wholesaleEnabled: c.wholesaleEnabled ?? false,
    kycStatus: c.kycStatus,
    selfLimit: c.selfLimit,
    xauStale: c.xauStale,
    fxStale: c.fxStale,
    giftBoxProductId: c.giftBoxProductId,
  };
}
 
/** Full cart read model (per-line price/line/premium/accessory, summary,
 *  milestone gate, fulfilment, unlocks, progress, KYC). Returns CartView JSON. */
export function cartView(linesJson: string, ctxJson: string): string {
  if (!BOOT) return JSON.stringify({ error: "not_bootstrapped" });
  const lines = JSON.parse(linesJson) as CartLine[];
  return JSON.stringify(deriveCart(lines, buildCartCtx(ctxJson)));
}
 
/** Milestone conversion (1g splits → combibar). Returns the new lines JSON. */
export function cartApplyMilestone(linesJson: string, variant: string, ctxJson: string): string {
  if (!BOOT) return JSON.stringify({ error: "not_bootstrapped" });
  const lines = JSON.parse(linesJson) as CartLine[];
  return JSON.stringify(applyMilestone(lines, variant as MilestoneVariant, buildCartCtx(ctxJson)));
}
 
/** Reconcile the auto gift-box line after a gift toggle. Returns `{ lines, changed }` JSON. */
export function cartReconcileGiftBox(linesJson: string, ctxJson: string): string {
  if (!BOOT) return JSON.stringify({ error: "not_bootstrapped" });
  const lines = JSON.parse(linesJson) as CartLine[];
  return JSON.stringify(reconcileGiftBox(lines, buildCartCtx(ctxJson)));
}
 
/** Add an item to the cart, merging by itemKey. `fromWholesale` → "better
 *  premium wins" on a same-item merge. Returns the new lines JSON. Pure (no
 *  bootstrap needed — tier lookup only). */
export function cartMergeAdd(linesJson: string, incomingJson: string, fromWholesale: boolean): string {
  const lines = JSON.parse(linesJson) as CartLine[];
  const incoming = JSON.parse(incomingJson) as CartLine;
  return JSON.stringify(mergeAdd(lines, incoming, fromWholesale));
}
 
/** Re-stamp priceEach across all lines from the live spot. Returns the new lines JSON. */
export function cartRefreshPrices(linesJson: string, ctxJson: string): string {
  if (!BOOT) return JSON.stringify({ error: "not_bootstrapped" });
  const lines = JSON.parse(linesJson) as CartLine[];
  return JSON.stringify(refreshCartPrices(lines, buildCartCtx(ctxJson)));
}
 
// ── Checkout payment processing fee (4b-iii, contract §25) ───────────────────
// SAME formula the web PaymentMethodSelector uses (lib/payment-fees) → the iOS
// payment_method_picker shows a byte-identical fee. Charge stays server-auth.
export function paymentFee(methodJson: string, orderTotal: number | string): string {
  const m = JSON.parse(methodJson) as PaymentFeeMethod;
  const total = typeof orderTotal === "string" ? Number(orderTotal) : orderTotal;
  return JSON.stringify({
    feeSgd: feeAmountSgd(m, total),
    rateLabel: feeRateLabel(m, total),
  });
}
 
// ── Referee (first-order referral) discount — contract §25 ───────────────────
// The SAME `computeRefereeDiscountSgd` the web checkout + all 3 server charge
// paths use (5g + 25% caps). iOS CheckoutController calls this so the displayed
// referral discount is byte-identical to what the server charges — required now
// that iOS pins the displayed total (no drift on the money route). `itemsJson` =
// [{grams, quantity}]; spot + curveRate come from the cart summary (the same
// (spot, curve) pair the lock captures → matches the server's lockedSpot).
export function refereeDiscount(
  itemsJson: string,
  eligibleMaxGrams: number | string,
  refereeDiscountPct: number | string,
  lockedSpotSgd: number | string,
  curveRate: number | string,
): string {
  const items = JSON.parse(itemsJson || "[]") as { grams: number; quantity: number }[];
  const r = computeRefereeDiscountSgd({
    items,
    eligibleMaxGrams:   Number(eligibleMaxGrams),
    refereeDiscountPct: Number(refereeDiscountPct),
    lockedSpotSgd:      Number(lockedSpotSgd),
    curveRate:          Number(curveRate),
  });
  return JSON.stringify(r);
}
 
// ── Promo + referral no-stacking projection — contract §45 ───────────────────
// The SAME computeCheckoutDiscounts the web cart + checkout run: percent on the
// whole merchandise subtotal (capped), and NO STACKING (promo vs referral —
// keep the larger, drop the other). iOS cart + checkout call this so the
// displayed discount is byte-identical to what the server charges
// (resolveOrderDiscounts) — no drift, no tamper-422 on both-code orders. Pure
// (no BOOT needed). `argsJson` = CheckoutDiscountArgs. Client = DISPLAY, server
// = AUTHORITATIVE. Returns CheckoutDiscountResult JSON.
export function checkoutDiscounts(argsJson: string): string {
  const args = JSON.parse(argsJson || "{}") as CheckoutDiscountArgs;
  return JSON.stringify(computeCheckoutDiscounts({
    promo: args.promo ?? null,
    referralDiscountSgd: Number(args.referralDiscountSgd) || 0,
    merchandiseSubtotalSgd: Number(args.merchandiseSubtotalSgd) || 0,
    shippingFeeSgd: Number(args.shippingFeeSgd) || 0,
  }));
}
 
// ── Sell-to-Us (buyback) on-device view — contract §27 ───────────────────────
