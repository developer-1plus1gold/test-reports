/**
 * ezy2ship Shipment Booking API v1.0.3 — the SOAP envelopes and their responses
 * (2026-11-21). Pure: built and parsed here, sent by lib/singpost/booking-client.ts.
 *
 * Element names, order and the success codes are those of SingPost's guide
 * (sections "Step 1–3", Appendices A, C, D). Optional fields we never use are
 * left out rather than sent empty; the guide marks every one of them Optional
 * or Conditional (international only).
 */
import { el, find, childText, parseXml, XmlParseError, type XmlNode } from "@/lib/singpost/xml";
import type { SingpostAddress, SingpostShipment } from "@/lib/singpost/shipment";
import type { SingpostCollectionType } from "@/lib/singpost/settings";
 
export const SOAP_ENV_NS = "http://schemas.xmlsoap.org/soap/envelope/";
 
/** Which call, for the SOAPAction header: `${namespace}#CreateShipment`. */
export type SingpostOperation = "CreateShipment" | "GetShipmentLabels" | "AssignShipments";
 
export const CREATE_OK = "STATUS_SUCCESS";
export const LABEL_OK = "STATUS_GET_SHIPMENT_LABEL_OK";
export const ASSIGN_OK = "STATUS_SUCCESS";
/** createShipment refused our SenderReference as already used (Appendix A). */
export const DUPLICATE_SENDER_REFERENCE = "STATUS_ERROR_DUPLICATE_SENDER_REFERENCE";
/** assignShipments: the shipment is already confirmed (Appendix D) — a retry that already landed. */
export const ALREADY_ASSIGNED = "STATUS_SHIPMENT_IS_ALREADY_ASSIGNED_OR_DELETED";
export const INVALID_TICKET = "STATUS_INVALID_TICKET";
 
/** A SingPost refusal: their code (Appendix A/C/D) and message. */
export class SingpostApiError extends Error {
  constructor(readonly code: string, message: string) {
    super(`SingPost ${code}: ${message}`);
    this.name = "SingpostApiError";
  }
}
 
const money = (n: number) => n.toFixed(2);
const kg = (n: number) => String(Math.round(n * 1000) / 1000);
const cm = (n: number) => n.toFixed(2);
 
function envelope(namespace: string, body: ReturnType<typeof el>): string {
  return `<?xml version="1.0" encoding="UTF-8"?>` +
    `<soapenv:Envelope xmlns:soapenv="${SOAP_ENV_NS}" xmlns:sz="${escapeAttr(namespace)}">` +
    `<soapenv:Header/><soapenv:Body>${body.xml}</soapenv:Body></soapenv:Envelope>`;
}
 
function escapeAttr(s: string): string {
  return s.replace(/&/g, "&amp;").replace(/"/g, "&quot;").replace(/</g, "&lt;");
}
 
function address(tag: "ShipFromAddress" | "ShipToAddress", a: SingpostAddress) {
  return el(tag,
    el("Name1", a.name1),
    a.name2 ? el("Name2", a.name2) : null,
    el("AddressLine1", a.line1),
    a.line2 ? el("AddressLine2", a.line2) : null,
    a.line3 ? el("AddressLine3", a.line3) : null,
    el("State", "Singapore"),
    el("Town", "Singapore"),
    el("Postcode", a.postcode),
    el("Country", a.country),
    el("Contact", el("EmailAddress", a.email), el("PhoneNumber", a.phone)),
  );
}
 
/** Step 1 — createShipment (SG domestic, one parcel, one content line). */
export function buildCreateShipmentXml(namespace: string, ticket: string, s: SingpostShipment): string {
  return envelope(namespace, el("sz:createShipment",
    el("Ticket", ticket),
    el("ShipmentData", el("Shipments",
      el("ShipmentType", "REGULAR_SHIPMENT"),
      el("CarrierCode", s.carrierCode),
      el("ServiceCode", s.serviceCode),
      el("ShipByAddressCode", "BA"),
      el("TotalValueCurrency", "SGD"),
      el("TotalValue", money(s.totalValueSgd)),
      el("AccountNumber", s.accountNumber),
      el("SenderReference", s.senderReference),
      el("DeliveryInstruction", s.deliveryInstruction),
      el("NonDeliveryInstruction", "S"),
      el("ItemType", "P"),
      address("ShipFromAddress", s.from),
      address("ShipToAddress", s.to),
      el("HandlingUnits",
        el("Weight", kg(s.parcel.weightKg)),
        el("PackageLength", cm(s.parcel.lengthCm)),
        el("PackageWidth", cm(s.parcel.widthCm)),
        el("PackageHeight", cm(s.parcel.heightCm)),
        el("ContentDetailItems",
          el("ItemDescription", s.item.description),
          el("ItemWeight", kg(s.item.weightKg)),
          el("ItemQuantity", s.item.quantity),
          el("ItemCurrency", "SGD"),
          el("TotalAmount", money(s.item.totalSgd)),
        ),
      ),
    )),
  ));
}
 
/** Step 2 — getShipmentLabels. `Link` 0 = base64 in the response (we serve the bytes ourselves). */
export function buildGetLabelXml(namespace: string, ticket: string, trackingNumber: string): string {
  return envelope(namespace, el("sz:getShipmentLabels",
    el("Ticket", ticket),
    el("ShipmentNumber", trackingNumber),
    el("Link", 0),
  ));
}
 
export interface CollectionRequest { date: string; from: string; to: string }
 
/** Step 3 — assignShipments: confirm the shipment for first mile. */
export function buildAssignXml(
  namespace: string, ticket: string, trackingNumber: string,
  collectionType: SingpostCollectionType, request?: CollectionRequest,
): string {
  if (collectionType === "COLLECTION_REQUEST" && !request) {
    throw new Error("COLLECTION_REQUEST needs a collection date and window");
  }
  return envelope(namespace, el("sz:assignShipments",
    el("Ticket", ticket),
    el("data",
      el("CollectionType", collectionType),
      collectionType === "COLLECTION_REQUEST" ? el("CollectionRequest",
        el("CollectionDate", request!.date),
        el("CollectionTimeFrom", request!.from),
        el("CollectionTimeTo", request!.to),
      ) : null,
      el("Shipments", el("ShipmentNumber", trackingNumber)),
    ),
  ));
}

/** Parse a SOAP response, surfacing a SOAP Fault as a SingpostApiError. */
function parseSoap(xml: string): XmlNode {
  let root: XmlNode;
  try {
    root = parseXml(xml);
  } catch (err) {
    throw new SingpostApiError("bad_response", err instanceof XmlParseError ? `unreadable response (${err.message})` : "unreadable response");
  }
  const fault = find(root, "Fault");
  if (fault) {
    throw new SingpostApiError(childText(fault, "faultcode") ?? "soap_fault", childText(fault, "faultstring") ?? "SOAP fault");
  }
  return root;
}
 
/** createShipment → the tracking number, or throw with SingPost's code. */
export function parseCreateShipmentResponse(xml: string): string {
  const data = find(find(parseSoap(xml), "createShipmentResponse"), "data");
  const code = childText(data, "code");
  if (code !== CREATE_OK) {
    throw new SingpostApiError(code ?? "no_code", childText(data, "message") ?? "createShipment did not succeed");
  }
  const tracking = childText(find(data, "Shipment"), "ShipmentNumber");
  if (!tracking) throw new SingpostApiError("no_tracking", "createShipment succeeded but returned no tracking number");
  return tracking;
}
 
/** getShipmentLabels (Link 0) → the label PDF bytes. */
export function parseLabelResponse(xml: string): Uint8Array {
  const ret = find(find(parseSoap(xml), "getShipmentLabelsResponse"), "return");
  const code = childText(ret, "code");
  if (code !== LABEL_OK) throw new SingpostApiError(code ?? "no_code", childText(ret, "message") ?? "getShipmentLabels did not succeed");
  const data = (childText(ret, "data") ?? "").replace(/\s+/g, "");
  if (!data || !/^[A-Za-z0-9+/]+={0,2}$/.test(data)) throw new SingpostApiError("no_label", "label was not base64");
  const bytes = Uint8Array.from(Buffer.from(data, "base64"));
  // A PDF starts "%PDF". Anything else is not a label we can print.
  if (bytes.length < 5 || Buffer.from(bytes.subarray(0, 4)).toString("latin1") !== "%PDF") {
    throw new SingpostApiError("no_label", "label is not a PDF");
  }
  return bytes;
}
 
export interface AssignResult { manifestNumber: string | null; collectionDocketNumber: string | null; alreadyAssigned: boolean }
 
/** assignShipments → manifest (+ docket for a collection request). "Already assigned" is success. */
export function parseAssignResponse(xml: string): AssignResult {
  const a = find(find(parseSoap(xml), "assignShipmentsResponse"), "ShipmentAssigns");
  const code = childText(a, "code");
  if (code === ALREADY_ASSIGNED) return { manifestNumber: null, collectionDocketNumber: null, alreadyAssigned: true };
  if (code !== ASSIGN_OK) throw new SingpostApiError(code ?? "no_code", childText(a, "message") ?? "assignShipments did not succeed");
  return {
    manifestNumber: childText(a, "ManifestNumber"),
    collectionDocketNumber: childText(a, "CollectionDocketNumber"),
    alreadyAssigned: false,
  };
}
