/**
 * The small amount of XML the SingPost APIs need (2026-11-21).
 *
 * Both SingPost APIs speak XML: Shipment Booking (ezy2ship) is SOAP 1.1 and the
 * Tracking API is a plain XML POST. Their responses are tiny, flat documents, so
 * rather than add a dependency we build requests with an escaping element
 * helper and read responses with a strict, minimal parser:
 *
 *   - elements, attributes (read, ignored by the callers), text, CDATA,
 *     comments, the XML declaration and processing instructions;
 *   - the five predefined entities plus decimal / hex character references;
 *   - namespace prefixes are kept on `name` and stripped on `local`, because
 *     SingPost's responses use whatever prefix they like (`sz:`, `ns2:`, none).
 *
 * Malformed input (an unclosed or mismatched tag, a stray `<`) throws
 * XmlParseError — the clients turn that into an API error, never a guess. DTDs
 * are refused outright: there is no reason for a carrier to send one, and
 * refusing them removes every entity-expansion attack by construction.
 *
 * Pure: no server imports.
 */
 
export class XmlParseError extends Error {
  constructor(message: string) {
    super(message);
    this.name = "XmlParseError";
  }
}
 
export interface XmlNode {
  /** Qualified name as written, e.g. "sz:createShipmentResponse". */
  name: string;
  /** Name without its namespace prefix, e.g. "createShipmentResponse". */
  local: string;
  attrs: Record<string, string>;
  children: XmlNode[];
  /** Concatenated text directly inside this element (not its descendants). */
  text: string;
}
 
// ─── Building ───────────────────────────────────────────────────────────────
 
const ESCAPES: Record<string, string> = { "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&apos;" };
 
/**
 * Escape text for an XML element or attribute. Characters XML 1.0 cannot carry
 * at all (C0 controls other than tab / LF / CR, lone surrogates, U+FFFE/U+FFFF)
 * are dropped rather than escaped — a character reference to them is itself
 * ill-formed, and a customer's pasted address can contain them.
 */
export function escapeXml(value: string): string {
  return value
    // eslint-disable-next-line no-control-regex
    .replace(/[\u0000-\u0008\u000B\u000C\u000E-\u001F￾￿]|[\uD800-\uDBFF](?![\uDC00-\uDFFF])|(?<![\uD800-\uDBFF])[\uDC00-\uDFFF]/g, "")
    .replace(/[&<>"']/g, (c) => ESCAPES[c]);
}
 
/** A child of an element: nested markup, a scalar to escape, or nothing. */
export type XmlChild = XmlFragment | string | number | null | undefined;
 
/** Already-built markup, so it is never escaped twice. */
export class XmlFragment {
  constructor(readonly xml: string) {}
  toString(): string { return this.xml; }
}
 
const NAME = /^[A-Za-z_][\w.-]*(?::[A-Za-z_][\w.-]*)?$/;
 
/**
 * `<name>children</name>`, or `<name/>` when there are none. Scalars are
 * escaped; fragments are inserted as they are. A name that is not a plain XML
 * name throws — names are ours, never data, so this is a programming error.
 */
export function el(name: string, ...children: XmlChild[]): XmlFragment {
  if (!NAME.test(name)) throw new Error(`Invalid XML element name: ${name}`);
  const inner = children
    .filter((c) => c !== null && c !== undefined)
    .map((c) => (c instanceof XmlFragment ? c.xml : escapeXml(String(c))))
    .join("");
  return new XmlFragment(inner === "" ? `<${name}/>` : `<${name}>${inner}</${name}>`);
}
 
// ─── Parsing ────────────────────────────────────────────────────────────────
 
const NAMED_ENTITIES: Record<string, string> = { amp: "&", lt: "<", gt: ">", quot: '"', apos: "'" };
 
function decodeEntities(s: string): string {
  return s.replace(/&(#x[0-9a-fA-F]+|#[0-9]+|[A-Za-z]+);/g, (whole, ref: string) => {
    if (ref[0] === "#") {
      const code = ref[1] === "x" || ref[1] === "X" ? parseInt(ref.slice(2), 16) : parseInt(ref.slice(1), 10);
      if (!Number.isFinite(code) || code < 0 || code > 0x10ffff) throw new XmlParseError(`Invalid character reference ${whole}`);
      return String.fromCodePoint(code);
    }
    const named = NAMED_ENTITIES[ref];
    if (named === undefined) throw new XmlParseError(`Unknown entity ${whole}`);
    return named;
  });
}
 
const localName = (qname: string): string => qname.slice(qname.indexOf(":") + 1);
 
/**
 * Parse a document and return its root element. Throws XmlParseError on
 * anything that is not well-formed, on a DTD, and on an empty document.
 */
export function parseXml(input: string): XmlNode {
  const src = input.charCodeAt(0) === 0xfeff ? input.slice(1) : input;
  const stack: XmlNode[] = [];
  let root: XmlNode | null = null;
  let i = 0;
 
  const appendText = (t: string) => {
    const top = stack[stack.length - 1];
    if (top) top.text += t;
    else if (t.trim() !== "") throw new XmlParseError("Text outside the root element");
  };
 
  while (i < src.length) {
    const lt = src.indexOf("<", i);
    if (lt === -1) { appendText(decodeEntities(src.slice(i))); break; }
    if (lt > i) appendText(decodeEntities(src.slice(i, lt)));
 
    if (src.startsWith("<!--", lt)) {
      const end = src.indexOf("-->", lt + 4);
      if (end === -1) throw new XmlParseError("Unterminated comment");
      i = end + 3;
    } else if (src.startsWith("<![CDATA[", lt)) {
      const end = src.indexOf("]]>", lt + 9);
      if (end === -1) throw new XmlParseError("Unterminated CDATA section");
      appendText(src.slice(lt + 9, end));
      i = end + 3;
    } else if (src.startsWith("<?", lt)) {
      const end = src.indexOf("?>", lt + 2);
      if (end === -1) throw new XmlParseError("Unterminated processing instruction");
      i = end + 2;
    } else if (src.startsWith("<!", lt)) {
      throw new XmlParseError("Document type declarations are not accepted");
    } else if (src[lt + 1] === "/") {
      const end = src.indexOf(">", lt);
      if (end === -1) throw new XmlParseError("Unterminated end tag");
      const name = src.slice(lt + 2, end).trim();
      const open = stack.pop();
      if (!open || open.name !== name) throw new XmlParseError(`Mismatched end tag </${name}>`);
      i = end + 1;
    } else {
      const end = findTagEnd(src, lt);
      const raw = src.slice(lt + 1, end);
      const selfClosing = raw.endsWith("/");
      const body = selfClosing ? raw.slice(0, -1) : raw;
      const m = /^([A-Za-z_][\w.:-]*)/.exec(body);
      if (!m) throw new XmlParseError("Invalid start tag");
      const node: XmlNode = { name: m[1], local: localName(m[1]), attrs: parseAttrs(body.slice(m[1].length)), children: [], text: "" };
      const parent = stack[stack.length - 1];
      if (parent) parent.children.push(node);
      else if (root) throw new XmlParseError("More than one root element");
      else root = node;
      if (!selfClosing) stack.push(node);
      i = end + 1;
    }
  }
 
  if (stack.length > 0) throw new XmlParseError(`Unclosed element <${stack[stack.length - 1].name}>`);
  if (!root) throw new XmlParseError("Empty document");
  return root;
}
 
/** Index of the `>` closing the tag at `lt`, skipping `>` inside quoted attribute values. */
function findTagEnd(src: string, lt: number): number {
  let quote: string | null = null;
  for (let j = lt + 1; j < src.length; j++) {
    const c = src[j];
    if (quote) { if (c === quote) quote = null; }
    else if (c === '"' || c === "'") quote = c;
    else if (c === "<") throw new XmlParseError("Unexpected < inside a tag");
    else if (c === ">") return j;
  }
  throw new XmlParseError("Unterminated start tag");
}
 
function parseAttrs(s: string): Record<string, string> {
  const attrs: Record<string, string> = {};
  const re = /\s*([A-Za-z_][\w.:-]*)\s*=\s*("([^"]*)"|'([^']*)')/y;
  let pos = 0;
  while (pos < s.length) {
    if (/^\s*$/.test(s.slice(pos))) break;
    re.lastIndex = pos;
    const m = re.exec(s);
    if (!m) throw new XmlParseError("Malformed attribute");
    attrs[m[1]] = decodeEntities(m[3] ?? m[4] ?? "");
    pos = re.lastIndex;
  }
  return attrs;
}

/** First direct child with this local name. */
export function child(node: XmlNode | null | undefined, local: string): XmlNode | null {
  return node?.children.find((c) => c.local === local) ?? null;
}
 
/** Every direct child with this local name. */
export function childrenNamed(node: XmlNode | null | undefined, local: string): XmlNode[] {
  return node ? node.children.filter((c) => c.local === local) : [];
}
 
/** First descendant (depth-first, document order) with this local name. */
export function find(node: XmlNode | null | undefined, local: string): XmlNode | null {
  if (!node) return null;
  for (const c of node.children) {
    if (c.local === local) return c;
    const hit = find(c, local);
    if (hit) return hit;
  }
  return null;
}
 
/** Trimmed text of the first direct child with this local name, or null when absent or empty. */
export function childText(node: XmlNode | null | undefined, local: string): string | null {
  const t = child(node, local)?.text.trim();
  return t ? t : null;
}
