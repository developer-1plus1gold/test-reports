/**
 * Book an order with SingPost from Dispatch (2026-11-21).
 *
 * Called by POST /api/admin/orders/[id]/singpost. Same gates as every other
 * courier booking: switched on + configured, the order may move to SingPost
 * (lib/singpost/dispatch.ts), packed, and dispatchBlocker (consignment gold,
 * MAG group-delivery hold, marketplace cancel) — the one dispatch gate.
 *
 *   1. createShipment     → tracking number          (nothing is written if it fails)
 *   2. assignShipments    → confirmed for collection  (the configured first-mile type)
 *   3. one guarded write: shipper 'singpost', the service, tracking number,
 *      public tracking link, status 'booked' — or 'failed' with stage "created"
 *      when step 2 failed, so the next tap only confirms (confirmSingpost).
 *
 * The customer's delivery fee is never touched. What SingPost charges us is on
 * the monthly rate card, not per booking, so no trip cost is recorded.
 */

/** Today's date in Singapore, YYYY-MM-DD — the collection-request day. */
export function sgToday(nowMs = Date.now()): string {
  return new Date(nowMs + 8 * 3600_000).toISOString().slice(0, 10);
}
 
/** The collection request for COLLECTION_REQUEST, from the configured window; undefined otherwise. */
export function collectionRequestFor(c: SingpostConfig, nowMs = Date.now()): CollectionRequest | undefined {
  if (c.collectionType !== "COLLECTION_REQUEST" || !c.collectionWindow) return undefined;
  return { date: sgToday(nowMs), from: c.collectionWindow.from, to: c.collectionWindow.to };
}
 
/** A SingPost refusal as one line for the operator and shipper_error. */
function reason(err: unknown): string {
  if (err instanceof SingpostApiError) return `${err.code}: ${err.message.replace(/^SingPost [^:]+: /, "")}`.slice(0, 300);
  return err instanceof Error ? err.message.slice(0, 300) : "unknown error";
}
 
/** Switched on, credentials set, and the service's codes saved — else a 409 saying which. */
async function readiness(db: SupabaseClient, service: SingpostService): Promise<{ ok: true; settings: SingpostSettings } | { ok: false; result: BookResult }> {
  if (!isSingpostBookingConfigured()) {
    return { ok: false, result: fail(409, "SingPost is not connected in this environment yet (no API credentials).") };
  }
  const settings = await loadSingpostSettings(db);
  if (!settings.enabled) return { ok: false, result: fail(409, "SingPost is switched off — turn it on in Admin → Shippers first.") };
  const missing = missingSettings(settings.config, service);
  if (missing.length) {
    return { ok: false, result: fail(409, `SingPost settings are incomplete for ${SINGPOST_SERVICE_LABEL[service]}: ${missing.join(", ")} (Admin → Shippers).`) };
  }
  return { ok: true, settings };
}

/** Retry step 2 only, for a shipment created but not confirmed. */
async function confirmSingpost(db: SupabaseClient, order: OrderRow, config: SingpostConfig, nowMs: number): Promise<BookResult> {
  const trackingNo = order.shipper_order_id as string;
  const goldBlocker = await dispatchBlocker(db, order.id);
  if (goldBlocker) return fail(409, goldBlocker, { code: "awaiting_gold" });
 
  let assigned;
  try {
    assigned = await assignShipment(trackingNo, config.collectionType!, collectionRequestFor(config, nowMs));
  } catch (err) {
    const msg = reason(err);
    console.error(`[singpost/book] confirm ${order.id} ${trackingNo}:`, msg);
    await guardedUpdate(db, order, { shipper_error: `Created on SingPost but not confirmed for collection — ${msg}` });
    return fail(502, `SingPost did not confirm ${trackingNo} for collection — ${msg}`, { trackingNo });
  }
  const events = Array.isArray(order.shipper_status_events) ? order.shipper_status_events as Array<{ singpost_reference?: unknown }> : [];
  const ref = [...events].reverse().find((e) => typeof e?.singpost_reference === "string")?.singpost_reference;
  const ok = await guardedUpdate(db, order, {
    shipper_status: "booked",
    shipper_error: null,
    shipper_status_events: withEvent(order, {
      status: "booked", at: new Date(nowMs).toISOString(),
      singpost_reference: ref, singpost_stage: "assigned",
      ...(assigned.manifestNumber ? { manifest: assigned.manifestNumber } : {}),
      ...(assigned.collectionDocketNumber ? { docket: assigned.collectionDocketNumber } : {}),
      note: assigned.alreadyAssigned
        ? `SingPost says ${trackingNo} was already confirmed for collection`
        : `SingPost ${trackingNo} confirmed for collection (${config.collectionType})`,
    }),
  });
  if (!ok) return fail(409, "The order changed while this ran — refresh and try again", { trackingNo });
  return { status: 200, body: { ok: true, trackingNo, shipper_status: "booked", confirmed: true } };
}
