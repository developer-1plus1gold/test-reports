/**
 * Order → SingPost shipment (2026-11-21). Pure: no env, no I/O.
 *
 * Applies the limits of the ezy2ship Shipment Booking API v1.0.3 (createShipment
 * field table) BEFORE anything is sent, so a parcel SingPost would refuse is
 * refused here with a message the operator can act on:
 *
 *   Name1 / Name2 / AddressLine1–3   35 characters each
 *   Postcode (SG)                    6 digits
 *   PhoneNumber                      20, digits + "-" only (and a leading "+")
 *   ShipTo EmailAddress              20 (as printed in the guide; Sender is 100)
 *   SenderReference                  20, unique per shipment
 *   DeliveryInstruction              50
 *   TotalValue                       Float[7,2] → at most 99,999.99
 *   Weight                           Float[2,3] kg, > 0
 *   PackageLength/Width/Height       Float[3,2] cm, > 0
 *   ContentDetailItems               1–20; item weights must equal the parcel weight
 *
 * What we always send (Mukesh, 2026-11-20): NonDeliveryInstruction "S" (return
 * to sender), ItemType "P", ShipByAddressCode "BA", no InsuredValue (an annual
 * policy covers us, not per-parcel cover), no COD, no DeliveryDate/Time (a
 * promised slot is a customer-facing decision not yet made).
 *
 * Nothing on the label names the contents: the one content line is "Sealed
 * package", as for SPX (carrier-sender-anonymity.test.ts).
 */
import type { SingpostConfig } from "@/lib/singpost/settings";

export const MAX_SENDER_REFERENCE = 20;
export const MAX_SHIPTO_EMAIL = 20;

/** Collapse whitespace and drop characters SingPost (and XML) cannot carry. */
export function cleanText(s: string | null | undefined): string {
  return String(s ?? "")
    // eslint-disable-next-line no-control-regex
    .replace(/[\u0000-\u001F\u007F]/g, " ")
    .replace(/\s+/g, " ")
    .trim();
}
 
/**
 * Wrap text into lines of at most `max` characters, breaking at spaces, and
 * after a comma when that is the only break. A single word longer than `max`
 * is split hard — rare (a long building name with no spaces), but it must not
 * be cut off.
 */
export function wrapText(text: string, max = MAX_LINE): string[] {

/**
 * Our two address lines in SingPost's three 35-character lines. Each of ours
 * starts its own line when that fits (block + street, then unit + building);
 * otherwise the two are wrapped as one text. Returns null when even that needs
 * more than three lines — SingPost would cut the address, so we refuse instead.
 */
export function splitAddressLines(line1: string | null | undefined, line2: string | null | undefined): [string, string, string] | null {
  const a = wrapText(line1 ?? "");
  const b = wrapText(line2 ?? "");
  let lines = [...a, ...b];
  if (lines.length > 3) lines = wrapText([cleanText(line1), cleanText(line2)].filter(Boolean).join(", "));
  if (lines.length === 0 || lines.length > 3) return null;
  return [lines[0], lines[1] ?? "", lines[2] ?? ""];
}
 
/** A Singapore postcode: six digits once spaces are removed, else null. */
export function normaliseSgPostcode(raw: string | null | undefined): string | null {
  const s = String(raw ?? "").replace(/\s+/g, "");
  return /^\d{6}$/.test(s) ? s : null;
}
 
/**
 * A phone SingPost accepts: digits and "-" only, an optional leading "+", at
 * most 20 characters. A bare 8-digit Singapore number gets +65 (as everywhere
 * else in the app — lib/auth/phone-utils.ts). Null when nothing usable remains.
 */
export function normaliseSingpostPhone(raw: string | null | undefined): string | null {
  const t = String(raw ?? "").trim();
  if (!t) return null;
  const digits = t.replace(/\D/g, "");
  const phone = t.startsWith("+") ? `+${digits}` : /^[3689]\d{7}$/.test(digits) ? `+65${digits}` : digits;
  return digits.length >= 8 && phone.length <= 20 ? phone : null;
}
 
/** A person's name in Name1 + Name2 (35 each), or null when it needs more. */
export function splitName(raw: string | null | undefined): [string, string] | null {
  const lines = wrapText(raw ?? "");
  if (lines.length === 0 || lines.length > 2) return null;
  return [lines[0], lines[1] ?? ""];
}
 
/**
 * SenderReference: unique per shipment, at most 20 characters. Our short order
 * number (what the packing slip and Dispatch show), with a suffix for a
 * re-booking so SingPost's duplicate-reference check never refuses a genuine
 * second parcel: 1P1G-1A2B3C4D, then 1P1G-1A2B3C4D-2.
 */
export function singpostSenderReference(orderId: string, attempt = 1): string {
  const short = orderId.replace(/[^A-Za-z0-9]/g, "").slice(0, 8).toUpperCase();
  const ref = attempt > 1 ? `1P1G-${short}-${attempt}` : `1P1G-${short}`;
  return ref.slice(0, MAX_SENDER_REFERENCE);
}
 
const round = (n: number, dp: number) => Math.round(n * 10 ** dp) / 10 ** dp;
 
/** Build the shipment or throw SingpostMappingError naming every problem at once. */
export function buildSingpostShipment(input: ShipmentInput): SingpostShipment {
  const problems: string[] = [];
  const r = input.recipient;
 
  const toName = splitName(r.name);
  if (!toName) problems.push(r.name ? "recipient name is longer than 70 characters" : "recipient name is missing");
  const toLines = splitAddressLines(r.line1, r.line2);
  if (!toLines) problems.push(r.line1 ? "delivery address is too long for SingPost's three 35-character lines" : "delivery address is missing");
  const toPostcode = normaliseSgPostcode(r.postalCode);
  if (!toPostcode) problems.push("delivery postcode must be 6 digits");
  const toPhone = normaliseSingpostPhone(r.phone);
  if (!toPhone) problems.push("recipient phone is missing or not a phone number");
 
  const s = input.config.sender;
  const fromName = splitName(s.name);
  if (!fromName) problems.push("sender name is invalid");
  const fromPostcode = normaliseSgPostcode(s.postcode);
  if (!fromPostcode) problems.push("sender postcode must be 6 digits");
  const fromPhone = normaliseSingpostPhone(input.senderPhone);
  if (!fromPhone) problems.push("sender phone is not a phone number");
  const fromLines = [s.line1, s.line2, s.line3].map(cleanText);
  if (!fromLines[0] || fromLines.some((l) => l.length > MAX_LINE)) problems.push("sender address lines must be 1–35 characters");
 
  const email = cleanText(input.contactEmail);
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) problems.push("contact email is not an email address");
  else if (email.length > MAX_SHIPTO_EMAIL) problems.push(`contact email must be at most ${MAX_SHIPTO_EMAIL} characters (SingPost's recipient email limit)`);
 
  const value = round(Number(input.goodsSgd), 2);
  if (!Number.isFinite(value) || value <= 0) problems.push("the parcel has no goods value");
  else if (value > MAX_TOTAL_VALUE_SGD) problems.push("goods value is above S$99,999.99, the most SingPost's TotalValue field holds — book by hand");
 
  for (const [k, v] of [["carrier code", input.carrierCode], ["service code", input.serviceCode], ["account number", input.accountNumber]] as const) {
    if (!cleanText(v)) problems.push(`${k} is missing`);
  }
 
  const p = input.config.parcel;
  const goldKg = (Math.max(0, Number(input.totalGrams) || 0) + PACKAGING_GRAMS) / 1000;
  const weightKg = round(Math.max(p.weightKg, goldKg), 3);
  if (weightKg > 99.999) problems.push("parcel weight is above SingPost's 99.999 kg field");
  const dims = [p.lengthCm, p.widthCm, p.heightCm].map((d) => round(d, 2));
  if (dims.some((d) => !(d > 0) || d > 999.99)) problems.push("parcel dimensions must be between 0 and 999.99 cm");
 
  if (problems.length > 0) throw new SingpostMappingError(problems);
 
  return {
    carrierCode: cleanText(input.carrierCode),
    serviceCode: cleanText(input.serviceCode),
    accountNumber: cleanText(input.accountNumber),
    senderReference: singpostSenderReference(input.orderId, input.attempt),
    totalValueSgd: value,
    deliveryInstruction: SINGPOST_DELIVERY_INSTRUCTION,
    from: {
      name1: fromName![0], name2: fromName![1],
      line1: fromLines[0], line2: fromLines[1], line3: fromLines[2],
      postcode: fromPostcode!, country: "SG", email, phone: fromPhone!,
    },
    to: {
      name1: toName![0], name2: toName![1],
      line1: toLines![0], line2: toLines![1], line3: toLines![2],
      postcode: toPostcode!, country: "SG", email, phone: toPhone!,
    },
    parcel: { weightKg, lengthCm: dims[0], widthCm: dims[1], heightCm: dims[2] },
    item: { description: SINGPOST_ITEM_DESCRIPTION, weightKg, quantity: 1, totalSgd: value },
  };
}
