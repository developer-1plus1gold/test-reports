/**
 * The house letterhead for internal governance documents, served as HTML.
 *
 * WHY HTML AND NOT PDF (operator decision, 2026-09-08). These documents were
 * rendered through `generatePolicyPdf`, the customer-facing policy renderer.
 * That gave a PDF with the wrong letterhead and a flat, technical structure —
 * they read as engineering notes rather than as internal policy documents.
 * The operator's instruction was: take the header and footer from the IPPC,
 * keep the output HTML, and print to PDF from the browser when needed. The
 * `@media print` block below is what makes that print cleanly on A4.
 *
 * ⚠️ THE CHROME IS A DELIBERATE SNAPSHOT OF `content/governance/ippc-v1.3.html`,
 * NOT A SHARED IMPORT. The IPPC is a versioned, signed compliance artefact
 * (v1.2, June 2026, with a Compliance Officer sign-off block). Reaching into it
 * to extract a stylesheet would mean editing a document that is supposed to be
 * immutable between versions. So its styling was copied here once, and THIS is
 * the going-forward chrome for governance documents authored after 2026-09-08.
 * If the IPPC is reissued at v1.3, re-sync deliberately — do not assume drift
 * is a bug.
 *
 * A document is a BODY FRAGMENT in `content/governance/*.html` plus frontmatter;
 * this module supplies the page, the letterhead, the cover table and the closing
 * rule. Fragments therefore hold only content, and every document changes shape
 * together when this file changes.
 */

   */

}

/** Escape a frontmatter value before it lands in HTML. */
function esc(s: string): string {
  return s.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;");
}

/**
 * Wrap a body fragment in the house letterhead.
 *
 * `body` is trusted repository content (`content/governance/*.html`, PR-gated
 * like every policy text in this codebase) — it is authored HTML, not user
 * input, and there is no path by which an operator or customer can supply it.
 * Frontmatter VALUES are escaped anyway, because they land in attributes and
 * headings and escaping them costs nothing.
 */
/**
 * @param signature Sign-off state for this document. Optional so an unwired
 *   caller still renders a document rather than throwing — but when it is
 *   omitted NO block is drawn at all, deliberately. A block that silently
 *   reported "not yet signed" because the caller forgot to pass state would be
 *   indistinguishable from a genuinely unsigned document, and this whole
 *   mechanism exists so those two can never look alike.
 */
export function renderGovernanceDocument(
  meta: DocMeta,
  body: string,
  signature?: { state: SignatureState; generatedAt: Date },
): string {
  const updated = humanDate(meta.last_updated);
