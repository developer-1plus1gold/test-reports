/**
 * PDPA retention — Phase 0 (2026-09-07): the schedule as data + the pure
 * eligibility classifier behind the REPORT-ONLY hub view.
 *
 * Nothing here deletes or anonymises anything. It computes "what WOULD be
 * eligible under the schedule" so the blast radius is visible before any
 * destructive phase is built. The schedule of record is
 * content/governance/pdpa-retention-schedule.html.
 */
 
/** R3 — LLM/agent input bodies: keep the decision + metadata, not the raw prompt. */
export const LLM_INPUT_RETENTION_DAYS = 90;
/**
 * R2 — un-transacted (no AML hold) account inactivity window.
 *
 * ▶ TWO CONSTANTS FOR ONE WINDOW, DELIBERATELY. The classifier needs a day
 * count to compare a timestamp against; the schedule states the window in
 * months, and that is the wording a regulator reads. Declaring the months
 * figure rather than DERIVING it is the fix for a real defect found on
 * 2026-09-11: the tab rendered `DAYS / 365 | 0` months — dividing a day count
 * by a YEAR — and so displayed **"2 months inactive"** against a schedule that
 * says 24. Enforcement was never wrong (the cron has always used the 730), but
 * the governance hub was telling its reader we anonymise a year sooner than we
 * do. `retention-window-consistency.test.ts` pins the two to each other.
 */
export const UNTRANSACTED_INACTIVE_DAYS = 730;
/** The same window as the schedule of record states it (content/governance/pdpa-retention-schedule.html). */
export const UNTRANSACTED_INACTIVE_MONTHS = 24;
/** R1 — AML record-keeping: anonymise only AFTER this, never within it. */
export const AML_RETENTION_YEARS = 5;

  action: string;
  /** true = a destructive phase computes a live eligible count for it;

  live: boolean;
  note?: string;
}

/** ISO cutoff `days` before `now` — anything older than the return value is
 *  eligible. Pure. */
export function cutoffIso(now: Date, days: number): string {
  return new Date(now.getTime() - days * 86_400_000).toISOString();
}
 
/** ISO cutoff `years` before `now`, calendar-correct (handles leap years). */
export function yearsCutoffIso(now: Date, years: number): string {
  const d = new Date(now.getTime());
  d.setFullYear(d.getFullYear() - years);
  return d.toISOString();
}
 
export interface AccountRow {
  id: string;
  last_activity_at: string | null;
  created_at: string;
}
 
export interface AccountRetentionCounts {
  /** Transacting customers whose LAST order is older than the 5-yr AML window. */
  transactedPast5y: number;
  /** Never-transacted accounts inactive beyond the 24-month window. */
  untransactedInactive: number;
  /** Everything else — not yet eligible. */
  active: number;
}
 
export interface AccountRetentionSelection {
  /** R1 — transacting, last order older than the 5-yr AML window. */
  transactedPast5y: string[];
  /** R2 — never transacted, inactive beyond the 24-month window. */
  untransactedInactive: string[];
  /** Not yet eligible. */
  active: string[];
}
 
/**
 * Pure: partition accounts into the retention classes, returning the ids in
 * each. `classifyAccountRetention` (counts) is derived from this, so the
 * report and the anonymisation cron can never disagree about WHICH accounts
 * are eligible vs how many.
 *
 * An account "has transacted" iff it appears in `lastOrderIsoByCustomer`. The
 * AML hold means a transacting account is NEVER eligible inside 5 years, so it
 * can only fall into `transactedPast5y` once its last order clears the window.
 */
export function selectAccountRetention(
  accounts: AccountRow[],
  lastOrderIsoByCustomer: Map<string, string>,
  now: Date = new Date(),
  opts: { inactiveDays?: number; amlYears?: number } = {},
): AccountRetentionSelection {
  const inactiveCutoff = cutoffIso(now, opts.inactiveDays ?? UNTRANSACTED_INACTIVE_DAYS);
  const amlCutoff = yearsCutoffIso(now, opts.amlYears ?? AML_RETENTION_YEARS);
 
  const sel: AccountRetentionSelection = { transactedPast5y: [], untransactedInactive: [], active: [] };
 
  for (const a of accounts) {
    const lastOrder = lastOrderIsoByCustomer.get(a.id);
    if (lastOrder) {
      // Transacting → AML hold; eligible only once the LAST order clears 5y.
      if (lastOrder < amlCutoff) sel.transactedPast5y.push(a.id);
      else sel.active.push(a.id);
    } else {
      // Never transacted → no AML hold; eligible once inactive past the window.
      const activitySignal = a.last_activity_at ?? a.created_at;
      if (activitySignal < inactiveCutoff) sel.untransactedInactive.push(a.id);
      else sel.active.push(a.id);
    }
  }
  return sel;
}
 
/** Counts, derived from the selection so the two can never diverge. */
export function classifyAccountRetention(
  accounts: AccountRow[],
  lastOrderIsoByCustomer: Map<string, string>,
  now: Date = new Date(),
  opts: { inactiveDays?: number; amlYears?: number } = {},
): AccountRetentionCounts {
  const s = selectAccountRetention(accounts, lastOrderIsoByCustomer, now, opts);
  return {
    transactedPast5y: s.transactedPast5y.length,
    untransactedInactive: s.untransactedInactive.length,
    active: s.active.length,
  };
}
 
/**
 * The accounts that must NEVER be anonymised by the retention cron, or that
 * have already been anonymised (so we neither re-process them nor keep counting
 * them as "pending"). Every set is by account id.
 */
export interface AnonymisationExclusions {
  /** External-channel house customers (Shopee / in-person) — operational rows. */
  houseIds: Set<string>;
  /** Admin-team accounts (matched by phone) — never anonymise an operator. */
  adminIds: Set<string>;
  /** Internal/demo accounts (e.g. demo@ / @1plus1gold.com). */
  internalIds: Set<string>;
  /** Already anonymised (tombstoned / deleted_at) — idempotency. */
  alreadyAnonymisedIds: Set<string>;
}
 
export type ExclusionReason = "house" | "admin" | "internal" | "already_anonymised";
 
export interface AnonymisationTargets {
  /** Ids that survive every exclusion and will be anonymised. */
  targets: string[];
  /** Ids removed, with the first reason that excluded them (for the log). */
  excluded: Array<{ id: string; reason: ExclusionReason }>;
}
 
/**
 * Pure: given eligible ids and the exclusion sets, decide the final targets.
 * This is the safety gate between "eligible under the schedule" and "actually
 * anonymised" — it is deliberately pure + tested so the destructive call is
 * only ever made on an id that survived every exclusion. Order of precedence
 * (internal/house/admin first) only affects the recorded reason, not whether an
 * id is excluded.
 */
export function filterAnonymisationTargets(
  ids: string[],
  ex: AnonymisationExclusions,
): AnonymisationTargets {
  const targets: string[] = [];
  const excluded: Array<{ id: string; reason: ExclusionReason }> = [];
  for (const id of ids) {
    if (ex.internalIds.has(id)) excluded.push({ id, reason: "internal" });
    else if (ex.houseIds.has(id)) excluded.push({ id, reason: "house" });
    else if (ex.adminIds.has(id)) excluded.push({ id, reason: "admin" });
    else if (ex.alreadyAnonymisedIds.has(id)) excluded.push({ id, reason: "already_anonymised" });
    else targets.push(id);
  }
  return { targets, excluded };
}
