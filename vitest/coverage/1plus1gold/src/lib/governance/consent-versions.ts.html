/**
 * PDPA G10 — consent versioning.
 *
 * Records WHICH version of Terms + Privacy a customer accepted, and when, so
 * we can later evidence it (PDPC expectation). The "version" is deliberately
 * the policy document's own `last_updated` frontmatter date — so editing
 * content/terms.md or content/privacy.md IS the version bump; there is no
 * second discipline to keep in sync.
 *
 * `stampSignupConsent` is called from the AUTH paths only (WhatsApp/email OTP
 * verify + Google/Apple callback), never the money path. The server stamps the
 * CURRENT versions — the value is never accepted from the client, so it cannot
 * be forged. It writes ONCE (first sign-in): the `.is("consent_accepted_at",
 * null)` guard means a later re-sign-in never re-stamps a newer version, which
 * is what makes "the version in force on the date you registered" a fact we
 * can stand behind (content/terms.md §13). Fail-soft in every branch — a
 * consent-log hiccup must never break sign-in.
 */
import type { SupabaseClient } from "@supabase/supabase-js";
import { loadMarkdown } from "@/lib/content/loadMarkdown";
import { getClientIp } from "@/lib/rate-limit";
 
export interface ConsentVersions {
  terms:   string | null;
  privacy: string | null;
}
 
/**
 * The versions currently in force = the `last_updated` date of each policy
 * doc. Fail-soft to null per-doc (a missing/damaged file must not break
 * sign-in). Reads the same files every surface serves via /api/content, so
 * the recorded version always matches what the customer could actually read.
 */
export async function getCurrentConsentVersions(): Promise<ConsentVersions> {
  const [terms, privacy] = await Promise.all([
    loadMarkdown("terms").then((d) => d.last_updated).catch(() => null),
    loadMarkdown("privacy").then((d) => d.last_updated).catch(() => null),
  ]);
  return { terms, privacy };
}
 
const UA_MAX = 512;
 
/**
 * Stamp the current Terms/Privacy versions onto a profile at first sign-in.
 *
 * Idempotent by design: the update is gated on `consent_accepted_at IS NULL`,
 * so the FIRST successful sign-in wins and every later one is a no-op. Returns
 * quietly on any error (fail-soft — sign-in must never fail because a consent
 * row could not be written).
 *
 * @param admin  a service-role client (bypasses RLS to write the profile)
 * @param userId the auth user id whose profile to stamp
 * @param req    the incoming request (for client IP + user-agent, evidence only)
 */
export async function stampSignupConsent(
  admin: SupabaseClient,
  userId: string | undefined,
  req: Request,
): Promise<void> {
  if (!userId) return;
  try {
    const { terms, privacy } = await getCurrentConsentVersions();
    const ua = (req.headers.get("user-agent") ?? "").slice(0, UA_MAX) || null;
    const ip = getClientIp(req, userId); // anti-spoof form: ip:/uid:/fp:
    const { error } = await admin
      .from("profiles")
      .update({
        terms_version:       terms,
        privacy_version:     privacy,
        consent_accepted_at: new Date().toISOString(),
        consent_ip:          ip,
        consent_user_agent:  ua,
      })
      .eq("id", userId)
      .is("consent_accepted_at", null); // first sign-in wins; never overwrite
    if (error) console.warn("[consent] stampSignupConsent failed (non-fatal):", error.message);
  } catch (err) {
    console.warn("[consent] stampSignupConsent threw (non-fatal):", err instanceof Error ? err.message : err);
  }
}
