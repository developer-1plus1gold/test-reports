/**
 * The system map (docs/architecture.html) is a BODY FRAGMENT — it also
 * publishes unchanged as an Artifact — so the admin route wraps it in a
 * minimal document. Kept here, not in the route file: a route.ts may export
 * only handlers + segment config (route-exports-valid.test.ts).
 */
export const ARCHITECTURE_CSP =
  "default-src 'none'; style-src 'unsafe-inline' https://fonts.googleapis.com; font-src https://fonts.gstatic.com; img-src data:; frame-ancestors 'none'";
 
export function wrapArchitectureFragment(fragment: string): string {
  return `<!doctype html>\n<html lang="en">\n<head>\n<meta charset="utf-8">\n<meta name="viewport" content="width=device-width, initial-scale=1">\n<meta name="robots" content="noindex, nofollow">\n<title>1Plus1Gold — System map</title>\n</head>\n<body>\n${fragment}\n</body>\n</html>\n`;
}
