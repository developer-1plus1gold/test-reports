/**
 * runRetentionAnonymisation — the orchestration behind the
 * retention-anonymise-accounts cron (PDPA Phase 2+3 / R1 + R2).
 *
 * Lives in a lib (not the route file) for two reasons: a Next.js App Router
 * `route.ts` may only export route handlers + config (exporting anything else
 * fails `next build`), and keeping it here lets the unit tests drive it with a
 * mock admin + injectable clock without importing the route.
 *
 * Anonymises eligible, non-excluded accounts and logs each. Never throws for a
 * single account failure — it records the error and moves on, so one bad row
 * can't strand the rest. See the route for the full contract.
 */

import { resolveAnonymisationExclusions } from "@/lib/governance/retention-exclusions";
