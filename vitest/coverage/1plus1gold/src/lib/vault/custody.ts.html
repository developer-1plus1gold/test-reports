/**
 * Vault custody creation — turns a `shipper='vault'` order into allocated,
 * serial-tracked custody records (one `vault_holdings` row per physical bar).
 *
 * Lifecycle (mirrors how a normal order reserves stock, then DOESN'T dispatch):
 *   1. The order is created with shipper='vault' (no delivery address, fee 0).
 *   2. On order/payment, `consumeStockForOrderItem` FIFO-reserves inventory rows
 *      for each gold line and links them via `order_fulfilments` — EXACTLY as a
 *      shipped order. (Unlike Gold Yield, vault does NOT skip the consume: we
 *      want a specific serialized bar allocated to the customer.)
 *   3. THEN this helper reads back those reserved rows and creates one
 *      `vault_holdings` row per inventory row — created `awaiting_storage` so it
 *      shows in "My Vault" right away (badged "Securing"). It flips to `stored`
 *      via markVaultHoldingsStoredForOrder when admin marks the order stored
 *      after packing.
 *
 * Idempotent: upserts on the UNIQUE `inventory_id` (one holding per physical
 * bar), so a webhook retry / double-fire can't create duplicates. Best-effort:
 * never throws — a custody-row failure must not unwind a committed payment;
 * admin can reconcile. Charge stays server-authoritative elsewhere; this only
 * records custody.
 */

/**

 * unit, allocating any shortfall now via FIFO.
 *
 * Why this exists: a PRESALE vault order reserves nothing at order time —
 * `consumeStockForOrderItem` finds no `available` inventory, so no
 * `order_fulfilments` link is created, so `createVaultHoldingsForOrder` makes
 * no holding, so the bar is never marked sold and COGS never flows (revenue is
 * still recognised at `paid`, overstating gross profit). By the time admin marks
 * the order *stored*, the physical bar IS in hand, so this is the correct moment
 * to allocate the specific serialized row.

 * use `reserveInventoryForOrderItem` (single FIFO pick) and DON'T fall back to
 * the `products.stock_qty` decrement that `consumeStockForOrderItem` allows.
 * If no row is available the shortfall is simply left unfilled — mark-stored
 * then stays blocked (count 0), which is the correct "no bar yet" signal.
 *

/**
 * NON-THROWING BY CONTRACT (2026-07-19 post-payment audit).
 *
 * Called from all three charge paths immediately before customer-facing work
 * (marking the payment done, claiming promo/referral, sending the receipt). It
 * previously contained no error handling whatsoever — every read was unchecked
 * and any exception escaped, which would have stranded a paying customer with
 * no receipt over a vault-holding failure.
 *
 * A missing holding is recoverable: `mark-vault-stored` ensure-creates holdings
 * that weren't created at order time. A missing receipt is not recoverable in
 * the same quiet way. So this fails loudly to the log and returns 0.
 * See feedback_backend_ops_never_degrade_customer.
 */
export async function createVaultHoldingsForOrder(
  client: Client,
  orderId: string,
  status: VaultHoldingCreateStatus = "awaiting_storage",

      `holdings are missing and "My Vault" will not show them until mark-vault-stored ` +
      `ensure-creates them. Order + payment are unaffected:`,
      err instanceof Error ? err.message : err,
    );
    return 0;
  }
}

/**

 * packing. Two steps so it's robust to inventory timing:
 *   1. Ensure-create any bars that weren't holdinged at order time (e.g. stock
 *      was received between order and packing) directly as `stored`.
 *   2. Flip every remaining `awaiting_storage` holding for the order to `stored`
 *      and stamp `stored_at` to the actual storage time.
 *
 * Returns the number of `stored` holdings the order now has. The caller treats
 * 0 as "no bar allocated" and refuses to advance the order.
 */

  //    (admin is marking it stored), so FIFO-allocate the shortfall. No-op for
  //    in-stock orders that already reserved at order time.

  await createVaultHoldingsForOrder(client, orderId, "stored");

    .from("vault_holdings")
    .update({ status: "stored", stored_at: new Date().toISOString() })
    .eq("order_id", orderId)

    .from("vault_holdings")

/**
 * Cancellation teardown — remove a cancelled/refunded order's ACTIVE vault
 * holdings (awaiting_storage / stored). Terminal historical records
 * (withdrawn / liquidated) are preserved. The underlying inventory reservation
 * is released separately by releaseInventoryForOrder. Best-effort: logs and
 * returns 0 on error, never throws. Returns the number of holdings removed.
 */
export async function voidVaultHoldingsForOrder(client: Client, orderId: string): Promise<number> {
  const { data, error } = await client
    .from("vault_holdings")
    .delete()
    .eq("order_id", orderId)
    .not("status", "in", "(withdrawn,liquidated)")
    .select("id");
  if (error) {
    console.error("[vault/custody] voidVaultHoldingsForOrder failed:", error.message, "order", orderId);
    return 0;
  }
  return (data ?? []).length;
}
