/**
 * Dispatch → "Ready to hand over": has this order's SHIPPING LABEL been printed?
 * (Mukesh 2026-11-27: "once it moves to ready to handover it gets trickier to
 * identify which order needs the shipping label … if a label is printed then it
 * becomes reprint shipping label".)
 *
 * The board splits the Ready stage into "Needs shipping label" and "Labelled —
 * ready to hand over", and every label button reads "Reprint …" once printed.
 *
 * A print is recorded WITH the label it printed (`labelKey`: the courier, plus the
 * tracking number for a courier whose label carries one). When the order is
 * rebooked or switched to another courier the key no longer matches, so the old
 * print stops counting and the order is back under "Needs shipping label" — the
 * paper on the parcel is wrong — without every booking / switch / cancel path
 * having to remember to clear it.
 *
 * Every courier prints a label, our own 4x6 included (Lalamove, Grab,
 * self-collect: Mukesh "we do print the shipping (our own) label for these orders
 * also"). Only gold kept in our vault has no parcel.
 */
 
import { statusRank } from "@/lib/shipping/spx-public-track";
 
export type LabelOrder = {
  shipper?: string | null;
  shipper_status?: string | null;
  shipper_order_id?: string | null;
  source_channel?: string | null;
  shipping_label_printed_at?: string | null;
  shipping_label_printed_key?: string | null;
};
 
/** No parcel leaves: nothing to label. */
const NO_LABEL_SHIPPERS = new Set(["vault"]);
 
/**
 * Couriers whose label is OUR 4x6 sticker with no carrier tracking number on it
 * that matters: a driver takes the parcel straight to the customer (Lalamove
 * lanes, Grab) or the customer collects it. Rebooking a driver does not make the
 * sticker wrong, so the key is the courier alone.
 */
const OWN_STICKER_SHIPPERS = new Set(["lalamove", "afternoon", "evening", "grab", "self_pickup"]);
 
const norm = (s: string | null | undefined) => String(s ?? "").trim().toLowerCase();
 
/** Whether this order gets a shipping label at all. */
export function needsLabel(o: LabelOrder): boolean {
  return !NO_LABEL_SHIPPERS.has(norm(o.shipper));
}
 
/**
 * The identity of the label a print produced. A carrier label (SPX AWB, NinjaVan
 * waybill, SingPost, a marketplace's label) is tied to its tracking number; our
 * own sticker only to the courier. A marketplace order is keyed by its channel
 * too, since its carrier is the marketplace's.
 */
export function labelKey(o: LabelOrder): string {
  const shipper = norm(o.shipper) || norm(o.source_channel);
  if (OWN_STICKER_SHIPPERS.has(shipper)) return shipper;
  return `${shipper}:${String(o.shipper_order_id ?? "").trim()}`;
}
 
/** Printed for the label the order carries NOW (a rebook / switch voids it). */
export function isLabelPrinted(o: LabelOrder): boolean {
  return !!o.shipping_label_printed_at && o.shipping_label_printed_key === labelKey(o);
}
 
/** The Ready stage's two parts. Vault orders count as labelled (nothing to print). */
export function readySubStage(o: LabelOrder): "needs_label" | "labelled" {
  return needsLabel(o) && !isLabelPrinted(o) ? "needs_label" : "labelled";
}
 
export const READY_SUB_META: Record<"needs_label" | "labelled", { title: string; hint: string }> = {
  needs_label: { title: "Needs shipping label",           hint: "Packed — open the order and print the label (step 3); SPX, NinjaVan, SingPost and marketplace labels come with the booking" },
  labelled:    { title: "Labelled — ready to hand over", hint: "Book the courier, drop off, or hand to the collector" },
};
 
/**
 * The carrier already has the parcel (picked up or later), so its label was
 * printed — even when the print predates the record (before 2026-11-27) or was
 * made outside Dispatch (Seller Center). Mukesh 2026-11-27: picked-up SPX tiles
 * still read "Print SPX label".
 */
export function isWithCarrier(o: LabelOrder): boolean {
  return statusRank(o.shipper_status) >= statusRank("picked_up");
}
 
/** Button text: "Print shipping label" → "Reprint shipping label" once printed or picked up. */
export function labelButtonText(o: LabelOrder, noun: string = norm(o.shipper) === "self_pickup" ? "collection label" : "shipping label"): string {
  return `${isLabelPrinted(o) || isWithCarrier(o) ? "Reprint" : "Print"} ${noun}`;
}
 
/** "Printed 14:32" (today) or "Printed 26 Nov 14:32", Singapore time. */
export function printedStamp(o: LabelOrder, nowMs: number = Date.now()): string | null {
  if (!isLabelPrinted(o)) return null;
  const at = new Date(o.shipping_label_printed_at!);
  const tz = { timeZone: "Asia/Singapore" } as const;
  const time = at.toLocaleTimeString("en-GB", { ...tz, hour: "2-digit", minute: "2-digit" });
  const day = (d: Date) => d.toLocaleDateString("en-GB", { ...tz, day: "numeric", month: "short" });
  return day(at) === day(new Date(nowMs)) ? `Printed ${time}` : `Printed ${day(at)} ${time}`;
}
