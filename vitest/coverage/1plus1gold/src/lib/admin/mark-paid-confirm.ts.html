/**
 * The confirmation an operator sees before marking a customer order paid.
 *
 * ▶ WHY THIS IS A SHARED MODULE AND NOT A STRING AT EACH BUTTON (2026-09-18).
 * `PATCH /api/admin/orders/[id]/payment` is reachable from FOUR buttons across
 * TWO screens — Admin → Payments (the COD list, desktop + mobile, and the
 * External Marketplace Payouts list) and Books → Cash & Receivables (desktop +
 * mobile). Books warned that a receipt goes out; Payments said nothing at all.
 *
 * That asymmetry cost a real customer message: on 2026-09-18 a COD order from
 * 1 September was settled in the books and the customer received a "Payment
 * received with thanks" WhatsApp AND email seventeen days after the fact, at
 * midnight his time. Nothing was broken — the route has always sent the paid
 * receipt, and its own comment says so — but the button that fired it read as
 * pure bookkeeping. Mukesh: *"I see it now...that I just marked this customer as
 * paid in the books..and that has triggered the payment receipt."*
 *
 * A comment cannot keep two copies of a warning in step; one function can. The
 * same failure shape as the footer's three trust signals (2026-09-10) and the
 * six copies of the shipping price rule (2026-09-15) — a rule written out N
 * times is a rule that will be updated N-1 times.
 *
 * ▶ THE MESSAGE NAMES THE CUSTOMER, NOT JUST THE ORDER. "Mark A0F60040 as paid?"
 * is a question about a row. "Rajesh Arora will be sent a receipt" is a question
 * about a person, which is the decision actually being made.
 *
 * ▶ IT IS DELIBERATELY NOT CHANNEL-CONDITIONAL. The client does not know whether
 * this customer has a phone, a real email, or (for a house/channel booking
 * account) neither — `getChannels` resolves that server-side at send time. A
 * confirm that promised "WhatsApp and email" and then sent one, or none, would
 * be worse than one that states the intent. So the COD wording states both
 * channels (true for every real customer) and the external wording is hedged,
 * because a marketplace payout is usually booked against a house account that
 * reaches nobody.
 */
 
/**
 * `S$1,341.03` — formatted here rather than by the caller's `useFmt()` on
 * purpose. That hook renders `"Loading…"` for a non-SGD display currency until
 * FX lands (2026-04-28 decision: never fabricate a rate), and a money
 * confirmation that reads "Mark A0F60040 (Loading…) as paid?" is worse than no
 * number. The books figure is SGD, so the confirm shows SGD.
 *
 * Hand-grouped rather than `toLocaleString` — Node's ICU build has already
 * produced a test/runtime split in this repo once (en-GB "Sept" vs "Sep",
 * 2026-07-30), and there is no reason to take that risk for a thousands
 * separator.
 */
export function formatSgd(n: number): string {
  const safe = Number.isFinite(n) ? n : 0;
  const [whole, cents] = Math.abs(safe).toFixed(2).split(".");
  const grouped = whole.replace(/\B(?=(\d{3})+(?!\d))/g, ",");
  return `${safe < 0 ? "-" : ""}S$${grouped}.${cents}`;
}
 
export type MarkPaidKind = "cod" | "external";
 
export interface MarkPaidConfirmInput {
  /** Display reference the operator sees — e.g. `A0F60040`. */
  ref: string;
  /** The amount shown on THIS screen, in SGD. For an external payout that is
   *  the NET expected cash, not the gross — the same number the row displays. */
  amountSgd: number;
  /** Customer name or phone, when we have one. */
  customerLabel?: string | null;
  kind: MarkPaidKind;
}
 
export interface MarkPaidConfirm {
  title: string;
  /** One paragraph per entry — rendered as separate lines by the dialog. */
  lines: string[];
  confirmLabel: string;
}
 
/**
 * Structured rather than a single string so the branded `useAdminDialog()` can
 * render real paragraphs on both screens. `markPaidConfirmText` joins it for a
 * plain-text context (and for tests, which assert the sentences rather than the
 * markup).
 */
export function markPaidConfirm({
  ref, amountSgd, customerLabel, kind,
}: MarkPaidConfirmInput): MarkPaidConfirm {
  const amount = formatSgd(amountSgd);
  const who = (customerLabel ?? "").trim() || "The customer";
 
  if (kind === "external") {
    return {
      title: `Mark payout received for ${ref}?`,
      lines: [
        `Records ${amount} as received from the marketplace.`,
        "If the order belongs to a customer account we can reach, they will also be " +
        "sent a payment-confirmation receipt immediately.",
        "This cannot be undone here.",
      ],
      confirmLabel: "Mark payout received",
    };
  }
 
  return {
    title: `Mark ${ref} as paid?`,
    lines: [
      `Records ${amount} as received.`,
      `${who} will be sent a payment-confirmation receipt by WhatsApp and email, immediately.`,
      "This cannot be undone here.",
    ],
    confirmLabel: "Mark as paid",
  };
}
 
/** The same confirmation as plain text. */
export function markPaidConfirmText(input: MarkPaidConfirmInput): string {
  const c = markPaidConfirm(input);
  return [c.title, "", ...c.lines].join("\n");
}
