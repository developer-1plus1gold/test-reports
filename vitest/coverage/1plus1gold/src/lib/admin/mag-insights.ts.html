/**
 * Tiles for Admin → MAG (Mukesh 2026-11-24, "just like Customer and COD").
 *
 * Pure: GET /api/admin/mag/insights loads the rows, this aggregates. Grams, counts and money — no
 * names.
 *
 *   In custody — customer grams on our bars: allocations `allocated` + `sell_requested` (a
 *                sell-back in progress is still ours to hold). Same rule as Vault Custody.
 *   Bars       — listings `active`: their total grams, grams sold to customers, and grams still
 *                available to buy (mag_listing_availability.grams_remaining).
 *   Bought     — grams a customer paid for, dated by `allocated_at` (a sell-back SPLITS a row but
 *                keeps the original allocation date, so sold grams still count where they were
 *                bought). Checkout holds (`held`) and abandoned ones (`released`) are not sales.
 *   Sold back  — `sold_back` rows, dated by `closed_at` (the payout).
 *   Value      — grams in custody × today's spot (SGD/g), the same basis as Vault Custody's
 *                sell-back value before premiums; null when spot is unavailable.
 * Internal / supplier accounts are left out of the customer figures (nonSalesCustomerIds).
 */
import { lastSgtMonths, sgtDateOf, sgtToday } from "@/lib/finance/sgt-window";
import { nonSalesCustomerIds, type SalesExclusions } from "@/lib/sales/sales-exclusions";
 
export const MAG_CHART_MONTHS = 6;
 
export interface MagInsightAllocation {
  customer_id: string;
  listing_id: string;
  grams: number;
  status: string;
  allocated_at: string | null;
  closed_at: string | null;
  updated_at: string;
}
 
export interface MagInsightBar {
  listing_id: string;
  serial: string | null;
  status: string;
  grams_total: number;
  grams_remaining: number | null;
}
 
export interface MagInsights {
  custody: { grams: number; customers: number; valueSgd: number | null; avgPerCustomer: number | null };
  bars: { active: number; gramsTotal: number; gramsSold: number; gramsAvailable: number };
  boughtThisMonth: { grams: number; customers: number };
  boughtLastMonth: { grams: number; customers: number };
  soldBackThisMonth: number;
  soldBackLastMonth: number;
  /** Grams in custody per bar, largest first. */
  byBar: Array<{ serial: string; grams: number }>;
  byMonth: Array<{ month: string; grams: number; soldBack: number }>;
}
 
const r4 = (n: number) => Math.round(n * 10000) / 10000 + 0;
const SOLD = new Set(["allocated", "sell_requested", "sold_back", "redeemed", "delivered"]);
const HELD = new Set(["allocated", "sell_requested"]);
 
export function computeMagInsights(
  allocations: MagInsightAllocation[],
  bars: MagInsightBar[],
  exclusions: SalesExclusions,
  spotSgdPerG: number | null,
  now: Date,
): MagInsights {
  const months = lastSgtMonths(sgtToday(now), MAG_CHART_MONTHS);
  const thisKey = months[months.length - 1].key;
  const lastKey = months[months.length - 2].key;
  const monthOf = (iso: string) => sgtDateOf(iso).slice(0, 7);
  const nonSales = nonSalesCustomerIds(exclusions);
  const rows = allocations.filter((a) => !nonSales.has(a.customer_id));
 
  const held = rows.filter((a) => HELD.has(a.status));
  const heldGrams = held.reduce((s, a) => s + a.grams, 0);
  const holders = new Set(held.map((a) => a.customer_id));
 
  const bought = (key: string) => {
    const b = rows.filter((a) => SOLD.has(a.status) && a.allocated_at && monthOf(a.allocated_at) === key);
    return { grams: r4(b.reduce((s, a) => s + a.grams, 0)), customers: new Set(b.map((a) => a.customer_id)).size };
  };
  const soldBack = (key: string) => r4(rows
    .filter((a) => a.status === "sold_back" && monthOf(a.closed_at ?? a.updated_at) === key)
    .reduce((s, a) => s + a.grams, 0));
 
  const active = bars.filter((b) => b.status === "active");
  const serialOf = new Map(bars.map((b) => [b.listing_id, b.serial ?? b.listing_id.slice(0, 8)]));
  const perBar = new Map<string, number>();
  for (const a of held) perBar.set(a.listing_id, (perBar.get(a.listing_id) ?? 0) + a.grams);
  const allHeldOnActive = allocations.filter((a) => HELD.has(a.status) && active.some((b) => b.listing_id === a.listing_id));
 
  return {
    custody: {
      grams: r4(heldGrams),
      customers: holders.size,
      valueSgd: spotSgdPerG != null ? Math.round(heldGrams * spotSgdPerG * 100) / 100 : null,
      avgPerCustomer: holders.size ? Math.round((heldGrams / holders.size) * 10) / 10 : null,
    },
    bars: {
      active: active.length,
      gramsTotal: r4(active.reduce((s, b) => s + Number(b.grams_total), 0)),
      gramsSold: r4(allHeldOnActive.reduce((s, a) => s + a.grams, 0)),
      gramsAvailable: r4(active.reduce((s, b) => s + Number(b.grams_remaining ?? 0), 0)),
    },
    boughtThisMonth: bought(thisKey),
    boughtLastMonth: bought(lastKey),
    soldBackThisMonth: soldBack(thisKey),
    soldBackLastMonth: soldBack(lastKey),
    byBar: [...perBar.entries()]
      .map(([id, grams]) => ({ serial: serialOf.get(id) ?? id.slice(0, 8), grams: r4(grams) }))
      .sort((a, b) => b.grams - a.grams),
    byMonth: months.map((m) => ({ month: m.key, grams: bought(m.key).grams, soldBack: soldBack(m.key) })),
  };
}
