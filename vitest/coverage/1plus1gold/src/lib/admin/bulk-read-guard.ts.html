/**
 * Bulk-read guard for admin routes that return whole datasets of personal data
 * (the customer list, the compliance-case list). PDPA review 2026-09-07.
 *
 * G8 made these reads AUDITED; this adds the DEFENCE on top — a proportionate
 * ladder keyed on the acting admin session:
 *
 *   ≤ THROTTLE_PER_HOUR      → allowed
 *   > THROTTLE_PER_HOUR      → 429 throttle (a cooldown; no logout, no page)
 *   > BREACH_PER_HOUR        → 429 + escalate (Control Tower + WhatsApp) +
 *                              FORCE RE-2FA (revoke all admin sessions)
 *
 * The two tiers are independent rate-limit buckets, so a mild overage only
 * throttles — the heavy response (log everyone out) is reserved for sustained
 * abuse, i.e. an actual mass-export pattern.
 *
 * SCOPE / honest limit: this defends a misused or compromised admin SESSION.
 * It does NOT stop someone holding the Supabase service key or DB password —
 * those bypass the app (and RLS) entirely and are a secret-management concern.
 */
 
import { NextResponse } from "next/server";
import { checkRateLimit } from "@/lib/rate-limit";
import { adminSessionId } from "@/lib/admin/audit";
import { revokeAdminSessionsBefore } from "@/lib/admin/session-revocation";
import { sendAdminAlert } from "@/lib/twilio";
import { openSuspectedIncident } from "@/lib/governance/breach-incident";
import { getAdminPhonesCsv } from "@/lib/admin-team";
 
// Tunable. A two-person admin team opening a list a handful of times an hour is
// nowhere near these; 40+ full-dataset pulls/hour is not a human browsing.
export const THROTTLE_PER_HOUR = 40;
export const BREACH_PER_HOUR = 80;
const HOUR_MS = 60 * 60 * 1000;
 
export type BulkReadAction = "allow" | "throttle" | "breach";
 
/** Pure decision from the two bucket results. Breach outranks throttle. */
export function decideBulkReadAction(r: { throttleLimited: boolean; breachLimited: boolean }): BulkReadAction {
  if (r.breachLimited) return "breach";
  if (r.throttleLimited) return "throttle";
  return "allow";
}
 
// eslint-disable-next-line @typescript-eslint/no-explicit-any
type SupabaseClient = any;
 
const AGENT = "bulk_read_guard";
 
/**
 * Guard a bulk-PII read. Returns a 429 NextResponse to return to the caller
 * when throttled/breached, or null to proceed. On a breach it also escalates
 * and forces re-2FA (deduped so it pages/revokes at most once per session-hour).
 */
export async function guardBulkRead(
  req: Request,
  sb: SupabaseClient,
  opts: { resourceType: string },
): Promise<NextResponse | null> {
  const session = adminSessionId(req);
  const base = `admin-bulkread:${opts.resourceType}:${session}`;
 
  // Increment BOTH buckets every call so the breach counter keeps climbing even
  // while the throttle bucket is already returning 429.
  const [throttle, breach] = await Promise.all([
    checkRateLimit(`${base}:throttle`, THROTTLE_PER_HOUR, HOUR_MS),
    checkRateLimit(`${base}:breach`, BREACH_PER_HOUR, HOUR_MS),
  ]);
 
  const action = decideBulkReadAction({ throttleLimited: throttle.limited, breachLimited: breach.limited });
  if (action === "allow") return null;
 
  if (action === "breach") {

    const hourBucket = Math.floor(Date.now() / HOUR_MS);
    const alertKey = `${opts.resourceType}:${session}:${hourBucket}`;

        `if not, treat as a potential data-exfiltration attempt.`;

      // only reached it if somebody later remembered to type it in — and the
      // PDPA clock would then run from when they typed, not from when we

          `All admin sessions were signed out automatically (re-login + WhatsApp OTP required) and ` +
          `further bulk reads were refused with HTTP 429. Acting admin session (hashed): ${session}.`,
        severity: "high",
        dataCategories:
          opts.resourceType === "compliance_cases"

        assessmentNotes:
          "AUTO-OPENED by the bulk-read guard — not yet assessed by a human. Confirm whether this was " +
          "a legitimate operator export or an unauthorised session. If legitimate, record that here and " +
          "close as not_notifiable. If not, determine the volume and categories actually retrieved from " +
          "the admin_access_log audit rows for this session before deciding notifiability.",
        remediation:
          "Automatic, already applied: all admin sessions revoked (forced re-authentication with WhatsApp " +
          "OTP) and the bulk-read endpoints rate-limited to HTTP 429 for the remainder of the hour.",
        affectedCount: null,
        detectedBy: "bulk-read-guard",
      });
 
      // Force re-2FA: revoke every session issued before now.

}
