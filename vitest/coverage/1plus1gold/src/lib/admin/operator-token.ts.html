/**
 * The operator key (Mukesh 2026-11-25): lets Claude make production data changes through the
 * SAME admin API routes the admin screens call (products, POs, receipts, dispatch, orders,
 * suppliers …), so validation, triggers and side effects run exactly as when a person uses
 * the screens — instead of hand-written SQL or a data-migration PR.
 *
 *   • Its own secret, `OPS_AGENT_TOKEN` (never ADMIN_SECRET), sent as the `x-ops-token` header.
 *     The value lives only in Vercel's env and in this cloud environment's network
 *     credentials (the proxy attaches it); nobody pastes it into chat or code.
 *   • Full admin reach for now (Mukesh: "this operator should be able to work like an admin").
 *   • Every call it makes is written to `admin_operator_log` (method, path, query, IP), so its
 *     changes can be told apart from a person's and reviewed.
 *   • Revoke by rotating or removing OPS_AGENT_TOKEN in Vercel — nothing else depends on it.
 *   • Off unless configured: no env (or one shorter than MIN_TOKEN_LENGTH) accepts nothing.
 */
import { timingSafeEqual } from "crypto";
 
export const OPERATOR_HEADER = "x-ops-token";
 
/** A token shorter than this is treated as not configured (guards a typo'd short value). */
export const MIN_TOKEN_LENGTH = 32;
 
/** Pure: does the presented header match the configured operator token? Constant-time. */
export function operatorTokenValid(presented: string | null | undefined, configured: string | undefined): boolean {
  const want = (configured ?? "").trim();
  const got = (presented ?? "").trim();
  if (want.length < MIN_TOKEN_LENGTH || got.length === 0) return false;
  const a = Buffer.from(got);
  const b = Buffer.from(want);
  return a.length === b.length && timingSafeEqual(a, b);
}
 
/** Pure: the row written for one operator call. Query strings are kept (they carry ids and
 *  filters, never secrets — the token travels in a header). */
export function operatorLogRow(req: Request, ip: string): { method: string; path: string; query: string | null; ip: string } {
  const url = new URL(req.url);
  return { method: req.method, path: url.pathname.slice(0, 300), query: url.search ? url.search.slice(0, 500) : null, ip: ip.slice(0, 64) };
}
