/**
 * Post-take-over SHELL account detection — pure helpers for Admin → Customers.
 *
 * WHY THIS EXISTS. When a WhatsApp-OTP customer later signs up with
 * Google/Apple and re-links the same phone, `link-phone`'s HIGH-C2 take-over
 * flow correctly transfers `profiles.phone` to the real account and NULLs it on
 * the original synthetic one. That leftover synthetic account then renders in
 * Admin → Customers as a completely BLANK row: no phone (transferred away) and
 * no email (its synthetic `phone_<digits>@auth.1plus1gold.com` address is
 * deliberately nulled for display). The behaviour is correct; the row just
 * looks like a data bug, and it cost real investigation time on 2026-08-21.
 *
 * ▶ This is a LABELLING fix, deliberately not an identity fix. Do NOT wire
 * `resolveCustomerIdentity()` into the admin surface — `/api/admin/users`
 * already reads `profiles.phone` and already nulls synthetic emails on
 * purpose. The only useful change is to say what the blank row IS.
 *
 * Pure + unit-tested — no DB, no network.
 */
 
/** Matches the synthetic address minted for WhatsApp-OTP signups. */
const SYNTHETIC_EMAIL_RE = /^phone_(\d+)@auth\.1plus1gold\.com$/i;
 
/** Digits of the phone encoded in a synthetic email, or null. */
export function phoneDigitsFromSyntheticEmail(email: string | null | undefined): string | null {
  if (!email) return null;
  const m = SYNTHETIC_EMAIL_RE.exec(email.trim());
  return m ? m[1] : null;
}
 
/** Compare phone numbers by digits only — `profiles.phone` is E.164 ("+65…")
 *  while the synthetic email carries bare digits ("65…"). */
function digitsOnly(phone: string | null | undefined): string {
  return (phone ?? "").replace(/\D/g, "");
}
 
export interface ShellProfileInput {
  id:    string;
  phone: string | null;
  /** The RAW auth email (before synthetic addresses are nulled for display). */
  rawEmail: string | null;
  /** Display name, used to label the account the phone moved to. */
  fullName: string | null;
}
 
/**
 * Is this row a post-take-over shell? Synthetic OTP email AND no phone left.
 *
 * Exported so the Admin → Customers labelling and the dashboard's
 * "New Customers Today" count share ONE definition and cannot drift apart.
 */
export function isShellAccount(row: { phone: string | null; rawEmail: string | null }): boolean {
  return !!phoneDigitsFromSyntheticEmail(row.rawEmail) && !digitsOnly(row.phone);
}
 
export interface ShellInfo {
  /** True when this is a synthetic OTP account whose phone has been taken over. */
  isPhoneShell: boolean;
  /** The account that now holds the phone, when we can identify it. */
  movedToId:   string | null;
  movedToName: string | null;
  /** The phone that moved away, E.164-formatted for display. */
  movedPhone:  string | null;
}
 
/**
 * Identify post-take-over shells and, where possible, where each phone went.
 *
 * A row is a shell when BOTH hold:
 *  - its auth email is synthetic (so it was created by WhatsApp OTP), and
 *  - `profiles.phone` is now NULL (so the phone was transferred away).
 *
 * A synthetic account that still holds its phone is a perfectly normal OTP
 * customer and is NOT flagged — 17 of 19 such accounts display fine.
 *
 * The destination is resolved by matching the digits encoded in the synthetic
 * email against the phone another profile now holds. `profiles.phone` carries a
 * UNIQUE index (`profiles_phone_unique`), so at most one account can match.
 * When no holder is found the row is still flagged a shell — we just cannot say
 * where the phone went, which is honest rather than silent.
 */
export function resolveShellAccounts(profiles: ShellProfileInput[]): Map<string, ShellInfo> {
  // digits → the profile that currently holds that phone.
  const holderByDigits = new Map<string, ShellProfileInput>();
  for (const p of profiles) {
    const d = digitsOnly(p.phone);
    if (d) holderByDigits.set(d, p);
  }
 
  const out = new Map<string, ShellInfo>();
  for (const p of profiles) {
    const digits = phoneDigitsFromSyntheticEmail(p.rawEmail);
    if (!isShellAccount(p)) continue;
 
    const holder = holderByDigits.get(digits!);
    // Guard against a row somehow matching itself.
    const moved = holder && holder.id !== p.id ? holder : null;
    out.set(p.id, {
      isPhoneShell: true,
      movedToId:   moved?.id ?? null,
      movedToName: moved?.fullName ?? null,
      movedPhone:  `+${digits}`,
    });
  }
  return out;
}
