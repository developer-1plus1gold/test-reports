//
// payment-fees.ts — the SINGLE source for the customer-facing payment-processing
// fee math (pct + flat + alt-pct-below-threshold + min-fee).
//
// Consumed by the PWA `PaymentMethodSelector` AND bundled into the on-device
// engine (`engine.paymentFee`) so the iOS `payment_method_picker` shows a
// byte-identical fee — no Swift re-implementation of money math (the cart-core
// pattern). The CHARGE stays server-authoritative regardless. Contract §25.
//
 
export interface PaymentFeeMethod {
  customer_fee_pct:            number;
  customer_flat_fee_sgd?:      number;
  customer_fee_pct_alt?:       number;
  provider_fee_threshold_sgd?: number;
  provider_min_fee_sgd?:       number;
}
 
/** Computed SGD processing fee for an order total. Rounds to cents. */
export function feeAmountSgd(method: PaymentFeeMethod, orderTotal: number): number {
  if (orderTotal <= 0) return 0;
  const thresh  = method.provider_fee_threshold_sgd ?? 0;
  const flatFee = method.customer_flat_fee_sgd ?? 0;
  const altPct  = method.customer_fee_pct_alt ?? 0;
  const minFee  = method.provider_min_fee_sgd ?? 0;
 
  const pct  = thresh > 0 && orderTotal <= thresh ? altPct : method.customer_fee_pct;
  const flat = thresh > 0 && orderTotal <= thresh ? 0 : flatFee;
  const effectiveMin = thresh > 0 && orderTotal <= thresh ? minFee : 0;
 
  let amount = orderTotal * pct / 100 + flat;
  if (effectiveMin > 0) amount = Math.max(amount, effectiveMin);
  return Math.round(amount * 100) / 100;
}
