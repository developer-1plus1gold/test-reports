/**
 * Consignment PO notifications (Step 6).
 *
 * Two audiences on a new consignment PO:
 *   • Admins — always, via the Meta-approved ADMIN_ALERT template (sendAdminAlert).
 *   • The supplier — AUTOMATICALLY when they have a WhatsApp number on file AND a
 *     supplier-PO template is configured; otherwise the operator forwards the
 *     message manually from the consignment queue (the pre-existing fallback).
 *
 * An automatic, business-initiated WhatsApp to a supplier is outside the 24-hour
 * session window, so Meta requires a PRE-APPROVED template. We can't approve one
 * from code — so the send is gated on TWILIO_SUPPLIER_PO_TEMPLATE_SID. Until that
 * env var points at an approved template, `sendSupplierConsignmentPo` reports
 * `sent:false, reason:'no_template'` and the flow falls back to manual. When the
 * template is live, set the env var and auto-send switches on with no code change.
 */
 
import { sendWhatsAppTemplate } from "@/lib/twilio";
 
export interface SupplierPoMessage {
  supplierWhatsapp: string | null | undefined;
  supplierName: string;
  poNumber: string;
  productLabel: string;
  quantity: number;
  gramsEach: number;
}
 
export type SupplierNotifyReason = "sent" | "no_number" | "no_template" | "error";
 
export interface SupplierNotifyResult {
  sent: boolean;
  reason: SupplierNotifyReason;
  sid?: string | null;
  detail?: string;
}
 
/**
 * Attempt to auto-notify the supplier of a consignment PO. Never throws — a
 * notification failure must not break the PO sync; the caller falls back to the
 * manual route and the admin alert flags it.
 */
/** E.164: leading + and 8–15 digits. Same shape the manage route enforces. */
const E164 = /^\+\d{8,15}$/;
 
export async function sendSupplierConsignmentPo(msg: SupplierPoMessage): Promise<SupplierNotifyResult> {
  const number = (msg.supplierWhatsapp ?? "").trim();
  if (!number) return { sent: false, reason: "no_number" };
  // Defence-in-depth: the Suppliers screen validates E.164 on write, but never
  // hand an unvalidated number to Twilio — a malformed value that reached the
  // row by any other path is treated as "no number on file" (manual fallback),
  // not blindly sent.
  if (!E164.test(number)) return { sent: false, reason: "no_number", detail: "supplier WhatsApp not E.164" };
 
  const templateSid = process.env.TWILIO_SUPPLIER_PO_TEMPLATE_SID?.trim();
  if (!templateSid) return { sent: false, reason: "no_template" };
 
  // Template variables (structured for Meta approval — don't stuff everything
  // into one var). {1}=supplier, {2}=item summary, {3}=PO number.
  const item = `${msg.quantity}× ${msg.productLabel} (${msg.gramsEach}g each)`;
  try {
    const res = await sendWhatsAppTemplate(number, templateSid, {
      "1": msg.supplierName,
      "2": item,
      "3": msg.poNumber,
    });

    return { sent: true, reason: "sent", sid: res.sid };
  } catch (err) {
    return { sent: false, reason: "error", detail: err instanceof Error ? err.message : String(err) };
  }
}
 
/** Human-readable one-liner for the admin alert, explaining the supplier leg. */
export function supplierNotifySummary(r: SupplierNotifyResult): string {
  switch (r.reason) {
    case "sent":        return "supplier auto-notified on WhatsApp";
    case "no_number":   return "no supplier WhatsApp on file — forward the message manually from the consignment queue";
    case "no_template": return "supplier WhatsApp template not configured — forward the message manually from the consignment queue";
    case "error":       return `supplier auto-notify failed (${r.detail ?? "unknown"}) — forward manually from the consignment queue`;
  }
}
