/**
 * What procure-on-sale costs us versus simply buying the stock upfront.
 *
 * Consignment is convenience with a price: no capital tied up, no shipment to
 * pay for, no bars sitting in the vault — but the supplier charges more per
 * piece than their wholesale sheet. This module puts a number on that gap so the
 * premium is a decision rather than an assumption.
 *
 * TWO COMPARISONS, BECAUSE THEY MEAN DIFFERENT THINGS
 *
 *   vs SHEET PRICE    the supplier's own asking price for the piece, before any
 *                     shipping or FX. A like-for-like "what would they charge me
 *                     for this same bar" — the cleanest read on the consignment
 *                     mark-up itself.
 *
 *   vs LANDED COST    what buying upfront ACTUALLY costs delivered: sheet price
 *                     + shipping + wire + the bank's FX spread. This is the
 *                     honest business comparison, because consignment avoids all
 *                     of those too — we collect the piece ourselves.
 *
 * Reporting only the first would overstate the consignment penalty (it ignores
 * the costs consignment saves); reporting only the second hides how much of the
 * gap is the supplier's mark-up versus logistics we would have paid anyway.
 *
 * ⚠ THE ORDER-SIZE ASSUMPTION IS LOAD-BEARING
 *   Landed cost spreads a PER-SHIPMENT flat fee (shipping + wire, often US$250+)
 *   across the grams ordered. Costing a single 1g bar as its own shipment lands
 *   the entire flat fee on one gram and produces an absurd upfront cost — which
 *   would make consignment look brilliant for exactly the wrong reason. Callers
 *   MUST pass a realistic `assumedOrderGrams`, and the figure must be shown
 *   next to the result so the reader can judge it.
 */
 
import {
  computeLandedCost,
  spotUsdPerGram,
  type SpotContext,
  type ShippingConfig,
  type SupplierLineInput,
} from "@/lib/pricing/landed-cost";
import { computeConsignmentCost, type ConsignmentTerms } from "./cost";
import { roundSgd } from "@/lib/order-pricing";
 
export interface UpfrontComparisonInput extends ConsignmentTerms {
  /** SGD per gram — the same spot the consignment cost is quoted at. */
  spotPerGSgd: number;
  gramsEach: number;
  /** The supplier's sheet line for this product. Null when it is not on a sheet. */
  supplierLine: SupplierLineInput | null;
  spot: SpotContext;
  shipping: ShippingConfig;
  /**
   * Total grams a realistic upfront order would carry. The per-shipment flat
   * fees are spread across this, NOT across one piece. Required — there is no
   * safe default, because the wrong one silently flatters consignment.
   */
  assumedOrderGrams: number;
}
 
export interface UpfrontComparison {
  /** What consignment costs us for one piece. */
  consignmentCostSgd: number;
  /** Supplier's sheet ask for the piece, before shipping/FX. Null if not on a sheet. */
  sheetItemCostSgd: number | null;
  /** Sheet price + shipping + wire + FX spread, at the assumed order size. */
  landedPerPieceSgd: number | null;
  /** The logistics slice consignment avoids entirely. */
  shippingPlusFxPerPieceSgd: number | null;
  /** consignment − sheet ask. The supplier's mark-up for holding the metal. */
  excessVsSheetSgd: number | null;
  /** consignment − landed. What procure-on-sale really costs us, net. */
  excessVsLandedSgd: number | null;
  /** excessVsLanded as a percentage of landed. Null when landed is 0/unknown. */
  excessVsLandedPct: number | null;
  /** Echoed back so the assumption travels with the number. */
  assumedOrderGrams: number;
 
  // ── Like-for-like PREMIUM figures (what to compare the premium field to) ────
  /**
   * The sheet ask expressed as a per-piece premium: sheet cost − the metal value
   * inside it. For a `per_piece` line this IS the supplier's dollar ask, in SGD.
   */
  sheetPremiumPerPieceSgd: number | null;
  /**
   * The same figure once the piece is actually delivered — landed cost − metal
   * value. Deliberately includes the FX markup on the metal itself, because that
   * is money genuinely paid above spot to own the bar.
   */
  landedPremiumPerPieceSgd: number | null;
  /**
   * What OUR consignment terms charge over spot for one piece:
   * `premium + spread × grams`. The premium field alone is NOT comparable to the
   * two figures above — the per-gram spread is part of our mark-up too, and
   * omitting it flatters consignment by more the heavier the piece.
   */
  consignmentPremiumPerPieceSgd: number;
}
 
export function compareConsignmentToUpfront(
  input: UpfrontComparisonInput,
): UpfrontComparison {
  const { unitCostSgd: consignmentCostSgd } = computeConsignmentCost({
    spotPerGSgd:   input.spotPerGSgd,
    spreadPerGSgd: input.spreadPerGSgd,
    premiumSgd:    input.premiumSgd,
    gramsEach:     input.gramsEach,
    quantity:      1,
  });
 
  // Our own over-spot charge — available even with no baseline at all, because
  // it is derived purely from the terms being edited.
  const consignmentPremiumPerPieceSgd = roundSgd(
    input.premiumSgd + input.spreadPerGSgd * input.gramsEach,
  );
 
  const empty: UpfrontComparison = {
    consignmentCostSgd,
    sheetItemCostSgd: null,
    landedPerPieceSgd: null,
    shippingPlusFxPerPieceSgd: null,
    excessVsSheetSgd: null,
    excessVsLandedSgd: null,
    excessVsLandedPct: null,
    assumedOrderGrams: input.assumedOrderGrams,
    sheetPremiumPerPieceSgd: null,
    landedPremiumPerPieceSgd: null,
    consignmentPremiumPerPieceSgd,
  };
 
  // No sheet line (or no usable ask) → report the consignment cost alone rather
  // than inventing a comparison. A fabricated baseline is worse than none.
  if (!input.supplierLine) return empty;
 
  const landed = computeLandedCost(
    input.supplierLine,
    input.spot,
    input.shipping,
    input.assumedOrderGrams,
  );
  if (!landed) return empty;
 
  const sheetItemCostSgd  = roundSgd(landed.itemCostSgd);
  const landedPerPieceSgd = roundSgd(landed.landedPerPieceSgd);
  const excessVsLandedSgd = roundSgd(consignmentCostSgd - landedPerPieceSgd);
 
  // Metal value the SHEET figures are built on — derived from the same
  // spotUsdPerOz × usdToSgd basis `computeLandedCost` used, NOT from the
  // caller's `spotPerGSgd`. The two spot representations can differ in the last
  // cents, and here the whole point is a subtraction: any mismatch would show up
  // as phantom premium rather than cancelling out.
  const metalRefSgd = spotUsdPerGram(input.spot.spotUsdPerOz) * input.gramsEach * input.spot.usdToSgd;
 
  return {
    consignmentCostSgd,
    sheetItemCostSgd,
    landedPerPieceSgd,
    shippingPlusFxPerPieceSgd: roundSgd(landed.shippingPlusFxPerPieceSgd),
    excessVsSheetSgd:  roundSgd(consignmentCostSgd - sheetItemCostSgd),
    excessVsLandedSgd,
    excessVsLandedPct: landedPerPieceSgd === 0
      ? null
      : roundSgd((excessVsLandedSgd / landedPerPieceSgd) * 100),
    assumedOrderGrams: input.assumedOrderGrams,
    sheetPremiumPerPieceSgd:  roundSgd(landed.itemCostSgd       - metalRefSgd),
    landedPremiumPerPieceSgd: roundSgd(landed.landedPerPieceSgd - metalRefSgd),
    consignmentPremiumPerPieceSgd,
  };
}
