/**
 * Which supplier-sheet line to price a consignment product against.
 *
 * Setting a consignment premium is a judgement — "is S$60 a fair per-piece
 * mark-up for a 10g bar?" — and the only honest yardstick is what the SAME piece
 * costs on a wholesale sheet. That yardstick is rarely a clean lookup:
 *
 *   • The consignment supplier may publish NO sheet at all. BullionKing is
 *     exactly this (`fetch_mode = 'none'` — consignment-only), so a
 *     supplier-scoped lookup finds nothing for every product it consigns. The
 *     comparison has to reach the supplier that DOES publish (MTB).
 *   • The exact SKU may be on the sheet but carry no ask (`ask_kind = 'none'` —
 *     the supplier is out of stock, so the line exists with a blank price). On
 *     the 2026-07-23 MTB sheet that is true of BOTH the 20g PAMP Fortuna and the
 *     100g LBMA — two of our seven consigned products.
 *
 * So the resolution ladder, per Mukesh: exact product first, and when that is
 * unavailable, **the cheapest line at the same weight**. A same-weight line is a
 * real, buyable alternative at that gram size, which is the decision actually in
 * front of the operator ("could I stock this weight cheaper myself?"). Cheapest
 * rather than closest-by-name because it sets the HARDEST bar for the premium —
 * a premium that beats the cheapest alternative beats all of them.
 *
 * Every result carries WHERE it came from. A comparison figure whose provenance
 * is invisible gets read as like-for-like when it is not, and a premium gets set
 * against the wrong bar.
 */
 
import { itemCostUsd, type SpotContext, type SupplierLineInput } from "@/lib/pricing/landed-cost";
 
/** How the baseline was found — always shown next to the number it produced. */
export type BaselineMatch = "exact" | "same_weight";
 
export interface SheetCandidate {
  supplierId: string;
  supplierName: string;
  /** Normalised `productKey(product_name, weight_label)` — the same identity the
   *  pricing-strategy module links on. Raw names never match; see the route. */
  productKey: string;
  /** Human name, for provenance. */
  productName: string;
  line: SupplierLineInput;
}
 
export interface SheetBaseline {
  supplierId: string;
  supplierName: string;
  productName: string;
  match: BaselineMatch;
  line: SupplierLineInput;
}
 
/**
 * Same weight, allowing for stored-numeric noise. 1 oz is 31.1035 g on both
 * sides, so the tolerance only has to absorb rounding, never bridge sizes — a
 * loose one would let a 20g line stand in for a 25g product.
 */
function sameWeight(a: number, b: number): boolean {

  return Math.abs(a - b) <= Math.max(0.01, b * 0.001);
}
 
/**
 * Pick the sheet line to compare a consignment product against.
 *
 * Returns null when nothing is priceable — the caller must then show no
 * comparison at all. An invented baseline is worse than an absent one: it looks
 * like evidence.
 */
export function resolveSheetBaseline(args: {
  /** `products.supplier_product_key`. Null when the product was never linked. */
  productKey: string | null;
  /** Grams in one piece of OUR product. */
  grams: number;
  /** The consignment supplier — its own sheet wins an exact match if it has one. */
  preferSupplierId?: string | null;
  candidates: SheetCandidate[];
  spot: SpotContext;
}): SheetBaseline | null {
  const { productKey, grams, preferSupplierId, candidates, spot } = args;
 
  // A line with no usable ask cannot be a baseline. This is not an edge case —
  // it is the normal "supplier is out of this SKU" state, and it is what makes
  // the same-weight fallback load-bearing rather than defensive.
  const priced = candidates
    .map((c) => ({ c, costUsd: itemCostUsd(c.line, spot) }))
    .filter((x): x is { c: SheetCandidate; costUsd: number } => x.costUsd != null && x.costUsd > 0);
 
  if (priced.length === 0) return null;
 
  const cheapest = (xs: typeof priced) =>
    xs.reduce((best, x) => (x.costUsd < best.costUsd ? x : best));
 
  const asBaseline = (x: (typeof priced)[number], match: BaselineMatch): SheetBaseline => ({
    supplierId:   x.c.supplierId,
    supplierName: x.c.supplierName,
    productName:  x.c.productName,
    match,
    line:         x.c.line,
  });
 
  if (productKey) {
    const exact = priced.filter((x) => x.c.productKey === productKey);
    if (exact.length > 0) {
      // Our own supplier's sheet first when it has one; otherwise cheapest, so a
      // product listed by two suppliers resolves the same way every load.
      const own = preferSupplierId ? exact.filter((x) => x.c.supplierId === preferSupplierId) : [];
      return asBaseline(cheapest(own.length > 0 ? own : exact), "exact");
    }
  }
 
  const sameGrams = priced.filter((x) => sameWeight(x.c.line.grams, grams));
  if (sameGrams.length === 0) return null;
  return asBaseline(cheapest(sameGrams), "same_weight");
}
