/**
 * Deterministic transaction monitoring (Path B) — the pure kernel.
 *
 * Extracted from /api/orders/create so the structuring/velocity/CTR rules are
 * unit-testable. See docs/compliance-triggers-redesign.md for the calibration
 * rationale (why structuring is shape-based, not a raw order count).
 *
 * Rules (windows in days):
 *   R1  same-day cumulative ≥ S$20,000                         → flag (medium)
 *        (suppressed when it merely restates a single cash CTR order — see below)
 *   R2  ≥3 orders each < S$20,000 aggregating ≥ S$20,000 / 30d → STRUCTURING (high)
 *   R3  30-day cumulative ≥ S$100,000                          → flag (high)
 *   R5  ≥5 orders each ≥ S$5,000 / 30d                         → VELOCITY (high)
 *   CTR cash (cod/in_person) current order ≥ S$20,000          → mandatory CTR
 *
 * Path B NEVER drafts an STR (deterministic monitoring; STR authorship is a
 * human/officer or Path A sanctions-path judgement). Max severity is "high".
 */
 
export const CTR_THRESHOLD_SGD          = 20_000; // MAS PSPM cash reporting threshold
export const STRUCTURING_MIN_PIECES     = 3;      // deliberate splitting needs ≥3 pieces
export const VELOCITY_MIN_ORDERS        = 5;      // "5 instances" benchmark…
export const VELOCITY_MIN_ORDER_SGD     = 5_000;  // …anchored to material orders
export const VELOCITY_CUMULATIVE_100K   = 100_000;
export const CASH_METHODS = ["cod", "in_person"] as const;
 
export interface MonitorOrder { total_sgd: number | string; created_at: string; }
 
export interface MonitorResult {
  flags:     string[];
  riskLevel: "medium" | "high";
  ctr:       { applies: boolean; sameDayTotalSgd: number };
  raw:       Record<string, unknown>;
}
 
function sgtMidnightMs(now: Date): number {
  const sgtOffset = 8 * 3600 * 1000;
  const sgt = new Date(now.getTime() + sgtOffset);
  return Date.UTC(sgt.getUTCFullYear(), sgt.getUTCMonth(), sgt.getUTCDate()) - sgtOffset;
}
 
/**
 * @param priorOrders  the customer's orders in the last ~90d (NOT incl. the current one)
 * @param currentOrderSgd  the order being placed now
 */
export function evaluateTransactionMonitoring(params: {
  priorOrders:     MonitorOrder[];
  currentOrderSgd: number;
  paymentMethod:   string;
  now?:            Date;
}): MonitorResult {
  const { priorOrders, currentOrderSgd, paymentMethod } = params;
  const now = params.now ?? new Date();
  const isCash = (CASH_METHODS as readonly string[]).includes(paymentMethod);
 
  const since30dMs      = now.getTime() - 30 * 24 * 60 * 60 * 1000;
  const todayStartMs    = sgtMidnightMs(now);
  const amt = (o: MonitorOrder) => Number(o.total_sgd);
  const ms  = (o: MonitorOrder) => new Date(o.created_at).getTime();
 
  const flags: string[] = [];
 
  // ── CTR — mandatory cash threshold report ──────────────────────────────────
  const ctrApplies = isCash && currentOrderSgd >= CTR_THRESHOLD_SGD;
 
  // ── R1: same-day cumulative ≥ threshold ────────────────────────────────────
  const todayPrior = priorOrders.filter((o) => ms(o) >= todayStartMs).reduce((s, o) => s + amt(o), 0);
  const todayTotal = todayPrior + currentOrderSgd;
  // Suppress the pure duplicate: a single cash CTR order (no same-day priors)
  // is already fully captured by the CTR case — don't also raise a monitoring
  // flag that restates the same fact. Genuine same-day accumulation (priors>0)
  // still flags for either payment type.
  if (todayTotal >= CTR_THRESHOLD_SGD && (todayPrior > 0 || !ctrApplies)) {
    flags.push(`Same-day cumulative spend S$${todayTotal.toFixed(2)} meets S$${CTR_THRESHOLD_SGD.toLocaleString()} reference threshold`);
  }
 
  // ── R2: threshold-evasion structuring ──────────────────────────────────────
  const orders30d = priorOrders.filter((o) => ms(o) >= since30dMs);
  const subThresholdPieces = [
    ...orders30d.filter((o) => amt(o) < CTR_THRESHOLD_SGD).map(amt),
    ...(currentOrderSgd < CTR_THRESHOLD_SGD ? [currentOrderSgd] : []),
  ];
  const subThresholdSum = subThresholdPieces.reduce((s, v) => s + v, 0);
  const isStructuring = subThresholdPieces.length >= STRUCTURING_MIN_PIECES && subThresholdSum >= CTR_THRESHOLD_SGD;
  if (isStructuring) {
    flags.push(`Possible structuring: ${subThresholdPieces.length} sub-S$${CTR_THRESHOLD_SGD.toLocaleString()} orders aggregating S$${subThresholdSum.toFixed(2)} in 30 days`);
  }
 
  // ── R3: 30-day cumulative ≥ 100k ───────────────────────────────────────────
  const prior30dTotal = orders30d.reduce((s, o) => s + amt(o), 0);
  const total30d = prior30dTotal + currentOrderSgd;
  const is100k = total30d >= VELOCITY_CUMULATIVE_100K;
  if (is100k) flags.push(`30-day cumulative spend S$${total30d.toFixed(2)}`);
 
  // ── R5: sustained velocity (≥5 material orders / 30d) ──────────────────────
  const materialCount =
    orders30d.filter((o) => amt(o) >= VELOCITY_MIN_ORDER_SGD).length +
    (currentOrderSgd >= VELOCITY_MIN_ORDER_SGD ? 1 : 0);
  const isVelocity = materialCount >= VELOCITY_MIN_ORDERS;
  if (isVelocity) flags.push(`${materialCount} orders ≥ S$${VELOCITY_MIN_ORDER_SGD.toLocaleString()} in 30 days — sustained high velocity`);
 
  const riskLevel: "medium" | "high" =
    (is100k || isStructuring || isVelocity) ? "high" : "medium";
 
  return {
    flags,
    riskLevel,
    ctr: { applies: ctrApplies, sameDayTotalSgd: todayTotal },
    raw: {
      checkType:        "transaction_monitoring",
      paymentMethod,
      currentOrderSgd,
      todayTotalSgd:    todayTotal,
      cumulative30dSgd: total30d,
      structuringPieces: subThresholdPieces.length,
      structuringSumSgd: subThresholdSum,
      materialOrderCount: materialCount,
      isStructuring,
      isVelocity,
      is100k,
      ctrApplies,
    },
  };
}
