/**
 * AML screening status interpretation for the Compliance Agent (Path A).
 *
 * Classifies a provider's AML screening result into ONE OF THREE real
 * states — replacing the old single `flagged` boolean that conflated "genuine
 * sanctions/PEP hit", "screening pending", and "screening not available" into
 * one value (which is what manufactured the 2026-08-09 false-positive critical;
 * see docs/compliance-triggers-redesign.md).
 *
 *   cleared        — GREEN: sanctions + PEP + adverse-media clear.
 *   sanctions_hit  — RED carrying a FATAL label (SANCTIONS/PEP/WATCH_LIST/…) —
 *                    a genuine AML hit.
 *   unavailable    — no applicant / fetch failed / review pending, OR a RED with
 *                    only recoverable (document-quality) labels. This is the
 *                    ABSENCE of a clear screen, NOT a hit.
 */
 
 
export type AmlStatus = "cleared" | "sanctions_hit" | "unavailable";
 
export interface AmlInfo {
  /** Prose consumed verbatim by the agent runtime context + STR draft. */
  text:   string;
  status: AmlStatus;
  /** true only for a genuine sanctions/PEP hit (status === "sanctions_hit"). */
  sanctionsHit: boolean;
  /** true when there is NO clear screen (hit OR unavailable) — i.e. not cleared. */
  notCleared: boolean;
}
 
function mk(status: AmlStatus, text: string): AmlInfo {
  return {
    text,
    status,
    sanctionsHit: status === "sanctions_hit",
    notCleared:   status !== "cleared",
  };
}
 
// ── Didit AML screening ───────────────────────────────────────────────────────
 
/**
 * Dataset families in a Didit AML hit that constitute a genuine screening hit.
 *
 * Matched as a lowercase SUBSTRING because Didit qualifies its labels —
 * "PEP Level 2", "Warnings and Regulatory Enforcement" — and an exact-match set
 * would silently miss every graded variant. Mirrors the categories already in
 * KYC_FATAL_REJECT_LABELS so the two providers assert the same thing.
 */
const DIDIT_RISK_DATASETS = [
  "sanction",
  "pep",
  "adverse media",
  "warning",
  "regulatory",
  "fitness and probity",
  "watch",
  "criminal",
  "terror",
  "crime",
] as const;
 
/** Didit marks a hit a reviewer has dismissed with exactly this review_status. */
const DIDIT_DISMISSED_REVIEW_STATUS = "false positive";
 
function hitIsDismissed(hit: DiditAmlHitLike): boolean {
  return (hit.review_status ?? "").trim().toLowerCase() === DIDIT_DISMISSED_REVIEW_STATUS;
}
 
/**
 * Whether one live hit should be treated as a screening hit.
 *
 * Fails CLOSED on anything unrecognised: a hit carrying no datasets, or only
 * datasets outside the list above, still counts. An AML dataset we have never
 * seen is not evidence of innocence, and a lookup that returns "not in my list"
 * and is then used as a clear signal is the same defect shape as a non-total
 * switch — it passes every test whose fixture happens to avoid the unknown
 * input. The cost of failing closed is an operator reviewing a case that turns
 * out to be benign; the cost of failing open is releasing a sanctions match.
 */
function hitIsRisk(hit: DiditAmlHitLike): boolean {
  const datasets = (hit.datasets ?? []).map((d) => String(d).toLowerCase());
  if (datasets.length === 0) return true;
  return datasets.some(
    (d) => DIDIT_RISK_DATASETS.some((risk) => d.includes(risk)) || !isKnownBenignDataset(d),
  );
}
 
/**
 * There is currently no dataset Didit returns that we consider benign — every
 * family they document (PEP, Sanctions, Adverse Media, Warnings and Regulatory
 * Enforcement, Fitness and Probity) is a risk signal. This exists so that if a
 * genuinely non-risk dataset is added later it is allow-listed HERE, explicitly,
 * rather than by loosening `hitIsRisk` into failing open.
 */
function isKnownBenignDataset(_dataset: string): boolean {
  return false;
}
 
interface DiditAmlHitLike {
  datasets?: string[];
  review_status?: string;
  caption?: string;
  match_score?: number;
  risk_score?: number;
}
 
interface DiditAmlScreeningLike {
  status?: string;
  total_hits?: number;
  hits?: DiditAmlHitLike[];
  score?: number | null;
}
 
/**
 * Classify Didit's `aml_screenings[]` into the same three states the former
 * interpreter produces, so every downstream consumer — the compliance agent
 * prompt, the STR draft, the S$20k escalation rule — is unchanged by the
 * provider swap.
 *
 * A session with NO screening at all is `unavailable`, never `cleared`. That is
 * the case that fires if the configured workflow omits the AML Screening step
 * (Didit's "Free KYC" workflow does), and it must read as "we did not screen",
 * because at or above S$20k an unavailable screen is what forces the CDD
 * escalation. Silently treating an absent screen as a pass would remove the
 * control while every test stayed green.
 *
 * Multiple screenings can appear (one per workflow node). The worst wins.
 */
export function interpretDiditAmlStatus(
  screenings: DiditAmlScreeningLike[] | null | undefined,
): AmlInfo {
  if (!screenings || screenings.length === 0) {
    return mk(
      "unavailable",
      "Status not available — the Didit session carries no AML screening (workflow may omit the AML Screening step, or the session did not complete)",
    );
  }
 
  const liveHits: DiditAmlHitLike[] = [];
  const dismissed: DiditAmlHitLike[] = [];
  let anyPending = false;
  let anyComputed = false;
 
  for (const s of screenings) {
    const status = (s.status ?? "").trim().toLowerCase();
    // `score` is null until screening has actually run; In Review / In Progress
    // mean a result exists but is not final.
    if (s.score === null || s.score === undefined) anyPending = true;
    else anyComputed = true;
    if (status === "in review" || status === "in progress" || status === "not started") {
      anyPending = true;
    }
    for (const hit of s.hits ?? []) {
      (hitIsDismissed(hit) ? dismissed : liveHits).push(hit);
    }
  }
 
  const riskHits = liveHits.filter(hitIsRisk);
  if (riskHits.length > 0) {
    const detail = riskHits
      .slice(0, 5)
      .map((h) => {
        const sets = (h.datasets ?? []).join("/") || "unclassified dataset";
        const score = typeof h.risk_score === "number" ? ` risk ${h.risk_score}` : "";
        return `${h.caption ?? "unnamed"} [${sets}]${score}`;
      })
      .join("; ");
    const more = riskHits.length > 5 ? ` (+${riskHits.length - 5} more)` : "";
    return mk(
      "sanctions_hit",
      `FAILED — Didit AML screening returned ${riskHits.length} live hit(s): ${detail}${more}`,
    );
  }
 
  if (anyPending && !anyComputed) {
    return mk(
      "unavailable",
      "Pending — Didit AML screening has not produced a result yet; sanctions/PEP/adverse-media screening not completed",
    );
  }
  if (anyPending) {
    return mk(
      "unavailable",
      "Not cleared — at least one Didit AML screening node is still pending; treat as not-screened until it completes",
    );
  }
 
  const dismissedNote = dismissed.length
    ? ` (${dismissed.length} hit(s) dismissed as false positives by a reviewer)`
    : "";
  return mk(
    "cleared",
    `PASSED — Didit AML screening cleared: no live sanctions, PEP or adverse-media hits${dismissedNote}`,
  );
}
