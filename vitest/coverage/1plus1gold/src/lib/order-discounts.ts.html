/**
 * Single source for applying promo + referral discounts to an order at charge
 * time. Shared verbatim by every charge path — orders/create (COD),
 * payments/hitpay/create, payments/stripe/create-intent — so the three can never
 * drift on how a discount is computed, capped, apportioned, or which one wins.
 *
 * Two rules live here and nowhere else:
 *   1. Percent promos use `computePromoDiscountSgd` on the whole merchandise
 *      subtotal (gold + accessories), bounded by the mandatory cap.
 *   2. NO STACKING — a promo and a referral discount are mutually exclusive on a
 *      single order; we keep whichever saves the customer more and drop the other
 *      (the dropped code is neither charged for nor claimed/recorded).
 *
 * See [[feedback_pricing_single_source]] and [[feedback-money-route-charge-integrity]].
 */
import { computePromoDiscountSgd, resolveExclusiveDiscount, type PromoType } from "@/lib/promo-discount";
import { applyDiscountsSequentially, type DiscountApportionment } from "@/lib/discounts";
import { roundSgd } from "@/lib/order-pricing";
 
/** The minimal promo fields the resolver needs — read straight from promo_codes. */
export interface PromoMeta {
  type: PromoType | string | null;
  value_sgd?: number | null;
  percent?: number | null;
  max_discount_sgd?: number | null;
  /** Optional IRAS scope; null = pro-rata across IPM + taxable. */
  scope?: "ipm_only" | "taxable_only" | null;
}
 
export interface ResolveOrderDiscountsArgs {
  /** promo_codes row (already fetched + min_order validated), or null if none entered. */
  promo: PromoMeta | null;
  /** Referral discount already computed by computeRefereeDiscountSgd, or 0. */
  referralDiscountSgd: number;
  /** Gold subtotal (IPM-exempt). */
  goldSubtotalSgd: number;
  /** Accessory subtotal (taxable). */
  accessorySubtotalSgd: number;
  /** Shipping fee before any free-shipping promo. */
  shippingFeeSgd: number;
}
 
export interface ResolvedOrderDiscounts {
  /** Amount taken off the merchandise subtotal by the promo (0 if dropped/free-shipping). */
  promoDiscountSgd: number;
  /** Effective promo type after no-stacking (null if the promo was dropped). */
  promoType: PromoType | string | null;
  /** True only when a surviving free_shipping promo waived the fee. */
  freeShipping: boolean;
  /** Shipping fee after a surviving free_shipping promo. */
  effectiveShippingFeeSgd: number;
  /** Referral discount after no-stacking (0 if dropped). */
  referralDiscountSgd: number;
  /** Whether the caller should still claim/record the promo code. */
  promoKept: boolean;
  /** Whether the caller should still claim/record the referral code. */
  referralKept: boolean;
  /** IRAS apportionment of the promo discount across IPM/taxable. */
  promoApportioned: DiscountApportionment;
  /** IRAS apportionment of the referral discount across IPM/taxable. */
  referralApportioned: DiscountApportionment;
  /** roundSgd(max(0, gold + acc + effShip − promo − referral)). */
  grandTotalSgd: number;
}
 
/**
 * The promo's comparable "benefit" for the no-stacking contest. A free_shipping
 * promo's benefit is the fee it would waive; value/percent it's the discount.
 */
function promoBenefitSgd(discountSgd: number, freeShipping: boolean, shippingFeeSgd: number): number {
  return freeShipping ? Math.max(0, shippingFeeSgd) : discountSgd;
}
 
export function resolveOrderDiscounts(args: ResolveOrderDiscountsArgs): ResolvedOrderDiscounts {
  const { promo, goldSubtotalSgd, accessorySubtotalSgd, shippingFeeSgd } = args;
  const merchandiseSubtotalSgd = Math.max(0, goldSubtotalSgd) + Math.max(0, accessorySubtotalSgd);
  const referralIn = Math.max(0, args.referralDiscountSgd);
 
  // ── Raw promo discount (before no-stacking) ────────────────────────────────
  let promoDiscountSgd = 0;
  let freeShipping = false;
  let promoType: PromoType | string | null = null;
  if (promo && promo.type) {
    promoType = promo.type;
    const d = computePromoDiscountSgd(
      {
        type: promo.type as PromoType,
        value_sgd: promo.value_sgd,
        percent: promo.percent,
        max_discount_sgd: promo.max_discount_sgd,
      },
      merchandiseSubtotalSgd,
    );
    promoDiscountSgd = d.discountSgd;
    freeShipping = d.freeShipping;
  }
 
  // ── No stacking — keep the larger benefit, drop the other ──────────────────
  let referralDiscountSgd = referralIn;
  let promoKept = promoType != null;
  let referralKept = referralIn > 0;
 
  if (promoKept && referralKept) {
    const keep = resolveExclusiveDiscount(
      promoBenefitSgd(promoDiscountSgd, freeShipping, shippingFeeSgd),
      referralDiscountSgd,
    );
    if (keep === "promo") {
      referralDiscountSgd = 0;
      referralKept = false;
    } else {
      // "referral" (or a degenerate "none" where both are 0) — drop the promo.
      promoDiscountSgd = 0;
      freeShipping = false;
      promoType = null;
      promoKept = false;
    }
  }
 
  const effectiveShippingFeeSgd = freeShipping ? 0 : shippingFeeSgd;
 
  // ── IRAS apportionment — promo first, then referral (loser contributes 0) ──
  const { apportioned } = applyDiscountsSequentially(
    { ipm: Math.max(0, goldSubtotalSgd), taxable: Math.max(0, accessorySubtotalSgd) },
    [
      { amount: promoDiscountSgd, scope: promo?.scope ?? null },
      { amount: referralDiscountSgd, scope: null },
    ],
  );
 
  const grandTotalSgd = roundSgd(
    Math.max(0, goldSubtotalSgd + accessorySubtotalSgd + effectiveShippingFeeSgd - promoDiscountSgd - referralDiscountSgd),
  );
 
  return {
    promoDiscountSgd,
    promoType,
    freeShipping,
    effectiveShippingFeeSgd,
    referralDiscountSgd,
    promoKept,
    referralKept,
    promoApportioned: apportioned[0],
    referralApportioned: apportioned[1],
    grandTotalSgd,
  };
}
