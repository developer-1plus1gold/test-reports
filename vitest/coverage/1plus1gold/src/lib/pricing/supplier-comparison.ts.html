/**
 * Product-centric supplier comparison — Step 3 of the Supplier & Pricing hub
 * (docs/supplier-pricing-hub-design.md §3). Answers the question the operator
 * actually has: "for each product I sell, what does EVERY supplier cost me, and
 * where should I buy it?"
 *
 * The spine is OUR sellable catalogue (shop-visible gold products). For each
 * product we walk every ACTIVE supplier and resolve its landed cost through the
 * one resolver (`resolveLandedCost`) — so the number here can never drift from
 * the strategy / consignment pages. The two models resolve differently but
 * land on the same per-gram figure:
 *   - sheet   → the supplier's own sheet line for this product (exact SKU, else
 *     cheapest same-weight — `resolveSheetBaseline`), landed.
 *   - manual  → the in-force per-product term ($ per piece or % over spot).
 * A supplier that can't price a product (no matching sheet line, no manual
 * term) is null — an absent comparison, never an invented one.
 *
 * Against each supplier's cost we set OUR retail & wholesale price (premium over
 * spot) and compute the margin, then flag the cheapest source per product.
 *
 * PURE of side effects beyond the reads it's handed the `db` for; the money math
 * is entirely delegated to the kernels. Testable by mocking `db`, spot & premiums.
 */

/** Margin selling at `sell` having paid `cost`, as a % of the sell price. Null
 *  when either side is missing or the sell price is non-positive. */
function marginPct(sell: number | null, cost: number | null): number | null {
  if (sell == null || cost == null || !(sell > 0)) return null;
  return ((sell - cost) / sell) * 100;
}

/** Fetch PAID 1g-card sales and aggregate the realized premium per refinery. Two

async function resolveSplit1gPremiums(db: SupabaseClient): Promise<Record<string, Split1gPremium>> {
  const { data: curve1g } = await db
    .from("products")
    .select("id, refinery")
    .eq("product_type", "curve")
    .eq("grams", 1);
  const refById = new Map<string, string | null>();
  for (const p of (curve1g ?? []) as Array<{ id: string; refinery: string | null }>) refById.set(p.id, p.refinery);
  if (refById.size === 0) return {};
 
  const { data: items } = await db
    .from("order_items")
    .select("premium_applied_pct, product_id, orders!inner(payment_status, status)")
    .in("product_id", [...refById.keys()])
    .eq("orders.payment_status", "paid")
    .not("premium_applied_pct", "is", null);
 
  // eslint-disable-next-line @typescript-eslint/no-explicit-any
  const rows = ((items ?? []) as any[]).map((r) => {
    const ord = Array.isArray(r.orders) ? r.orders[0] : r.orders;
    return {
      refinery: refById.get(r.product_id) ?? null,
      premiumPct: r.premium_applied_pct != null ? Number(r.premium_applied_pct) : null,
      orderStatus: ord?.status ?? null,

  });
