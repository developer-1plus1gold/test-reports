/**
 * THE cost of a bar (Mukesh 2026-11-14, "one cost everywhere"): the cheapest
 * price that RESTOCKS this bar across every active supplier — the exact sheet
 * line, the same refiner's bar, or a manual / consignment term — named by its
 * supplier. The 2 % card fee is out (landed-resolver has no fee input).
 *
 * Every margin and pricing comparison reads this one number: the Pricing
 * Strategy page, the margin monitor, the competitor heat map, MAG Pool offers,
 * the daily pricing review and both simulators (via the supplier comparison's
 * `bestLandedPerGramSgd`). `cost-definition-guard.test.ts` keeps it that way.
 *
 * An "any brand, same weight" sheet line is NEVER a restock — on 11-14 it priced
 * a 1 g Nadir off a China Gold Panda coin. The supplier comparison still shows
 * such cells, as a reference.
 *
 * What we PAID for stock we already hold is a separate, display-only figure
 * (lib/pricing/daily-review.ts `paidPremiumPct`).
 */
import type { SupplierCostCell } from "@/lib/pricing/supplier-comparison";
 
export interface RestockCost { supplierId: string; landedPerGramSgd: number }
 
/** The cheapest cell that restocks this bar, or null when no supplier can. */
export function cheapestRestock(cells: readonly Pick<SupplierCostCell, "supplierId" | "landedPerGramSgd" | "basis">[]): RestockCost | null {
  let best: RestockCost | null = null;
  for (const c of cells) {
    if (!isRestockBasis(c.basis) || c.landedPerGramSgd == null) continue;
    if (!best || c.landedPerGramSgd < best.landedPerGramSgd) best = { supplierId: c.supplierId, landedPerGramSgd: c.landedPerGramSgd };
  }
  return best;
}
 
/** True for a basis that prices THIS bar (or the same refiner's). */
export function isRestockBasis(basis: SupplierCostCell["basis"]): boolean {
  return basis !== "same_weight_sheet" && basis !== "none";
}
 
/** THE cost of each product, by product id, with its supplier's name. */
export type RestockCostMap = ReadonlyMap<string, { sgdPerG: number; supplier: string | null }>;
 
/**
 * Pure: the restock cost of every product the comparison priced — its `best`,
 * which is cheapestRestock above. The one map every reader is built from.
 */
export function restockCostsFromComparison(cmp: {
  suppliers: readonly { id: string; name: string }[];
  rows: readonly { productId: string; bestSupplierId: string | null; bestLandedPerGramSgd: number | null }[];
}): RestockCostMap {
  const names = new Map(cmp.suppliers.map(sp => [sp.id, sp.name]));
  const out = new Map<string, { sgdPerG: number; supplier: string | null }>();
  for (const r of cmp.rows) {
    if (r.bestLandedPerGramSgd != null && r.bestSupplierId) {
      out.set(r.productId, { sgdPerG: r.bestLandedPerGramSgd, supplier: names.get(r.bestSupplierId) ?? null });
    }
  }
  return out;
}
 
/**
 * A product whose COST is another product's (`products.cost_reference_product_id`,
 * Mukesh 2026-11-25): the 1 g split cards are cut from a CombiBar, so no sheet
 * restocks them; their cost is held to the 1 g Nadir's. Pure: every row with a
 * reference takes that row's best supplier + landed SGD/g. A reference to a
 * missing row, to a row with no restock, or of a different weight leaves the
 * row's own best alone — a per-gram cost only carries across the same grams.
 * One level only: a reference to a row that itself has a reference reads that
 * row's OWN best, so a cycle cannot loop.
 */
export function applyCostReferences<R extends {
  productId: string;
  grams: number;
  costReferenceProductId?: string | null;
  bestSupplierId: string | null;
  bestLandedPerGramSgd: number | null;
}>(rows: readonly R[]): R[] {
  const own = new Map(rows.map(r => [r.productId, r]));
  return rows.map(r => {
    const ref = r.costReferenceProductId ? own.get(r.costReferenceProductId) : undefined;
    if (!ref || ref.grams !== r.grams || ref.bestSupplierId == null || ref.bestLandedPerGramSgd == null) return r;
    return { ...r, bestSupplierId: ref.bestSupplierId, bestLandedPerGramSgd: ref.bestLandedPerGramSgd };
  });
}
