/**
 * Daily pricing review (Mukesh 2026-11-14, option 1) — the 11-14 manual review,
 * every day at 16:00 SGT, after the afternoon competitor fetch.
 *
 * Pure: the cron (`/api/cron/pricing-review`) gathers the data, this builds the
 * table and the flags, and the `pricing_reviewer` agent only writes words around
 * them. Every number — the rows, the checks and each flag's suggested target — is
 * computed HERE (agent contract §0.2, math is code); the model may only say
 * whether to act on a flag and why.
 *
 * One row per thing a customer can buy:
 *   curve     a Premium Curve bar on the shop, at the curve rate for ONE piece
 *             (a first purchase — the most a customer pays);
 *   standard  a standard bar on the shop, at its first tier;
 *   mag_open  a MAG open-pool bar type with an active listing, at its first tier;
 *   mag_offer an active own-bar / group-pool offer, at its flat %.
 * All percentages are over spot and leave the 2 % supplier card fee out (the
 * 11-14 comparison rule).
 *
 * COST (Mukesh 11-14, option 1) is the cheapest RESTOCK across every active
 * supplier — what replacing the bar costs today (MTB sheet landed, BullionKing
 * consignment, StoneX…), with the supplier named. It drives every flag. PAID is
 * the premium we actually paid for the bars we hold, against the spot on the day
 * we bought them — spot-neutral, so a bar bought before a price drop does not
 * read as a loss. Shown beside the cost; it drives nothing. A MAG bar is already
 * owned, so its cost IS its paid premium.
 *
 * Checks (thresholds approved by Mukesh 11-14):
 *   above_competitor     more than 1 point above the cheapest COMPARABLE
 *                        competitor for the same bar (fresh lists only);
 *   below_cost           priced below our cost;
 *   thin_margin          under 0.5 points over cost;
 *   offer_over_outright  a MAG offer more than 0.5 points above the same bar
 *                        bought outright on the shop;
 *   curve_rises          the curve's anchors go UP with weight somewhere.
 * Standing decisions are kept as ACCEPTED exceptions: still shown, never
 * suggested for change.
 */
 
export const COMPETITOR_GAP_PTS = 1.0;
export const THIN_MARGIN_PTS = 0.5;
export const OFFER_OVER_OUTRIGHT_PTS = 0.5;
/** A suggested target sits this far above the competitor or outright price it
 *  answers — "just above", the 11-14 rule. */
export const TARGET_ABOVE_PTS = 0.5;
 
export type ReviewPool = "curve" | "standard" | "mag_open" | "mag_offer";
export type FlagCode = "above_competitor" | "below_cost" | "thin_margin" | "offer_over_outright" | "curve_rises";
 
export interface ReviewRow {
  /** Stable within a day: `<pool>:<productId>` (`mag_offer:<offerId>` for offers). */
  key: string;
  pool: ReviewPool;
  productId: string;
  label: string;
  grams: number;
  ourPct: number;
  /** Cheapest restock % over spot (a MAG bar: the premium we paid for it). */
  costPct: number | null;
  /** Where that cost comes from: the cheapest supplier, or "what we paid". */
  costSource: string | null;
  /** Premium we paid for the units we hold (weighted), % over that day's spot. */
  paidPct: number | null;
  /** ourPct − costPct, points. */
  marginPts: number | null;
  competitor: { name: string; product: string | null; pct: number } | null;
  /** MAG offers only: the same bar bought outright on the shop. */
  outrightPct?: number | null;
  /** MAG offers only: "2 months". */
  duration?: string;
}
 
export interface ReviewFlag {
  /** `<code>:<rowKey>` — what a suggestion refers to. */
  id: string;
  code: FlagCode;
  rowKey: string | null;
  /** One plain line, computed (no model). */
  message: string;
  /** Where the change is made (a curve bar changes the curve anchor at its weight). */
  appliesTo: string;
  /** The suggested % over spot, computed; null when the fix is not a single price. */
  targetPct: number | null;
  /** A standing decision covers it: shown, never suggested. */
  accepted: string | null;
}
 
export interface ReviewInput {
  spotSgdPerG: number;
  /** The Premium Curve anchors, % over spot. */
  curveAnchors: ReadonlyArray<{ grams: number; pct: number }>;
  /** Shop-visible curve + standard bars with today's outright %. */
  shopBars: ReadonlyArray<{ productId: string; label: string; grams: number; pool: "curve" | "standard"; ourPct: number }>;
  /** MAG open-pool bar types (shop-visible, an active public listing). */
  magBars: ReadonlyArray<{ productId: string; label: string; grams: number; ourPct: number; paidPct: number | null }>;
  /** Active MAG pool offers. */
  magOffers: ReadonlyArray<{ offerId: string; productId: string; label: string; grams: number; duration: string; offerPct: number; outrightPct: number | null }>;
  /** Cheapest restock per gram (card fee out) and its supplier, by product id. */
  costByProduct: ReadonlyMap<string, { sgdPerG: number; source: string | null }>;
  /** Premium we paid for held stock, % over that day's spot, by product id. */
  paidPctByProduct: ReadonlyMap<string, number>;
  /** The cheapest comparable, fresh competitor price per gram, by product id. */
  competitorByProduct: ReadonlyMap<string, { name: string; product: string | null; sgdPerG: number }>;
}
 
export interface PricingReview {
  spotSgdPerG: number;
  rows: ReviewRow[];
  flags: ReviewFlag[];
  /** Flags that need a decision (not accepted). */
  openFlags: number;
}
 
/** Standing decisions (CLAUDE.md) the review must not re-raise. */
export interface AcceptedException {
  code: FlagCode;
  pool: ReviewPool;
  /** Match by weight (± 0.01 g) and, when set, a label substring. */
  grams: number;
  labelIncludes?: string;
  reason: string;
}
 
// The 1 g "small-bar pricing accepted" exceptions were retired 2026-11-15: the 1 g Nadir is now a
// standard bar priced at PM Marketplace's level, so a competitor undercutting it must surface.
export const ACCEPTED_EXCEPTIONS: readonly AcceptedException[] = [
  { code: "below_cost", pool: "mag_open", grams: 31.1035, labelIncludes: "Maple", reason: "1 oz Maple MAG stays at spot + 2.8 % (Mukesh 11-11)" },
  { code: "thin_margin", pool: "mag_open", grams: 31.1035, labelIncludes: "Maple", reason: "1 oz Maple MAG stays at spot + 2.8 % (Mukesh 11-11)" },
];
 
const r2 = (n: number) => Math.round(n * 100) / 100;
const pct = (n: number) => `${r2(n)}%`;
const sameGrams = (a: number, b: number) => Math.abs(a - b) < 0.01;
 
function acceptedFor(code: FlagCode, row: ReviewRow, list: readonly AcceptedException[]): string | null {
  const hit = list.find(e => e.code === code && e.pool === row.pool && sameGrams(e.grams, row.grams)
    && (!e.labelIncludes || row.label.toLowerCase().includes(e.labelIncludes.toLowerCase())));
  return hit?.reason ?? null;
}
 
function appliesTo(row: ReviewRow): string {
  switch (row.pool) {
    case "curve": return `Premium Curve anchor at ${r2(row.grams)} g (Admin → Premiums)`;
    case "standard": return `${row.label} tier 1 (Admin → Products)`;
    case "mag_open": return `MAG bar type ${row.label} (Admin → Products)`;
    case "mag_offer": return `MAG pool offer ${row.label}, ${row.duration ?? ""} (Admin → MAG → Pool offers)`;
  }
}
 
/** Build the day's review. Deterministic: same input, same output. */
export function buildPricingReview(input: ReviewInput, accepted: readonly AcceptedException[] = ACCEPTED_EXCEPTIONS): PricingReview {
  const spot = input.spotSgdPerG;
  const costPctOf = (sgdPerG: number | null | undefined) =>
    sgdPerG != null && Number.isFinite(sgdPerG) && spot > 0 ? r2((sgdPerG / spot - 1) * 100) : null;
 
  const paidOf = (id: string) => { const p = input.paidPctByProduct.get(id); return p == null ? null : r2(p); };
  const rows: ReviewRow[] = [];
  const withCost = (row: Omit<ReviewRow, "marginPts">): ReviewRow =>
    ({ ...row, marginPts: row.costPct != null ? r2(row.ourPct - row.costPct) : null });
 
  for (const b of input.shopBars) {
    const c = input.competitorByProduct.get(b.productId);
    const cost = input.costByProduct.get(b.productId);
    rows.push(withCost({
      key: `${b.pool}:${b.productId}`, pool: b.pool, productId: b.productId, label: b.label, grams: b.grams,
      ourPct: r2(b.ourPct), costPct: costPctOf(cost?.sgdPerG), costSource: cost ? cost.source : null,
      paidPct: paidOf(b.productId),
      competitor: c && spot > 0 ? { name: c.name, product: c.product, pct: r2((c.sgdPerG / spot - 1) * 100) } : null,
    }));
  }
  for (const m of input.magBars) {
    rows.push(withCost({
      key: `mag_open:${m.productId}`, pool: "mag_open", productId: m.productId, label: m.label, grams: m.grams,
      ourPct: r2(m.ourPct), costPct: m.paidPct == null ? null : r2(m.paidPct),
      costSource: m.paidPct == null ? null : "what we paid", paidPct: m.paidPct == null ? null : r2(m.paidPct), competitor: null,
    }));
  }
  for (const o of input.magOffers) {
    rows.push(withCost({
      key: `mag_offer:${o.offerId}`, pool: "mag_offer", productId: o.productId, label: o.label, grams: o.grams,
      ourPct: r2(o.offerPct), costPct: costPctOf(input.costByProduct.get(o.productId)?.sgdPerG),
      costSource: input.costByProduct.get(o.productId)?.source ?? null, paidPct: null, competitor: null,
      outrightPct: o.outrightPct == null ? null : r2(o.outrightPct), duration: o.duration,
    }));
  }
  const poolRank: Record<ReviewPool, number> = { curve: 0, standard: 1, mag_open: 2, mag_offer: 3 };
  rows.sort((a, b) => poolRank[a.pool] - poolRank[b.pool] || a.grams - b.grams || a.label.localeCompare(b.label));
 
  const flags: ReviewFlag[] = [];
  const add = (code: FlagCode, row: ReviewRow | null, message: string, targetPct: number | null, where: string) => {
    flags.push({
      id: `${code}:${row?.key ?? "curve"}`, code, rowKey: row?.key ?? null, message, appliesTo: where,
      targetPct: targetPct == null ? null : r2(targetPct),
      accepted: row ? acceptedFor(code, row, accepted) : null,
    });
  };
  // Never suggest a price under cost + the thin-margin line.
  const floorOf = (row: ReviewRow) => (row.costPct != null ? row.costPct + THIN_MARGIN_PTS : -Infinity);
 
  for (const row of rows) {
    if (row.competitor && row.ourPct - row.competitor.pct > COMPETITOR_GAP_PTS) {
      const target = Math.max(row.competitor.pct + TARGET_ABOVE_PTS, floorOf(row));
      add("above_competitor", row,
        `${row.label}: ${pct(row.ourPct)}, ${r2(row.ourPct - row.competitor.pct)} pts above ${row.competitor.name}${row.competitor.product ? ` (${row.competitor.product})` : ""} at ${pct(row.competitor.pct)}.`,
        target < row.ourPct ? target : null, appliesTo(row));
    }
    if (row.costPct != null && row.ourPct < row.costPct) {
      add("below_cost", row, `${row.label}: ${pct(row.ourPct)} is below our cost of ${pct(row.costPct)}${row.costSource ? ` (${row.costSource})` : ""}.`,
        row.costPct + THIN_MARGIN_PTS, appliesTo(row));
    } else if (row.marginPts != null && row.marginPts < THIN_MARGIN_PTS) {
      add("thin_margin", row, `${row.label}: only ${r2(row.marginPts)} pts over our cost (${pct(row.costPct!)}${row.costSource ? `, ${row.costSource}` : ""}).`,
        row.costPct! + THIN_MARGIN_PTS, appliesTo(row));
    }
    if (row.pool === "mag_offer" && row.outrightPct != null && row.ourPct - row.outrightPct > OFFER_OVER_OUTRIGHT_PTS) {
      add("offer_over_outright", row,
        `${row.label} (${row.duration}): offer ${pct(row.ourPct)} is ${r2(row.ourPct - row.outrightPct)} pts above the same bar bought outright (${pct(row.outrightPct)}).`,
        Math.max(row.outrightPct + OFFER_OVER_OUTRIGHT_PTS, floorOf(row)), appliesTo(row));
    }
  }
 
  const anchors = [...input.curveAnchors].sort((a, b) => a.grams - b.grams);
  const rises = anchors.slice(1).flatMap((a, i) => (a.pct > anchors[i].pct ? [`${r2(anchors[i].grams)} g ${pct(anchors[i].pct)} → ${r2(a.grams)} g ${pct(a.pct)}`] : []));
  if (rises.length) {
    add("curve_rises", null, `The Premium Curve rises with weight: ${rises.join("; ")}.`, null, "Premium Curve anchors (Admin → Premiums)");
  }
 
  return { spotSgdPerG: r2(spot), rows, flags, openFlags: flags.filter(f => !f.accepted).length };
}
 
/** "Pricing review 14 Nov: 2 to decide (…)" / "…: all clear". Deterministic. */
export function reviewHeadline(review: PricingReview, sgtDate: string): string {
  const open = review.flags.filter(f => !f.accepted);
  const day = new Date(`${sgtDate}T00:00:00Z`).toLocaleDateString("en-SG", { day: "numeric", month: "short", timeZone: "UTC" });
  if (!open.length) return `Pricing review ${day}: all clear — ${review.rows.length} prices checked.`;
  const first = open.slice(0, 2).map(f => f.message.split(":")[0]).join(", ");
  return `Pricing review ${day}: ${open.length} to decide (${first}${open.length > 2 ? ", …" : ""}).`.slice(0, 280);
}
 
/** Today's date in Singapore, YYYY-MM-DD. */
export function sgtDate(now: Date = new Date()): string {
  return new Date(now.getTime() + 8 * 3600_000).toISOString().slice(0, 10);
}
 
/**
 * The premium we paid for a set of held units, % over the spot on the day each
 * was bought, card fee out; equal-weighted (units of one product share a weight).
 * Units with no recorded spot are skipped; null when none is usable.
 */
export function paidPremiumPct(units: ReadonlyArray<{ costPerG: number; feePerG?: number | null; spotAtPurchase: number | null }>): number | null {
  const pcts = units.flatMap(u => {
    const fee = u.feePerG != null && u.feePerG > 0 ? u.feePerG : 0;
    const spot = u.spotAtPurchase;
    if (!(spot && spot > 0) || !(u.costPerG > 0)) return [];
    return [((u.costPerG - fee) / spot - 1) * 100];
  });
  return pcts.length ? pcts.reduce((a, b) => a + b, 0) / pcts.length : null;
}
