/**
 * Weekend / holiday pricing — what customers pay and are paid while the gold
 * market is closed (Mukesh 2026-11-22). Settings: Admin → Premiums → "When markets
 * are closed" (table `market_closed_pricing`, migration 20261122b). Ships OFF.
 *
 * ▶ ONE NUMBER, NOT TWENTY-FIVE FORMULAS. Every customer price is
 * spot × grams × (1 + premium); there is no single price function, but there is a
 * single SPOT. So the buffer is applied to the spot customer prices are built from
 * (the "pricing spot"), at the few places that hand spot to customer pricing:
 * the pricing snapshot (web + app displays), the app engine bootstrap, the checkout
 * lock + its 1 % guard, the charge, marketplaces and the AI calculator. Internal
 * tools (strategy, margins, supplier comparison) keep reading getSpotPayload and
 * the TRUE spot — they never import this file.
 *
 * ▶ THE TICKER SHOWS THE TRUE SPOT (Mukesh: customers compare it with Google); the
 * closed-market line discloses the buffer instead ("… and our prices include a
 * small 0.5% weekend allowance"). Buyback is the mirror: 'reduce' pays off spot × (1 − pct), 'pause'
 * refuses new buybacks until the reopen.
 */

const pct = (v: unknown) => {
  const n = Number(v);
  return Number.isFinite(n) && n >= 0 && n <= MAX_PCT ? n : 0;
};
 
export function parseSettingsRow(row: unknown): MarketClosedSettings {
  if (!row || typeof row !== "object") return MARKET_CLOSED_OFF;
  const r = row as Record<string, unknown>;
  const mode = r.buyback_mode === "pause" || r.buyback_mode === "reduce" ? r.buyback_mode : "off";
  return { sellBufferPct: pct(r.sell_buffer_pct), buybackMode: mode, buybackReducePct: pct(r.buyback_reduce_pct) };
}
 
// ── Pure ─────────────────────────────────────────────────────────────────────
 
/** The sell buffer in force now (0 while open). */
export function activeSellBufferPct(m: MarketStatus, s: MarketClosedSettings): number {
  return m.open ? 0 : s.sellBufferPct;
}
 
/** The spot every customer SELL price is built from. */
export function pricingSpot(marketSpot: number, m: MarketStatus, s: MarketClosedSettings): number {
  const b = activeSellBufferPct(m, s);
  return b > 0 ? marketSpot * (1 + b / 100) : marketSpot;
}
 
/** The buyback reduction in force now (0 while open or not in 'reduce' mode). */
export function activeBuybackReducePct(m: MarketStatus, s: MarketClosedSettings): number {
  return !m.open && s.buybackMode === "reduce" ? s.buybackReducePct : 0;
}
 
/** The spot every customer BUYBACK quote is built from. */
export function buybackSpot(marketSpot: number, m: MarketStatus, s: MarketClosedSettings): number {
  const r = activeBuybackReducePct(m, s);
  return r > 0 ? marketSpot * (1 - r / 100) : marketSpot;
}
 
export function buybacksPaused(m: MarketStatus, s: MarketClosedSettings): boolean {
  return !m.open && s.buybackMode === "pause";
}
 
const fmtPct = (n: number) => `${Number(n.toFixed(3))}%`;
 
/**
 * The sell-to-us line while closed (same voice as the ticker's, Mukesh 2026-11-22);
 * null while open or when buybacks run normally.
 */
export function buybackClosedNote(m: MarketStatus, s: MarketClosedSettings, reopenLabel: string): string | null {
  if (m.open) return null;
  const lead = `We are open as usual. The global gold spot price doesn't update until ${reopenLabel}`;
  if (s.buybackMode === "pause") return `${lead}, so sell-back requests reopen then. Buying gold continues as normal.`;
  const r = activeBuybackReducePct(m, s);
  if (r > 0) return `${lead}, so ${m.reason === "holiday" ? "holiday" : "weekend"} buyback quotes are ${fmtPct(r)} below it.`;
  return null;
}

export async function loadMarketClosedSettings(db?: SupabaseClient, now = Date.now()): Promise<MarketClosedSettings> {

  try {

}

/**
 * Context for the AI assistants while a weekend buffer is in force: they price with
 * the pricing spot, so tell them the TRUE spot to quote and the buffer to mention —
 * otherwise the prompt's "Spot:" line would present the buffered figure as the spot.
 * Empty when no buffer applies.
 */
export function aiMarketClosedContext(cs: CustomerSpot): string {
  if (cs.sellBufferPct <= 0) return "";
  return `\n\nGOLD MARKET CLOSED (weekend/holiday): the true spot is SGD ${cs.marketSgd.toFixed(2)}/g. `
    + `While the market is closed, customer prices are built from SGD ${cs.pricingSgd.toFixed(2)}/g — a ${Number(cs.sellBufferPct.toFixed(3))}% weekend buffer. `
    + `When asked for the spot price, quote SGD ${cs.marketSgd.toFixed(2)}/g and say weekend prices include the ${Number(cs.sellBufferPct.toFixed(3))}% buffer until the market reopens.`;
}
