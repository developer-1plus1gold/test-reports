/**
 * Extract per-product source URLs from a supplier PDF's LINK ANNOTATIONS (#2).
 *
 * The vision parser reads TEXT, not PDF hyperlinks. The MTB sheet hyperlinks each
 * product row (the "→") to wholesale.mtbmetals.com/product/{id}. We read those
 * link annotations + the text positions with pdfjs and pair each URL with the
 * product name on the same row (by y-overlap), so the caller can match a parsed
 * line to its URL by product identity.
 *
 * Fail-soft: any pdfjs error → returns [] (URLs are a nice-to-have, never block parsing).
 */
 
import { loadPdfjs } from "@/lib/pdfjs-node";
 
export interface PdfProductUrl {
  /** The product-name text found on the same row as the link (raw). */
  text: string;
  url: string;
  page: number;
}
 
export async function extractPdfProductUrls(bytes: Uint8Array): Promise<PdfProductUrl[]> {
  try {
    // Legacy build runs on the main thread in Node (no worker).
    const pdfjs = await loadPdfjs();
    const doc = await pdfjs.getDocument({ data: bytes, isEvalSupported: false, useSystemFonts: true }).promise;
    const out: PdfProductUrl[] = [];
 
    for (let p = 1; p <= doc.numPages; p++) {
      const page = await doc.getPage(p);
      const [annots, text] = await Promise.all([page.getAnnotations({ intent: "display" }), page.getTextContent()]);
 
      // Text items with position: { str, x, y }. y is the baseline (PDF space, origin bottom-left).
      const items = (text.items as Array<{ str: string; transform: number[]; width: number }>)
        .filter((i) => i.str && i.str.trim())
        .map((i) => ({ str: i.str, x: i.transform[4], y: i.transform[5], w: i.width }));
 
      for (const a of annots as Array<{ subtype?: string; url?: string; rect?: number[] }>) {
        if (a.subtype !== "Link" || !a.url || !a.rect) continue;
        if (!/^https?:\/\//i.test(a.url)) continue;
        const [x1, y1, x2, y2] = a.rect;
        const yLo = Math.min(y1, y2) - 2;
        const yHi = Math.max(y1, y2) + 2;
        // The MTB link rect spans the WHOLE product-name row — gather the text
        // items inside it (y in the row band, x within the rect), left-to-right.
        const rowText = items
          .filter((i) => i.y >= yLo && i.y <= yHi && i.x >= Math.min(x1, x2) - 2 && i.x <= Math.max(x1, x2) + 2)
          .sort((m, n) => m.x - n.x)
          .map((i) => i.str)
          .join("")
          .replace(/\s+/g, " ")
          .replace(/→\s*$/, "")
          .trim();
        if (rowText) out.push({ text: rowText, url: a.url, page: p });
      }
    }
    await doc.cleanup();
    return out;
  } catch (e) {
    console.error("[pdf-product-urls] extraction failed:", (e as Error).message);
    return [];
  }
}
