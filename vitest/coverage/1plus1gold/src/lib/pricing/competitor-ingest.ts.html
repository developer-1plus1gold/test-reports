/** bullionking.sg — WooCommerce gold category, prices server-rendered in the HTML.

/** UOB "unit" → grams + a catalog-style label. "100 GM"→100g, "2.5 GM"→2.5g,
 *  "1 KILOBAR"/"1 KILOCERT"→1kg, "1 OZ"→1oz. Returns null for fractional-oz
 *  coins ("1/10 OZ") and anything unrecognised. */
export function parseUobUnitGrams(unit: string): { grams: number; label: string } | null {
  const u = unit.trim().toUpperCase();
  const gm = u.match(/^([\d.]+)\s*GM\b/);
  if (gm) {
    const g = parseFloat(gm[1]);
    return Number.isFinite(g) && g > 0 ? { grams: g, label: `${g}g` } : null;

  if (/\bKILO(BAR|CERT)\b/.test(u)) return { grams: 1000, label: "1kg" };

  if (oz && !u.includes("/")) {
    const n = parseFloat(oz[1]);
    return Number.isFinite(n) && n > 0 ? { grams: n * OZ_GRAMS, label: `${n}oz` } : null;
  }
  return null; // fractional-oz coins etc.

    const price = t.bankSell ? parseMoney(t.bankSell) : null;
    if (!price) continue;
    const key = gramsKey(w.grams);
    if (filter && !filter.has(key)) continue; // only favourite sizes
    const ex = cheapest.get(key);
    if (!ex || price < ex.priceSgd) {
      cheapest.set(key, { productName: `UOB ${w.label} Gold Bar`, weightGrams: w.grams, weightLabel: w.label, priceSgd: price, availability: "live" });
    }
  }
  return [...cheapest.values()];
}

/** Drop scraped rows whose per-gram price is implausible vs live spot — a guard
 *  against a site layout change feeding garbage prices into the benchmark (a
 *  spuriously LOW price is the dangerous case: it would wrongly become the
 *  "cheapest competitor"). Retail is ~always ≥ spot; themed bars run up to ~4×.
 *  Only applied to AUTO scrapes (manual/CSV are operator-trusted). */
export function dropImplausibleRows(
  rows: CompetitorRowInput[], spotSgdPerGram: number,
): { kept: CompetitorRowInput[]; dropped: number } {
  if (!(spotSgdPerGram > 0)) return { kept: rows, dropped: 0 };
  const lo = spotSgdPerGram * 0.7, hi = spotSgdPerGram * 5;
  const kept = rows.filter((r) => {
    const perG = r.priceSgd / r.weightGrams;
    return perG >= lo && perG <= hi;
  });
  return { kept, dropped: rows.length - kept.length };
}
 
/** The gram sizes we actively benchmark = the weights of our starred supplier
 *  products (Supplier Sheets ★). Returns rounded gram keys for matching. */
export async function getFavoriteGrams(db: SupabaseClient): Promise<Set<number>> {
  const { data } = await db.from("supplier_product_favorites").select("label, product_key");
  const out = new Set<number>();
  for (const f of data ?? []) {
    const w = parseWeightGrams((f.label as string) ?? "") ?? parseWeightGrams((f.product_key as string) ?? "");
    if (w) out.add(gramsKey(w.grams));
  }
  return out;
}

/** Validate + persist a new dated competitor price list. Throws on a bad insert. */
export async function saveCompetitorList(
  db: SupabaseClient,
  competitorId: string,
  source: "manual" | "csv" | "auto",
  rawRows: CompetitorRowInput[],
  spotSgdPerGram?: number | null,
): Promise<{ listId: string; saved: number }> {
  const clean = [] as { product_name: string; weight_grams: number; weight_label: string; price_sgd: number; availability: string }[];
  for (const r of rawRows) {
    const grams = Number(r.weightGrams);
    const price = Number(r.priceSgd);
    if (!Number.isFinite(grams) || grams <= 0) continue;
    if (!Number.isFinite(price) || price <= 0) continue;
    const name = (r.productName ?? "").toString().trim().slice(0, 200) || `${grams} gram`;
    clean.push({
      product_name: name,
      weight_grams: grams,
      weight_label: (r.weightLabel ?? "").toString().trim().slice(0, 60) || `${grams} gram`,
      price_sgd: price,
      availability: AVAILS.has((r.availability ?? "").toString()) ? (r.availability as string) : "live",
    });
  }
  if (clean.length === 0) throw new Error("No valid rows (each needs a positive weight in grams and a positive SGD price)");
 
  const listDate = new Date().toISOString().slice(0, 10);
  const { data: list, error: listErr } = await db
    .from("competitor_price_lists")
    .insert({
      competitor_id: competitorId, list_date: listDate, source, parse_status: "parsed",
      spot_sgd_per_gram: spotSgdPerGram != null && Number.isFinite(spotSgdPerGram) ? spotSgdPerGram : null,
    })
    .select("id").maybeSingle();
  if (listErr || !list) throw new Error("Failed to create price list");
 
  const { error: rowsErr } = await db.from("competitor_prices").insert(
    clean.map((c) => ({ list_id: list.id, competitor_id: competitorId, list_date: listDate, currency: "SGD", ...c })),
  );
  if (rowsErr) {
    await db.from("competitor_price_lists").delete().eq("id", list.id);
    throw new Error("Failed to save prices");
  }
  return { listId: list.id, saved: clean.length };
}
