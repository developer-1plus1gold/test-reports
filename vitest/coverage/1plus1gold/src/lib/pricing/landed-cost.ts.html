/**
 * Pricing-strategy kernel — DETERMINISTIC money math for the supplier-pricing
 * module. The LLM parser only EXTRACTS + CLASSIFIES each supplier line (grams,
 * unitKind, askKind, askValue); all cost/margin arithmetic lives here so it is
 * testable and never hallucinated (mirrors the order-pricing kernel split).
 *
 * Units: spot is XAU. 1 troy oz = 31.1035 g (TROY_OZ). Supplier asks are USD;
 * we convert to SGD with the live USD→SGD rate. Landed cost adds configurable
 * shipping (flat fee + per-gram rate).
 *
 * The brief's CRITICAL caveat — how to read the supplier "ask":
 *   unitKind 'per_piece' ($ per bar/piece, gram bars + multigram):
 *       itemCostUsd = spotUsdPerG × grams + askDollar
 *       e.g. 1g $13 → spot/g + 13 ;  10g $21 → spot/g×10 + 21 ;  12×1g $100 → spot/g×12 + 100
 *   unitKind 'per_oz' ($ premium per troy oz, oz bars):
 *       itemCostUsd = (spotUsdPerOz + askDollar) × ozCount      (ozCount = grams/TROY_OZ)
 *       e.g. 1oz $45 → (spot/oz + 45)×1
 *   unitKind 'percent' (% of spot per oz, coins quoted like 103.15%):
 *       itemCostUsd = spotUsdPerG × grams × (askPercent/100)    (the % already INCLUDES spot)
 *       e.g. 1oz coin 103.15% → spot/oz × 1.0315
 */
 
import { TROY_OZ } from "@/lib/utils";
 
export type UnitKind = "per_piece" | "per_oz" | "percent";
export type AskKind = "dollar" | "percent" | "none";
export type MarginStatus = "green" | "amber" | "red";
 
export interface ShippingConfig {
  flatFeeUsd: number;     // per-shipment shipping flat fee (USD)
  perGramFeeUsd: number;  // per-gram shipping/handling (USD)
  wireFeeUsd?: number;    // per-shipment wire-transfer flat fee (USD) — spread like the flat fee
  // Bank FX spread: SGD charged ABOVE the live mid-market rate per USD wired (e.g.
  // bank 1.2862 − live 1.2840 = 0.0022). Applied to the whole landed USD on convert.
  fxMarkupSgdPerUsd?: number;
}
 
export interface SpotContext {
  spotUsdPerOz: number;   // XAU/USD per troy oz
  usdToSgd: number;       // 1 USD = X SGD
}
 
export interface SupplierLineInput {
  grams: number;          // canonical grams of metal content
  unitKind: UnitKind;
  askKind: AskKind;
  askValue: number | null; // USD amount, or percent (e.g. 103.15), null if no ask
}
 
export interface LandedCost {
  itemCostUsd: number;        // supplier cost for the whole PIECE (before shipping)
  shippingPerGramUsd: number; // per-gram fee + the flat fee's per-gram share
  landedPerGramUsd: number;
  landedPerGramSgd: number;
  landedPerPieceSgd: number;  // landedPerGramSgd × the piece's grams
  // ── Per-PIECE breakdown for the Supplier-Sheet table (USD + SGD sides) ──
  // The columns split landed into "supplier price" + "shipping & currency charges".
  // On the SGD side, the currency-charges bucket ALSO absorbs the FX markup, so the
  // three columns stay internally consistent: supplierSgd + shipping&fxSgd = landedSgd.
  shippingPerPieceUsd: number;        // shippingPerGramUsd × grams
  landedPerPieceUsd: number;          // landedPerGramUsd × grams
  itemCostSgd: number;                // itemCostUsd × the live mid rate (no FX markup)
  shippingPlusFxPerPieceSgd: number;  // landedPerPieceSgd − itemCostSgd (shipping + FX markup)
  // ── Quote-builder decomposition (2026-08-01) ──────────────────────────────
  // The Quote Builder absorbs the flat shipping + wire fee and spreads it BY QTY
  // (per piece) over the ACTUAL quote — not a fixed order size — because that's
  // the only allocation that doesn't distort a thin-premium large bar (a fixed
  // fee spread by weight/value scales WITH value and doubles a 1%-premium 100g
  // bar; by piece it's inverse to value, so cheap small bars — already
  // wide-premium — absorb it and big low-premium bars stay near spot). These two
  // fields let the quote re-amortise: cost EX the flat freight (composition-
  // independent, includes per-gram fee + FX markup), and the flat freight for
  // ONE shipment in SGD (flat + wire, converted at the FX-marked rate).
  costExFlatFreightPerPieceSgd: number;  // (item + per-gram fee) × grams × (rate + fx markup)
  flatFreightPerShipmentSgd: number;     // (flat fee + wire fee) × (rate + fx markup)
}
 
export const spotUsdPerGram = (spotUsdPerOz: number): number => spotUsdPerOz / TROY_OZ;
 
/**
 * Supplier cost for the whole item (USD), BEFORE shipping, per the caveat table.
 * Returns null when the line has no usable ask (askKind 'none' / null value).
 */
export function itemCostUsd(line: SupplierLineInput, spot: SpotContext): number | null {
  const { grams, unitKind, askKind, askValue } = line;
  if (askKind === "none" || askValue == null || !Number.isFinite(askValue)) return null;
  if (!Number.isFinite(grams) || grams <= 0) return null;
 
  const spotG = spotUsdPerGram(spot.spotUsdPerOz);
  const ozCount = grams / TROY_OZ;
 
  switch (unitKind) {
    case "per_piece":
      // $ premium for the whole piece, added to the metal value.
      return spotG * grams + askValue;
    case "per_oz":
      // $ premium per troy oz, added to spot/oz, scaled by oz content.
      return (spot.spotUsdPerOz + askValue) * ozCount;
    case "percent":
      // ask is the TOTAL price as a % of spot (includes spot), per oz of content.
      return spotG * grams * (askValue / 100);
  }
}
 
/**
 * Landed cost per gram (and per piece), in USD and SGD.
 *
 * Shipping/fees have these parts:
 *  • per-gram fee — a per-gram cost, added to every gram directly.
 *  • flat fee + wire-transfer fee — PER-SHIPMENT costs. Both are spread over
 *    `gramsOrdered` (the total order quantity), NOT over a single piece's weight
 *    — otherwise the $250 flat fee lands entirely on a 1g bar (→ absurd per-gram
 *    cost). `gramsOrdered` defaults to the piece's own grams when not supplied.
 *  • FX markup — the bank's spread over the live mid-market USD→SGD rate. The
 *    whole landed USD is wired and converted at (live + markup), so it applies
 *    proportionally to item + shipping + wire.
 */
export function computeLandedCost(
  line: SupplierLineInput,
  spot: SpotContext,
  shipping: ShippingConfig,
  gramsOrdered?: number,
): LandedCost | null {
  const item = itemCostUsd(line, spot);
  if (item == null) return null;
  const orderG = gramsOrdered && gramsOrdered > 0 ? gramsOrdered : line.grams;
  // Flat shipping fee + flat wire-transfer fee both spread over the order.
  const flatTotalUsd = Math.max(0, shipping.flatFeeUsd) + Math.max(0, shipping.wireFeeUsd ?? 0);
  const flatPerGramUsd = flatTotalUsd / orderG;
  const shippingPerGramUsd = Math.max(0, shipping.perGramFeeUsd) + flatPerGramUsd;
  const landedPerGramUsd = item / line.grams + shippingPerGramUsd;
  const effectiveUsdToSgd = spot.usdToSgd + Math.max(0, shipping.fxMarkupSgdPerUsd ?? 0);
  const landedPerGramSgd = landedPerGramUsd * effectiveUsdToSgd;
  const landedPerPieceSgd = landedPerGramSgd * line.grams;
  const itemCostSgd = item * spot.usdToSgd; // supplier price at the live mid rate (markup lives in the shipping&currency bucket)
  // Cost EXCLUDING the flat (shipment) freight — item + per-gram fee, at the
  // FX-marked rate. Composition-independent, so the quote builder can add the
  // by-qty flat-freight share on top.
  const exFlatPerGramUsd = item / line.grams + Math.max(0, shipping.perGramFeeUsd);
  const costExFlatFreightPerPieceSgd = exFlatPerGramUsd * line.grams * effectiveUsdToSgd;
  return {
    itemCostUsd: item,
    shippingPerGramUsd,
    landedPerGramUsd,
    landedPerGramSgd,
    landedPerPieceSgd,
    shippingPerPieceUsd: shippingPerGramUsd * line.grams,
    landedPerPieceUsd: landedPerGramUsd * line.grams,
    itemCostSgd,
    shippingPlusFxPerPieceSgd: landedPerPieceSgd - itemCostSgd,
    costExFlatFreightPerPieceSgd,
    flatFreightPerShipmentSgd: flatTotalUsd * effectiveUsdToSgd,
  };
}

/**
 * Hard floor on the retail gross margin any AUTOMATED premium proposal may
 * imply. Lives here, in the pricing library, so every producer of premium
 * proposals shares one definition — there is exactly one today
 * (`/api/cron/pricing-strategist`), and this is where a second one would find it
 * rather than re-deriving the rule.
 *
 * Why it is code and not prompt prose: on 2026-08-16 the pricing_strategist filed
 * a 10g proposal implying ~1.5% gross margin while its prompt asked for ≥4% — its
 * own rationale said "preserving a ~1.5% gross margin". A money rule stated only
 * in a prompt does not hold. Agent contract §0.2 — math is code.
 *
 * NOTE this bounds what an AGENT may PROPOSE. It is deliberately NOT applied to
 * the operator's own premium edits: 1g CombiBars are a deliberate loss-leader
 * selling below landed cost (accepted policy, 2026-08-12), and premium writes are
 * human-gated by design (§0.4). See `docs/pricing-benchmark-redesign.md`.
 */
export const MIN_RETAIL_GROSS_MARGIN_PCT = 4;
 
/** Gross margin on revenue implied by a premium over spot, given landed cost. */
export function impliedRetailMarginPct(
  premiumPct: number,
  spotSgdPerGram: number,
  costSgdPerGram: number,
): number | null {
  const retail = spotSgdPerGram * (1 + premiumPct / 100);
  if (!Number.isFinite(retail) || retail <= 0) return null;
  return ((retail - costSgdPerGram) / retail) * 100;
}
 
/**
 * May an automated proposal at this premium be filed?
 *
 * **Fails CLOSED** — when the margin cannot be computed (non-positive retail,
 * missing inputs) the answer is `false`. This is a money rule: "we could not
 * check" must not read as "it passed". Callers distinguish the two by testing
 * `impliedMarginPct == null` (unevaluable) vs a number below the floor.
 */
export function passesRetailMarginFloor(
  premiumPct: number,
  spotSgdPerGram: number,
  costSgdPerGram: number,
  floorPct: number = MIN_RETAIL_GROSS_MARGIN_PCT,
): { ok: boolean; impliedMarginPct: number | null } {
  const m = impliedRetailMarginPct(premiumPct, spotSgdPerGram, costSgdPerGram);
  return { ok: m != null && m >= floorPct, impliedMarginPct: m };
}
 
/** Below this, a premium move is churn rather than a decision worth an operator tap. */
export const MIN_PREMIUM_DELTA_PCT = 0.5;
 
/** The shape any premium proposal must have to be gated (agent output, validated). */
export interface GateablePremiumProposal {
  grams: number;
  label: string;
  currentPremiumPct: number;
  proposedPremiumPct: number;
  direction: "raise" | "lower" | "hold";
}
 
export interface ProposalGateResult<T> {
  /** Proposals that cleared every gate — the only ones a caller may file. */
  allowed: T[];
  /** Dropped for implying a retail gross margin under the floor. */
  blockedByFloor: Array<{ grams: number; label: string; proposedPremiumPct: number; impliedMarginPct: number }>;
  /**
   * Dropped because the floor could NOT be evaluated — no landed cost for that
   * size, or no spot. Surfaced separately from `blockedByFloor` because it means
   * "we could not check", not "we checked and it failed"; a non-zero count here
   * is a data-quality signal (a product with no cost basis) rather than a pricing
   * one, and the operator should be able to tell the two apart.
   */
  blockedNoCostBasis: Array<{ grams: number; label: string; proposedPremiumPct: number }>;
  /** Dropped as a hold or a sub-threshold move. */
  droppedAsChurn: number;
}
 
/**
 * THE gate every producer of automated premium proposals must pass its output
 * through before writing anything. Enforcement lives here, in the pricing library
 * — not in a route — so a second producer inherits it by construction rather than
 * by remembering to re-implement it. Pure and synchronous; the caller does the I/O.
 *
 * Gates, in order: drop holds → drop sub-`MIN_PREMIUM_DELTA_PCT` churn → drop
 * anything implying a retail gross margin below `floorPct`.
 *
 * `costByGrams` should carry the HIGHEST landed cost among products sharing a
 * size: proposals are per-size and a per-denomination curve premium covers every
 * product at that gram, so the dearest one binds.
 *
 * **FAILS CLOSED.** A size with no landed cost, or a null spot, cannot be
 * evaluated — and an unevaluable proposal is withheld, not passed through
 * (`blockedNoCostBasis`). An earlier cut of this failed OPEN on the reasoning
 * that "nothing to floor against is not a reason to drop a proposal"; that is the
 * wrong default for a money rule, because the case where cost is unknown is
 * exactly the case where we cannot tell whether the price is profitable
 * (security_agent_pr, 2026-08-16). The cost of failing closed is a withheld
 * proposal, visible in the counter; the cost of failing open is an unpriceable
 * product reaching the queue unchecked.
 *
 * Why this is code and not prompt prose: on 2026-08-16 pricing_strategist filed a
 * 10g proposal implying ~1.5% gross margin while its prompt asked for ≥4%, its own
 * rationale saying "preserving a ~1.5% gross margin". Agent contract §0.2 — math
 * is code. See `docs/pricing-benchmark-redesign.md`.
 */
export function gatePremiumProposals<T extends GateablePremiumProposal>(
  proposals: T[],
  ctx: {
    spotSgdPerGram: number | null;
    costByGrams: Map<number, number>;
    floorPct?: number;
    minDeltaPct?: number;
  },
): ProposalGateResult<T> {
  const floorPct = ctx.floorPct ?? MIN_RETAIL_GROSS_MARGIN_PCT;
  const minDelta = ctx.minDeltaPct ?? MIN_PREMIUM_DELTA_PCT;
  const blockedByFloor: ProposalGateResult<T>["blockedByFloor"] = [];
  const blockedNoCostBasis: ProposalGateResult<T>["blockedNoCostBasis"] = [];
  let droppedAsChurn = 0;
 
  const allowed = (proposals ?? []).filter((p) => {
    if (p.direction === "hold"
      || !Number.isFinite(p.proposedPremiumPct)
      || Math.abs(p.proposedPremiumPct - p.currentPremiumPct) < minDelta) {
      droppedAsChurn++;
      return false;
    }
    const cost = ctx.costByGrams.get(p.grams);
    const { ok, impliedMarginPct } = cost != null && ctx.spotSgdPerGram != null
      ? passesRetailMarginFloor(p.proposedPremiumPct, ctx.spotSgdPerGram, cost, floorPct)
      : { ok: false, impliedMarginPct: null };
    if (ok) return true;
    if (impliedMarginPct == null) {
      // Could not evaluate — withhold, and say so distinctly.
      blockedNoCostBasis.push({ grams: p.grams, label: p.label, proposedPremiumPct: p.proposedPremiumPct });
      return false;
    }
    blockedByFloor.push({
      grams: p.grams, label: p.label, proposedPremiumPct: p.proposedPremiumPct,
      impliedMarginPct: Number(impliedMarginPct.toFixed(2)),
    });
    return false;
  });
 
  return { allowed, blockedByFloor, blockedNoCostBasis, droppedAsChurn };
}

  const tier3 = t.spotSgdPerGram * (1 + t.currentPremiumPct / 100);
  const suggestive = tier1 * (1 + t.currentPremiumPct / 100);
 
  const wholesaleMargin = grossMarginPct(tier2, tier1);
  const retailMargin = grossMarginPct(tier3, tier1);
 
  const comp = t.cheapestCompetitorSgd ?? null;
  const vsBest = comp != null && comp > 0 ? (tier3 / comp - 1) * 100 : null;
 
  return {
    tier1CostSgd: tier1,
    tier2WholesaleSgd: tier2,
    tier3RetailSgd: tier3,
    suggestiveRetailSgd: suggestive,
    tier1PremiumPct: premiumOverSpotPct(tier1, t.spotSgdPerGram),
    tier2PremiumPct: premiumOverSpotPct(tier2, t.spotSgdPerGram),
    tier3PremiumPct: premiumOverSpotPct(tier3, t.spotSgdPerGram),
    wholesaleMarginPct: wholesaleMargin,
    retailMarginPct: retailMargin,
    vsBestCompetitorPct: vsBest,
    status: marginStatus(tier3, tier2, comp, wholesaleMargin, t.amberWholesaleFloorPct, t.competitorComparable ?? true),
  };
}
 
/**
 * Status per the brief:
 *  • Red   — retail below wholesale (selling at a loss / near-zero margin).
 *  • Amber — retail above the cheapest competitor (being undercut), OR wholesale
 *            margin below the configured floor.
 *  • Green — retail below cheapest competitor and above wholesale, margin healthy.
 */
/**
 * @param competitorComparable — when false the benchmark is only the cheapest
 *   DIFFERENT product at this weight (WEIGHT grade), so it must not raise amber.
 *   Defaults true to preserve existing callers.
 */
export function marginStatus(
  tier3Retail: number,
  tier2Wholesale: number,
  cheapestCompetitorSgd: number | null,
  wholesaleMarginPct: number,
  amberFloorPct: number,
  competitorComparable: boolean = true,
): MarginStatus {
  if (tier3Retail < tier2Wholesale) return "red";
  const undercut = competitorComparable && cheapestCompetitorSgd != null && cheapestCompetitorSgd > 0 && tier3Retail > cheapestCompetitorSgd;
  if (undercut || wholesaleMarginPct < amberFloorPct) return "amber";
  return "green";
}
 
// ── Per-denomination margin CURVE model (Three-Tier Strategy, 2026-06-14) ─────
// Margin varies by denomination (1g ~23% → 100g ~3.5%), so the control is a curve
// of anchors by gram, interpolated for in-between sizes (same log-space interp as
// the shop premium curve). Decision "Both": premium-over-spot DRIVES the price
// (price = spot × (1+premium)); separate margin-over-cost FLOORS guard (amber).
 
export interface CurveAnchor { grams: number; value: number }
 
/**
 * Log-space linear interpolation across curve anchors, clamped flat beyond the
 * ends. Mirrors `getPremiumForWeight` so the strategy curve behaves like the shop
 * premium curve. Returns 0 if there are no usable anchors.
 */
export function interpAnchors(grams: number, anchors: CurveAnchor[]): number {
  const sorted = anchors
    .filter((a) => a.grams > 0 && Number.isFinite(a.value))
    .sort((a, b) => a.grams - b.grams);
  if (sorted.length === 0) return 0;
  if (grams <= sorted[0].grams) return sorted[0].value;
  const last = sorted[sorted.length - 1];
  if (grams >= last.grams) return last.value;
  for (let i = 0; i < sorted.length - 1; i++) {
    const a = sorted[i], b = sorted[i + 1];
    Eif (grams >= a.grams && grams <= b.grams) {
      const t = (Math.log(grams) - Math.log(a.grams)) / (Math.log(b.grams) - Math.log(a.grams));
      return a.value + t * (b.value - a.value);
    }
  }
  return last.value;
}
 
export interface CurveTierInputs {
  costSgdPerGram: number;            // landed cost SGD/g
  grams: number;
  spotSgdPerGram: number;
  retailPremiumPct: number;          // over spot (resolved for this grams) → retail price
  wholesalePremiumPct: number;       // over spot → wholesale price
  retailFloorPct: number;            // min gross margin over cost (amber below)
  wholesaleFloorPct: number;         // min gross margin over cost (amber below)
  cheapestCompetitorSgd?: number | null;
  /** Like-for-like (E/S/R) benchmark? Only then may it drive amber — see TierInputs. */
  competitorComparable?: boolean;
  currentRetailPremiumPct?: number | null;    // from products (display only)
  currentWholesalePremiumPct?: number | null; // from products (display only)
}
 
export interface CurveTierResult {
  costPerGramSgd: number;
  wholesalePerGramSgd: number;
  retailPerGramSgd: number;
  costPerPieceSgd: number;
  wholesalePerPieceSgd: number;
  retailPerPieceSgd: number;
  costPremiumPct: number;            // cost over spot (derived)
  retailPremiumPct: number;          // over spot (echo of the resolved curve)
  wholesalePremiumPct: number;       // over spot (echo)
  wholesaleMarginPct: number;        // premium SPREAD = wholesale premium − cost premium (pts over spot)
  retailMarginPct: number;           // premium SPREAD = retail premium − cost premium (pts over spot)
  retailFloorPct: number;            // echo of the resolved floor
  wholesaleFloorPct: number;         // echo
  currentRetailPremiumPct: number | null;
  currentWholesalePremiumPct: number | null;
  vsBestCompetitorPct: number | null;
  status: MarginStatus;
}
 
export function computeCurveTiers(t: CurveTierInputs): CurveTierResult {
  const spot = t.spotSgdPerGram;
  const cost = t.costSgdPerGram;
  const retail = spot * (1 + t.retailPremiumPct / 100);
  const wholesale = spot * (1 + t.wholesalePremiumPct / 100);
 
  // "Margin" = premium SPREAD over cost, in points over spot ((price − cost) / spot),
  // i.e. sell premium − cost premium. This is how gold pricing is reasoned about
  // (buy at +11.8%, sell at +15% → 3.2 pts) and the floors are set in the same unit.
  const costPremiumPct = premiumOverSpotPct(cost, spot);
  const retailMargin = t.retailPremiumPct - costPremiumPct;
  const wholesaleMargin = t.wholesalePremiumPct - costPremiumPct;
 
  const comp = t.cheapestCompetitorSgd ?? null;
  const vsBest = comp != null && comp > 0 ? (retail / comp - 1) * 100 : null;
 
  // Red — retail below wholesale (mispriced) OR below cost (true loss). Amber —
  // undercut by a competitor OR a margin under its per-denomination floor. Else green.
  let status: MarginStatus;
  if (retail < wholesale || retail < cost) {
    status = "red";
  } else {
    const undercut = (t.competitorComparable ?? true) && comp != null && comp > 0 && retail > comp;
    const thin = retailMargin < t.retailFloorPct || wholesaleMargin < t.wholesaleFloorPct;
    status = undercut || thin ? "amber" : "green";
  }
 
  return {
    costPerGramSgd: cost,
    wholesalePerGramSgd: wholesale,
    retailPerGramSgd: retail,
    costPerPieceSgd: cost * t.grams,
    wholesalePerPieceSgd: wholesale * t.grams,
    retailPerPieceSgd: retail * t.grams,
    costPremiumPct,
    retailPremiumPct: t.retailPremiumPct,
    wholesalePremiumPct: t.wholesalePremiumPct,
    wholesaleMarginPct: wholesaleMargin,
    retailMarginPct: retailMargin,
    retailFloorPct: t.retailFloorPct,
    wholesaleFloorPct: t.wholesaleFloorPct,
    currentRetailPremiumPct: t.currentRetailPremiumPct ?? null,
    currentWholesalePremiumPct: t.currentWholesalePremiumPct ?? null,
    vsBestCompetitorPct: vsBest,
    status,
  };
}
