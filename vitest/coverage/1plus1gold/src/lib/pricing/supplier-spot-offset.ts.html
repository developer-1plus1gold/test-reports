/**
 * Supplier spot offset, measured at sheet upload (2026-10-08).
 *
 * StoneX prices every bar off its OWN gold spot, which runs above ours — about
 * +0.34% on a weekday and more than double that at weekends. Their portal sits
 * behind a login, so for now the operator reads StoneX's spot off the portal and
 * types it in when uploading the day's sheet. We record our own spot at the same
 * moment and the offset between them; the landed kernel then prices the metal at
 * the supplier's spot (`ShippingConfig.spotOffsetPct`) until the next upload.
 *
 * Kept pure so the bounds — the part that catches a typo — are testable.
 */
 
import { TROY_OZ } from "@/lib/utils";
 
/** Bounds match the DB check on `supplier_cost_config.spot_offset_pct`. A value
 *  outside them is almost certainly a typo (a per-gram price typed into the per-oz
 *  field reads as ≈ −97%), and silently storing it would misprice every quote. */
export const SPOT_OFFSET_MIN_PCT = -5;
export const SPOT_OFFSET_MAX_PCT = 10;
 
export type SpotOffsetResult =
  | { ok: true; supplierSpotSgdPerOz: number; ourSpotSgdPerOz: number; offsetPct: number }
  | { ok: false; error: string };
 
/** Parse the optional form value. Empty → null (no reading); junk → error. */
export function parseSupplierSpotInput(raw: unknown): { value: number | null; error?: string } {
  if (raw == null) return { value: null };
  const s = String(raw).replace(/[,\s]|S\$/gi, "").trim();
  if (s === "") return { value: null };
  const n = Number(s);
  if (!Number.isFinite(n) || n <= 0) return { value: null, error: "Supplier spot must be a positive number (S$ per oz)" };
  return { value: n };
}

}
 
/** How far the operator's displayed "our spot" may sit from a fresh server read
 *  before the upload is refused. Gold moves ~0.1%/min at its busiest, so this
 *  admits a few minutes between reading the two spots and picking the file,
 *  while refusing a page left open for hours (or a tampered value). */
export const OUR_SPOT_TOLERANCE_PCT = 0.5;
 
/**
 * Pick the "our spot" to measure the supplier offset against (2026-10-08).
 *
 * The operator reads the supplier's spot off its portal and our live spot off
 * the upload card at the same moment; that displayed figure is the right one to
 * compare against, because a server read seconds or minutes later sees a market
 * that has moved (~0.1% during New York hours — the size of the offset itself).
 * The displayed figure is only trusted when it is close to a fresh server read.
 * No displayed figure (an older client) → the server read is used, as before.
 */
export function reconcileOurSpot(
  displayedSgdPerOz: number | null,
  serverSgdPerGram: number,
): { ok: true; ourSpotSgdPerGram: number; source: "displayed" | "server" } | { ok: false; error: string } {
  if (!(serverSgdPerGram > 0)) return { ok: false, error: "Our live spot is unavailable — try again in a minute" };
  if (displayedSgdPerOz == null) return { ok: true, ourSpotSgdPerGram: serverSgdPerGram, source: "server" };
  const serverOz = serverSgdPerGram * TROY_OZ;
  const driftPct = (displayedSgdPerOz / serverOz - 1) * 100;
  if (!Number.isFinite(driftPct) || Math.abs(driftPct) > OUR_SPOT_TOLERANCE_PCT) {
    return {
      ok: false,
      error: `Our spot was S$${displayedSgdPerOz.toFixed(2)}/oz when you entered the supplier's spot and is S$${serverOz.toFixed(2)}/oz now. Re-read the supplier's spot and enter it again.`,
    };
  }
  return { ok: true, ourSpotSgdPerGram: displayedSgdPerOz / TROY_OZ, source: "displayed" };
}
