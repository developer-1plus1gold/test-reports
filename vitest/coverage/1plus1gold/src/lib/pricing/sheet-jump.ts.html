/**
 * Premium-jump guard for supplier price-sheet uploads (Mukesh 2026-11-18).
 *
 * WHY: the 2026-11-17 StoneX catalogue came in with every PAMP premium up 40–180 % overnight
 * (1 g S$22.23 → S$35.42), its volume tiers gone and Fortuna = Rose to the cent, while no competitor
 * (BullionKing's own shop included) moved. The upload saved it without comment, so it became the
 * newest sheet: it moved our restock cost and the StoneX-indexed BullionKing premium at once, and
 * nobody was asked whether the download was the right view of the catalogue.
 *
 * WHAT: compare a new sheet's per-piece S$ premiums with the supplier's previous sheet, line by line
 * (same product name + weight). When enough lines move by at least `thresholdPct` either way, the
 * upload route refuses to save and asks the operator to confirm. A real market move is confirmed in
 * one click; a wrong download is caught before it prices anything. A fall is flagged as well as a
 * rise, because a premium that drops by half understates our cost just as dangerously.
 *
 * Only `ask_kind = "dollar"` lines are compared: a percent-of-spot ask (≈100–110) barely moves even
 * when its premium doubles, so the same threshold would never fire on it.
 */
 
export interface JumpComparableLine {
  product_name: string | null;
  weight_grams: number | null;
  ask_kind: string | null;
  ask_value: number | null;
  price_tiers?: ReadonlyArray<unknown> | null;
}
 
export interface JumpMovedLine {
  product_name: string;
  prev: number;
  next: number;
  changePct: number;
}
 
export interface PremiumJump {
  /** Lines priced on both sheets. */
  matched: number;
  /** Lines that moved by at least the threshold, largest move first. */
  moved: JumpMovedLine[];
  /** Median change across all matched lines (%). */
  medianChangePct: number;
  /** Lines that had a quantity ladder before and have a single price (or none) now. */
  tiersLost: number;
}
 
export interface JumpOptions {
  /** A line counts as moved at |change| ≥ this (%). */
  thresholdPct?: number;
  /** At least this many moved lines … */
  minMoved?: number;
  /** … and at least this share of the matched lines. */
  minShare?: number;
}
 
export const JUMP_DEFAULTS: Required<JumpOptions> = { thresholdPct: 30, minMoved: 3, minShare: 0.2 };
 
const keyOf = (l: JumpComparableLine): string | null => {
  const name = (l.product_name ?? "").toLowerCase().replace(/[®™]/g, "").replace(/[^a-z0-9]+/g, " ").trim();
  return name ? `${name}|${l.weight_grams ?? ""}` : null;
};
 
const priced = (l: JumpComparableLine): l is JumpComparableLine & { ask_value: number } =>
  l.ask_kind === "dollar" && typeof l.ask_value === "number" && Number.isFinite(l.ask_value) && l.ask_value > 0;
 
const ladderLength = (l: JumpComparableLine): number => (Array.isArray(l.price_tiers) ? l.price_tiers.length : 0);
 
function median(xs: number[]): number {
  const s = [...xs].sort((a, b) => a - b);
  const m = Math.floor(s.length / 2);
  return s.length % 2 ? s[m] : (s[m - 1] + s[m]) / 2;
}
 
/**
 * Pure: the jump between two sheets, or null when the new sheet is in line with the previous one
 * (or there is too little overlap to judge).
 */
export function detectPremiumJump(
  prev: readonly JumpComparableLine[],
  next: readonly JumpComparableLine[],
  opts: JumpOptions = {},
): PremiumJump | null {
  const o = { ...JUMP_DEFAULTS, ...opts };
  const before = new Map<string, JumpComparableLine & { ask_value: number }>();
  for (const l of prev) {
    const k = keyOf(l);
    if (k && priced(l) && !before.has(k)) before.set(k, l);
  }
 
  const changes: number[] = [];
  const moved: JumpMovedLine[] = [];
  let tiersLost = 0;
  const seen = new Set<string>();
  for (const l of next) {
    const k = keyOf(l);
    if (!k || seen.has(k) || !priced(l)) continue;
    const p = before.get(k);
    if (!p) continue;
    seen.add(k);
    const changePct = ((l.ask_value - p.ask_value) / p.ask_value) * 100;
    changes.push(changePct);
    if (Math.abs(changePct) >= o.thresholdPct) {
      moved.push({ product_name: l.product_name as string, prev: p.ask_value, next: l.ask_value, changePct });
    }
    if (ladderLength(p) > 1 && ladderLength(l) <= 1) tiersLost++;
  }
 
  const matched = changes.length;
  if (matched === 0 || moved.length < o.minMoved || moved.length / matched < o.minShare) return null;
  moved.sort((a, b) => Math.abs(b.changePct) - Math.abs(a.changePct));
  return { matched, moved, medianChangePct: median(changes), tiersLost };
}
 
/** One line for the operator: "41 of 52 premiums moved 30 %+ vs the 2026-11-16 sheet (median +38 %)…". */
export function describeJump(j: PremiumJump, comparedSheetDate: string, thresholdPct = JUMP_DEFAULTS.thresholdPct): string {
  const sign = (n: number) => `${n >= 0 ? "+" : ""}${n.toFixed(0)}%`;
  const examples = j.moved.slice(0, 3)
    .map((m) => `${m.product_name} S$${m.prev.toFixed(2)} → S$${m.next.toFixed(2)} (${sign(m.changePct)})`)
    .join("; ");
  return `${j.moved.length} of ${j.matched} premiums moved ${thresholdPct}%+ against the ${comparedSheetDate} sheet ` +
    `(median ${sign(j.medianChangePct)})` +
    (j.tiersLost ? `, and ${j.tiersLost} lost their quantity tiers` : "") +
    `. For example: ${examples}.`;
}
