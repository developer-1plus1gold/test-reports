/**
 * Supplier-agnostic landed-cost resolver — the spine of the Supplier & Pricing
 * hub (docs/supplier-pricing-hub-redesign.md).
 *
 * Every screen that needs "what does this supplier cost us for this product?"
 * calls `resolveLandedCost` and dispatches here, instead of hardcoding the MTB
 * sheet path. The branches REUSE the existing money kernels verbatim —
 * `computeLandedCost` (sheet) and `computeConsignmentCost` (formula) — so the
 * resolver can never produce a number that drifts from the pricing pages.
 *
 * MODELS (redesign 2026-07-25 — collapsing to two + an orthogonal flag):
 *   sheet   — a parsed daily-sheet line + our landed shipping/FX.
 *   manual  — a per-product term you enter, either:
 *               • dollar_per_piece: (spot/g + spread/g) × grams + premium/piece
 *               • pct_over_spot:    spot/g × (1 + premium%), quoted already-landed
 *             `is_consignment_source` (inventory + procure-on-sale) is an
 *             ORTHOGONAL supplier flag, not a pricing model — a consignment
 *             supplier is a `manual` supplier with a `dollar_per_piece` term.
 *
 * PURE — no DB, no clock, no spot fetch. Callers resolve the supplier's model +
 * inputs and the live spot, then pass them in.
 */
 
import {
  computeLandedCost,
  type SupplierLineInput,
  type ShippingConfig,
  type SpotContext,
} from "@/lib/pricing/landed-cost";
import { computeConsignmentCost } from "@/lib/consignment/cost";
 
export type SupplierPricingModel = "sheet" | "manual";
 
/**
 * A manual supplier's per-product cost term. Discriminated on `kind` so a $ term
 * and a % term can't cross-contaminate. Mirrors `supplier_product_terms`.
 */
export type ManualTerm =
  | { kind: "dollar_per_piece"; premiumSgd: number; spreadPerGSgd: number }
  | { kind: "pct_over_spot"; premiumPct: number };
 
/** Live pricing context for ONE piece of a product. */
export interface LandedContext {
  /** XAU/USD + USD→SGD — the sheet path's USD arithmetic runs off this. */
  spot: SpotContext;
  /** Live spot SGD per gram — the manual (dollar/piece + %-over-spot) arithmetic runs off this. */
  spotSgdPerGram: number;
  /** Grams of metal in ONE piece. */
  grams: number;
}
 
/**
 * The per-model inputs. A discriminated union so each model carries exactly what
 * it needs and nothing it doesn't — no bag of optional fields that silently
 * cross-contaminate.
 */
export type SupplierPricing =
  | { model: "sheet"; line: SupplierLineInput; shipping: ShippingConfig; gramsOrdered?: number }
  | { model: "manual"; term: ManualTerm };
 
export interface ResolvedLandedCost {
  model: SupplierPricingModel;
  /** The cost the NEXT sale incurs, per gram, in SGD, LANDED (all-in). */
  landedPerGramSgd: number;
  /** landedPerGramSgd × grams. */
  landedPerPieceSgd: number;
  /** Supplier's own price/g BEFORE our-side shipping + FX markup. For the sheet
   *  model this is the raw metal+ask at the live mid rate; for the manual model
   *  the supplier price IS the landed price, so this equals landedPerGramSgd.
   *  Used by the cost-history snapshot (landed − supplier = our add-on). */
  supplierPerGramSgd: number;
  /** Landed cost per piece EXCLUDING the flat (shipment) shipping + wire fee —
   *  composition-independent. The Quote Builder adds the by-qty freight share on
   *  top. For the manual model there is no separable flat freight, so this equals
   *  landedPerPieceSgd and `flatFreightPerShipmentSgd` is 0. */
  costExFlatFreightPerPieceSgd: number;
  /** The supplier's flat shipping + wire fee for ONE shipment, in SGD (at the
   *  FX-marked rate). Spread by qty across the quote's pieces. 0 for manual. */
  flatFreightPerShipmentSgd: number;
}
 
const finitePos = (n: number): boolean => Number.isFinite(n) && n > 0;
 
/** Per-gram + per-piece landed for a fixed $-premium term (consignment math).
 *  `(spot/g + spread/g) × grams + premium/piece`. Returns null on bad inputs. */
function dollarPerPieceCost(
  spotSgdPerGram: number,
  spreadPerGSgd: number,
  premiumSgd: number,
  grams: number,
): { landedPerGramSgd: number; landedPerPieceSgd: number } | null {
  if (!finitePos(spotSgdPerGram)) return null;
  // Reuse the ONE consignment formula — spread scales with weight, premium does
  // not. Per gram = unit cost ÷ grams (matches strategy-data + margin monitor).
  const cc = computeConsignmentCost({
    spotPerGSgd: spotSgdPerGram,
    spreadPerGSgd,
    premiumSgd,
    gramsEach: grams,
    quantity: 1,
  });
  return { landedPerGramSgd: cc.unitCostSgd / grams, landedPerPieceSgd: cc.unitCostSgd };
}
 
/** Per-gram + per-piece landed for a %-over-spot term: `spot/g × (1 + pct/100)`,
 *  quoted already-landed. Returns null on a non-finite / negative premium. */
function pctOverSpotCost(
  spotSgdPerGram: number,
  premiumPct: number,
  grams: number,
): { landedPerGramSgd: number; landedPerPieceSgd: number } | null {
  if (!finitePos(spotSgdPerGram)) return null;
  if (!Number.isFinite(premiumPct) || premiumPct < 0) return null;
  const perG = spotSgdPerGram * (1 + premiumPct / 100);
  return { landedPerGramSgd: perG, landedPerPieceSgd: perG * grams };
}
 
/**
 * Resolve a supplier's landed cost for one product piece. Returns null when the
 * inputs can't yield a cost (e.g. a sheet line with no usable ask, or a
 * non-positive spot/grams) — callers treat null as "no cost basis from this
 * supplier" rather than guessing a number.
 */
export function resolveLandedCost(
  pricing: SupplierPricing,
  ctx: LandedContext,
): ResolvedLandedCost | null {
  if (!finitePos(ctx.grams)) return null;
 
  switch (pricing.model) {
    case "sheet": {
      // Reuse the sheet kernel exactly — grams come from the matched line.
      const lc = computeLandedCost(pricing.line, ctx.spot, pricing.shipping, pricing.gramsOrdered);
      if (!lc) return null;
      const g = pricing.line.grams;
      Iif (!finitePos(g)) return null;
      return {
        model: "sheet",
        landedPerGramSgd: lc.landedPerGramSgd,
        landedPerPieceSgd: lc.landedPerPieceSgd,
        supplierPerGramSgd: lc.itemCostSgd / g,
        costExFlatFreightPerPieceSgd: lc.costExFlatFreightPerPieceSgd,
        flatFreightPerShipmentSgd: lc.flatFreightPerShipmentSgd,
      };
    }
 
    case "manual": {
      // A per-product term you enter — the go-forward path for every non-sheet
      // supplier (consignment or not). Both kinds are quoted already-landed, so
      // supplierPerGramSgd == landedPerGramSgd.
      const c =
        pricing.term.kind === "dollar_per_piece"
          ? dollarPerPieceCost(ctx.spotSgdPerGram, pricing.term.spreadPerGSgd, pricing.term.premiumSgd, ctx.grams)
          : pctOverSpotCost(ctx.spotSgdPerGram, pricing.term.premiumPct, ctx.grams);
      if (!c) return null;
      return {
        model: "manual",
        landedPerGramSgd: c.landedPerGramSgd,
        landedPerPieceSgd: c.landedPerPieceSgd,
        supplierPerGramSgd: c.landedPerGramSgd,
        // Manual terms are quoted already-landed — no separable shipment freight.
        costExFlatFreightPerPieceSgd: c.landedPerPieceSgd,
        flatFreightPerShipmentSgd: 0,
      };
    }
  }
}
