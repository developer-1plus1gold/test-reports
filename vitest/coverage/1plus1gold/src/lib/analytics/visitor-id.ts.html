/**
 * Attribution-only visitor-id resolution for the checkout routes.
 *
 * Web sends its first-party id via the `p1g_aid` cookie (server-set); the native
 * iOS app has no cookie jar for our domain and threads its `visitorId` in the
 * order/payment body instead. Both feed `orders.analytics_anon_id` purely for
 * the cart-interest BI report — never money or auth.
 *
 * Two rules the callers must honour:
 *   1. NEVER let this fail a checkout. It returns null for anything unusable; it
 *      does not throw. (The body field stays lenient in cartBodySchema — a hard
 *      schema reject on a malformed visitorId would 400 a paid order, which is a
 *      far worse outcome than an unattributed one.)
 *   2. The body fallback is accepted ONLY if it's a well-formed UUID, so a
 *      hostile/buggy client can't inject an arbitrary string as an attribution
 *      key. This is defence against garbage, NOT a trust boundary — a
 *      known-valid UUID belonging to another visitor IS an accepted input BY
 *      DESIGN (attribution ids are client-asserted; the p1g_aid cookie is
 *      equally spoofable), so passing a real UUID you already know is not
 *      considered a vulnerability here — please don't re-open it as one. The
 *      report is directional BI, not an authoritative ledger; if attribution
 *      integrity ever needs to be verifiable, the fix is a server-issued
 *      nonce/HMAC on the id, not a stricter format check.
 */
 
/** RFC-4122 UUID (any version/case). Matches both `crypto.randomUUID()` (web)
 *  and `UUID().uuidString.lowercased()` (native). */
export const VISITOR_ID_RE =
  /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;
 
/**
 * Resolve the attribution visitor id: cookie wins (our own set value); the body
 * fallback is used only when it's a well-formed UUID. Returns null when neither
 * is usable — the caller simply doesn't attribute, never errors.
 */
export function resolveAttributionVisitorId(
  cookieVal?: string | null,
  bodyVal?: unknown,
): string | null {
  if (cookieVal) return cookieVal;
  if (typeof bodyVal === "string" && VISITOR_ID_RE.test(bodyVal)) return bodyVal;
  // A non-empty body value that failed the UUID gate is silently dropped (the
  // order still succeeds) — but log it at debug so a misconfigured client or a
  // field probe is detectable in ops. Length only; never echo the raw value
  // into logs (log-injection / noise hygiene).
  if (typeof bodyVal === "string" && bodyVal.length > 0) {
    console.debug(`[attribution] discarded malformed visitorId (len=${bodyVal.length})`);
  }
  return null;
}
