/**
 * Sub-source: the platform WITHIN a channel — which Social (instagram/tiktok/…),
 * which search engine, which AI engine, paid_search vs paid_social. Lets the
 * dashboard divide "Social" (and the rest) into the specific platforms without a
 * second classifier drifting from resolveChannel. null = no meaningful split
 * (e.g. Direct). Prefers the referrer host, falls back to utm_source.
 */
export function resolveSubSource(
  channel: Channel,
  input: { referrerHost?: string | null; utm?: Utm; adPlatform?: AdPlatform | null },
): string | null {
  const host = (input.referrerHost ?? "").toLowerCase();
  const src  = input.utm?.source?.toLowerCase().trim() || null;
  const ad   = input.adPlatform ?? null;
 
  switch (channel) {
    case "paid":
      if (ad === "google_ads" || ad === "microsoft_ads") return "paid_search";
      if (ad === "tiktok_ads") return "paid_social";
      if ((src && SOCIAL_TOKENS.has(src)) || (host && socialPlatformOfHost(host))) return "paid_social";
      if ((src && SEARCH_TOKENS.has(src)) || (host && searchEngineOfHost(host)))   return "paid_search";
      return "paid_other";
    case "social":
      return socialPlatformOfHost(host) ?? tokenPlatform(src) ?? src ?? "other";
    case "ai":
      return aiEngineOfHost(host) ?? tokenPlatform(src) ?? src ?? "other";
    case "organic":
      return searchEngineOfHost(host) ?? tokenPlatform(src) ?? src ?? "other";
    case "referral":
      return (host || null) ?? src ?? "other";
    case "email":
      return src ?? "email";
    case "direct":
    default:
      return src ?? null;
  }
}
