/**
 * The admin stacking order — one place, so two layers can never tie again.
 *
 * WHY (2026-11-04, Mukesh: "almost all popups on mobile — I can't scroll down to
 * the final button; in landscape the buttons show"): the mobile bottom nav and
 * AdminModal were BOTH z-index 50. On a tie the later element in the document
 * paints on top, and the nav is rendered after <main>, so it covered the bottom
 * ~64px of every bottom-sheet dialog — exactly where its submit button sits.
 * Landscape "fixed" it because a phone on its side is ≥768px wide, where the
 * nav is `md:hidden`. The 2026-09-01 AdminModal fix (five scroll/viewport
 * causes) could not see this: the panel scrolled to its end correctly, and the
 * end was under the nav.
 *
 * Every admin overlay takes its z-index from here. `admin-layers.test.ts`
 * asserts the ORDER (not the numbers) and that no admin file hard-codes a
 * z-index class on a `fixed inset-0` overlay.
 */
export const ADMIN_Z = {
  /** Mobile nav drawer (and its backdrop) — under the bottom nav, by design. */
  sidebarDrawer: 40,
  /** Fixed mobile bottom nav. */
  bottomNav: 50,
  /** Every dialog, sheet and lightbox — must cover the bottom nav. */
  modal: 60,
  /** AdminDialog (confirm/alert) — opens on top of an open modal. */
  dialog: 100,
} as const;
