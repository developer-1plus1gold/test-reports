/**
 * Global network-error monitor.
 *
 * Wraps `window.fetch` ONCE per page load and tracks consecutive
 * network-level failures (thrown errors — not 4xx/5xx HTTP). When the
 * failure count crosses a threshold within a moving window, sets
 * `hasIssue: true` on a Zustand store. The SiteShell renders a
 * `<NetworkIssueBanner />` when this is true.
 *
 * Detects mid-session network problems that arise from:
 *   - Carrier-level content filters (e.g. Singtel Mobile Protect SG
 *     intercepting the TLS handshake when blocking a domain) —
 *     primary motivation for this monitor (2026-05-27)
 *   - Lost WiFi / cellular signal mid-session
 *   - Backend outage where every request 5xx-throws or DNS-fails
 *
 * Browsers do NOT expose the exact reason for a fetch failure (by
 * design — would let attackers fingerprint blocked sites). So we
 * can't tell TLS-MITM from DNS failure from offline. The banner copy
 * lists both common causes and asks the user to try WiFi.
 *
 * Does NOT modify fetch BEHAVIOUR — pass-through wrapper. The real
 * fetch's promise is returned unchanged; we just observe.
 */
import { create } from "zustand";
 
interface NetworkState {
  /** True when ≥THRESHOLD consecutive fetch errors in the last WINDOW_MS. */
  hasIssue: boolean;
  /** Most recent failed URL (string) — for logs only, not displayed. */
  lastFailedUrl: string | null;
  /** Internal: timestamps of recent failures. */
  _recentFailures: number[];
  /** Called by the fetch wrapper after a throw. */
  _recordFailure: (url: string) => void;
  /** Called by the fetch wrapper after a successful fetch. */
  _recordSuccess: () => void;
  /** Manual dismiss — banner stays hidden until next failure burst. */
  dismiss: () => void;
}
 
const WINDOW_MS = 30_000;     // 30s rolling window
const THRESHOLD = 2;          // ≥2 throws within window → flag
const DISMISS_GRACE_MS = 60_000; // After dismiss, suppress for 60s
 
let lastDismissAt = 0;
 
export const useNetworkStore = create<NetworkState>((set, get) => ({
  hasIssue: false,
  lastFailedUrl: null,
  _recentFailures: [],
 
  _recordFailure: (url) => {
    const now = Date.now();
    if (now - lastDismissAt < DISMISS_GRACE_MS) return; // user dismissed recently
    const recent = [...get()._recentFailures, now].filter(t => now - t < WINDOW_MS);
    const shouldFlag = recent.length >= THRESHOLD;
    set({
      _recentFailures: recent,
      lastFailedUrl:   url,
      hasIssue:        shouldFlag || get().hasIssue,
    });
  },
 
  _recordSuccess: () => {
    // Clear the issue flag on the first successful fetch after it was set.
    // Keeps the recent-failures window for diagnostic accuracy on subsequent failures.
    if (get().hasIssue) {
      set({ hasIssue: false, _recentFailures: [] });
    }
  },
 
  dismiss: () => {
    lastDismissAt = Date.now();
    set({ hasIssue: false, _recentFailures: [] });
  },
}));
 
/**
 * Install the global fetch wrapper ONCE per page load. Idempotent —
 * stores a marker on window so HMR / double-import doesn't double-wrap
 * (which would call the original fetch twice per call).
 */
declare global {
  interface Window {
    __1p1g_network_monitor_installed?: boolean;
  }
}
 
export function installNetworkMonitor(): void {
  if (typeof window === "undefined") return;
  if (window.__1p1g_network_monitor_installed) return;
  window.__1p1g_network_monitor_installed = true;
 
  const originalFetch = window.fetch.bind(window);
 
  window.fetch = async function wrappedFetch(...args: Parameters<typeof fetch>): Promise<Response> {
    const url = typeof args[0] === "string"
      ? args[0]
      : args[0] instanceof URL
        ? args[0].toString()
        : (args[0] as Request).url;
    try {
      const res = await originalFetch(...args);
      // Network-level success — even a 5xx counts as "we reached the server".
      // The banner is for "can't reach server at all" (TLS / DNS / offline),
      // NOT for server-side errors which have their own UI.
      useNetworkStore.getState()._recordSuccess();
      return res;
    } catch (err) {
      // Skip recording for intentional aborts — those aren't network problems.
      const name = (err as Error)?.name;
      if (name !== "AbortError") {
        useNetworkStore.getState()._recordFailure(url);
      }
      throw err;
    }
  };
}
