/**
 * Delivery P&L simulator — pure engine (no I/O).
 *
 * Ported from Mukesh's prototype (docs/pl-simulator-design.md) with the SAME
 * field names and units so the two can be compared line by line. Units:
 *   - money in S$ unless a field name says otherwise (lPick / lXferFee /
 *     lStoreRate / lStoreMin are US$ and are converted with `fx`)
 *   - fields named pPaid / pSaver / sxShare / failRate / payAbs / gst / prem
 *     are PERCENT (9 means 9%)
 *
 * One deliberate change from the prototype: a bar carries `costPct` (landed
 * cost as % over spot) instead of a fixed margin. Gold profit is then
 *   sale − cost = spot·g·(1+prem/100) − spot·g·(1+costPct/100)
 * so changing the premium moves the margin, and changing spot moves cost
 * with it. `costPctFromMargin` converts a brief margin into the same model,
 * which reproduces the prototype's base case exactly.
 *
 * Read-only by construction: nothing here writes anything. The page owns state.
 */

export function compute(st: SimState, override?: Partial<SimState>): SimResult {

  const scale = s.scale ?? 1;
  const paid = s.pPaid / 100;
  const saver = s.pSaver / 100;
  const free = Math.max(0, 1 - paid - saver);
  const g = s.gst / 100;
  const pay = s.payAbs / 100;
 
  const bars0 = s.bars.map((b) => {
    const sale = s.spot * b.g * (1 + b.prem / 100);
    const cost = s.spot * b.g * (1 + b.costPct / 100);
    return { ...b, sale, cost, gp: sale - cost };
  });
  const ppd = bars0.reduce((a, b) => a + b.ppd, 0);
  const dayFactor = scale * s.delDays * WPM;
  const parcelsM = ppd * dayFactor;
  const colM = s.colDays * WPM;
 
  const delivAvg = (free + paid) * s.aEve + saver * s.aSaver;
  const failAvg = (s.failRate / 100) * s.aRe;
  const airVarPP = s.aPack + delivAvg + failAvg;
  const feePP = paid * s.paidFee + saver * (s.saverFee ?? 0);
 
  const sx = s.sxShare / 100;
  const own = 1 - sx;
  const picksPerDay = (sx > 0 ? 1 : 0) + (own > 0 ? 1 : 0);
  const airCol = colM * s.aCol;
  const airFlat = s.aFlat + s.aStore;
  const loomisPicks = colM * picksPerDay * s.lPick * s.fx;
  const ownGramsDay = bars0.reduce((a, b) => a + b.ppd * scale * b.g, 0) * own * (s.delDays / 7);
  const stockKg = (ownGramsDay * s.lStockDays) / 1000;

      feePP,
      fulPP,
      contrib,
      contribPctOfSale: ratio(contrib, b.sale),

      monthly: contrib * b.ppd * dayFactor,
      salesM: barSalesM,
      shareOfSalesPct: ratio(barSalesM, salesM),
      shareOfParcelsPct: ratio(b.ppd, ppd),
      chip: contrib < 0 ? "loses" : contrib < b.gp * 0.4 ? "thin" : "healthy",
    };
  });
 
  return {
    s,
    bars,
    ppd: ppd * scale,

    colM,
    free,
    paid,
    saver,

    feeM,
    payM,

    contribM,

    fixedPP,
    varPP,

    picksPerDay,

/**
 * Premium each bar needs so its contribution per parcel equals `targetPP`
 * (0 = break even). Fulfilment per parcel does not depend on price, so this
 * is closed-form. Read-only: it never changes anything.
 */
export function requiredPremiums(r: SimResult, targetPP: number): RequiredPremium[] {
  const s = r.s;
  const pay = s.payAbs / 100;
  return r.bars.map((b) => {
    const spotG = s.spot * b.g;
    const saleNeeded = (targetPP + spotG * (1 + b.costPct / 100) - r.feePP * (1 - pay) + r.fulPP) / (1 - pay);
    const requiredPrem = (saleNeeded / spotG - 1) * 100;
    const m = marginPct(requiredPrem, b.costPct);
    return {
      id: b.id,
      name: b.name,
      currentPrem: b.prem,
      requiredPrem,
      deltaPts: requiredPrem - b.prem,
      marginAtRequiredPct: m,
      belowFloor: m < RETAIL_MARGIN_FLOOR_PCT,
    };
  });
}

export function sensitivity(st: SimState): SensitivityRow[] {
  const base = compute(st).netM;
  const baseScale = st.scale ?? 1;
  const row = (key: SensitivityRow["key"], label: string, lowLabel: string, highLabel: string, lo: Partial<SimState>, hi: Partial<SimState>): SensitivityRow => {
    const lowNetM = compute(st, lo).netM;
    const highNetM = compute(st, hi).netM;
    const d = (v: number) => (base !== 0 ? (v - base) / Math.abs(base) : null);
    return { key, label, lowLabel, highLabel, lowNetM, highNetM, lowDelta: lowNetM - base, highDelta: highNetM - base, lowDeltaPct: d(lowNetM), highDeltaPct: d(highNetM) };
  };
  return [
    row("spot", "Gold spot", "−10%", "+10%", { spot: st.spot * 0.9 }, { spot: st.spot * 1.1 }),
    row("volume", "Volume", "−50%", "+50%", { scale: baseScale * 0.5 }, { scale: baseScale * 1.5 }),
    row("airpak", "Airpak rates", "−20%", "+20%", scaleAirpak(st, 0.8), scaleAirpak(st, 1.2)),
  ];
}

          ? { paidFee: fee, pPaid: t, pSaver: Math.min(st.pSaver, 100 - t) }
          : { saverFee: fee, pSaver: t, pPaid: Math.min(st.pPaid, 100 - t) };
      const netM = compute(st, o).netM;
      return { fee, takeUpPct: t, netM, deltaM: netM - base, deltaPct: base !== 0 ? (netM - base) / Math.abs(base) : null };
    }),
  );
