/**
 * embed-bridge-validate.ts — input sanitisers for the Calculator chart embeds'
 * `window.__calcSet*` native→WebView bridges.
 *
 * The embeds expose global functions the native SDUI panel calls to push live
 * params (grams / budget / premium / label) without a reload. Because they are
 * GLOBAL `window` properties, any script sharing the WebView origin — not just
 * our native code — can call them. Untrusted numbers/strings must therefore be
 * clamped and sanitised before they reach React state, so a co-tenant or
 * tampered caller can't inject NaN/Infinity, absurd values, or arbitrary text
 * into a customer-facing pricing chart.
 * (security_agent embed-window-bridge-unvalidated-2026, 2026-06-04.)
 */
 
/** Finite, > 0, capped. Returns undefined for anything else. */
export function sanitizeGrams(v: unknown): number | undefined {
  return typeof v === "number" && Number.isFinite(v) && v > 0 && v <= 100_000 ? v : undefined;
}
 
/** Finite, > 0, capped (SGD). Returns undefined for anything else. */
export function sanitizeBudget(v: unknown): number | undefined {
  return typeof v === "number" && Number.isFinite(v) && v > 0 && v <= 1_000_000_000 ? v : undefined;
}
 
/**
 * Finite, >= 0, clamped to [0, 100]. The chart accepts premium either as a
 * fraction (0.08) or a percent (8); 100 is a generous ceiling that rejects
 * NaN/Infinity/negatives/absurd injection without clipping legitimate values
 * in either unit. Returns undefined for anything else.
 */
export function sanitizePremium(v: unknown): number | undefined {
  return typeof v === "number" && Number.isFinite(v) && v >= 0 && v <= 100 ? v : undefined;
}
 
/**
 * Coerce to a safe display string: drop control characters (code < 0x20 or
 * 0x7F DEL), trim, truncate to 40 chars. Returns undefined if empty.
 */
export function sanitizeLabel(v: unknown): string | undefined {
  if (typeof v !== "string") return undefined;
  const clean = Array.from(v)
    .filter((ch) => {
      const c = ch.charCodeAt(0);
      return c >= 0x20 && c !== 0x7f;
    })
    .join("")
    .trim()
    .slice(0, 40);
  return clean.length > 0 ? clean : undefined;
}
