/** Validate an admin edit. Returns the lane or every problem, field by field. */
export function parseLanePatch(raw: unknown): { ok: true; lane: LanePatch } | { ok: false; errors: string[] } {
  const r = lanePatchSchema.safeParse(raw);
  if (r.success) return { ok: true, lane: r.data };
  return { ok: false, errors: r.error.issues.map((i) => `${i.path.join(".") || "lane"}: ${i.message}`) };
}

/**
 * What the lane promises that its carrier, as configured, cannot do. Empty =
 * nothing to warn about. Only SingPost has published booking cut-offs; the
 * on-demand couriers (Lalamove, Grab) and SPX take bookings all day.
 */
export function laneWarnings(
  lane: Pick<LanePatch, "carrier" | "carrier_service" | "enabled" | "days">,
  cutoffs: BookingCutoffs | null,
  /** SingPost's own on/off (Admin → Shippers → SingPost). Omitted = not checked. */
  singpostEnabled?: boolean,
): string[] {
  if (lane.carrier !== "singpost") return [];
  const out: string[] = [];
  if (lane.enabled && singpostEnabled === false) {
    out.push("SingPost is switched off (Admin → Shippers → SingPost), so orders on this lane cannot be booked. Switch SingPost on first, or keep this lane off.");
  }
  const service = lane.carrier_service && cutoffs ? cutoffs[lane.carrier_service] : null;
  if (!service) return out;
  for (const d of [...lane.days].sort((a, b) => a.day_of_week - b.day_of_week)) {
    if (!d.dispatch) continue;
    const day = DAY_NAMES[d.day_of_week];
    const booking = service[d.day_of_week];
    if (!booking) {
      out.push(`${day}: SingPost does not take this service on ${day}s, but the lane dispatches that day.`);
      continue;
    }
    const until = hhmmToMinutes(d.orders_until);
    const cut = hhmmToMinutes(booking);
    if (until == null) {
      out.push(`${day}: orders are taken all day, but SingPost's booking cut-off is ${hm(booking)}.`);
    } else if (cut != null && until >= cut) {
      out.push(`${day}: pay-by ${hm(d.orders_until!)} leaves no time to book before SingPost's ${hm(booking)} cut-off.`);
    }
  }
  return out;
}

/** "Every day" · "Mon–Fri" · "Mon–Sat" · "Mon, Wed, Fri" · "none" (Monday-first runs). */
export function daysLabel(dows: readonly number[]): string {
  const set = new Set(dows);
  if (set.size === 7) return "Every day";
  if (set.size === 0) return "none";
  const order = [1, 2, 3, 4, 5, 6, 0].filter((d) => set.has(d));
  const runs: number[][] = [];
  for (const d of order) {
    const last = runs.at(-1);
    const prev = last?.at(-1);
    if (last && prev != null && [1, 2, 3, 4, 5, 6, 0].indexOf(d) === [1, 2, 3, 4, 5, 6, 0].indexOf(prev) + 1) last.push(d);
    else runs.push([d]);
  }
  return runs.map((r) => (r.length >= 3 ? `${SHORT[r[0]]}–${SHORT[r.at(-1)!]}` : r.map((d) => SHORT[d]).join(", "))).join(", ");
}
 
/**

 * "1 day after dispatch" read as a riddle). One wording, used by the editor's dropdown
 * and the collapsed card alike.
 */
export function deliveryRuleLabel(lane: Pick<LanePatch, "today_only" | "days_after">): string {
  if (lane.today_only) return "Same day only";
  if (lane.days_after <= 0) return "Same day as dispatch";
  if (lane.days_after === 1) return "Next delivery day";

}

/** "15:00" → "3", "15:30" → "3:30"; with the meridiem when asked. */
function clock12(t: string, withMeridiem: boolean): string {
  const [h, m] = t.slice(0, 5).split(":").map(Number);
  const h12 = h % 12 === 0 ? 12 : h % 12;
  const base = m ? `${h12}:${String(m).padStart(2, "0")}` : String(h12);
  return withMeridiem ? `${base} ${h < 12 ? "am" : "pm"}` : base;
}
 
/**
 * The lane's delivery window as a short label for its tile ("3–6 pm",
 * "11 am–2 pm"), when every delivery day shares one; "times vary" when they
 * differ; null when the lane has no window. Lets two lanes with the same name
 * be told apart at a glance (Mukesh 2026-11-22).
 */
export function laneWindowLabel(days: LanePatch["days"]): string | null {
  const wins = [...new Set(days.filter((d) => d.delivers && d.window_start && d.window_end)
    .map((d) => `${d.window_start!.slice(0, 5)}-${d.window_end!.slice(0, 5)}`))];
  if (wins.length === 0) return null;
  if (wins.length > 1) return "times vary";
  const [from, to] = wins[0].split("-");
  const samePeriod = (Number(from.slice(0, 2)) < 12) === (Number(to.slice(0, 2)) < 12);
  return `${clock12(from, !samePeriod)}–${clock12(to, true)}`;
}
