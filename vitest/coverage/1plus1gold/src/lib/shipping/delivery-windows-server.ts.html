export type WindowCheck =
  | { ok: true; start: string | null; end: string | null }
  | { ok: false; status: 400 | 409; error: string };
 
/**
 * Validate an optional customer-chosen window on an order body. Absent → the
 * unscheduled behaviour every order had before (old app builds send none).
 * Present → it must be a slot we still offer, or the order is refused with 409
 * so the customer picks again rather than being promised a closed slot.
 */
export async function checkSubmittedWindow(
  shipperType: string,
  start: unknown,
  end: unknown,
  now: Date = new Date(),
): Promise<WindowCheck> {
  if (start == null && end == null) return { ok: true, start: null, end: null };
  if (shipperType !== "lalamove") return { ok: true, start: null, end: null };
  if (typeof start !== "string" || typeof end !== "string") {
    return { ok: false, status: 400, error: "Invalid delivery window." };
  }
  let offered = false;
  try {
    offered = isWindowOffered(await loadWindowInput("lalamove", now), start, end);
  } catch (err) {
    console.error("[delivery-windows] re-check failed:", err);
    return { ok: false, status: 409, error: "We couldn't confirm your delivery window. Please choose it again." };
  }
  if (!offered) {
    return { ok: false, status: 409, error: "That delivery window is no longer available. Please choose another." };
  }
  return { ok: true, start: new Date(start).toISOString(), end: new Date(end).toISOString() };
}
