/**
 * Loads the next-day evening delivery settings + promise from the DB. Shared by
 * /api/shipping-options (what checkout shows) and the order routes (the
 * re-check), so both see exactly the same promise and the same fee.
 *
 * Unlike assertShipperEnabled (which fails OPEN for live carriers), this fails
 * CLOSED: the evening option is a promise plus a fee the server sets, so if we
 * cannot load the settings we refuse the option rather than guess either.
 */
import { createClient } from "@supabase/supabase-js";
import { checkSubmittedWindow, loadWindowInput } from "@/lib/shipping/delivery-windows-server";
import { netShipperFee, type ShipperFeeRow } from "@/lib/shipping/shipper-fee";
import { eveningPromise, isEveningWindowPromised, type EveningPromise } from "@/lib/shipping/evening";
 
export const EVENING_SHIPPER_KEY = "evening";
export const EVENING_DEFAULT_NAME = "Next-day evening delivery";
 
export interface EveningSettings {
  enabled: boolean;
  displayName: string;
  notes: string | null;
  /** What the customer pays: flat_fee_sgd ± discount/premium (netShipperFee). */
  feeSgd: number;
  grossFee: number;
  discountPct: number;
  premiumPct: number;
}
 
function admin() {
  return createClient(process.env.NEXT_PUBLIC_SUPABASE_URL!, process.env.SUPABASE_SERVICE_KEY!);
}
 
/** Missing row → disabled. The option only exists once the migration seeds it. */
export async function loadEveningSettings(): Promise<EveningSettings> {
  const { data, error } = await admin()
    .from("shipper_settings")
    .select("enabled, display_name, notes, flat_fee_sgd, discount_pct, modifier_type")
    .eq("shipper", EVENING_SHIPPER_KEY)
    .maybeSingle();
  if (error) throw new Error(`evening_settings_query_failed: ${error.message}`);
  const row = data as ({ enabled: boolean | null; display_name: string | null; notes: string | null } & ShipperFeeRow) | null;
  // Same pricing as every other flat-fee carrier — no evening-only fee logic.
  const f = netShipperFee(row, 0);
  return {
    enabled: row?.enabled === true,
    displayName: row?.display_name?.trim() || EVENING_DEFAULT_NAME,
    notes: row?.notes?.trim() || null,
    feeSgd: f.fee,
    grossFee: f.grossFee,
    discountPct: f.discountPct,
    premiumPct: f.premiumPct,
  };
}
 
export interface EveningOption {
  settings: EveningSettings;
  promise: EveningPromise | null;
}
 
/** Never throws: a failed load hides the option. */
export async function fetchEveningOption(now: Date = new Date()): Promise<EveningOption | null> {
  try {
    const [settings, input] = await Promise.all([loadEveningSettings(), loadWindowInput(EVENING_SHIPPER_KEY, now)]);
    return { settings, promise: eveningPromise(input) };
  } catch (err) {
    console.error("[evening] load failed:", err);
    return null;
  }
}

/**
 * Validate an evening choice on an order body. The window must be one we
 * promise right now; the fee is NOT taken from the client — the caller must
 * charge `feeSgd`.
 */
export async function checkEveningChoice(start: unknown, end: unknown, now: Date = new Date()): Promise<EveningCheck> {
  if (typeof start !== "string" || typeof end !== "string") {
    return { ok: false, status: 400, error: "Please choose your evening delivery window." };
  }
  try {
    const [settings, input] = await Promise.all([loadEveningSettings(), loadWindowInput(EVENING_SHIPPER_KEY, now)]);
    if (!settings.enabled) {
      return { ok: false, status: 409, error: "Evening delivery is not currently available. Please choose another shipping option." };
    }
    if (!isEveningWindowPromised(input, start, end)) {
      return { ok: false, status: 409, error: "Your evening delivery date has changed. Please review it and try again." };
    }
    return { ok: true, start: new Date(start).toISOString(), end: new Date(end).toISOString(), feeSgd: settings.feeSgd };
  } catch (err) {
    console.error("[evening] re-check failed:", err);
    return { ok: false, status: 409, error: "We couldn't confirm your evening delivery. Please choose it again." };
  }
}
