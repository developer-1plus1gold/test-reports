 *
 * Loads the selected orders, validates each is eligible for a consolidated
 * Lalamove trip, and geocodes each dropoff's postal code (one OneMap call each,

  paidFee:        number;                       // what this customer paid for shipping
}
 
export interface BatchPrep {
  ready:       PreparedDropoff[];

  serviceType: "MOTORCYCLE" | "SEDAN";
}
 
/** A shape error the route should turn into a 400. */
export interface BatchPrepError { error: string; status: number }
 
export function isBatchPrepError(x: BatchPrep | BatchPrepError): x is BatchPrepError {
  return (x as BatchPrepError).error !== undefined;
}
