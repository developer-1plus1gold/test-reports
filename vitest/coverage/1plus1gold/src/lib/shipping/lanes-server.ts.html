/**
 * Loads the checkout delivery lanes (Admin → Shippers → Delivery lanes) and
 * runs the order-route re-check. Shared by /api/shipping-options and every
 * order route, so the picker and the 409 can never disagree. Pure rules:
 * ./lanes.ts. Design: docs/singpost-integration-plan.md → "Delivery lanes".
 *
 * Fails CLOSED: a lane is a promise plus a fee the server sets, so a settings
 * read that fails offers no lane — never a guess.
 */
import { createClient, type SupabaseClient } from "@supabase/supabase-js";
import { normaliseMinOrder } from "@/lib/shipping/min-order";
import { netShipperFee } from "@/lib/shipping/shipper-fee";
import {
  LANE_CARRIERS, LANE_DISPATCH_KINDS, findPromisedLaneOffer, isLaneKey, laneOffer, offerFitsOrder,
  type LaneCarrier, type LaneConfig, type LaneDay, type LaneDispatchKind, type LaneOffer,
} from "@/lib/shipping/lanes";
import { minOrderMessage } from "@/lib/shipping/min-order";
 
function admin(): SupabaseClient {
  return createClient(process.env.NEXT_PUBLIC_SUPABASE_URL!, process.env.SUPABASE_SERVICE_KEY!);
}
 
interface LaneRow {
  key: string; display_name: string; notes: string | null; enabled: boolean; sort_order: number;
  today_only: boolean; days_after: number; observes_holidays: boolean;
  flat_fee_sgd: number | string; modifier_type: string; discount_pct: number | string;
  min_order_sgd: number | string | null; max_order_sgd: number | string | null;
  carrier: string; carrier_service: string | null; dispatch_kind: string | null; show_carrier_logo: boolean | null;
}
interface LaneDayRow {
  lane_key: string; day_of_week: number; dispatch: boolean; orders_until: string | null;
  delivers: boolean; window_start: string | null; window_end: string | null;
}
 
export const LANE_SELECT =
  "key, display_name, notes, enabled, sort_order, today_only, days_after, observes_holidays, flat_fee_sgd, modifier_type, discount_pct, min_order_sgd, max_order_sgd, carrier, carrier_service, dispatch_kind, show_carrier_logo";
export const LANE_DAY_SELECT = "lane_key, day_of_week, dispatch, orders_until, delivers, window_start, window_end";
 
/** Rows → engine configs. Pure; a row with an unknown carrier is dropped (never offered). */
export function toLaneConfigs(rows: readonly LaneRow[], dayRows: readonly LaneDayRow[]): LaneConfig[] {
  const byLane = new Map<string, Partial<Record<number, LaneDay>>>();
  for (const d of dayRows) {
    const m = byLane.get(d.lane_key) ?? {};
    m[d.day_of_week] = {
      dispatch: d.dispatch === true, ordersUntil: d.orders_until, delivers: d.delivers === true,
      windowStart: d.window_start, windowEnd: d.window_end,
    };
    byLane.set(d.lane_key, m);
  }
  return rows
    .filter((r) => (LANE_CARRIERS as readonly string[]).includes(r.carrier))
    .map((r) => {
      const f = netShipperFee({ flat_fee_sgd: r.flat_fee_sgd, discount_pct: r.discount_pct, modifier_type: r.modifier_type }, 0);
      const max = Number(r.max_order_sgd);
      return {
        key: r.key,
        enabled: r.enabled === true,
        displayName: r.display_name.trim(),
        notes: r.notes?.trim() || null,
        sortOrder: Number(r.sort_order) || 0,
        todayOnly: r.today_only === true,
        daysAfter: Number(r.days_after) || 0,
        days: byLane.get(r.key) ?? {},
        observesHolidays: r.observes_holidays === true,
        fee: { fee: f.fee, grossFee: f.grossFee, discountPct: f.discountPct, premiumPct: f.premiumPct },
        minOrderSgd: normaliseMinOrder(r.min_order_sgd),
        maxOrderSgd: r.max_order_sgd != null && Number.isFinite(max) && max > 0 ? max : null,
        carrier: r.carrier as LaneCarrier,
        carrierService: r.carrier_service,
        dispatchKind: (LANE_DISPATCH_KINDS as readonly string[]).includes(r.dispatch_kind ?? "") ? r.dispatch_kind as LaneDispatchKind : null,
        showCarrierLogo: r.show_carrier_logo === true,
      };
    });
}
 
export interface LaneSettings { lanes: LaneConfig[]; holidays: Set<string> }
 
/** Every lane (on and off) + the holiday list. Throws on a read error. */
export async function loadLaneSettings(db: SupabaseClient = admin()): Promise<LaneSettings> {
  const [l, d, h] = await Promise.all([
    db.from("delivery_lanes").select(LANE_SELECT).order("sort_order"),
    db.from("delivery_lane_days").select(LANE_DAY_SELECT),
    db.from("delivery_holidays").select("holiday_date"),
  ]);
  const err = l.error ?? d.error ?? h.error;
  if (err) throw new Error(`delivery_lanes_query_failed: ${err.message}`);
  return {
    lanes: toLaneConfigs((l.data ?? []) as LaneRow[], (d.data ?? []) as LaneDayRow[]),
    holidays: new Set(((h.data ?? []) as Array<{ holiday_date: string }>).map((x) => String(x.holiday_date).slice(0, 10))),
  };
}
 
/** Same as loadLaneSettings, but an unreadable table offers no lanes (logged) — the checkout still loads. */
export async function loadLaneSettingsOrEmpty(): Promise<LaneSettings> {
  try {
    return await loadLaneSettings();
  } catch (err) {
    console.error("[lanes] settings load failed:", err);
    return { lanes: [], holidays: new Set() };
  }
}
 
/**
 * The lane an order's delivery belongs to: its own `delivery_lane`, else (an
 * order from before lanes) the "next day" lane of its Dispatch section
 * (delivery_option afternoon / evening). null = no lane (Express, other carriers).
 */
export function laneForOrder(lanes: readonly LaneConfig[], o: { delivery_lane?: string | null; delivery_option?: string | null }): LaneConfig | null {
  const own = o.delivery_lane ? lanes.find((l) => l.key === o.delivery_lane) : undefined;
  if (own) return own;
  if (!o.delivery_option) return null;
  const same = lanes.filter((l) => l.dispatchKind === o.delivery_option);
  return same.find((l) => !l.todayOnly) ?? same[0] ?? null;
}

export interface LaneChoice {
  key: string;
  carrier: LaneCarrier;
  carrierService: string | null;
  /** = orders.delivery_option: the Dispatch section (Lalamove/Grab lanes), else null. */
  dispatchKind: LaneDispatchKind | null;
  /** The fee the order MUST carry (never the client's). */
  feeSgd: number;
  /** SGT YYYY-MM-DD promised. */
  date: string;
  start: string | null;
  end: string | null;
  minOrderSgd: number | null;
  maxOrderSgd: number | null;
}
 
export type LaneCheck = { ok: true; lane: LaneChoice } | { ok: false; status: 400 | 409; error: string };
 
export const LANE_GONE_MESSAGE = "That delivery option is no longer available. Please review your shipping choice and try again.";
 
function choiceOf(cfg: LaneConfig, o: LaneOffer): LaneChoice {
  return {
    key: cfg.key, carrier: cfg.carrier, carrierService: cfg.carrierService, dispatchKind: cfg.dispatchKind, feeSgd: o.fee,
    date: o.date, start: o.start, end: o.end, minOrderSgd: cfg.minOrderSgd, maxOrderSgd: cfg.maxOrderSgd,
  };
}
 
/**
 * Validate a lane choice: the lane must be on and its offer now (or 15 minutes
 * ago) must be for `date`. The caller charges `lane.feeSgd`.
 */
export async function checkLaneChoice(laneKey: unknown, date: unknown, now: Date = new Date()): Promise<LaneCheck> {
  if (!isLaneKey(laneKey) || typeof date !== "string" || !/^\d{4}-\d{2}-\d{2}$/.test(date)) {
    return { ok: false, status: 400, error: "Please choose a delivery option." };
  }
  let s: LaneSettings;
  try {
    s = await loadLaneSettings();
  } catch (err) {
    console.error("[lanes] re-check failed:", err);
    return { ok: false, status: 409, error: "We couldn't confirm your delivery option. Please choose it again." };
  }
  const cfg = s.lanes.find((c) => c.key === laneKey);
  const offer = cfg?.enabled ? findPromisedLaneOffer(cfg, now, date, s.holidays) : null;
  if (!cfg || !offer) return { ok: false, status: 409, error: LANE_GONE_MESSAGE };
  return { ok: true, lane: choiceOf(cfg, offer) };
}
 
/**
 * Installed app builds (before the lane picker) send the carrier, not the lane:
 * "spx", or "evening" + a window. Map them onto the lane that makes the same
 * promise right now: the first enabled SPX lane, or the enabled lane whose
 * offer has exactly that window. null = no lane matches (the caller refuses).
 */
export async function legacyLaneChoice(raw: "spx" | "evening", start: unknown, end: unknown, now: Date = new Date()): Promise<LaneCheck | null> {
  let s: LaneSettings;
  try {
    s = await loadLaneSettings();
  } catch (err) {
    console.error("[lanes] legacy re-check failed:", err);
    return { ok: false, status: 409, error: "We couldn't confirm your delivery option. Please choose it again." };
  }
  const earlier = new Date(now.getTime() - 15 * 60_000);
  for (const cfg of s.lanes) {
    if (!cfg.enabled) continue;
    const offers = [laneOffer(cfg, now, s.holidays), laneOffer(cfg, earlier, s.holidays)].filter((o): o is LaneOffer => o !== null);
    const match = raw === "spx"
      ? (cfg.carrier === "spx" ? offers[0] : undefined)
      : offers.find((o) => o.start != null && o.end != null && typeof start === "string" && typeof end === "string"
          && Date.parse(o.start) === Date.parse(start) && Date.parse(o.end) === Date.parse(end));
    if (match) return { ok: true, lane: choiceOf(cfg, match) };
  }
  return null;
}
 
/** The lane's minimum and maximum against the order value (gold + accessories before discounts). */
export function checkLaneOrderValue(lane: Pick<LaneChoice, "minOrderSgd" | "maxOrderSgd">, orderValueSgd: number):
  { ok: true } | { ok: false; status: 409; error: string } {
  if (offerFitsOrder(lane, orderValueSgd)) return { ok: true };
  if (lane.maxOrderSgd != null && orderValueSgd > lane.maxOrderSgd) {
    return { ok: false, status: 409, error: `This delivery option is available on orders up to S$${lane.maxOrderSgd.toFixed(2)}. Please choose another.` };
  }
  return { ok: false, status: 409, error: minOrderMessage(lane.minOrderSgd as number) };
}
 
/**
 * The lane a pending payment row carries (orderData.lane, written by our own
 * gateway-create). Re-validated for shape — a malformed row creates the order
 * without a lane rather than with a guessed one.
 */
export function parseStoredLane(v: unknown): LaneChoice | null {
  if (!v || typeof v !== "object") return null;
  const l = v as Record<string, unknown>;
  const fee = Number(l.feeSgd);
  const num = (x: unknown) => (x == null ? null : Number.isFinite(Number(x)) ? Number(x) : null);
  if (!isLaneKey(l.key) || !(LANE_CARRIERS as readonly string[]).includes(String(l.carrier))
    || !Number.isFinite(fee) || fee < 0 || typeof l.date !== "string" || !/^\d{4}-\d{2}-\d{2}$/.test(l.date)) return null;
  const iso = (x: unknown) => (typeof x === "string" && !Number.isNaN(Date.parse(x)) ? x : null);
  return {
    key: l.key, carrier: l.carrier as LaneCarrier, carrierService: typeof l.carrierService === "string" ? l.carrierService : null,
    dispatchKind: (LANE_DISPATCH_KINDS as readonly string[]).includes(String(l.dispatchKind)) ? l.dispatchKind as LaneDispatchKind : null,
    feeSgd: fee, date: l.date, start: iso(l.start), end: iso(l.end), minOrderSgd: num(l.minOrderSgd), maxOrderSgd: num(l.maxOrderSgd),
  };
}
 
/** The orders columns a lane order is created with (shipper = the lane's default carrier). */
export function laneOrderFields(lane: LaneChoice, deliveryAddress: unknown) {
  return {
    shipper:                lane.carrier,
    shipper_service:        lane.carrier === "singpost" ? lane.carrierService
                          : lane.carrier === "spx" ? "NEXT_DAY"
                          : lane.carrier === "lalamove" ? "MOTORCYCLE" : null,
    shipper_status:         "pending",
    shipping_fee_sgd:       lane.feeSgd,
    delivery_address:       deliveryAddress,
    delivery_window_start:  lane.start,
    delivery_window_end:    lane.end,
    // Dispatch section (Afternoon / Evening) for a Lalamove/Grab lane — what every
    // downstream reader (Dispatch, receipts, reschedule, reminders) keys on.
    delivery_option:        lane.dispatchKind,
    delivery_lane:          lane.key,
    promised_delivery_date: lane.date,
  };
}
