import { HANDOVER_INSTRUCTION } from "@/lib/shipping/handover";
 
/**
 * GrabExpress API — pure rules (2026-11-07). No server imports: the Dispatch
 * screen and the tests read these too.
 *
 * Contract: Grab's official docs, https://developer.grab.com/docs/grab-express/
 * (summary + the choices we made: docs/grab-express-api-reference.md).
 */
 
/** Grab's delivery states → our `orders.shipper_status` vocabulary (the one
 *  Lalamove and SPX already use, so every Dispatch rule keeps working).
 *  Grab spells the cancelled state two ways: CANCELED in Get Delivery Details,
 *  CANCELLED in the Tracking Webhook. Both are mapped. */
const STATUS_MAP: Readonly<Record<string, string>> = {
  QUEUEING:         "booked",            // scheduled order, waiting for its pickup window
  ALLOCATING:       "assigning_driver",  // finding a driver
  PENDING_PICKUP:   "on_going",          // driver assigned, job not started
  PICKING_UP:       "on_going",          // driver on the way to us
  PENDING_DROP_OFF: "picked_up",         // parcel collected, on the way to the customer
  IN_DELIVERY:      "out_for_delivery",  // driver at the customer's address
  IN_RETURN:        "returning",         // drop-off failed; parcel coming back to us
  COMPLETED:        "delivered",
  RETURNED:         "failed",            // parcel back with us — re-book or hand over
  CANCELED:         "failed",
  CANCELLED:        "failed",
  FAILED:           "failed",            // no driver, driver cancelled, or Grab ops cancelled
};
 
/** Our status for a Grab state, or null for a state we do not know — an
 *  unknown state must never overwrite what we have. */
export function mapGrabStatus(grabStatus: unknown): string | null {
  if (typeof grabStatus !== "string") return null;
  return STATUS_MAP[grabStatus.trim().toUpperCase()] ?? null;
}
 
/** Grab phones are E.164 digits WITHOUT the "+" (their docs). */
export function grabPhone(e164: string): string | null {
  const digits = e164.replace(/[^\d]/g, "");
  return digits.length >= 8 && digits.length <= 15 ? digits : null;
}
 
export interface GrabPoint {
  address:     string;
  keywords?:   string;
  cityCode:    "SIN";
  coordinates: { latitude: number; longitude: number };
}
 
export interface GrabBookingConfig {
  serviceType:        "INSTANT" | "SAME_DAY" | "BULK";
  vehicleType:        string;
  paymentMethod:      "CASHLESS" | "CASH";
  /** Our ops inbox. Grab requires an email for sender AND recipient; we send
   *  ours for both so no customer email leaves our system — we notify the
   *  customer ourselves (WhatsApp), as we do for Lalamove and SPX. */
  contactEmail:       string;
}
 
export interface GrabBookingInput {
  orderId:      string;
  totalGrams:   number;
  origin:       { name: string; phone: string; address: string; lat: number; lng: number };
  recipient:    { name: string; phone: string; line1: string; line2?: string | null; postalCode: string };
  destination:  { lat: number; lng: number };
  /** Pickup window for a scheduled trip (grabPickupWindow); null books now. */
  pickupWindow?: { from: Date; to: Date } | null;
  /** Grab's `highValue` flag: goods above the admin's threshold, or the
   *  operator's per-order choice (lib/shipping/handover.ts isHighValue). */
  highValue:    boolean;
}
 
export interface GrabDeliveryRequest {
  merchantOrderID: string;
  serviceType:     string;
  vehicleType:     string;
  paymentMethod:   string;
  payer:           "SENDER";
  highValue:       boolean;
  packages:        Array<{
    name: string; description: string; quantity: number;
    dimensions: { height: number; width: number; depth: number; weight: number };
  }>;
  sender:      { firstName: string; email: string; phone: string; instruction: string };
  recipient:   { firstName: string; email: string; phone: string; instruction: string };
  origin:      GrabPoint;
  destination: GrabPoint;
  schedule?:   { pickupTimeFrom: string; pickupTimeTo: string };
  /** One Grab promo code; Grab requires paymentMethod alongside it (we always send one). */
  promoCode?:  string;
}
 
/** A small padded box. Grab requires dimensions; their default ceiling is
 *  50 × 50 × 50 cm, and a bar parcel is far below it. */
export const PARCEL_DIMENSIONS_CM = { height: 5, width: 15, depth: 10 } as const;
/** Packaging on top of the metal (box, card, padding), in grams. */
export const PACKAGING_GRAMS = 250;
/** How wide a pickup window we give Grab for a scheduled trip when the order
 *  has no window end (their docs recommend one to three hours). */
export const SCHEDULE_WINDOW_MINUTES = 60;
/** A Grab bike trip across Singapore, pickup to door, with margin. */
export const GRAB_TRIP_MINUTES = 45;
/** How long before the customer's window opens a driver may collect. */
export const PICKUP_LEAD_MINUTES = 30;
/** The narrowest pickup window we hand Grab. */
const MIN_PICKUP_WINDOW_MINUTES = 30;
/** Only a window this far out is scheduled; nearer, the trip goes now. */
const SCHEDULE_MIN_AHEAD_MINUTES = 30;
 
const MIN = 60_000;
 
/**
 * The pickup window for a customer's DELIVERY window, so the parcel ARRIVES
 * inside it (2026-11-13). Before this the pickup window was the delivery window's
 * first hour: a 19:30–21:00 order became pickup 19:30–20:30, and a driver
 * collecting at 20:25 lands after 21:00.
 *
 *   from = window start − PICKUP_LEAD (the trip itself fills that gap);
 *   to   = window end − GRAB_TRIP, never narrower than 30 minutes;
 *   no window end → from + SCHEDULE_WINDOW_MINUTES.
 *
 * null (book now) when there is no window, or it starts within 30 minutes.
 */
export function grabPickupWindow(
  startIso: string | null | undefined,
  endIso: string | null | undefined,
  now: Date = new Date(),
): { from: Date; to: Date } | null {
  if (!startIso) return null;
  const start = new Date(startIso).getTime();
  if (!Number.isFinite(start) || start - now.getTime() <= SCHEDULE_MIN_AHEAD_MINUTES * MIN) return null;
  // Grab refuses a pickup window in the past; never earlier than five minutes out.
  const from = Math.max(start - PICKUP_LEAD_MINUTES * MIN, now.getTime() + 5 * MIN);
  const end = endIso ? new Date(endIso).getTime() : NaN;
  const to = Number.isFinite(end) && end > start
    ? Math.max(end - GRAB_TRIP_MINUTES * MIN, from + MIN_PICKUP_WINDOW_MINUTES * MIN)
    : from + SCHEDULE_WINDOW_MINUTES * MIN;
  return { from: new Date(from), to: new Date(to) };
}
/** What the package is called. Neutral on purpose: it never names the contents
 *  (guarded by carrier-sender-anonymity.test.ts). */
export const GRAB_PACKAGE_NAME = "Parcel";
/** The package description carries the hand-over rule, because Grab shows the
 *  driver the package but not (yet) the recipient instruction. */
export const GRAB_PACKAGE_DESCRIPTION = `Sealed parcel. ${HANDOVER_INSTRUCTION}`;
 
function coords(lat: number, lng: number, what: string): { latitude: number; longitude: number } {
  if (!Number.isFinite(lat) || !Number.isFinite(lng) || Math.abs(lat) > 90 || Math.abs(lng) > 180) {
    throw new Error(`${what} coordinates are missing or invalid`);
  }
  return { latitude: lat, longitude: lng };
}
 
/**
 * The Create Delivery Request body for one order. Throws on anything Grab
 * would reject, so a bad order fails here with a message we wrote, not at
 * Grab with one we have to decode.
 *
 * Nothing about the contents or value is declared: a neutral package with no
 * price — the same rule as Lalamove, where we send no item details at all.
 * `highValue` comes from the order (threshold or the operator's choice). The
 * hand-over instruction (lib/shipping/handover.ts) goes in every field Grab
 * shows a driver.
 */
export function buildGrabDeliveryRequest(input: GrabBookingInput, cfg: GrabBookingConfig): GrabDeliveryRequest {
  const senderPhone = grabPhone(input.origin.phone);
  if (!senderPhone) throw new Error("Our pickup phone (LALAMOVE_PICKUP_PHONE) is not a valid phone number");
  const recipientPhone = grabPhone(input.recipient.phone);
  if (!recipientPhone) throw new Error("The recipient phone number is invalid");
  if (!input.recipient.name.trim()) throw new Error("The recipient name is missing");
  if (!/^\d{6}$/.test(input.recipient.postalCode.trim())) throw new Error("The delivery postal code is invalid");
 
  const line2 = input.recipient.line2?.trim();
  const destAddress = [input.recipient.line1.trim(), line2, `Singapore ${input.recipient.postalCode.trim()}`]
    .filter(Boolean).join(", ");
 
  const grams = Number.isFinite(input.totalGrams) && input.totalGrams > 0 ? input.totalGrams : 0;
  const body: GrabDeliveryRequest = {
    merchantOrderID: input.orderId,
    serviceType:     cfg.serviceType,
    vehicleType:     cfg.vehicleType,
    paymentMethod:   cfg.paymentMethod,
    payer:           "SENDER",
    highValue:       input.highValue === true,
    packages: [{
      name:        GRAB_PACKAGE_NAME,
      description: GRAB_PACKAGE_DESCRIPTION,
      quantity:    1,
      dimensions:  { ...PARCEL_DIMENSIONS_CM, weight: Math.ceil(grams + PACKAGING_GRAMS) },
    }],
    // sender.instruction is what Grab shows the driver at pickup.
    sender:    { firstName: input.origin.name, email: cfg.contactEmail, phone: senderPhone, instruction: HANDOVER_INSTRUCTION },
    recipient: { firstName: input.recipient.name.trim().slice(0, 100), email: cfg.contactEmail, phone: recipientPhone, instruction: HANDOVER_INSTRUCTION },
    origin: {
      address: input.origin.address, cityCode: "SIN",
      coordinates: coords(input.origin.lat, input.origin.lng, "Pickup"),
    },
    destination: {
      address: destAddress, cityCode: "SIN",
      ...(line2 ? { keywords: line2 } : {}),
      coordinates: coords(input.destination.lat, input.destination.lng, "Delivery"),
    },
  };
  if (input.pickupWindow) {
    body.schedule = { pickupTimeFrom: input.pickupWindow.from.toISOString(), pickupTimeTo: input.pickupWindow.to.toISOString() };
  }
  return body;
}
 
/** The fee from a Grab quote, in S$. Grab returns the amount in major units
 *  (S$11 is `11`); anything not in SGD, or not a positive number, is refused
 *  rather than recorded as a cost. */
export function quoteAmountSgd(quote: { amount?: unknown; currency?: { code?: unknown } | null } | null | undefined): number | null {
  if (!quote || quote.currency?.code !== "SGD") return null;
  const n = typeof quote.amount === "number" ? quote.amount : Number(quote.amount);
  return Number.isFinite(n) && n > 0 && n < 1000 ? Math.round(n * 100) / 100 : null;
}
 
/**
 * ▶ CONFIRM IN THE SANDBOX (test code VALIDPROMO): Grab's docs call `quote.amount`
 * "Delivery cost" and `discountInfo.amount` the discount, but do not say whether
 * `amount` is already net of it. We read `amount` as the price BEFORE the
 * discount. If the sandbox shows it is after, flip this one constant — every
 * price and saving in Dispatch and on the order derives from discountedQuote().
 */
export const GRAB_AMOUNT_IS_BEFORE_DISCOUNT = true;
 
/** A quote's price and promo saving, S$. `priceSgd` is what we would pay;
 *  `discountSgd` is null when no code applied. */
export function discountedQuote(
  quote: { amount?: unknown; currency?: { code?: unknown } | null; discountInfo?: { success?: unknown; amount?: unknown } | null } | null | undefined,
): { priceSgd: number | null; discountSgd: number | null } {
  const amount = quoteAmountSgd(quote);
  const d = quote?.discountInfo;
  const raw = d && d.success === true ? Number(d.amount) : NaN;
  const discount = Number.isFinite(raw) && raw > 0 ? Math.round(raw * 100) / 100 : null;
  if (amount === null || discount === null) return { priceSgd: amount, discountSgd: null };
  const price = GRAB_AMOUNT_IS_BEFORE_DISCOUNT ? Math.max(0, Math.round((amount - discount) * 100) / 100) : amount;
  return { priceSgd: price, discountSgd: discount };
}
