import type { SupabaseClient } from "@supabase/supabase-js";
 
/**
 * Gold lines on an order that have NO physical bar yet because they were sold
 * on consignment and the supplier bar has not been received.
 *
 * A consignment sale has no inventory row until Admin → Consignment Queue →
 * Confirm → Receive, which mints the bar and links it to the order line via
 * order_fulfilments. Until then the order can look ready: it is paid, it can be
 * packed, and its packing slip printed with only the accessories on it (found
 * 2026-09-29 on two PAMP orders). Every step that would send the parcel out —
 * packing, booking a courier, handing over, storing in the vault — must refuse
 * while this list is non-empty.
 *
 * `awaiting_supplier` and `confirmed` are the pre-receipt states; `received`
 * and `cancelled` are not awaiting anything.
 */
export const AWAITING_CONSIGNMENT_STATUSES = ["awaiting_supplier", "confirmed"] as const;
 
export interface AwaitingGoldLine {
  order_item_id: string;
  product_label: string;
  quantity:      number;
  status:        string;
}
 
/**
 * Returns the awaiting lines, or `null` when the lookup failed. Callers that
 * gate an action must treat `null` as "blocked" (fail closed): a transient DB
 * error must never let a parcel go out without its gold.
 */
export async function findAwaitingGold(
  db: SupabaseClient,
  orderId: string,
): Promise<AwaitingGoldLine[] | null> {
  const { data, error } = await db
    .from("consignment_procurements")
    .select("order_item_id, quantity, status, products(label)")
    .eq("order_id", orderId)
    .in("status", [...AWAITING_CONSIGNMENT_STATUSES]);
 
  if (error) {
    console.error("[findAwaitingGold] consignment_procurements lookup failed:", error.message);
    return null;
  }
  return ((data ?? []) as Array<{
    order_item_id: string | null;
    quantity: number | null;
    status: string;
    products: { label: string | null } | { label: string | null }[] | null;
  }>).map((r) => {
    const p = Array.isArray(r.products) ? r.products[0] : r.products;
    return {
      order_item_id: r.order_item_id ?? "",
      product_label: p?.label ?? "Gold item",
      quantity:      Number(r.quantity ?? 1),
      status:        r.status,
    };
  });
}
 
/** One sentence naming what is missing, for 409 bodies and UI banners. */
export function awaitingGoldMessage(lines: AwaitingGoldLine[]): string {
  const list = lines.map((l) => `${l.quantity} × ${l.product_label}`).join(", ");
  return `Gold not yet received from the supplier: ${list}. Receive it in Admin → Consignment Queue first.`;
}
 
/**
 * Gate helper for server routes: returns a 409-ready error string when the
 * order must not ship yet, else null. Fails closed on a lookup error.
 */
export async function awaitingGoldBlocker(
  db: SupabaseClient,
  orderId: string,
): Promise<string | null> {
  const lines = await findAwaitingGold(db, orderId);
  if (lines === null) return "Could not confirm the order's gold has been received — try again.";
  return lines.length > 0 ? awaitingGoldMessage(lines) : null;
}
