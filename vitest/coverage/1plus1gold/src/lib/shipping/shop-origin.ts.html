/**
 * Where couriers collect from: our dispatch address and contact (2026-11-07).
 *
 * ONE reader for every courier. The env vars keep their historical LALAMOVE_
 * prefix because Lalamove was the first integration and PROD already has them
 * set; GrabExpress reads the same values, so the two can never disagree about
 * where the parcel is. The defaults are the long-standing Lalamove ones —
 * unchanged, so this refactor cannot move a Lalamove pickup.
 */
export interface ShopOrigin {
  name:    string;
  /** E.164 with a leading +, e.g. "+6591234567". */
  phone:   string;
  address: string;
  /** Decimal-degree strings, as Lalamove's API wants them. */
  lat:     string;
  lng:     string;
}
 
export function shopOrigin(): ShopOrigin {
  return {
    name:    (process.env.LALAMOVE_PICKUP_NAME    ?? "1p1g").trim(),
    phone:   (process.env.LALAMOVE_PICKUP_PHONE   ?? "+6591234567").trim(),
    address: (process.env.LALAMOVE_PICKUP_ADDRESS ?? "93 Meyer Road, #25-02, The Meyerise, Singapore 437986").trim(),
    lat:     (process.env.LALAMOVE_PICKUP_LAT     ?? "1.29620246948925").trim(),
    lng:     (process.env.LALAMOVE_PICKUP_LNG     ?? "103.892923386471").trim(),
  };
}
