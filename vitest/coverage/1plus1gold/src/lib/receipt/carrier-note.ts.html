import type { SupabaseClient } from "@supabase/supabase-js";
 
/**
 * The customer-facing carrier note (`shipper_settings.notes`) for an order's
 * shipper — the SAME admin-editable copy shown at checkout under the carrier
 * (e.g. "Best effort next day"). Rendered on the receipt beside the carrier so a
 * customer's confirmation reflects current dispatch expectations, and so a
 * temporary change (e.g. "Shipping on 20 Aug" during a shipping pause) reaches
 * every new receipt with no code/deploy — just an admin edit.
 *
 * Best-effort: returns null on any miss (no shipper, no row, error) so the
 * receipt simply omits the note rather than failing.
 */
export async function fetchCarrierNote(
  supabaseAdmin: SupabaseClient,
  shipper: string | null | undefined,
): Promise<string | null> {
  if (!shipper) return null;
  try {
    const { data } = await supabaseAdmin
      .from("shipper_settings")
      .select("notes")
      .eq("shipper", shipper)
      .maybeSingle();
    const note = (data as { notes?: string | null } | null)?.notes;
    const trimmed = typeof note === "string" ? note.trim() : "";
    return trimmed || null;
  } catch {
    return null;
  }
}
 
/**
 * The Delivery line for a receipt, corrected for 1+1Gold MAG (testing item 14).
 *
 * A MAG purchase is booked under the `vault` carrier, so it used to print the
 * vault note — "Customer pays shipping at withdrawal". A MAG holding is never
 * withdrawn and shipped: it is grams of a shared bar held in our custody. So:
 *  - every line is MAG  → MAG wording replaces the carrier label and note;
 *  - some lines are MAG → the carrier wording stays (it is right for the
 *    physical bars) and a MAG sentence is appended;
 *  - no MAG lines       → unchanged.
 * Pure: both receipt writers call it with the same inputs, so their cache keys
 * (which include serviceType + carrierNote) stay identical.
 */
export const MAG_RECEIPT_SERVICE = "1+1Gold MAG — held in custody";
export const MAG_RECEIPT_NOTE =
  "Allocated grams of a shared bar held in our custody. Nothing is shipped — sell back or convert from My Account.";
 
export function receiptFulfilment(input: {
  serviceType: string | null;
  carrierNote: string | null;
  itemCount: number;
  magItemCount: number;
  /** "Tue 6 Oct, 10:00–12:00" when the customer chose a Lalamove window. It
   *  REPLACES the carrier's generic note (e.g. "within 1–2 hours"), which would
   *  otherwise contradict the slot the customer actually booked. */
  deliveryWindow?: string | null;
}): { serviceType: string | null; carrierNote: string | null } {
  const { serviceType, itemCount, magItemCount, deliveryWindow } = input;
  const carrierNote = deliveryWindow ? `Delivery window: ${deliveryWindow} (SGT)` : input.carrierNote;
  if (magItemCount <= 0) return { serviceType, carrierNote };
  if (magItemCount >= itemCount) return { serviceType: MAG_RECEIPT_SERVICE, carrierNote: MAG_RECEIPT_NOTE };
  const magLine = "MAG grams on this order stay allocated in our custody and are not shipped.";
  return { serviceType, carrierNote: carrierNote ? `${carrierNote} ${magLine}` : magLine };
}
