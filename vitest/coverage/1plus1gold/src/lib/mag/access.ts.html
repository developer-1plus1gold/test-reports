/**
 * MAG — gradual roll-out gate (operator 2026-09-23).
 *
 * MAG is visible to everyone, but BUYING it needs either the global switch
 * `mag_settings.open_to_all` or the customer's own `customers.mag_enabled`.
 * Only buying is gated: sell-back and redemption of grams a customer already
 * holds are never checked here.
 *
 * PER BAR (owner 2026-11-11): being a MEMBER of a reserved bar or group pool
 * (`mag_listing_members`) is itself the admin's explicit choice, so it counts
 * as access FOR THAT BAR — a member may buy it without `mag_enabled`. Public
 * bars keep the customer-wide rule. The pure rule both sides share lives in
 * lib/mag/buyable.ts (client-safe); this module does the reads.
 *
 * Fails CLOSED: a read error means "not allowed". Refusing a purchase is
 * recoverable (the customer asks again); selling MAG to someone the operator
 * has not enabled is the thing the roll-out exists to prevent.
 */
import type { SupabaseClient } from "@supabase/supabase-js";
import { magListingsPurchasable } from "./buyable";
 
export const MAG_ACCESS_DENIED =
  "MAG is opening gradually. Request early access on the Shop page and we'll let you know when your account is enabled.";
 
export interface MagAccess {
  openToAll: boolean;
  enabled: boolean;      // customers.mag_enabled
  requested: boolean;    // an early-access request is on file
  canBuy: boolean;
}
 
export async function readMagOpenToAll(supabase: SupabaseClient): Promise<boolean> {
  const { data, error } = await supabase.from("mag_settings").select("open_to_all").eq("id", 1).maybeSingle();
  if (error) { console.error("[mag/access] settings read failed:", error.message); return false; }
  return data?.open_to_all === true;
}
 
export async function getMagAccess(supabase: SupabaseClient, customerId: string | null): Promise<MagAccess> {
  const openToAll = await readMagOpenToAll(supabase);
  if (!customerId) return { openToAll, enabled: false, requested: false, canBuy: openToAll };
  const [cust, reqRow] = await Promise.all([
    supabase.from("customers").select("mag_enabled").eq("id", customerId).maybeSingle(),
    supabase.from("mag_access_requests").select("customer_id").eq("customer_id", customerId).maybeSingle(),
  ]);
  if (cust.error) console.error("[mag/access] customer read failed:", cust.error.message);
  const enabled = !cust.error && cust.data?.mag_enabled === true;
  const requested = !reqRow.error && !!reqRow.data;
  return { openToAll, enabled, requested, canBuy: openToAll || enabled };
}
 
/**
 * The ACTIVE reserved/group bars this customer is a member of, optionally
 * narrowed to `listingIds`. Fails CLOSED: a read error returns an empty set.
 */
export async function readPrivateMemberListingIds(
  supabase: SupabaseClient,
  customerId: string,
  listingIds?: readonly string[],
): Promise<Set<string>> {
  if (listingIds && listingIds.length === 0) return new Set();
  let q = supabase.from("mag_listing_members").select("listing_id").eq("customer_id", customerId);
  if (listingIds) q = q.in("listing_id", [...listingIds]);
  const { data: members, error: mErr } = await q;
  if (mErr) { console.error("[mag/access] membership read failed:", mErr.message); return new Set(); }
  const ids = [...new Set(((members ?? []) as { listing_id: string }[]).map(m => m.listing_id))];
  if (ids.length === 0) return new Set();
  // Membership counts only on a private bar: a stray member row on a public
  // bar (access switched back to public) must not bypass the roll-out.
  const { data: bars, error: lErr } = await supabase
    .from("mag_listings").select("id").in("id", ids).in("access_type", ["reserved", "group"]).eq("status", "active");
  if (lErr) { console.error("[mag/access] listing read failed:", lErr.message); return new Set(); }
  return new Set(((bars ?? []) as { id: string }[]).map(b => b.id));
}
 
/**
 * Server-side purchase gate for the order + payment routes: may this customer
 * buy grams of EVERY listing in `listingIds`? Customer-wide access covers all
 * bars; otherwise each listing must be a private bar they are a member of.
 */
export async function magPurchaseAllowed(
  supabase: SupabaseClient,
  customerId: string,
  listingIds: readonly string[],
): Promise<boolean> {
  const access = await getMagAccess(supabase, customerId);
  if (access.canBuy) return true;
  const member = await readPrivateMemberListingIds(supabase, customerId, listingIds);
  return magListingsPurchasable(access.canBuy, listingIds, member);
}
 
/**
 * Admin → MAG → Customer view: the customer-wide access plus the private bars
 * whose membership lets them buy without it. Same reads as the purchase gate.
 */
export async function getMagAccessSummary(
  supabase: SupabaseClient,
  customerId: string,
): Promise<MagAccess & { memberListingIds: string[] }> {
  const [access, member] = await Promise.all([
    getMagAccess(supabase, customerId),
    readPrivateMemberListingIds(supabase, customerId),
  ]);
  return { ...access, memberListingIds: [...member].sort() };
}
