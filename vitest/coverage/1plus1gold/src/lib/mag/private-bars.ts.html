/**
 * Admin → MAG → Customer view: every active PRIVATE bar (reserved + group),
 * who its members are, and the margin at today's spot (owner 2026-11-11).
 * Pure so the route stays a thin read and the arithmetic is testable.
 *
 * Margin is measured on the LOWEST premium tier — the cheapest price any
 * member can be charged — so a thin figure is the worst case, not an average.
 * The margin leaves the supplier card fee out (owner 2026-11-14): the listing's
 * `cost_per_g_sgd` includes it, so the bar's `fee_per_g_sgd` (inventory) is taken
 * off — the same split as gross profit, where the fee is its own P&L line (#767).
 */
import type { MagAccessType, MagTier } from "./listings";
import { memberGramsBought } from "./pool-admin";
 
/** Below this the margin is flagged amber; below 0 red. */
export const MAG_THIN_MARGIN_PCT = 1;
 
export type MarginFlag = "negative" | "thin" | "ok";
 
/**
 * Margin % at one tier: (spot × (1 + tier%/100) − cost) / price × 100.
 * `premiumPct` is the stored PERCENT (2.8 = 2.8%) — standard_premium_tiers'
 * unit, which getTierPremium divides by 100. (Until 2026-11-14 this read it as a
 * fraction, so a 2.8% bar priced at spot × 3.8 and every margin read ~70%: the
 * Customer view could never flag a thin or negative bar.) Null when it cannot be
 * computed (no spot, no price).
 */
export function magTierMarginPct(spotSgdPerG: number | null, premiumPct: number, costPerG: number): number | null {
  if (spotSgdPerG == null || !Number.isFinite(spotSgdPerG) || spotSgdPerG <= 0) return null;
  Iif (!Number.isFinite(premiumPct) || !Number.isFinite(costPerG)) return null;
  const price = spotSgdPerG * (1 + premiumPct / 100);
  Iif (!(price > 0)) return null;
  return ((price - costPerG) / price) * 100;
}
 
export function marginFlag(pct: number | null): MarginFlag | null {
  if (pct == null) return null;
  if (pct < 0) return "negative";
  if (pct < MAG_THIN_MARGIN_PCT) return "thin";
  return "ok";
}
 
/** The tier with the smallest premium — the worst-case price. */
export function lowestPremiumTier(tiers: MagTier[]): MagTier | null {
  return tiers.reduce<MagTier | null>((lo, t) => (lo == null || t.premium_pct < lo.premium_pct ? t : lo), null);
}

  groupName: string | null;

  gramsTotal: number;
  gramsRemaining: number;

  costPerG: number;

  members: PrivateBarMember[];

type Num = number | string;
 
export interface PrivateBarInputs {
  listings: {
    id: string; serial: string; product_id: string; access_type: string | null; group_name: string | null;
    window_ends_at: string | null; grams_total: Num; cost_per_g_sgd: Num;
    /** The bar's supplier card fee per gram (inventory.fee_per_g_sgd); 0/absent = none. */
    fee_per_g_sgd?: Num | null;
  }[];
  products: {
    id: string; label: string;
    standard_premium_tiers: { min_qty: Num; max_qty: Num | null; premium_pct: Num }[] | null;
  }[];
  availability: { listing_id: string; grams_remaining: Num }[];
  members: { listing_id: string; customer_id: string; max_grams: Num | null }[];
  allocations: { listing_id: string; customer_id: string | null; grams: Num | null; status: string; hold_expires_at?: string | null }[];
  names: Map<string, string>;
  spotSgdPerG: number | null;
  nowMs: number;
}
 
const isPrivate = (a: string | null): a is "reserved" | "group" => a === "reserved" || a === "group";
 
export function buildPrivateBarRows(i: PrivateBarInputs): PrivateBarRow[] {
  const productById = new Map(i.products.map(p => [p.id, p]));
  const remaining = new Map(i.availability.map(a => [a.listing_id, Number(a.grams_remaining)]));
  const rows: PrivateBarRow[] = [];
  for (const l of i.listings) {
    const accessType = l.access_type;
    if (!isPrivate(accessType)) continue;
    const p = productById.get(l.product_id);
    const tiers: MagTier[] = (p?.standard_premium_tiers ?? []).map(t => ({
      min_qty: Number(t.min_qty), max_qty: t.max_qty == null ? null : Number(t.max_qty), premium_pct: Number(t.premium_pct),
    })).sort((a, b) => a.min_qty - b.min_qty);
    const bought = memberGramsBought(i.allocations.filter(a => a.listing_id === l.id), i.nowMs);
    const fee = Number(l.fee_per_g_sgd ?? 0);
    const costPerG = Number(l.cost_per_g_sgd) - (Number.isFinite(fee) && fee > 0 ? fee : 0);
    const lo = lowestPremiumTier(tiers);
    const pct = lo ? magTierMarginPct(i.spotSgdPerG, lo.premium_pct, costPerG) : null;
    rows.push({
      listingId: l.id,
      serial: l.serial,
      label: p?.label ?? "MAG bar",
      accessType,
      groupName: accessType === "group" ? l.group_name : null,
      windowEndsAt: l.window_ends_at,
      gramsTotal: Number(l.grams_total),
      gramsRemaining: Math.max(0, remaining.get(l.id) ?? 0),
      costPerG,
      tiers,
      members: i.members.filter(m => m.listing_id === l.id).map(m => ({
        customerId: m.customer_id,
        name: i.names.get(m.customer_id) ?? null,
        maxGrams: m.max_grams == null ? null : Number(m.max_grams),
        gramsBought: bought.get(m.customer_id) ?? 0,
      })).sort((a, b) => (a.name ?? "").localeCompare(b.name ?? "")),
      lowestTierMarginPct: pct,
      marginFlag: marginFlag(pct),
    });
  }
  return rows.sort((a, b) =>
    a.accessType.localeCompare(b.accessType)
    || (a.groupName ?? "").localeCompare(b.groupName ?? "")
    || a.serial.localeCompare(b.serial));
}
