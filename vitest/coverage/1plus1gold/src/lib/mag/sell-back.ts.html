/**
 * MAG — sell back (design: docs/allocated-gold-design.md §4.4).
 *
 * The bar never left our vault, so a sell-back is a buyback with no pickup:
 * the request is created at `received` with `shipping_type = 'vault'` and
 * `mag_listing_id` set, then runs the ordinary admin lifecycle (approve →
 * record payout → complete). The grams move `allocated → sell_requested`
 * atomically in `mag_request_sell_back` (FIFO, splitting the last row), stay
 * counted as live until the request resolves, and then settle:
 *   completed           → `sold_back` (grams return to the listing, derived)
 *   rejected / cancelled → back to `allocated`
 *
 * No inventory row is ever touched — the physical bar stays `mag_listed`.
 */
import type { SupabaseClient } from "@supabase/supabase-js";
 
/** Quote formula shared with /api/buybacks and /api/vault/sell — never diverge. */
export function quoteMagSellBack(spotSgdPerG: number | null, grams: number, buybackPremiumPct: number): number | null {
  if (spotSgdPerG == null || !(spotSgdPerG > 0) || !(grams > 0)) return null;
  return Math.round(spotSgdPerG * grams * (1 + buybackPremiumPct / 100) * 100) / 100;
}

  return !!err && /mag_sell_insufficient|mag_sell_invalid_grams/.test(err.message ?? "");

/**

  return !!err && /mag_delivery_locked/.test(err.message ?? "");

/**
 * If this sale takes the customer's LAST whole gram on the bar and they own the
 * piece's fraction, claim the fraction for this request (house_sell_request_id)
 * and return its grams; else 0. Called after mag_request_sell_back has moved
 * the grams to sell_requested, so "no allocated grams left" is exact.
 */
async function claimFractionForSale(
  supabase: SupabaseClient, customerId: string, listingId: string, requestId: string,
): Promise<number> {
  const { count, error } = await supabase
    .from("mag_allocations").select("id", { count: "exact", head: true })
    .eq("customer_id", customerId).eq("listing_id", listingId).eq("status", "allocated");
  if (error || (count ?? 0) > 0) return 0;
  const { data } = await supabase
    .from("mag_listings")
    .update({ house_sell_request_id: requestId })
    .eq("id", listingId).eq("house_owner_customer_id", customerId).is("house_sell_request_id", null)
    .select("house_grams");
  const row = (data ?? [])[0] as { house_grams: number | string } | undefined;
  return row ? Number(row.house_grams) : 0;
}
