/**
 * MAG access pools — how listings and cart lines group into sections.
 * One ordering for the Shop and the cart so they cannot disagree:
 *   1. the caller's reserved bar(s)
 *   2. one section per group pool (by group name)
 *   3. the public pool
 * Design: docs/mag-reserved-pools-design.md §3.
 */
import type { MagAccessType, MagListingPublic } from "./listings";
 
export interface MagSection<T> {
  key: string;
  access: MagAccessType;
  title: string;
  items: T[];
}
 
export const MAG_SECTION_TITLES = {
  reserved: "Your reserved bar",
  public:   "Public pool",
  groupFallback: "Your group pool",
} as const;
 
/** Generic, non-disclosing heading for the private sub-sections a caller is not in. */
export const MAG_PRIVATE_CONTACT_TITLE = "Reserved & group bars";
export const MAG_PRIVATE_CONTACT_COPY =
  "Some bars are reserved for one customer or pooled for a group, with a window to buy. Contact us for more information.";
 
function sectionFor(access: MagAccessType, groupName: string | null | undefined): { key: string; title: string } {
  if (access === "reserved") return { key: "reserved", title: MAG_SECTION_TITLES.reserved };
  if (access === "group") {
    const name = groupName?.trim() || MAG_SECTION_TITLES.groupFallback;
    return { key: `group:${name}`, title: name };
  }
  return { key: "public", title: MAG_SECTION_TITLES.public };
}
 
const rank = (a: MagAccessType) => (a === "reserved" ? 0 : a === "group" ? 1 : 2);
 
export function groupMagSections<T>(
  items: T[],
  accessOf: (t: T) => MagAccessType | undefined,
  groupOf: (t: T) => string | null | undefined,
): MagSection<T>[] {
  const byKey = new Map<string, MagSection<T>>();
  for (const it of items) {
    const access = accessOf(it) ?? "public";
    const { key, title } = sectionFor(access, groupOf(it));
    let s = byKey.get(key);
    if (!s) { s = { key, access, title, items: [] }; byKey.set(key, s); }
    s.items.push(it);
  }
  return [...byKey.values()].sort((a, b) => rank(a.access) - rank(b.access) || a.title.localeCompare(b.title));
}
 
export function groupMagListings(listings: MagListingPublic[]): MagSection<MagListingPublic>[] {
  return groupMagSections(listings, l => l.accessType, l => l.groupName);
}
 
/** Grams the caller may still buy on a listing: remaining grams, minus their cap headroom on a private bar. */
export function magBuyableGrams(l: Pick<MagListingPublic, "gramsRemaining" | "accessType" | "myCapGrams" | "myBoughtGrams">): number {
  let n = Math.max(0, l.gramsRemaining);
  if (l.accessType !== "public" && l.myCapGrams != null) {
    n = Math.min(n, Math.max(0, l.myCapGrams - (l.myBoughtGrams ?? 0)));
  }
  return n;
}
 
/** "Buy by 31 Oct 2026" in Singapore time, or null for a public bar. */
export function magWindowLabel(windowEndsAt: string | null | undefined): string | null {
  if (!windowEndsAt) return null;
  const t = Date.parse(windowEndsAt);
  if (!Number.isFinite(t)) return null;
  const d = new Date(t + 8 * 3600_000);
  const months = ["Jan","Feb","Mar","Apr","May","Jun","Jul","Aug","Sep","Oct","Nov","Dec"];
  return `Buy by ${d.getUTCDate()} ${months[d.getUTCMonth()]} ${d.getUTCFullYear()}`;
}
 
export type MagCartEntry<T> =
  | { kind: "item"; item: T }
  | { kind: "heading"; key: string; title: string }
  | { kind: "contact" };
 
/**
 * Cart layout: non-MAG lines first in their own order, then every MAG
 * sub-section (reserved → groups → public) under its heading, then the
 * contact-us row. Headings and the contact row only appear when the cart
 * holds MAG.
 */
export function magCartEntries<T extends { productType: string; magAccess?: MagAccessType; magGroupName?: string }>(
  items: T[],
): MagCartEntry<T>[] {
  const others = items.filter(i => i.productType !== "mag");
  const mag = items.filter(i => i.productType === "mag");
  const out: MagCartEntry<T>[] = others.map(item => ({ kind: "item", item }));
  if (mag.length === 0) return out;
  const sections = groupMagSections(mag, i => i.magAccess, i => i.magGroupName);
  for (const s of sections) {
    out.push({ kind: "heading", key: s.key, title: `${s.title}` });
    for (const item of s.items) out.push({ kind: "item", item });
  }
  out.push({ kind: "contact" });
  return out;
}
