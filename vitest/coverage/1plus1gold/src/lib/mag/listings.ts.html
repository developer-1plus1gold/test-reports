/**
 * 1+1Gold MAG — reading listings for the shop, the admin and the account page.
 * The ordinary shop catalog excludes MAG (lib/shop-catalog.ts), so this is the
 * only place MAG products + their per-serial availability are read.
 */
import type { SupabaseClient } from "@supabase/supabase-js";
import { listingImage } from "@/lib/mag/listing-source";
 
export interface MagTier { min_qty: number; max_qty: number | null; premium_pct: number }
 
export interface MagListingPublic {
  listingId:      string;
  productId:      string;
  label:          string;
  refinery:       string | null;
  weightDisplay:  string | null;
  /** The bar's own photo (its serial's CoA photo) → legacy listing photo → bar-type image. */
  imageUrl:       string | null;
  /** The bar's own video from its serial (Serials screen), if any. */
  videoUrl:       string | null;
  serial:         string;
  /** Refinery-stamped serial on the physical bar, when recorded. */
  manufacturerSerial: string | null;
  fineness:       number | null;
  isLbma:         boolean;
  dimensions:     string | null;
  gstExempt:      boolean;
  gramsTotal:     number;
  gramsRemaining: number;
  tiers:          MagTier[];
  /** products.low_stock_threshold, read as GRAMS for MAG (summed across the
   *  bar type's listings — see lib/mag/stock.ts). null = no low-stock warning. */
  lowStockThreshold?: number | null;
}
 
interface ProductRow {
  id: string; label: string; refinery: string | null; weight_display: string | null;
  shop_visible: boolean; product_type: string;
  fineness: number | null; is_lbma: boolean | null; dimensions: string | null; gst_exempt: boolean | null;
  standard_premium_tiers: MagTier[] | null;
  low_stock_threshold: number | null;
  product_media: { url: string; kind?: string | null; sort_order?: number | null }[] | null;
}
 
function firstImage(media: ProductRow["product_media"]): string | null {
  const imgs = (media ?? []).filter(m => (m.kind ?? "image") === "image")
    .sort((a, b) => (a.sort_order ?? 0) - (b.sort_order ?? 0));
  return imgs[0]?.url ?? null;
}
 
/**
 * Active listings on shop-visible MAG products, soonest-to-sell-out last.
 * Sold-out listings are included (gramsRemaining = 0) so the shop can say so.
 */
export async function fetchPublicMagListings(supabase: SupabaseClient): Promise<MagListingPublic[]> {
  const { data: listings, error } = await supabase
    .from("mag_listings")
    .select("id, product_id, serial, grams_total, photo_url, status")
    .eq("status", "active");
  if (error) throw new Error(error.message);
  const rows = (listings ?? []) as { id: string; product_id: string; serial: string; grams_total: number; photo_url: string | null }[];
  if (rows.length === 0) return [];
 
  const ids = rows.map(r => r.id);
  const productIds = [...new Set(rows.map(r => r.product_id))];
  const [{ data: avail, error: aErr }, { data: products, error: pErr }, { data: media }] = await Promise.all([
    supabase.from("mag_listing_availability").select("listing_id, grams_remaining").in("listing_id", ids),
    supabase.from("products")
      .select("id, label, refinery, weight_display, shop_visible, product_type, fineness, is_lbma, dimensions, gst_exempt, low_stock_threshold, standard_premium_tiers(min_qty, max_qty, premium_pct), product_media(url, kind, sort_order)")
      .in("id", productIds),
    // Media is best-effort: a failed read falls back to the bar-type image
    // rather than hiding MAG from the shop.
    supabase.from("nfc_serials").select("serial, image_url, video_url, manufacturer_serial").in("serial", rows.map(r => r.serial)),
  ]);
  if (aErr || pErr) throw new Error((aErr ?? pErr)!.message);
  type SerialMedia = { serial: string; image_url: string | null; video_url: string | null; manufacturer_serial: string | null };
  const mediaBySerial = new Map(((media ?? []) as SerialMedia[]).map(m => [m.serial, m]));
 
  const remaining = new Map((avail ?? []).map((a: { listing_id: string; grams_remaining: number }) => [a.listing_id, Number(a.grams_remaining)]));
  const productById = new Map(((products ?? []) as ProductRow[]).map(p => [p.id, p]));
 
  return rows.flatMap(r => {
    const p = productById.get(r.product_id);
    if (!p || !p.shop_visible || p.product_type !== "mag") return [];
    return [{
      listingId:      r.id,
      productId:      p.id,
      label:          p.label,
      refinery:       p.refinery,
      weightDisplay:  p.weight_display,
      imageUrl:       listingImage(mediaBySerial.get(r.serial)?.image_url, r.photo_url, firstImage(p.product_media)),
      videoUrl:       mediaBySerial.get(r.serial)?.video_url ?? null,
      serial:         r.serial,
      manufacturerSerial: mediaBySerial.get(r.serial)?.manufacturer_serial ?? null,
      fineness:       p.fineness == null ? null : Number(p.fineness),
      isLbma:         !!p.is_lbma,
      dimensions:     p.dimensions,
      gstExempt:      p.gst_exempt !== false,
      gramsTotal:     Number(r.grams_total),
      gramsRemaining: Math.max(0, remaining.get(r.id) ?? 0),
      tiers:          (p.standard_premium_tiers ?? []).map(t => ({ min_qty: Number(t.min_qty), max_qty: t.max_qty == null ? null : Number(t.max_qty), premium_pct: Number(t.premium_pct) })),
      lowStockThreshold: p.low_stock_threshold == null ? null : Number(p.low_stock_threshold),
    }];
  }).sort((a, b) => b.gramsRemaining - a.gramsRemaining || a.label.localeCompare(b.label));
}
 
export interface MagHoldingRow {
  listingId: string;
  serial:    string;
  label:     string;
  grams:     number;   // allocated — owned and free to sell or redeem
  pendingSaleGrams: number; // sell_requested — owned, locked in an open sell-back
  paidSgd:   number;   // Σ price_per_g × grams over allocated grams (what the customer paid)
  /** Σ price_per_g × grams over the sell_requested grams. Kept separate so
   *  paidSgd / grams stays the cost of the FREE grams (sell-to-us reads it),
   *  while a display of what the customer HOLDS adds the two (2026-10-05). */
  pendingSalePaidSgd: number;
  buybackEligible:   boolean;
  buybackPremiumPct: number;
  /** Bar type's purchase tiers — a redeem at checkout is valued at the MAG buy price. */
  tiers:             MagTier[];
}
 
/**
 * A customer's owned grams, one row per bar: `allocated` grams plus grams
 * locked in an open sell-back (`sell_requested`, shown separately — still
 * theirs until the buyback completes).
 */
export async function fetchCustomerMagHoldings(supabase: SupabaseClient, customerId: string): Promise<MagHoldingRow[]> {
  const { data, error } = await supabase
    .from("mag_allocations")
    .select("listing_id, grams, status, price_per_g_sgd, mag_listings(serial, products(label, buyback_eligible, buyback_premium_pct, standard_premium_tiers(min_qty, max_qty, premium_pct)))")
    .eq("customer_id", customerId)
    .in("status", ["allocated", "sell_requested"]);
  if (error) throw new Error(error.message);
  const by = new Map<string, MagHoldingRow>();
  for (const r of (data ?? []) as unknown as {
    listing_id: string; grams: number; status: string; price_per_g_sgd: number | null;
    mag_listings: { serial: string; products: { label: string; buyback_eligible?: boolean | null; buyback_premium_pct?: number | null; standard_premium_tiers?: Array<{ min_qty: number; max_qty: number | null; premium_pct: number }> | null } | null } | null;
  }[]) {
    const p = r.mag_listings?.products;
    const cur = by.get(r.listing_id) ?? {
      listingId: r.listing_id,
      serial:    r.mag_listings?.serial ?? "",
      label:     p?.label ?? "Allocated gold",
      grams: 0, pendingSaleGrams: 0, paidSgd: 0, pendingSalePaidSgd: 0,
      buybackEligible:   !!p?.buyback_eligible,
      buybackPremiumPct: Number(p?.buyback_premium_pct ?? 0),
      tiers: (p?.standard_premium_tiers ?? []).map(t => ({ min_qty: Number(t.min_qty), max_qty: t.max_qty == null ? null : Number(t.max_qty), premium_pct: Number(t.premium_pct) })),
    };
    if (r.status === "sell_requested") {
      cur.pendingSaleGrams += Number(r.grams);
      cur.pendingSalePaidSgd += Number(r.grams) * Number(r.price_per_g_sgd ?? 0);
    } else {
      cur.grams   += Number(r.grams);
      cur.paidSgd += Number(r.grams) * Number(r.price_per_g_sgd ?? 0);
    }
    by.set(r.listing_id, cur);
  }
  return [...by.values()].map(h => ({
    ...h,
    paidSgd: Math.round(h.paidSgd * 100) / 100,
    pendingSalePaidSgd: Math.round(h.pendingSalePaidSgd * 100) / 100,
  }));
}
 
/**
 * Validate an admin-entered MAG listing photo URL. Returns the URL to store,
 * or null (no photo) for anything unacceptable.
 *
 * Only hosts the site CSP `img-src` actually allows are accepted: a same-origin
 * path ("/images/…", never protocol-relative "//"), or https on our Supabase
 * storage. Anything else would be stored happily and then render as a broken
 * image for every customer (blocked by CSP), and would make the customer's
 * browser call a host we don't control.
 */
export function normaliseMagPhotoUrl(raw: unknown): string | null {
  if (typeof raw !== "string") return null;
  const v = raw.trim();
  if (!v || v.length > 500) return null;
  if (v.startsWith("/") && !v.startsWith("//") && !v.includes("\\")) return v;
  let u: URL;
  try { u = new URL(v); } catch { return null; }
  if (u.protocol !== "https:" || u.username || u.password) return null;
  const host = u.hostname.toLowerCase();
  if (host.endsWith(".supabase.co")) return u.toString();
  return null;
}
