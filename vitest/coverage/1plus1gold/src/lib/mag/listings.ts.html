export function isMagAccessType(v: unknown): v is MagAccessType {
  return v === "public" || v === "reserved" || v === "group";
}
 
/** True once a private bar's buying window has passed. */
export function magWindowClosed(l: Pick<MagListingPublic, "accessType" | "windowEndsAt">, now = Date.now()): boolean {
  return l.accessType !== "public" && !!l.windowEndsAt && Date.parse(l.windowEndsAt) <= now;
}

  const imgs = (media ?? []).filter(m => (m.kind ?? "image") === "image")

/**
 * Active PUBLIC listings on shop-visible MAG products, soonest-to-sell-out last.
 * Sold-out listings are included (gramsRemaining = 0); the SHOP drops them for
 * a signed-out visitor with magShopVisible.
 *
 * Reserved/group bars are NEVER returned here: this reader feeds cached and
 * anonymous surfaces (/shop, Home tiles, pricing overview), and a private
 * serial must not reach them. Per-customer reads use fetchMagListingsForCustomer.
 */
/**
 * What the shop shows (owner 2026-11-14): an open-pool bar while it has grams
 * to sell, to everyone; once it is fully sold, only to customers who still hold
 * grams in it. A reserved/group bar is already limited to its members, who keep
 * seeing it sold out or not.
 */
export function magShopVisible(l: Pick<MagListingPublic, "accessType" | "gramsRemaining" | "myHeldGrams">): boolean {
  return l.accessType !== "public" || l.gramsRemaining > 0 || (l.myHeldGrams ?? 0) > 0;
}

    paidSgd: Math.round(h.paidSgd * 100) / 100,
    pendingSalePaidSgd: Math.round(h.pendingSalePaidSgd * 100) / 100,
  }));
}
 
/**
 * Validate an admin-entered MAG listing photo URL. Returns the URL to store,
 * or null (no photo) for anything unacceptable.
 *
 * Only hosts the site CSP `img-src` actually allows are accepted: a same-origin
 * path ("/images/…", never protocol-relative "//"), or https on our Supabase
 * storage. Anything else would be stored happily and then render as a broken
 * image for every customer (blocked by CSP), and would make the customer's
 * browser call a host we don't control.
 */
export function normaliseMagPhotoUrl(raw: unknown): string | null {
  if (typeof raw !== "string") return null;
  const v = raw.trim();
  if (!v || v.length > 500) return null;
  if (v.startsWith("/") && !v.startsWith("//") && !v.includes("\\")) return v;
  let u: URL;
  try { u = new URL(v); } catch { return null; }
  if (u.protocol !== "https:" || u.username || u.password) return null;
  const host = u.hostname.toLowerCase();
  if (host.endsWith(".supabase.co")) return u.toString();
  return null;
}
