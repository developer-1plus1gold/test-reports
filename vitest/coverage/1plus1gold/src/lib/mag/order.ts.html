/**
 * 1+1Gold MAG (My Allocated Gold) — the order-path helpers.
 * Design: docs/allocated-gold-design.md (v3).
 *
 * A MAG cart line is `{ productType:"mag", productId:<bar type>, grams:1,
 * quantity:<grams wanted>, magListingId:<one physical serial> }`. The money is
 * priced by the ordinary kernel (priceOrder → standard tiers); this module owns
 * only the GRAMS: validating the cart, holding grams while a payment is open,
 * and turning the hold into an allocation once the order exists.
 *
 * Over-allocation is impossible by construction — the DB trigger
 * `mag_allocations_guard` locks the listing row and raises P0001
 * `mag_over_allocation` when live grams would exceed the bar. Everything here
 * treats that error as "sold out", never as a crash.
 */

/** True when a Postgres error is the trigger's over-allocation / inactive refusal. */
export function isMagSoldOutError(err: { code?: string; message?: string } | null | undefined): boolean {
  return !!err && (err.code === "P0001" || /mag_over_allocation|not active/i.test(err.message ?? ""));
}
 
/**
 * Reserve the grams for an open payment session. All lines are inserted in ONE
 * statement, so a sold-out line rolls back the whole hold.
 */
export async function holdMagGrams(
  supabase: SupabaseClient,
  args: { customerId: string; holdRef: string; lines: MagLine[]; minutes?: number },
): Promise<{ ok: true } | { ok: false; soldOut: boolean; error: string }> {
  const expires = new Date(Date.now() + (args.minutes ?? MAG_HOLD_MINUTES) * 60_000).toISOString();
  const { error } = await supabase.from("mag_allocations").insert(
    args.lines.map(l => ({
      listing_id:      l.listingId,
      customer_id:     args.customerId,
      grams:           l.grams,
      cost_per_g_sgd:  l.costPerG,
      status:          "held",
      hold_ref:        args.holdRef,
      hold_expires_at: expires,
    })),
  );
  if (!error) return { ok: true };
  if (isMagSoldOutError(error)) {
    return { ok: false, soldOut: true, error: "Someone just bought these grams — please refresh your cart." };
  }
  console.error("[mag] hold failed:", error.message);
  return { ok: false, soldOut: false, error: "Could not reserve your gold — please try again." };
}
 
/**
 * Turn a hold into the customer's allocation once the order exists. Each held
 * row is matched to its order_item by listing (one line per listing). If a hold
 * expired and the grams were taken meanwhile, the trigger refuses — reported
 * back so the caller can page an operator (the customer has paid).
 */
export async function allocateMagHold(
  supabase: SupabaseClient,
  args: {
    holdRef: string;
    orderId: string;
    /** listingId → { orderItemId, pricePerG } */
    byListing: Map<string, { orderItemId: string | null; pricePerG: number | null }>;
  },
): Promise<{ ok: true; allocated: number } | { ok: false; error: string }> {
  const { data: held, error } = await supabase
    .from("mag_allocations")
    .select("id, listing_id")
    .eq("hold_ref", args.holdRef)
    .eq("status", "held");
  if (error) return { ok: false, error: error.message };
  const now = new Date().toISOString();
  let allocated = 0;
  for (const row of (held ?? []) as { id: string; listing_id: string }[]) {
    const link = args.byListing.get(row.listing_id);
    const { error: upErr } = await supabase
      .from("mag_allocations")
      .update({
        status:          "allocated",
        order_id:        args.orderId,
        order_item_id:   link?.orderItemId ?? null,
        price_per_g_sgd: link?.pricePerG ?? null,
        allocated_at:    now,
        hold_expires_at: null,
      })
      .eq("id", row.id)
      .eq("status", "held");
    if (upErr) return { ok: false, error: upErr.message };
    allocated++;
  }
  if (allocated === 0) return { ok: false, error: `no held grams for ${args.holdRef}` };
  return { ok: true, allocated };
}
 
/** Give held grams back (payment failed / expired / order creation failed). */

    .from("mag_allocations")
    .update({ status: "released", closed_at: new Date().toISOString() })

/**
 * Map each MAG listing to its order_item id + per-gram price, read back from the
 * order. One line per listing; the order_item for a listing is found by product
 * + quantity (a MAG cart cannot contain the same product twice at equal grams
 * for two listings without the per-line magListingId — so fall back to order).
 */
/**
 * Never throws: the customer has already paid, so allocation must proceed even
 * if the link read fails. Returns an empty map on failure (grams still
 * allocate, unlinked) and `linkError` so the caller pages an operator.
 */
export async function linkMagOrderItemsSafe(
  supabase: SupabaseClient,
  orderId: string,
  lines: MagLine[],
): Promise<{ byListing: Map<string, { orderItemId: string | null; pricePerG: number | null }>; linkError: string | null }> {
  try {
    const byListing = await linkMagOrderItems(supabase, orderId, lines);
    const missing = lines.filter(l => !byListing.get(l.listingId)?.orderItemId).map(l => l.listingId);
    return { byListing, linkError: missing.length ? `no order item matched listing(s) ${missing.join(",")}` : null };
  } catch (e) {
    return { byListing: new Map(), linkError: (e as Error).message };
  }
}
 
export async function linkMagOrderItems(
  supabase: SupabaseClient,
  orderId: string,
  lines: MagLine[],
): Promise<Map<string, { orderItemId: string | null; pricePerG: number | null }>> {
  // ⚠️ order_items has NO created_at column. Selecting / ordering by it made
  // this query error on every call; the error was discarded and every MAG
  // allocation was written with order_item_id + price NULL, which later broke
  // sell-back and redeem (2026-09-28). Order by `id` purely for determinism,
  // and never swallow the error: an unlinked allocation cannot be sold back.
  const { data, error } = await supabase
    .from("order_items")
    .select("id, product_id, quantity, price_sgd")
    .eq("order_id", orderId)
    .order("id", { ascending: true });
  if (error) {
    console.error("[mag] order-item link query failed:", orderId, error.message);
    throw new Error(`MAG order-item link failed: ${error.message}`);
  }
  const rows = (data ?? []) as { id: string; product_id: string | null; quantity: number; price_sgd: number }[];
  const used = new Set<string>();
  const out = new Map<string, { orderItemId: string | null; pricePerG: number | null }>();
  for (const l of lines) {
    const hit = rows.find(r => !used.has(r.id) && r.product_id === l.productId && Number(r.quantity) === l.grams);
    if (hit) used.add(hit.id);
    out.set(l.listingId, { orderItemId: hit?.id ?? null, pricePerG: hit ? Number(hit.price_sgd) : null });
  }
  return out;
}
