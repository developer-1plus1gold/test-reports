/**
 * MAG own-bar / group-pool QUOTE PRICING (Mukesh 2026-11-27,
 * docs/mag-quote-automation-design.md §3).
 *
 * We buy the whole bar on day one; the customer pays for it gram by gram over the
 * duration. The money tied up in the unsold grams is the only cost an outright
 * sale does not have, so the quote that earns the outright price is
 *
 *     suggested %  =  outright % (1 bar)  +  cost of capital %
 *     cost of capital %  =  rate × years × ½
 *
 * The ½ is the equal spread (Mukesh: "based on the equal spread of the grams that
 * the customer will purchase over the duration"): on average half the bar is
 * unsold. The rate is Admin → MAG "Cost of capital (% a year)", 7 to start.
 *
 * Not covered, deliberately (same as every MAG bar today): the customer pays the
 * spot of the day they buy, so a fall in gold on unsold grams is ours.
 *
 * Pure — shared by the admin quote form, the pool-offer table and the tests.
 */
import { QUOTE_DURATIONS, type DurationUnit } from "./pool-offers";
 
export const DEFAULT_COST_OF_CAPITAL_PCT = 7;
/** Suggested % is rounded UP to this step, so rounding never eats margin. */
export const QUOTE_ROUND_STEP_PCT = 0.05;
 
/** The duration as a fraction of a year: N weeks = 7N/365, N months = N/12. */
export function durationYears(value: number, unit: DurationUnit): number {
  return unit === "weeks" ? (value * 7) / 365 : value / 12;
}
 
/** Cost of carrying the unsold grams, % of the bar's value, for an even spread. */
export function costOfCapitalPct(ratePctPerYear: number, value: number, unit: DurationUnit): number {
  if (!Number.isFinite(ratePctPerYear) || ratePctPerYear <= 0) return 0;
  return ratePctPerYear * durationYears(value, unit) * 0.5;
}
 
/** Round up to the next QUOTE_ROUND_STEP_PCT (tolerant of float noise: 3.30000001 stays 3.30). */
export function roundUpToStep(pct: number, step: number = QUOTE_ROUND_STEP_PCT): number {
  const n = Math.ceil(pct / step - 1e-9);
  return Math.round(n * step * 1000) / 1000;
}
 
/** The % that earns the outright price after carrying the bar for the duration. */
export function suggestedQuotePct(outrightPct: number, ratePctPerYear: number, value: number, unit: DurationUnit): number {
  return roundUpToStep(outrightPct + costOfCapitalPct(ratePctPerYear, value, unit));
}
 
/** Margin left at `quotePct`, % of value: quote − bar cost − cost of capital. */
export function marginLeftPct(quotePct: number, costPct: number, ratePctPerYear: number, value: number, unit: DurationUnit): number {
  return quotePct - costPct - costOfCapitalPct(ratePctPerYear, value, unit);
}
 
export interface QuoteOptionCalc {
  durationValue: number;
  durationUnit: DurationUnit;
  costOfCapitalPct: number;
  suggestedPct: number;
  /** Margin at the suggested % — equals the outright margin, plus rounding. */
  marginAtSuggestedPct: number | null;
}
 
/**
 * The suggestion for every quotable duration (2 weeks, 1, 2, 3, 6 months), so
 * the quote can offer the customer alternatives side by side.
 */
export function quoteOptionsFor(outrightPct: number, costPct: number | null, ratePctPerYear: number): QuoteOptionCalc[] {
  return QUOTE_DURATIONS.map((d) => {
    const coc = costOfCapitalPct(ratePctPerYear, d.durationValue, d.durationUnit);
    const suggestedPct = suggestedQuotePct(outrightPct, ratePctPerYear, d.durationValue, d.durationUnit);
    return {
      durationValue: d.durationValue,
      durationUnit: d.durationUnit,
      costOfCapitalPct: Math.round(coc * 1000) / 1000,
      suggestedPct,
      marginAtSuggestedPct: costPct == null ? null : Math.round(marginLeftPct(suggestedPct, costPct, ratePctPerYear, d.durationValue, d.durationUnit) * 1000) / 1000,
    };
  });
}
 
/** Parse the admin setting: a % a year from 0 to 50; anything else → the default. */
export function parseCostOfCapitalPct(v: unknown): number {
  if (v === null || v === undefined || v === "") return DEFAULT_COST_OF_CAPITAL_PCT;
  const n = Number(v);
  return Number.isFinite(n) && n >= 0 && n <= 50 ? n : DEFAULT_COST_OF_CAPITAL_PCT;
}
