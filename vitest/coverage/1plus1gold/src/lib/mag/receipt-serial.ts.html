/**
 * MAG — the bar's serial on customer documents (testing item 17,
 * 2026-09-28). A MAG line is a share of ONE serial-numbered bar, so the
 * receipt and the order-history PDF name that serial beside the line.
 *
 * WHICH bar (2026-11-13, MAG Agreement §4A): a receipt is the record of a sale,
 * so a MAG line names the bar its grams were BOUGHT from, even after an admin
 * moved them to another bar ("Move a holder" re-points mag_allocations.listing_id).
 * The bar bought from is the `from_serial` of the line's EARLIEST transfer in
 * `mag_transfers` (append-only); a line never moved is on the bar it was bought
 * from. Each move adds a note under the line (lib/mag/line-label `magMoveNote`).
 *
 * Resolved per ORDER LINE, not per allocation row: a line holds grams of one bar
 * (one order_items row per listing), and a partial sell-back after a move splits
 * a row by INSERT — the new row is not in any transfer's `allocation_ids`, but it
 * is still the same line, bought from the same bar.
 *
 * Two queries for any number of lines (allocations, then the transfers that
 * touch them). Fail-soft: a failed lookup returns an empty map — the document
 * still renders, just without the serial, rather than blocking a receipt. A failed
 * TRANSFER lookup falls back to the bar the grams are on now (the pre-§4A wording).
 */
import type { SupabaseClient } from "@supabase/supabase-js";
 
/** One allocation row of an order line, on the bar it is on now. */
export interface MagLineAllocation {
  id: string | null;
  orderItemId: string;
  grams: number;
  serial: string;
}
 
/** One `mag_transfers` row (the columns the receipt needs). */
export interface MagLineTransfer {
  allocationIds: string[];
  fromSerial: string;
  toSerial: string;
  grams: number;
  at: string;
}
 
/** A move as the line sees it: how many of the LINE's grams went, and where. */
export interface MagLineMove {
  grams: number;
  /** Every gram of the line moved (the note then says "Moved", not "6 g moved"). */
  wholeLine: boolean;
  toSerial: string;
  at: string;
}
 
export interface MagLineBar {
  /** The bar(s) the line was bought from (NFC serials, de-duplicated). */
  serials: string[];
  /** Moves of the line's grams, oldest first. */
  moves: MagLineMove[];
}
 
/**
 * Pure: each order line's bar-bought-from and its moves.
 *
 * Grams per move: when every row the transfer moved belongs to this line, the
 * transfer's own `grams` (exact even if a row was split later); when it moved
 * several lines' rows together, the sum of this line's moved rows as they stand.
 */

/** Order line → the bar(s) it was bought from, joined with ", ". */
export async function fetchMagSerialsByOrderItem(
  supabase: SupabaseClient,
  orderItemIds: string[],
): Promise<Map<string, string>> {
  const lines = await fetchMagLineBars(supabase, orderItemIds);
  return new Map([...lines].map(([k, v]) => [k, v.serials.join(", ")]));
}
 
/** "MAG in 100 gram bar" + "PAMP-AB12-CD34" → "MAG in 100 gram bar · Serial PAMP-AB12-CD34". */
export function labelWithMagSerial(label: string, serial: string | undefined | null): string {
  return serial ? `${label} · Serial ${serial}` : label;
}
