/**
 * MAG — moving a holder's grams from one OPEN bar to another (MAG Agreement §4A,
 * Mukesh 2026-11-13). Admin-only: Admin → MAG → "Move a holder".
 *
 * The database is the authority: `mag_transfer_holding` (migration
 * 20261113_mag_transfers.sql) re-checks every rule under row locks and raises
 * `mag_transfer_refused: <code>`. This module mirrors those rules so the admin
 * screen can say up front WHY a holder or a bar is not eligible, and owns the
 * one message table both paths use — the route maps the SQL code through
 * `transferRefusalMessage`, so the wording cannot drift between them.
 *
 * Pure (no server imports): the admin page, the route and the account views
 * all import it.
 */
import { barRefName } from "@/lib/serials/bar-serial";
import { sgtLongDate } from "./line-label";
 
export const TRANSFER_REASONS = ["consolidate", "sell_or_deliver_bar", "vault_or_audit", "other"] as const;
export type TransferReason = (typeof TRANSFER_REASONS)[number];
 
export const TRANSFER_REASON_LABELS: Record<TransferReason, string> = {
  consolidate:         "Consolidate the customer's grams onto one bar",
  sell_or_deliver_bar: "We need the bar free to sell or deliver it whole",
  vault_or_audit:      "Vault or audit reason",
  other:               "Other (note required)",
};
 
export const TRANSFER_NOTE_MAX = 500;
 
export function isTransferReason(v: unknown): v is TransferReason {
  return typeof v === "string" && (TRANSFER_REASONS as readonly string[]).includes(v);
}
 
/** Every refusal the SQL function can raise, with the admin-facing message. */
export const TRANSFER_REFUSALS = {
  invalid:                "The request is incomplete",
  same_bar:               "Pick a different bar to move the grams to",
  invalid_reason:         "Choose a reason for the move",
  note_required:          "Add a note — the reason is “Other”",
  source_delivered:       "This bar has been delivered — its grams left the vault with it",
  source_not_active:      "Only a bar that is on sale (active) can have grams moved off it",
  source_not_open:        "Grams can be moved only off an open bar, not a reserved or group bar",
  dest_not_active:        "That bar is not on sale (active)",
  dest_not_open:          "That bar is reserved or a group bar — grams can be moved only to an open bar",
  different_type:         "That bar is a different bar type — grams move only between bars of the same type",
  different_piece:        "That bar has a different piece weight",
  different_stock_source: "One bar is consignment stock and the other is ours — grams cannot move between them",
  delivery_in_progress:   "A delivery checkout is holding one of the bars — try again once it ends",
  owns_fraction:          "The customer owns the part of a gram on this piece — it stays with its piece, so their grams cannot move",
  fraction_in_checkout:   "Someone is buying the rest of this piece right now — try again once that checkout ends",
  checkout_open:          "The customer has a checkout open on this bar — try again once it is paid or expires",
  sell_back_open:         "The customer has an open sell-back or a checkout using these grams — finish or cancel it first",
  no_grams:               "The customer holds no grams on this bar",
  dest_full:              "That bar does not have enough unsold grams for all of the customer's grams",
} as const;
 
export type TransferRefusal = keyof typeof TRANSFER_REFUSALS;
 
export function transferRefusalMessage(code: TransferRefusal): string {
  return TRANSFER_REFUSALS[code];
}
 
/** "mag_transfer_refused: dest_full" (a Postgres error message) → "dest_full"; null when it is not a refusal. */
export function parseTransferRefusal(message: string | null | undefined): TransferRefusal | null {
  const m = /mag_transfer_refused:\s*([a-z_]+)/.exec(message ?? "");
  const code = m?.[1];
  return code && code in TRANSFER_REFUSALS ? (code as TransferRefusal) : null;
}
 
/** One bar, as the rules need to see it. */
export interface TransferBar {
  id: string;
  productId: string;
  serial: string;
  manufacturerSerial: string | null;
  label: string;
  status: string;
  accessType: string;
  pieceGrams: number | null;
  gramsTotal: number;
  /** Unsold whole grams (mag_listing_availability.grams_remaining). */
  gramsRemaining: number;
  /** A delivery checkout holds the bar now (delivery_hold_* not expired). */
  deliveryHoldLive: boolean;
  /** Who owns the part of a gram on the piece (null = 1Plus1Gold, or no fraction). */
  houseOwnerCustomerId: string | null;
  /** The fraction is in a live "Buy the rest" checkout. */
  houseHoldLive: boolean;
  /** Backed by consignment stock (inventory.source_type = 'consignment'). */
  consignment: boolean;
}
 
/** One customer's grams on the source bar, by state. */
export interface TransferHolder {
  customerId: string;
  allocatedGrams: number;
  /** Grams in a checkout hold that has not expired. */
  heldGrams: number;
  /** Grams in an open sell-back or a checkout redeem (sell_requested). */
  pendingGrams: number;
}
 
/**
 * Why this holder's grams cannot leave this bar, or null. The same rules as the
 * SQL function; when several apply, the screen may name a different one first.
 */
export function sourceRefusal(source: TransferBar, holder: TransferHolder): TransferRefusal | null {
  if (source.status === "delivered") return "source_delivered";
  if (source.status !== "active") return "source_not_active";
  if (source.accessType !== "public") return "source_not_open";
  if (source.deliveryHoldLive) return "delivery_in_progress";
  if (source.houseOwnerCustomerId && source.houseOwnerCustomerId === holder.customerId) return "owns_fraction";
  if (source.houseHoldLive) return "fraction_in_checkout";
  if (holder.heldGrams > 0) return "checkout_open";
  if (holder.pendingGrams > 0) return "sell_back_open";
  if (holder.allocatedGrams <= 0) return "no_grams";
  return null;
}
 
/** Why `dest` cannot take `grams` from `source`, or null. */
export function destinationRefusal(source: TransferBar, dest: TransferBar, grams: number): TransferRefusal | null {
  if (dest.id === source.id) return "same_bar";
  if (dest.status !== "active") return "dest_not_active";
  if (dest.accessType !== "public") return "dest_not_open";
  if (dest.productId !== source.productId) return "different_type";
  if ((dest.pieceGrams ?? null) !== (source.pieceGrams ?? null)) return "different_piece";
  if (dest.consignment !== source.consignment) return "different_stock_source";
  if (dest.deliveryHoldLive) return "delivery_in_progress";
  if (dest.gramsRemaining < grams) return "dest_full";
  return null;
}
 
export interface TransferOption { bar: TransferBar; refusal: TransferRefusal | null }
 
/**
 * The bars a holder's grams could go to: eligible ones first (most room first),
 * then the rest, each with its reason. When the holder cannot move at all, every
 * bar carries the holder's reason, so nothing reads as pickable.
 */
export function transferOptions(source: TransferBar, holder: TransferHolder, candidates: TransferBar[]): TransferOption[] {
  const blocked = sourceRefusal(source, holder);
  return candidates
    .filter((b) => b.id !== source.id)
    .map((bar) => ({ bar, refusal: blocked ?? destinationRefusal(source, bar, holder.allocatedGrams) }))
    .sort((a, b) =>
      Number(a.refusal !== null) - Number(b.refusal !== null)
      || b.bar.gramsRemaining - a.bar.gramsRemaining
      || a.bar.serial.localeCompare(b.bar.serial));
}
 
/** A body the route accepts: trimmed, capped note; null when it is empty. */
export function cleanTransferNote(v: unknown): string | null {
  if (typeof v !== "string") return null;
  const t = v.replace(/\s+/g, " ").trim().slice(0, TRANSFER_NOTE_MAX);
  return t || null;
}
 
/** The refusal for a reason + note pair, or null (mirrors the SQL guard). */
export function reasonRefusal(reason: unknown, note: string | null): TransferRefusal | null {
  if (!isTransferReason(reason)) return "invalid_reason";
  if (reason === "other" && !note) return "note_required";
  return null;
}
 
/** A bar named for a customer: refinery bar no. first, then our serial. */
export function barName(serial: string, manufacturerSerial?: string | null): string {
  return barRefName(serial, manufacturerSerial);
}
 
/** "13 November 2026" in Singapore time. */
export function transferDate(iso: string): string {
  return sgtLongDate(iso);
}
 
/** What the customer sees: one moved holding. */
export interface MagTransferRecord {
  id: string;
  grams: number;
  fromSerial: string;
  fromManufacturerSerial: string | null;
  toSerial: string;
  toManufacturerSerial: string | null;
  at: string;
}
 
/** "Moved 12 g from bar B240888 (serial 100G-…) to bar B240999 (serial 100G-…) on 13 November 2026." */
export function transferLine(r: MagTransferRecord): string {
  return `Moved ${r.grams} g from bar ${barName(r.fromSerial, r.fromManufacturerSerial)} to bar ${barName(r.toSerial, r.toManufacturerSerial)} on ${transferDate(r.at)}.`;
}
