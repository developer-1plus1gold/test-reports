/**
 * MAG type cards + requests (operator 2026-10-25): the Shop always shows the
 * three ways to hold MAG, each with a short explanation and a request action,
 * even when no bar of that type is listed. A request is recorded in
 * `mag_pool_requests` and listed in Admin → MAG, where an admin sets up the bar
 * and its members. Design: docs/mag-type-cards-design.md.
 *
 * One definition for web (MagTypeCards) and the native Shop spec, so the copy
 * cannot drift between surfaces.
 */
export type MagPoolRequestType = "notify" | "own_bar" | "group_pool";
export type MagPoolRequestStatus = "open" | "done" | "declined";
 
export interface MagTypeCard {
  key: "public" | "reserved" | "group";
  title: string;
  copy: string;
  requestType: MagPoolRequestType;
  cta: string;
}
 
export const MAG_TYPE_CARDS: readonly MagTypeCard[] = [
  {
    key: "public", title: "Open pool", requestType: "notify", cta: "Notify me",
    copy: "Buy any number of grams from a shared bar open to everyone. If none is listed, ask us to tell you when the next bar opens.",
  },
  {
    key: "reserved", title: "Your own bar", requestType: "own_bar", cta: "Request my own bar",
    copy: "A whole serial-numbered bar reserved for you alone, bought in grams over a window you choose.",
  },
  {
    key: "group", title: "Group pool", requestType: "group_pool", cta: "Request a group pool",
    copy: "One bar shared by a group you name — family or friends — with each member buying their own grams.",
  },
] as const;
 
export const MAX_REQUEST_GRAMS = 1000;
const MAX_TEXT = 300;
 
export interface MagPoolRequestInput {
  requestType: MagPoolRequestType;
  grams: number | null;
  groupName: string | null;
  members: string | null;
  note: string | null;
}
 
/** Trim, drop control/bidi characters, cap length; empty → null. */
function cleanText(v: unknown, max = MAX_TEXT): string | null {
  if (typeof v !== "string") return null;
  // eslint-disable-next-line no-control-regex
  const s = v.replace(/[\u0000-\u001f\u007f‎‏‪-‮⁦-⁩]/g, " ").trim().slice(0, max);
  return s.length ? s : null;
}
 
export function validateMagPoolRequest(body: unknown):
  { ok: true; value: MagPoolRequestInput } | { ok: false; error: string } {
  const b = (body && typeof body === "object" ? body : {}) as Record<string, unknown>;
  const t = b.requestType;
  if (t !== "notify" && t !== "own_bar" && t !== "group_pool") return { ok: false, error: "Unknown request type" };
 
  let grams: number | null = null;
  if (b.grams !== undefined && b.grams !== null && b.grams !== "") {
    const g = Number(b.grams);
    if (!Number.isInteger(g) || g < 1 || g > MAX_REQUEST_GRAMS) {
      return { ok: false, error: `Grams must be a whole number from 1 to ${MAX_REQUEST_GRAMS}` };
    }
    grams = g;
  }
  const groupName = cleanText(b.groupName, 80);
  const members = cleanText(b.members);
  const note = cleanText(b.note);
  if (t === "group_pool" && !groupName) return { ok: false, error: "Please name your group" };
  return {
    ok: true,
    value: {
      requestType: t,
      grams: t === "notify" ? null : grams,
      groupName: t === "group_pool" ? groupName : null,
      members: t === "group_pool" ? members : null,
      note,
    },
  };
}
