/**
 * Admin → MAG — filtering the bar list (Mukesh, 2026-11-10: "difficult to figure
 * out bars of different status … including pending audit and audit dates").
 *
 * Audit rule: a bar that customers hold grams of is DUE for its audit when no
 * audit has been signed for it in the current calendar month (Singapore time) —
 * the same monthly period the vault audit runs on (lib/vault/audit-reminder.ts).
 * A bar nobody holds grams of is never due.
 *
 * Pure: the page filters what it already loaded; the counts on each chip come
 * from the same function, so a chip never says 3 and shows 2.
 */
 
export type StatusFilter = "all" | "active" | "draft" | "retired" | "delivered";
export type PoolFilter = "all" | "public" | "reserved" | "group";
export type AuditFilter = "all" | "due" | "never" | "audited";
export type HoldersFilter = "all" | "held" | "empty";
 
export interface AdminBarFilters {
  status: StatusFilter;
  pool: PoolFilter;
  audit: AuditFilter;
  holders: HoldersFilter;
  /** Matches serial, bar no. (refinery serial), bar type or group name. */
  q: string;
}
 
export const DEFAULT_FILTERS: AdminBarFilters = { status: "all", pool: "all", audit: "all", holders: "all", q: "" };
 
export interface AdminBarRow {
  status: string;
  access_type: string;
  serial: string;
  product_label: string | null;
  group_name: string | null;
  manufacturer_serial?: string | null;
  /** Grams customers hold now: allocated (incl. open sell-backs) + live checkout holds. */
  grams_allocated: number;
  grams_held: number;
  last_audit_at: string | null;
}
 
/** "2026-11" — the calendar month in Singapore time. */
function sgMonth(iso: string | number | Date): string {
  const d = new Date(iso);
  const parts = new Intl.DateTimeFormat("en-CA", { timeZone: "Asia/Singapore", year: "numeric", month: "2-digit" }).formatToParts(d);
  return `${parts.find(p => p.type === "year")?.value}-${parts.find(p => p.type === "month")?.value}`;
}
 
export function holdsGrams(r: Pick<AdminBarRow, "grams_allocated" | "grams_held">): boolean {
  return Number(r.grams_allocated) + Number(r.grams_held) > 0;
}
 
export type AuditState = "due" | "ok" | "not_needed";
 
/** Due = customers hold grams and nothing signed this month; ok = signed this month. */
export function auditState(r: Pick<AdminBarRow, "grams_allocated" | "grams_held" | "last_audit_at" | "status">, nowMs = Date.now()): AuditState {
  if (r.status === "delivered" || !holdsGrams(r)) return "not_needed";
  return r.last_audit_at && sgMonth(r.last_audit_at) === sgMonth(nowMs) ? "ok" : "due";
}
 
export function matchesFilters(r: AdminBarRow, f: AdminBarFilters, nowMs = Date.now()): boolean {
  if (f.status !== "all" && r.status !== f.status) return false;
  if (f.pool !== "all" && r.access_type !== f.pool) return false;
  if (f.holders === "held" && !holdsGrams(r)) return false;
  Iif (f.holders === "empty" && holdsGrams(r)) return false;
  if (f.audit === "due" && auditState(r, nowMs) !== "due") return false;
  if (f.audit === "never" && r.last_audit_at) return false;
  Iif (f.audit === "audited" && !r.last_audit_at) return false;
  const q = f.q.trim().toLowerCase();
  if (q) {
    const hay = [r.serial, r.manufacturer_serial, r.product_label, r.group_name].filter(Boolean).join(" ").toLowerCase();
    if (!hay.includes(q)) return false;
  }
  return true;
}
 
/** How many rows each option would show, given the OTHER filters (chip counts). */
export function countFor<K extends keyof Omit<AdminBarFilters, "q">>(
  rows: AdminBarRow[], f: AdminBarFilters, key: K, value: AdminBarFilters[K], nowMs = Date.now(),
): number {
  const g = { ...f, [key]: value } as AdminBarFilters;
  return rows.filter(r => matchesFilters(r, g, nowMs)).length;
}
