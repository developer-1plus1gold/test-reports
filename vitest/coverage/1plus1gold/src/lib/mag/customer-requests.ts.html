export async function loadCustomerRequestViews(db: SupabaseClient, customerId: string, now: Date = new Date()): Promise<CustomerRequestView[]> {
  const { data, error } = await db.from("mag_pool_requests")
    .select("id, request_type, stage, status, group_name, quote_grams, duration_value, duration_unit, premium_pct, quote_options, quote_expires_at, quoted_at, accepted_at, declined_at, product_id, listing_id, created_at, resolved_at, schedule_consent_at, products:product_id(label)")
    .eq("customer_id", customerId).neq("request_type", "notify")
    .order("created_at", { ascending: false }).limit(20);
  if (error) throw new Error(error.message);
  const rows = (data ?? []) as unknown as CustomerRequestRow[];

  // weight, shop page and one-bar outright premium. Read only when a quote is open.
  const quotedIds = [...new Set(rows.filter((r) => r.stage === "quoted" && r.product_id).map((r) => r.product_id!))];
  const prices = quotedIds.length ? await loadPricing(db, quotedIds, now) : null;
  // The buying schedule (Agreement §1B): on the quote while the rules are on; on a live
  // bar whose customer agreed to it.
  const enabled = await rhythmEnabled(db);
  const scheduled = rows.filter((r) => r.stage === "live" && r.listing_id && (r as { schedule_consent_at?: string | null }).schedule_consent_at);

  const [{ data: rhythm }, held] = listingIds.length
    ? await Promise.all([
        db.from("mag_listings").select("id, rhythm_schedule, rhythm_state, window_ends_at").in("id", listingIds),
        heldOnListings(db, listingIds, now),
      ])
    : [{ data: [] as unknown[] }, new Map<string, { held: number; pending: number }>()];
  const rhythmBy = new Map(((rhythm ?? []) as { id: string; rhythm_schedule: unknown; rhythm_state: unknown; window_ends_at: string | null }[]).map((x) => [x.id, x]));
  return Promise.all(rows.map(async (r) => {

    const pr = prices && r.product_id ? prices.products.get(r.product_id) : null;
    if (prices && pr) v = withPrices(v, { spotSgdPerG: prices.spotSgdPerG, spotAt: prices.spotAt, ...pr });
    if (enabled && v.canRespond && r.quote_grams != null) v = withSchedule(v, Math.round(Number(r.quote_grams)));

}

/** Where a released customer's grams are now: the bar the release moved them to (their
 *  mag_transfers row from the rhythm), else this bar if they still hold grams on it. */
async function releasedDestination(db: SupabaseClient, listingId: string, customerId: string): Promise<ReleasedTo | null> {
  const { data: moved } = await db.from("mag_transfers").select("to_listing_id, to_serial, grams")
    .eq("from_listing_id", listingId).eq("customer_id", customerId).eq("actor", RHYTHM_AGENT)
    .order("created_at", { ascending: false }).limit(1).maybeSingle();
  const m = moved as { to_listing_id: string; to_serial: string; grams: number } | null;
  const where = m ? m.to_listing_id : listingId;
  const { data: bar } = await db.from("mag_listings").select("serial, product_id, products(label)").eq("id", where).maybeSingle();
  const b = bar as { serial: string; product_id: string; products: { label: string } | null } | null;
  if (!b) return null;
  let grams = m ? Number(m.grams) : 0;
  if (!m) {
    const { data: rows } = await db.from("mag_allocations").select("grams").eq("listing_id", listingId).eq("customer_id", customerId).eq("status", "allocated");
    grams = ((rows ?? []) as { grams: number }[]).reduce((t, x) => t + Number(x.grams), 0);
  }
  const { data: tier } = await db.from("standard_premium_tiers").select("premium_pct").eq("product_id", b.product_id)
    .order("min_qty", { ascending: true }).limit(1).maybeSingle();
  const pct = (tier as { premium_pct?: unknown } | null)?.premium_pct;
  return { grams, serial: b.serial, label: (b.products?.label ?? "MAG bar").split(" — ")[0], pct: pct == null ? null : Number(pct), moved: !!m };
}
