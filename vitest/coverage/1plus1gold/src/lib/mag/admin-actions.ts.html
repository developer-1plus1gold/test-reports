/**
 * Admin → MAG → "Needs your action" (Mukesh 2026-11-28). One list of what an admin has
 * to do or watch on MAG, gathered from the same rows the sections show, so the top of
 * the screen and the sections can never disagree. Pure: the route reads, this sorts.
 *
 *   act   (red)  — only an admin can move it: a new request to quote, an accepted quote
 *                  to confirm, a set-up that stopped, a bar audit due, and — only while
 *                  MAG is invite-only — a customer asking for access.
 *   watch (gold) — waiting on someone else: a bar on order from BullionKing, a quote
 *                  about to expire, a reserved bar behind its schedule, a consolidation.
 *
 * Each item names the SECTION that handles it (`target`), so the screen can open and
 * scroll to it.
 */
 
export type MagActionTone = "act" | "watch";
export type MagActionTarget = "requests" | "bars" | "schedules" | "moves" | "settings";
 
export interface MagActionItem {
  key: string;
  tone: MagActionTone;
  title: string;
  detail: string;
  target: MagActionTarget;
  /** Sort key — the oldest / most urgent first within a tone. */
  at: string;
}
 
export interface ActionRequestRow {
  id: string; customerName: string; requestType: "notify" | "own_bar" | "group_pool";
  stage: string; status: string; createdAt: string; quoteExpiresAt: string | null;
  setupError: string | null; label: string;
}
export interface ActionBarRow {
  id: string; serial: string; label: string; status: string;
  grams_allocated: number; grams_held: number; last_audit_at: string | null;
}
export interface ActionScheduleRow {
  listingId: string; serial: string; label: string; mode: "enforced" | "dry_run" | "legacy";
  strikes: number; catchUp: { dueAt?: string; grams?: number } | null; held: number;
}
export interface ActionConsolidationRow { customerName: string; label: string; kind: "full" | "partial" | "blocked"; gramsMoved: number }
export interface ActionAccessRequest { customerName: string; requestedAt: string }
 
export interface MagActionInputs {
  requests: ActionRequestRow[];
  /** Audit state already decided by lib/mag/admin-filters auditState (only "due" rows). */
  auditDue: ActionBarRow[];
  schedules: ActionScheduleRow[];
  consolidations: ActionConsolidationRow[];
  /** Only read while MAG is invite-only (mag_settings.open_to_all = false). */
  accessRequests: ActionAccessRequest[];
  inviteOnly: boolean;
  nowMs: number;
}
 
const DAY = 86_400_000;
/** Sort key for an item with no deadline (a consolidation): last within its tone. */
const NEVER_URGENT = "9999-12-31T00:00:00Z";
/** A quote expiring within this is worth a nudge (the reminder job sends one at 2 days too). */
export const QUOTE_EXPIRY_WATCH_DAYS = 2;
 
const KIND: Record<ActionRequestRow["requestType"], string> = { notify: "Notify me", own_bar: "Own bar", group_pool: "Group pool" };
const fmt = (iso: string) => new Date(iso).toLocaleDateString("en-GB", { day: "numeric", month: "short", timeZone: "Asia/Singapore" });
 
export function buildMagActionItems(i: MagActionInputs): MagActionItem[] {
  const out: MagActionItem[] = [];
 
  for (const r of i.requests) {
    if (r.status !== "open" || r.requestType === "notify") continue;
    const who = `${r.customerName} · ${KIND[r.requestType]} · ${r.label}`;
    if (r.setupError) {
      out.push({ key: `req-err-${r.id}`, tone: "act", title: "Set-up stopped", detail: `${who} — ${r.setupError}`, target: "requests", at: r.createdAt });
    } else if (r.stage === "open") {
      out.push({ key: `req-new-${r.id}`, tone: "act", title: "Send a quote", detail: who, target: "requests", at: r.createdAt });
    } else if (r.stage === "accepted") {
      out.push({ key: `req-acc-${r.id}`, tone: "act", title: "Confirm set-up", detail: `${who} — the customer accepted`, target: "requests", at: r.createdAt });
    } else if (r.stage === "awaiting_bar") {
      out.push({ key: `req-po-${r.id}`, tone: "watch", title: "Bar on order", detail: `${who} — listed when the purchase order is received`, target: "requests", at: r.createdAt });
    } else Eif (r.stage === "quoted" && r.quoteExpiresAt) {
      const left = Date.parse(r.quoteExpiresAt) - i.nowMs;
      if (left > 0 && left <= QUOTE_EXPIRY_WATCH_DAYS * DAY) {
        out.push({ key: `req-exp-${r.id}`, tone: "watch", title: "Quote expires soon", detail: `${who} — valid until ${fmt(r.quoteExpiresAt)}`, target: "requests", at: r.quoteExpiresAt });
      }
    }
  }
 
  for (const b of i.auditDue) {
    out.push({ key: `audit-${b.id}`, tone: "act", title: "Bar audit due", detail: `${b.serial} · ${b.label} — customers hold grams, not audited this month`, target: "bars", at: b.last_audit_at ?? "1970-01-01T00:00:00Z" });
  }
 
  for (const s of i.schedules) {
    if (s.mode !== "enforced") continue;
    if (s.catchUp?.dueAt) {
      out.push({ key: `sched-${s.listingId}`, tone: "watch", title: "Behind its buying schedule", detail: `${s.serial} · ${s.label} — catch up to ${s.catchUp.grams ?? "?"} g by ${fmt(s.catchUp.dueAt)} (holds ${s.held} g)`, target: "schedules", at: s.catchUp.dueAt });
    }
  }
 
  for (const c of i.consolidations) {
    if (c.kind === "blocked") continue;
    out.push({ key: `move-${c.customerName}-${c.label}`, tone: "watch", title: "Consolidation suggested", detail: `${c.customerName} · ${c.label} — ${c.gramsMoved} g can move onto one bar`, target: "moves", at: NEVER_URGENT });
  }
 
  if (i.inviteOnly) {
    for (const a of i.accessRequests) {
      out.push({ key: `access-${a.customerName}-${a.requestedAt}`, tone: "act", title: "Asks to buy MAG", detail: `${a.customerName} — MAG is invite-only`, target: "settings", at: a.requestedAt });
    }
  }
 
  const rank: Record<MagActionTone, number> = { act: 0, watch: 1 };
  return out.sort((a, b) => rank[a.tone] - rank[b.tone] || a.at.localeCompare(b.at));
}
