/**
 * 1+1Gold MAG — who holds a listing (admin holders view). Pure, so the grouping
 * is testable without a DB.
 *
 * Holders = customers with `allocated` or `sell_requested` grams (a pending
 * sell-back is still theirs). Grams that left through `sold_back` / `redeemed`
 * are reported as listing-level totals, and unexpired `held` grams as the
 * amount sitting in someone's checkout — none of those is a holder.
 */
export interface MagAllocationRow {
  customer_id:     string;
  grams:           number | string;
  status:          string;
  price_per_g_sgd: number | string | null;
  allocated_at:    string | null;
  hold_expires_at: string | null;
  order_id:        string | null;
}
 
export interface MagHolder {
  customerId:        string;
  grams:             number;   // allocated (sellable)
  pendingSaleGrams:  number;   // sell_requested
  paidSgd:           number;   // Σ grams × price, over both
  avgPricePerG:      number | null;
  firstAllocatedAt:  string | null;
  orders:            number;
}
 
export interface MagHoldersSummary {
  holders:        MagHolder[];
  gramsOwned:     number;
  gramsSoldBack:  number;
  gramsRedeemed:  number;
  gramsInCheckout: number;
}
 
export function summariseMagHolders(rows: MagAllocationRow[], nowMs: number): MagHoldersSummary {
  const by = new Map<string, MagHolder & { _orders: Set<string> }>();
  let soldBack = 0, redeemed = 0, inCheckout = 0;
  for (const r of rows) {
    const g = Number(r.grams) || 0;
    if (r.status === "sold_back") { soldBack += g; continue; }
    if (r.status === "redeemed")  { redeemed += g; continue; }
    if (r.status === "held") {
      if (r.hold_expires_at && Date.parse(r.hold_expires_at) > nowMs) inCheckout += g;
      continue;
    }
    if (r.status !== "allocated" && r.status !== "sell_requested") continue;
    const h = by.get(r.customer_id) ?? {
      customerId: r.customer_id, grams: 0, pendingSaleGrams: 0, paidSgd: 0,
      avgPricePerG: null, firstAllocatedAt: null, orders: 0, _orders: new Set<string>(),
    };
    if (r.status === "allocated") h.grams += g; else h.pendingSaleGrams += g;
    h.paidSgd += g * (Number(r.price_per_g_sgd) || 0);
    if (r.allocated_at && (!h.firstAllocatedAt || r.allocated_at < h.firstAllocatedAt)) h.firstAllocatedAt = r.allocated_at;
    Eif (r.order_id) h._orders.add(r.order_id);
    by.set(r.customer_id, h);
  }
  const holders = [...by.values()].map(({ _orders, ...h }) => {
    const total = h.grams + h.pendingSaleGrams;
    return {
      ...h,
      paidSgd: Math.round(h.paidSgd * 100) / 100,
      avgPricePerG: total > 0 ? Math.round((h.paidSgd / total) * 100) / 100 : null,
      orders: _orders.size,
    };
  }).sort((a, b) => (b.grams + b.pendingSaleGrams) - (a.grams + a.pendingSaleGrams));
  return {
    holders,
    gramsOwned: holders.reduce((s, h) => s + h.grams + h.pendingSaleGrams, 0),
    gramsSoldBack: soldBack,
    gramsRedeemed: redeemed,
    gramsInCheckout: inCheckout,
  };
}
