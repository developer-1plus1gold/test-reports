/** Admin: take back a quote before the bar is set up. */
export async function withdrawQuote(db: SupabaseClient, id: string): Promise<ActionResult<{ stage: MagRequestStage }>> {
  const r = await loadRequest(db, id);
  if (!r) return fail(404, "Request not found");
  if (!CAN.withdraw(r.stage)) return fail(409, `Nothing to withdraw at this stage (${r.stage})`);
  if (!(await moveStage(db, id, r.stage, "withdrawn"))) return fail(409, "The request changed meanwhile — reload and try again");
  return { ok: true, value: { stage: "withdrawn" } };
}
