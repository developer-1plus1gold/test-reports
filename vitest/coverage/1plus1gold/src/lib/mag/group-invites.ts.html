/**
 * MAG group-pool SHARE LINKS (docs/mag-quote-automation-design.md §5).
 *
 * When a group pool is set up, its requester gets a link
 * (https://1plus1gold.com/mag/join/<token>). Anyone who opens it sees the bar,
 * its closing date and its price, signs in, and taps Join — which makes them a
 * member of that bar (membership = access to buy from it; the MAG Agreement is
 * still accepted at their first purchase, as for every MAG buyer).
 *
 * Limits: up to `max_members` members (the requester included), until the bar's
 * closing date, and the requester can turn the link off at any time. Admins are
 * told of every join. The token is 256 random bits; the preview shows the bar,
 * never who else is in it.
 */

async function loadInvite(db: SupabaseClient, token: string): Promise<InviteRow | null> {
  if (!isInviteToken(token)) return null;
  const { data } = await db.from("mag_group_invites")
    .select("id, listing_id, request_id, owner_customer_id, token, max_members, expires_at, revoked_at")
    .eq("token", token).maybeSingle();
  return (data as InviteRow | null) ?? null;
}
 
async function memberCount(db: SupabaseClient, listingId: string): Promise<number> {
  const { data } = await db.from("mag_listing_members").select("customer_id").eq("listing_id", listingId).limit(100);
  return (data ?? []).length;
}
 
function stateOf(inv: InviteRow, members: number, now: Date): InviteState {
  if (inv.revoked_at || now.getTime() > Date.parse(inv.expires_at)) return "closed";
  return members >= inv.max_members ? "full" : "open";
}

/** The requester turns their link off (or back on). */
export async function setInviteActive(db: SupabaseClient, requestId: string, ownerId: string, active: boolean): Promise<boolean> {
  const { data } = await db.from("mag_group_invites")
    .update({ revoked_at: active ? null : new Date().toISOString() })
    .eq("request_id", requestId).eq("owner_customer_id", ownerId).select("id");
  return (data?.length ?? 0) > 0;
}
 
/** The requester's own view of their link (account card). */
export async function ownerInvite(db: SupabaseClient, requestId: string, ownerId: string): Promise<{ url: string; active: boolean; members: number; maxMembers: number } | null> {
  const { data } = await db.from("mag_group_invites")
    .select("listing_id, token, max_members, revoked_at, expires_at, owner_customer_id")
    .eq("request_id", requestId).eq("owner_customer_id", ownerId).maybeSingle();
  const inv = data as { listing_id: string; token: string; max_members: number; revoked_at: string | null; expires_at: string } | null;
  if (!inv) return null;
  return {
    url: joinUrl(inv.token),
    active: !inv.revoked_at && Date.now() <= Date.parse(inv.expires_at),
    members: await memberCount(db, inv.listing_id),
    maxMembers: inv.max_members,
  };
}
