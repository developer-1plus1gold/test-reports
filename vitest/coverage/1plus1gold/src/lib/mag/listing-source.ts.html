/**
 * 1+1Gold MAG — where a listing's facts come from (operator, 2026-09-28).
 *
 * A MAG listing is ONE physical bar we bought. Its cost, weight and serial are
 * therefore facts about that bar, not numbers an admin types:
 *   - cost per gram  ← inventory.cost_per_g_sgd (the PO / receipt cost basis);
 *   - grams          ← the bar's product weight (must be a whole number of grams);
 *   - serial         ← the bar's CoA serial, generated on the standard Serials
 *                      screen for the bar's product;
 *   - photo + video  ← that serial's nfc_serials.image_url / video_url, managed on
 *                      the Serials screen — so there is no MAG-specific upload.
 * Typing them by hand is what broke the cost link (testing item 24): the first
 * DEV listings carried a cost unrelated to any bar we own.
 *
 * Pure so the rules are testable without a database.
 */
 
export interface InventoryBar {
  id:             string;
  status:         string;
  product_id:     string;
  cost_per_g_sgd: number | string | null;
  nfc_serial:     string | null;
  product_grams:  number | string | null;
  product_type:   string | null;
}
 
export interface SerialRow {
  serial:     string;
  status:     string;
  product_id: string | null;
}
 
export type ListingSource =
  | { ok: true; grams: number; costPerG: number; serial: string }
  | { ok: false; status: number; error: string };
 
/** Product types whose inventory can never back a MAG listing. */
const NOT_A_BAR = new Set(["curve", "accessory", "consumable", "fulfilment_consumable", "equipment", "mag"]);
 
/**
 * @param magGrams the chosen MAG bar type's weight (products.grams of the `mag`
 *   product). The physical bar must weigh exactly this — a 50g bar can never
 *   back a "MAG in 100g bar" listing (operator, 2026-10-01).
 */
export function resolveListingSource(bar: InventoryBar | null, serial: SerialRow | null, magGrams: number | string | null): ListingSource {
  if (!bar) return { ok: false, status: 404, error: "Inventory bar not found" };
  if (bar.status !== "available") {
    return { ok: false, status: 409, error: "That bar is not available (already sold, reserved or listed)" };
  }
  if (bar.product_type && NOT_A_BAR.has(bar.product_type)) {
    return { ok: false, status: 400, error: "Only a physical gold bar can be listed as MAG" };
  }
  const grams = Number(bar.product_grams);
  if (!Number.isInteger(grams) || grams < 1) {
    return { ok: false, status: 400, error: "MAG needs a bar of a whole number of grams (e.g. 50g, 100g, 1kg)" };
  }
  const typeGrams = magGrams == null || magGrams === "" ? NaN : Number(magGrams);
  if (!Number.isFinite(typeGrams) || typeGrams <= 0) {
    return { ok: false, status: 400, error: "This MAG bar type has no weight set — set its grams in Admin → Products" };
  }
  if (typeGrams !== grams) {
    return { ok: false, status: 400, error: `This bar weighs ${grams}g but the MAG bar type is ${typeGrams}g — choose a ${typeGrams}g bar` };
  }
  const costPerG = Number(bar.cost_per_g_sgd);
  if (!Number.isFinite(costPerG) || costPerG <= 0) {
    return { ok: false, status: 400, error: "This bar has no purchase cost recorded — receive it against its PO first" };
  }
  if (!serial) {
    return { ok: false, status: 400, error: "Register a serial for this bar on the Serials screen first" };
  }
  if (serial.status !== "unassigned") {
    return { ok: false, status: 409, error: "That serial is already assigned or voided" };
  }
  if (serial.product_id && serial.product_id !== bar.product_id) {
    return { ok: false, status: 400, error: "That serial belongs to a different product than the bar" };
  }
  if (bar.nfc_serial && bar.nfc_serial !== serial.serial) {
    return { ok: false, status: 409, error: `This bar is already bound to serial ${bar.nfc_serial}` };
  }
  return { ok: true, grams, costPerG, serial: serial.serial };
}
 
/**
 * The picture a customer sees for a listing: the serial's own photo first
 * (the actual bar), then a legacy per-listing photo, then the bar type's
 * product image.
 */
export function listingImage(serialImage: string | null | undefined, legacyPhoto: string | null | undefined, productImage: string | null | undefined): string | null {
  return serialImage || legacyPhoto || productImage || null;
}
