/**
 * Sell-to-Us — the 1+1Gold MAG half of the one "Your gold" selection list.
 *
 * MAG grams and physical bars are sold back on separate paths: physical bars
 * travel (pickup / drop-off, inspection), MAG grams never leave our vault
 * (/api/mag/sell, no pickup). One request can't carry both, so the selection
 * model refuses to mix them: once anything of one kind is selected, the other
 * kind is locked until the selection is cleared. Pure + tested.
 */
import type { MagHoldingRow } from "./listings";
import { quoteMagSellBack } from "./sell-back";
 
export type SellKind = "mag" | "physical";
 
/** Which kind the current selection belongs to (null = nothing selected). */
export function activeSellKind(magSelectedCount: number, physicalSelectedCount: number): SellKind | null {
  if (magSelectedCount > 0) return "mag";
  if (physicalSelectedCount > 0) return "physical";
  return null;
}
 
/** May an item of `kind` be selected given the current selection? */
export function canSelectKind(kind: SellKind, active: SellKind | null): boolean {
  return active === null || active === kind;
}
 
/** A MAG holding can be sold back: eligible product + free grams (not locked in a sell-back). */
export function isMagSellable(h: Pick<MagHoldingRow, "buybackEligible" | "grams">): boolean {
  return h.buybackEligible && h.grams >= 1;
}
 
/** Whole grams only (the API sells in 1-gram steps), within [1, max]. */
export function clampMagGrams(grams: number, max: number): number {
  const cap = Math.floor(max);
  if (!(cap >= 1)) return 0;
  const g = Math.floor(Number.isFinite(grams) ? grams : 1);
  return Math.min(cap, Math.max(1, g));
}
 
export interface MagSellLine {
  listingId: string;
  label: string;
  serial: string;
  grams: number;
  buybackPremiumPct: number;
  /** Estimated payout at today's spot (null when spot is unknown). */
  quoteSgd: number | null;
}
 
/** The selected MAG lines (selection = listingId → grams), sellable rows only, grams clamped. */
export function magSellLines(
  holdings: MagHoldingRow[],
  selection: Record<string, number>,
  spotSgdPerG: number | null,
): MagSellLine[] {
  const out: MagSellLine[] = [];

    if (!Object.prototype.hasOwnProperty.call(selection, h.listingId) || !isMagSellable(h)) continue;

      listingId: h.listingId, label: h.label, serial: h.serial, grams,
      buybackPremiumPct: h.buybackPremiumPct,
      quoteSgd: quoteMagSellBack(spotSgdPerG, grams, h.buybackPremiumPct),
    });

}
 
/** Count / grams / payout over the selected MAG lines. Payout is null if any line can't be quoted. */
export function magSellTotals(lines: MagSellLine[]): { count: number; grams: number; payout: number | null } {
  let payout: number | null = 0;
  let grams = 0;
  for (const l of lines) {
    grams += l.grams;
    payout = payout === null || l.quoteSgd === null ? null : payout + l.quoteSgd;
  }
  return {
    count: lines.length,
    grams,
    payout: lines.length === 0 ? null : payout === null ? null : Math.round(payout * 100) / 100,
  };
}
 
export interface MagChartPoint {
  label: string;
  /** What the customer paid per gram (allocated grams only). */
  mag_cost_g: number | null;
  /** What we pay per gram today: spot × (1 + buyback premium). */
  mag_receive_g: number;
}
 
/**
 * MAG holdings as cost-vs-receive points for the Sell-to-Us chart. One point
 * per bar with free grams; cost/g is the customer's weighted paid price.
 */
export function magChartPoints(holdings: MagHoldingRow[], spotSgdPerG: number | null): MagChartPoint[] {
  if (spotSgdPerG == null || !(spotSgdPerG > 0)) return [];
  return holdings
    .filter(h => h.grams > 0)
    .map(h => ({
      label: `MAG ${h.serial || h.label}`.trim(),
      mag_cost_g: h.paidSgd > 0 ? Math.round((h.paidSgd / h.grams) * 100) / 100 : null,
      mag_receive_g: Math.round(spotSgdPerG * (1 + h.buybackPremiumPct / 100) * 100) / 100,
    }));
}
