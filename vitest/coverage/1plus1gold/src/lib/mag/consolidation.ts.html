/**
 * MAG — consolidation suggestions (Admin → MAG, Mukesh 2026-11-13).
 *
 * "A person holds 10 g in bar 1, 15 g in bar 2 and 25 g in bar 3 — it makes
 * sense to move this person to bar 3 and consolidate all 50 g on a single serial
 * number." This planner finds those cases across every open bar and proposes the
 * moves. It is ADVISORY: each move is made through the existing "Move a holder"
 * route, and `mag_transfer_holding` re-checks every rule under row locks. The
 * rules used here are the ones in ./transfer.ts (`sourceRefusal`,
 * `destinationRefusal`) — never re-stated — so the plan and the database agree.
 *
 * Pure (no server imports).
 *
 * How a destination is chosen, per (customer, bar type):
 *   1. Among the bars the customer already holds grams on, ordered by
 *        their grams there (most first — fewest grams moved),
 *        then the grams everyone holds on the bar (most first — keeps bars full),
 *        then the oldest bar,
 *      the first bar that can take EVERY movable holding (cumulative room counts:
 *      moves into one bar add up against its free grams). Holdings that cannot
 *      move (the customer owns the part of a gram, a payment or sell-back is in
 *      progress, …) stay where they are; the plan works around them, and a bar
 *      the customer is stuck on is preferred when that leaves fewer bars.
 *   2. If none can, a bar of the same type they hold nothing on, with room for
 *      ALL of their grams — only when that leaves them on exactly one bar.
 *   3. Otherwise a PARTIAL plan: into a held bar (same order), as many whole
 *      holdings as fit, largest first — only when it reduces their bar count.
 *   A "bar type" here is product + piece weight + stock source (consignment vs
 *   ours): the transfer rules never move grams across any of the three.
 */

/** Bars whose grams can move between each other (transfer rule 2 + stock source). */
export function barTypeKey(b: Pick<TransferBar, "productId" | "pieceGrams" | "consignment">): string {
  return `${b.productId}|${b.pieceGrams ?? ""}|${b.consignment ? "c" : "o"}`;
}
 
const liveGrams = (h: TransferHolder) => h.allocatedGrams + h.heldGrams + h.pendingGrams;
const heldByAll = (b: TransferBar) => Math.max(0, b.gramsTotal - b.gramsRemaining);
 
/** The destination order of rule 1: the customer's grams there, then everyone's, then oldest. */
function byDestinationOrder(mine: Map<string, number>) {
  return (a: ConsolidationBar, b: ConsolidationBar) =>
    (mine.get(b.id) ?? 0) - (mine.get(a.id) ?? 0)
    || heldByAll(b) - heldByAll(a)
    || Date.parse(a.createdAt) - Date.parse(b.createdAt)
    || a.serial.localeCompare(b.serial);
}
 
interface Movable { holding: ConsolidationHolding; bar: ConsolidationBar }
 
/**
 * Moves into `dest`, largest holding first, each checked by the transfer rules
 * against the room left after the earlier moves. `all` = refuse unless every
 * holding fits (rule 1/2); otherwise skip the ones that do not (rule 3).
 */
function packInto(dest: ConsolidationBar, movable: Movable[], all: boolean): ConsolidationMove[] | null {
  const moves: ConsolidationMove[] = [];
  let used = 0;
  for (const m of [...movable].sort((a, b) => b.holding.allocatedGrams - a.holding.allocatedGrams || a.bar.serial.localeCompare(b.bar.serial))) {
    if (m.bar.id === dest.id) continue;
    const g = m.holding.allocatedGrams;
    const refusal = destinationRefusal(m.bar, dest, used + g);
    if (refusal) { if (all) return null; continue; }
    moves.push({ fromListingId: m.bar.id, toListingId: dest.id, grams: g });
    used += g;
  }
  return moves;
}
 
function planCustomerType(
  customerId: string,
  holdings: ConsolidationHolding[],
  barById: Map<string, ConsolidationBar>,
  sameType: ConsolidationBar[],
): ConsolidationSuggestion {
  const views: HoldingView[] = [];
  const movable: Movable[] = [];
  const stuckBars = new Set<string>();
  for (const h of holdings) {
    const bar = barById.get(h.listingId)!;
    const refusal = sourceRefusal(bar, h);
    views.push({
      listingId: h.listingId, grams: liveGrams(h), allocatedGrams: h.allocatedGrams,
      refusal, refusalText: refusal ? transferRefusalMessage(refusal) : null,
    });
    if (refusal) stuckBars.add(bar.id);
    else movable.push({ holding: h, bar });
  }
  const mine = new Map(holdings.map((h) => [h.listingId, liveGrams(h)]));
  views.sort((a, b) => b.grams - a.grams);
  const barsBefore = holdings.length;
  const first = sameType[0];
  const base = {
    customerId, productId: first.productId, label: first.label, pieceGrams: first.pieceGrams,
    consignment: first.consignment, holdings: views, barsBefore,
  };
  const barsAfterFor = (destId: string, moves: ConsolidationMove[]) => {
    const moved = new Set(moves.map((m) => m.fromListingId));
    return new Set([destId, ...holdings.map((h) => h.listingId).filter((id) => !moved.has(id))]).size;
  };
  const suggestion = (kind: SuggestionKind, dest: ConsolidationBar, moves: ConsolidationMove[], isNew: boolean): ConsolidationSuggestion => {
    const barsAfter = barsAfterFor(dest.id, moves);
    return {
      ...base, kind: kind === "full" && barsAfter > 1 ? "partial" : kind,
      destinationId: dest.id, destinationIsNew: isNew, moves,
      gramsMoved: moves.reduce((s, m) => s + m.grams, 0), barsBefore, barsAfter,
    };
  };
 
  const held = holdings.map((h) => barById.get(h.listingId)!).sort(byDestinationOrder(mine));
 
  // Rule 1 — a bar they are on that takes every movable holding; fewest bars left wins, then the order.
  let best: ConsolidationSuggestion | null = null;
  for (const dest of held) {
    const moves = packInto(dest, movable, true);
    if (!moves || moves.length === 0) continue;
    const s = suggestion("full", dest, moves, false);
    if (!best || s.barsAfter < best.barsAfter) best = s;
  }
  if (best && best.barsAfter === 1) return best;
 
  // Rule 2 — a bar of the type they are not on, with room for all of it, when that leaves ONE bar.
  if (stuckBars.size === 0) {
    const fresh = sameType
      .filter((b) => !mine.has(b.id))
      .sort((a, b) => heldByAll(b) - heldByAll(a) || Date.parse(a.createdAt) - Date.parse(b.createdAt) || a.serial.localeCompare(b.serial));
    for (const dest of fresh) {
      const moves = packInto(dest, movable, true);
      Eif (moves && moves.length > 0) return suggestion("full", dest, moves, true);
    }
  }
  if (best) return best;
 
  // Rule 3 — partial: as many whole holdings as fit, into a bar they are on.
  for (const dest of held) {
    const moves = packInto(dest, movable, false) ?? [];
    if (moves.length === 0) continue;
    const s = suggestion("partial", dest, moves, false);
    Eif (s.barsAfter < barsBefore && (!best || s.barsAfter < best.barsAfter)) best = s;
  }
  if (best) return best;
 
  return { ...base, kind: "blocked", destinationId: null, destinationIsNew: false, moves: [], gramsMoved: 0, barsAfter: barsBefore };
}
 
/**
 * Open bars (access_type 'public') whose every holder could be moved to other
 * same-type bars with room — so the bar could be sold or delivered whole.
 * Greedy: largest holder first; each goes to a bar they already hold (it also
 * consolidates them), else the bar with the least room that still fits.
 */
function freeableBars(bars: ConsolidationBar[], holdingsByBar: Map<string, ConsolidationHolding[]>): FreeableBar[] {
  const out: FreeableBar[] = [];
  const byType = new Map<string, ConsolidationBar[]>();
  for (const b of bars) byType.set(barTypeKey(b), [...(byType.get(barTypeKey(b)) ?? []), b]);
  const barsOf = new Map<string, Set<string>>();
  for (const [listingId, hs] of holdingsByBar) {
    for (const h of hs) barsOf.set(h.customerId, (barsOf.get(h.customerId) ?? new Set()).add(listingId));
  }
  for (const bar of bars) {
    const holders = holdingsByBar.get(bar.id) ?? [];
    if (holders.length === 0 || bar.houseOwnerCustomerId) continue;
    if (holders.some((h) => sourceRefusal(bar, h) !== null)) continue;
    const others = (byType.get(barTypeKey(bar)) ?? []).filter((b) => b.id !== bar.id);
    const used = new Map<string, number>();
    const moves: FreeableBar["moves"] = [];
    let ok = true;
    for (const h of [...holders].sort((a, b) => b.allocatedGrams - a.allocatedGrams)) {
      const fits = others.filter((d) => destinationRefusal(bar, d, (used.get(d.id) ?? 0) + h.allocatedGrams) === null);
      const onIt = barsOf.get(h.customerId) ?? new Set<string>();
      const pick = fits.find((d) => onIt.has(d.id))
        ?? [...fits].sort((a, b) => (a.gramsRemaining - (used.get(a.id) ?? 0)) - (b.gramsRemaining - (used.get(b.id) ?? 0)))[0];
      if (!pick) { ok = false; break; }
      used.set(pick.id, (used.get(pick.id) ?? 0) + h.allocatedGrams);
      moves.push({ customerId: h.customerId, fromListingId: bar.id, toListingId: pick.id, grams: h.allocatedGrams });
    }
    if (ok) out.push({ listingId: bar.id, label: bar.label, gramsHeld: moves.reduce((s, m) => s + m.grams, 0), moves });
  }
  return out.sort((a, b) => a.gramsHeld - b.gramsHeld);
}
 
/**
 * The plan for every customer on ≥2 open bars of one type. Bars that are not
 * open (reserved / group) are ignored, as are holdings on them; delivered bars
 * too. Suggestions: biggest saving first (bars removed, then grams on the line).
 */
export function planConsolidation(bars: ConsolidationBar[], holdings: ConsolidationHolding[]): ConsolidationPlan {
  const open = bars.filter((b) => b.accessType === "public" && b.status !== "delivered");
  const barById = new Map(open.map((b) => [b.id, b]));
  const live = holdings.filter((h) => barById.has(h.listingId) && liveGrams(h) > 0);
 
  const byType = new Map<string, ConsolidationBar[]>();
  for (const b of open) byType.set(barTypeKey(b), [...(byType.get(barTypeKey(b)) ?? []), b]);
 
  const groups = new Map<string, ConsolidationHolding[]>();
  for (const h of live) {
    const k = `${h.customerId}#${barTypeKey(barById.get(h.listingId)!)}`;
    groups.set(k, [...(groups.get(k) ?? []), h]);
  }
 
  const suggestions: ConsolidationSuggestion[] = [];
  for (const [k, hs] of groups) {
    if (new Set(hs.map((h) => h.listingId)).size < 2) continue;
    const typeKey = k.slice(k.indexOf("#") + 1);
    suggestions.push(planCustomerType(hs[0].customerId, hs, barById, byType.get(typeKey)!));
  }
  const rank: Record<SuggestionKind, number> = { full: 0, partial: 1, blocked: 2 };
  suggestions.sort((a, b) =>
    rank[a.kind] - rank[b.kind]
    || (b.barsBefore - b.barsAfter) - (a.barsBefore - a.barsAfter)
    || b.holdings.reduce((s, h) => s + h.grams, 0) - a.holdings.reduce((s, h) => s + h.grams, 0)
    || a.customerId.localeCompare(b.customerId));
 
  const holdingsByBar = new Map<string, ConsolidationHolding[]>();
  for (const h of live) holdingsByBar.set(h.listingId, [...(holdingsByBar.get(h.listingId) ?? []), h]);
  const activeOpen = open.filter((b) => b.status === "active");
  return { suggestions, freeableBars: freeableBars(activeOpen, holdingsByBar) };
}
 
/** The note on each move of an applied suggestion: "Consolidation: 3 bars → 1". */
export function consolidationNote(s: Pick<ConsolidationSuggestion, "barsBefore" | "barsAfter">): string {
  return `Consolidation: ${s.barsBefore} bars → ${s.barsAfter}`;
}
