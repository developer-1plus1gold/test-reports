/**
 * Tiers the Savings Simulator models MAG with: the available listing whose
 * 1-gram premium is lowest (a saver's first purchase is the smallest), or
 * null when nothing is listed.
 */
export function cheapestMagTiers(listings: MagListingPublic[]): MagListingPublic["tiers"] | null {
  const usable = listings.filter((l) => l.gramsRemaining >= 1 && l.tiers.length > 0);
  if (usable.length === 0) return null;
  const at1 = (l: MagListingPublic) => computeStandardLineTotal(1, 1, 1, l.tiers).premiumRate;
  return usable.reduce((a, b) => (at1(b) < at1(a) ? b : a)).tiers;
}
