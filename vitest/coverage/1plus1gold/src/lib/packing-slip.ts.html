export interface PackingSlipBar {
  /** Inventory row id (last 8 hex shown on slip as the Bar ID fallback) */
  inventory_id:  string;
  /** order_item this bar fulfils — used to swap in the operator-attached NFC
   *  card (nfc_serials) at print time. Optional: conversion slips don't set it. */
  order_item_id?: string | null;

  product_label?: string | null;
  /** Customer NFC card serial (1G-XXXX-XXXX) affixed to this bar. */
  nfc_serial?:   string | null;
  /** Manufacturer serial printed on the bar (e.g. a PAMP/Valcambi number),
   *  from the operator-attached card. When present it's shown as the slip's
   *  primary bar identifier (standard bars, PAMP splits) instead of Bar ID. */
  manufacturer_serial?: string | null;

  source_label:  string;
  rfid_tag:      string | null;
  serial_number: string | null;
  /** The combibar serial this piece belongs to (own serial if received whole,
   *  else the parent combibar's serial/lot). Shown as the slip "Bar ID" —
   *  the RFID box number is internal and intentionally NOT shown. */
  combibar_serial?: string | null;
  lot_reference: string | null;
}
 
/**
 * Swap each bar's NFC serial for the card the operator physically ATTACHED to
 * that line at packing (`nfcByItem`, sourced from `nfc_serials` keyed by
 * order_item) instead of `inventory.nfc_serial` (the box the card was generated
 * against at receipt — usually a different serial, never what ships).
 *
 * Cards aren't bound to a specific inventory row, so within an order_item we
 * assign them positionally. When a bar's line has no card attached at its
 * position yet, the serial is blanked so a stale/wrong serial never prints.
 * Bars whose order_item has no entry in `nfcByItem` (e.g. conversion slips,
 * non-NFC rows) are left untouched.
 */
export function attachOperatorSerials<B extends PackingSlipBar>(
  bars: B[],
  nfcByItem: Record<string, { serial: string; manufacturer_serial?: string | null }[]>,
): B[] {
  const cursor: Record<string, number> = {};
  return bars.map((b) => {
    const oi = b.order_item_id;
    if (!oi || !(oi in nfcByItem)) return b;
    const i = cursor[oi] ?? 0;
    cursor[oi] = i + 1;
    const card = nfcByItem[oi]?.[i];
    // Attach the card's manufacturer serial too (standard bars / PAMP splits)
    // so the slip can show it as the primary bar identifier.
    return { ...b, nfc_serial: card?.serial ?? null, manufacturer_serial: card?.manufacturer_serial ?? null };
  });
}

function escapeHtml(s: string): string {
  return s.replace(/[&<>"']/g, c => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", "\"": "&quot;", "'": "&#39;" }[c]!));
}
 
function formatLalamoveId(id: string): string {
  if (id.length <= 6) return escapeHtml(id);
