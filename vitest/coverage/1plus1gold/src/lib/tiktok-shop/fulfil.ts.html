 * TikTok's shipped / delivered come back through the order import (forward only).
 * A refusal is stored on the ledger row (`fulfil_error`) and shown on the order.

/** The one package of an order from Get Order Detail, or why not. */
export function onePackage(order: unknown): { ok: true; packageId: string; provider: string | null } | { ok: false; error: string } {
  const o = (order ?? {}) as { packages?: Array<{ id?: string }>; shipping_provider?: string };
  const ids = [...new Set((o.packages ?? []).map((p) => p.id).filter((v): v is string => !!v))];
  if (ids.length === 0) return { ok: false, error: "TikTok has no package for this order yet" };
  if (ids.length > 1) return { ok: false, error: "TikTok split this order into several packages — ship it in Seller Center" };
  return { ok: true, packageId: ids[0], provider: o.shipping_provider || null };
}

/** The TikTok package of one order, for the Dispatch tile (null = not a TikTok order). */
