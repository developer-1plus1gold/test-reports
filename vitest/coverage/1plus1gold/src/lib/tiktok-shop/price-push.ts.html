/**
 * TikTok Shop price + stock push (step C part 2, 2026-11-17) — TikTok's TRANSPORT for the
 * shared engine (lib/marketplace/price-push), which owns every guard: the switch, stale
 * gold, due cadence, > 5 % Approve, below-cost refusal, stock-only ticks.
 *
 * TikTok takes prices and stock per PRODUCT (doc ids in docs/tiktok-shop-integration.md):
 *   POST /product/202309/products/{product_id}/prices/update
 *        { skus: [{ id, price: { amount, currency: "SGD" } }] }
 *   POST /product/202309/products/{product_id}/inventory/update
 *        { skus: [{ id, inventory: [{ warehouse_id, quantity }] }] }  — the total in that warehouse
 * Inventory can answer code 0 with `data.errors[]` per SKU; those count as refused.
 * The warehouse is the shop's default SALES warehouse from the catalog snapshot.
 */
import type { SupabaseClient } from "@supabase/supabase-js";
import { ttsCall } from "./client";
import { getShopCredentials } from "./connection";
import {
  runMarketplacePush, syncMarketplacePushCards,
  type PushChannel, type PushHttp, type PushMode, type PushResult, type SkuUpdate,
} from "@/lib/marketplace/price-push";
 
/** The default SALES warehouse in the stored catalog snapshot, or null. Pure. */
export function salesWarehouse(snapshot: unknown): string | null {
  const ws = ((snapshot as { warehouses?: { warehouses?: Array<{ id: string; type?: string; is_default?: boolean; effect_status?: string }> } } | null)
    ?.warehouses?.warehouses ?? []).filter((w) => w.type === "SALES_WAREHOUSE" && w.effect_status !== "DISABLED");
  return (ws.find((w) => w.is_default) ?? ws[0])?.id ?? null;
}
 
/** Per-SKU refusals inside a successful inventory reply. Pure. */
export function inventoryErrors(data: unknown): string | null {
  const errs = (data as { errors?: Array<{ code?: number; message?: string; detail?: { extra_errors?: Array<{ message?: string }> } }> } | null)?.errors;
  if (!Array.isArray(errs) || errs.length === 0) return null;
  return errs.map((e) => e.detail?.extra_errors?.[0]?.message ?? e.message ?? String(e.code ?? "refused")).join("; ").slice(0, 200);
}
 
const msg = (e: unknown) => (e instanceof Error ? e.message : String(e)).slice(0, 200);
 
export const TIKTOK_SHOP_PUSH: PushChannel = {
  channel: "tiktok_shop",
  label: "TikTok Shop",
  agentName: "tiktok_shop_sync",
  page: "/admin/marketplaces?tab=tiktok_shop",
  async connect(db, now, http) {
    const creds = await getShopCredentials(db, now);
    if (!creds) return null;
    const { data: conn } = await db.from("marketplace_connections").select("api_shapes").eq("channel", "tiktok_shop").maybeSingle();
    const warehouseId = salesWarehouse((conn?.api_shapes as Record<string, unknown> | null)?.tiktokCatalog);
    const base = { appKey: creds.appKey, appSecret: creds.appSecret, accessToken: creds.token, shopCipher: creds.shopCipher, fetchImpl: http.fetchImpl, method: "POST" as const };
    return {
      async send(updates: SkuUpdate[]) {
        const failures = new Map<string, string>();
        for (const u of updates) {
          const product = encodeURIComponent(u.itemId);
          try {
            if (u.price != null) {
              await ttsCall(`/product/202309/products/${product}/prices/update`, {
                ...base, body: { skus: [{ id: u.skuId, price: { amount: u.price.toFixed(2), currency: "SGD" } }] },
              });
            }
            Eif (u.quantity != null) {
              if (!warehouseId) throw new Error("no TikTok sales warehouse — run the catalog sync");
              const data = await ttsCall(`/product/202309/products/${product}/inventory/update`, {
                ...base, body: { skus: [{ id: u.skuId, inventory: [{ warehouse_id: warehouseId, quantity: Math.max(0, Math.trunc(u.quantity)) }] }] },
              });
              const refused = inventoryErrors(data);
              if (refused) throw new Error(refused);
            }
          } catch (e) { failures.set(u.sellerSku, msg(e)); }
        }
        return failures;
      },
    };
  },
};
 
export function runTikTokShopPush(db: SupabaseClient, now: Date, mode: PushMode, http: PushHttp = {}): Promise<PushResult> {
  return runMarketplacePush(db, now, mode, TIKTOK_SHOP_PUSH, http);
}
