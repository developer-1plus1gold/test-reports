/**
 * TikTok Shop request signature — PURE.
 *
 * Algorithm, from TikTok Shop "Sign your API request" (partner.tiktokshop.com, doc
 * 678e3a3d4ddec3030b238faf; same in its Go, Java, Node and Python samples):
 *   1. every query parameter except `sign` and `access_token`, keys sorted, as {key}{value};
 *   2. the API path in front;
 *   3. the raw JSON body appended, unless the request is multipart/form-data;
 *   4. wrapped as  app_secret + string + app_secret;
 *   5. HMAC-SHA256 with the app secret as key, lowercase hex.
 * The access token (header `x-tts-access-token`, API 202309+) is never signed.
 */
import { createHmac, timingSafeEqual } from "node:crypto";
 
export type TtsQuery = Record<string, string>;
 
export function signTtsRequest(appSecret: string, path: string, query: TtsQuery, body?: string | null): string {
  const params = Object.keys(query)
    .filter((k) => k !== "sign" && k !== "access_token")
    .sort()
    .map((k) => `${k}${query[k]}`)
    .join("");
  const input = `${appSecret}${path}${params}${body ?? ""}${appSecret}`;
  return createHmac("sha256", appSecret).update(input, "utf8").digest("hex");
}
