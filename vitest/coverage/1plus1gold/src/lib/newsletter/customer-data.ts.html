/**
 * Per-customer data fetcher for the newsletter "Your month" section.
 *
 * Runs at customer-view time (server-side, /account/newsletter route handler)
 * — produces the deterministic input the block library evaluates against.
 *
 * No LLM calls. All values come from existing site tables.
 */
 
import type { SupabaseClient } from "@supabase/supabase-js";
import type { CustomerNewsletterData } from "./blocks";
import { aggregateOrders, filterPaid, filterByWindow, PAID_STATUSES, type OrderAgg } from "@/lib/order-aggregations";
import { nextPurchaseRate } from "@/lib/gold";
import { fetchPremiums, toPremiumMap } from "@/lib/premiums";
import { IN_CUSTODY_STATUSES } from "@/lib/vault/liquidation";
 
const COMBIBAR_50G  = 50;
const COMBIBAR_100G = 100;
 
interface AccumulationRow {
  grams:               string | number;
  recorded_at:         string;
  family_member_id:    string | null;
}
 
interface ProfileRow {
  full_name:        string | null;
  created_at:       string;
}
 
interface TreeBranchRow {
  id:               string;
  name:             string;
  declared_grams:   string | number;  // customer-set goal grams for this branch
  is_self:          boolean;
}
 
/**
 * Build CustomerNewsletterData for (customer, period). period is the month
 * being summarised (the prior month, since the cron runs on the 1st).
 */
export async function buildCustomerNewsletterData(
  supabase: SupabaseClient,
  customerId: string,
  periodYear: number,
  periodMonth: number,
): Promise<CustomerNewsletterData | null> {
  const periodStart = isoDate(periodYear, periodMonth, 1);
  const periodEnd   = isoEndOfMonth(periodYear, periodMonth);
  const periodStartIso = periodStart + "T00:00:00Z";
  const periodEndIso   = periodEnd   + "T23:59:59Z";
 
  // ── Profile first (cheap, also acts as the "customer exists?" gate) ──────
  const { data: profile } = await supabase
    .from("profiles")
    .select("full_name, created_at")
    .eq("id", customerId)
    .maybeSingle();
  if (!profile) return null;
 
  const p = profile as ProfileRow;
  const fullName  = (p.full_name ?? "").trim();
  const firstName = fullName.split(/\s+/)[0] || "there";
  const joinedAt = new Date(p.created_at);
  const tenureMonths = monthsBetween(joinedAt, new Date(periodEnd));
  const joinedThisMonth =
    joinedAt.getUTCFullYear() === periodYear && joinedAt.getUTCMonth() + 1 === periodMonth;
 
  // ── Streak window: 24 months ending in period (single grouped fetch) ────
  const streakWindowStartDate = new Date(Date.UTC(periodYear, periodMonth - 24, 1));
  const streakWindowEndDate   = new Date(Date.UTC(periodYear, periodMonth, 0, 23, 59, 59));
  const ytdStartIso = `${periodYear}-01-01T00:00:00Z`;
  const trailing3StartDate = new Date(Date.UTC(periodYear, periodMonth - 3, 1));
  const trailing3EndDate   = new Date(Date.UTC(periodYear, periodMonth, 0, 23, 59, 59));
  const periodStartCompact = periodStart.replace(/-/g, "");
  const periodEndCompact   = periodEnd.replace(/-/g, "");
 
  // ── Single parallel batch for every remaining query ─────────────────────
  // Earlier this was ~25+ sequential awaits — including a 24-iteration
  // streak loop. Newsletter page paint time was 5–8s on customers with
  // many orders. Folding them into one Promise.all takes the network
  // round-trip down to roughly the slowest single query (~700ms typical).
  // Premium fetch goes via the REST API at content.supabase — included
  // in the same batch so it doesn't add another sequential hop.
  const [
    accRes,           // accumulation rows up to period end
    ordersThisMonthRes,
    lastOrderRes,
    streakOrdersRes,  // paid-order created_ats over the last 24 months (streak)
    branchesRes,
    holdingsRes,
    lifetimeOrdersRes,
    gyBarsCountRes,
    gySalesRes,
    bbRowsRes,
    convRowsRes,
    spotOpenRes,
    spotCloseRes,
    spotSeriesRes, // daily SGD/g for the period — line chart on the market page
    custRes,
    threeMoAccRes,
    vaultHoldingsRes,
    premiumRows,
  ] = await Promise.all([
    supabase
      .from("accumulation")
      .select("grams, recorded_at, family_member_id")
      .eq("customer_id", customerId)
      .lte("recorded_at", periodEndIso)
      .order("recorded_at", { ascending: true }),
    supabase
      .from("orders")
      .select("id, total_grams, spot_sgd, total_sgd, status, created_at")
      .eq("customer_id", customerId)
      .gte("created_at", periodStartIso)
      .lte("created_at", periodEndIso)
      .in("status", [...PAID_STATUSES]),
    supabase
      .from("orders")
      .select("created_at")
      .eq("customer_id", customerId)
      .in("status", [...PAID_STATUSES])
      .order("created_at", { ascending: false })
      .limit(1),
    supabase
      .from("orders")
      .select("created_at")
      .eq("customer_id", customerId)
      .in("status", [...PAID_STATUSES])
      .gte("created_at", streakWindowStartDate.toISOString())
      .lte("created_at", streakWindowEndDate.toISOString()),
    supabase
      .from("tree_branches")
      .select("id, name, declared_grams, is_self")
      .eq("customer_id", customerId),
    supabase
      .from("customer_gold_holdings")
      .select("curve_grams_purchased, standard_grams_purchased, curve_yielded")
      .eq("customer_id", customerId)
      .maybeSingle(),
    supabase
      .from("orders")
      .select(`
        id, status, created_at, grand_total_sgd, total_sgd, total_grams,
        order_items ( quantity, price_sgd, grams_each, premium_applied_pct,
                      products ( product_type ) )
      `)
      .eq("customer_id", customerId)
      .in("status", [...PAID_STATUSES]),
    supabase
      .from("gold_yield_bars")
      .select("id", { count: "exact", head: true })
      .eq("user_id", customerId),
    supabase
      .from("gold_yield_sales")
      .select("payout_to_user_sgd, gold_yield_bars!inner(user_id)")
      .eq("gold_yield_bars.user_id", customerId),
    supabase
      .from("buyback_requests")
      .select("status, paid_at, total_payout_sgd, shipping_fee_sgd, created_at, buyback_request_lines(grams)")
      .eq("customer_id", customerId),
    supabase
      .from("conversions")
      .select("status, target_50g_count, target_100g_count, conversion_fee_sgd, total_charged_sgd, created_at")
      .eq("customer_id", customerId),
    supabase
      .from("gold_price_history")
      .select("price_per_gram")
      .eq("currency", "SGD")
      .gte("date", periodStartCompact)
      .order("date", { ascending: true })
      .limit(1)
      .maybeSingle(),
    supabase
      .from("gold_price_history")
      .select("price_per_gram")
      .eq("currency", "SGD")
      .lte("date", periodEndCompact)
      .order("date", { ascending: false })
      .limit(1)
      .maybeSingle(),
    // Daily spot prices for the period — feeds the spot-trend line chart
    // on the market page. SGD/g is what the customer sees on-screen so
    // we keep the chart in their primary unit. Cap at 40 rows: months
    // have ~22 trading days; some months that span longer windows could
    // hit higher counts, but 40 is a safe ceiling.
    supabase
      .from("gold_price_history")
      .select("date, price_per_gram")
      .eq("currency", "SGD")
      .gte("date", periodStartCompact)
      .lte("date", periodEndCompact)
      .order("date", { ascending: true })
      .limit(40),
    supabase
      .from("customers")
      .select("preferred_currency")
      .eq("id", customerId)
      .maybeSingle(),
    supabase
      .from("accumulation")
      .select("grams")
      .eq("customer_id", customerId)
      .gte("recorded_at", trailing3StartDate.toISOString())
      .lte("recorded_at", trailing3EndDate.toISOString()),
    // Vault custody — grams currently held in our vault (in-custody statuses
    // only; terminal withdrawn/liquidated have left). Same set the dashboard +
    // /api/vault/dashboard use so the newsletter can't drift.
    supabase
      .from("vault_holdings")
      .select("grams, status")
      .eq("customer_id", customerId)
      .in("status", [...IN_CUSTODY_STATUSES]),
    fetchPremiums().catch(() => []),
  ]);
 
  // ── Accumulation totals for the period ──────────────────────────────────
  const accRows = (accRes.data ?? []) as AccumulationRow[];
  const totalGrams = accRows.reduce((s, r) => s + Number(r.grams), 0);
  // Net accumulation-ledger movement in the period — used ONLY to derive the
  // true holdings at period start for milestone crossing. This INCLUDES
  // `cancelled` reversals of prior-month orders, so it can go negative (e.g. a
  // June purchase cancelled in July). It must NOT be shown to the customer as
  // "grams added" — the displayed `gramsAddedThisMonth` (below) is computed
  // from THIS month's paid purchases only, so a delayed cancellation of an
  // older order never renders a nonsensical negative "Grams Added" (2026-08-01,
  // Mukesh's account showed "-2g added across 1 purchases").
  const ledgerNetThisMonth = accRows
    .filter((r) => r.recorded_at >= periodStart && r.recorded_at <= periodEndIso)
    .reduce((s, r) => s + Number(r.grams), 0);
  const totalGramsAtPeriodStart = totalGrams - ledgerNetThisMonth;
 
  // ── Purchase counts + YTD avg premium + last-order distance ─────────────
  const purchasesThisMonth = (ordersThisMonthRes.data ?? []).length;
  // avgPremiumPctYTD is computed lower down via the shared aggregator
  // (after ordersAgg is built — see the "Lifetime aggregator inputs"
  // block). Using premium_applied_pct snapshots with the 0-100% sane-band
  // filter, not derived-from-grand_total which goes negative on
  // discounted orders.
  const lastOrderDate = (lastOrderRes.data?.[0] as { created_at: string } | undefined)?.created_at;
  const monthsSinceLastPurchase = lastOrderDate
    ? monthsBetween(new Date(lastOrderDate), new Date(periodEnd))
    : 999;
 
  // ── Streak — bucket the 24-month window client-side ─────────────────────
  // Replaces 24 sequential per-month head-count queries. Each buckets a
  // paid order's created_at into a "YYYY-MM" key; consecutive months ending
  // in the period with ≥1 paid order = the streak.
  const streakOrdersData = (streakOrdersRes.data ?? []) as { created_at: string }[];
  const monthsWithPaidOrders = new Set<string>();
  for (const r of streakOrdersData) {
    const d = new Date(r.created_at);
    monthsWithPaidOrders.add(`${d.getUTCFullYear()}-${d.getUTCMonth() + 1}`);
  }
  let purchaseStreakMonths = 0;
  for (let i = 0; i < 24; i++) {
    const target = new Date(Date.UTC(periodYear, periodMonth - 1 - i, 1));
    const key    = `${target.getUTCFullYear()}-${target.getUTCMonth() + 1}`;
    if (monthsWithPaidOrders.has(key)) purchaseStreakMonths++;
    else break;
  }
 
  // ── Tree branches (using the already-fetched accRows) ───────────────────
  const branchRows = (branchesRes.data ?? []) as TreeBranchRow[];
  const treeBranches: CustomerNewsletterData["treeBranches"] = [];
  for (const b of branchRows) {
    if (b.is_self) continue;
    const allocatedGrams = accRows
      .filter((r) => r.family_member_id === b.id)
      .reduce((s, r) => s + Number(r.grams), 0);
    const firedThisMonth = accRows.some(
      (r) => r.family_member_id === b.id
          && r.recorded_at >= periodStart
          && r.recorded_at <= periodEndIso,
    );
    treeBranches.push({
      name:           b.name,
      allocatedGrams,
      goalGrams:      Number(b.declared_grams),
      firedThisMonth,
    });
  }
 
  // ── Milestone detection ─────────────────────────────────────────────────
  const milestone = detectMilestone({
    totalGramsBefore: totalGramsAtPeriodStart,
    totalGramsAfter:  totalGrams,
    purchasesThisMonth,
    joinedThisMonth,
    isFirstPurchase:  joinedThisMonth ? purchasesThisMonth >= 1 : false,
  });
 
  // ── Pace projection (3-month avg from threeMoAccRes) ────────────────────
  const last3MonthsAvg = ((threeMoAccRes.data ?? []) as { grams: number | string }[])
    .reduce((s, r) => s + Number(r.grams), 0) / 3;
  const monthsToNext50g  = totalGrams < COMBIBAR_50G  && last3MonthsAvg > 0
    ? Math.ceil((COMBIBAR_50G  - totalGrams) / last3MonthsAvg) : null;
  const monthsToNext100g = totalGrams < COMBIBAR_100G && last3MonthsAvg > 0
    ? Math.ceil((COMBIBAR_100G - totalGrams) / last3MonthsAvg) : null;
 
  const returnedAfterGap   = monthsSinceLastPurchase >= 3 && purchasesThisMonth >= 1;
  const dormantWithBalance = !joinedThisMonth && monthsSinceLastPurchase >= 3 && totalGrams >= 1 && purchasesThisMonth === 0;
 
  // ── Holdings (typed split) ──────────────────────────────────────────────
  const holdings = holdingsRes.data as { curve_grams_purchased: number | string; standard_grams_purchased: number | string; curve_yielded: number | string } | null;
  const curveGramsLifetime    = holdings ? Number(holdings.curve_grams_purchased)    : 0;
  const standardGramsLifetime = holdings ? Number(holdings.standard_grams_purchased) : 0;
  const curveYielded          = holdings ? Number(holdings.curve_yielded)            : 0;
  const positionInCycle       = curveGramsLifetime % 50;
  const cyclesCompleted       = Math.floor(curveGramsLifetime / 50);
 
  // ── Lifetime aggregator inputs ──────────────────────────────────────────
  type LtOrder = {
    id: string; status: string; created_at: string;
    grand_total_sgd: number | string | null; total_sgd: number | string | null; total_grams: number | string;
    order_items: { quantity: number; price_sgd: number | string; grams_each: number | string;
                   premium_applied_pct: number | string | null;
                   products: { product_type: string | null } | null }[];
  };
  const ordersAgg: OrderAgg[] = ((lifetimeOrdersRes.data as LtOrder[] | null) ?? []).map(o => ({
    id:              o.id,
    status:          o.status,
    created_at:      o.created_at,
    grand_total_sgd: o.grand_total_sgd != null ? Number(o.grand_total_sgd) : null,
    total_sgd:       Number(o.total_sgd ?? 0),
    total_grams:     Number(o.total_grams),
    items: (o.order_items ?? []).map(i => ({
      product_type:        i.products?.product_type ?? null,
      grams_each:          Number(i.grams_each),
      quantity:            Number(i.quantity),
      price_sgd:           Number(i.price_sgd),
      premium_applied_pct: i.premium_applied_pct != null ? Number(i.premium_applied_pct) : null,
    })),
  }));
  const lifetimeAgg = aggregateOrders(filterPaid(ordersAgg));
  const curveStats     = lifetimeAgg.byType.curve;
  const standardStats  = lifetimeAgg.byType.standard;
  const accessoryStats = lifetimeAgg.byType.accessory;
 
  // YTD avg premium — same shared aggregator, just windowed to this
  // calendar year. Shares the math + sane-band filter [0,100] used by
  // dashboard, admin analytics, and order-history PDF; previous local
  // implementation derived premium from grand_total_sgd / total_grams,
  // which goes negative when an order had a discount applied (referral,
  // promo). Filter window is [Jan 1 of period year, period end inclusive).
  const ytdAgg = aggregateOrders(filterPaid(filterByWindow(
    ordersAgg,
    new Date(ytdStartIso),
    new Date(new Date(periodEndIso).getTime() + 1),
  )));
  const avgPremiumPctYTD = ytdAgg.totals.avgPremium;
 
  // Single pass over lifetime order items — period grams + splits drawer.
  let curveGramsAddedThisMonth    = 0;
  let standardGramsAddedThisMonth = 0;
  let accessorySpendThisMonth     = 0;
  let curve50Grams  = 0;
  let curve100Grams = 0;
  for (const o of ordersAgg) {
    const inPeriod = o.created_at && o.created_at >= periodStart && o.created_at <= periodEndIso;
    for (const i of o.items) {
      const g = i.grams_each * i.quantity;
      const sgd = i.price_sgd * i.quantity;
      const t = i.product_type ?? "curve";
      if (inPeriod) {
        if (t === "standard")       standardGramsAddedThisMonth += g;
        else if (t === "accessory") accessorySpendThisMonth     += sgd;
        else                        curveGramsAddedThisMonth    += g;
      }
      if (t === "curve") {
        if (Number(i.grams_each) === 50)  curve50Grams  += 50  * i.quantity;
        if (Number(i.grams_each) === 100) curve100Grams += 100 * i.quantity;
      }
    }
  }
  const splitsInDrawerGrams = Math.max(0, curveGramsLifetime - curve50Grams - curve100Grams);
 
  // Displayed "Grams Added this month" = gold grams from THIS month's PAID
  // purchases (curve + standard), NOT the raw accumulation-ledger net
  // (ledgerNetThisMonth), which includes reversals of prior-month orders and
  // can go negative. ordersAgg is already PAID-only, so cancelled orders drop
  // out cleanly and this is always ≥ 0. (Accessories carry 0 grams.)
  const gramsAddedThisMonth = curveGramsAddedThisMonth + standardGramsAddedThisMonth;
 
  // ── Next purchase rate ─────────────────────────────────────────────────
  const premMap = toPremiumMap(premiumRows);
  const nextPurchaseRatePct = premMap.size > 0
    ? nextPurchaseRate(positionInCycle, premMap) * 100
    : null;
 
  // ── Gold Yield ──────────────────────────────────────────────────────────
  const gyBarCount = gyBarsCountRes.count ?? 0;
  let goldYield: CustomerNewsletterData["goldYield"] = null;
  if (gyBarCount > 0 || curveYielded > 0) {
    const earningsPaidSgd = ((gySalesRes.data as { payout_to_user_sgd: number | string | null }[] | null) ?? [])
      .reduce((s, r) => s + Number(r.payout_to_user_sgd ?? 0), 0);
    goldYield = { barsEnrolled: gyBarCount, gramsYielded: curveYielded, earningsPaidSgd };
  }
 
  // ── Vault custody ─────────────────────────────────────────────────────────
  const vaultHeld = (vaultHoldingsRes.data as { grams: number | string; status: string }[] | null) ?? [];
  const vaultGramsInCustody = vaultHeld.reduce((s, h) => s + Number(h.grams), 0);
  // Grams still being secured (ordered, not yet physically stored) — these can't
  // be withdrawn/sold yet, so the drawer softens its copy for them.
  const vaultGramsSettling = vaultHeld
    .filter((h) => h.status === "awaiting_storage")
    .reduce((s, h) => s + Number(h.grams), 0);
  let vault: CustomerNewsletterData["vault"] = null;
  if (vaultGramsInCustody > 0) {
    vault = { gramsInCustody: vaultGramsInCustody, gramsSettling: vaultGramsSettling, holdingsCount: vaultHeld.length };
  }
 
  // ── Buyback ─────────────────────────────────────────────────────────────
  type BbRow = { status: string; paid_at: string | null; total_payout_sgd: number | string | null; shipping_fee_sgd: number | string | null; created_at: string; buyback_request_lines: { grams: number | string }[] };
  const bbList = (bbRowsRes.data ?? []) as BbRow[];
  const paidBb = bbList.filter(b => b.status === "completed" || b.status === "paid" || !!b.paid_at);
  let buyback: CustomerNewsletterData["buyback"] = null;
  if (paidBb.length > 0) {
    const totalGrams = paidBb.reduce((s, b) =>
      s + (b.buyback_request_lines ?? []).reduce((g, l) => g + Number(l.grams), 0), 0);
    const grossPayout = paidBb.reduce((s, b) => s + Number(b.total_payout_sgd ?? 0), 0);
    const pickupFees  = paidBb.reduce((s, b) => s + Number(b.shipping_fee_sgd ?? 0), 0);
    const netPayout   = Math.max(0, grossPayout - pickupFees);
    const monthGrams  = paidBb
      .filter(b => b.created_at >= periodStart && b.created_at <= periodEndIso)
      .reduce((s, b) => s + (b.buyback_request_lines ?? []).reduce((g, l) => g + Number(l.grams), 0), 0);
    buyback = {
      totalGramsSold:     totalGrams,
      totalPayoutSgd:     netPayout,
      avgPerGramSgd:      totalGrams > 0 ? netPayout / totalGrams : null,
      gramsSoldThisMonth: monthGrams,
    };
  }
 
  // ── Conversion ──────────────────────────────────────────────────────────
  type ConvRow = { status: string; target_50g_count: number | string | null; target_100g_count: number | string | null; conversion_fee_sgd: number | string | null; total_charged_sgd: number | string | null; created_at: string };
  const convList = (convRowsRes.data ?? []) as ConvRow[];
  const completedConv = convList.filter(c => c.status === "completed" || c.status === "delivered");
  let conversion: CustomerNewsletterData["conversion"] = null;
  if (completedConv.length > 0) {
    conversion = {
      count50g:           completedConv.reduce((s, c) => s + Number(c.target_50g_count ?? 0), 0),
      count100g:          completedConv.reduce((s, c) => s + Number(c.target_100g_count ?? 0), 0),
      totalFeesSgd:       convList
        .filter(c => !["rejected", "cancelled"].includes(c.status))
        .reduce((s, c) => s + Number(c.total_charged_sgd ?? c.conversion_fee_sgd ?? 0), 0),
      completedThisMonth: completedConv.filter(c =>
        c.created_at >= periodStart && c.created_at <= periodEndIso
      ).length,
    };
  }
 
  // ── Spot move ───────────────────────────────────────────────────────────
  const spotOpenPerG  = spotOpenRes.data  ? Number((spotOpenRes.data  as { price_per_gram: number | string }).price_per_gram) : null;
  const spotClosePerG = spotCloseRes.data ? Number((spotCloseRes.data as { price_per_gram: number | string }).price_per_gram) : null;
  const spotPctChange = spotOpenPerG && spotClosePerG && spotOpenPerG > 0
    ? ((spotClosePerG - spotOpenPerG) / spotOpenPerG) * 100
    : null;
 
  // ── Daily spot series for the period (chart data) ──────────────────────
  // Each row { date: 'YYYYMMDD', price_per_gram: number } from gold_price_history.
  // Convert to a clean series for the SVG line chart on the market page.
  const spotSeriesRaw = (spotSeriesRes.data ?? []) as { date: string; price_per_gram: number | string }[];
  const spotSeries: { date: string; sgdPerG: number }[] = spotSeriesRaw.map(r => ({
    date:    r.date,
    sgdPerG: Number(r.price_per_gram),
  }));
 
  // ── Monthly accumulation timeline (trailing 12 months) ─────────────────
  // Bucket the accumulation rows we already fetched by YYYY-MM. For each
  // month, sum grams_added; carry a running cumulative total. Drives the
  // accumulation line/area chart on page 3. Months with zero activity
  // still emit a point so the line draws continuously.
  const monthlyBucket = new Map<string, number>();
  for (const r of accRows) {
    const d = new Date(r.recorded_at);
    const key = `${d.getUTCFullYear()}-${String(d.getUTCMonth() + 1).padStart(2, "0")}`;
    monthlyBucket.set(key, (monthlyBucket.get(key) ?? 0) + Number(r.grams));
  }
  const monthlyAccumulation: { yearMonth: string; gramsAdded: number; gramsCumulative: number }[] = [];
  let runningTotal = 0;
  for (let i = 11; i >= 0; i--) {
    const target = new Date(Date.UTC(periodYear, periodMonth - 1 - i, 1));
    const key = `${target.getUTCFullYear()}-${String(target.getUTCMonth() + 1).padStart(2, "0")}`;
    const added = monthlyBucket.get(key) ?? 0;
    runningTotal += added;
    monthlyAccumulation.push({ yearMonth: key, gramsAdded: added, gramsCumulative: runningTotal });
  }
  // Backfill cumulative for months before the 12-month window so the
  // first point in the series isn't artificially low. Add the
  // pre-window total to every cumulative point.
  const preWindowGrams = totalGrams - monthlyAccumulation.reduce((s, m) => s + m.gramsAdded, 0);
  for (const m of monthlyAccumulation) m.gramsCumulative += preWindowGrams;
 
  // ── Preferred currency ──────────────────────────────────────────────────
  const preferredCurrency = (custRes.data as { preferred_currency: string | null } | null)?.preferred_currency ?? "SGD";
 
  return {
    customerId,
    firstName,
    fullName,
    periodYear,
    periodMonth,
    tenureMonths,
    totalGrams,
    gramsAddedThisMonth,
    purchasesThisMonth,
    avgPremiumPctYTD,
    purchaseStreakMonths,
    monthsSinceLastPurchase,
    milestone,
    treeBranches,
    monthsToNext50g,
    monthsToNext100g,
    returnedAfterGap,
    joinedThisMonth,
    dormantWithBalance,
    curve: {
      gramsLifetime:        curveGramsLifetime,
      positionInCycle,
      cyclesCompleted,
      lifetimeSpendSgd:     curveStats.revenue,
      avgCostPerGramSgd:    curveStats.avgCostPerGram,
      avgPremiumPct:        curveStats.avgPremium,
      nextPurchaseRatePct,
      splitsInDrawerGrams,
      gramsAddedThisMonth:  curveGramsAddedThisMonth,
    },
    standard: {
      gramsLifetime:        standardGramsLifetime,
      itemsLifetime:        standardStats.itemCount,
      lifetimeSpendSgd:     standardStats.revenue,
      avgCostPerGramSgd:    standardStats.avgCostPerGram,
      avgPremiumPct:        standardStats.avgPremium,
      gramsAddedThisMonth:  standardGramsAddedThisMonth,
    },
    accessory: {
      itemsLifetime:        accessoryStats.itemCount,
      lifetimeSpendSgd:     accessoryStats.revenue,
      spendThisMonth:       accessorySpendThisMonth,
    },
    goldYield,
    vault,
    buyback,
    conversion,
    spotMove: {
      currency:  "SGD",
      openPerG:  spotOpenPerG,
      closePerG: spotClosePerG,
      pctChange: spotPctChange,
    },
    spotSeries,
    monthlyAccumulation,
    preferredCurrency,
  };
}
 
function detectMilestone(opts: {
  totalGramsBefore: number;
  totalGramsAfter:  number;
  purchasesThisMonth: number;
  joinedThisMonth: boolean;
  isFirstPurchase: boolean;
}): CustomerNewsletterData["milestone"] {
  if (opts.isFirstPurchase) return "first_purchase";
  if (opts.totalGramsBefore < 100 && opts.totalGramsAfter >= 100) return "first_combibar_100g";
  if (opts.totalGramsBefore < 50  && opts.totalGramsAfter >= 50)  return "first_combibar_50g";
  if (opts.totalGramsBefore < 25  && opts.totalGramsAfter >= 25)  return "halfway_to_50g";
  if (opts.totalGramsBefore < 10  && opts.totalGramsAfter >= 10)  return "first_10g";
  return null;
}

function isoDate(year: number, month: number, day: number): string {
  return `${year}-${String(month).padStart(2, "0")}-${String(day).padStart(2, "0")}`;
}
 
function isoEndOfMonth(year: number, month: number): string {
  const d = new Date(Date.UTC(year, month, 0));
  return `${d.getUTCFullYear()}-${String(d.getUTCMonth() + 1).padStart(2, "0")}-${String(d.getUTCDate()).padStart(2, "0")}`;
}
 
function monthsBetween(a: Date, b: Date): number {
  return (b.getUTCFullYear() - a.getUTCFullYear()) * 12
       + (b.getUTCMonth() - a.getUTCMonth());
}
