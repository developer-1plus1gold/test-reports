/**
 * Newsletter fan-out — send a published brief to every subscribed customer via
 * their available channels (email + WhatsApp). Shared by the hourly cron
 * (`/api/cron/newsletter-notify`) and the manual admin trigger
 * (`/api/admin/newsletters/send-now`).
 *
 * Opt-out model (2026-07-03 directive): the newsletter is DEFAULT-ON. A customer
 * is a recipient unless they explicitly unsubscribed
 * (`newsletter_preferences.unsubscribed_at IS NOT NULL`). Internal (demo/test)
 * customers are always excluded. Delivery fans out to every channel the customer
 * has (the "newsletter" category is DEFAULT_ON in notify-customer.ts, so an
 * un-narrowed customer receives it on both email and WhatsApp).
 *
 * Idempotent: a customer already marked notified for this brief
 * (`newsletter_views.notification_sent_at`) is skipped unless `force`.
 */
 
import type { SupabaseClient } from "@supabase/supabase-js";
import { getInternalCustomerIds } from "@/lib/finance/internal-customers";

export interface FanoutOptions {
  /** Compute recipients + resolved channels but send nothing. */
  dryRun?: boolean;
  /** Re-send even to customers already marked notified for this brief. */
  force?: boolean;
  /** Target a specific brief; defaults to the latest published one. */
  briefId?: string;
}
 
export interface FanoutRecipient {
  customerId: string;
  firstName:  string;
  email:      boolean;   // has a real (non-synthetic) email
  whatsapp:   boolean;   // has a phone
}
 
export interface FanoutResult {
  ok:        boolean;
  reason?:   string;
  dryRun:    boolean;
  brief?:    { id: string; year: number; month: number; monthYear: string };
  eligible:  number;          // subscribed, non-internal
  alreadyDone: number;        // already notified for this brief
  skipped?:  number;          // resolved but unreachable (deleted / no channel)
  sent:      number;
  recipients?: FanoutRecipient[];   // dry-run: who would be sent to
  errors?:   { customerId: string; error: string }[];
}
 
function truncate(s: string, n: number): string {
  return s.length > n ? s.slice(0, n - 1) + "…" : s;
}
 
/** One `newsletter_deliveries` row per channel that was actually attempted. */
type ChannelOutcome = { ok: boolean; reason?: string };
 
async function recordDeliveries(
  db: SupabaseClient,
  customerId: string,
  briefId: string,
  result: { whatsapp: ChannelOutcome; email: ChannelOutcome },
): Promise<void> {
  // A channel the customer does not have is reported as not-ok with a reason,
  // and must NOT be logged as a failed delivery — "no phone on file" is not a
  // delivery failure, and counting it as one would make every email-only
  // customer look like a broken WhatsApp send.
  const rows = ([["email", result.email], ["whatsapp", result.whatsapp]] as const)
    .filter(([, r]) => r.ok || (r.reason && !NOT_ATTEMPTED_REASONS.has(r.reason)))
    .map(([channel, r]) => ({
      customer_id:     customerId,
      market_brief_id: briefId,
      channel,
      status:          r.ok ? "sent" : "failed",
      error:           r.ok ? null : (r.reason ?? "unknown"),
      sent_at:         new Date().toISOString(),
    }));
  if (rows.length === 0) return;
 
  try {
    // The fan-out is idempotent and cron-retried, so the same
    // (customer, brief, channel) can legitimately be attempted twice; upsert on
    // the unique index keeps channel totals from double-counting a retry.
    await db.from("newsletter_deliveries")
      .upsert(rows, { onConflict: "customer_id,market_brief_id,channel" });
  } catch (err) {
    console.error("[newsletter/fanout] delivery telemetry write failed:",
      err instanceof Error ? err.message : String(err));
  }
}
