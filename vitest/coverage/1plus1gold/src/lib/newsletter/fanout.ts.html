/**
 * Newsletter fan-out — send a published brief to every subscribed customer via
 * their available channels (email + WhatsApp). Shared by the hourly cron
 * (`/api/cron/newsletter-notify`) and the manual admin trigger
 * (`/api/admin/newsletters/send-now`).
 *
 * Opt-out model (2026-07-03 directive): the newsletter is DEFAULT-ON. A customer
 * is a recipient unless they explicitly unsubscribed
 * (`newsletter_preferences.unsubscribed_at IS NOT NULL`). Internal (demo/test)
 * customers are always excluded. Delivery fans out to every channel the customer
 * has (the "newsletter" category is DEFAULT_ON in notify-customer.ts, so an
 * un-narrowed customer receives it on both email and WhatsApp).
 *
 * Idempotent: a customer already marked notified for this brief
 * (`newsletter_views.notification_sent_at`) is skipped unless `force`.
 *
 * EMAIL IS SPREAD OVER SEVERAL DAYS (2026-10-06, Mukesh's call — stay on the
 * Resend free plan, 100 emails/day). WhatsApp goes to everyone on the first
 * run, as before. Email is capped per rolling 24h at
 * NEWSLETTER_EMAIL_DAILY_CAP minus EVERY email Resend already sent in that
 * window (receipts, OTP codes, alerts included), so transactional mail always
 * keeps TRANSACTIONAL_EMAIL_RESERVE of headroom. Email-only customers go
 * first — for them the email is the only copy; customers who already have the
 * WhatsApp get their email copy on a later run. A customer is locked
 * (`notification_sent_at`) only once nothing is left pending, and the
 * per-channel lock is `newsletter_deliveries`, so a later run sends only the
 * channel still owed. If the backlog needs more than MAX_SPREAD_DAYS, the
 * admins are told once per brief — that is the signal to consider the paid plan.
 */

/** Resend free plan: 100 emails per day across ALL email the app sends. */
export const RESEND_DAILY_LIMIT = 100;
/** Always left free for receipts, sign-in codes and alerts. */
export const TRANSACTIONAL_EMAIL_RESERVE = 30;
/** Most newsletter emails that may go out in any rolling 24h. */
export const NEWSLETTER_EMAIL_DAILY_CAP = RESEND_DAILY_LIMIT - TRANSACTIONAL_EMAIL_RESERVE;
/** A backlog needing more days than this alerts the admins. */
export const MAX_SPREAD_DAYS = 4;
 
/**
 * How many newsletter emails this run may send, given everything Resend has
 * already sent in the last 24 hours (newsletter and transactional alike).
 * Counting ALL sends is what keeps the reserve real: a busy order day leaves
 * less room for the newsletter, never less room for a receipt.
 */
export function newsletterEmailBudget(sentLast24h: number): number {
  const used = Number.isFinite(sentLast24h) ? Math.max(0, sentLast24h) : RESEND_DAILY_LIMIT;
  return Math.max(0, Math.min(NEWSLETTER_EMAIL_DAILY_CAP, RESEND_DAILY_LIMIT - TRANSACTIONAL_EMAIL_RESERVE - used));
}
 
/** Days needed to clear `pendingEmails` at the daily cap (today counts as one). */
export function emailSpreadDays(pendingEmails: number): number {
  return pendingEmails <= 0 ? 0 : Math.ceil(pendingEmails / NEWSLETTER_EMAIL_DAILY_CAP);
}
 
/**
 * Email queue order: customers who have NO WhatsApp first (the email is their
 * only copy), then everyone else. Stable within each group so a backlog drains
 * in the same order run after run.
 */
export function orderEmailQueue<T extends { customerId: string; whatsapp: boolean }>(rows: T[]): T[] {
  return [...rows].sort((a, b) =>
    (a.whatsapp === b.whatsapp ? 0 : a.whatsapp ? 1 : -1) || a.customerId.localeCompare(b.customerId));
}
 
/** Channels already settled for one customer on this brief. */

/**
 * Which channels are still owed. A channel with a `sent` row is done. A
 * `failed` row also counts as done once the customer has received the
 * newsletter on some channel — the same rule as before this change (a
 * customer who got one channel was locked and never retried). If nothing has
 * ever succeeded, failures are retried, as before.
 */
export function pendingChannels(
  r: { whatsapp: boolean; email: boolean },
  done: DeliveredChannels | undefined,
  force: boolean,
): { whatsapp: boolean; email: boolean } {
  if (force || !done) return { whatsapp: r.whatsapp, email: r.email };
  const anySent = done.whatsapp === "sent" || done.email === "sent";
  const settled = (s?: "sent" | "failed") => s === "sent" || (s === "failed" && anySent);
  return { whatsapp: r.whatsapp && !settled(done.whatsapp), email: r.email && !settled(done.email) };
}
