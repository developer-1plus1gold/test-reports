/**
 * Singapore recipient-phone normalization.
 *
 * Delivery is SG-only (checkout locks the country to Singapore), so a recipient
 * phone must be a valid SG number in +65 E.164 form for the courier (NinjaVan /
 * Lalamove) to reach it. The web checkout used a free-text field with no
 * enforcement, so a number missing the country code stored malformed — e.g.
 * "+97304637" (a real order: the +65 was dropped, leaving "+" + the 8 local
 * digits). This normalizes whatever the operator/customer typed — with or
 * without +65, spaces, or a leading 0/00 — into "+65XXXXXXXX", and reports
 * validity so the form can reject genuinely bad input.
 *
 * SG local numbers are 8 digits; the leading digit is 6 (landline), 8/9
 * (mobile) or 3 (VoIP).
 */
 
const SG_LOCAL_RE = /^[3689]\d{7}$/;
 
export function normalizeSgPhone(input: string | null | undefined): { e164: string; valid: boolean } {
  const raw = (input ?? "").trim();
  let digits = raw.replace(/\D/g, "");
  // Strip an international prefix: 0065… or +65… → 65…, then drop the 65.
  if (digits.startsWith("0065")) digits = digits.slice(4);
  if (digits.startsWith("65") && digits.length === 10) digits = digits.slice(2);
  const valid = SG_LOCAL_RE.test(digits);
  return { e164: valid ? "+65" + digits : raw, valid };
}
