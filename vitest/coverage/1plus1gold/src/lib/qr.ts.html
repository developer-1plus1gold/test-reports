/**
 * QR geometry — the shared half of every QR we print.
 *
 * Extracted from src/lib/coa-qr.ts (2026-09-17) when the parcel insert card
 * became the second printed artefact to carry a QR. The CoA's header had
 * accumulated three hard-won rules that are properties of PRINTED QRs in
 * general, not of certificates, and a second copy of them is a second thing to
 * get wrong:
 *
 * ▶ THE QUIET ZONE COMES FROM THE LIBRARY, not from CSS. The emitted path is
 *   offset by QUIET_ZONE modules and the viewBox is widened to match, so the
 *   4-module border the QR spec requires is part of the geometry itself.
 *   Correctness then does not depend on a padding value in a component someone
 *   may later restyle. Plate padding around it is purely aesthetic.
 *
 * ▶ PATH DATA, NEVER AN SVG STRING. The obvious implementation is
 *   `QRCode.toString({type:"svg"})` piped into `dangerouslySetInnerHTML`. Do
 *   not. The attacker playbook's F5 disposes every CSP-chain finding on the
 *   standing fact that `src/` carries no XSS sink beyond JsonLd.tsx, whose
 *   `application/ld+json` payload is non-executable — a fact now asserted by
 *   `src/__tests__/dangerously-set-inner-html-inventory.test.ts` rather than
 *   counted by hand. Adding an injection site to sell a decorative square would
 *   invalidate a standing security disposition. Returning geometry lets a React
 *   caller render an ordinary <svg><path/></svg>, which has no injection
 *   surface at all, and lets an HTML-string caller interpolate a value that is
 *   structurally incapable of carrying markup (see `assertPathSafe`).
 *
 * ▶ SVG, NOT A PNG DATA-URI: vector, so it prints sharp at any size on any
 *   printer, and it is a fraction of the bytes.
 *
 * Two things deliberately stay with the CALLER, because they are editorial
 * rather than mechanical: the error-correction level (a payload's robustness
 * budget is a judgement about where it will be handled) and the colour (near-
 * black ink on a white plate — tinting a QR is the classic way to trade scan
 * reliability for decoration, and a code nobody can scan fails at its one job).
 */
import QRCode from "qrcode";
 
/** Quiet zone in modules. 4 is the QR spec's minimum; below it, scanners fail. */
export const QUIET_ZONE = 4;
 
export interface QrGeometry {
  /** SVG path data for every dark module, in a `viewBox="0 0 size size"` space. */
  path: string;
  /** Side of the square viewBox, in modules — code width plus both quiet zones. */
  size: number;
}
 
/**
 * Encode `payload` and return its dark modules as SVG path data.
 *
 * `ecLevel` is the caller's call: "M" (15%) is the everyday default, "Q" (25%)
 * buys real robustness for paper that gets folded, handled and smudged — worth
 * it whenever the payload is short enough that the extra redundancy does not
 * push the code into a denser version.
 */
export function qrGeometry(payload: string, ecLevel: "L" | "M" | "Q" | "H"): QrGeometry {
  const qr = QRCode.create(payload, { errorCorrectionLevel: ecLevel });
  const n = qr.modules.size;
  const d = qr.modules.data;
 
  // Emit one path command per RUN of adjacent dark modules rather than one per
  // module — same rendering, materially smaller document, and it keeps a
  // printed artefact's markup from being mostly QR.
  const parts: string[] = [];
  for (let y = 0; y < n; y++) {
    let run = 0;
    for (let x = 0; x <= n; x++) {
      const dark = x < n && d[y * n + x] === 1;
      if (dark) { run++; continue; }
      if (run > 0) {
        parts.push(`M${x - run + QUIET_ZONE} ${y + QUIET_ZONE}h${run}v1h-${run}z`);
        run = 0;
      }
    }
  }
 
  return { path: parts.join(""), size: n + QUIET_ZONE * 2 };
}
 
/**
 * Guard for the HTML-STRING callers only (an artefact built as a string and
 * handed to `iframe srcdoc`, which is same-origin and therefore a real sink if
 * anything attacker-shaped ever reaches it).
 *
 * A JSX caller needs nothing — React escapes attribute values. A string caller
 * is interpolating straight into markup, so it asserts what it is about to
 * interpolate. This is cheap insurance against a future refactor that routes a
 * customer-supplied string through `qrGeometry` and into a document: path data
 * is a closed alphabet, so anything outside it means the assumption broke.
 */
export function assertPathSafe(path: string): string {
  if (!/^[Mmhvz0-9 .,-]*$/.test(path)) {
    throw new Error("qr: path data contains unexpected characters — refusing to interpolate into markup");
  }
  return path;
}
