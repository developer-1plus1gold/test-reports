/**
 * Automation-monitor pattern — shared helper for type='automation' agents.
 *
 * An automation monitor watches some condition (a query against a domain
 * table) and reflects matches into the unified Control Tower queue via
 * agent_decisions. Two passes per cron tick:
 *
 *   Forward:  rows currently matching the condition → ensure each has
 *             an open agent_decisions row pointing at it. Idempotent
 *             via lookup on (related_entity_type, related_entity_id).
 *
 *   Backward: open agent_decisions rows whose underlying condition no
 *             longer matches → flip to status='executed' (self-healing).
 *             This is what makes the queue stay accurate without manual
 *             housekeeping.
 *
 * No LLM. No agent_run_log writes. agent_events row written for each
 * forward-create + backward-close so the audit trail shows the cron's
 * activity per tick.
 */
 
import type { SupabaseClient } from "@supabase/supabase-js";
import { sendAdminAlert } from "@/lib/twilio";
import { getAdminPhonesCsv } from "@/lib/admin-team";
 
export interface MonitorMatch {
  /** The id of the row in the source domain table. */
  id: string;
  /** Free-form summary for the queue card (e.g. "Order ord_abc payment stuck 67min"). */
  summary: string;
  /** Anything else worth carrying into action_payload for the detail panel. */
  payload?: Record<string, unknown>;
  /**
   * Override the run-level severity for THIS match. Defaults to `args.severity`,
   * so every existing caller is unaffected.
   *
   * ▶ WHY PER-MATCH. A monitor watches several conditions under one agent, and
   * they are not equally urgent. `shipping_monitor` is the case that forced it:
   * "you have not pressed Book yet" belongs in the queue, while "the carrier has
   * had this for 45 minutes and gives up at 120" needs to reach a phone. Only
   * `escalate` pings (see the forward pass below), so with one run-level
   * severity the monitor could either ping for everything or — as it did — ping
   * for nothing.
   *
   * Splitting into two runMonitor calls is NOT an alternative: both passes key on
   * (agentName, entityType), so the second call's backward pass would close every
   * row the first call had just opened.
   */
  severity?: "escalate" | "approve" | "notify";
}
 
export interface MonitorRunArgs {
  sb:                  SupabaseClient;
  agentName:           string;
  severity:            "escalate" | "approve" | "notify";
  /** Used for related_entity_type on agent_decisions rows. */
  entityType:          string;
  /** Current matches from the source-of-truth query. */
  matches:             MonitorMatch[];
}
 
export interface MonitorRunResult {
  opened:  number;
  closed:  number;
  matched: number;
}

/**
 * Record one observability tick for an automation agent. Called by every
 * cron at the end of its run with a snapshot payload describing what it
 * observed — healthy or not. Surfaced on the agent detail page as
 * "Latest observation" so admin can see "checked 12s ago, all healthy".
 *
 * Soft-fails. Storage cost ≈ 1 row per cron tick per agent.
 */
export async function recordMonitorTick(
  sb:        SupabaseClient,
  agentName: string,
  payload:   Record<string, unknown>,
): Promise<void> {
  const { error } = await sb.from("agent_events").insert({
    agent_name: agentName,
    event_type: "monitor_tick",
    actor:      "system:cron",
    payload,
  });
  if (error) {
    console.warn(`[automation-monitor:${agentName}] tick insert failed: ${error.message}`);
  }
}
