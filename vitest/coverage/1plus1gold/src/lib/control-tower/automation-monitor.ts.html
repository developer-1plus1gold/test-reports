/**
 * Automation-monitor pattern — shared helper for type='automation' agents.
 *
 * An automation monitor watches some condition (a query against a domain
 * table) and reflects matches into the unified Control Tower queue via
 * agent_decisions. Two passes per cron tick:
 *
 *   Forward:  rows currently matching the condition → ensure each has
 *             an open agent_decisions row pointing at it. Idempotent
 *             via lookup on (related_entity_type, related_entity_id).
 *
 *   Backward: open agent_decisions rows whose underlying condition no
 *             longer matches → flip to status='executed' (self-healing).
 *             This is what makes the queue stay accurate without manual
 *             housekeeping.
 *
 * No LLM. No agent_run_log writes. agent_events row written for each
 * forward-create + backward-close so the audit trail shows the cron's
 * activity per tick.
 */
 
import type { SupabaseClient } from "@supabase/supabase-js";
import { sendAdminAlert } from "@/lib/twilio";
import { getAdminPhonesCsv } from "@/lib/admin-team";
 
export interface MonitorMatch {
  /** The id of the row in the source domain table. */
  id: string;
  /** Free-form summary for the queue card (e.g. "Order ord_abc payment stuck 67min"). */
  summary: string;
  /** Anything else worth carrying into action_payload for the detail panel. */
  payload?: Record<string, unknown>;
}
 
export interface MonitorRunArgs {
  sb:                  SupabaseClient;
  agentName:           string;
  severity:            "escalate" | "approve" | "notify";
  /** Used for related_entity_type on agent_decisions rows. */
  entityType:          string;
  /** Current matches from the source-of-truth query. */
  matches:             MonitorMatch[];
}
 
export interface MonitorRunResult {
  opened:  number;
  closed:  number;
  matched: number;
}
 
export async function runMonitor(args: MonitorRunArgs): Promise<MonitorRunResult> {
  const matchIds = new Set(args.matches.map((m) => m.id));
 
  // ── Forward pass: open new decisions for any unmatched current rows.
  // Look up existing open decisions for this agent so we don't double-write.
  const { data: existingOpen } = await args.sb
    .from("agent_decisions")
    .select("id, related_entity_id, status")
    .eq("agent_name", args.agentName)
    .eq("related_entity_type", args.entityType)
    .in("status", ["proposed", "edited", "approved"]);
 
  const existingByEntity = new Map<string, { id: string; status: string }>();
  for (const e of (existingOpen ?? []) as Array<{ id: string; related_entity_id: string; status: string }>) {
    existingByEntity.set(e.related_entity_id, { id: e.id, status: e.status });
  }
 
  const toOpen = args.matches.filter((m) => !existingByEntity.has(m.id));
  let opened = 0;
  if (toOpen.length > 0) {
    const rows = toOpen.map((m) => ({
      agent_name:          args.agentName,
      severity:            args.severity,
      summary:             m.summary,
      status:              "proposed",
      action_payload:      m.payload ?? {},
      related_entity_type: args.entityType,
      related_entity_id:   m.id,
    }));
    const { error } = await args.sb.from("agent_decisions").insert(rows);
    if (error) {
      console.warn(`[automation-monitor:${args.agentName}] open insert failed: ${error.message}`);
    } else {
      opened = rows.length;
      // Audit event per opened row.
      await args.sb.from("agent_events").insert(
        toOpen.map((m) => ({
          agent_name:          args.agentName,
          event_type:          "automation_monitor_opened",
          actor:               "system:cron",
          payload:             { entity_type: args.entityType, entity_id: m.id, summary: m.summary },
        })),
      );
 
      // Immediate WhatsApp ping for freshly-opened escalate-severity rows.
      // The severity ladder in CLAUDE.md says escalate = "agent stops; pages
      // Mukesh on WhatsApp." Pre-2026-05-20 this was a documentation-only
      // contract — escalate rows only surfaced in the 8am/8pm digest. After
      // the 2026-05 price-service outage (Railway, since retired — silent for
      // ~12h until 8am digest
      // would have fired), wiring the immediate ping here makes escalate
      // mean what it says.
      //
      // Soft-fails: WhatsApp errors are logged and never block the cron's
      // primary job (queue maintenance). De-dup is already provided by the
      // forward-pass logic — repeated ticks won't re-open the same row, so
      // a single WhatsApp is sent per incident, not per tick.
      if (args.severity === "escalate") {
        const adminPhones = getAdminPhonesCsv();
        if (adminPhones) {
          const source = `Agent: ${args.agentName}`;
          for (const m of toOpen) {
            try {
              await sendAdminAlert(adminPhones, "critical", source, m.summary);
            } catch (err) {
              console.warn(
                `[automation-monitor:${args.agentName}] WhatsApp escalate failed:`,
                err instanceof Error ? err.message : err,
              );
            }
          }
        }
      }
    }
  }
 
  // ── Backward pass: close open decisions whose entity no longer matches.
  const toClose = ((existingOpen ?? []) as Array<{ id: string; related_entity_id: string; status: string }>)
    .filter((e) => !matchIds.has(e.related_entity_id) && e.status !== "approved");
 
  let closed = 0;
  if (toClose.length > 0) {
    const ids = toClose.map((e) => e.id);
    const { error } = await args.sb
      .from("agent_decisions")
      .update({
        status:           "executed",
        resolved_at:      new Date().toISOString(),
        resolution_notes: "Automation monitor: condition no longer matches",
      })
      .in("id", ids);
    if (error) {
      console.warn(`[automation-monitor:${args.agentName}] close update failed: ${error.message}`);
    } else {
      closed = ids.length;
      await args.sb.from("agent_events").insert(
        toClose.map((e) => ({
          agent_name:          args.agentName,
          event_type:          "automation_monitor_closed",
          related_decision_id: e.id,
          actor:               "system:cron",
          payload:             { reason: "condition_no_longer_matches" },
        })),
      );
    }
  }
 
  return {
    opened,
    closed,
    matched: args.matches.length,
  };
}
 
/**
 * Record one observability tick for an automation agent. Called by every
 * cron at the end of its run with a snapshot payload describing what it
 * observed — healthy or not. Surfaced on the agent detail page as
 * "Latest observation" so admin can see "checked 12s ago, all healthy".
 *
 * Soft-fails. Storage cost ≈ 1 row per cron tick per agent.
 */
export async function recordMonitorTick(
  sb:        SupabaseClient,
  agentName: string,
  payload:   Record<string, unknown>,
): Promise<void> {
  const { error } = await sb.from("agent_events").insert({
    agent_name: agentName,
    event_type: "monitor_tick",
    actor:      "system:cron",
    payload,
  });
  if (error) {
    console.warn(`[automation-monitor:${agentName}] tick insert failed: ${error.message}`);
  }
}
