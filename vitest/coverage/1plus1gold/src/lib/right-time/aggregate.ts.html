export function median(xs: number[]): number {
  const s = [...xs].sort((a, b) => a - b);
  const m = Math.floor(s.length / 2);
  return s.length % 2 ? s[m] : (s[m - 1] + s[m]) / 2;
}
 
/**
 * Average of the EXTREME_SAMPLE highest and lowest buy prices. Never a single
 * purchase: below EXTREME_SAMPLE buys both are null.
 */
export function extremeBuyLevels(buys: number[]): { buyersHighSgd: number | null; buyersLowSgd: number | null } {
  if (buys.length < EXTREME_SAMPLE) return { buyersHighSgd: null, buyersLowSgd: null };
  const sorted = [...buys].sort((a, b) => a - b);
  const avg = (xs: number[]) => r2(xs.reduce((a, b) => a + b, 0) / xs.length);
  return { buyersHighSgd: avg(sorted.slice(-EXTREME_SAMPLE)), buyersLowSgd: avg(sorted.slice(0, EXTREME_SAMPLE)) };
}

/** Date parts in Singapore time. */
function sgt(iso: string) {
  const d = new Date(new Date(iso).getTime() + SGT_OFFSET_MS);
  return { dow: d.getUTCDay(), dom: d.getUTCDate(), hour: d.getUTCHours(), ymd: ymdOf(d) };
}
function ymdOf(d: Date): string {
  return `${d.getUTCFullYear()}${String(d.getUTCMonth() + 1).padStart(2, "0")}${String(d.getUTCDate()).padStart(2, "0")}`;
}
function ymdToMs(ymd: string): number {
  return Date.UTC(+ymd.slice(0, 4), +ymd.slice(4, 6) - 1, +ymd.slice(6, 8));
}

/**
 * One report per selectable period. A period longer than our history simply
 * uses what we have: orders and the price window both start no earlier than
 * the first order, so "1Y" and "All" coincide until we are a year old.
 */
export function buildPeriodReports(
  orders: RtOrder[],
  items: RtItem[],
  history: RtSpotPoint[],
  yearly: RtYearPoint[] = [],
  nowMs: number = Date.now(),
): RightTimePeriods {
  const firstYmd = orders.length ? orders.map((o) => sgt(o.createdAt).ymd).sort()[0] : null;
  const sinceMs = firstYmd ? ymdToMs(firstYmd) : undefined;
  const periods = {} as Record<RtPeriod, RightTimeReport>;
  for (const p of RT_PERIODS) {
    const cutoff = p === "ALL" ? -Infinity : nowMs - PERIOD_DAYS[p] * DAY_MS;
    const os = orders.filter((o) => new Date(o.createdAt).getTime() >= cutoff);
    const ids = new Set(os.map((o) => o.id));
    const from = Math.max(cutoff, sinceMs ?? -Infinity);
    periods[p] = buildRightTimeReport(os, items.filter((i) => ids.has(i.orderId)), history, yearly, {
      levelsFromMs: Number.isFinite(from) ? from : undefined,
    });
  }
  return { dataSince: firstYmd, periods };
}
 
/** Annual series from 2015 on, sorted, positive prices only. */
