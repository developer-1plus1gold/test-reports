/**
 * Is this product our signature 1g card — the ONLY thing we repack? A 1g curve
 * piece from Valcambi (separated from a CombiBar). Every other piece, including
 * a sealed 5g/10g Valcambi bar on the curve, keeps its "Sealed Assay" badge
 * (bug fixed 2026-10-30: the check had no weight, so whole curve bars read as
 * repacked). The iOS shop spec carries the same rule as a visible_when.
 */
export function isSignatureCard(p: { product_type: string; refinery?: string | null; grams: number }): boolean {
  return p.product_type === "curve" && Number(p.grams) === 1 && (p.refinery ?? "").toLowerCase().includes("valcambi");
}
