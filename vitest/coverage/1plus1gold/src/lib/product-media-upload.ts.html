/**
 * True when `path` is a key the mint route could have produced for THIS
 * product: directly under `${productId}/`, in the minted filename shape. A key
 * under another product, a traversal, or an arbitrary name is refused.
 */
export function isMintedProductMediaPath(productId: string, path: string): boolean {
  const prefix = productMediaPrefix(productId);
  if (!path.startsWith(prefix)) return false;
  const file = path.slice(prefix.length);
  if (file.includes("/")) return false;
  return MINTED_RE.test(file);
}
 
/** The kind a minted path holds, from its extension. */
export function kindForMintedPath(path: string): ProductMediaKind {
  return /\.(mp4|webm|mov)$/i.test(path) ? "video" : "image";
