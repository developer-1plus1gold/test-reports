/**
 * Serial-photo batch → which card each photo belongs to (2026-11-06).
 *
 * The admin "Attach batch" action reads every photo in a `serial-inbox` batch
 * (one bar or card per photo) with the serial_photo_reader agent, then calls
 * this to decide. It is PURE and deliberately strict: a photo that cannot be
 * tied to exactly one card, with no contradiction, is never attached — it stays
 * in the inbox and the operator sees why. A missed photo costs one manual
 * attach; a WRONG photo on a certificate is a customer-facing error.
 */
import { isValidSerialFormat, normaliseSerial } from "@/lib/nfc/serial";
 
export interface PhotoReading {
  /** Number printed on the assay card ("Certificate Number" on PAMP cards). */
  certificate_number: string | null;
  /** Number engraved on the bar itself, when legible. */
  engraved_number: string | null;
  /** Our own NFC card serial (e.g. "1G-NUE6-24DD") if the photo shows one of ours. */
  card_serial: string | null;
  /** 0–1: how sure the reader is of the numbers it returned. */
  confidence: number;
}
 
export interface SerialRow {
  serial: string;
  manufacturer_serial: string | null;
  status: string;
  image_url: string | null;
}
 
export type PhotoDecision =
  | { action: "attach"; serial: string; matchedOn: "card_serial" | "manufacturer_serial"; value: string }
  | { action: "skip"; reason: string };
 
/** Readings below this are never trusted to attach on their own. */
export const MIN_READ_CONFIDENCE = 0.8;
 
/** Upper-case, strip spaces — refiner certificate numbers are short alphanumerics. */
export function cleanMakerSerial(v: string | null | undefined): string | null {
  const s = (v ?? "").replace(/\s+/g, "").toUpperCase();
  return /^[A-Z0-9-]{4,20}$/.test(s) ? s : null;
}
 
/** The value to look up, and on which column. Null when nothing usable was read
 *  or the photo contradicts itself. */
export function lookupKey(r: PhotoReading):
  | { on: "card_serial" | "manufacturer_serial"; value: string }
  | { conflict: string }
  | null {
  const card = r.card_serial ? normaliseSerial(r.card_serial) : null;
  if (card && isValidSerialFormat(card)) return { on: "card_serial", value: card };
 
  const cert = cleanMakerSerial(r.certificate_number);
  const engraved = cleanMakerSerial(r.engraved_number);
  if (cert && engraved && cert !== engraved) {
    return { conflict: `the card says ${cert} but the bar reads ${engraved}` };
  }
  const value = cert ?? engraved;
  return value ? { on: "manufacturer_serial", value } : null;
}
 
/**
 * Decide one photo. `rows` are the nfc_serials rows matching the lookup value
 * (by `serial` for card_serial, by `manufacturer_serial` otherwise), any status.
 */
export function decidePhoto(r: PhotoReading, rows: SerialRow[], opts: { overwrite?: boolean } = {}): PhotoDecision {
  if (!(r.confidence >= MIN_READ_CONFIDENCE)) {
    return { action: "skip", reason: `the numbers were not read clearly (confidence ${Math.round((r.confidence || 0) * 100)}%)` };
  }
  const key = lookupKey(r);
  if (!key) return { action: "skip", reason: "no serial or certificate number was readable" };
  if ("conflict" in key) return { action: "skip", reason: key.conflict };
 
  const live = rows.filter((row) => row.status !== "voided");
  if (live.length === 0) {
    return { action: "skip", reason: `${key.value} is not a received bar (no card exists for it yet)` };
  }
  if (live.length > 1) {
    return { action: "skip", reason: `${key.value} matches ${live.length} cards — attach it by hand` };
  }
  const row = live[0];
  if (row.image_url && !opts.overwrite) {
    return { action: "skip", reason: `${row.serial} already has a photo` };
  }
  return { action: "attach", serial: row.serial, matchedOn: key.on, value: key.value };
}
