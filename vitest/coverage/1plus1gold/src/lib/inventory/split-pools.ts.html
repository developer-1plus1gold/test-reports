/**
 * CombiBar↔split pool status (Stage 1, read-only). An unbroken CombiBar is
 * sellable whole AND its grams back the 1g-split pool; bundles draw N grams from
 * the same pool. Availability is computed from grams:
 *
 *   pool_free_grams = available 1g split rows + Σ(available CombiBar grams − committed)
 *   unit avail   = floor(pool_free_grams)
 *   N-pack avail = floor(pool_free_grams / N)
 *   whole avail  = count of available CombiBars of that SKU with 0 committed grams
 *
 * Backed by v_split_pool_grams + v_combibar_committed + split_pool_members.
 *
 * ⚠️ THIS FORMULA EXISTS TWICE — here (display) and in plpgsql
 * `recompute_product_stock_qty` (20260717h), which WRITES products.stock_qty and
 * therefore governs what the shop will actually sell. They must agree, and
 * nothing in the database can enforce that, so `poolAvailability()` below is the
 * single TypeScript definition and `split-pools-drift.test.ts` asserts the SQL
 * branches still match it. Same discipline as the pricing kernel's zero-drift CI.
 */
 
import type { SupabaseClient } from "@supabase/supabase-js";
 
/**
 * Units of a pool member buyable right now.
 *
 * `unit` is FLOORED (fixed 2026-07-20): it previously returned raw
 * `pool_free_grams` here while the plpgsql wrote `floor(...)` into stock_qty, so
 * a pool holding fractional grams would have shown e.g. 67.5 on the pool page
 * and 67 in the shop. No pool carries fractional grams today, but six
 * fractional-gram products exist — it was one catalogue change from being real.
 */
export function poolAvailability(
  role: "unit" | "bundle" | "source",
  poolFreeGrams: number,
  memberGrams: number,
  wholeSellableCount: number,
): number {
  if (role === "unit")   return Math.max(0, Math.floor(poolFreeGrams));
  if (role === "bundle") return memberGrams > 0 ? Math.max(0, Math.floor(poolFreeGrams / memberGrams)) : 0;
  return Math.max(0, wholeSellableCount);
}
 
/**
 * How many NFC cards a packed order line needs.
 *
 * A pool `bundle` product ("1g x 5", "1g x 10") is ONE catalogue row that ships
 * as N physically separate 1g pieces, each with its own NFC card, certificate
 * and buyback path. So the card count is `quantity × packGrams`, not `quantity`.
 *
 * `packGrams` is `split_pool_members.grams` for the line's product where the
 * role is `bundle` — pass null/undefined for every other product (a plain 1g
 * split, a standard bar, an accessory), which needs one card per unit.
 *
 * Before 2026-07-21 three call sites each used `quantity` directly, so a
 * "1g x 10" line accepted 1 card and the packing gate called it ready with 9
 * cards missing. This is the single definition — see the callers in
 * /api/admin/serials/assign, NfcRowEditor and PackingChecklistPanel.
 */
export function cardsRequired(quantity: number, packGrams?: number | null): number {
  const qty  = Math.max(0, Math.floor(Number(quantity) || 0));
  const pack = Number(packGrams);
  // Only a genuine multi-piece bundle multiplies. A 1-gram `unit` member has
  // grams = 1, which must NOT be treated as a multiplier of anything else.
  if (Number.isFinite(pack) && pack > 1) return qty * Math.floor(pack);
  return qty;
}
