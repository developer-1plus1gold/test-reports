/**
 * stock_adjustments — the append-only ledger of deliberate counter changes.
 *
 * WHY THIS EXISTS (drift investigation, 2026-07-19): `products.stock_qty` was
 * the only quantity in the system that could move without leaving a trace. Every
 * other movement writes a row somewhere — PO receipt, packing event, giveaway,
 * gold reservation — but a hand-edit of the counter wrote nothing. On PROD the
 * Gift Box showed 75 units gone with only 72 explained, and the missing 3 were
 * permanently unattributable. Not because the number was wrong, but because
 * nothing recorded WHY.
 *
 * With this ledger, `inventory_reconciliation.on_hand_expected` adds
 * sum(delta) back, so a documented adjustment reconciles to zero and only
 * UNDOCUMENTED movement shows as drift. That is the entire point: a non-zero
 * delta should mean "something is wrong", not "someone corrected this once".
 *
 * Invariants enforced at the DB (see the migration), not just here:
 *   - append-only: UPDATE/DELETE are rejected by trigger. Fix a mistake with a
 *     compensating row, never by editing history.
 *   - never gold: curve/standard on-hand is derived from serialized `inventory`
 *     rows, so an adjustment there would be silently ignored by the view.
 *   - delta <> 0, and a note of >= 3 real characters is REQUIRED.
 */
 
import type { SupabaseClient } from "@supabase/supabase-js";
 
/**
 * Ordered most-likely-first. `sold_unrecorded` leads because it is the reason
 * the reconciliation screen's drift most often has — and because its absence
 * caused a real mis-entry on 2026-07-19 (Mukesh: "I did not find paid order in
 * the reasons and thus used sample", which wrongly booked a marketing expense
 * for units whose cost was already in COGS). A dropdown with no honest option
 * produces dishonest data.
 */
export const ADJUSTMENT_REASONS = [
  { code: "sold_unrecorded", label: "Sold — never decremented", hint: "Went out on a paid order but the counter never dropped" },
  { code: "damaged",   label: "Damaged",            hint: "Unusable — crushed, torn, printed wrong" },
  { code: "lost",      label: "Lost / missing",     hint: "Cannot be found at count time" },
  { code: "found",     label: "Found",              hint: "Turned up — stock we already had" },
  { code: "miscount",  label: "Miscount corrected", hint: "The recorded number was simply wrong" },
  { code: "sample",    label: "Sample / giveaway",  hint: "Used outside an order (photos, a gift)" },
  { code: "write_off", label: "Write-off",          hint: "Deliberately scrapped or disposed of" },
  { code: "other",     label: "Other",              hint: "Anything else — explain in the note" },
] as const;
 
export type AdjustmentReasonCode = (typeof ADJUSTMENT_REASONS)[number]["code"];

/**
 * Write one adjustment row and (by default) move the counter by the same delta.
 *
 * The two writes are deliberately ordered ledger-FIRST. If the counter update
 * then fails, we are left with a ledger row and an unmoved counter — which shows
 * up as drift on the reconciliation screen, i.e. LOUD. The reverse order could
