/**
 * Single source of truth for 1Plus1Gold's regulatory-identity copy.
 *
 * Why a helper rather than ad-hoc strings: the PSPM licence is issued
 * by the Ministry of Law (Singapore), NOT by MAS. Earlier copy across
 * Footer / email / receipt / about / personas / voice / AI prompts
 * incorrectly said "MAS PSPM Licence" / "MAS-licensed" / "regulated by
 * MAS". Centralising the correct phrasing means future regulator
 * rewordings touch ONE file.
 *
 * MAS still legitimately appears in code where its actual role applies
 * — AML/CFT supervision of PSPM dealers, STR reporting via SONAR, FATF
 * jurisdiction lists, CTR thresholds. Those references must NOT use
 * this helper; they're correct as-is.
 *
 * Pairs with: apps/ios-customer/Sources/Design/RegulatoryText.swift.
 * The two files duplicate strings deliberately (iOS can't import TS).
 * Any rewording here must be mirrored in the Swift constants and vice
 * versa — see memory/feedback_regulatory_helper_pair.md.
 */
 
import { BUSINESS_CONFIG } from "@/lib/business-config";
 
/** Compact one-liner for Footer mobile row, email + receipt regulatory rows.
 *  Example: "Singapore PSPM Licence PS20260000988 · MinLaw-registered · IPM-exempt dealer" */
export function regulatoryShortLine(): string {
  return `Singapore PSPM Licence ${BUSINESS_CONFIG.pspmLicence} · MinLaw-registered · IPM-exempt dealer`;
}
 
/** Full sentence for legal/about pages and terms-page metadata.
 *  Example: "Registered Dealer under the PSPM Act (Ministry of Law, Singapore) — PSPM Licence PS20260000988 — AML/CFT supervised by MAS." */
export function regulatoryLongLine(): string {
  return `Registered Dealer under the PSPM Act (Ministry of Law, Singapore) — PSPM Licence ${BUSINESS_CONFIG.pspmLicence} — AML/CFT supervised by MAS.`;
}
 
/** Trust strip used on About hero + Footer chip.
 *  Example: "Singapore-licensed · PSPM PS20260000988 · IPM-exempt dealer" */
export function regulatoryTrustStrip(): string {
  return `Singapore-licensed · PSPM ${BUSINESS_CONFIG.pspmLicence} · IPM-exempt dealer`;
}
 
/** Inline clause for voice + AI-persona prompts where we describe
 *  ourselves to an LLM.
 *  Example: "Singapore physical gold dealer (MinLaw-licensed under the PSPM Act, AML supervised by MAS)" */
export function regulatorClause(): string {
  return "Singapore physical gold dealer (MinLaw-licensed under the PSPM Act, AML supervised by MAS)";
}
 
/** Footer column header label for the licence row. */
export const PSPM_LICENCE_LABEL = "PSPM Licence (MinLaw)";
 
/** Two-line paragraph used in Footer's Legal & Compliance column
 *  + About data KV-row blurb beneath the licence number. */
export const REGULATOR_BLURB =
  "Gold dealing activities regulated under the Precious Stones and Precious Metals Act (PSPM). Licensed by the Ministry of Law (Singapore).";
