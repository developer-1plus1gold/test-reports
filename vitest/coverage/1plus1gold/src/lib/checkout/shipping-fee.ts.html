// Sidebar/summary shipping-fee resolution for checkout.
//
// Extracted from checkout/page.tsx so the decision is pure + unit-tested. The
// subtle invariant this guards: a NO-SHIPPING fulfilment type (gold_yield,
// vault, self_pickup) must resolve to fee 0 — NEVER null. The checkout's
// chargeAmount short-circuits to 0 when shippingFee === null, so a null here
// makes displayedTotalSgd 0; the server then sees displayed(0) ≪ grand(gold)
// and rejects the order with "Cart total mismatch — please refresh and try
// again" (orders/create + hitpay/create). gold_yield + vault previously fell
// through to the trailing null and broke vault COD orders (2026-06-26).
 
export type ShipperType = "lalamove" | "ninjavan" | "spx" | "self_pickup" | "gold_yield" | "vault";
 
/**
 * The ONE rule for what a Lalamove delivery costs the customer once a live
 * quote is in hand: they pay the LOWER of our fixed price and the live quote,
 * and the live quote alone when no fixed price is configured.
 *
 * ▶ WHY THIS IS EXTRACTED (2026-09-15). This rule existed SIX times — here plus
 * FIVE hand-written copies in `checkout/page.tsx`: the step-summary line, the
 * carrier radio row, the order-total breakdown, the processing-fee base, and
 * `buildSelectedShipping`. ⚠️ The GrabExpress design doc (§15) recorded it as
 * existing "twice"; that count came from reading the two named call sites rather
 * than grepping, and it was wrong. Five of the six drive what the customer READS
 * and one drives the fee actually SENT to `orders/create`, so all six must agree
 * and nothing asserted that they did.
 *
 * The failure mode if a future edit touches one copy is SILENT: the customer is
 * shown one number and charged another, and a divergence inside the checkout's
 * ±5c price-match tolerance trips no guard at all. Folding them was named the
 * precondition for adding a second carrier precisely because a per-carrier copy
 * of a money rule is how mis-charging gets introduced — and with six copies,
 * adding Grab meant editing six places correctly or mis-charging silently.
 *
 * Total by construction (never null) — callers that have no quote yet decide
 * what to show; this answers only "given this quote, what is the fee?".
 */
export function lalamoveCustomerFee(netFee: number, lalamoveFixedFeeSgd: number | null): number {
  const fixedFee = lalamoveFixedFeeSgd ?? 0;
  return fixedFee > 0 ? Math.min(fixedFee, netFee) : netFee;
}
 
/**
 * Resolve the shipping fee shown in the order summary for the selected method.
 *   • null  → not yet resolvable (renders a "—" placeholder; charge is held)
 *   • 0     → genuinely free (no-shipping fulfilment, or a free promo elsewhere)
 *   • >0    → the chargeable delivery fee
 */
export function resolveSidebarShippingFee(
  step: number,
  shipperType: ShipperType,
  quote: { netFee: number } | null,
  lalamoveFixedFeeSgd: number | null,
  ninjavanFee: number | null,
  spxFee: number | null = null,
): number | null {
  // Fee isn't shown until the customer reaches the shipping step.
  if (step < 3) return null;
  // No-shipping fulfilment types are always free — pin to 0, never null.
  if (shipperType === "gold_yield" || shipperType === "vault" || shipperType === "self_pickup") return 0;
  if (shipperType === "ninjavan") return ninjavanFee ?? null;
  // SPX is a fixed-fee carrier like NinjaVan (no live quote).
  Iif (shipperType === "spx") return spxFee ?? null;
  Eif (shipperType === "lalamove") {
    // Both available → the shared rule decides (customer pays the lower).
    if (quote) return lalamoveCustomerFee(quote.netFee, lalamoveFixedFeeSgd);
    // Fixed price set → show it immediately while the live quote loads.
    const fixedFee = lalamoveFixedFeeSgd ?? 0;
    if (fixedFee > 0) return fixedFee;
    // No fixed fee and no live quote yet.
    return null;
  }
  return null;
}
