/**
 * The order-confirmation delivery block (checkout/page.tsx OrderConfirmed), by the checkout's
 * shipping type. A vault order and a self-pickup go to no address, so they show no To / Address
 * rows and no driver note (Mukesh 2026-11-28: a vault MAG order read "To: Self Pickup",
 * "Address: , Singapore" and "Driver tracking will be available…").
 */
/** What the confirmation's delivery block shows: a vault order and a self-pickup go to no address. */
export type ConfirmationDeliveryKind = "vault" | "self_pickup" | "courier";
export function confirmationDeliveryKind(type: string | null | undefined): ConfirmationDeliveryKind {
  return type === "vault" ? "vault" : type === "self_pickup" ? "self_pickup" : "courier";
}
export const CONFIRMATION_DELIVERY_NOTE: Record<ConfirmationDeliveryKind, string> = {
  // The same sentence as the checkout's Vault Storage option card (checkout/page.tsx).
  vault: "Once payment is confirmed, your gold is held in allocated, serial-tracked custody. No shipping — withdraw it to you anytime from your Vault.",
  self_pickup: "You collect this order in person — nothing is sent to an address.",
  courier: "Driver tracking will be available in your account once payment is confirmed and the driver is assigned.",
};
