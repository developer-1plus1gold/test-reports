 * Triple-tier resilience:
 *   Tier 1  XAU: goldapi.io      (paid unlimited plan, primary)
 *   Tier 2  XAU: metals-api.com  (graceful no-op if METALS_API_KEY unset)
 *   Tier 3  any: Supabase singleton `spot_price_cache` table (last-known-good)
 *
 *   Tier 1  FX:  fastforex.io    (premium interbank, primary if FASTFOREX_KEY)
 *   Tier 2  FX:  open.er-api.com (free, no key, bank-aggregated)

 * surface as a 503 on `/api/payments/.../create-intent`. Catalogue browsing

export type ProviderName = "goldapi" | "metals_dev" | "fastforex" | "metals_dev_fx" | "open_er_api";
export type ProviderFailureReason = "timeout" | "rate_limited" | "http_error" | "network" | "bad_response";
export type ProviderFailure = { reason: ProviderFailureReason; detail: string; at: number };
 
const providerFailures: Partial<Record<ProviderName, ProviderFailure>> = {};
 
/** How long a recorded failure stays reportable. Beyond this it says nothing about the current tick. */
const PROVIDER_FAILURE_WINDOW_MS = 15 * 60_000;
 
export function classifyProviderError(err: unknown): ProviderFailure["reason"] {
  if (err instanceof Error) {
    if (err.name === "TimeoutError" || err.name === "AbortError" || /timeout/i.test(err.message)) return "timeout";
    if (/^HTTP 429\b/.test(err.message)) return "rate_limited";
    if (/^HTTP \d{3}/.test(err.message)) return "http_error";
    if (err.name === "TypeError") return "network"; // undici "fetch failed"
  }
  return "bad_response";
}
 
function recordFailure(provider: ProviderName, err: unknown): void {
  providerFailures[provider] = {
    reason: classifyProviderError(err),
    detail: (err instanceof Error ? err.message : String(err)).slice(0, 200),
    at: Date.now(),
  };
}
 
function recordSuccess(provider: ProviderName): void {
  delete providerFailures[provider];
}
 
/** Recent upstream failures on this instance, newest view. Empty when every provider is healthy. */
export function getProviderFailures(now: number = Date.now()): Partial<Record<ProviderName, ProviderFailure>> {
  const out: Partial<Record<ProviderName, ProviderFailure>> = {};
  for (const [k, v] of Object.entries(providerFailures) as [ProviderName, ProviderFailure][]) {
    if (now - v.at <= PROVIDER_FAILURE_WINDOW_MS) out[k] = v;
  }
  return out;
}
 
/** One-line human summary, e.g. "goldapi timeout, metals_dev timeout". */
export function summariseProviderFailures(f: Partial<Record<ProviderName, ProviderFailure>>): string {
  return (Object.entries(f) as [ProviderName, ProviderFailure][])
    .map(([k, v]) => `${k} ${v.reason.replace("_", " ")}`)
    .join(", ");
}
