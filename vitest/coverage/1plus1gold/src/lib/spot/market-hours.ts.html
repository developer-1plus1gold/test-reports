/**
 * Is the gold market open? (Mukesh 2026-11-22: customers asked why the spot price
 * had not moved for two days — the market was closed.)
 *
 * Pure and dependency-free, so the server, the web ticker and tests share it. The
 * pricing snapshot carries its answer to every surface (web + app); the weekend
 * sell buffer and buyback adjustment (lib/pricing/market-closed.ts) switch on it.
 *
 * ▶ THE SCHEDULE IS THE GOLD MARKET'S OWN, IN NEW YORK TIME. Spot gold trades
 * round the clock from Sunday 18:00 to Friday 17:00 New York time (Globex). Working
 * in America/New_York means daylight saving needs no code: in Singapore the close is
 * Sat 06:00 (Nov–Mar) or Sat 05:00 (Mar–Nov), the reopen Mon 07:00 or Mon 06:00.
 * The 17:00–18:00 daily maintenance hour is NOT treated as closed: spot barely
 * pauses, and a note that flashed up every morning would teach customers to ignore it.
 *
 * ▶ HOLIDAYS: New Year's Day, Good Friday and Christmas Day close the market from
 * 17:00 the evening before to 18:00 on the day. Good Friday is computed from Easter
 * each year, so nothing needs updating. Overlapping closures merge (Good Friday runs
 * straight into the weekend). A holiday on a weekend changes nothing.
 */
 
const NY = "America/New_York";
const CLOSE_HOUR = 17; // Friday (and the evening before a holiday), New York
const OPEN_HOUR = 18;  // Sunday (and the evening of a holiday), New York
 
export type MarketStatus = {
  open: boolean;
  /** Why it is closed; null while open. */
  reason: "weekend" | "holiday" | null;
  /** When this closure began (ISO); null while open. */
  closedSince: string | null;
  /** When trading resumes (ISO); null while open. */
  reopensAt: string | null;
};
 
const nyParts = (d: Date) => {
  const f = new Intl.DateTimeFormat("en-US", {
    timeZone: NY, year: "numeric", month: "numeric", day: "numeric", hour: "numeric", minute: "numeric", hourCycle: "h23",
  });
  const p = Object.fromEntries(f.formatToParts(d).map((x) => [x.type, x.value]));
  return { y: +p.year, m: +p.month, d: +p.day, h: +p.hour, min: +p.minute };
};
 
/** The UTC instant of a New York wall-clock time (handles EST/EDT). */
export function nyWallToUtc(y: number, m: number, d: number, h: number, min = 0): Date {
  const want = Date.UTC(y, m - 1, d, h, min);
  let t = want + 5 * 3600_000; // EST guess
  for (let i = 0; i < 3; i++) {
    const p = nyParts(new Date(t));
    const got = Date.UTC(p.y, p.m - 1, p.d, p.h, p.min);
    if (got === want) break;
    t += want - got;
  }
  return new Date(t);
}
 
/** Easter Sunday (Gregorian), month 1-based. */
export function easterSunday(year: number): { m: number; d: number } {
  const a = year % 19, b = Math.floor(year / 100), c = year % 100;
  const d = Math.floor(b / 4), e = b % 4, f = Math.floor((b + 8) / 25), g = Math.floor((b - f + 1) / 3);
  const h = (19 * a + b - d - g + 15) % 30, i = Math.floor(c / 4), k = c % 4;
  const l = (32 + 2 * e + 2 * i - h - k) % 7, m = Math.floor((a + 11 * h + 22 * l) / 451);
  const month = Math.floor((h + l - 7 * m + 114) / 31);
  return { m: month, d: ((h + l - 7 * m + 114) % 31) + 1 };
}
 
/** Full-closure holidays for `year`, as New York calendar dates. */
export function marketHolidays(year: number): Array<{ y: number; m: number; d: number }> {
  const e = easterSunday(year);
  const gf = new Date(Date.UTC(year, e.m - 1, e.d - 2));
  return [
    { y: year, m: 1, d: 1 },
    { y: gf.getUTCFullYear(), m: gf.getUTCMonth() + 1, d: gf.getUTCDate() },
    { y: year, m: 12, d: 25 },
  ];
}
 
type Interval = { start: number; end: number; holiday: boolean };
 
function closuresAround(now: Date): Interval[] {
  const out: Interval[] = [];
  const base = nyParts(now);
  // Weekly closes: the Friday on or before each of the surrounding weeks.
  for (let w = -2; w <= 2; w++) {
    const probe = new Date(Date.UTC(base.y, base.m - 1, base.d + w * 7));
    const dow = probe.getUTCDay();
    const fri = new Date(Date.UTC(probe.getUTCFullYear(), probe.getUTCMonth(), probe.getUTCDate() - ((dow + 2) % 7)));
    const sun = new Date(Date.UTC(fri.getUTCFullYear(), fri.getUTCMonth(), fri.getUTCDate() + 2));
    out.push({
      start: nyWallToUtc(fri.getUTCFullYear(), fri.getUTCMonth() + 1, fri.getUTCDate(), CLOSE_HOUR).getTime(),
      end: nyWallToUtc(sun.getUTCFullYear(), sun.getUTCMonth() + 1, sun.getUTCDate(), OPEN_HOUR).getTime(),
      holiday: false,
    });
  }
  for (const y of [base.y - 1, base.y, base.y + 1]) {
    for (const h of marketHolidays(y)) {
      const eve = new Date(Date.UTC(h.y, h.m - 1, h.d - 1));
      out.push({
        start: nyWallToUtc(eve.getUTCFullYear(), eve.getUTCMonth() + 1, eve.getUTCDate(), CLOSE_HOUR).getTime(),
        end: nyWallToUtc(h.y, h.m, h.d, OPEN_HOUR).getTime(),
        holiday: true,
      });
    }
  }
  // Merge overlapping / touching closures.
  out.sort((a, b) => a.start - b.start);
  const merged: Interval[] = [];
  for (const c of out) {
    const last = merged.at(-1);
    if (last && c.start <= last.end) {
      last.end = Math.max(last.end, c.end);
      last.holiday ||= c.holiday;
    } else merged.push({ ...c });
  }
  return merged;
}
 
export function marketStatus(now: Date = new Date()): MarketStatus {
  const t = now.getTime();
  const hit = closuresAround(now).find((c) => t >= c.start && t < c.end);
  if (!hit) return { open: true, reason: null, closedSince: null, reopensAt: null };
  return {
    open: false,
    reason: hit.holiday ? "holiday" : "weekend",
    closedSince: new Date(hit.start).toISOString(),
    reopensAt: new Date(hit.end).toISOString(),
  };
}

/** "0.5%" — trailing zeros dropped. */
const fmtPct = (n: number) => `${Number(n.toFixed(3))}%`;
 
/**
 * The customer line under the ticker while the market is closed (Mukesh
 * 2026-11-22: lead with "we are open" — "Gold markets are closed" read as the
 * shop being closed):
 *   "We are open as usual. The global gold spot price doesn't update on weekends,
 *    so it stays at Friday's close. Live prices return Mon 7:00 am."
 * With a weekend sell buffer in force (lib/pricing/market-closed.ts) it adds
 * "… and our prices include a small 0.5% weekend allowance." — the ticker itself
 * always shows the TRUE spot. Holidays say "over the holiday" (Good Friday runs
 * into the weekend, so naming the day would read wrong on the Saturday).
 * Null while open — the ticker shows nothing extra then.
 */
export function marketClosedNote(s: MarketStatus, sellBufferPct = 0): string | null {
  if (s.open || !s.closedSince || !s.reopensAt) return null;
  const closeDay = new Intl.DateTimeFormat("en-SG", { timeZone: NY, weekday: "long" }).format(new Date(s.closedSince));
  const holiday = s.reason === "holiday";
  const when = holiday ? "over the holiday" : "on weekends";
  const allowance = sellBufferPct > 0
    ? ` and our prices include a small ${fmtPct(sellBufferPct)} ${holiday ? "holiday" : "weekend"} allowance`
    : "";
  return `We are open as usual. The global gold spot price doesn't update ${when}, so it stays at ${closeDay}'s close${allowance}. Live prices return ${reopenLabel(s)}.`;
}
