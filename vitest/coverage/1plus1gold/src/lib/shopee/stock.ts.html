/**
 * Sellable-stock derivation — the SINGLE source for both the server exporter
 * (build-listings.ts → the downloaded sheet) AND the admin preview
 * (shopee-export/page.tsx → the Stock column + warnings). They MUST agree.
 *
 * Mirrors the shop's own display (shop-pricing `deriveStatus` /
 * `stockAvailabilityChip`): sellable count = `stock_qty` DIRECTLY, which is the
 * derived available-to-sell count (and for CombiBar<->split pool members, the
 * pool-derived count). The legacy `stock_reserved` counter was never subtracted
 * — it held nonsense (1g Split-Valcambi once read 22 on-hand / 51 reserved) and
 * was dropped entirely in the 2026-07-17 flag-model cleanup.
 * Presale/sold_out → 0 (not fulfillable on Shopee). null stock_qty = untracked
 * → 0 + a warning (admin sets a Shopee stock manually).
 */
export function sellableStock(p: {
  stock_qty: number | null;
  status: string;
}): { stock: number; untracked: boolean } {
  if (p.stock_qty == null) return { stock: 0, untracked: true };
  const unavailable = p.status === "sold_out" || p.status === "presale" || p.stock_qty <= 0;
  return { stock: unavailable ? 0 : Math.max(0, p.stock_qty), untracked: false };
}
