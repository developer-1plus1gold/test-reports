/** All shop-visible products (INCLUDING sold_out/presale — the exporter lists
 *  them out-of-stock rather than dropping them), media ordered. */
export async function fetchExportProducts(): Promise<ShopeeSourceProduct[]> {
  const { data, error } = await admin()
    .from("products")
    .select(PRODUCT_SELECT)
    .eq("shop_visible", true)
    .neq("product_type", "accessory") // accessories aren't gold listings; excluded from the gold channel
    .order("sort_order", { ascending: true });
  if (error) throw new Error(`shopee product fetch failed: ${error.message}`);
  return ((data ?? []) as unknown as ShopeeSourceProduct[]).map((p) => ({
    ...p,
    grams: Number(p.grams),
    product_media: Array.isArray(p.product_media)
      ? [...p.product_media]
          .filter((m) => (m.kind ?? "image") === "image")
          .sort((a, b) => Number(a.sort_order ?? 0) - Number(b.sort_order ?? 0))
      : [],
  }));
}

function absoluteUrl(url: string): string {
  if (/^https?:\/\//i.test(url)) return url;
  const base = (process.env.NEXT_PUBLIC_SITE_URL || "https://1plus1gold.com").replace(/\/$/, "");
  return `${base}${url.startsWith("/") ? "" : "/"}${url}`;
}
