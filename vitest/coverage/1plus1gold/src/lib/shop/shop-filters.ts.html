/**
 * Shop filter rules that span product lines — pure, so the page and tests
 * share one definition.
 *
 * The Shop's Type filter (operator 2026-09-23) chooses which lines show:
 * Premium Curve ("curve" — 1g Splits to 1oz bars), Standard bars and coins ("standard"), 1+1Gold MAG
 * ("mag"). An empty selection means "all", matching FilterChipGroup.
 *
 * MAG is bought by the gram, so the Weight filter does not apply to it; the
 * Refiner filter and "Show sold-out" do.
 */
import type { MagListingPublic } from "@/lib/mag/listings";
 
export type ShopLine = "curve" | "standard" | "mag";
 
export const SHOP_TYPE_OPTIONS: { value: ShopLine; label: string }[] = [
  { value: "curve",    label: "Premium Curve" },
  { value: "standard", label: "Standard" },
  { value: "mag",      label: "1+1Gold MAG" },
];
 
export function lineVisible(typeFilter: ReadonlySet<string>, line: ShopLine): boolean {
  return typeFilter.size === 0 || typeFilter.has(line);
}
 
export function filterMagListings(
  listings: readonly MagListingPublic[],
  opts: { refineryFilter: ReadonlySet<string>; showSoldOut: boolean },
): MagListingPublic[] {
  return listings.filter((l) => {
    if (!opts.showSoldOut && l.gramsRemaining <= 0) return false;
    if (opts.refineryFilter.size > 0 && !(l.refinery && opts.refineryFilter.has(l.refinery))) return false;
    return true;
  });
}
 
/** Refiner options across every visible line, A–Z, de-duplicated. */
export function refineryOptionsFor(
  products: readonly { product_type: string; refinery: string | null }[],
  listings: readonly MagListingPublic[],
  typeFilter: ReadonlySet<string>,
): { value: string; label: string }[] {
  const names = new Set<string>();
  for (const p of products) {
    if ((p.product_type === "curve" || p.product_type === "standard")
        && lineVisible(typeFilter, p.product_type) && p.refinery?.trim()) names.add(p.refinery);
  }

}
