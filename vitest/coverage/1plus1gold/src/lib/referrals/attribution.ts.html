/**
 * Whether the customer can still be linked to a referrer ("Were you referred?"
 * in My Rewards): no link yet, no referral recorded, no paid order. Mirrors the
 * attach_referral() RPC, which stays the authority.
 */
export async function canAttachReferral(db: SupabaseClient, userId: string): Promise<boolean> {
  const [attr, used, paid] = await Promise.all([
    db.from("referral_attributions").select("referred_id", { count: "exact", head: true }).eq("referred_id", userId),
    db.from("referrals").select("id", { count: "exact", head: true }).eq("referred_id", userId),
    db.from("orders").select("id", { count: "exact", head: true }).eq("customer_id", userId).in("status", [...PAID_STATUSES]),
  ]);
  return (attr.count ?? 0) === 0 && (used.count ?? 0) === 0 && (paid.count ?? 0) === 0;
}

/**
 * The eligible gold value of a first order for an ATTRIBUTED referral: gold
 * pieces no heavier than the code's per-piece limit (MAG and accessories never
 * count), valued at the price paid, capped at the programme's 5g. Mirrors the
 * referee-discount rule in lib/referral-discount.ts, but on the order's actual
 * line prices: the order has already been charged, so this is what was bought.
 */
export function eligibleSubtotalFromLines(lines: RewardLine[], eligibleMaxGrams: number): number {
  const capGrams = Math.min(eligibleMaxGrams, MAX_REFERRAL_CAP_GRAMS);
  let gramsLeft = capGrams;
  let subtotal = 0;
  const eligible = lines
    .filter((l) => (l.productType === "curve" || l.productType === "standard")
      && l.gramsEach > 0 && l.gramsEach <= eligibleMaxGrams && l.quantity > 0)
    // Lightest pieces first — the same order lib/referral-discount fills the 5g in.
    .sort((a, b) => a.gramsEach - b.gramsEach);
  for (const l of eligible) {
    for (let i = 0; i < l.quantity && gramsLeft > 0; i++) {
      const take = Math.min(l.gramsEach, gramsLeft);
      subtotal += (l.priceSgd / l.gramsEach) * take;
      gramsLeft -= take;
    }
    if (gramsLeft <= 0) break;
  }
  return Math.round(subtotal * 100) / 100;
}
