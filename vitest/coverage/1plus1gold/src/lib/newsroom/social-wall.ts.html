/**
 * The social wall — our published Instagram / LinkedIn posts, read from the
 * drumbeat pipeline.
 *
 * ▶ THE ROWS ARE NOT COPIED. marketing_content_pieces already carries
 * published_at and platform_permalinks and is already maintained by the
 * publisher. Duplicating it into newsroom_posts would buy nothing and create a
 * sync problem. So this reads that table directly — which makes the projection
 * the load-bearing part of the file.
 *
 * ⚠️ EXPLICIT FIELD LIST, NEVER A SPREAD. That table also holds
 * `core_asset_brief` and `caption_direction` (the instructions we give the AI),
 * `register` / `pillar` / `chakra` (our content framework), `notes` and
 * `estimated_effort_hrs`. Spreading the row onto a public surface would publish
 * our content strategy. That is exactly the F8 defect of 2026-08-21, where
 * /api/payments/methods spread its row and published what we pay our payment
 * provider and how much of it we absorb. Guarded by
 * src/__tests__/newsroom-guards.test.ts.
 *
 * The same applies one table over: marketing_drafts holds `original_content` —
 * the AI's draft BEFORE the operator edited it. Publishing that would expose
 * our editing process on every post. Only the approved/edited caption body for
 * the matching channel is read.
 *
 * ▶ WHAT REACHES THE PAGE: an image, a link, and (on hover) one line of the
 * caption. No date, no format label, no channel name as text — a wall with
 * metadata under every tile reads as a table of contents, and our posting
 * cadence is competitive information besides. See docs/newsroom-design.md §4.
 */
 
import { createClient } from "@supabase/supabase-js";
import { safeExternalUrl } from "./types";
 
/** Channels we surface, in tap-priority order. Instagram first: it is the
 *  visual channel, so its post is the one a tile's imagery came from. */
const WALL_CHANNELS = ["instagram", "linkedin"] as const;
export type WallChannel = (typeof WALL_CHANNELS)[number];
 
export interface WallTile {
  id: string;
  /** Signed URL for the tile's media. A tile without one is dropped, never blank. */
  mediaUrl: string;
  /**
   * What `mediaUrl` points at, so the renderer knows which element to use.
   *
   * ▶ DELIBERATELY DISTINCT FROM `isVideo` BELOW, which looks similar and
   * answers a different question. `isVideo` is about the POST (is the thing
   * waiting on Instagram a reel) and drives the play glyph. `mediaKind` is about
   * the ASSET WE HAPPEN TO HOLD and drives img-vs-video. They usually agree and
   * do not have to: a reel piece with an uploaded still is a video post shown
   * through an image.
   */
  mediaKind: "image" | "video";
  /** Where tapping the tile goes — the primary channel's live post. */
  href: string;
  primaryChannel: WallChannel;
  /** The other channel's post, when the piece went to both. Hover-only. */
  secondary: { channel: WallChannel; href: string } | null;
  /** First line of the published caption. Hover-only; never always-on. */
  captionLine: string | null;
  isVideo: boolean;
}
 
function admin() {
  const url = process.env.NEXT_PUBLIC_SUPABASE_URL;
  const key = process.env.SUPABASE_SERVICE_KEY;
  return url && key ? createClient(url, key) : null;
}
 
/** Video formats, per the piece's `format` enum. Drives the play glyph only. */
const VIDEO_FORMATS = new Set(["short_reel", "long_reel"]);
 
function firstLine(body: unknown): string | null {
  if (typeof body !== "string") return null;
  const line = body.split("\n").map((s) => s.trim()).find((s) => s.length > 0);
  if (!line) return null;
  return line.length > 120 ? `${line.slice(0, 117)}…` : line;
}
 
/**
 * Why the wall reports a STATUS and not just tiles.
 *
 * Every failure in here used to `return []`, and the section renders its follow
 * CTA alone on an empty array — so a dead database, a storage-signing outage and
 * "we have not posted anything yet" were PIXEL-IDENTICAL, with a bare
 * `catch {}` swallowing the difference. The standing question in this codebase
 * is *what would this look like if it were silently doing nothing?*, and the
 * answer here was *exactly the same*, on the page an award reel drives traffic to.
 *
 * The fix is NOT to make the page loud — a visitor must never see a stack trace,
 * so the page stays fail-soft and still renders the CTA. The fix is to make the
 * failure DISTINGUISHABLE, so a second caller (the newsroom-wall-health cron)
 * can read the same projection and alert. One projection, two callers: the page
 * ignores `status`, the cron acts on it.
 */
export type WallStatus =
  /** Tiles were produced. */
  | "ok"
  /** Nothing is published yet. Legitimately empty — never alerts. */
  | "empty"
  /** Supabase env missing. A config/deploy fault. */
  | "no_client"
  /** The pieces read failed. */
  | "query_error"
  /** Pieces ARE published, but none has both a live permalink and media. */
  | "no_candidates"
  /** Candidates existed but nothing signed — a storage fault. */
  | "sign_failed"
  /** Unexpected throw. */
  | "threw";
 
export interface WallResult {
  tiles: WallTile[];
  status: WallStatus;
  /** Operator-facing detail. NEVER rendered to a visitor. */
  detail?: string;
}
 
/**
 * True when an empty wall means something is WRONG rather than quiet.
 *
 * `empty` is deliberately NOT degraded: a wall with nothing published yet is
 * correct behaviour, and alerting on it would train us to ignore the alert.
 */
export function isWallDegraded(status: WallStatus): boolean {
  return status !== "ok" && status !== "empty";
}
 
function wallFail(status: WallStatus, detail: string): WallResult {
  // Logged at error level so a degraded wall leaves a trace even if nobody is
  // running the cron yet. The cron is what turns this into an actual page.
  console.error(`[newsroom-wall] degraded: ${status} — ${detail}`);
  return { tiles: [], status, detail };
}
 
/**
 * Build the wall. Never throws; returns a status so an empty wall can be told
 * apart from a broken one. An absent tile beats a dead one: we have already had
 * permalinks go dead when posts were deleted (df209eaa, 2026-09-11).
 *
 * Honours `wall_visible` — the OPERATOR's hide switch. The link-health cron
 * never writes it (docs/newsroom-design.md §4b: flag for review, never
 * auto-hide, or one transient 429 silently empties the wall).
 */
export async function getWallTiles(limit = 9): Promise<WallResult> {
  const sb = admin();
  if (!sb) return wallFail("no_client", "SUPABASE env missing");
 
  try {
    // EXPLICIT columns — see the header. Nothing here is our strategy.
    const { data: pieces, error } = await sb
      .from("marketing_content_pieces")
      .select("id, format, published_at, platform_permalinks")
      .eq("status", "published")
      .eq("wall_visible", true)
      .not("published_at", "is", null)
      .order("published_at", { ascending: false })
      .limit(limit * 3); // over-fetch: some pieces have no usable link or image
 
    if (error) return wallFail("query_error", error.message);
    // Split from the error case on purpose: "no rows" is a legitimate quiet
    // wall, "the read failed" is an outage. Collapsing them is the whole defect.
    if (!pieces?.length) return { tiles: [], status: "empty" };
 
    const ids = pieces.map((p) => p.id as string);
 
    const [{ data: assets }, { data: drafts }] = await Promise.all([
      sb.from("marketing_assets")
        .select("content_piece_id, storage_path, mime_type, asset_kind")
        .in("content_piece_id", ids),
      sb.from("marketing_drafts")
        .select("content_piece_id, content, target, status")
        .eq("draft_kind", "caption")
        .in("status", ["approved", "edited"])
        .in("content_piece_id", ids),
    ]);
 
    /**
     * Pick one asset per piece, PREFERRING an image.
     *
     * ▶ AN EARLIER CUT TOOK IMAGES ONLY, and measuring it is what killed that.
     * Of the 18 published pieces on production, 7 carry an image and **11 are
     * video-only** — the reels, which are the best work on the channel this
     * wall exists to point at. An images-only wall silently omits them, and an
     * omission looks exactly like a design choice.
     *
     * A video's storage object is genuinely not a thumbnail, so the fix is not
     * to feed it to an <img>: the renderer draws it in a muted <video
     * preload="metadata">, whose first frame IS the poster. Same trade already
     * accepted for the CoA authenticity clip (2026-08-10), and `media-src` on
     * both CSP headers already allows the Supabase storage host — checked, not
     * assumed.
     */
    const assetFor = new Map<string, { path: string; kind: "image" | "video" }>();
    for (const a of assets ?? []) {
      const pid = a.content_piece_id as string;
      const path = a.storage_path as string | null;
      const mime = typeof a.mime_type === "string" ? a.mime_type : "";
      if (!path) continue;
      const kind = mime.startsWith("image/") ? "image" : mime.startsWith("video/") ? "video" : null;
      if (!kind) continue;
      const held = assetFor.get(pid);
      // First image wins outright; a video is only kept while no image has been
      // seen, and is replaced the moment one is.
      if (!held || (held.kind === "video" && kind === "image")) {
        assetFor.set(pid, { path, kind });
      }
    }
 
    const captionFor = new Map<string, string>();
    for (const d of drafts ?? []) {
      const pid = d.content_piece_id as string;
      if (captionFor.has(pid)) continue;
      const line = firstLine((d.content as Record<string, unknown> | null)?.body);
      if (line) captionFor.set(pid, line);
    }
 
    /**
     * ▶ SIGN EVERY PATH IN ONE CALL, NOT ONE PER TILE.
     *
     * This used to `await createSignedUrl()` inside the loop. Vercel runs in
     * iad1 (Washington DC) and Supabase is ap-southeast-1 (Singapore), so each
     * of those was a round trip halfway around the world — taken NINE TIMES IN
     * SERIES before the page could render.
     *
     * Measured on production 2026-09-15: /newsroom took 6–11s against 190ms for
     * /shop. And the signed token differed on every request, which proves the
     * page was re-rendering for every visitor rather than serving its 300s
     * cache — so every reader paid all nine hops, every time.
     *
     * `createSignedUrls()` signs the whole batch in ONE request. Nine hops → one.
     *
     * ▶ Mapped BY PATH, not by index. The API echoes each requested path
     * verbatim and in order (verified against live storage rather than assumed),
     * but pairing by position would silently hand a tile another tile's URL if
     * that ever stopped holding — and a wall of wrong-but-plausible thumbnails
     * is the kind of wrong nobody reports.
     */
    type Candidate = {
      pid: string;
      path: string;
      kind: "image" | "video";
      href: string;
      primaryChannel: WallChannel;
      secondary: { channel: WallChannel; href: string } | null;
      isVideo: boolean;
    };
 
    const candidates: Candidate[] = [];
    for (const p of pieces) {
      if (candidates.length >= limit) break;
      const pid = p.id as string;
 
      const links = (p.platform_permalinks ?? {}) as Record<string, unknown>;
      const resolved = WALL_CHANNELS
        .map((ch) => ({ channel: ch, href: safeExternalUrl(links[ch] as string | undefined) }))
        .filter((x): x is { channel: WallChannel; href: string } => x.href !== null);
 
      // No live post to send anyone to → the tile has no purpose. Instagram
      // permalinks are fetched best-effort by the publisher and CANNOT be
      // rebuilt from the stored media id, so this genuinely happens.
      if (resolved.length === 0) continue;
 
      const asset = assetFor.get(pid);
      if (!asset) continue; // the media IS the content; no media, no tile
 
      candidates.push({
        pid,
        path: asset.path,
        kind: asset.kind,
        href: resolved[0].href,
        primaryChannel: resolved[0].channel,
        secondary: resolved[1] ?? null,
        isVideo: VIDEO_FORMATS.has(String(p.format)),
      });
    }
 
    if (candidates.length === 0) {
      // Published pieces exist but none is renderable. Benign early on; if it
      // persists it means every permalink died or every asset vanished, which
      // is precisely the df209eaa case. Degraded, so the cron looks at it.
      return wallFail("no_candidates", `${pieces.length} published, 0 renderable`);
    }
 
    // TTL comfortably longer than the page's 5-minute revalidate, so a cached
    // page can never serve an expired link.
    const { data: signed } = await sb.storage
      .from("marketing-content")
      .createSignedUrls(candidates.map((c) => c.path), 60 * 60 * 24 * 7);
 
    const urlFor = new Map<string, string>();
    for (const row of signed ?? []) {
      if (row.error || !row.path || !row.signedUrl) continue;
      urlFor.set(row.path, row.signedUrl);
    }
 
    // A path that failed to sign is dropped, exactly as the per-tile `continue`
    // did — one bad object must never take the whole wall down with it.
    const tiles: WallTile[] = candidates.flatMap((c) => {
      const mediaUrl = urlFor.get(c.path);
      if (!mediaUrl) return [];
      return [{
        id: c.pid,
        mediaUrl,
        mediaKind: c.kind,
        href: c.href,
        primaryChannel: c.primaryChannel,
        secondary: c.secondary,
        captionLine: captionFor.get(c.pid) ?? null,
        isVideo: c.isVideo,
      }];
    });
 
    // Candidates existed but NOTHING signed — storage is unhappy. Dropping one
    // object is normal; dropping all of them is not.
    if (tiles.length === 0) {
      return wallFail("sign_failed", `${candidates.length} candidates, 0 signed`);
    }
    return { tiles, status: "ok" };
  } catch (err) {
    return wallFail("threw", err instanceof Error ? err.message : String(err));
  }
}
