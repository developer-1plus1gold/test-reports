/**
 * How a probe's HTTP status becomes a link verdict.
 *
 * Pure on purpose: this is the rule most likely to be "tidied" into
 * `res.ok ? "ok" : "dead"` by someone who has not met Instagram's bot handling,
 * and that one-liner would be a REGRESSION, not a simplification. It lives
 * apart from the cron so it can be tested exhaustively with no network.
 *
 * ▶ THE ASYMMETRY IS THE WHOLE POINT. `dead` drives an alert that asks a human
 * to hide a post; `unknown` drives nothing. So the bar for `dead` is an
 * unambiguous, cacheable "this resource is gone" — 404 or 410, and nothing
 * else. Everything ambiguous degrades to `unknown`:
 *
 *   • 401 / 403  — Instagram and LinkedIn routinely serve these to datacentre
 *                  IPs for posts that are perfectly alive. This is the single
 *                  most likely response in production, and calling it `dead`
 *                  would mark the entire wall dead on the first run.
 *   • 429        — rate limited. Says nothing about the post.
 *   • 3xx        — a login bounce. We probe with redirect:"manual" precisely so
 *                  this stays visible as a 3xx instead of resolving to a 200
 *                  login page that would read as a false `ok`.
 *   • 5xx        — the platform is down, not the post.
 *
 * The cost of the asymmetry is a real deletion sitting unflagged until a probe
 * gets through — acceptable, because the alternative (auto-hiding a live wall
 * on a 429) is the failure docs/newsroom-design.md §4b exists to prevent.
 */
export type LinkStatus = "ok" | "dead" | "unknown";
 
/** HTTP statuses that mean the resource is genuinely gone. Nothing else does. */
const GONE = new Set([404, 410]);
 
export function classifyLinkStatus(httpStatus: number): LinkStatus {
  if (GONE.has(httpStatus)) return "dead";
  // 2xx only. A 3xx is a redirect we deliberately did not follow (see header).
  if (httpStatus >= 200 && httpStatus < 300) return "ok";
  return "unknown";
}
