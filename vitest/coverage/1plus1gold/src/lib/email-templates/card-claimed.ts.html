/**
 * Card claimed — SECURITY notice email.
 *
 * Sent to the PREVIOUS holder when a card they held (that was NOT marked as a
 * gift — a plain hand-over) is claimed into someone else's account. Bearer
 * asset: whoever holds it can claim it, so this is an anti-theft signal, not a
 * warm thank-you (that's `giftAcceptedEmail`, used for genuine gifts).
 *
 * Payload parity with the (pending Meta approval) WhatsApp security template:
 * {1 = serial, 2 = claimant sanitised display}. Always sent — a security alert
 * is not gated by the customer's marketing/gift notification prefs.
 */
import { renderEmail, EMAIL_COLORS, EMAIL_FONTS } from "./layout";
 
export type CardClaimedEmailData = {
  holderFirstName?: string;
  serial:           string;     // NFC card serial — WhatsApp template var {{1}}
  recipientDisplay: string;     // sanitised "Priya, ending 4234" — var {{2}}
};
 
export function cardClaimedEmail(data: CardClaimedEmailData): { subject: string; html: string; text: string } {
  const greeting = data.holderFirstName ? `Hi ${data.holderFirstName},` : "Hi,";
  const content = `
    <p style="margin: 0 0 16px;">${escapeHtml(greeting)}</p>
    <p style="margin: 0 0 16px;">
      Your gold card <strong style="color: ${EMAIL_COLORS.goldDim};">${escapeHtml(data.serial)}</strong> was just added to another account by ${escapeHtml(data.recipientDisplay)}.
    </p>
    <p style="margin: 0 0 16px;">
      If you handed this card over, no action is needed — it now belongs to them.
    </p>
    <p style="margin: 0; font-family: ${EMAIL_FONTS.sans}; font-size: 13px; color: ${EMAIL_COLORS.muted};">
      If this wasn&rsquo;t you, please contact us right away.
    </p>
  `;
  return {
    subject: "Your gold card was claimed",
    html: renderEmail({
      preheader: `Card ${data.serial} was added to another account.`,
      content,
      cta: { label: "Contact us →", url: "https://www.1plus1gold.com/contact" },
    }),
    text: [
      greeting,
      "",
      `Your gold card ${data.serial} was just added to another account by ${data.recipientDisplay}.`,
      "If you handed this card over, no action is needed — it now belongs to them.",
      "If this wasn't you, please contact us right away: https://www.1plus1gold.com/contact",
      "",
      "1Plus1Gold · Singapore",
    ].join("\n"),
  };
}
 
function escapeHtml(s: string): string {
  return s.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&#39;");
}
