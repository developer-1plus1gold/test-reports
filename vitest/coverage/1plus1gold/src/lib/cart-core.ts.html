/**
 * cart-core.ts — the SINGLE source for Cart (+ Checkout) display derivation.
 *
 * Pure (no I/O, relative imports only so esbuild bundles it into the iOS
 * on-device engine — same constraint as shop-pricing.ts). Given the cart lines
 * + the live spot/premiums + the customer's curve position, it derives EVERY
 * value the cart UI shows (per-line price/line/premium/accessory-breakdown,
 * summary, curve position, milestone gate, fulfilment, free-accessory unlocks,
 * progress, KYC nudge) AND the pure cart transforms (refresh prices, milestone
 * conversion, gift-box reconcile, standard-tier hydrate).
 *
 * Consumed by BOTH the web cart/checkout pages AND the iOS kernel (engine.js
 * `cartView`/`applyMilestone`/`reconcileGiftBox`/`refreshCartPrices`) — one
 * implementation, no drift (the no-parallel-implementations rule). The CHARGE
 * stays server-authoritative (`/api/orders/create` recomputes + rejects a
 * mismatch > 1¢); this is DISPLAY only. Contract §25.
 */

/**
 * Build the cart line for a paid accessory from its catalog row. THE single
 * source for adding an accessory to the cart from the shop — both shop
 * add-buttons (grid card + list row) go through this. An accessory has g=0, so
 * its price never comes from grams: the line MUST carry `fixedPriceSgd` (the
 * list price) and `discountRule` (free-over-threshold rules), which is what
 * deriveCart prices it from. Omitting `fixedPriceSgd` was a real bug — a $5
 * accessory added at $0 — so it lives here once, tested, not duplicated inline.
 */
export function accessoryCartLine(
  p: { id: string; label: string; price_sgd: number; discount_rules: AccessoryDiscountRule[] | null },
  count: number,
  imageUrl: string | null,
): CartLine {
  return {
    productType:   "accessory",
    productId:     p.id,
    label:         p.label,
    g:             0,
    isAssay:       false,
    count,
    priceEach:     p.price_sgd,
    imageUrl,
    fixedPriceSgd: p.price_sgd,
    discountRule:  p.discount_rules,
  };
}

/** Available-to-sell cap for a cart line — same product resolution as
 *  isPresaleItem (productId first, then grams). null = untracked / made-to-order
 *  / presale / product-not-found → no cap. Mirrors web stockLeftOf
 *  (stock-display.ts). */
function stockLeftForItem(item: CartLine, catalog: CartCatalogProduct[]): number | null {

/** Best (highest threshold met) free rule — mirrors computeAccessoryLineTotal's

/** Fill `tiers` on standard lines that lack it (older carts / calculator adds
 *  stored standard items without tiers → rendered "Spot +0%"). Pure. */
export function hydrateStandardTiers(
  lines: CartLine[],
  catalog: CartCatalogProduct[],
  wholesaleEnabled = false,
): { lines: CartLine[]; changed: boolean } {
  let changed = false;
  const hydrated = lines.map((i) => {
    if (i.productType !== "standard") return i;
    const std = catalog.find((p) => (i.productId && p.id === i.productId) || (p.product_type === "standard" && p.grams === i.g));
    if (!std) {
      // Not in the catalog (server path passes catalog:[]; or a wholesale-only
      // product). Keep whatever the line carries — the server pre-set its
      // authoritative better-of tiers, and a missing/[] array is handled by the
      // pricing fallback below in deriveCart.
      return i;
    }
    const stdTiers = std.standard_premium_tiers && std.standard_premium_tiers.length > 0
      ? std.standard_premium_tiers
      : [{ min_qty: 1, max_qty: null, premium_pct: std.premium_pct ?? 0 }];
    const catalogWholesale = std.wholesale_premium_tiers && std.wholesale_premium_tiers.length > 0
      ? std.wholesale_premium_tiers
      : null;
    if (catalogWholesale) {
      // AUTHORITATIVE path — the catalog KNOWS this product's wholesale tiers, so
      // resolve EXACTLY like the server's priceOrder: better-of(standard, wholesale)
      // when the customer is wholesale-enabled AND the qty meets the wholesale
      // minimum (wholesaleTiersIfEligible), else standard. This OVERRIDES whatever
      // tiers the line arrived with — the single-source fix for the "cart total
      // mismatch" 422 (the wholesale surface embeds wholesale tiers in the line, a
      // SECOND pricing path that drifted from the server). iOS feeds the catalog
      // wholesale tiers + the gate via /api/cart|checkout/context.
      const eligibleWhl = wholesaleEnabled ? wholesaleTiersIfEligible(catalogWholesale, i.count) : undefined;
      changed = true;
      return { ...i, tiers: eligibleWhl ? pickBetterTiers(stdTiers, eligibleWhl, i.count) : stdTiers };
    }
    // The catalog has NO wholesale tiers for this product → it can't speak to
    // wholesale pricing, so never DISCARD the line's own usable tiers: a caller
    // whose catalog omits wholesale tiers (e.g. the web cart's /api/products) embeds
    // them in the line itself; overriding with bare standard would REGRESS it. Only
    // fill a missing/[] array (the "Spot +0%" bug) from the standard tiers.
    if (i.tiers && i.tiers.length > 0) return i;
    changed = true;
    return { ...i, tiers: stdTiers };
  });
  return { lines: hydrated, changed };
}
 
/** Wholesale tiers apply ONLY when the qty meets the wholesale MINIMUM — the

 *  below-min qty UP to the lowest tier, which would hand out wholesale pricing on
 *  sub-minimum Shop orders). Returns the wholesale tiers when eligible, else

/** Add `incoming` to `lines`, merging by itemKey (productId ?? type_g).
 *
 *  - Normal add: keep the existing line's tiers/price (first-wins), bump count.
 *  - Wholesale add (`fromWholesale`): when the SAME item is already in the cart
 *    (e.g. added earlier from the standard Shop), the BETTER (lower) premium for
 *    the merged quantity wins — so adding from Wholesale upgrades the line to
 *    wholesale pricing, but never downgrades a line that's already cheaper. When
 *    the incoming (wholesale) tiers win, adopt their unit priceEach too.
 *
 *  Shared by the web cart store + the iOS engine (no parallel implementation). */
export function mergeAdd(lines: CartLine[], incoming: CartLine, fromWholesale: boolean): CartLine[] {
  const key = keyOf(incoming);
  const idx = lines.findIndex((l) => keyOf(l) === key);
  if (idx < 0) return [...lines, incoming];
 
  const existing = lines[idx];
  const count = existing.count + incoming.count;
  const tiers = fromWholesale
    ? pickBetterTiers(existing.tiers, incoming.tiers, count)
    : (existing.tiers ?? incoming.tiers);
  // When the wholesale tiers won, the displayed unit price should follow them.
  const priceEach = fromWholesale && tiers === incoming.tiers ? incoming.priceEach : existing.priceEach;
  const merged: CartLine = { ...existing, count, tiers, priceEach };
  return lines.map((l, i) => (i === idx ? merged : l));
}
