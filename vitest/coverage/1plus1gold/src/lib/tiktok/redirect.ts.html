/**
 * TikTok Connect — the redirect URI and the consent-page URL. PURE (no I/O).
 *
 * ▶ The redirect URI must equal the one registered on the TikTok app (My Apps →
 * TikTok account holder redirect URL) EXACTLY, and TikTok requires `https`, a
 * trailing `/`, no query, no port (doc 1832209711206401). Only ONE redirect URL can be
 * active at a time, so Connect works on the PROD domain only — on DEV the admin card
 * says so instead of failing halfway (same as Lazada).
 *
 * ▶ The consent-page URL is GENERATED by TikTok (My Apps → Basic Information → "TikTok
 * account holder authorization URL"); its query parameters are not documented, so we
 * never build it. It is read from `TIKTOK_BIZ_AUTH_URL` (not a secret — it carries only
 * the public App ID, scopes and redirect URI) and we only append our `state`.
 */
export const TIKTOK_REDIRECT_URI = "https://1plus1gold.com/api/admin/marketing/tiktok/callback/";
 
/** True when this deployment is the one TikTok can redirect back to. */
export function canConnectHere(requestOrigin: string): boolean {
  return new URL(TIKTOK_REDIRECT_URI).origin === requestOrigin;
}
 
/** Fixed result codes the callback hands to the admin page — never free text from the
 *  request, so nothing attacker-supplied is reflected into the page. */
export const CONNECT_RESULTS = ["connected", "state_invalid", "missing_code", "denied", "not_configured", "failed", "prod_only"] as const;
export type ConnectResult = (typeof CONNECT_RESULTS)[number];
 
const TIKTOK_AUTH_ORIGIN = "https://www.tiktok.com";
const TIKTOK_AUTH_PATH = "/v2/auth/authorize";
 
/**
 * The consent URL for one Connect attempt: TikTok's generated URL + our `state`.
 * Returns null — never a half-right URL — when the configured URL is missing, is not
 * TikTok's authorize page, or names a redirect URI other than ours (that would send
 * the code somewhere else). `disable_auto_auth=1` always shows the consent screen, so
 * a reconnect is a deliberate tap rather than a silent bounce.
 */
export function buildAuthorizeUrl(configured: string | undefined, state: string): string | null {
  if (!configured) return null;
  let u: URL;
  try { u = new URL(configured.trim()); } catch { return null; }
  if (u.origin !== TIKTOK_AUTH_ORIGIN || u.pathname.replace(/\/$/, "") !== TIKTOK_AUTH_PATH) return null;
  const redirect = u.searchParams.get("redirect_uri");
  if (redirect !== null && redirect !== TIKTOK_REDIRECT_URI) return null;
  u.searchParams.set("state", state);
  u.searchParams.set("disable_auto_auth", "1");
  return u.toString();
}
