/**
 * Drumbeat → TikTok: one channel's publish step, end to end. Server-only.
 * Called by the marketing publisher for channel "tiktok" (video only).
 *
 *   1. a usable token for the connected account (Drumbeat → Channels → Connect TikTok)
 *   2. the media prefix verified with TikTok (automatic, idempotent)
 *   3. the approved video copied to the public tiktok-media bucket
 *   4. publish (draft → the account's TikTok inbox; direct → a public post)
 *   5. a short status poll; in-flight work is finished by the tiktok-publish-status cron
 *
 * The result uses the publisher's existing states, so nothing downstream changes:
 *   draft  → "manual_prepared" (Mark as posted after posting from the TikTok app)
 *   direct → "published" with `tt:<post id>` (or `tt-pending:<share_id>` until it lands)
 */

export interface DrumbeatTikTokResult {
  status: "published" | "manual_prepared" | "failed";
  platformPostId?: string;
  permalink?: string;
  manualPayload?: { caption: string; assetUrl: string };
  error?: string;
}
 
export interface DrumbeatTikTokInput {
  pieceId: string;
  video: { id: string; storagePath: string; mimeType: string };
  /** The channel's chosen cover image (Drumbeat cover picker) → custom_thumbnail_url. */
  cover?: { id: string; storagePath: string; mimeType: string } | null;
  caption: string;
}
