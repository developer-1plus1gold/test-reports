/**
 * iOS app-version policy: what the native app compares itself against to show
 * "Update available" (running < latest) or "Update required" (running < min).
 * Design: docs/ios-app-update-design.md (Mukesh, 2026-11-22).
 *
 * ▶ "LATEST" IS READ FROM APPLE, NEVER TYPED. Apple's public lookup returns the
 * version that is live on the App Store, so a TestFlight or ad-hoc build never
 * moves it and nobody has to remember to update it after a release. The route
 * that serves this is CDN-cached for 6 h, so Apple is asked a few times a day.
 *
 * ▶ FAILS QUIET. Any lookup problem yields `latest: null`, and the app shows
 * nothing. A broken Apple response must never put a banner in front of every
 * customer, let alone the hard screen.
 */
 
export const IOS_BUNDLE_ID = "com.oneplusonegold.ios";
 
/**
 * EMERGENCY ONLY — never set per release. A build below this sees a full-screen
 * "Update required" it cannot dismiss. Set it only when an installed build can no
 * longer work (e.g. it cannot complete a payment), and only to a version that is
 * already live on the App Store, or customers are locked out with nowhere to go.
 */
export const IOS_MIN_SUPPORTED_VERSION: string | null = null;
 
const LOOKUP_URL = `https://itunes.apple.com/lookup?bundleId=${IOS_BUNDLE_ID}&country=sg`;
const LOOKUP_TIMEOUT_MS = 5_000;
 
const VERSION_RE = /^\d{1,4}(?:\.\d{1,4}){0,3}$/;
 
/** A dotted numeric version ("1.0.14"), or null if the input is anything else. */
export function parseVersion(raw: unknown): string | null {
  if (typeof raw !== "string") return null;
  const v = raw.trim();
  return VERSION_RE.test(v) ? v : null;
}
 
/** Numeric, segment-wise compare: 1.0.10 > 1.0.9; missing segments count as 0. */
export function compareVersions(a: string, b: string): number {
  const pa = a.split(".").map(Number);
  const pb = b.split(".").map(Number);
  for (let i = 0; i < Math.max(pa.length, pb.length); i++) {
    const d = (pa[i] ?? 0) - (pb[i] ?? 0);
    if (d !== 0) return d < 0 ? -1 : 1;
  }
  return 0;
}
 
/** Only an https link to the App Store is ever handed to the app to open. */
export function safeStoreUrl(raw: unknown): string | null {
  if (typeof raw !== "string") return null;
  try {
    const u = new URL(raw);
    return u.protocol === "https:" && u.hostname === "apps.apple.com" ? u.toString() : null;
  } catch {
    return null;
  }
}
 
export type IosVersionPolicy = {
  /** Version live on the App Store; null when unknown (the app shows nothing). */
  latest: string | null;
  /** Emergency floor; null = no forced update. */
  minSupported: string | null;
  /** App Store page for the Update button; null when unknown. */
  storeUrl: string | null;
  /** Apple's numeric app id: the app opens its store page IN-APP
   *  (SKStoreProductViewController) with it; null → falls back to storeUrl. */
  appStoreId: string | null;
};
 
/** Apple's numeric track id as a digit string, or null. */
export function parseAppStoreId(raw: unknown): string | null {
  const s = typeof raw === "number" && Number.isSafeInteger(raw) && raw > 0 ? String(raw)
    : typeof raw === "string" ? raw.trim() : "";
  return /^\d{6,12}$/.test(s) ? s : null;
}
 
/** Read the live App Store version. Never throws. */
export async function getIosVersionPolicy(
  fetchImpl: typeof fetch = fetch,
): Promise<IosVersionPolicy> {
  const minSupported = parseVersion(IOS_MIN_SUPPORTED_VERSION);
  const fallbackStoreUrl = safeStoreUrl(process.env.NEXT_PUBLIC_APP_STORE_URL);
  try {
    const res = await fetchImpl(LOOKUP_URL, {
      signal: AbortSignal.timeout(LOOKUP_TIMEOUT_MS),
      headers: { Accept: "application/json" },
    });
    if (!res.ok) throw new Error(`lookup HTTP ${res.status}`);
    const body = (await res.json()) as { results?: Array<{ version?: unknown; trackViewUrl?: unknown; bundleId?: unknown; trackId?: unknown }> };
    const hit = body.results?.find((r) => r.bundleId === IOS_BUNDLE_ID);
    const latest = parseVersion(hit?.version);
    if (!latest) throw new Error("lookup returned no usable version");
    return {
      latest,
      minSupported,
      storeUrl: safeStoreUrl(hit?.trackViewUrl) ?? fallbackStoreUrl,
      appStoreId: parseAppStoreId(hit?.trackId),
    };
  } catch (err) {
    console.warn("[app-version] App Store lookup failed:", err instanceof Error ? err.message : err);
    return { latest: null, minSupported, storeUrl: fallbackStoreUrl, appStoreId: null };
  }
}
