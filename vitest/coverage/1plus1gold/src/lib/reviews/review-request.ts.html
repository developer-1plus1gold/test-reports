/**
 * The post-delivery review request — who gets asked, when, and the three
 * separate things that have to be true before a single message goes out.
 *
 * ▶ READ THIS BEFORE TURNING IT ON. The operator declined a WhatsApp review
 * campaign outright on 2026-09-16 — *"I do not want to be like a spam on
 * whatsapp, as people treat it very personally and in frustration end up
 * locking the chats"* — and reversed that on 2026-09-17 with an explicit
 * condition: *"lets build it and keep it ready. Don't turn it on."* So this
 * module's job is not only to decide who is eligible; it is to make an
 * accidental send structurally impossible while it waits.
 *
 * THREE INDEPENDENT GATES, and each one alone is enough to stop everything:
 *
 *  1. `REVIEW_REQUEST_ENABLED` must be exactly "true". Unset, empty, "1",
 *     "yes" and "TRUE" all mean OFF. An env var that must be spelled one way
 *     cannot be turned on by a typo, and it is off in dev and preview for free.
 *
 *  2. `REVIEW_REQUEST_START_AT` must be a valid date, and ONLY orders delivered
 *     AFTER it are eligible. ▶ THIS IS THE ONE THAT MATTERS MOST, and it is
 *     not a nicety: without it, the moment the switch flips, every customer who
 *     has ever received an order is messaged at once — a mass send is exactly
 *     the outcome the original objection was about. Set it to the moment you
 *     turn the feature on and the back catalogue is excluded by construction
 *     rather than by anyone remembering. A missing or unparseable value while
 *     ENABLED means SEND NOTHING: failing closed is the only safe direction for
 *     a switch whose failure mode is messaging thousands of people.
 *
 *  3. Per-customer state: asked once, ever, and never after a "no".
 *
 * ▶ AND A FOURTH THING, ADDED 2026-09-18, THAT LOOSENS RATHER THAN TIGHTENS —
 * `REVIEW_REQUEST_BACKFILL_UNTIL`, the one-time back-catalogue window. It lifts
 * the staleness bound and raises the per-run cap for a fixed period and then
 * expires on its own. It does NOT touch any of the three gates above. See
 * `reviewRequestBackfill` below for why it is shaped that way; the short
 * version is that a backfill is a decision about HOW OLD an order may be, never
 * about whether someone already said no.
 *
 * ▶ ONCE PER CUSTOMER, NOT PER ORDER. A customer who buys monthly must not be
 * asked monthly. The stamp lives on the person (`profiles.review_request_sent_at`).
 *
 * ▶ A DISMISSAL ON THE ACCOUNT CARD SILENCES THIS CHANNEL TOO, and that
 * direction is deliberate. Someone who pressed "Not now" on the quiet in-app
 * card has already answered; following them to WhatsApp afterwards is precisely
 * the nagging that was ruled out. The reverse does NOT hold — being messaged
 * does not hide the account card, because a passive card the customer chose to
 * look at is not an interruption.
 *
 * ▶ AND IT IS NOT A SENTIMENT FUNNEL. Eligibility never looks at order value,
 * complaint history or anything else predictive of a good review. Selectively
 * asking only customers you expect to be happy is review gating — against
 * Trustpilot's own guidelines, and the thing that makes a score worthless.
 *
 * ▶ CORRECTED 2026-09-18. This paragraph used to finish: *"The message itself
 * leads with 'if anything is not right, tell us first', which is the same
 * two-sided shape the account card already uses."* That sentence stopped being
 * true when the operator had the copy rewritten to a single humble ask, and it
 * is corrected rather than deleted because it also stated a WRONG reason: the
 * two-sided wording was never what kept this from being gating. The guard is
 * the paragraph above — eligibility consults nothing predictive of a good
 * review, and every delivered customer is asked. Copy cannot carry that
 * property, so changing the copy cannot lose it. (The complaint route still
 * sits unconditionally on the account card, where someone chose to look.)
 */
import { deriveLifecycle } from "@/lib/orders/lifecycle";
 
/** A settling period. Asking the hour a parcel lands is asking someone to
 *  review a box they have not opened. */
export const SETTLE_DAYS = 3;
/** Past this, the ask is stale — "how was the order you got five weeks ago" is
 *  a worse message, and the bound also caps any burst if the switch is flipped
 *  after a quiet spell. */
export const MAX_AGE_DAYS = 30;
/** Per run. A bounded mistake is a recoverable one. */
export const MAX_PER_RUN = 25;
 
/**
 * THE ONE-TIME BACKFILL — the two bounds that move while it is running.
 *
 * ▶ WHY THIS EXISTS. Operator, 2026-09-18: *"for Sunday message - lets send to
 * all - make this 1 time exception."* The standing 30-day staleness bound and
 * the 25-per-run cap both exist to keep an ordinary day's send small; a
 * deliberate one-off ask to the whole back catalogue is the case they were
 * never written for. So they lift, together, for a window — and then stop.
 *
 * ▶ IT EXPIRES BY ITSELF, WHICH IS THE WHOLE DESIGN. `REVIEW_REQUEST_BACKFILL_UNTIL`
 * is a wall-clock moment: past it, the normal bounds are back with no code
 * change, no follow-up PR and nobody remembering. This repo has paid for the
 * other shape — on 2026-08-17 the holiday `ops_notice` banner self-expired
 * exactly as designed while the carrier toggles beside it needed manual
 * reverts, and it was the manual ones that had to be chased. "One-time" is a
 * property you build, not a promise you make.
 *
 * ▶ WHAT IT DELIBERATELY DOES **NOT** LIFT — and this is the important half.
 * Not the sent stamp, not an in-app dismissal, not the settling period, and
 * NOT `REVIEW_REQUEST_START_AT`. Those four are what make the feature safe at
 * all; a backfill flag that swept them aside would be a single variable that
 * releases everyone, which is the mass-send this whole module is built to
 * prevent. During a backfill the cutoff is still the real floor — the operator
 * moves it back deliberately, as a separate visible act.
 */
export const BACKFILL_MAX_AGE_DAYS = 3650;
/** Still a cap. The point of a per-run cap is to bound a MISTAKE, and a
 *  backfill's intended size is the thing being lifted, not the safety margin —
 *  so this is raised to cover a real back catalogue in one run, not removed. */
export const BACKFILL_MAX_PER_RUN = 100;
/** How far ahead the window may be set. A date that can be 2099 is not a
 *  one-time exception, it is a permanent switch with extra steps — so a window
 *  longer than this is REFUSED rather than clamped, because a typo in the year
 *  should be loud on /admin/crons rather than quietly become the new normal. */
export const BACKFILL_MAX_WINDOW_DAYS = 14;
 
const DAY_MS = 86_400_000;
 
/** What these functions actually need: a bag of string env vars. Deliberately
 *  NOT NodeJS.ProcessEnv — that type requires NODE_ENV, so every test had to
 *  cast, and the cast type-checked under vitest and was then REJECTED by
 *  `tsc --noEmit`. The gate that matters is the one that caught it. */
export type ReviewRequestEnv = Record<string, string | undefined>;
 
/** Exactly "true". Anything else — including "TRUE", "1" and "yes" — is OFF. */
export function reviewRequestEnabled(env: ReviewRequestEnv = process.env): boolean {
  return env.REVIEW_REQUEST_ENABLED === "true";
}
 
/**
 * The cutoff. `null` means "no usable cutoff", which callers MUST treat as
 * send-nothing rather than send-everything — see gate 2 above.
 */
export function reviewRequestStartAt(env: ReviewRequestEnv = process.env): Date | null {
  const raw = (env.REVIEW_REQUEST_START_AT ?? "").trim();
  if (!raw) return null;
  const d = new Date(raw);
  return Number.isNaN(d.getTime()) ? null : d;
}
 
export type ReviewRequestBackfill =
  | { active: true; until: Date }
  | { active: false; reason: "unset" | "unparseable" | "expired" | "window_too_long" };
 
/**
 * Is the one-time backfill running right now?
 *
 * Unset, unparseable, already past, or set absurdly far ahead all resolve to
 * INACTIVE — i.e. every failure here fails toward FEWER messages. Note that is
 * the opposite direction from `reviewRequestStartAt`, where an unparseable
 * value means send nothing at all; the principle is the same one both times,
 * because there the safer branch is "stop" and here it is "carry on normally".
 */
export function reviewRequestBackfill(
  env: ReviewRequestEnv = process.env,
  now: Date = new Date(),
): ReviewRequestBackfill {
  const raw = (env.REVIEW_REQUEST_BACKFILL_UNTIL ?? "").trim();
  if (!raw) return { active: false, reason: "unset" };
  const until = new Date(raw);
  if (Number.isNaN(until.getTime())) return { active: false, reason: "unparseable" };
  if (until.getTime() <= now.getTime()) return { active: false, reason: "expired" };
  if (until.getTime() - now.getTime() > BACKFILL_MAX_WINDOW_DAYS * DAY_MS) {
    return { active: false, reason: "window_too_long" };
  }
  return { active: true, until };
}
 
export type ReviewRequestGate =
  | { send: true }
  | { send: false; reason: "disabled" | "no_start_at" };
 
/** The whole feature in one answer, so no caller can half-check it. */
export function reviewRequestGate(env: ReviewRequestEnv = process.env): ReviewRequestGate {
  if (!reviewRequestEnabled(env)) return { send: false, reason: "disabled" };
  if (!reviewRequestStartAt(env)) return { send: false, reason: "no_start_at" };
  return { send: true };
}
 
export type ReviewCandidateOrder = {
  status: string | null;
  shipper_status?: string | null;
  /** When the order reached `completed`. Orders that never did carry null. */
  lifecycle_completed_at?: string | null;
};
 
export type ReviewCandidate = {
  orders: ReviewCandidateOrder[];
  /** `profiles.review_prompt_dismissed_at` — they already said no in-app. */
  dismissedAt: string | null;
  /** `profiles.review_request_sent_at` — we already asked, once, ever. */
  sentAt: string | null;
};
 
export type ReviewDecision =
  | { ask: true; deliveredAt: Date }
  | { ask: false; reason: "already_asked" | "dismissed_in_app" | "nothing_delivered" | "before_cutoff" | "too_soon" | "too_old" };
 
/**
 * Decide for ONE customer. Pure — no clock, no env, no database — so every
 * branch is testable and the order of the checks is visible.
 *
 * The order is not arbitrary: the two "they have answered / we have asked"
 * cases come FIRST, so that a customer who said no can never be re-examined by
 * a later rule change to the timing.
 */
export function decideReviewRequest(
  c: ReviewCandidate,
  opts: { now: Date; startAt: Date; settleDays?: number; maxAgeDays?: number },
): ReviewDecision {
  if (c.sentAt) return { ask: false, reason: "already_asked" };
  if (c.dismissedAt) return { ask: false, reason: "dismissed_in_app" };
 
  const delivered = c.orders
    .filter((o) => deriveLifecycle(o.status ?? "", o.shipper_status ?? null) === "completed")
    .map((o) => (o.lifecycle_completed_at ? new Date(o.lifecycle_completed_at) : null))
    .filter((d): d is Date => d !== null && !Number.isNaN(d.getTime()));
 
  if (delivered.length === 0) return { ask: false, reason: "nothing_delivered" };
 
  // The MOST RECENT delivery is what the message is about — asking about an
  // older parcel when a newer one has just arrived reads as a system that is
  // not paying attention.
  const deliveredAt = new Date(Math.max(...delivered.map((d) => d.getTime())));
 
  if (deliveredAt <= opts.startAt) return { ask: false, reason: "before_cutoff" };
 
  const ageMs = opts.now.getTime() - deliveredAt.getTime();
  if (ageMs < (opts.settleDays ?? SETTLE_DAYS) * DAY_MS) return { ask: false, reason: "too_soon" };
  if (ageMs > (opts.maxAgeDays ?? MAX_AGE_DAYS) * DAY_MS) return { ask: false, reason: "too_old" };
 
  return { ask: true, deliveredAt };
}
