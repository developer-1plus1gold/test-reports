/** TikTok Shop product status (Search Products: DRAFT, PENDING, FAILED, ACTIVATE,

export function tiktokState(status: string | null): LiveState {
  switch ((status ?? "").toUpperCase()) {
    case "ACTIVATE": return "live";
    case "PENDING": case "SCHEDULED": return "pending";
    case "FAILED": return "rejected";
    case "SELLER_DEACTIVATED": case "PLATFORM_DEACTIVATED": case "FREEZE": return "inactive";

    default: return "unknown";
  }
}

/**
 * Reconcile ONE channel straight after its price push (Mukesh 2026-11-18) — only when the
 * push actually sent something, so the 5-minute stock-only ticks do not trigger it. Never
 * throws: a reconciliation problem must not fail the push that already succeeded; it is
 * returned for the cron's result line and lands on the channel's Control Tower card.
 * ▶ A marketplace that applies a price a little after accepting it would show here as a
 *   "price changed" drift; the next 6-hourly run (or push) clears it. Watch the first cards.
 */
export async function reconcileAfterPush(
  db: SupabaseClient, channel: string, push: { ran: boolean; sent: number }, now: Date, http: Http = {},
): Promise<{ drifts: number; error?: string } | null> {
  if (!push.ran || push.sent === 0) return null;
  const reader = RECONCILE_READERS.find((r) => r.channel === channel);
  if (!reader) return null;
  try {
    const r = await runReconcile(db, now, [reader], { [channel]: CHANNEL_JOBS[channel] ?? [] }, http);
    const c = r.channels[0];
    return { drifts: c?.drifts.length ?? 0, ...(c?.error ? { error: c.error } : {}) };
  } catch (e) {
    return { drifts: 0, error: (e instanceof Error ? e.message : String(e)).slice(0, 200) };
  }
}
