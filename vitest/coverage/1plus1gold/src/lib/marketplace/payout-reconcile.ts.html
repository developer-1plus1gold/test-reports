/**
 * Marketplace payout reconciliation — the rules (Mukesh 2026-11-24).
 *
 * Lazada and TikTok Shop pay us in BATCHES (a payout statement covering many orders). For each
 * order in a batch this decides what happens, and it never depends on our fee being right:
 *
 *   1. MATCH on the marketplace's own order number (orders.source_channel + external_order_id) —
 *      never on an amount. A batch lists exactly which orders it pays.
 *   2. FEES ARE THEIRS. The marketplace's fee and shipping for the order replace ours, so an
 *      estimate (or a waived commission during incubation) is corrected by the payout itself.
 *      No tolerance applies to fees.
 *   3. FLAG ONLY WHAT FEES CANNOT EXPLAIN: what the buyer paid (the marketplace's revenue for
 *      the order) differs from our order total by more than 5 cents; an adjustment (refund,
 *      penalty, compensation) on the order; a cancelled order that was paid out; or a paid
 *      order the marketplace gives no revenue figure for. A difference is never auto-closed.
 *   4. CLOSE when the batch is PAID and step 3 found nothing: the order's payout is received
 *      (payment_status → paid, paid_at = the batch's paid date).
 *   5. A VOUCHER THE MARKETPLACE PAID (Mukesh 11-24, option 1): when the marketplace credits MORE
 *      than the buyer paid, and the extra is no more than the discount on our order, the extra is
 *      the marketplace's share of a co-funded voucher — income to us, not a difference. It is
 *      recorded on the order (`marketplace_funded_discount_sgd`, counted as gold revenue by
 *      lib/finance/order-revenue) and the order closes. A larger extra, or any shortfall, is
 *      still a difference.
 *
 * Pure: the readers (lib/lazada/payouts, lib/tiktok-shop/payouts) produce PayoutBatch values,
 * the runner (payout-run.ts) loads our orders and applies the decisions.
 */
 
export type PayoutChannel = "lazada" | "tiktok_shop";
export type PayoutStatus = "pending" | "paid" | "failed";
 
/** One order inside a payout batch, as the marketplace settled it. All amounts SGD. */
export interface PayoutOrder {
  externalOrderId: string;
  /** What the buyer paid for the order, as the marketplace counts it; null when not given. */
  revenueSgd: number | null;
  /** Marketplace fees on the order (commission, payment, service…), ≥ 0. */
  feeSgd: number;
  /** Shipping the marketplace deducted from us, ≥ 0. */
  shippingSgd: number;
  /** Anything else on the order (refund, penalty, compensation) — signed, + is to us. */
  adjustmentSgd: number;
  /** What the marketplace pays us for the order. */
  settlementSgd: number;
}
 
export interface PayoutBatch {
  channel: PayoutChannel;
  ref: string;
  status: PayoutStatus;
  paidAt: string | null;
  /** The batch total the marketplace pays out; null when not given. */
  amountSgd: number | null;
  periodLabel: string | null;
  orders: PayoutOrder[];
  /** Charges in the batch tied to no order (ads, monthly fees), ≥ 0. */
  otherDeductionsSgd: number;
}
 
/** Our side of an order, read from `orders`. */
export interface OurOrder {
  id: string;
  externalOrderId: string;
  status: string;
  paymentStatus: string | null;
  grandTotalSgd: number;
  feeSgd: number | null;
  shippingSgd: number | null;
  /** The discount recorded on our order (the whole voucher, as the buyer saw it). */
  discountSgd: number;
  /** The share of that discount already recorded as paid by the marketplace. */
  fundedSgd: number;
}
 
export const REVENUE_TOLERANCE_SGD = 0.05;
 
export type OrderState = "closed" | "difference" | "pending" | "unknown" | "cancelled";
 
export interface OrderDecision {
  externalOrderId: string;
  orderId: string | null;
  state: OrderState;
  /** Write payment_status = paid (only when the order is still unpaid). */
  close: boolean;
  /** Fee / shipping to write on the order (null = leave as is). */
  setFeeSgd: number | null;
  setShippingSgd: number | null;
  /** The marketplace-funded share of the voucher to record (null = leave as is). */
  setFundedSgd: number | null;
  previousFeeSgd: number | null;
  note: string | null;
}
 
const r2 = (n: number) => Math.round(n * 100) / 100 + 0;
const money = (n: number) => `S$${r2(n).toFixed(2)}`;
const CANCELLED = /cancel|refund/i;
 
export function decideOrder(batch: PayoutBatch, po: PayoutOrder, ours: OurOrder | undefined): OrderDecision {
  const base = {
    externalOrderId: po.externalOrderId, orderId: ours?.id ?? null,
    close: false, setFeeSgd: null, setShippingSgd: null, setFundedSgd: null, previousFeeSgd: ours?.feeSgd ?? null, note: null,
  };
  if (!ours) return { ...base, state: "unknown", note: "No order of ours carries this marketplace order number" };
 
  if (CANCELLED.test(ours.status)) {
    return Math.abs(po.settlementSgd) < 0.005
      ? { ...base, state: "cancelled" }
      : { ...base, state: "difference", note: `Cancelled here, but the marketplace settled ${money(po.settlementSgd)} for it` };
  }
  if (batch.status !== "paid") return { ...base, state: "pending" };
 
  if (po.revenueSgd == null) {
    return { ...base, state: "difference", note: "The marketplace paid this order but gave no figure for what the buyer paid" };
  }
  // What the marketplace should credit for the order: what the buyer paid, plus any share of the
  // voucher already recorded as the marketplace's.
  const expected = r2(ours.grandTotalSgd + ours.fundedSgd);
  const revenueGap = r2(po.revenueSgd - expected);
  const voucherRoom = r2(Math.max(0, ours.discountSgd - ours.fundedSgd));
  let setFundedSgd: number | null = null;
  if (revenueGap > REVENUE_TOLERANCE_SGD && revenueGap <= voucherRoom + REVENUE_TOLERANCE_SGD) {
    setFundedSgd = r2(Math.min(ours.discountSgd, ours.fundedSgd + revenueGap));
  } else if (Math.abs(revenueGap) > REVENUE_TOLERANCE_SGD) {
    return {
      ...base, state: "difference",
      note: `Buyer paid ${money(po.revenueSgd)} per the marketplace; our order says ${money(expected)} (${revenueGap > 0 ? "+" : "−"}${money(Math.abs(revenueGap))})`,
    };
  }
  if (Math.abs(po.adjustmentSgd) >= 0.01) {
    return {
      ...base, state: "difference",
      note: `An adjustment of ${po.adjustmentSgd > 0 ? "+" : "−"}${money(Math.abs(po.adjustmentSgd))} on this order (refund, penalty or compensation)`,
    };
  }
 
  const fee = r2(po.feeSgd), ship = r2(po.shippingSgd);
  const feeChanged = ours.feeSgd == null || Math.abs(Number(ours.feeSgd) - fee) >= 0.01;
  const shipChanged = ours.shippingSgd == null || Math.abs(Number(ours.shippingSgd) - ship) >= 0.01;
  const notes = [
    feeChanged ? `Fee ${ours.feeSgd == null ? "—" : money(Number(ours.feeSgd))} → ${money(fee)} (the marketplace's)` : null,
    setFundedSgd != null ? `The marketplace paid ${money(setFundedSgd - ours.fundedSgd)} of the ${money(ours.discountSgd)} voucher` : null,
  ].filter(Boolean);
  return {
    ...base,
    state: "closed",
    close: ours.paymentStatus !== "paid",
    setFeeSgd: feeChanged ? fee : null,
    setShippingSgd: shipChanged ? ship : null,
    setFundedSgd,
    note: notes.length ? notes.join("; ") : null,
  };
}

export interface BatchReport {
  channel: PayoutChannel;
  ref: string;
  status: PayoutStatus;
  paidAt: string | null;
  amountSgd: number | null;
  ordersSgd: number;
  otherDeductionsSgd: number;
  closed: number;
  differences: number;
  unknown: number;
  /** Sum of (new fee − old fee) over the orders whose fee was corrected. */
  feeCorrectionSgd: number;
  /** Voucher share newly recorded as paid by the marketplace (income), this batch. */
  fundedVoucherSgd: number;
  decisions: OrderDecision[];
}
 
export function decideBatch(batch: PayoutBatch, ours: Map<string, OurOrder>): BatchReport {
  const decisions = batch.orders.map((po) => decideOrder(batch, po, ours.get(po.externalOrderId)));
  const feeCorrection = decisions.reduce(
    (s, d) => s + (d.setFeeSgd != null ? d.setFeeSgd - (d.previousFeeSgd ?? 0) : 0), 0);
  return {
    channel: batch.channel, ref: batch.ref, status: batch.status, paidAt: batch.paidAt, amountSgd: batch.amountSgd,
    ordersSgd: r2(batch.orders.reduce((s, o) => s + o.settlementSgd, 0)),
    otherDeductionsSgd: r2(batch.otherDeductionsSgd),
    closed: decisions.filter((d) => d.state === "closed").length,
    differences: decisions.filter((d) => d.state === "difference").length,
    unknown: decisions.filter((d) => d.state === "unknown").length,
    feeCorrectionSgd: r2(feeCorrection),
    fundedVoucherSgd: r2(decisions.reduce((s, d) => {
      if (d.setFundedSgd == null) return s;
      const o = ours.get(d.externalOrderId);
      return s + d.setFundedSgd - (o?.fundedSgd ?? 0);
    }, 0)),
    decisions,
  };
}
 
const CHANNEL_LABEL: Record<PayoutChannel, string> = { lazada: "Lazada", tiktok_shop: "TikTok Shop" };
 
/**
 * The admin WhatsApp for a newly PAID batch (the alert template's detail line; it must not start
 * or end with a variable — the template wraps it).
 */
export function payoutAlertDetail(r: BatchReport): string {
  const date = r.paidAt
    ? new Date(r.paidAt).toLocaleDateString("en-SG", { day: "numeric", month: "short", timeZone: "Asia/Singapore" })
    : "date not given";
  const parts = [
    `${CHANNEL_LABEL[r.channel]} payout ${r.amountSgd != null ? money(r.amountSgd) : "(amount not given)"} paid ${date}`,
    `${r.closed} order${r.closed === 1 ? "" : "s"} closed`,
  ];
  if (Math.abs(r.feeCorrectionSgd) >= 0.01) {
    parts.push(`fees corrected by ${r.feeCorrectionSgd > 0 ? "+" : "−"}${money(Math.abs(r.feeCorrectionSgd))}`);
  }
  if (r.fundedVoucherSgd >= 0.01) parts.push(`${money(r.fundedVoucherSgd)} of vouchers paid by ${CHANNEL_LABEL[r.channel]}`);
  if (r.otherDeductionsSgd >= 0.01) parts.push(`${money(r.otherDeductionsSgd)} other deductions`);
  parts.push(r.differences ? `${r.differences} to check in Admin → Payments` : "nothing to check");
  return `${parts.join(", ")}.`;
}
