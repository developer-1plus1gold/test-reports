/**
 * Marketplace reconciliation — what is ACTUALLY live on each marketplace, against what we
 * recorded (Mukesh 2026-11-18: "how are we ensuring all the updates in each direction are
 * working?"). Channel-agnostic: a marketplace plugs in a ChannelReader (lib/marketplace/
 * reconcile-readers.ts); Shopee joins by adding one reader, nothing here changes.
 *
 * What it catches — none of it was visible before:
 *  - a mapped listing that is MISSING on the marketplace (deleted / delisted there);
 *  - a listing that is NOT LIVE (pending review, rejected, deactivated) — and our stored
 *    status, which nothing ever read back, is corrected (TikTok listings sat at "pending"
 *    while buyers were ordering);
 *  - PRICE changed outside our push (a push that silently did not apply, or a hand edit in
 *    Seller Center);
 *  - STOCK above what we pushed — an OVERSELL risk (approve severity);
 *  - a marketplace SYNC JOB failing run after run (cron_runs), so a dead push is not mistaken
 *    for "nothing to do".
 * Live values are written back to marketplace_listings, so the next price/stock push (which
 * plans against those values) corrects any drift by itself; the card says what was found.
 *
 * Pure functions here; the runner (I/O) is runReconcile.
 */

/** How we store a live state on marketplace_listings.marketplace_status. "active" for live
 *  on every channel — Lazada's own word, and what the rest of the code filters on. */
export function storedStatus(state: LiveState): string {
  return state === "live" ? "active" : state;
}
 
const money = (n: number) => `S$${n.toFixed(2)}`;
 
/** Live first: when we cannot tell which listing is ours, the one a buyer can order from. */
const STATE_RANK: Record<LiveState, number> = { live: 0, pending: 1, draft: 2, rejected: 3, unknown: 4, inactive: 5, deleted: 6 };
 
/**
 * Pure: the ONE marketplace listing to judge for our row, among every listing that carries
 * its seller SKU. A SKU can have several (TikTok Shop, 11-19: eight listings were recreated
 * and the 11-17 originals deactivated, both keeping the same SKU). Taking "the last one in
 * the list" judged the dead originals: a card of 6 false "oversell" and 8 "not live" lines
 * (11-25), and their status / stock / price were written back over the live listings' values.
 * Our own item wins when we recorded it; otherwise the most sellable state.
 */
export function pickLiveListing(candidates: readonly LiveListing[], ourItemId?: string | null): LiveListing | undefined {
  if (candidates.length === 0) return undefined;
  if (ourItemId) {
    const mine = candidates.find((c) => c.itemId === ourItemId);
    if (mine) return mine;
  }
  return [...candidates].sort((a, b) => STATE_RANK[a.state] - STATE_RANK[b.state])[0];
}

export interface CronRunRow { cron_name: string; status: string; ran_at: string; detail?: string | null }
 
/** Pure: jobs whose last `streak` runs all failed. Rows newest-first or any order. */
export function failingJobs(runs: readonly CronRunRow[], jobs: readonly string[], streak = 3): Drift[] {
  const out: Drift[] = [];
  for (const job of jobs) {
    const recent = runs.filter((r) => r.cron_name === job).sort((a, b) => b.ran_at.localeCompare(a.ran_at)).slice(0, streak);
    if (recent.length === streak && recent.every((r) => r.status !== "ok" && r.status !== "success")) {
      out.push({ kind: "job_failing", sellerSku: null, name: job, detail: `${job} failed its last ${streak} runs — latest: ${(recent[0].detail ?? recent[0].status).slice(0, 160)}` });
    }
  }
  return out;
}
 
const KIND_ORDER: DriftKind[] = ["oversell", "missing", "job_failing", "not_live", "price"];
 
/** Pure: the one-line card summary for a channel. */
export function summarise(label: string, drifts: readonly Drift[]): string {
  const counts = KIND_ORDER.map((k) => [k, drifts.filter((d) => d.kind === k).length] as const).filter(([, n]) => n > 0);
  const words: Record<DriftKind, string> = { oversell: "oversell risk", missing: "missing", job_failing: "sync job failing", not_live: "not live", price: "price changed" };
  return `${label}: ${counts.map(([k, n]) => `${n} ${words[k]}`).join(" · ")}`;
}
 
/** Oversell and failing jobs need a decision; the rest is for awareness. */
export function severityOf(drifts: readonly Drift[]): "approve" | "notify" {
  return drifts.some((d) => d.kind === "oversell" || d.kind === "job_failing" || d.kind === "missing") ? "approve" : "notify";
}

export const RECONCILE_AGENT = "marketplace_reconcile";
/** One card TYPE per channel, so a run for one channel (after its price push) can never
 *  close another channel's open card — runMonitor closes by agent + entity type. */
export const reconcileEntityType = (channel: string) => `marketplace_channel:${channel}`;
 
/** Stable Control Tower entity id per channel: one open card per channel, closed when clean. */
export function channelEntityId(channel: string): string {
  const h = createHash("sha256").update(`marketplace_reconcile:${channel}`).digest("hex");
  return `${h.slice(0, 8)}-${h.slice(8, 12)}-${h.slice(12, 16)}-${h.slice(16, 20)}-${h.slice(20, 32)}`;
}
