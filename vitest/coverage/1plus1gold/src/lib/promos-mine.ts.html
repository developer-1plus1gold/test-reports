/**
 * getRedeemableRewards — the customer's "My Rewards" wallet contents: promo codes
 * they can still redeem (assigned + all-customer, in-window, not exhausted).
 *
 * Extracted from `/api/promos/mine` so BOTH that route AND the cart context
 * (`buildCartContext` → the in-cart picker, §45 PROMO-SURF-2) consume ONE
 * implementation — no parallel query, no drift.
 *
 * Display only — the discount is always recomputed server-authoritatively at
 * validate/charge. Never trust these amounts for the charge.
 */
 
import type { SupabaseClient } from "@supabase/supabase-js";
 
export interface RedeemableReward {
  code: string;
  /** "" when the admin didn't set a title (never null — so the iOS SDUI spec can
   *  gate the derived-label rows with safe string equality). */
  title: string;
  type: string;
  valueSgd: number | null;
  percent: number | null;
  maxDiscountSgd: number | null;
  minOrderSgd: number | null;
  /** "" when no expiry (never null — see title). */
  expiresAt: string;
  source: "assigned" | "all";
}
 
type Row = {
  id: string; code: string; title: string | null; type: string;
  value_sgd: number | null; percent: number | null; max_discount_sgd: number | null;
  min_order_sgd: number | null; per_customer_limit?: number | null;
  max_uses: number | null; uses_count: number | null; used?: boolean;
  starts_at: string | null; expires_at: string | null;
};
 
function shape(c: Row, source: "assigned" | "all"): RedeemableReward {
  return {
    code:           c.code,
    // title/expiresAt coalesced to "" (never null) so the iOS SDUI spec can gate
    // the derived-label + expiry rows with safe string equality. Behaviour-
    // identical for the web (an empty string is falsy → client derives the label).
    title:          c.title ?? "",
    type:           c.type,
    valueSgd:       c.type === "value" ? Number(c.value_sgd) : null,
    percent:        c.type === "percent" ? Number(c.percent) : null,
    maxDiscountSgd: c.type === "percent" ? Number(c.max_discount_sgd) : null,
    minOrderSgd:    c.min_order_sgd != null ? Number(c.min_order_sgd) : null,
    expiresAt:      c.expires_at ?? "",
    source,
  };
}
 
/** Codes the caller can still redeem — assigned to them (admin 'assigned' codes
 *  AND referral reward codes, which are promo_codes with assigned_to = the
 *  referrer) + all-customer codes in-window, not globally exhausted, and not
 *  exhausted by THIS customer (per_customer_limit via the redemption ledger).
 *  Assigned first, then shared. Empty for an anonymous / errored caller. */
export async function getRedeemableRewards(
  userId: string,
  supabase: SupabaseClient,
): Promise<RedeemableReward[]> {
  const [assignedRes, allRes] = await Promise.all([
    supabase
      .from("promo_codes")
      .select("id, code, title, type, value_sgd, percent, max_discount_sgd, min_order_sgd, max_uses, uses_count, used, starts_at, expires_at")
      .eq("assigned_to", userId),
    supabase
      .from("promo_codes")
      .select("id, code, title, type, value_sgd, percent, max_discount_sgd, min_order_sgd, per_customer_limit, max_uses, uses_count, starts_at, expires_at")
      .eq("audience", "all"),
  ]);
 
  const assigned = (assignedRes.data ?? []) as Row[];
  const allCodes = (allRes.data ?? []) as Row[];
 
  // My redemption counts for the all-customer codes (per_customer_limit gate).
  const allIds = allCodes.map((c) => c.id);
  const myCounts: Record<string, number> = {};
  if (allIds.length) {
    const { data: reds } = await supabase
      .from("promo_redemptions")
      .select("promo_code_id")
      .eq("customer_id", userId)
      .in("promo_code_id", allIds);
    for (const r of reds ?? []) {
      const id = (r as { promo_code_id: string }).promo_code_id;
      myCounts[id] = (myCounts[id] ?? 0) + 1;
    }
  }
 
  const now = Date.now();
  const inWindow = (c: Row) =>
    (!c.starts_at || Date.parse(c.starts_at) <= now) &&
    (!c.expires_at || Date.parse(c.expires_at) >= now);
  const notGloballyExhausted = (c: Row) => c.max_uses == null || (c.uses_count ?? 0) < c.max_uses;
 
  const assignedOut = assigned
    .filter((c) => inWindow(c) && notGloballyExhausted(c) && !c.used)
    .map((c) => shape(c, "assigned"));
 
  const allOut = allCodes
    .filter((c) => inWindow(c) && notGloballyExhausted(c) &&
      (c.per_customer_limit == null || (myCounts[c.id] ?? 0) < c.per_customer_limit))
    .map((c) => shape(c, "all"));
 
  return [...assignedOut, ...allOut];
}
