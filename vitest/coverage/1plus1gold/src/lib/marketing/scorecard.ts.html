/**
 * End-January discoverability scorecard (docs/marketing-scorecard-design.md §2).
 *
 * Targets agreed with Mukesh 2026-10-14. Every value is READ from data a cron
 * already fills — nothing on the panel is typed in by hand — so the scorecard
 * cannot drift from the work it measures.
 */
 
export const SCORECARD_TARGET_DATE = "2027-01-31";
 
export type ScorecardKey =
  | "linking_sites" | "directories" | "nonbrand_clicks" | "brand_clicks" | "ai_mentions" | "cost_per_order";
 
export const SCORECARD_TARGETS: Record<ScorecardKey, { label: string; baseline: number | string; target: number | string; unit?: string }> = {
  linking_sites:   { label: "Sites genuinely linking to us", baseline: 1,   target: 15 },
  directories:     { label: "Directory listings",            baseline: 0,   target: 6 },
  nonbrand_clicks: { label: "Non-brand Google clicks",        baseline: 0,   target: 100, unit: "/ 30 days" },
  brand_clicks:    { label: "Brand Google clicks",            baseline: 140, target: 250, unit: "/ 30 days" },
  ai_mentions:     { label: "AI answers mentioning us",       baseline: 7,   target: 40,  unit: "of 390" },
  cost_per_order:  { label: "Cost per order known",           baseline: "unknown", target: "known" },
};
 
/** Links that exist without an outreach row (pre-programme, found by the Ahrefs pass 2026-10-12). */
export const KNOWN_LINKING_DOMAINS = ["scamadviser.com"];
 
/** 13 weekly probe runs × 30 prompts = the agreed 390 denominator. */
export const AI_PROBE_WINDOW_RUNS = 13;
 
/**
 * Is a Search Console query a search for us by name? Normalises spacing and
 * punctuation so "1 plus 1 gold", "1plus1 gold" and "1+1 gold" all match.
 */
export function isBrandQuery(q: string | null | undefined): boolean {
  const s = (q ?? "").toLowerCase().replace(/[^a-z0-9+]/g, "");
  return /1plus1|1\+1gold|oneplusone/.test(s);
}
 
const domainOf = (d: string | null | undefined) =>
  (d ?? "").toLowerCase().replace(/^https?:\/\//, "").replace(/^www\./, "").split("/")[0];
 
export interface ScorecardProspect {
  domain: string; ask_type: string; status: string; link_check_status: string | null;
}
export interface ScorecardGscRow { query: string | null; clicks: number }
export interface ScorecardProbe { probe_date: string; ok: boolean; mentioned: boolean }
 
export interface ScorecardMetric {
  key: ScorecardKey; label: string; unit?: string;
  baseline: number | string; target: number | string;
  value: number | string;
  /** 0–1 progress from baseline to target; null where it is not numeric. */
  progress: number | null;
  source: string;
  note?: string;
}

export function buildScorecard(input: {
  prospects: ScorecardProspect[];
  gsc:       ScorecardGscRow[];   // last 30 days
  probes:    ScorecardProbe[];
  costPerOrderKnown: boolean;
  webCostPerOrder: number | null;
}): ScorecardMetric[] {
  const t = SCORECARD_TARGETS;
  const linked = (p: ScorecardProspect) => p.link_check_status === "linked" || p.status === "won";
 
  const linkingDomains = new Set(KNOWN_LINKING_DOMAINS);
  for (const p of input.prospects) {
    if (p.ask_type !== "directory_listing" && linked(p)) linkingDomains.add(domainOf(p.domain));
  }
  const directories = new Set(
    input.prospects.filter((p) => p.ask_type === "directory_listing" && linked(p)).map((p) => domainOf(p.domain)),
  ).size;
 
  let brand = 0, nonbrand = 0;
  for (const r of input.gsc) {
    if (isBrandQuery(r.query)) brand += r.clicks; else nonbrand += r.clicks;
  }
 
  const runDates = [...new Set(input.probes.map((p) => p.probe_date))].sort().reverse().slice(0, AI_PROBE_WINDOW_RUNS);
  const inWindow = input.probes.filter((p) => p.ok && runDates.includes(p.probe_date));
  const mentions = inWindow.filter((p) => p.mentioned).length;
 
  const n = (k: ScorecardKey) => t[k] as { baseline: number; target: number; label: string; unit?: string };
 
  return [
    { key: "linking_sites", ...n("linking_sites"), value: linkingDomains.size,
      progress: progress(linkingDomains.size, n("linking_sites").baseline, n("linking_sites").target),
      source: "Outreach link-check (daily) + known links" },
    { key: "directories", ...n("directories"), value: directories,
      progress: progress(directories, 0, n("directories").target),
      source: "Directory-listing prospects the link-check confirms live" },
    { key: "nonbrand_clicks", ...n("nonbrand_clicks"), value: nonbrand,
      progress: progress(nonbrand, 0, n("nonbrand_clicks").target),
      source: "Google Search Console (daily, ~3-day lag)",
      note: "Google hides rare searches, so this is a minimum." },
    { key: "brand_clicks", ...n("brand_clicks"), value: brand,
      progress: progress(brand, n("brand_clicks").baseline, n("brand_clicks").target),
      source: "Google Search Console (daily, ~3-day lag)" },
    { key: "ai_mentions", ...n("ai_mentions"), unit: `of ${inWindow.length}`, value: mentions,
      progress: progress(mentions, 0, n("ai_mentions").target),
      source: `Weekly AI answer check, last ${runDates.length} run${runDates.length === 1 ? "" : "s"}`,
      note: runDates.length < AI_PROBE_WINDOW_RUNS
        ? `Target is out of 390 (13 runs); ${runDates.length} runs so far.` : undefined },
    { key: "cost_per_order", label: t.cost_per_order.label, baseline: t.cost_per_order.baseline,
      target: t.cost_per_order.target,
      value: input.costPerOrderKnown
        ? (input.webCostPerOrder !== null ? `S$${input.webCostPerOrder.toFixed(0)} (web, last month)` : "known")
        : "unknown",
      progress: input.costPerOrderKnown ? 1 : 0,
      source: "Marketing spend in the books ÷ real orders" },
  ];
}
