/**
 * Google Veo (Gemini Developer API) image-to-video adapter — ASYNC.
 *
 * Veo renders over minutes, so it can't finish inside one serverless request.
 * The flow is two-phase:
 *   1. veoStartImageToVideo() → POST :predictLongRunning with the source image
 *      inline + a motion prompt → returns the operation name (a handle).
 *   2. veoPollOperation(name) → GET the operation; while running returns
 *      {done:false}; on completion returns {done:true, videoUrl} (or an error).
 * A poll cron drives phase 2 and ingests the finished MP4.
 *
 * The video URI Veo returns is itself on generativelanguage.googleapis.com and
 * must be downloaded WITH the API key — veoDownloadVideo() handles that.
 *
 * Env: GEMINI_API_KEY. The 'animate-veo' profile ships DISABLED until the exact
 * model slug + response shape are verified live (this is a preview API), so the
 * parsing here is deliberately defensive — it searches the response for a video
 * URI rather than assuming one fixed path.
 */

/** Download the finished video (Veo URIs require the API key). Returns bytes. */
export async function veoDownloadVideo(videoUrl: string): Promise<{ data: Buffer; contentType: string }> {
  const key = process.env.GEMINI_API_KEY;
  if (!key) throw new Error("GEMINI_API_KEY not set");
  // Google file URIs live on the same host; we send the key, so the host MUST
  // be Google — a public-host string check alone would let the key leak.
  assertHostAllowed(videoUrl, GOOGLE_API_HOSTS, "veoDownloadVideo");
  const res = await fetch(videoUrl, { headers: { "x-goog-api-key": key } });
  if (!res.ok) throw new Error(`Veo download failed: HTTP ${res.status}`);
  const contentType = res.headers.get("content-type") || "video/mp4";
  return { data: Buffer.from(await res.arrayBuffer()), contentType };
}
