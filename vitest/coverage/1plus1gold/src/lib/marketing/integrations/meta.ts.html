/**
 * Meta Graph API publishing (Instagram).
 *
 * Two-step publish (per Meta's documented flow):
 *   1. POST /me/media — create a media container with the asset URL +
 *      caption. Returns container ID.
 *   2. POST /me/media_publish — publish the container. Returns the IG
 *      post ID.
 *
 * Asset URL must be publicly accessible (Meta fetches it). For our
 * pipeline, we pass a Supabase Storage signed URL with ≥1h TTL — that's
 * plenty of time for Meta's fetch + publish flow.
 *
 * Env:
 *   META_IG_BUSINESS_ACCOUNT_ID  — IG Business Account ID (numeric)
 *   META_GRAPH_ACCESS_TOKEN      — long-lived Page access token (60-day TTL)
 */

  thumbnailUrl?: string;

/** Reply to an Instagram comment (the public conversational-engagement loop).
 *  POST /{comment-id}/replies. Returns the new reply's id. */
export async function replyToInstagramComment(commentId: string, message: string): Promise<{ id: string }> {
  const token = process.env.META_GRAPH_ACCESS_TOKEN;
  if (!token) throw new Error("META_GRAPH_ACCESS_TOKEN not set");
  const r = await fetch(`${META_GRAPH}/${commentId}/replies`, {
    method: "POST",
    headers: { Authorization: `Bearer ${token}`, "Content-Type": "application/json" },
    body: JSON.stringify({ message }),
  });
  if (!r.ok) {
    throw new Error(`IG comment reply failed: HTTP ${r.status} ${await r.text().catch(() => "")}`);
  }
  return await r.json() as { id: string };
}
