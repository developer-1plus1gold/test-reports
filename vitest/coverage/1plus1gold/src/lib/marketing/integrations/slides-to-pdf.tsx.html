/**
 * Compose ordered carousel slides into a single PDF for a LinkedIn document
 * ("carousel") post — the swipeable slide-deck format, the native way to show a
 * multi-slide piece on LinkedIn (a ugcPost carries only one image, so an 8-slide
 * carousel would otherwise post as slide 1 only).
 *
 * Reuses the project's existing PDF tooling: @react-pdf/renderer (renderToBuffer,
 * as the receipt/history PDFs do) + sharp (as image-optimize does). Slides are
 * WebP; react-pdf embeds JPEG/PNG only, so each is decoded + re-encoded to JPEG
 * via sharp first. The deck takes the FIRST slide's aspect (a designed carousel
 * is uniform); any odd-sized slide is padded to that canvas with the brand matte
 * so the deck reads consistently and nothing is cropped.
 */
 
import { renderToBuffer, Document, Page, Image } from "@react-pdf/renderer";
import sharp from "sharp";
import { isOwnSupabaseUrl } from "@/lib/safe-fetch-url";
 
const MATTE = "#16110A";      // brand background
const MAX_EDGE = 1600;        // bound the page (and PDF) size
const MAX_SLIDES = 20;        // LinkedIn document carousels cap ~20; bound memory (each slide is fetched + base64'd in RAM)
 
/**
 * @param slideUrls Ordered, fetchable URLs (signed storage URLs) for each slide.
 * @returns PDF bytes (ArrayBuffer), one full-bleed page per slide.
 */
export async function slidesToPdf(slideUrls: string[]): Promise<ArrayBuffer> {
  if (slideUrls.length === 0) throw new Error("slidesToPdf: no slides supplied");
  if (slideUrls.length > MAX_SLIDES) {
    throw new Error(`slidesToPdf: too many slides (${slideUrls.length} > ${MAX_SLIDES})`);
  }
 
  const buffers = await Promise.all(
    slideUrls.map(async (url) => {
      // SSRF guard: every slide URL is one of OUR Supabase signed URLs. Pin the
      // host to our own project rather than fetching whatever URL we're handed —
      // mirrors marketing storage ingestFromUrl. Blocks internal SSRF and
      // public-host exfil even if a crafted asset row reached this far.
      if (!isOwnSupabaseUrl(url)) throw new Error("slidesToPdf: refusing slide URL off our storage host");
      const r = await fetch(url);
      if (!r.ok) throw new Error(`slidesToPdf: slide fetch failed HTTP ${r.status}`);
      return Buffer.from(await r.arrayBuffer());
    }),
  );
 
  // Deck aspect = the first slide's (a designed carousel is uniform), capped so
  // the PDF stays a sensible size.
  const meta0 = await sharp(buffers[0]).metadata();
  let pageW = meta0.width ?? 1080;
  let pageH = meta0.height ?? 1350;
  const longEdge = Math.max(pageW, pageH);
  Iif (longEdge > MAX_EDGE) {
    const scale = MAX_EDGE / longEdge;
    pageW = Math.round(pageW * scale);
    pageH = Math.round(pageH * scale);
  }
 
  const dataUris = await Promise.all(
    buffers.map(async (buf) => {
      const jpeg = await sharp(buf)
        .resize(pageW, pageH, { fit: "contain", background: MATTE })
        .jpeg({ quality: 85 })
        .toBuffer();
      return `data:image/jpeg;base64,${jpeg.toString("base64")}`;
    }),
  );
 
  const doc = (
    <Document>
      {dataUris.map((uri, i) => (
        <Page key={i} size={[pageW, pageH]} style={{ backgroundColor: MATTE }}>
          {/* eslint-disable-next-line jsx-a11y/alt-text -- react-pdf Image has no alt */}
          <Image src={uri} style={{ width: pageW, height: pageH }} />
        </Page>
      ))}
    </Document>
  );
 
  const buf = await renderToBuffer(doc);
  // Return a standalone ArrayBuffer (Node Buffers view a pooled ArrayBuffer).
  return buf.buffer.slice(buf.byteOffset, buf.byteOffset + buf.byteLength) as ArrayBuffer;
}
