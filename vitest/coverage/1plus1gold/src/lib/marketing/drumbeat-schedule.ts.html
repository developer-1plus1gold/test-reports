/**
 * Drumbeat scheduling — the derived-date helpers.
 *
 * A content piece has exactly TWO operator-authored scheduling values:
 *   post_date    — which day it goes out (a calendar date)
 *   publish_time — what time, in Asia/Singapore
 *
 * Everything else (post_day, week_start, week_number, scheduled_at) is a STORED
 * GENERATED column on `marketing_content_pieces`. The functions here reproduce
 * those definitions for the ADMIN UI — so the operator can see the publish
 * moment update as they type, before saving.
 *
 * ⚠️ THESE MUST STAY IN LOCKSTEP WITH migration 20260912_drumbeat_derived_dates.
 * They are a second reader of the same rule, not a second source of truth: the
 * database computes the value that is actually used, and `drumbeat-schedule`
 * only previews it. `drumbeat-schedule.test.ts` pins each one against the exact
 * SQL expression it mirrors, so a change to either side fails at PR time.
 *
 * WHY THIS EXISTS AT ALL (2026-09-12): the table used to carry TWO independently
 * edited dates — `post_date` on the calendar list and `scheduled_at` on the
 * piece-detail page — and only `scheduled_at` drove publishing. A piece moved to
 * 15 Sep by editing `post_date` published on the 12th, because `scheduled_at`
 * was never touched. The fix is structural (one date, the rest generated); this
 * module is the half that makes the consequence visible while editing.
 */

export const SGT_OFFSET = "+08:00";

/** Parse "YYYY-MM-DD" as a UTC midnight Date. Returns null on anything else.
 *  UTC deliberately: a runner in a negative-offset zone would otherwise land on
 *  the previous day and shift every derived value (the 2026-09-09 bucket-key
 *  lesson — a date-only string must never be handed to a local-time parser). */
export function parsePostDate(postDate: string): Date | null {
  if (!/^\d{4}-\d{2}-\d{2}$/.test(postDate)) return null;
  const d = new Date(`${postDate}T00:00:00Z`);
  if (Number.isNaN(d.getTime())) return null;
  // Reject a well-formed but non-existent date (2026-02-30 parses as 03-02).
  if (d.toISOString().slice(0, 10) !== postDate) return null;
  return d;
}
 
/** ISO day-of-week: Mon=1 … Sun=7. Mirrors `extract(isodow from post_date)`. */
export function isoDayOfWeek(postDate: string): number | null {
  const d = parsePostDate(postDate);
  if (!d) return null;
  return d.getUTCDay() === 0 ? 7 : d.getUTCDay();
}
 
/** The weekday label. Mirrors the CASE over extract(isodow) in the migration —
 *  deliberately NOT `toLocaleDateString`, which is locale- and ICU-dependent
 *  (Node renders en-GB September as "Sept" where browsers render "Sep"). */
export function postDayLabel(postDate: string): string | null {
  const dow = isoDayOfWeek(postDate);
  return dow === null ? null : DAY_LABELS[dow - 1];
}

/** "HH:MM" (or "HH:MM:SS" as Postgres returns it) → "HH:MM"; null if unusable. */
export function normalisePublishTime(value: unknown): string | null {
  if (typeof value !== "string") return null;
  const m = /^(\d{2}):(\d{2})(?::\d{2})?$/.exec(value.trim());
  if (!m) return null;
  const h = Number(m[1]), min = Number(m[2]);
  if (h > 23 || min > 59) return null;
  return `${m[1]}:${m[2]}`;
}

/**
 * The one line the operator reads before saving: "Tue 15 Sep 2026, 07:30 SGT".
 *
 * Built from the string parts rather than from a Date, so it says the same
 * thing in every timezone the admin is opened in — the publish moment is a
 * Singapore wall-clock fact and rendering it in the viewer's zone is what made
 * the two dates look unrelated in the first place.
 */
export function publishMomentLabel(postDate: string, publishTime: string): string | null {
  const time = normalisePublishTime(publishTime);
  const day = postDayLabel(postDate);
  if (!day || !time) return null;
  const [y, m, d] = postDate.split("-");
  return `${day} ${Number(d)} ${MONTH_LABELS[Number(m) - 1]} ${y}, ${time} SGT`;
}

/**
 * Group pieces into weeks, LATEST FIRST — both between weeks and WITHIN one.
 *
 * Sorts on `week_start` (a real date) rather than `week_number`, for the
 * year-rollover reason on `isoWeekStart`.
 *
 * ⚠️ The within-week order is DESCENDING too (operator's call, 2026-09-12). The
 * first cut kept it ascending on the reasoning that "a week should still read as
 * a week" — that was a decision made on the operator's behalf, and he reversed
 * it: newest-first the whole way down is what he wants, so the most recent piece
 * is at the top of the list wherever you are in it. Do not "restore" ascending.
 */
export function groupByWeekLatestFirst<T extends { week_start: string; post_date: string }>(
  pieces: T[],
): Array<{ weekStart: string; pieces: T[] }> {
  const map = new Map<string, T[]>();
  for (const p of pieces) {
    const arr = map.get(p.week_start) ?? [];
    arr.push(p);
    map.set(p.week_start, arr);
  }
  return [...map.entries()]
    .sort(([a], [b]) => (a < b ? 1 : a > b ? -1 : 0))
    .map(([weekStart, group]) => ({
      weekStart,
      pieces: [...group].sort((x, y) => (x.post_date < y.post_date ? 1 : x.post_date > y.post_date ? -1 : 0)),
    }));
}
