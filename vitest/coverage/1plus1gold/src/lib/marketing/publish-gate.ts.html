/**
 * Brand sign-off gate for the marketing publisher.
 *
 * The publisher may only push a piece live once it has been vetted. There are
 * two equivalent vetting signals, and EITHER satisfies the gate:
 *
 *   1. Operator approval — every asset actually being published is
 *      operator-approved (status `approved` or already `published_to_channel`).
 *      An operator clicking Approve on each asset IS the human sign-off this
 *      gate exists to ensure, so it stands on its own.
 *   2. AI brand review — the latest marketing_brand_reviews row is `passed`
 *      or `override_granted` (handled in the route; the AI review is an
 *      optional assist for un-approved/raw assets).
 *
 * A raw `generated` asset that was never approved or reviewed is genuinely
 * un-vetted and still blocks publish.
 */
 
/**
 * Whether a piece's lifecycle status permits a publish call. A normal publish
 * needs `scheduled`/`brand_reviewed`. A `replace`-scoped REPOST is a deliberate
 * re-push of specific channels and is expected on an already-`published` piece
 * (e.g. re-pushing LinkedIn now that it carries an image), so `published` is
 * allowed when reposting. `force` bypasses the check entirely.
 */
export function canPublishStatus(status: string, isRepost: boolean, force: boolean): boolean {
  if (force) return true;
  if (status === "scheduled" || status === "brand_reviewed") return true;
  if (isRepost && status === "published") return true;
  return false;
}
 
/**
 * The channels a publish call actually targets: the explicit `channels` subset
 * and/or the `replace` (repost) scope, intersected with the piece's plan;
 * neither given → every target channel (a full publish).
 *
 * Every approval gate must scope to THIS set. The bug this fixes: a per-channel
 * publish (`channels: ["instagram"]`) was rejected because a DIFFERENT target
 * channel's caption/asset wasn't approved yet — the route iterated all
 * target_channels instead of just the ones being published.
 */
export function scopePublishChannels(
  targetChannels: string[],
  channels?: string[] | null,
  replace?: string[] | null,
): string[] {
  const requested = new Set(
    [...(channels ?? []), ...(replace ?? [])].filter((c): c is string => typeof c === "string"),
  );
  return requested.size > 0 ? targetChannels.filter((c) => requested.has(c)) : targetChannels;
}
 
/** Asset statuses that represent an explicit operator sign-off. */
export const OPERATOR_APPROVED_ASSET_STATUSES = ["approved", "published_to_channel"] as const;
 
/**
 * True when the assets being published alone satisfy the brand sign-off —
 * i.e. there is at least one and every one is operator-approved. When false,
 * the caller must fall back to the AI brand-review gate.
 */
export function assetsAreOperatorApproved(statuses: Array<string | null | undefined>): boolean {
  return statuses.length > 0 &&
    statuses.every((s) => (OPERATOR_APPROVED_ASSET_STATUSES as readonly string[]).includes(s ?? ""));
}
