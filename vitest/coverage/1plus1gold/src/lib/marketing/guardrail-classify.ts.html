/**
 * Deterministic classification of an operator rejection reason into a brand
 * guardrail's (scope, kind). Pure code — NOT an LLM agent.
 *
 * Design (per docs/agent-contract.md §0.1 "default to extend, don't spawn an
 * agent for a small job"): reject-time classification is a deterministic
 * heuristic so a reject never blocks on inference, and the EXISTING
 * marketing_guardrail_curator re-classifies/normalizes any heuristic miss on
 * its daily pass (its `rewrite` action already re-types kind+scope). The
 * heuristic captures the high-signal split: phrasing that asks for something to
 * be present/added → positive_directive; everything else → negative_exclusion.
 */
 
// Vocabulary lives in ONE place (curation-action.ts). This module is the upstream
// EMITTER of scopes, so a local copy drifting narrower than the writers' allow-lists
// is exactly how a scoped rule got silently clamped to 'all' before 2026-08-27.
import type { GuardrailKindValue, GuardrailScopeValue } from "./curation-action";
 
export type GuardrailKind = GuardrailKindValue;
export type GuardrailScope = GuardrailScopeValue;
 
export interface GuardrailClassification {
  scope:     GuardrailScope;
  kind:      GuardrailKind;
  rule_text: string;
  rationale: string;
}
 
export interface ClassifyInput {
  reason: string;
  /** caption → copy rule (scope 'caption'); asset → image-op rule (scope 'all');
   *  response → reply copy rule (scope 'response'); brief → brief-spec rule
   *  (scope 'brief'). */
  source: "caption" | "asset" | "response" | "brief";
}
 
export function classifyRejection(input: ClassifyInput): GuardrailClassification {
  const reason = input.reason.trim();
  const lower = reason.toLowerCase();
  // "X is missing", "include X", "add X", "always/ensure/must have X" → ADD-this.
  const positive =
    /\b(missing|include|add|always|ensure|must (?:have|include|show)|should (?:have|include|show)|needs? (?:to|a)|don'?t forget|remember to)\b/.test(lower);
  const kind: GuardrailKind = positive ? "positive_directive" : "negative_exclusion";
  return {
    scope: input.source === "caption" ? "caption" : input.source === "response" ? "response" : input.source === "brief" ? "brief" : "all",
    kind,
    rule_text: reason,
    rationale: `Auto-classified from an operator rejection (${input.source}); curator refines on its daily pass.`,
  };
}
