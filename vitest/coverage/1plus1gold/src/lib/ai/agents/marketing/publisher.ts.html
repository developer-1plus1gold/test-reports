/**
 * MarketingPublisher — pushes operator-approved pieces to channels.
 *
 * Channel handlers:
 *   - instagram      → Meta Graph (Feed posts, Reels, Stories)
 *   - linkedin       → REST /ugcPosts (company page)
 *   - whatsapp_status → no API; prepare asset + caption + send admin
 *                       WhatsApp alert with the deep-link payload
 *
 * Gating: refuses to publish unless
 *   - marketing_content_pieces.status = 'scheduled'
 *   - marketing_brand_reviews.status IN ('passed', 'override_granted')
 *   - asset_id resolves + storage path is readable
 *
 * agent_type=automation in the registry (no LLM call in the publish
 * happy path). Inherits BaseAgent for the audit-log plumbing.
 */

export interface PublisherAssetRef {
  id:           string;
  storagePath:  string;
  mimeType:     string;
  kind:         string;          // generated_image | composed_reel | feature_capture | ...
}
 
export interface PublisherInput {
  piece:          ContentPieceRow;
  /** The approved caption variant body — channel-specific text + alt-text. */
  captionsByChannel: Record<Channel, { text: string; alt_text?: string }>;
  /** Resolved media per channel, in the operator's assigned order. Instagram
   *  composes a carousel from the image list when 2+ images are assigned (else
   *  a single image; a video-led list is a Reel). The route fills every target
   *  channel — an explicit assignment, else the most-recent-approved fallback. */
  assetsByChannel: Record<Channel, PublisherAssetRef[]>;
  /** Hashtags (optional, channel-specific). */
  hashtagsByChannel?: Record<Channel, string[]>;
  /** Channel → existing platform_post_ids entry. A channel that already has an
   *  entry is SKIPPED (idempotent — a cron re-dispatch of a still-`scheduled`
   *  piece must not re-post IG/LinkedIn or re-prepare WhatsApp) unless it's in
   *  replaceChannels. */
  alreadyPosted?: Record<string, string>;
  /** Channels to force-republish even if alreadyPosted has them. For an
   *  existing real post we DELETE it first where the platform supports it
   *  (LinkedIn) so we don't leave a duplicate. */
  replaceChannels?: string[];
  /** Restrict this publish to a subset of the piece's target channels. When
   *  set, ONLY these channels are published (intersected with target_channels);
   *  every other channel is left untouched. Omitted/empty → publish all target
   *  channels (the "Publish all" path). This is what enables publishing one
   *  channel at a time on the FIRST publish, not just via idempotency-skip on
   *  subsequent runs. */
  onlyChannels?: string[];
}
 
export interface PublisherChannelResult {
  channel:         Channel;
  status:          "published" | "manual_prepared" | "failed" | "skipped";
  platformPostId?: string;
  permalink?:      string;
  manualPayload?:  { caption: string; assetUrl: string };   // whatsapp_status
  error?:          string;
}
 
export interface PublisherOutput {
  pieceId:        string;
  perChannel:     PublisherChannelResult[];
}
