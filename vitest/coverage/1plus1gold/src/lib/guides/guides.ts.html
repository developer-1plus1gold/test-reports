/**
 * /guides — answer pages for the questions people ask search engines and AI
 * assistants about buying gold in Singapore (2026-10-11, discoverability
 * option 1).
 *
 * WHY THIS FILE IS BUILT THE WAY IT IS
 *  - The Q&A blocks are NOT new copy. Each guide references existing questions
 *    in lib/faq-data.ts by their exact text, so a guide can never state a fact
 *    the FAQ doesn't — and when a policy changes, editing the FAQ updates every
 *    guide, the FAQ page, the FAQPage JSON-LD, llms.txt and the iOS FAQ at once.
 *    `guideFaqs()` THROWS on an unknown question, so a reworded FAQ breaks the
 *    build (guides.test.ts) instead of silently dropping an answer.
 *  - The only new prose is each guide's title, description and short intro.
 *    Keep the intros to restatements of facts already published elsewhere in
 *    the app — never introduce a number, a rate or a promise here.
 *  - Live numbers (spot, bar prices by size) are rendered by the page from the
 *    pricing snapshot + the shop pricing kernel, never written into this file.
 */
 
import { FAQ_CATEGORIES, type FaqItem } from "@/lib/faq-data";
 
export interface GuideLink {
  href: string;
  label: string;
}
 
export interface Guide {
  slug: string;
  /** <h1> + <title> stem. Phrased as the question people actually search. */
  title: string;
  /** Meta description, ≤155 chars (Google + WhatsApp preview truncation). */
  description: string;
  /** Short plain-text intro paragraphs. Facts only; no new numbers. */
  intro: string[];
  /** Exact question text from FAQ_CATEGORIES, in display order. */
  faqQuestions: string[];
  /** Render today's live spot price (SGD + USD, per gram + per troy ounce). */
  showLiveSpot?: boolean;
  /** Render the live price-by-size table (lib/guides/price-today), optionally
   *  narrowed to one refinery. Prices come from the shop pricing kernel. */
  priceTable?: { refinery?: string; caption: string };
  /** Render the premium-vs-spread explainer (we quote premium, never spread). */
  explainSpread?: boolean;
  related: GuideLink[];
}

const FAQ_BY_QUESTION: ReadonlyMap<string, FaqItem> = new Map(
  FAQ_CATEGORIES.flatMap((c) => c.items).map((it) => [it.q, it]),
);
 
/** The FAQ items a guide references. Throws on an unknown question so a
 *  reworded FAQ fails the build rather than silently dropping an answer. */
export function guideFaqs(guide: Guide): FaqItem[] {
  return guide.faqQuestions.map((q) => {
    const item = FAQ_BY_QUESTION.get(q);
    if (!item) throw new Error(`guide "${guide.slug}" references unknown FAQ question: ${q}`);
    return item;
  });
}
 
export function getGuide(slug: string): Guide | undefined {
  return GUIDES.find((g) => g.slug === slug);
}
 
/** schema.org FAQPage for one guide — same plain-text answers as the page. */
export function guideFaqJsonLd(guide: Guide): Record<string, unknown> {
  return {
    "@context": "https://schema.org",
    "@type": "FAQPage",
    mainEntity: guideFaqs(guide).map((it) => ({
      "@type": "Question",
      name: it.q,
      acceptedAnswer: { "@type": "Answer", text: it.a },
    })),
  };
}
 
/** schema.org BreadcrumbList: Home › Guides › <guide>. */
export function guideBreadcrumbJsonLd(guide: Guide, siteUrl: string): Record<string, unknown> {
  return {
    "@context": "https://schema.org",
    "@type": "BreadcrumbList",
    itemListElement: [
      { "@type": "ListItem", position: 1, name: "Home", item: `${siteUrl}/` },
      { "@type": "ListItem", position: 2, name: "Guides", item: `${siteUrl}/guides` },
      { "@type": "ListItem", position: 3, name: guide.title, item: `${siteUrl}/guides/${guide.slug}` },
    ],
  };
}
