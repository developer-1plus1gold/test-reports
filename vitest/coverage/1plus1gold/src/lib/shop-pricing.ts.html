function deriveStatus(p: ShopPriceable): { soldOut: boolean; presale: boolean; lowStock: boolean; inStock: boolean } {
  const presale   = p.status === "presale";
  const stockLeft = p.stock_qty !== null ? Math.max(0, p.stock_qty) : null;
  const soldOut   = p.status === "sold_out" || (stockLeft !== null && stockLeft <= 0 && !presale);
  const lowStock  = !soldOut && stockLeft !== null && stockLeft > 0 &&
                    p.low_stock_threshold !== null && stockLeft <= p.low_stock_threshold;
  // Tracked + countable + healthy: shows "{stock_qty} in stock". Untracked
  // (stockLeft === null) products show nothing (made-to-order / unlimited).
  const inStock   = !soldOut && !presale && !lowStock && stockLeft !== null && stockLeft > 0;
  return { soldOut, presale, lowStock, inStock };
}

/**
 * Price every product in the catalog. Mirrors the web cards' per-type logic:
 *   curve     → spot × g × (1 + getPremiumForWeight(g)); floor = best combibar rate
 *   standard  → spot × g × (1 + getTierPremium(1, tiers)); tier-range for the badge
 *   accessory → fixed price_sgd
 * Returns each product with `.pricing` attached; the input type is preserved.
 */
export function priceShopCatalog<T extends ShopPriceable>(
  catalog: T[],
  spotSgd: number | null,
  premiums: PremiumAnchor[],
): PricedProduct<T>[] {
  // decimal premium per anchor weight (mirrors toPremiumMap: pct/100).
  const premiumMap = new Map<number, number>(premiums.map((r) => [r.g, r.premium_pct / 100]));
  const floorPremiumPct = premiumMap.size > 0 ? Math.min(...premiumMap.values()) * 100 : null;
 
  return catalog.map((p) => {
    const status = deriveStatus(p);
    const isGift = /gift/i.test(p.label);
    const sortedTiers = p.standard_premium_tiers && p.standard_premium_tiers.length > 0
      ? [...p.standard_premium_tiers].sort((a, b) => a.min_qty - b.min_qty) : null;
    let pricing: ShopPricing = {
      priceSgd: null, premiumRate: null, premiumPct: null,
      floorPremiumPct: null, tierFirstPct: null, tierLastPct: null,
      premiumLabel: null, floorPremiumLabel: null, tierFirstLabel: null, tierLastLabel: null,
      minQty: 1, ...status,
      spotSubtotalSgd: spotSgd !== null && p.product_type !== "accessory" ? spotSgd * p.grams : null,
      isSplit: p.product_type === "curve" && p.grams <= 10,
      tierRows: sortedTiers
        ? sortedTiers.map((t) => ({
            rangeLabel: t.max_qty != null ? `${t.min_qty}–${t.max_qty} units` : `${t.min_qty}+ units`,
            premiumLabel: formatPremium(t.premium_pct / 100),
          }))
        : [],
      atMarketMedian: p.product_type === "standard" && !sortedTiers,
      bestFor:  p.product_type === "accessory" ? (isGift ? "Gifting 1g splits" : "Storing 1g splits") : null,
      category: p.product_type === "accessory" ? (isGift ? "Gift-Ready" : "Storage") : null,
    };
 
    if (p.product_type === "curve") {
      const rate = premiumMap.size > 0 ? getPremiumForWeight(p.grams, premiumMap) : null;
      pricing = {
        ...pricing,
        premiumRate: rate,
        premiumPct: rate !== null ? rate * 100 : null,
        priceSgd: spotSgd !== null && rate !== null ? spotSgd * p.grams * (1 + rate) : null,
        // splits (≤10g) show the floor they converge toward; combibars don't.
        floorPremiumPct: p.grams <= 10 ? floorPremiumPct : null,
      };
    } else if (p.product_type === "standard") {
      const rate = sortedTiers ? getTierPremium(1, sortedTiers) : null;
      pricing = {
        ...pricing,
        premiumRate: rate,
        premiumPct: rate !== null ? rate * 100 : null,
        priceSgd: spotSgd !== null && rate !== null ? spotSgd * p.grams * (1 + rate) : null,
        tierFirstPct: sortedTiers ? sortedTiers[0].premium_pct : null,
        tierLastPct: sortedTiers && sortedTiers.length > 1 ? sortedTiers[sortedTiers.length - 1].premium_pct : null,
        minQty: minQtyForTiers(p.standard_premium_tiers),
      };
    } else {
      // accessory — fixed price.
      pricing = { ...pricing, priceSgd: p.price_sgd };
    }
 
    // PWA-formatted labels (formatPremium takes a DECIMAL → "X.XX%"). premiumRate
    // is already decimal; the *Pct fields are percentages, so /100 first.
    pricing = {
      ...pricing,
      premiumLabel:      pricing.premiumRate    !== null ? formatPremium(pricing.premiumRate)        : null,
      floorPremiumLabel: pricing.floorPremiumPct !== null ? formatPremium(pricing.floorPremiumPct / 100) : null,
      tierFirstLabel:    pricing.tierFirstPct   !== null ? formatPremium(pricing.tierFirstPct / 100)  : null,
      tierLastLabel:     pricing.tierLastPct    !== null ? formatPremium(pricing.tierLastPct / 100)   : null,
    };
 
    // tierRows is ALSO exposed at the product top level so an SDUI `repeater`
    // (which resolves only single-level `item.<key>`, not `item.pricing.x`) can
    // render the "Premium tiers" list. The rest stays under `pricing` (accessed
    // via deep interpolation, which DOES support `item.pricing.x`).
    return { ...p, pricing, tierRows: pricing.tierRows, detailChips: buildDetailChips(p, pricing) } as PricedProduct<T>;
  });
}

/** Qty-aware pricing for ONE product at a chosen quantity — the SINGLE source
 *  both the web shop's qty editor AND the iOS on-device qty node consume. The
 *  unit price is qty-DEPENDENT for standard (the tier crossed by `qty`); curve +
 *  accessory unit prices are qty-independent (the line just multiplies). */
export interface ShopLine {
  qty: number;
  priceEach: number | null;   // unit price AT this qty (null ⇒ no live price → "Enquire")
  lineSgd: number | null;     // priceEach × qty
  premiumPct: number | null;  // applied premium as a percentage (for the subline)
  premiumLabel: string | null; // formatPremium(rate) — "23.00%" (matches the web)
}
 
/**
 * Price ONE product at `qty`. Mirrors the web rows' per-type pricing EXACTLY:
 *   curve     → spot × g × (1 + getPremiumForWeight(g)); line = each × qty (qty-indep)
 *   standard  → computeStandardLineTotal(qty, …): each = spot × g × (1 + tier(qty))
 *   accessory → fixed price_sgd; line = each × qty
 * Used by `engine.shopLine` (iOS) and the web qty editors so the line value is
 * byte-identical across platforms (no parallel implementation — the CHARGE
 * stays server-authoritative). Contract §23.
 */
export function shopLineForProduct(
  p: ShopPriceable,
  qty: number,
  spotSgd: number | null,
  premiums: PremiumAnchor[],
  /** The signed-in customer's accumulated curve grams (paid history, mod 50g
   *  applied inside computeOrderTotal). Pass 0 for guests. Makes the curve line
   *  quote the price for the customer's ACTUAL curve position — i.e. what the
   *  cart will charge them — so the shop matches the logged-in user. */
  accumulatedCurveGrams = 0,
): ShopLine {
  const n = Math.max(1, Math.floor(Number(qty) || 1));
 
  if (p.product_type === "accessory") {
    const each = p.price_sgd;
    return { qty: n, priceEach: each, lineSgd: each != null ? each * n : null, premiumPct: null, premiumLabel: null };
  }
 
  if (p.product_type === "curve") {
    const premiumMap = new Map<number, number>(premiums.map((r) => [r.g, r.premium_pct / 100]));
    if (spotSgd === null || premiumMap.size === 0) {
      return { qty: n, priceEach: null, lineSgd: null, premiumPct: null, premiumLabel: null };
    }
    // Follow the premium curve as qty accumulates: qty × grams move DOWN the
    // curve toward the floor, anchored at the customer's accumulated position.
    // This is the SAME primitive the cart prices curve with (computeOrderTotal),
    // so the shop line matches the cart (and the eventual charge) instead of
    // flat-multiplying the single-piece rate. Standalone-line estimate (this
    // line's grams + the customer's history); the cart blends multiple lines and
    // stays server-authoritative. Contract §23.
    const totalGrams = p.grams * n;
    const { totalSgd, curveRate } = computeOrderTotal(totalGrams, spotSgd, premiumMap, accumulatedCurveGrams);
    return {
      qty: n,
      priceEach: totalSgd / n,    // blended per-unit at the accumulated curve rate
      lineSgd: totalSgd,
      premiumPct: curveRate * 100,
      premiumLabel: formatPremium(curveRate),
    };
  }
 
  // standard — tier premium is a function of qty.
  const tiers = p.standard_premium_tiers;
  if (spotSgd !== null && tiers && tiers.length > 0) {
    const b = computeStandardLineTotal(n, p.grams, spotSgd, tiers);
    return { qty: n, priceEach: b.priceEach, lineSgd: b.lineSgd, premiumPct: b.premiumRate * 100, premiumLabel: formatPremium(b.premiumRate) };
  }
  return { qty: n, priceEach: null, lineSgd: null, premiumPct: null, premiumLabel: null };
}

    switch (key) {
      case "weight-asc":  return (a.grams - b.grams) || byRefinery(a, b) || bySortOrder(a, b);
      case "weight-desc": return (b.grams - a.grams) || byRefinery(a, b) || bySortOrder(a, b);
      case "refinery":    return byRefinery(a, b) || (a.grams - b.grams) || bySortOrder(a, b);
      // "default" = the admin's curated order (products.sort_order). Explicit so
      // it's correct as the SHOP DEFAULT sort (was `return 0`). 2026-06-12.
      default:            return bySortOrder(a, b) || (a.grams - b.grams);
    }
  };
}
 
/**
 * Filter → sort → group the already-priced STANDARD products per the section's
 * controls. Returns groups: a single ""-keyed group when not grouping, else one
 * per refinery ("Other" last). Mirrors the web shop's useMemo pipeline exactly.
 */
export function shopStandardView<T extends ShopPriceable>(
  standard: PricedProduct<T>[],
  state: ShopViewState,
): ShopGroup<T>[] {
  const grams = new Set(state.gramsFilter ?? []);
  const refineries = new Set(state.refineryFilter ?? []);
 
  const filtered = standard.filter((p) => {
    if (!state.showSoldOut && p.status === "sold_out") return false;
    if (grams.size > 0 && !grams.has(String(p.grams))) return false;
    if (refineries.size > 0 && !(p.refinery && refineries.has(p.refinery))) return false;
    return true;
  });
 
  const sorted = [...filtered].sort(sortComparator<T>(state.sort ?? "default"));
 
  if (!state.groupByRefinery) return [{ key: "", products: sorted }];
 
  const groups = new Map<string, PricedProduct<T>[]>();
  for (const p of sorted) {
    const k = (p.refinery && p.refinery.trim()) || "Other";
    (groups.get(k) ?? groups.set(k, []).get(k)!).push(p);
  }
  return [...groups.keys()]
    .sort((a, b) => (a === "Other" ? 1 : b === "Other" ? -1 : a.localeCompare(b)))
    .map((key) => ({ key, products: groups.get(key)! }));
}
 
export interface ShopFilterOption { value: string; label: string; }
 
/**
 * Distinct weight + refinery filter options for the Standard section's chips,
 * derived from the catalog (mirrors the web shop: grams asc with weight_display
 * labels; refineries A–Z). Shared so web + iOS surface the same chips.
 */
export function shopFilterOptions<T extends ShopPriceable & { weight_display?: string | null }>(
  standard: PricedProduct<T>[],
): { gramsOptions: ShopFilterOption[]; refineryOptions: ShopFilterOption[] } {
  const gramsMap = new Map<number, string>();
  for (const p of standard) {
    if (!gramsMap.has(p.grams)) {
      const wd = p.weight_display && p.weight_display.trim() ? p.weight_display : `${p.grams}g`;
      gramsMap.set(p.grams, wd);
    }
  }
  const gramsOptions = [...gramsMap.entries()]
    .sort((a, b) => a[0] - b[0])
    .map(([g, label]) => ({ value: String(g), label }));
 
  const refineryOptions = [...new Set(
    standard.map((p) => p.refinery).filter((r): r is string => !!r && r.trim().length > 0),
  )].sort((a, b) => a.localeCompare(b)).map((r) => ({ value: r, label: r }));
 
  return { gramsOptions, refineryOptions };
}
 
/** Partition a priced catalog into the shop's render sections. */
export function partitionShop<T extends ShopPriceable>(priced: PricedProduct<T>[]) {
  const curve = priced.filter((p) => p.product_type === "curve");
  return {
    curveSplits:    curve.filter((p) => p.grams <= 10),
    curveCombibars: curve.filter((p) => p.grams > 10),
    standard:       priced.filter((p) => p.product_type === "standard"),
    accessories:    priced.filter((p) => p.product_type === "accessory"),
  };
}
