/**
 * Airwallex client — hosted Checkout Session only (2026-11-07).
 *
 * Built from Airwallex's official docs (docs/airwallex-integration-design.md,
 * D1: Checkout Session). We never put card data on our pages or in the app:
 * the server creates a Checkout Session and the customer is redirected to its
 * `url` — the same pattern as HitPay and Razorpay. Ships switched off: the
 * airwallex_* methods are seeded is_active=false and nothing calls Airwallex
 * until the credentials below are set.
 *
 *   Auth      POST /api/v1/authentication/login (x-client-id + x-api-key) →
 *             a Bearer token valid 30 min. Cached per instance, refreshed a
 *             little early, and once more on a 401 (docs: reuse until expiry).
 *   Session   POST /api/v1/checkout/checkout_sessions/create → { id, url } (a
 *             "beta" API needing version ≥ 2026-05-29, sent as x-api-version).
 *             The PaymentIntent exists only once the customer completes; read
 *             it from GET …/checkout_sessions/{id}.payment_intent_id.
 *   Webhook   x-signature = hex HMAC-SHA256(x-timestamp + raw body, secret).
 *
 * Env (Vercel DEV = sandbox, PROD = production; never committed):
 *   AIRWALLEX_ENV            "production" for live; anything else → sandbox
 *   AIRWALLEX_CLIENT_ID      Developer → API keys (Checkout Session read+write)
 *   AIRWALLEX_API_KEY
 *   AIRWALLEX_ACCOUNT_ID     Settings → Account details (required by sessions)
 *   AIRWALLEX_WEBHOOK_SECRET the webhook subscription's secret
 *   AIRWALLEX_API_VERSION    optional, default 2026-05-29
 */

export function airwallexSessionTypes(methodKey: string): string[] {
  const type = AIRWALLEX_METHOD_MAP[methodKey];
  if (!type) return [];
  return type === "card" ? [type, ...AIRWALLEX_CARD_WALLETS] : [type];
}

/** The Create a Checkout Session body for one order. Pure — tested directly. */
export function buildCheckoutSessionBody(args: {
  accountId:       string;
  referenceNumber: string;
  amountSgd:       number;
  methodKey:       string;
  description:     string;
  successUrl:      string;
  backUrl:         string;
  email?:          string;
  name?:           string;
  cardNetworks?:   readonly string[] | null;
}): Record<string, unknown> {
  const type = AIRWALLEX_METHOD_MAP[args.methodKey];
  if (!type) throw new Error(`airwallex: unknown method ${args.methodKey}`);
  if (!Number.isFinite(args.amountSgd) || args.amountSgd <= 0) throw new Error("airwallex: amount must be positive");
  const networks = type === "card" && args.cardNetworks
    ? args.cardNetworks.map((n) => n.toLowerCase()).filter((n) => AIRWALLEX_CARD_NETWORKS.has(n))
    : [];
  const customer: Record<string, string> = { country_code: "SG" };
  if (args.email) customer.email = args.email.slice(0, 120);
  if (args.name) customer.name = args.name.slice(0, 100);
  return {
    account_id: args.accountId,
    mode:       "PAYMENT",
    ui_mode:    "HOSTED",
    // Idempotency: a retried create with the same reference returns the same session.
    request_id: args.referenceNumber,
    currency:   "SGD",
    line_items: [{
      quantity: 1,
      price: { currency: "SGD", unit_amount: Math.round(args.amountSgd * 100) / 100, product: { name: args.description.slice(0, 200) } },
    }],
    payment_intent_data: { merchant_order_id: args.referenceNumber },
    // Only the method the customer chose (and priced) at checkout is offered.
    payment_options: {
      payment_method_types: airwallexSessionTypes(args.methodKey),
      auto_capture: true,
      ...(networks.length ? { allowed_card_networks: networks } : {}),
    },
    customer_data: customer,
    // Back to our confirming page straight away (it polls the server fact).
    success_url: args.successUrl,
    back_url:    args.backUrl,
    // No hosted_completion_page: Airwallex refuses it (400) unless the session
    // creates an invoice or saves a payment source. A plain PAYMENT session
    // redirects to success_url on completion anyway (DEV sandbox, 2026-11-07).
    metadata: { reference_number: args.referenceNumber },
  };
}

/**
 * Our processing cost for a payment, in S$. Airwallex puts the fee on its
 * financial-transaction ledger, not on the PaymentIntent; the row may appear
 * after the payment, so this returns null when it is not there yet.
 */
export async function paymentFeeSgd(sourceId: string): Promise<number | null> {
  const r = await airwallexRequest<{ items?: Array<{ fee?: number | null; currency?: string | null; status?: string | null }> }>(
    "GET", "/api/v1/financial_transactions", { query: { source_id: sourceId } },
  );
  const fees = (r.items ?? []).filter((t) =>
    (t.currency ?? "SGD") === "SGD" && typeof t.fee === "number" && Number.isFinite(t.fee)
    && String(t.status ?? "").toUpperCase() !== "CANCELLED");
  if (!fees.length) return null;
  const total = Math.round(fees.reduce((sum, t) => sum + Math.abs(t.fee as number), 0) * 100) / 100;
  // A ledger row can appear PENDING before its fee is set; 0 there means "not
  // known yet", not "free" — recordAirwallexFee never overwrites, so storing it
  // would lock in a wrong 0. Settled rows are taken at face value.
  const settled = fees.some((t) => String(t.status ?? "").toUpperCase() === "SETTLED");
  return total === 0 && !settled ? null : total;
}
 
/**
 * True when the paid method differs from the one priced at checkout. The
 * session is restricted to one method, so this is a should-never-happen
 * check: the order still completes (they paid), an admin is alerted.
 */
export function paymentMethodMismatch(chosenKey: string, paidType: string | null | undefined): boolean {
  const allowed = airwallexSessionTypes(chosenKey);
  return allowed.length > 0 && !!paidType && !allowed.includes(paidType);
}

function safeEqualHex(a: string, b: string): boolean {
  const ab = Buffer.from(a, "utf8");
  const bb = Buffer.from(b, "utf8");
  return ab.length === bb.length && crypto.timingSafeEqual(ab, bb);
}
 
/**
 * Airwallex webhook: x-signature = hex HMAC-SHA256(x-timestamp + raw body,
 * subscription secret); x-timestamp in ms, outside WEBHOOK_TOLERANCE_MS of now
 * is refused. Fails closed (no secret / no headers → false).
 */
export function verifyAirwallexWebhook(
  rawBody: string,
  timestamp: string | null | undefined,
  signature: string | null | undefined,
  nowMs: number = Date.now(),
): boolean {
  const secret = env("AIRWALLEX_WEBHOOK_SECRET");
  if (!secret || !timestamp || !signature) return false;
  const ts = Number(timestamp);
  if (!Number.isFinite(ts) || Math.abs(nowMs - ts) > WEBHOOK_TOLERANCE_MS) return false;
  const expected = crypto.createHmac("sha256", secret).update(`${timestamp}${rawBody}`).digest("hex");
  return safeEqualHex(expected, signature.trim().toLowerCase());
}
