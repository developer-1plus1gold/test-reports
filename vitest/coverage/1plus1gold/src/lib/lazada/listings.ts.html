/**
 * Lazada listing ↔ our product mapping — PURE (step 2a,
 * docs/lazada-integration-design.md). Steps 2b (stock + price) and 2c (orders) key on
 * this map: a Lazada SellerSku that is not mapped is never guessed at.
 *
 * Sources:
 *   • `parseLazadaProducts` reads GET /products/get. Field names are parsed
 *     DEFENSIVELY (Lazada mixes cases: SellerSku / ShopSku / SkuId inside `skus[]`,
 *     `attributes.name`); anything missing just leaves that field null. The sync keeps
 *     the raw shape report so a surprise is visible, not silent.
 *   • `suggestProduct` proposes a match: an exact SKU match (case-insensitive) is
 *     confirmed automatically — it is the same code on both sides — otherwise the one
 *     product with the same weight AND brand (and not a contradicting minted/cast form)
 *     is SUGGESTED for a one-tap confirm. Two or more candidates → no suggestion; the
 *     admin picks. Reuses the competitor matcher's vocabulary so "PAMP", "Argor",
 *     "1 oz", "31.1 g" mean the same thing everywhere.
 */
import { parseBrand, productBrand, parseForm, productForm, weightsMatch } from "@/lib/pricing/competitor-match";
import { parseWeightGrams } from "@/lib/pricing/competitor-ingest";
 
export interface LazadaListing {
  sellerSku: string;
  itemId: string | null;
  skuId: string | null;
  shopSku: string | null;
  name: string;
  status: string | null;
  quantity: number | null;
  price: number | null;
  specialPrice: number | null;
  /** Lazada's category id for the listing (primary_category) — read, never typed. */
  category: string | null;
}
 
const str = (v: unknown): string | null => (v == null || v === "" ? null : String(v));
const num = (v: unknown): number | null => {
  const n = typeof v === "string" ? Number(v) : typeof v === "number" ? v : NaN;
  return Number.isFinite(n) ? n : null;
};
const pick = (o: Record<string, unknown>, ...keys: string[]) => {
  for (const k of keys) if (o[k] !== undefined && o[k] !== null && o[k] !== "") return o[k];
  return undefined;
};
 
/** One row per Lazada SKU (a product with variations has several). */
export function parseLazadaProducts(body: unknown): LazadaListing[] {
  const data = ((body ?? {}) as { data?: { products?: unknown } }).data;
  const products = Array.isArray(data?.products) ? (data!.products as Array<Record<string, unknown>>) : [];
  const out: LazadaListing[] = [];
  for (const p of products) {
    const attrs = (p.attributes ?? {}) as Record<string, unknown>;
    const name = String(pick(attrs, "name", "name_en") ?? pick(p, "name") ?? "").trim();
    const skus = Array.isArray(p.skus) ? (p.skus as Array<Record<string, unknown>>) : [];
    for (const s of skus) {
      const sellerSku = str(pick(s, "SellerSku", "seller_sku", "sellerSku"));
      if (!sellerSku) continue;
      out.push({
        sellerSku,
        itemId: str(pick(p, "item_id", "itemId")),
        skuId: str(pick(s, "SkuId", "sku_id", "skuId")),
        shopSku: str(pick(s, "ShopSku", "shop_sku")),
        name,
        status: str(pick(s, "Status", "status")) ?? str(pick(p, "status")),
        quantity: num(pick(s, "quantity", "Available", "sellableStock")),
        price: num(pick(s, "price")),
        specialPrice: num(pick(s, "special_price")),
        category: str(pick(p, "primary_category", "primaryCategory")),
      });
    }
  }
  return out;
}
 
export interface OurProduct {
  id: string;
  label: string;
  grams: number;
  sku: string | null;
  refinery: string | null;
  status: string;
}
 
export type MatchSource = "sku" | "suggested" | "none";
 
export interface Suggestion {
  productId: string | null;
  source: MatchSource;
  /** Why — shown on the mapping row. */
  reason: string;
}
 
export function suggestProduct(listing: Pick<LazadaListing, "sellerSku" | "name">, products: OurProduct[]): Suggestion {
  const key = listing.sellerSku.trim().toLowerCase();
  const bySku = products.filter((p) => p.sku && p.sku.trim().toLowerCase() === key);
  if (bySku.length === 1) return { productId: bySku[0].id, source: "sku", reason: `Same SKU (${bySku[0].sku})` };
 
  const w = parseWeightGrams(listing.name);
  const brand = parseBrand(listing.name);
  if (!w || brand === "UNKNOWN") {
    return { productId: null, source: "none", reason: !w ? "No weight in the Lazada title" : "No known brand in the Lazada title" };
  }
  const form = parseForm(listing.name);
  const candidates = products.filter((p) => {
    if (p.status === "archived") return false;
    if (!weightsMatch(Number(p.grams), w.grams)) return false;
    if (productBrand(p) !== brand) return false;
    const pf = productForm(p);
    return form === "UNKNOWN" || pf === "UNKNOWN" || pf === form;
  });
  if (candidates.length === 1) {
    return { productId: candidates[0].id, source: "suggested", reason: `Same weight (${w.label}) and brand` };
  }
  return { productId: null, source: "none",
    reason: candidates.length === 0 ? `No ${w.label} ${brand} product of ours` : `${candidates.length} of our products fit — pick one` };
}
