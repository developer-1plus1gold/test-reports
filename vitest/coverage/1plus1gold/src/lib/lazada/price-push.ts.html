/**
 * Lazada price + stock push (step 2b part 2, docs/lazada-integration-design.md §4/§4b).
 *
 * One function, three callers: the cron (only while the switch is ON), "Push now", and
 * "Approve" on a held row. Every price comes from plannedUpdate — the same pure rule the
 * preview shows — so what the admin saw is what Lazada gets.
 *
 * Guards (the push REFUSES rather than publishes a bad price):
 *   • gold price: the same staleness the shop's checkout uses (xau/fx stale or older than
 *     15 min → nothing is sent);
 *   • no live base → no price; below restock cost after fees → never sent;
 *   • a move > JUMP_CAP waits for an admin Approve (its stock still goes);
 *   • only 'sku' / 'confirmed' mappings, only listings that changed.
 *
 * Stock-only runs (`{ kind: "stock" }`, 2c part 2): on every 5-minute tick where no price
 * push is due — or the gold price is stale — the cron sends just the listings whose stock
 * changed. Stock needs no gold price, so a website / Lazada / manual sale reaches Lazada
 * within ~5 min whatever the price cadence. They never touch the price-cadence state.
 *
 * ▶ Since 2026-11-17 the run and its guards live in lib/marketplace/price-push (shared
 * with TikTok Shop); this file is Lazada's transport and its names for the callers.
 *
 * Lazada call (verified in Lazada's Node SDK, lazada-api 2.6.0): POST
 * /product/price_quantity/update with form field `payload` = XML, max 50 SKUs per call
 * (20 recommended — we send 20).
 */

const esc = (s: string) => s.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&apos;");
 
/** The UpdatePriceQuantity payload. Only the fields being changed are included. */
export function buildPriceQuantityXml(items: SkuUpdate[]): string {
  const skus = items.map((u) =>
    "<Sku>"
    + `<ItemId>${esc(u.itemId)}</ItemId><SkuId>${esc(u.skuId)}</SkuId><SellerSku>${esc(u.sellerSku)}</SellerSku>`
    + (u.price != null ? `<Price>${u.price.toFixed(2)}</Price>` : "")
    + (u.quantity != null ? `<Quantity>${Math.max(0, Math.trunc(u.quantity))}</Quantity>` : "")
    + "</Sku>").join("");
  return `<?xml version="1.0" encoding="UTF-8" ?><Request><Product><Skus>${skus}</Skus></Product></Request>`;
}

// ── The run ──────────────────────────────────────────────────────────────────
 
export const LAZADA_PUSH: PushChannel = {
  channel: "lazada",
  label: "Lazada",
  agentName: "lazada_sync",
  page: "/admin/marketplaces?tab=lazada",
  async connect(db, now, http) {
    const auth = await getAccessToken(db, now);
    if (!auth) return null;
    return {
      async send(updates: SkuUpdate[]) {
        const failures = new Map<string, string>();
        for (let i = 0; i < updates.length; i += BATCH) {
          const batch = updates.slice(i, i + BATCH);
          try {
            const body = await lazadaCall("/product/price_quantity/update", { payload: buildPriceQuantityXml(batch) },
              { appKey: auth.appKey, appSecret: auth.appSecret, accessToken: auth.token, fetchImpl: http.fetchImpl });
            for (const [sku, msg] of failedSkus(body)) failures.set(sku, msg);
          } catch (err) {
            const msg = err instanceof Error ? err.message.slice(0, 200) : "push failed";
            for (const u of batch) failures.set(u.sellerSku, msg);
          }
        }
        return failures;
      },
    };
  },
};
 
export function runLazadaPricePush(db: SupabaseClient, now: Date, mode: PushMode, http: PushHttp = {}): Promise<PushResult> {
  return runMarketplacePush(db, now, mode, LAZADA_PUSH, http);
