/**
 * Lazada Open Platform HTTP client — the ONE place that talks to Lazada.
 *
 * Gateways (verified in Lazada's SDKs): api.lazada.sg/rest for seller APIs,
 * auth.lazada.com/rest for the token endpoints. Every call is signed the same way;
 * READ APIs (e.g. /orders/get) must be GET with the params in the query string —
 * Lazada answers a POST to them with `UnsupportedHTTPMethod` (seen live 2026-11-16) —
 * while writes and the token endpoints are POST. A body `code` other than "0" is an error. Params (which include the access token)
 * are NEVER logged or put in error messages — only the path, Lazada's code/message
 * and its request_id.
 */
import { signLazadaRequest, type LazadaParams } from "./sign";
 
export const LAZADA_API_GATEWAY = "https://api.lazada.sg/rest";
export const LAZADA_AUTH_GATEWAY = "https://auth.lazada.com/rest";
/** Seller authorization page (Lazada "Seller authorization introduction"). Confirmed live
 *  on the first Connect; force_auth makes Lazada always show the consent screen. */
export const LAZADA_AUTHORIZE_URL = "https://auth.lazada.com/oauth/authorize";
 
const PARTNER_ID = "1plus1gold-ts-1";
const TIMEOUT_MS = 20_000;
 
export class LazadaApiError extends Error {
  constructor(message: string, readonly code: string, readonly requestId: string | null) { super(message); }
}
 
/**
 * Pure: Lazada's own reason behind a refusal. A bare `E501 Update product failed` (or
 * `E005 Invalid Request Format`) carries the real cause — the field and why — only in the
 * body's `detail` array, which the error used to drop (2026-11-23: five listings refused
 * every price push for two days with nothing to say why). Field + message only; never
 * any request param.
 */
export function errorDetail(body: Record<string, unknown>): string {
  const detail = body.detail;
  if (!Array.isArray(detail)) return "";
  const parts = (detail as Array<Record<string, unknown>>)
    .map((d) => [d.field, d.message].filter((v) => v != null && String(v).trim() !== "").map(String).join(": "))
    .filter(Boolean);
  return [...new Set(parts)].join("; ").slice(0, 300);
}
 
export interface LazadaCallOptions {
  appKey: string;
  appSecret: string;
  accessToken?: string;
  gateway?: "api" | "auth";
  /** GET for Lazada's read APIs, POST (default) for writes and the token endpoints. */
  method?: "GET" | "POST";
  fetchImpl?: typeof fetch;
  now?: () => number;
  /** Test seam for the throttle wait. */
  sleep?: (ms: number) => Promise<void>;
  /** File parts (e.g. UploadVideoBlock's `file`): sent multipart/form-data, NOT signed —
   *  Lazada signs the text params only, as its own SDKs do. */
  files?: Record<string, { data: Uint8Array; filename: string; contentType: string }>;
  /** Longer than the default for uploads. */
  timeoutMs?: number;
}
 
/** Lazada's per-second throttle ("ApiCallLimit … this ban will last N seconds"). */
export const THROTTLE_RETRIES = 3;
/**
 * Refusals that mean "too fast, try again", so they are retried like ApiCallLimit. E0901 ("Limit
 * service request speed in server side temporarily", code 901) is Lazada's server-side form of the
 * same throttle — it refused the 1 oz price update on 11-23 when a refused batch was re-sent one
 * SKU at a time. Like ApiCallLimit, it is rejected before Lazada acts, so a retry is safe.
 */
export function isThrottle(e: LazadaApiError): boolean {
  return e.code === "ApiCallLimit" || e.code === "901" || /\bE0901\b/.test(e.message);
}
const MAX_THROTTLE_WAIT_MS = 5_000;
 
/** Pure: the wait Lazada asks for in an ApiCallLimit message (+ a margin), capped. */
export function throttleWaitMs(message: string): number {
  const m = /last\s+(\d+)\s+second/i.exec(message);
  const secs = m ? Number(m[1]) : 1;
  return Math.min(MAX_THROTTLE_WAIT_MS, secs * 1000 + 250);
}
 
/**
 * One Lazada call. An ApiCallLimit refusal (Lazada throttles calls made back to back — first
 * seen 2026-11-18 on the licence run: migrate → read → update → read-back inside a second) is
 * retried after the wait Lazada names, re-signed with a fresh timestamp. Safe for writes too:
 * a throttled request is rejected at the gateway before Lazada acts on it.
 */
export async function lazadaCall<T = Record<string, unknown>>(
  apiPath: string,
  apiParams: LazadaParams,
  opts: LazadaCallOptions,
): Promise<T> {
  const sleep = opts.sleep ?? ((ms: number) => new Promise<void>((r) => setTimeout(r, ms)));
  for (let attempt = 0; ; attempt++) {
    try {
      return await lazadaCallOnce<T>(apiPath, apiParams, opts);
    } catch (e) {
      if (!(e instanceof LazadaApiError) || !isThrottle(e) || attempt >= THROTTLE_RETRIES) throw e;
      await sleep(throttleWaitMs(e.message));
    }
  }
}
 
async function lazadaCallOnce<T>(
  apiPath: string,
  apiParams: LazadaParams,
  opts: LazadaCallOptions,
): Promise<T> {
  const params: LazadaParams = {
    ...apiParams,
    app_key: opts.appKey,
    sign_method: "sha256",
    timestamp: String(opts.now ? opts.now() : Date.now()),
    partner_id: PARTNER_ID,
    ...(opts.accessToken ? { access_token: opts.accessToken } : {}),
  };
  params.sign = signLazadaRequest(opts.appSecret, apiPath, params);
 
  const base = opts.gateway === "auth" ? LAZADA_AUTH_GATEWAY : LAZADA_API_GATEWAY;
  const encoded = new URLSearchParams(params).toString();
  const timeout = AbortSignal.timeout(opts.timeoutMs ?? TIMEOUT_MS);
  const res = opts.files
    ? await (opts.fetchImpl ?? fetch)(base + apiPath, { method: "POST", body: multipart(params, opts.files), signal: timeout })
    : opts.method === "GET"
    ? await (opts.fetchImpl ?? fetch)(`${base}${apiPath}?${encoded}`, { method: "GET", signal: timeout })
    : await (opts.fetchImpl ?? fetch)(base + apiPath, {
        method: "POST",
        headers: { "Content-Type": "application/x-www-form-urlencoded;charset=utf-8" },
        body: encoded,
        signal: timeout,
      });
  const text = await res.text();
  let body: Record<string, unknown>;
  try {
    body = JSON.parse(text) as Record<string, unknown>;
  } catch {
    throw new LazadaApiError(`Lazada ${apiPath}: HTTP ${res.status}, non-JSON response`, `http_${res.status}`, null);
  }
  const code = body.code == null ? "0" : String(body.code);
  const requestId = typeof body.request_id === "string" ? body.request_id : null;
  if (!res.ok || code !== "0") {
    const msg = String(body.message ?? `HTTP ${res.status}`).slice(0, 200);
    const why = errorDetail(body);
    throw new LazadaApiError(`Lazada ${apiPath}: ${code} ${msg}${why ? ` — ${why}` : ""}`, code, requestId);
  }
  return body as T;
}
 
/** Signed text params + file parts as multipart/form-data (fetch sets the boundary). */
function multipart(params: LazadaParams, files: NonNullable<LazadaCallOptions["files"]>): FormData {
  const form = new FormData();
  for (const [k, v] of Object.entries(params)) form.append(k, v);
  for (const [k, f] of Object.entries(files)) form.append(k, new Blob([f.data as BlobPart], { type: f.contentType }), f.filename);
  return form;
}
 
/** The URL the admin is sent to for Connect. */
export function lazadaAuthorizeUrl(appKey: string, redirectUri: string, state: string): string {
  const u = new URL(LAZADA_AUTHORIZE_URL);
  u.searchParams.set("response_type", "code");
  u.searchParams.set("force_auth", "true");
  u.searchParams.set("redirect_uri", redirectUri);
  u.searchParams.set("client_id", appKey);
  u.searchParams.set("state", state);
  return u.toString();
}
