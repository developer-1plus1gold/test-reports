/** One state for the order from its items' statuses (Lazada keeps a status per item). */
export function orderState(o: LazadaOrder, items: LazadaItem[]): OrderState {
  const statuses = (items.length ? items.map((i) => i.status ?? "") : (o.statuses ?? [])).map((s) => s.toLowerCase());
  const live = statuses.filter((s) => !CANCELED.has(s));
  if (live.length === 0) return "canceled";
  if (live.some((s) => RETURNED.has(s))) return "returned";
  if (live.every((s) => PAID_STATUSES.has(s))) return "paid";
  return "unpaid";
}
 
/**
 * Pure: a paid Lazada order → the createExternalOrder input. Each Lazada item row is ONE
 * unit; rows of the same SKU at the same price are grouped. Canceled item rows are dropped.
 * The channel fee is an ESTIMATE from our Lazada pricing settings (commission + processing
 * % of the item prices) until the finance API is read — the order API does not carry it.
 */
export function mapLazadaOrder(o: LazadaOrder, items: LazadaItem[], productBySku: Map<string, string>, cfg: MarketplaceCostConfig): MapResult {
  const live = items.filter((i) => !CANCELED.has((i.status ?? "").toLowerCase()));
  const unknownSkus = [...new Set(live.map((i) => i.sku ?? "").filter((s) => !productBySku.has(s)))];
  if (unknownSkus.length) return { ok: false, unknownSkus: unknownSkus.map((s) => s || "(blank SKU)") };
 
  const lines = new Map<string, { productId: string; quantity: number; priceSgd: number }>();
  for (const i of live) {
    const productId = productBySku.get(i.sku!)!;
    const price = r2(money(i.item_price));
    const key = `${productId}|${price}`;
    const line = lines.get(key) ?? { productId, quantity: 0, priceSgd: price };
    line.quantity += 1;
    lines.set(key, line);
  }
  const gross = live.reduce((s, i) => s + money(i.item_price), 0);
  const pct = (cfg.commissionPct + cfg.otherPct) / 100;
  return {
    ok: true,
    input: {
      channel: CHANNEL,
      externalOrderId: externalId(o),
      buyerLabel: "Lazada buyer",
      orderDate: sgtDate(o.created_at),
      channelFeeSgd: r2(gross * pct),
      sellerVoucherSgd: r2(live.reduce((s, i) => s + money(i.voucher_seller), 0)),
      absorbedShippingSgd: r2(live.reduce((s, i) => s + money(i.shipping_fee_discount_seller), 0)),
      shipTo: { postal_code: o.address_shipping?.post_code?.replace(/\*/g, "").trim() || null },
      items: [...lines.values()],
    },
  };
}
 
/**
 * Pure: the buyer's checkout on Lazada, from the live item rows. Lazada reports "buyer paid"
 * per item (`paid_price`, after both vouchers) plus the shipping the buyer paid per item.
 */
export function lazadaBuyerFigures(items: LazadaItem[]): BuyerFigures {
  const live = items.filter((i) => !CANCELED.has((i.status ?? "").toLowerCase()));
  const hasPaid = live.length > 0 && live.every((i) => i.paid_price != null && Number.isFinite(Number(i.paid_price)));
  const buyerShipping = live.reduce((s, i) => s + money(i.shipping_amount), 0);
  return {
    listSgd: r2(live.reduce((s, i) => s + money(i.item_price), 0)),
    sellerVoucherSgd: r2(live.reduce((s, i) => s + money(i.voucher_seller), 0)),
    platformVoucherSgd: r2(live.reduce((s, i) => s + money(i.voucher_platform), 0)),
    platformShippingDiscountSgd: r2(live.reduce((s, i) => s + money(i.shipping_fee_discount_platform), 0)),
    buyerShippingSgd: r2(buyerShipping),
    otherBuyerChargesSgd: 0,
    reportedPaidSgd: hasPaid ? r2(live.reduce((s, i) => s + Number(i.paid_price), 0) + buyerShipping) : null,
    source: {
      items: live.map((i) => ({
        item_price: i.item_price, paid_price: i.paid_price, voucher_seller: i.voucher_seller, voucher_platform: i.voucher_platform,
        shipping_amount: i.shipping_amount, shipping_fee_original: i.shipping_fee_original,
        shipping_fee_discount_seller: i.shipping_fee_discount_seller, shipping_fee_discount_platform: i.shipping_fee_discount_platform,
      })),
    },
  };
}
 
/** Pure: every live item is at or past Ready to ship on Lazada. */
export function lazadaArranged(items: LazadaItem[]): boolean {
  const live = items.map((i) => (i.status ?? "").toLowerCase()).filter((st) => !CANCELED.has(st));
  return live.length > 0 && live.every((st) => st === "ready_to_ship" || st === "ready_to_ship_pending" || st === "shipped" || st === "delivered" || st === "confirmed");
}

export async function syncLazadaOrders(db: SupabaseClient, now: Date, mode: ImportMode, http: Http = {}, opts: ImportOptions = {}): Promise<OrderSyncResult> {
  const r = await runOrderImport(db, now, mode, LAZADA_ORDERS, http, opts);
  // Then arrange the shipment for paid orders, when the channel's auto-dispatch is on.
  if (r.ran) {
    r.autoDispatch = await runAutoDispatch(db, now, LAZADA_DISPATCH, http).catch((e: unknown) => {
      console.error("[auto-dispatch]", e instanceof Error ? e.message : e);
      return undefined;
    });
  }
  return r;
}
