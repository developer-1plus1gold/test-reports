/**
 * Create a Lazada listing for one of our products (Mukesh 2026-11-19: put the 1 g and 2.5 g
 * Nadir on Lazada and TikTok). Server-only, except the pure helpers.
 *
 * The listing is CLONED from a sibling we already sell on Lazada (same refinery where we
 * have one): its category and product attributes, and its parcel, are what Lazada already
 * accepted for a bar like this — nothing is typed by hand. What changes is ours: the name,
 * the description (the shop-tile facts text the Shopee and TikTok listings use, no
 * "investment" wording), our SKU, the price from the Lazada pricing settings (the same
 * plannedUpdate the price push sends, refused below restock cost), our stock, and the
 * gallery: the product's own images as JPG (Lazada takes JPG/PNG only) + our PSPM
 * registration certificate LAST, like every other listing.
 *
 * Shapes (Lazada API reference, open.lazada.com):
 *  - GetProductItem GET /product/item/get?item_id → data.primary_category, data.attributes,
 *    data.skus[] (package_* live on the SKU)
 *  - MigrateImage  POST /image/migrate (see licence-image.ts)
 *  - CreateProduct POST /product/create, payload `<Request><Product><PrimaryCategory/>
 *    <Images/><Attributes/><Skus><Sku/></Skus></Product></Request>` → data.item_id,
 *    data.sku_list[{ seller_sku, sku_id, shop_sku }]
 * A new listing goes through Lazada's review; it is recorded `pending` and the reconcile
 * read moves it to `active`. The price + stock push take it over from there.
 */

// XML 1.0 forbids most control characters; one stray one makes Lazada answer "E005 Invalid
// Request Format" for the whole payload, so they are dropped before escaping.
// eslint-disable-next-line no-control-regex
const xml = (s: string) => s.replace(/[\u0000-\u0008\u000B\u000C\u000E-\u001F\uFFFE\uFFFF]/g, "").replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&apos;");
 
/** Product attributes that are the sibling's OWN content, never copied onto a new listing. */
const OWN_CONTENT = new Set(["name", "description", "short_description", "name_ms", "description_ms", "short_description_ms", "name_en", "description_en", "video", "promotion_whitebkg_image",
  // Set by Lazada itself (how the sibling was created), not a listing attribute.
  "source"]);
const XML_NAME = /^[A-Za-z_][A-Za-z0-9_]*$/;
/** SKU fields copied from the sibling: the parcel (what Lazada's carriers already accepted). */
const PARCEL_FIELDS = ["package_weight", "package_length", "package_width", "package_height", "package_content"] as const;
 
export interface Template {
  itemId: string;
  categoryId: string;
  attributes: Record<string, string>;
  parcel: Partial<Record<(typeof PARCEL_FIELDS)[number], string>>;
}
 
/** Pure: GetProductItem data → the parts a new listing inherits. */
export function templateOf(itemId: string, data: unknown): Template {
  const d = (data ?? {}) as { primary_category?: unknown; attributes?: Record<string, unknown>; skus?: Array<Record<string, unknown>> };
  const categoryId = String(d.primary_category ?? "");
  if (!/^\d+$/.test(categoryId)) throw new Error("the sibling listing has no Lazada category");
  const attributes: Record<string, string> = {};
  for (const [k, v] of Object.entries(d.attributes ?? {})) {
    // Only names that are valid XML element names travel (they become <k>…</k>).
    if (!XML_NAME.test(k) || OWN_CONTENT.has(k) || v == null || typeof v === "object" || v === "") continue;
    attributes[k] = String(v);
  }
  const sku = (Array.isArray(d.skus) ? d.skus[0] : null) ?? {};
  const parcel: Template["parcel"] = {};
  for (const f of PARCEL_FIELDS) if (sku[f] != null && sku[f] !== "") parcel[f] = String(sku[f]);
  return { itemId, categoryId, attributes, parcel };
}

/** Pure: a weight in the sibling's own format ("5G" → "1G", "2.5G"; "5 g" → "1 g"). */
export function weightLike(siblingValue: string, grams: number): string {
  const g = Number.isInteger(grams) ? String(grams) : String(Number(grams.toFixed(2)));
  const m = /^\s*[\d.]+\s*([A-Za-z]+)\s*$/.exec(siblingValue);
  return m ? siblingValue.replace(/[\d.]+/, g) : `${g}G`;
}
 
/** Our refinery → Lazada's brand name EXACTLY as its dropdown spells it (a near miss is C011: "Argor-Heraeus" was refused, Seller Center lists "Argor Heraeus", 11-25) and certification. */
const LAZADA_BRANDS: ReadonlyArray<{ key: string; brand: string; certifications: string }> = [
  { key: "argor",    brand: "Argor Heraeus",         certifications: "Argor Heraeus" },
  { key: "heraeus",  brand: "Argor Heraeus",         certifications: "Argor Heraeus" },
  { key: "valcambi", brand: "Valcambi Suisse",      certifications: "Valcambi" },
  { key: "pamp",     brand: "Pamp",                 certifications: "PAMP Suisse" },
  { key: "nadir",    brand: "Nadir Metal Refinery", certifications: "Nadir" },
];
 
/**
 * Pure: the brand a listing must carry when the sibling it is cloned from is ANOTHER
 * refinery's (Mukesh 2026-11-25: the first Argor-Heraeus KineBars had no Argor sibling on
 * Lazada, so the 1 g previewed as brand "Pamp" and the 5 g as "Nadir Metal Refinery").
 * Only when the sibling's brand names a DIFFERENT known refinery: a sibling already of our
 * refinery, a "No Brand" sibling, or a refinery we do not know keeps the sibling's brand.
 */
export function brandOverride(ourRefinery: string, siblingBrand: string | undefined): { brand: string; certifications: string } | null {
  const ours = LAZADA_BRANDS.find((x) => ourRefinery.toLowerCase().includes(x.key));
  const sibText = (siblingBrand ?? "").toLowerCase();
  const sib = LAZADA_BRANDS.find((x) => sibText.includes(x.key));
  if (!ours || !sib || sib.brand === ours.brand) return null;
  return { brand: ours.brand, certifications: ours.certifications };
}
 
export function createProductXml(t: Template, l: { name: string; descriptionHtml: string; sellerSku: string; price: number; quantity: number; images: string[]; grams: number; refinery?: string | null }): string {
  if (!(l.price > 0)) throw new Error("no price");
  if (!l.images.length) throw new Error("no images");
  Iif (!(l.grams > 0)) throw new Error("no weight");
  // The sibling's weight and box contents describe the SIBLING (first live preview, 11-19:
  // a 1 g bar would have gone up as weight "5G", "1 x 5 gram Gold Bar"): ours replace them.
  const attrs: Record<string, string> = { ...t.attributes, name: l.name, description: l.descriptionHtml };
  if (attrs.weight_of_gold) attrs.weight_of_gold = weightLike(attrs.weight_of_gold, l.grams);
  // The sibling's brand describes the SIBLING too, when it is another refinery's.
  const own = brandOverride(l.refinery || l.name, attrs.brand);
  if (own) {
    attrs.brand = own.brand;
    if ("certifications" in attrs) attrs.certifications = own.certifications;
  }
  const parcel: Record<string, string> = { ...t.parcel, package_content: `1 x ${l.name}` } as Record<string, string>;
  const attrXml = Object.entries(attrs).map(([k, v]) => `<${k}>${xml(v)}</${k}>`).join("");
  const parcelXml = Object.entries(parcel).map(([k, v]) => `<${k}>${xml(String(v))}</${k}>`).join("");
  const imgs = l.images.slice(0, LAZADA_MAX_IMAGES).map((u) => `<Image>${xml(u)}</Image>`).join("");
  return `<?xml version="1.0" encoding="UTF-8" ?><Request><Product><PrimaryCategory>${t.categoryId}</PrimaryCategory>`
    + `<Images>${imgs}</Images><Attributes>${attrXml}</Attributes>`
    + `<Skus><Sku><SellerSku>${xml(l.sellerSku)}</SellerSku><quantity>${Math.max(0, Math.floor(l.quantity))}</quantity>`
    + `<price>${l.price.toFixed(2)}</price>${parcelXml}</Sku></Skus></Product></Request>`;
}
 
/** Pure: the sibling to clone — a live, confirmed Lazada listing of the same refinery where
 *  there is one (Nadir for Nadir), else of the nearest weight. */
export function pickSibling(p: { label: string; grams: number }, siblings: Array<{ item_id: string; label: string; grams: number }>): string | null {
  const brand = (p.label.match(/(nadir|pamp|valcambi|argor|perth|maple|heraeus)/i)?.[1] ?? "").toLowerCase();
  const pool = brand ? siblings.filter((s) => s.label.toLowerCase().includes(brand)) : [];
  const from = pool.length ? pool : siblings;
  if (!from.length) return null;
  return [...from].sort((a, b) => Math.abs(a.grams - p.grams) - Math.abs(b.grams - p.grams))[0].item_id;
}

/** Creates one Lazada listing per product id. Each product stands alone: one failure never stops the rest. */
export async function createLazadaListings(db: SupabaseClient, productIds: string[], now: Date, http: Http = {}, opts: { dryRun?: boolean } = {}): Promise<{ rows: LazadaCreateRow[]; error?: string }> {
  const auth = await getAccessToken(db, now);
  if (!auth) return { rows: [], error: "Lazada is not connected" };
  const call = <T>(path: string, params: Record<string, string>, method?: "GET") =>
    lazadaCall<T>(path, params, { appKey: auth.appKey, appSecret: auth.appSecret, accessToken: auth.token, fetchImpl: http.fetchImpl, ...(method ? { method } : {}) });
 
  const [products, { data: mapped }, cfg, spot] = await Promise.all([
    fetchExportProducts(),
    db.from("marketplace_listings").select("product_id, item_id, seller_sku, marketplace_status, match_source").eq("channel", CHANNEL).not("product_id", "is", null),
    loadPriceConfig(db, CHANNEL),
    freshSpot(),
  ]);
  if (spot == null) return { rows: [], error: "The gold price is not fresh — try again in a minute." };
  const rowsMapped = (mapped ?? []) as Array<{ product_id: string; item_id: string | null; seller_sku: string; marketplace_status: string | null; match_source: string }>;
  const already = new Set(rowsMapped.map((r) => r.product_id));
  const byId = new Map(products.map((p) => [p.id, p]));
  const siblings = rowsMapped
    .filter((r) => r.item_id && r.marketplace_status === "active" && (ACTIONABLE_MATCH as readonly string[]).includes(r.match_source))
    .map((r) => ({ item_id: r.item_id!, label: byId.get(r.product_id)?.label ?? "", grams: Number(byId.get(r.product_id)?.grams ?? 0) }));
  const [premiums, restock] = await Promise.all([fetchPremiums(), loadRestockCosts(db).catch(() => new Map() as Awaited<ReturnType<typeof loadRestockCosts>>)]);
 
  // The licence, once: Lazada hosts it, every new gallery ends with it. A dry run sends nothing.
  const lic = opts.dryRun ? null : await call<{ data?: { image?: { url?: string } } }>("/image/migrate", { payload: migrateImageXml(lazadaLicenceUrl()) });
  const licenceUrl = opts.dryRun ? lazadaLicenceUrl() : lic?.data?.image?.url;
  if (!licenceUrl) return { rows: [], error: "Lazada did not return a URL for the licence image" };
 
  const templates = new Map<string, Template>();
  const rows: LazadaCreateRow[] = [];
  for (const id of productIds) {
    const p = byId.get(id) as ShopeeSourceProduct | undefined;
    if (!p) { rows.push({ productId: id, title: id, ok: false, error: "not a gold product on our shop" }); continue; }
    const title = p.label;
    if (already.has(id)) { rows.push({ productId: id, title, ok: false, error: "already on Lazada (a Lazada listing is mapped to it)" }); continue; }
    try {
      const siblingId = pickSibling({ label: p.label, grams: Number(p.grams) }, siblings);
      if (!siblingId) throw new Error("no live Lazada listing to copy the category from");
      let t = templates.get(siblingId);
      if (!t) {
        t = templateOf(siblingId, (await call<{ data?: unknown }>("/product/item/get", { item_id: siblingId }, "GET"))?.data);
        templates.set(siblingId, t);
      }
      const base = shopLineForProduct(p as unknown as ShopPriceable, 1, spot, premiums, 0).priceEach;
      const cost = restock.get(p.id);
      const plan = plannedUpdate({ baseSgd: base ?? null, status: p.status, stock_qty: p.stock_qty }, { price: null, qty: null, specialPrice: null }, cfg,
        { restockCostSgd: cost ? cost.sgdPerG * Number(p.grams) : null });
      if (plan.price == null) throw new Error(plan.notes.join("; ") || "no price");
 
      const own = { ...p, product_media: p.product_media.filter((m) => (m.kind ?? "image") === "image") };
      const sellerSku = defaultSku(p);
      if (opts.dryRun) {
        const images = [...own.product_media.map((m) => m.url).slice(0, LAZADA_MAX_IMAGES - 1), licenceUrl];
        const payload = createProductXml(t, { name: title, descriptionHtml: tiktokDescription(p), sellerSku, price: plan.price, quantity: plan.stock ?? 0, images, grams: Number(p.grams), refinery: p.refinery });
        rows.push({ productId: p.id, title, ok: true, sellerSku, price: plan.price, payload, templateItemId: t.itemId, templateAttributeKeys: Object.keys(t.attributes) });
        continue;
      }
      const jpgs = (await hostShopeeImages([own]))[p.id] ?? [];
      const images: string[] = [];
      for (const u of jpgs.slice(0, LAZADA_MAX_IMAGES - 1)) {
        const m = await call<{ data?: { image?: { url?: string } } }>("/image/migrate", { payload: migrateImageXml(u) });
        if (m?.data?.image?.url) images.push(m.data.image.url);
      }
      if (!images.length) throw new Error("none of the product's images could be sent to Lazada");
      images.push(licenceUrl);
 
      const created = await call<{ data?: { item_id?: number | string; sku_list?: Array<{ seller_sku?: string; sku_id?: number | string; shop_sku?: string }> } }>(
        "/product/create",
        { payload: createProductXml(t, { name: title, descriptionHtml: tiktokDescription(p), sellerSku, price: plan.price, quantity: plan.stock ?? 0, images, grams: Number(p.grams), refinery: p.refinery }) },
      );
      const itemId = created?.data?.item_id;
      const sku = created?.data?.sku_list?.find((s) => s.seller_sku === sellerSku) ?? created?.data?.sku_list?.[0];
      if (itemId == null || sku?.sku_id == null) throw new Error("Lazada did not return the item and SKU ids");
      const iso = now.toISOString();
      const { error: dbErr } = await db.from("marketplace_listings").upsert({
        channel: CHANNEL, seller_sku: sellerSku, item_id: String(itemId), sku_id: String(sku.sku_id), shop_sku: sku.shop_sku ?? null,
        name: title.slice(0, 500), marketplace_status: "pending", marketplace_qty: plan.stock ?? 0, marketplace_price: plan.price,
        category: t.categoryId, product_id: p.id, match_source: "confirmed", match_reason: "Created from admin",
        matched_at: iso, last_seen_at: iso, last_pushed_at: iso, updated_at: iso,
      }, { onConflict: "channel,seller_sku" });
      if (dbErr) throw new Error(`listed on Lazada (${itemId}) but our mapping did not save — confirm it in Admin → Marketplaces`);
      rows.push({ productId: p.id, title, ok: true, itemId: String(itemId), sellerSku, price: plan.price });
    } catch (e) {
      rows.push({ productId: id, title, ok: false, error: (e instanceof Error ? e.message : String(e)).slice(0, 300) });
    }
  }
  return { rows };
}
