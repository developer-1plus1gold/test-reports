/**
 * Lazada: add our PSPM registration certificate as the LAST gallery image of each listing
 * (Mukesh 2026-11-18, same as TikTok Shop). Server-only, except the pure helpers.
 *
 * Shapes are Lazada's own API reference (open.lazada.com, read 2026-11-18):
 *  - MigrateImage  POST /image/migrate, payload `<Request><Image><Url>…</Url></Image></Request>`
 *    → data.image { url, hash_code }. JPG/PNG only, ≤ 1 MB — hence the .jpg copy of the
 *    licence (the WebP one is TikTok's and our site's).
 *  - GetProductItem GET /product/item/get?item_id → data.images (the PRODUCT gallery — where
 *    our listings keep their pictures; every skus[].Images is empty), data.skus[], data.attributes.
 *  - UpdateProduct POST /product/update, payload `<Request><Product><ItemId/>…</Product></Request>`
 *    ("update attributes or SKUs of an existing product"); `Product/Images/Image` is the
 *    gallery element of the same product schema (CreateProduct).
 *
 * SAFETY — this writes to live listings, so it reads before and after each one:
 *  - we send ItemId + Images + the SKU's identity only (SkuId, SellerSku — Lazada refuses an
 *    update without Skus: "E001 Parameter Skus is mandatory", first live run 2026-11-18); no
 *    price, quantity or attributes, and the read-back below proves price and stock did not move;
 *  - the gallery sent is the listing's CURRENT images + the licence (UpdateProduct replaces
 *    the list) — never more than Lazada's 8; a listing already showing the licence is skipped;
 *  - after the write we read the listing back: the gallery must be the old one + licence and
 *    name, status, price and stock must be unchanged. The first listing that does not read back
 *    that way STOPS the batch, so a wrong assumption costs at most one listing.
 */

export function migrateImageXml(url: string): string {
  return `<?xml version="1.0" encoding="UTF-8" ?><Request><Image><Url>${xml(url)}</Url></Image></Request>`;
}
 
/** Pure: UpdateProduct payload that sets ONLY the product gallery; the SKU is named (Lazada
 *  requires Skus) with no price or quantity. */
export function galleryUpdateXml(itemId: string, sku: { skuId: string; sellerSku: string }, images: readonly string[]): string {
  if (!/^\d{1,20}$/.test(itemId)) throw new Error("invalid Lazada item id");
  if (!/^\d{1,20}$/.test(sku.skuId)) throw new Error("invalid Lazada sku id");
  const imgs = images.map((u) => `<Image>${xml(u)}</Image>`).join("");
  return `<?xml version="1.0" encoding="UTF-8" ?><Request><Product><ItemId>${itemId}</ItemId><Images>${imgs}</Images>`
    + `<Skus><Sku><SkuId>${sku.skuId}</SkuId><SellerSku>${xml(sku.sellerSku)}</SellerSku></Sku></Skus></Product></Request>`;
}
 
/** True when this gallery already shows the licence (Lazada names migrated files by hash). */
export function hasLicence(images: readonly string[], licence: { url: string; hash: string | null }): boolean {
  return images.some((u) => u === licence.url || (!!licence.hash && u.toLowerCase().includes(licence.hash.toLowerCase())));
}
 
export type GalleryPlan = { action: "add"; images: string[] } | { action: "skip"; reason: string };
 
/** Pure: what to send for one listing. */
export function planGallery(current: readonly string[], licence: { url: string; hash: string | null }): GalleryPlan {
  const own = current.filter(Boolean);
  if (own.length === 0) return { action: "skip", reason: "the listing has no images on Lazada" };
  if (hasLicence(own, licence)) return { action: "skip", reason: "already shows the licence" };
  if (own.length >= LAZADA_MAX_IMAGES) return { action: "skip", reason: `already has ${own.length} images (Lazada's maximum is ${LAZADA_MAX_IMAGES}) — remove one in Seller Center first` };
  return { action: "add", images: [...own, licence.url] };
}

/** Pure: did the write do exactly what we meant? Returns the differences (empty = good). */
export function verifyAfter(before: ListingSnapshot, after: ListingSnapshot, licence: { url: string; hash: string | null }): string[] {
  const diffs: string[] = [];
  if (after.images.length !== before.images.length + 1) {
    diffs.push(`gallery has ${after.images.length} images, expected ${before.images.length + 1}`);
  } else {
    if (before.images.some((u, i) => !sameFile(after.images[i], u))) diffs.push("the existing images changed order or content");

    if (!sameFile(last, licence.url) && !hasLicence([last], licence)) diffs.push("the last image is not the licence");

  }

}
