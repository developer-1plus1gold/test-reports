const PRICE_UNIT_VALUES = new Set<PriceUnit>(PRICE_UNITS.map((u) => u.value));
const CURRENCY_VALUES   = new Set<string>(ALL_PREF_CURRENCIES);
 
/** A validated, snake_case patch ready to write to `profiles`. */
export interface PreferencesPatch {
  timezone?:             string;
  price_unit?:           PriceUnit;
  preferred_currencies?: string[];
}
 
/**
 * Validate an inbound PATCH body. Accepts any subset of the three scalars.
 * Returns `{ patch }` with only the supplied, valid keys, or `{ error }`
 * with a human-readable message (caller returns it as a 400).
 *
 * Rules:
 *   • timezone             — must be one of PREF_TIMEZONES
 *   • price_unit           — must be one of PRICE_UNITS
 *   • preferred_currencies — array of known codes, de-duplicated, length ≥ 1
 *     (the PWA blocks an empty selection — account/page.tsx:3193)
 */
export function validatePreferencesPatch(body: unknown): { patch: PreferencesPatch } | { error: string } {
  if (typeof body !== "object" || body === null) {
    return { error: "Body must be a JSON object" };
  }
  const b = body as Record<string, unknown>;
  const patch: PreferencesPatch = {};
 
  if ("timezone" in b && b.timezone !== undefined) {
    if (typeof b.timezone !== "string" || !TIMEZONE_VALUES.has(b.timezone)) {
      return { error: "Invalid timezone" };
    }
    patch.timezone = b.timezone;
  }
 
  if ("price_unit" in b && b.price_unit !== undefined) {
    if (typeof b.price_unit !== "string" || !PRICE_UNIT_VALUES.has(b.price_unit as PriceUnit)) {
      return { error: "price_unit must be one of: gram, oz, kg" };
    }
    patch.price_unit = b.price_unit as PriceUnit;
  }
 
  if ("preferred_currencies" in b && b.preferred_currencies !== undefined) {
    if (!Array.isArray(b.preferred_currencies)) {
      return { error: "preferred_currencies must be an array" };
    }
    const codes = b.preferred_currencies.filter((c): c is string => typeof c === "string");
    for (const c of codes) {
      if (!CURRENCY_VALUES.has(c)) return { error: `Unknown currency: ${c}` };
    }
    const unique = Array.from(new Set(codes));
    if (unique.length === 0) {
      return { error: "Select at least one currency" };
    }
    patch.preferred_currencies = unique;
  }
 
  if (Object.keys(patch).length === 0) {
    return { error: "No valid preference fields supplied" };
  }
  return { patch };
}
