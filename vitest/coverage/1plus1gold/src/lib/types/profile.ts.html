/**
 * Account → Profile: the cross-platform DATA CONTRACT for the editable
 * profile-identity fields on the `profiles` row.
 *
 * Single source of truth for the REST endpoint /api/profile (server) + the
 * SDUI Profile spec + the PWA (which writes these via the browser supabase
 * client today — no API, so iOS couldn't reach them; mirrors the Q3
 * /api/account/preferences pattern).
 *
 * NOTE — `phone` is intentionally NOT writable here. Changing the WhatsApp
 * number is a verification flow (`/api/auth/link-phone` → OTP on the new
 * number), not a bulk field save. GET returns it for display; PATCH ignores it.
 */
 
import { COUNTRY_CODE_BY_NAME } from "./countries";
 
/** A pending WhatsApp-number change awaiting the 48h review window. */
export interface PendingPhoneChange {
  new_phone:  string;
  expires_at: string;   // ISO — when the review window closes
}
 
/** Profile identity as returned by GET /api/profile. */
export interface Profile {
  full_name:     string | null;
  phone:         string | null;   // display-only here; changed via link-phone
  address_line1: string | null;
  address_line2: string | null;
  city:          string | null;
  country:       string | null;   // ISO 3166-1 alpha-2 (e.g. "SG")
  /** The authed user's email (auth.users.email). Synthetic phone-OTP users
   *  carry phone_<digits>@auth.1plus1gold.com — see email_is_synthetic. */
  email:         string | null;
  /** True when `email` is the synthetic phone-OTP placeholder (no real email
   *  on file). Drives the EmailRow "Add" vs "Change" affordance on iOS. */
  email_is_synthetic: boolean;
  /** Non-null when a WhatsApp-number change is pending review. Drives the
   *  in-surface pending banner (was bootstrap-only before Q4). */
  pending_phone_change: PendingPhoneChange | null;
  /** Presentation convenience: the address fields filtered + comma-joined
   *  exactly as the PWA view renders them (line1, line2, city, country-name),
   *  or null when empty. Lets the SDUI view show one "Address" value with a
   *  byte-exact "Not set" fallback (the comma-join + country-name lookup can't
   *  be expressed in SDUI predicates). Raw fields are still returned for edit. */
  address_display: string | null;
}
 
/** Fields a client may PATCH. Phone is server-controlled (link-phone). */

 * Returns `{ patch }` (only supplied keys) or `{ error }`. Lenient: unknown
 * keys (incl. `phone`) are ignored. Country is upper-cased to a 2-letter code.

          // to upsert `profiles` unvalidated, so on PROD 93 of 115 rows store
          // "Singapore". The country <select> lists ISO codes as its option
          // values, so such a row matches no option, onChange never fires, and
          // the name is sent straight back — which a strict 2-letter check
          // rejected with a 400, making the profile unsaveable for 81% of
          // customers. Accept the name and normalise it, so the row heals on
          // the owner's first save. Anything we cannot resolve is still an error.
          const byName = COUNTRY_CODE_BY_NAME[raw.toLowerCase()];
          if (!byName) return { error: "country must be a 2-letter ISO code" };
          patch.country = byName;
        }

  }
 
  if (Object.keys(patch).length === 0) return { error: "No valid profile fields supplied" };
  return { patch };
}
