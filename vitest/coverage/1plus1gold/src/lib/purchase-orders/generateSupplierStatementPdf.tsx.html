/**
 * Supplier statement PDF — one document covering several purchase orders, with
 * every line grouped under its PO (Mukesh, 2026-10-27: "a single statement of
 * all the PO lines (under each PO) ... the 2% as a separate col ... and how did
 * we arrive at total of each line"; then "add the supplier statement button").
 *
 * Same split as the PO PDF:
 *   - `buildSupplierStatementModel()` is PURE — it turns raw rows into what the
 *     sheet prints, so the per-line arithmetic and the serial assignment are
 *     unit-testable without rendering.
 *   - `generateSupplierStatementPdf()` renders it through the shared brand chrome.
 *
 * Processing fee — two cases, because both exist in the data:
 *   - Header `processing_fee_pct > 0`: the fee sits on top of the stored unit
 *     cost, so unit price = unit × (1 + p/100).
 *   - Header pct = 0 but the fee was folded INTO the unit cost
 *     (`foldProcessingFeeIntoLines`): the operator states the folded % and the
 *     sheet splits it back out, base = unit / (1 + f/100). With f = 0 the fee
 *     column simply reads 0.00 — nothing is invented.
 *
 * Serials are the MANUFACTURER bar serials (`inventory.serial_number`), never
 * the NFC serial — the supplier knows the bar by what is stamped on it.
 *
 * Dates + settlement (Mukesh, 2026-11-03: "add dates to the statement"): every
 * PO carries its order date, received date and the payments RECORDED against it
 * (ledger po_payment rows, or consignment supplier settlements), each dated. The
 * sheet opens with an account summary — one row per PO with total, paid, paid on
 * and balance — so a supplier sees what is settled and what is still open.
 */

/**
 * Received dates + recorded payments per PO, from the three sources. Ordinary
 * PO: receipts + ledger `po_payment` rows. Consignment PO: per-bar
 * `consignment_procurements` (confirmed_at, supplier_paid_*; the amount falls
 * back to the confirmed cost, as in lib/finance/consignment-payments). The two
 * never overlap — consignment never writes a po_payment row. Cancelled bars are
 * skipped.
 */
export function collectStatementDatesAndPayments(src: {
  receipts: { po_id: string; received_at: string | null }[];
  consignment: {
    purchase_order_id: string; status: string | null; confirmed_at: string | null;
    supplier_paid_at: string | null; supplier_paid_amount_sgd: number | string | null; confirmed_cost_sgd: number | string | null;
  }[];
  ledger: { related_po_id: string; entry_date: string | null; amount_sgd: number | string | null }[];
}): { receivedDates: Map<string, (string | null)[]>; payments: Map<string, StatementRawPayment[]> } {
  const receivedDates = new Map<string, (string | null)[]>();
  const payments = new Map<string, StatementRawPayment[]>();
  const add = <T,>(m: Map<string, T[]>, k: string, v: T) => m.set(k, [...(m.get(k) ?? []), v]);
  for (const r of src.receipts) add(receivedDates, r.po_id, r.received_at);
  for (const c of src.consignment) {
    if (c.status === "cancelled") continue;
    add(receivedDates, c.purchase_order_id, c.confirmed_at);
    if (c.supplier_paid_at) {
      add(payments, c.purchase_order_id, { date: c.supplier_paid_at, amountSgd: c.supplier_paid_amount_sgd ?? c.confirmed_cost_sgd });
    }
  }
  for (const l of src.ledger) add(payments, l.related_po_id, { date: l.entry_date, amountSgd: l.amount_sgd });
  return { receivedDates, payments };
}
 
/** "2026-11-03" or an ISO timestamp → the SGT calendar date, YYYY-MM-DD. */
export function sgtDate(v: string | null | undefined): string | null {
  if (!v) return null;
  if (/^\d{4}-\d{2}-\d{2}$/.test(v)) return v;
  const t = Date.parse(v);
  if (!Number.isFinite(t)) return null;
  return new Date(t + 8 * 3600_000).toISOString().slice(0, 10);
}
 
const MONTHS = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
/** YYYY-MM-DD → "3 Nov 2026"; null → "—". */
export function fmtStatementDate(d: string | null | undefined): string {
  if (!d) return "—";
  const [y, m, day] = d.split("-").map(Number);
  return `${day} ${MONTHS[m - 1]} ${y}`;
}
 
const r2 = (n: number) => Math.round(n * 100) / 100;
const num = (v: unknown): number | null => {
  if (v === null || v === undefined || v === "") return null;
  const n = Number(v);
  return Number.isFinite(n) ? n : null;
};
const fmtPct = (p: number) => `${Number(p.toFixed(4))}%`;
const fmtFactor = (p: number) => String(Number((1 + p / 100).toFixed(6)));
 
export function buildSupplierStatementModel(input: {
  pos: StatementRawPo[];
  /** % already folded into unit costs on POs whose header pct is 0. */
  foldedFeePct: number;
  /** Per-PO override of `foldedFeePct`, keyed by po_number — for a PO whose
   *  prices were recorded WITHOUT the folded fee (0), so the sheet does not split
   *  a fee out of a price that never contained one. */
  foldedFeePctByPo?: Record<string, number>;
  /** Payments already made against these POs (Mukesh, 2026-11-02). */
  payments?: StatementPayment[];
  generatedAt?: string;
}): StatementModel {
  const folded = Number.isFinite(input.foldedFeePct) && input.foldedFeePct > 0 ? input.foldedFeePct : 0;
  const pcts = new Set<number>();
 
  const sections: StatementSection[] = input.pos.map((po) => {
    const headerPct = poProcessingFeePct(po.header);
    const override = input.foldedFeePctByPo?.[po.header.po_number];
    const poFolded = override != null && Number.isFinite(override) && override >= 0 ? override : folded;
    const headerSpot = num(po.header.spot_sgd_at_creation);
    // Each unit is assigned to at most one line.
    const pool = po.units.map((u) => ({ ...u, used: false }));
    const lines = [...po.lines].sort((a, b) => (a.sort_order ?? 0) - (b.sort_order ?? 0));
 
    const rows: StatementRow[] = lines.map((l) => {
      const qty = Number(l.quantity_ordered ?? 0);
      const grams = num(l.product?.grams) ?? 0;
      const unit = num(l.unit_cost_sgd) ?? num(l.unit_cost) ?? 0;
 
      let base: number, fee: number, unitPrice: number, pct: number;
      if (headerPct > 0) {
        pct = headerPct;
        base = r2(unit);
        fee = r2(unit * headerPct / 100);
        unitPrice = r2(base + fee);
      } else {
        pct = poFolded;
        unitPrice = r2(unit);
        base = r2(unit / (1 + poFolded / 100));
        fee = r2(unitPrice - base);
      }
      pcts.add(pct);
 
      const consignSpot = pool.find((u) => u.product_id === l.product_id && num(u.supplier_spot_sgd_per_g) !== null);
      const spot = num(l.spot_sgd_per_g) ?? num(consignSpot?.supplier_spot_sgd_per_g) ?? headerSpot;
      const gold = spot !== null && grams > 0 ? r2(spot * grams) : null;
      const premium = gold !== null ? r2(base - gold) : null;
 
      // Serials: same product, closest unit cost first, serial as tie-break.
      const picked = pool
        .filter((u) => !u.used && u.product_id === l.product_id && u.serial_number)
        .sort((a, b) =>
          Math.abs((num(a.cost_per_g_sgd) ?? 0) * grams - unit) - Math.abs((num(b.cost_per_g_sgd) ?? 0) * grams - unit)
          || String(a.serial_number).localeCompare(String(b.serial_number)))
        .slice(0, Math.max(0, qty));
      picked.forEach((u) => { u.used = true; });
      const serials = picked.map((u) => String(u.serial_number)).sort().join(", ") || "—";
 
      const factor = fmtFactor(pct);
      const how = gold !== null && premium !== null
        ? `(${fmtSgd(gold)} + ${fmtSgd(premium)}) × ${factor} × ${qty}`
        : `${fmtSgd(base)} × ${factor} × ${qty}`;
 
      return {
        label: l.product?.label ?? "—",
        grams, serials, qty, spot, gold, premium, base, fee, unitPrice,
        feeFactor: 1 + pct / 100,
        lineTotal: r2(unitPrice * qty),
        how,
      };
    });
 
    const total = r2(rows.reduce((s, r) => s + r.lineTotal, 0));
    // Same-day payments on one PO read as one line (consignment settles per bar).
    const byDate = new Map<string, number>();
    for (const p of po.payments ?? []) {
      const amt = num(p.amountSgd);
      if (amt === null || amt <= 0) continue;
      const d = sgtDate(p.date) ?? "";
      byDate.set(d, r2((byDate.get(d) ?? 0) + amt));
    }
    const payments = [...byDate.entries()]
      .map(([d, amountSgd]) => ({ date: d || null, amountSgd }))
      .sort((a, b) => String(a.date ?? "").localeCompare(String(b.date ?? "")));
    const paid = r2(payments.reduce((s, p) => s + p.amountSgd, 0));
    const balance = r2(total - paid);
    const received = (po.receivedDates ?? []).map(sgtDate).filter((d): d is string => !!d).sort();
 
    return {
      poNumber: po.header.po_number,
      supplierName: po.header.supplier_name ?? null,
      rows,
      bars: rows.reduce((s, r) => s + r.qty, 0),
      feeTotal: r2(rows.reduce((s, r) => s + r.fee * r.qty, 0)),
      total,
      orderDate: sgtDate(po.header.order_date) ?? sgtDate(po.header.created_at),
      receivedDate: received.length ? received[received.length - 1] : null,
      payments,
      paid,
      balance,
      // Within a cent counts as settled (per-bar rounding on consignment).
      status: balance < 0.01 ? "settled" : paid > 0 ? "part-paid" : "open",
    };
  });
 
  const grandTotal = r2(sections.reduce((s, x) => s + x.total, 0));
  const recorded: StatementPayment[] = sections.flatMap((sec) =>
    sec.payments.map((p) => ({ label: `${sec.poNumber} · paid ${fmtStatementDate(p.date)}`, amountSgd: p.amountSgd, date: p.date })));
  const extra = (input.payments ?? [])
    .filter((p) => Number.isFinite(p.amountSgd) && p.amountSgd > 0)
    .map((p) => ({ label: p.label.trim() || "Payment received", amountSgd: r2(p.amountSgd), date: p.date ?? null }));
  const payments = [...recorded, ...extra];
  const paymentsTotal = r2(payments.reduce((s, p) => s + p.amountSgd, 0));
 
  const suppliers = [...new Set(sections.map((s) => s.supplierName).filter(Boolean))] as string[];
  const feeLabel = pcts.size === 1 ? `${fmtPct([...pcts][0])} FEE` : "FEE";
  return {
    sections,
    supplierLine: suppliers.length ? suppliers.join(", ") : "Supplier",
    feeLabel,
    bars: sections.reduce((s, x) => s + x.bars, 0),
    feeTotal: r2(sections.reduce((s, x) => s + x.feeTotal, 0)),
    grandTotal,
    payments,
    paymentsTotal,
    balanceDue: r2(grandTotal - paymentsTotal),
    generatedAt: input.generatedAt ?? new Date().toISOString(),
  };
}

export async function generateSupplierStatementPdf(m: StatementModel): Promise<Buffer> {
  const cols: { h: string; w: number; right?: boolean; v: (r: StatementRow) => string }[] = [
    { h: "BAR", w: 15, v: (r) => `${r.label} (${gramsLabel(r.grams)})` },
    { h: "BAR SERIAL(S)", w: 15, v: (r) => r.serials },
    { h: "QTY", w: 4, right: true, v: (r) => String(r.qty) },
    { h: "SPOT / G", w: 7, right: true, v: (r) => (r.spot !== null ? r.spot.toFixed(4) : "—") },
    { h: "GOLD VALUE", w: 8, right: true, v: (r) => (r.gold !== null ? fmtSgd(r.gold) : "—") },
    { h: "PREMIUM", w: 7, right: true, v: (r) => (r.premium !== null ? fmtSgd(r.premium) : "—") },
    { h: "BASE", w: 8, right: true, v: (r) => fmtSgd(r.base) },
