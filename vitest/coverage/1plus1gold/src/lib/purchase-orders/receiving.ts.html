/**
 * Remove `quantity` received units of a product belonging to `receiptId` —
 * deletes that many *available* inventory rows for eligible products, or
 * decrements products.stock_qty for accessories. Returns a reason string if it
 * cannot (units already consumed / stock already drawn down) so the caller can
 * block; in that case NOTHING is mutated.
 */
export async function removeReceivedUnits(
  client: Client,
  opts: {

    productId: string;
    product:   ReceivingProduct;
    quantity:  number;
  },
): Promise<{ ok: true } | { ok: false; reason: string }> {

      .from("inventory")
      .select("id, status")
      .eq("source_type", "purchase_order")
      .eq("source_id", receiptId)
      .eq("product_id", productId);
    const available = (rows ?? []).filter((r: any) => r.status === "available");
    if (available.length < quantity) {
      const consumed = (rows ?? []).length - available.length;
      return {
        ok: false,
        reason: `${consumed} of these received unit(s) are already sold/allocated — un-allocate them before reducing or deleting this receipt.`,
      };
    }
    const toDelete = available.slice(0, quantity).map((r: any) => r.id);
    await client.from("inventory").delete().in("id", toDelete);

  const { data: prod } = await client.from("products").select("stock_qty").eq("id", productId).single();
  const current = Number(prod?.stock_qty ?? 0);
  if (current < quantity) {

  }
  await client.from("products").update({ stock_qty: current - quantity }).eq("id", productId);
  return { ok: true };
}
 
/**

 * every product can be reversed (no consumed units / no over-drawn accessory
 * stock) BEFORE mutating anything, so a block leaves the receipt untouched.
 * Does NOT delete the receipt/receipt-line rows — the caller does that after.
 *

 * carry that serial, and re-recording the receipt reuses it via issueNfcSerial.

      if (available < qty) {
        return { ok: false, reason: `${(rows ?? []).length - available} received unit(s) are already sold/allocated — un-allocate them first.` };
      }
    } else {
      const { data: prod } = await client.from("products").select("stock_qty").eq("id", productId).single();
      if (Number(prod?.stock_qty ?? 0) < qty) {
        return { ok: false, reason: `Some of this accessory was already sold — stock would go negative; cannot reverse.` };
      }
    }

/** Recompute PO status from total received vs ordered across all lines. */
export async function updatePoReceivedStatus(client: Client, poId: string): Promise<void> {
  const { data: lines } = await client
    .from("purchase_order_lines")
    .select("id, quantity_ordered")
    .eq("po_id", poId);
  if (!lines?.length) return;
 
  const totalOrdered = lines.reduce((s: number, l: any) => s + Number(l.quantity_ordered), 0);
  let totalReceived = 0;
  for (const line of lines) {
    const { data: rls } = await client
      .from("purchase_order_receipt_lines")
      .select("quantity_received")
      .eq("po_line_id", line.id);
    totalReceived += (rls ?? []).reduce((s: number, r: any) => s + Number(r.quantity_received), 0);
  }
 
  const newStatus = totalReceived === 0 ? "confirmed" : totalReceived < totalOrdered ? "partially_received" : "fully_received";
  await client.from("purchase_orders").update({ status: newStatus }).eq("id", poId);
}
