/**
 * Shared purchase-order total math so the list card, the detail summary, and
 * any future surface always agree. Line total = Σ(unit_cost_sgd × qty); grand
 * total adds the header costs (shipping + payment + insurance + other), all
 * stored in SGD on the PO header, plus the processing fee (a % of the goods
 * value, so its amount is derived here rather than stored). landed_cost_sgd is NOT used here — it already
 * bakes in allocated freight, so summing it AND the header fees double-counts.
 */
 
export interface PoTotalsLine {
  unit_cost_sgd:    number | string | null;
  quantity_ordered: number | string | null;
}
 
export interface PoTotalsHeader {
  shipping_fee?:  number | string | null;
  payment_fee?:   number | string | null;
  insurance_fee?: number | string | null;
  other_costs?:   number | string | null;
  /** Processing fee as a % of the goods value. The S$ amount is derived. */
  processing_fee_pct?: number | string | null;
}
 
export function poLineTotalSgd(lines: PoTotalsLine[] | null | undefined): number {
  return (lines ?? []).reduce(
    (s, l) => s + Number(l.unit_cost_sgd ?? 0) * Number(l.quantity_ordered ?? 0),
    0,
  );
}
 
export function poHeaderCostSgd(po: PoTotalsHeader): number {
  return (
    Number(po.shipping_fee ?? 0) +
    Number(po.payment_fee ?? 0) +
    Number(po.insurance_fee ?? 0) +
    Number(po.other_costs ?? 0)
  );
}
 
/** A usable processing-fee percent (0 when unset or invalid). */
export function poProcessingFeePct(po: PoTotalsHeader): number {
  const p = Number(po.processing_fee_pct ?? 0);
  return Number.isFinite(p) && p > 0 ? p : 0;
}
 
/** S$ processing fee = goods value × %, rounded to the cent. */
export function poProcessingFeeSgd(
  po: PoTotalsHeader,
  lines: PoTotalsLine[] | null | undefined,
): number {
  const pct = poProcessingFeePct(po);
  return pct > 0 ? Math.round(poLineTotalSgd(lines) * pct) / 100 : 0;
}
 
export function poGrandTotalSgd(
  po: PoTotalsHeader,
  lines: PoTotalsLine[] | null | undefined,
): number {
  return poLineTotalSgd(lines) + poHeaderCostSgd(po) + poProcessingFeeSgd(po, lines);
}
 
export interface FoldableLine {
  id:            string;
  unit_cost:     number | string | null;
  unit_cost_sgd: number | string | null;
}
 
/**
 * Spread a header processing fee into the lines: each line's unit cost is
 * marked up by the same percent, so the fee becomes part of each bar's cost
 * (its own landed cost, premium, and the supplier's PO PDF) instead of one
 * header line. The header percent then goes to 0. Grand total is preserved
 * up to rounding (unit costs are stored to 4 dp).
 *
 * unit_cost_sgd is re-derived from the folded unit_cost × fx — the same rule
 * the line PATCH route uses — so the two columns cannot disagree.
 */
export function foldProcessingFeeIntoLines(
  lines: FoldableLine[],
  pct: number,
  fxRateToSgd: number,
): { id: string; unit_cost: number; unit_cost_sgd: number }[] {
  if (!(Number.isFinite(pct) && pct > 0)) return [];
  const factor = 1 + pct / 100;
  const fx = Number.isFinite(fxRateToSgd) && fxRateToSgd > 0 ? fxRateToSgd : 1;
  const r4 = (n: number) => Math.round(n * 10_000) / 10_000;
  return lines.map((l) => {
    const unitCost = r4(Number(l.unit_cost ?? 0) * factor);
    return { id: l.id, unit_cost: unitCost, unit_cost_sgd: r4(unitCost * fx) };
  });
}
