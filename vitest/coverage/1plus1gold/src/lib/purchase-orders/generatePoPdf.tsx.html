/**
 * Purchase-order PDF — a branded, printable copy of one PO (Mukesh, 2026-10-16:
 * "a pdf function for the PO to print the PO").
 *
 * Two halves, deliberately split:
 *   - `buildPoPdfModel()` is PURE. It turns the raw rows (PO header, lines,
 *     receipts, received inventory) into exactly what the sheet prints, so the
 *     arithmetic and the serial grouping are unit-testable without rendering.
 *   - `generatePoPdf()` renders that model through the shared @react-pdf brand
 *     chrome (src/lib/pdf/branding.tsx), like the quote + statements PDFs.
 *
 * Money follows the PO screen exactly: line unit costs are in the PO's purchase
 * currency, the header costs (shipping/payment/insurance/other) are stored in
 * SGD, and the grand total is the same `poLineTotalSgd + poHeaderCostSgd` the
 * detail page shows — so the printout can never disagree with the app.
 */

export function buildPoPdfModel(input: {
  header: PoPdfRawHeader;
  lines: PoPdfRawLine[];
  receipts: PoPdfRawReceipt[];
  units: PoPdfRawUnit[];
  generatedAt: string;
}): PoPdfModel {
  const { header: po, generatedAt } = input;
  const fxRate = num(po.fx_rate_to_sgd) || 1;
  const headerSpot = num(po.spot_sgd_at_creation) > 0 ? num(po.spot_sgd_at_creation) : null;
  const currency = po.purchase_currency || "SGD";
  const processingPct = poProcessingFeePct({ processing_fee_pct: num(po.processing_fee_pct) });
  const inclFactor = 1 + processingPct / 100;
 
  const sorted = [...input.lines].sort((a, b) => num(a.sort_order) - num(b.sort_order));
  const labelByProduct = new Map<string, string>();
  const lines: PoPdfLine[] = sorted.map((l, i) => {
    const qty = num(l.quantity_ordered);
    const unitCost = num(l.unit_cost);
    const grams = l.product?.grams == null ? null : num(l.product.grams);
    const label = l.product?.label?.trim() || "Unlinked item";
    Eif (l.product_id) labelByProduct.set(l.product_id, label);
    const lineSpot = num(l.spot_sgd_per_g) > 0 ? num(l.spot_sgd_per_g) : headerSpot;
    const goldValue = lineSpot != null && grams != null && grams > 0 ? lineSpot * grams : null;
    const premium = goldValue != null ? num(l.unit_cost_sgd) - goldValue : null;
    const inclUnit = num(l.unit_cost_sgd) * inclFactor;
    const inclPremium = goldValue != null ? inclUnit - goldValue : null;
    return {
      no: i + 1,
      label,
      grams,
      qty,
      unitCost,
      lineTotal: unitCost * qty,
      lineTotalSgd: num(l.unit_cost_sgd) * qty,
      spotSgdPerGram: lineSpot,
      premiumPerPieceSgd: premium,
      premiumPct: premium != null && goldValue ? (premium / goldValue) * 100 : null,
      incl: {
        unitCostSgd: inclUnit,
        lineTotalSgd: inclUnit * qty,
        premiumPerPieceSgd: inclPremium,
        premiumPct: inclPremium != null && goldValue ? (inclPremium / goldValue) * 100 : null,
      },
    };
  });
 
  const linesSgd = poLineTotalSgd(
    sorted.map((l) => ({ unit_cost_sgd: num(l.unit_cost_sgd), quantity_ordered: num(l.quantity_ordered) })),
  );
  const headerSgd = poHeaderCostSgd({
    shipping_fee: num(po.shipping_fee), payment_fee: num(po.payment_fee),
    insurance_fee: num(po.insurance_fee), other_costs: num(po.other_costs),
  });
  const processingSgd = poProcessingFeeSgd(
    { processing_fee_pct: processingPct },
    sorted.map((l) => ({ unit_cost_sgd: num(l.unit_cost_sgd), quantity_ordered: num(l.quantity_ordered) })),
  );
 
  // Group received units by receipt, then product — the serials are what an
  // auditor (or the supplier) reconciles the printout against.
  const receipts: PoPdfReceipt[] = [...input.receipts]
    .sort((a, b) => a.received_at.localeCompare(b.received_at))
    .map((r) => {
      const byProduct = new Map<string, string[]>();
      for (const u of input.units) {
        if (u.source_id !== r.id) continue;
        const key = u.product_id ?? "";
        const list = byProduct.get(key) ?? [];
        list.push((u.serial_number ?? "").trim() || "(no serial)");
        byProduct.set(key, list);
      }
      const items = [...byProduct.entries()].map(([pid, serials]) => ({
        label: labelByProduct.get(pid) ?? "Item",
        serials: [...serials].sort(),
        count: serials.length,
      }));
      return { receivedAt: r.received_at, notes: r.notes?.trim() || null, items };
    });
 
  const spots = lines.map((l) => l.spotSgdPerGram).filter((v): v is number => v != null);
  const minSpot = spots.length ? Math.min(...spots) : null;
  const maxSpot = spots.length ? Math.max(...spots) : null;
  const mixedSpots = minSpot != null && maxSpot != null && Math.abs(maxSpot - minSpot) > 1e-6;
  const priced = lines.filter((l) => l.premiumPerPieceSgd != null && l.spotSgdPerGram != null && l.grams);
  const premiumSgd = priced.length ? priced.reduce((acc, l) => acc + (l.premiumPerPieceSgd as number) * l.qty, 0) : null;
  const goldSgd = priced.reduce((acc, l) => acc + (l.spotSgdPerGram as number) * (l.grams as number) * l.qty, 0);
  // Premium incl. = the premium plus the processing fee those priced lines carry.
  const pricedGoods = priced.reduce((acc, l) => acc + l.lineTotalSgd, 0);
  const premiumInclSgd = premiumSgd != null ? premiumSgd + (pricedGoods * processingPct) / 100 : null;
 
  return {
    poNumber: po.po_number,
    supplierName: po.supplier_name,
    supplierContact: po.supplier_contact?.trim() || null,
    status: po.status,
    statusLabel: STATUS_LABELS[po.status] ?? humanise(po.status),
    paymentLabel: PAYMENT_LABELS[po.payment_status] ?? humanise(po.payment_status),
    paymentReference: po.payment_reference?.trim() || null,
    currency,
    fxRate,
    isNonSgd: currency !== "SGD",
    orderDate: po.order_date ?? null,
    createdAt: po.created_at,
    estimatedDelivery: po.estimated_delivery_date ?? null,
    spotSgdPerGram: mixedSpots ? null : (minSpot ?? headerSpot),
    spotRange: mixedSpots ? { min: minSpot as number, max: maxSpot as number } : null,
    notes: po.notes?.trim() || null,
    lines,
    totals: {
      pieces: lines.reduce((s, l) => s + l.qty, 0),
      grams: lines.reduce((s, l) => s + (l.grams ?? 0) * l.qty, 0),
      linesNative: lines.reduce((s, l) => s + l.lineTotal, 0),
      linesSgd,
      premiumSgd,
      premiumPct: premiumSgd != null && goldSgd > 0 ? (premiumSgd / goldSgd) * 100 : null,
      shippingSgd: num(po.shipping_fee),
      paymentSgd: num(po.payment_fee),
      insuranceSgd: num(po.insurance_fee),
      otherSgd: num(po.other_costs),
      headerSgd,
      processingPct,
      processingSgd,
      linesInclSgd: linesSgd + processingSgd,
      premiumInclSgd,
      premiumInclPct: premiumInclSgd != null && goldSgd > 0 ? (premiumInclSgd / goldSgd) * 100 : null,
      grandTotalSgd: linesSgd + headerSgd + processingSgd,
    },
    receipts,
    generatedAt,
  };
}

  { key: "grams", header: "Weight", flex: 0.75, align: "right" as const },
  { key: "qty", header: "Qty", flex: 0.45, align: "right" as const },
  { key: "spot", header: "Spot S$/g", flex: 0.95, align: "right" as const },
  { key: "prem", header: "Premium/pc", flex: 1.05, align: "right" as const },
  { key: "premPct", header: "Prem %", flex: 0.7, align: "right" as const },
  { key: "unit", header: "Unit cost", flex: 1.2, align: "right" as const },
  { key: "total", header: "Line total", flex: 1.3, align: "right" as const },
];
/** 2dp, unless the spot carries more precision (consignment-day spots are 4dp). */
const fmtSpot = (v: number) => v.toFixed(2);
const fmtPct = (p: number | null) => (p == null ? "—" : `${p.toFixed(2)}%`);
/** Processing-fee label: trims trailing zeros (2% not 2.000%). */
const fmtPctPlain = (p: number) => `${Number(p.toFixed(3))}%`;
 
function LinesTable({ m, cols, cellOf, totalOf }: {
  m: PoPdfModel;
  cols: { key: string; header: string; flex: number; align: "left" | "right" }[];
  cellOf: (l: PoPdfLine, key: string) => string;
  totalOf: (key: string) => string;
}) {
  return (
    <View>
      <View style={s.headRow}>
        {cols.map((c) => <Text key={c.key} style={{ ...s.headCell, flex: c.flex, textAlign: c.align }}>{c.header}</Text>)}
      </View>
      {m.lines.map((l) => (
        <View key={l.no} style={s.row} wrap={false}>
          {cols.map((c) => <Text key={c.key} style={{ ...s.cell, flex: c.flex, textAlign: c.align }}>{cellOf(l, c.key)}</Text>)}
        </View>
      ))}
      <View style={s.totalRow} wrap={false}>
        {cols.map((c) => <Text key={c.key} style={{ ...s.totalCell, flex: c.flex, textAlign: c.align }}>{totalOf(c.key)}</Text>)}
      </View>
    </View>
  );
}
 
function PoDoc({ m }: { m: PoPdfModel }) {
  const cols = m.isNonSgd ? [...COLS, { key: "sgd", header: "Line total (SGD)", flex: 1.5, align: "right" as const }] : COLS;
  const cellOf = (l: PoPdfLine, key: string): string => {
    switch (key) {
      case "no": return String(l.no);
      case "item": return l.label;
      case "grams": return l.grams == null ? "—" : fmtGrams(l.grams);
      case "qty": return String(l.qty);
      case "spot": return l.spotSgdPerGram == null ? "—" : fmtSpot(l.spotSgdPerGram);
      case "prem": return l.premiumPerPieceSgd == null ? "—" : fmtSgd(l.premiumPerPieceSgd);
      case "premPct": return fmtPct(l.premiumPct);
      case "unit": return fmtMoney(l.unitCost, m.currency);
      case "total": return fmtMoney(l.lineTotal, m.currency);
      case "sgd": return fmtSgd(l.lineTotalSgd);
      default: return "";
    }
  };
  const t = m.totals;
  const totalOf = (key: string): string => {
    switch (key) {
      case "item": return "Total";
      case "grams": return fmtGrams(t.grams);
      case "qty": return String(t.pieces);
      case "prem": return t.premiumSgd == null ? "—" : fmtSgd(t.premiumSgd);
      case "premPct": return fmtPct(t.premiumPct);
      case "total": return fmtMoney(t.linesNative, m.currency);
      case "sgd": return fmtSgd(t.linesSgd);
      default: return "";
    }
  };
  const headerCosts: [string, number][] = [
    [`Processing fee (${fmtPctPlain(t.processingPct)})`, t.processingSgd],
    ["Shipping", t.shippingSgd], ["Payment fee", t.paymentSgd],
    ["Insurance", t.insuranceSgd], ["Other costs", t.otherSgd],
  ].filter(([, v]) => (v as number) !== 0) as [string, number][];
  const hasProcessing = t.processingPct > 0;
  const inclCols = COLS.map((c) =>
    c.key === "prem" ? { ...c, header: "Premium/pc incl." }
      : c.key === "premPct" ? { ...c, header: "Prem % incl." }
      : c.key === "unit" ? { ...c, header: "Unit cost incl." }
      : c.key === "total" ? { ...c, header: "Line total incl." }
      : c);
  const inclCellOf = (l: PoPdfLine, key: string): string => {
    switch (key) {
      case "prem": return l.incl.premiumPerPieceSgd == null ? "—" : fmtSgd(l.incl.premiumPerPieceSgd);
      case "premPct": return fmtPct(l.incl.premiumPct);
      case "unit": return fmtSgd(l.incl.unitCostSgd);
      case "total": return fmtSgd(l.incl.lineTotalSgd);
      default: return cellOf(l, key);
    }
  };
  const inclTotalOf = (key: string): string => {
    switch (key) {
      case "prem": return t.premiumInclSgd == null ? "—" : fmtSgd(t.premiumInclSgd);
      case "premPct": return fmtPct(t.premiumInclPct);
      case "total": return fmtSgd(t.linesInclSgd);
      default: return totalOf(key);
    }
  };
  const footer = (
    <View style={{ position: "absolute", bottom: 0, left: 0, right: 0 }} fixed>
      <BrandFooter brandLine="1PLUS1GOLD — PURCHASE ORDER" trailing="1plus1gold.com" />
    </View>
  );
 
  return (
    <Document title={`Purchase Order ${m.poNumber}`} author="1Plus1Gold">
      <Page size="A4" style={s.page}>
        <BrandHeader title="PURCHASE ORDER" subtitle={m.poNumber} generatedAt={m.generatedAt} logoSize={22} />
        <View style={s.body}>
          {m.status === "cancelled" ? <Text style={s.cancelled}>THIS PURCHASE ORDER IS CANCELLED</Text> : null}
 
          <View style={s.parties}>
            <View style={s.party}>
              <Text style={s.partyLbl}>BUYER</Text>
              <Text style={s.partyName}>{BUSINESS_CONFIG.name}</Text>
              <Text style={s.partyText}>UEN {BUSINESS_CONFIG.uen} · {BUSINESS_CONFIG.jurisdiction}</Text>
              <Text style={s.partyText}>{BUSINESS_CONFIG.customerWhatsapp.e164} · 1plus1gold.com</Text>
            </View>
            <View style={s.party}>
              <Text style={s.partyLbl}>SUPPLIER</Text>
              <Text style={s.partyName}>{m.supplierName}</Text>
              {m.supplierContact ? <Text style={s.partyText}>{m.supplierContact}</Text> : null}
            </View>
          </View>
 
          <View style={s.metaRow}>
            <View style={s.metaCell}><Text style={s.metaLbl}>PO NUMBER</Text><Text style={s.metaVal}>{m.poNumber}</Text></View>
            <View style={s.metaCell}><Text style={s.metaLbl}>ORDER DATE</Text><Text style={s.metaVal}>{fmtIsoDate(m.orderDate ?? m.createdAt)}</Text></View>
            {m.estimatedDelivery ? (
              <View style={s.metaCell}><Text style={s.metaLbl}>EST. DELIVERY</Text><Text style={s.metaVal}>{fmtIsoDate(m.estimatedDelivery)}</Text></View>
            ) : null}
            <View style={s.metaCell}><Text style={s.metaLbl}>STATUS</Text><Text style={s.metaVal}>{m.statusLabel}</Text></View>
            <View style={s.metaCell}><Text style={s.metaLbl}>PAYMENT</Text><Text style={s.metaVal}>{m.paymentLabel}{m.paymentReference ? ` · ${m.paymentReference}` : ""}</Text></View>
            <View style={s.metaCell}><Text style={s.metaLbl}>CURRENCY</Text><Text style={s.metaVal}>{m.currency}{m.isNonSgd ? ` @ ${m.fxRate.toFixed(4)} SGD` : ""}</Text></View>
            {m.spotSgdPerGram != null ? (
              <View style={s.metaCell}><Text style={s.metaLbl}>SPOT S$/g</Text><Text style={s.metaVal}>{m.spotSgdPerGram.toFixed(2)}</Text></View>
            ) : m.spotRange ? (
              <View style={s.metaCell}><Text style={s.metaLbl}>SPOT S$/g</Text><Text style={s.metaVal}>{m.spotRange.min.toFixed(2)} – {m.spotRange.max.toFixed(2)} (per line)</Text></View>
            ) : null}
          </View>
 
          <Text style={s.sectionLbl}>ORDER LINES</Text>
          <LinesTable m={m} cols={cols} cellOf={cellOf} totalOf={totalOf} />
 
          <View style={s.sumWrap} wrap={false}>
            {m.isNonSgd ? (
              <View style={s.sumRow}><Text style={s.sumLbl}>Lines ({m.currency})</Text><Text style={s.sumVal}>{fmtMoney(t.linesNative, m.currency)}</Text></View>
            ) : null}
            <View style={s.sumRow}><Text style={s.sumLbl}>Lines (SGD)</Text><Text style={s.sumVal}>{fmtSgd(t.linesSgd)}</Text></View>
            {headerCosts.map(([lbl, v]) => (
              <View key={lbl} style={s.sumRow}><Text style={s.sumLbl}>{lbl}</Text><Text style={s.sumVal}>{fmtSgd(v)}</Text></View>
            ))}
            <View style={s.grandRow}><Text style={s.grandLbl}>Total (SGD)</Text><Text style={s.grandVal}>{fmtSgd(t.grandTotalSgd)}</Text></View>
          </View>
 
          {m.receipts.length > 0 ? (
            <View>
              <Text style={s.sectionLbl}>RECEIVED</Text>
              {m.receipts.map((r, i) => (
                <View key={i} wrap={false}>
                  <Text style={s.receiptHead}>{fmtIsoDate(r.receivedAt)}{r.notes ? ` — ${r.notes}` : ""}</Text>
                  {r.items.length === 0 ? <Text style={s.receiptItem}>No units recorded.</Text> : null}
                  {r.items.map((it, j) => (
                    <Text key={j} style={s.receiptItem}>{it.label} × {it.count}: {it.serials.join(", ")}</Text>
                  ))}
                </View>
              ))}
            </View>
          ) : null}
 
          {m.notes ? (
            <View wrap={false}>
              <Text style={s.sectionLbl}>NOTES</Text>
              <Text style={s.note}>{m.notes}</Text>
            </View>
          ) : null}
 
          <View style={s.signRow} wrap={false}>
            <View style={s.signBox}><View style={s.signLine} /><Text style={s.signLbl}>Authorised by — 1Plus1Gold</Text></View>
            <View style={s.signBox}><View style={s.signLine} /><Text style={s.signLbl}>Acknowledged by — {m.supplierName}</Text></View>
          </View>
        </View>
 
        {/* Pinned to the page bottom so a short PO doesn't float the footer mid-page. */}
        {footer}
      </Page>
 
      {hasProcessing ? (
        <Page size="A4" style={s.page}>
          <BrandHeader title="PURCHASE ORDER" subtitle={`${m.poNumber} · incl. processing fee`} generatedAt={m.generatedAt} logoSize={22} />
          <View style={s.body}>
            <Text style={s.sectionLbl}>ORDER LINES — INCLUSIVE OF PROCESSING FEE ({fmtPctPlain(t.processingPct)})</Text>
            <Text style={{ ...s.note, marginBottom: 6 }}>
              The same lines as page 1, with the {fmtPctPlain(t.processingPct)} processing fee added to each line pro rata to its value. Amounts in SGD.
            </Text>
            <LinesTable m={m} cols={inclCols} cellOf={inclCellOf} totalOf={inclTotalOf} />
            <View style={s.sumWrap} wrap={false}>
              <View style={s.sumRow}><Text style={s.sumLbl}>Lines (SGD)</Text><Text style={s.sumVal}>{fmtSgd(t.linesSgd)}</Text></View>
              <View style={s.sumRow}><Text style={s.sumLbl}>Processing fee ({fmtPctPlain(t.processingPct)})</Text><Text style={s.sumVal}>{fmtSgd(t.processingSgd)}</Text></View>
              <View style={s.grandRow}><Text style={s.grandLbl}>Lines incl. processing (SGD)</Text><Text style={s.grandVal}>{fmtSgd(t.linesInclSgd)}</Text></View>
            </View>
          </View>
          {footer}
        </Page>
      ) : null}
    </Document>
  );
}
 
export async function generatePoPdf(model: PoPdfModel): Promise<Buffer> {
  return renderToBuffer(<PoDoc m={model} />);
}
