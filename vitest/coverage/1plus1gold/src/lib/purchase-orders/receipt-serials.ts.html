/**
 * Manufacturer-serial capture at PO receipt (2026-10-06, Mukesh).
 *
 * Every inventory-eligible unit received against a PO MUST carry its
 * manufacturer serial — one per unit, so a qty-4 line needs 4 serials. On
 * receipt the server also issues OUR serial (nfc_serials) for each unit and
 * links it to the inventory row, so the CoA / NFC card exists the moment the bar
 * is in the vault instead of being created by hand later.
 *
 * Accessories are exempt (no per-unit inventory rows, no serial).
 *
 * This module is the pure half — validation + normalisation — so the browser
 * form and the API route share ONE rule. The DB half (conflict check, issuing)
 * lives in receiving.ts.
 */
 
import { isInventoryEligible } from "@/lib/inventory";
 
/** Letters/digits, optionally separated by . _ / - (checked after whitespace is
 *  stripped). Covers PAMP (D028511), Valcambi (AA995953), Nadir (A046039) and
 *  lets an odd refiner format through, while refusing markup or filter syntax. */
const SERIAL_RE = /^[A-Z0-9](?:[A-Z0-9._/-]{0,62}[A-Z0-9])?$/;
 
/** Strip ALL whitespace + uppercase — the card prints "AA 995953" but the
 *  registry stores "AA995953" (as entered by the serials tool), so a spaced and
 *  an unspaced entry must resolve to the same bar. */
export function normaliseManufacturerSerial(raw: string): string {
  return raw.replace(/\s+/g, "").toUpperCase();
}
 
export function requiresReceiptSerials(productType: string | null | undefined, grams: number): boolean {
  return isInventoryEligible(productType, grams);
}
 
export interface ReceiptSerialLine {
  key:         string;           // po_line_id (for error messages + mapping)
  productId:   string;
  label:       string;
  productType: string;
  grams:       number;
  quantity:    number;
  serials:     unknown;
}
 
export type ReceiptSerialResult =
  | { ok: true; serialsByKey: Map<string, string[]> }
  | { ok: false; error: string };
 
/**
 * Validate the serials for every line of one receipt. A product's serials must
 * be unique across the whole receipt (two lines can carry the same product).
 */
export function validateReceiptSerials(lines: ReceiptSerialLine[]): ReceiptSerialResult {
  const serialsByKey = new Map<string, string[]>();
  const seenPerProduct = new Map<string, Set<string>>();
 
  for (const l of lines) {
    if (!requiresReceiptSerials(l.productType, l.grams)) {
      serialsByKey.set(l.key, []);
      continue;
    }
    if (!Array.isArray(l.serials)) {
      return { ok: false, error: `${l.label}: enter the manufacturer serial for each of the ${l.quantity} unit(s)` };
    }
    if (l.serials.length !== l.quantity) {
      return { ok: false, error: `${l.label}: ${l.quantity} unit(s) received but ${l.serials.length} serial(s) entered — one serial per unit is required` };
    }
    const out: string[] = [];
    for (let i = 0; i < l.serials.length; i++) {
      const v = l.serials[i];
      const s = typeof v === "string" ? normaliseManufacturerSerial(v) : "";
      if (!s) return { ok: false, error: `${l.label}: serial for unit ${i + 1} is required` };
      if (!SERIAL_RE.test(s)) {
        return { ok: false, error: `${l.label}: "${s.slice(0, 40)}" is not a valid serial (letters, digits, . _ / -; max 64)` };
      }
      const seen = seenPerProduct.get(l.productId) ?? new Set<string>();
      if (seen.has(s)) return { ok: false, error: `${l.label}: serial ${s} entered more than once` };
      seen.add(s);
      seenPerProduct.set(l.productId, seen);
      out.push(s);
    }
    serialsByKey.set(l.key, out);
  }
  return { ok: true, serialsByKey };
}
