/**
 * Shipping country configuration.
 *
 * We currently ship within Singapore only.
 * Profile/account country is unrestricted — we collect that data for future planning.
 * This file is the single source of truth for shipping restrictions.
 *
 * To add a country later: add its ISO code to ALLOWED_SHIPPING_COUNTRIES
 * and update the checkout UI to show a country selector.
 */
 
export const ALLOWED_SHIPPING_COUNTRIES = ["SG"] as const;
export type  AllowedShippingCountry     = typeof ALLOWED_SHIPPING_COUNTRIES[number];
 
export const DEFAULT_SHIPPING_COUNTRY      = "SG"          as const;
export const DEFAULT_SHIPPING_COUNTRY_NAME = "Singapore"   as const;
 
/** Singapore postal codes are exactly 6 digits. */
export const SG_POSTAL_CODE_REGEX = /^\d{6}$/;
 
/** Returns true if the given ISO country code is allowed for shipping. */
export function isShippingAllowed(countryCode: string): boolean {
  return (ALLOWED_SHIPPING_COUNTRIES as readonly string[]).includes(countryCode.toUpperCase());
}
 
/** Validates a Singapore postal code string. */
export function isValidSgPostalCode(postalCode: string): boolean {
  return SG_POSTAL_CODE_REGEX.test(postalCode.trim());
}
 
/**
 * Physical carriers whose checkout availability is gated by
 * `shipper_settings.enabled`. `/api/shipping-options` (the UI list) already
 * hides a disabled carrier — but the money-path charge routes accepted a
 * client-supplied `shipperType` without re-checking, so a crafted checkout POST
 * could create an order for a carrier the UI never offered. That is the exact
 * "gated-off SPX / in_person incident" class: an order on a carrier with no
 * live fulfilment path. This set is the server-side counterpart of the
 * per-carrier `enabled` reads in `/api/shipping-options`.
 *
 * self_pickup / gold_yield / vault are NOT here — they are gated by their own
 * per-account enrolment/flag checks in the charge routes, not by a global
 * carrier toggle.
 */
export const ENABLED_GATED_CARRIERS = ["lalamove", "ninjavan", "spx"] as const;
 
/**
 * Server-authoritative check that a physical carrier is currently enabled.
 * Call in the PRE-PAYMENT charge routes (orders/create COD, hitpay/create,
 * stripe/create-intent) before minting an order/intent — NEVER in a post-payment
 * webhook, where rejecting would strand a customer who has already paid.
 *
 * Semantics mirror `/api/shipping-options`: a carrier is blocked only when its
 * `shipper_settings` row exists AND `enabled = false`. A missing row is
 * fail-open (the legacy lalamove default; the ninjavan/spx rows always exist
 * post-seed), so this can only ever TIGHTEN, never break a live carrier.
 */
export async function assertShipperEnabled(
  supabase: { from: (t: string) => any },
  shipper: string,
): Promise<{ ok: true } | { ok: false; error: string }> {
  if (!(ENABLED_GATED_CARRIERS as readonly string[]).includes(shipper)) {
    return { ok: true };
  }
  const { data, error } = await supabase
    .from("shipper_settings")
    .select("enabled")
    .eq("shipper", shipper)
    .maybeSingle();
  // Fail-open on a transient lookup error — availability is also schedule-gated
  // and the UI already hid a disabled carrier; a DB blip must not block a live
  // lalamove/ninjavan checkout. The explicit enabled=false row is the signal.
  // Log it so a *pattern* of gate-bypasses-via-DB-error is visible in
  // observability (the theoretical way a disabled carrier slips through).
  if (error) {
    console.error(`[assertShipperEnabled] fail-open: settings lookup for "${shipper}" errored — allowing checkout`, error);
    return { ok: true };
  }
  if (data && data.enabled === false) {
    return { ok: false, error: `${shipper} delivery is not currently available. Please choose another shipping option.` };
  }
  return { ok: true };
}
