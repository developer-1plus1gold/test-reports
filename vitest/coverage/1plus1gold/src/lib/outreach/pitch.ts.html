/**
 * Outreach — pitch parsing, validation and the mailto link. PURE.
 *
 * The rules in docs/outreach-automation-design.md §5 are enforced HERE, in
 * code, not trusted to the prompt:
 *  - under 150 words;
 *  - exactly one link, and it is to 1plus1gold.com;
 *  - every number in the draft appears in the facts block it was given;
 *  - a forum reply says it comes from the company.
 * A draft that fails is not saved; the route returns the problems.
 */
 
import type { ProspectKind } from "./domains";
 
export const MAX_PITCH_WORDS = 150;
export const MAX_SUBJECT_CHARS = 120;
 
export interface Pitch {
  subject: string | null;
  body: string;
}
 
export function parsePitch(text: string): Pitch {
  const fenced = text.match(/```(?:json)?\s*([\s\S]*?)\s*```/);
  const raw = (fenced ? fenced[1] : text).trim();
  let parsed: unknown;
  try {
    parsed = JSON.parse(raw);
  } catch (e) {
    throw new Error("Pitch output was not valid JSON: " + (e instanceof Error ? e.message : String(e)));
  }
  const o = (parsed ?? {}) as Record<string, unknown>;
  const body = typeof o.body === "string" ? o.body.trim() : "";
  if (!body) throw new Error("Pitch output had no body");
  const subject = typeof o.subject === "string" && o.subject.trim() ? o.subject.trim() : null;
  return { subject, body };
}
 
export function wordCount(s: string): number {
  return s.trim().split(/\s+/).filter(Boolean).length;
}
 
const URL_RE = /\bhttps?:\/\/[^\s)<>"']+|\b(?:www\.)?[a-z0-9-]+(?:\.[a-z0-9-]+)*\.(?:com|sg|gold|net|org|io)(?:\/[^\s)<>"']*)?/gi;
 
export function linksIn(s: string): string[] {
  return (s.match(URL_RE) ?? []).map((u) => u.replace(/[.,;:!?]+$/, ""));
}
 
/** Numeric tokens, commas dropped ("1,234.50" → "1234.50"; trailing ".0" kept as written). */
export function numbersIn(s: string): string[] {
  return (s.replace(URL_RE, " ").match(/\d[\d,]*(?:\.\d+)?/g) ?? []).map((n) => n.replace(/,/g, ""));
}
 
function canonicalNumber(n: string): string {
  const v = Number(n);
  return Number.isFinite(v) ? String(v) : n;
}
 
const DISCLOSURE = /\b(?:founder|co-founder|i run|we run|i work (?:at|for)|disclosure)\b/i;
 
export function validatePitch(p: Pitch, facts: string, kind: ProspectKind): string[] {
  const problems: string[] = [];
  const words = wordCount(p.body);
  if (words > MAX_PITCH_WORDS) problems.push(`Too long: ${words} words (max ${MAX_PITCH_WORDS}).`);
 
  const links = linksIn(p.body);
  const ours = links.filter((l) => /1plus1gold\.com/i.test(l));
  if (ours.length !== 1) problems.push(`Needs exactly one link to 1plus1gold.com (found ${ours.length}).`);
  if (links.length !== ours.length) problems.push("Links to other sites are not allowed.");
 
  const allowed = new Set(numbersIn(facts).map(canonicalNumber));
  const unsupported = [...new Set(numbersIn(`${p.subject ?? ""} ${p.body}`).map(canonicalNumber))]
    .filter((n) => !allowed.has(n));
  if (unsupported.length) problems.push(`Numbers not in the facts block: ${unsupported.join(", ")}.`);
 
  if (kind === "site") {
    if (!p.subject) problems.push("An email needs a subject.");
    else Iif (p.subject.length > MAX_SUBJECT_CHARS) problems.push(`Subject over ${MAX_SUBJECT_CHARS} characters.`);
  } else if (!(DISCLOSURE.test(p.body) && /1plus1gold/i.test(p.body))) {
    problems.push("A forum reply must say it comes from 1Plus1Gold (e.g. 'I'm Mukesh, founder of 1Plus1Gold').");
  }
  return problems;
}
 
/** mailto: with subject + body prefilled. Recipient may be empty (operator fills it). */
export function mailtoUrl(to: string | null, subject: string | null, body: string): string {
  const q: string[] = [];
  if (subject) q.push(`subject=${encodeURIComponent(subject)}`);
  q.push(`body=${encodeURIComponent(body)}`);
  return `mailto:${to ? encodeURIComponent(to).replace(/%40/g, "@") : ""}?${q.join("&")}`;
}
