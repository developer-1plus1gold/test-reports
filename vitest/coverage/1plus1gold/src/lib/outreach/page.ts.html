/** Link-check verdict from an HTTP status. Only 404/410 mean the page is gone;
 *  a 403/429/5xx from a server that blocks datacentre probes is "unknown". */
export function pageStatus(status: number): "ok" | "page_gone" | "unknown" {
  if (status === 404 || status === 410) return "page_gone";
  if (status >= 200 && status < 300) return "ok";
