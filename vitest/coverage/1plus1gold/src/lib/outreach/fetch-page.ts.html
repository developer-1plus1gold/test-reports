/**
 * The URL comes from an operator or from an AI engine's citation list, and our
 * server fetches it — so it must never reach an internal address. IP literals
 * and single-label / localhost names are refused outright; a public site is
 * always addressed by a domain name.
 */
export function isPublicWebUrl(u: URL): boolean {
  if (u.protocol !== "https:" && u.protocol !== "http:") return false;
  if (u.username || u.password) return false;
  const h = u.hostname.toLowerCase();
  if (h === "localhost" || h.endsWith(".localhost") || h.endsWith(".internal") || h.endsWith(".local")) return false;
  if (/^[\d.]+$/.test(h) || h.includes(":") || h.startsWith("[")) return false; // IPv4 / IPv6 literals
  return /^(?:[a-z0-9-]+\.)+[a-z]{2,}$/.test(h);
}

/**
 * The hostname check above cannot see where a public-looking name points, so
 * each hop also resolves the name and refuses it if ANY address is non-public.
 * A resolution failure is refused too. (A rebinding between this lookup and
 * the fetch's own lookup remains possible; this closes the static case.)
 */
async function resolvesPublic(host: string): Promise<boolean> {
  try {
    const addrs = await lookup(host, { all: true });
    return addrs.length > 0 && addrs.every((a) => !isPrivateAddress(a.address));
  } catch {
    return false;
  }
}
