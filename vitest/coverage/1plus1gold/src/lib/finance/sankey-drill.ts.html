/**
 * Geometry for the drill-down Sankey.
 *
 * Ported from the clickable prototype Mukesh approved on 2026-09-03
 * ("Money Flow Drill", artifact a3ddfb02). The published artifact is the record
 * — this was ported by READING it, not by recalling it, after the last time a
 * design was reproduced from memory and the wrong half shipped.
 *
 * Pure and DOM-free so the parts that are easy to get subtly wrong — the tail
 * fold, the label placement, the canvas height — are testable without a browser.
 * That is the same reasoning that pulled `paretoGeometry` and `flowRibbons` out
 * of their components after each was wrong once.
 *
 * ▶ ONE DIVERGENCE FROM THE PROTOTYPE, stated rather than buried. The prototype
 * drilled BOTH sides of the top-level diagram in place. In the app the drill is
 * rooted at the block you opened, so it is a one-sided fan-out. That is the
 * correct Sankey form here and the prototype's own design note says why: below
 * level 1 there is only one side — Rent and Travel do not flow *to* anything,
 * they partition Finance & G&A.
 */
 
/** Geometry constants, values as approved in the prototype. */
export const BARW = 13;      // bar width — the mark is the height, not the width
/** Horizontal channel between two levels. It is also all the room an INWARD
 *  label has, which is why those carry the name only — see the renderer. */
export const CH = 150;
export const LABW = 186;     // gutter for a label that points INWARD (name only)
/** The OUTERMOST gutter also carries the amount, so it needs more room. Sized
 *  from the longest real case — "20 g PAMP Fortuna Veriscan S$3,412.00" — which
 *  was clipped at 186 and read as a truncated product name. */
export const LABW_OUT = 240;
export const PADX = 8;
/** Room above the bars for TWO heading rows: the side captions ("which team
 *  spent it") and, under them, the name of the node each drilled column opened
 *  from. At 30 the two collided — they were 2px apart. */
export const TOPP = 48;
export const BOTP = 16;
export const ROW_MIN = 17;   // vertical slot a labelled row needs
export const GAP_MIN = 5;
export const GAP_MAX = 9;
export const TOP_N = 10;
export const MIN_SHARE = 0.05;
 
export interface DrillNode {
  /** Stable id within its level — the row name is unique among siblings here. */
  id: string;
  n: string;
  m?: string;
  /** Signed amount. Bar height is |a|; the sign drives the colour. */
  a: number;
  kids?: DrillNode[];
  /** True for the synthetic "N smaller" row, which opens rather than drills. */
  more?: boolean;
  moreKey?: string;
  /** The row this node came from, carried so the renderer can ask the caller's
   *  own accessors about it (profit, monthly series). Opaque here on purpose —
   *  the geometry must not know what a money row is. */
  src?: unknown;
  /**
   * Hold this row at the BOTTOM of its column whatever it is worth, and keep it
   * out of the tail fold.
   *
   * The one user is the net/shortfall block. It sits on the income side purely
   * so the two sides sum to the same total — it is not money anybody paid us —
   * but it is large, so ranking by size floated it above real revenue lines.
   * Mukesh, 2026-09-04: *"the shortfall is showing above the 2 small revenue
   * lines, which is a bit confusing to see."* Rank is about size; this row is
   * about arithmetic, so it does not compete on size.
   */
  pin?: "last";
}
 
export interface Box { nd: DrillNode; y: number; h: number }
 
/**
 * Fold the tail — by SHARE first, then by count.
 *
 * Strict proportionality plus one dominant node leaves the rest of a column too
 * thin to label, and an unlabelled hairline is not a row anyone can read. The
 * small ones collapse into a single "N smaller" that opens on tap; opening it
 * GROWS THE CANVAS rather than cramming, which is what `drillHeight` is for.
 */
export function foldTail(rows: DrillNode[], key: string, showAll: boolean): DrillNode[] {
  // Pinned rows are held out of the ranking AND the fold, then appended. Leaving
  // one in the fold would let the shortfall be swallowed into "N smaller", which
  // would hide the very number that makes the two sides balance.
  const pinned = rows.filter((r) => r.pin === "last");
  const rest   = rows.filter((r) => r.pin !== "last");
  const sorted = [...rest].sort((a, b) => Math.abs(b.a) - Math.abs(a.a));
  if (showAll) return [...sorted, ...pinned];
  const sum = sorted.reduce((a, r) => a + Math.abs(r.a), 0) || 1;
  let keep = sorted.filter((r) => Math.abs(r.a) / sum >= MIN_SHARE).slice(0, TOP_N);
  if (keep.length === 0) keep = sorted.slice(0, 1);
  const tail = sorted.filter((r) => !keep.includes(r));
  if (tail.length === 0) return [...keep, ...pinned];
  // Folding ONE row into "1 smaller" costs a row and saves nothing — it would
  // read as a hidden item where there is none.
  if (tail.length === 1) return [...keep, tail[0], ...pinned];
  return [...keep, {
    id: "more:" + key, more: true, moreKey: key,
    n: `${tail.length} smaller`,
    a: tail.reduce((x, y) => x + y.a, 0),
    m: "tap to open — the chart grows to fit",
  }, ...pinned];
}
 
/** Vertical room a drawn row gets. See `drillHeight`. */
export const ROW_PX = 52;
/**
 * Floor for the whole canvas.
 *
 * ⚠️ RAISED 440 → 620 on 2026-09-04, because the 40px/row change shipped the day
 * before was ALMOST INERT and Mukesh could not see it: *"I believe the vertical
 * scroll related changes are in prod, as I cannot see any change there."* He was
 * right. `foldTail` draws at most TOP_N + 1 = ELEVEN rows in a level unless the
 * reader opens the fold, and 11 × 40 = 440 — exactly the floor. So every normal
 * level landed on the floor either way and the diagram went from 400px to 440px,
 * a 40px change nobody would notice. The per-row rate only ever bites on an
 * expanded fold.
 *
 * The lesson, and it is the one this codebase keeps relearning: a per-unit rate
 * cannot change anything while a CAP upstream keeps the unit count below the
 * point where the rate applies. The floor is what a reader actually sees.
 */
export const MIN_MAP_PX = 620;
 
/**
 * Where a DRILLED level sits, and how tall it is.
 *
 * ⚠️ THE FIX FOR THE COMPLAINT THAT SURVIVED TWO EARLIER ATTEMPTS. Mukesh,
 * 2026-09-04: *"when I click on more items the system instead of expanding,
 * SHRINKS those items within the same window."* He was right both times.
 *
 * A drilled level used to be laid strictly inside its parent's band, because
 * that is what makes a Sankey a Sankey — children sum to the ribbon they came
 * from. But a block worth 5% of the total has 5% of the canvas, so opening
 * "N smaller" there put FORTY rows where eleven had been: more items, less room
 * each, and the fold's own promise ("the chart grows to fit") was false in
 * exactly the case a reader would try it. Raising the row rate could not help —
 * the rows were never getting the canvas, they were getting a slice of it. Nor
 * could sizing the canvas by the slice: 40 rows at 52px inside a 5% band asks
 * for a 41,600px canvas, so the ceiling simply binds and nothing improves.
 *
 * So an opened level now takes the room its rows NEED, growing out of its
 * parent's band and staying centred on it. The ribbon fans out from the thin
 * parent bar into the taller stack, which reads as zooming into that flow.
 *
 * ▶ THE TRADE, stated because it is a choice and not a constraint: thickness
 * stays strictly proportional WITHIN each column, so siblings remain comparable,
 * but the drilled branch is no longer to scale AGAINST the rest of the diagram.
 * Reading a drill as navigation — Mukesh's own words, *"the concept of drill
 * down is to go one by one"* — that is the right way round. If the cross-column
 * scale matters more than legibility on a small block, this is the line to
 * change.
 */
function openedBand(rowCount: number, parent: Box, top: number, band: number): [number, number] {
  // Never more than most of the canvas: the level you came from has to stay
  // visible, or drilling stops being a map and becomes a new page.
  const want = Math.min(rowCount * ROW_PX, band * 0.9);
  const h = Math.max(parent.h, want);
  const centred = parent.y + parent.h / 2 - h / 2;
  return [Math.max(top, Math.min(centred, top + band - h)), h];
}
 
/**
 * Canvas height comes from HOW MANY ROWS ARE DRAWN, not from the smallest share.
 *
 * Sizing it so a 0.07% row could hold a centred label wanted a 4,000px canvas —
 * the wrong lever. Bar height stays strictly the value, however thin that makes
 * the smallest bar, and label placement is solved separately (see `placeLabels`).
 *
 * ▶ Raised 30 → 40 px/row and the ceiling 1,200 → 4,000 (2026-09-04, Mukesh:
 *   "increase the section height … helps to see the data better when I expand it
 *   vertically during drill down").
 *
 *   The old 1,200 cap squeezed hardest exactly where there was most to read: a
 *   47-row vendor drill asked for 1,410px, got 1,200, and so drew ~25px per row
 *   INCLUDING its gaps. The more rows a level had, the less each one got — the
 *   same inversion as the Pareto bug of 2026-09-03, where longer labels stole
 *   plot area from the bars they described.
 *
 *   The ceiling stays (an unbounded canvas is a runaway on bad data) but is now
 *   far above any real level, so it is a backstop rather than a working limit.
 *   What makes that safe is that the CALLER now scrolls: `SankeyMap` renders into
 *   a bounded frame with overflow-y, so a tall canvas scrolls inside the diagram
 *   instead of stretching the page and pushing the detail rail out of reach.
 */
export function drillHeight(maxRowsInAnyLevel: number, min = MIN_MAP_PX, max = 4000): number {
  return Math.min(Math.max(min, maxRowsInAnyLevel * ROW_PX), max);
}
 
/**
 * Lay a level's rows into the vertical extent of their parent.
 *
 * Heights are strictly proportional to |amount| after the gaps are taken out, so
 * a thin bar is genuinely a small number. ⚠️ The cursor advances by the TRUE
 * height and only the drawn ink is floored — flooring the cursor accumulates,
 * and a side carrying sub-pixel rows then drifts out of line with its parent.
 * That exact bug cost a day on the top-level flow.
 */
export function layoutRows(rows: DrillNode[], y0: number, h0: number): Box[] {
  const sum = rows.reduce((a, r) => a + Math.abs(r.a), 0) || 1;
  const gaps = Math.max(rows.length - 1, 0);
  const gap = Math.min(GAP_MAX, Math.max(GAP_MIN, (h0 * 0.08) / (gaps || 1)));
  const scale = Math.max(h0 - gap * gaps, 1) / sum;
  let cur = y0;
  return rows.map((r) => {
    const h = Math.abs(r.a) * scale;
    const box = { nd: r, y: cur, h };
    cur += h + gap;
    return box;
  });
}
 
/**
 * Label y-positions: start at each bar's centre, then push apart so no two
 * collide — top-down, then bottom-up so the column stays inside the canvas.
 *
 * This is what lets bar height stay honest on skewed data: the bar keeps its
 * true size and the LABEL moves, with a leader line drawn back to it when it had
 * to. The alternative — a minimum bar height so every label fits — would break
 * the value encoding, which is the one thing a Sankey must not do.
 */
export function placeLabels(boxes: Box[], top: number, band: number): Map<Box, number> {
  const ls = boxes
    .map((b) => ({ b, want: b.y + b.h / 2, y: b.y + b.h / 2 }))
    .sort((p, q) => p.want - q.want);
  let prev = -Infinity;
  for (const l of ls) { l.y = Math.max(l.y, prev + ROW_MIN); prev = l.y; }
  let next = top + band + ROW_MIN;
  for (let i = ls.length - 1; i >= 0; i--) { ls[i].y = Math.min(ls[i].y, next - ROW_MIN); next = ls[i].y; }
  const m = new Map<Box, number>();
  for (const l of ls) m.set(l.b, l.y);
  return m;
}
 
/**
 * Build every level of the drill from a root and the open path.
 *
 * Level 0 is the root's own children; each further level is the children of the
 * node named at that depth in `path`. Everything you came through stays on
 * screen — which is the whole point of drilling in place rather than replacing
 * the view, and was Mukesh's specific correction to the first cut.
 */
export function buildDrill(opts: {
  rootRows: DrillNode[];
  path: string[];
  showAll: Record<string, boolean>;
  /** Height of the root bar the fan-out hangs off. */
  band: number;
  top?: number;
}): { levels: Box[][]; openIds: string[] } {
  const { rootRows, path, showAll, band } = opts;
  const top = opts.top ?? TOPP;
  const levels: Box[][] = [];
  const openIds: string[] = [];
 
  let rows = foldTail(rootRows, "d0", !!showAll["d0"]);
  if (rows.length === 0) return { levels, openIds };
  levels.push(layoutRows(rows, top, band));
 
  for (let d = 0; d < path.length; d++) {
    const parent = levels[d].find((b) => b.nd.id === path[d]);
    if (!parent) break;
    const kids = (parent.nd.kids ?? []).filter((k) => k.a !== 0);
    if (kids.length === 0) break;
    openIds.push(path[d]);
    const key = "d" + (d + 1);
    rows = foldTail(kids, key, !!showAll[key]);
    levels.push(layoutRows(rows, ...openedBand(rows.length, parent, top, band)));
  }
 
  // ▶ NOTHING OPENS ITSELF (Mukesh, 2026-09-04: *"ideally nothing should open
  // automatically, the concept of drill down is to go one by one"*).
  //
  // A sole child carrying the whole parent used to open itself, to spare the
  // reader a column that is a full-width ribbon from a bar to its identical
  // twin. That was the wrong trade twice over. A node the code opened cannot be
  // closed — clicking it removes it from the path and the rule instantly
  // re-opens it — so the reader got a control that did nothing, which is what
  // "gold desk does not collapse" was. And a pass-through column is only noise
  // if it appears unasked; if the reader clicks into it, showing exactly what
  // they clicked is the honest answer.
  //
  // So the rule is gone rather than patched, and `autoIds` with it. Every open
  // node is now one the reader opened, and every open node closes.
 
  return { levels, openIds };
}
 
 
/* ── the two-sided map ──────────────────────────────────────────────────────
   The approved prototype's actual shape: costs on the left, income on the right,
   the business between them, and BOTH SIDES DRILL IN PLACE. Opening a node adds
   a column growing away from the spine; nothing you came through leaves.
 
   Both sides fill the same band, which is honest here rather than a fudge: the
   shortfall is carried as a block on the income side precisely so the two sides
   sum to the same total (see the 2026-09-03 note on the flow diagram). If that
   ever stops being true the two stacks would no longer meet at the spine, which
   is a visible failure rather than a silent one. */
 
export interface MapGeometry {
  W: number; spineX: number; unit: number;
  /** x of the cost column at depth d — depth grows LEFTWARD, away from the spine. */
  xL: (d: number) => number;
  /** x of the income column at depth d — depth grows RIGHTWARD. */
  xR: (d: number) => number;
}
 
/**
 * The clickable box around a bar's LABEL.
 *
 * ⚠️ THIS IS WHY BLOCKS WOULD NOT COLLAPSE (Mukesh, 2026-09-04: *"some of the
 * items do not collapse back — example is gold desk, marketing expenses, in fact
 * many of the items on the side of money spent"*).
 *
 * A label needs a hit target, because a 2px bar cannot be clicked and the label
 * is the part the reader aims at. The first cut sized every one of them at
 * `LABW_OUT + BARW + 12` = 265px — the OUTER margin's width — regardless of
 * which column it belonged to. An inner column has only `CH` = 150px of channel
 * before the next column's bar, so those boxes ran straight over the NEIGHBOURING
 * COLUMN'S BARS. Marks are painted level by level, so the deeper column's
 * invisible box landed on top of the shallower one: once you drilled, clicking
 * "Gold desk" hit a level-1 label instead and drilled further rather than
 * collapsing. Nothing looked broken — the click simply went somewhere else.
 *
 * So the box is sized to the room the label ACTUALLY has: the full outer gutter
 * on the outermost column, and the inter-column channel less a margin anywhere
 * else. `labelBoxClearsNeighbour` states the invariant that makes it correct.
 */
export function labelHitBox(x: number, opts: { left: boolean; outermost: boolean }): { x: number; w: number } {
  const gutter = opts.outermost ? LABW_OUT + 12 : CH - 8;
  return { x: opts.left ? x - gutter : x, w: gutter + BARW };
}
 
/**
 * True when a label box cannot reach either neighbouring column's bar.
 *
 * Columns sit `BARW + CH` apart, so the neighbour's bar occupies
 * [x ± unit, x ± unit + BARW]. Exists so the invariant is asserted rather than
 * re-derived from the constants each time one of them moves.
 */
export function labelBoxClearsNeighbour(x: number, box: { x: number; w: number }): boolean {
  const unit = BARW + CH;
  return box.x > x - unit + BARW && box.x + box.w < x + unit;
}
 
export function mapGeometry(nL: number, nR: number): MapGeometry {
  const unit = BARW + CH;
  const spineX = PADX + LABW_OUT + nL * unit;
  return {
    unit, spineX,
    W: PADX * 2 + LABW_OUT + nL * unit + BARW + CH + nR * unit + LABW_OUT,
    xL: (d) => PADX + LABW_OUT + (nL - 1 - d) * unit,
    xR: (d) => spineX + BARW + CH + d * unit,
  };
}
 
/**
 * The height the whole map needs: driven by the busiest column on EITHER side,
 * so unfolding "N smaller" anywhere grows the canvas rather than cramming.
 */
export function mapHeight(levelsL: Box[][], levelsR: Box[][], min = MIN_MAP_PX): number {
  const most = Math.max(0, ...levelsL.map((l) => l.length), ...levelsR.map((l) => l.length));
  return drillHeight(most, min);
}
