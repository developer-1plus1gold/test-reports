/**
 * Findings — the decision layer of the money map.
 *
 * The 2026-09-02 mock carried a hand-written callout under the gold block
 * ("Shopee is the only channel earning a gross profit … the 1g Valcambi split is
 * both the best seller and the biggest single loss"). That prose was the most
 * decision-dense thing on the page and it did NOT survive the rebuild, because a
 * sentence true of one snapshot cannot be carried forward.
 *
 * So the findings are computed from the same rows the chart draws. Each one
 * names the node it concerns, so it can be opened from the finding; each states
 * the evidence rather than a verdict; and they are ranked by money at stake,
 * because that is the order a decision-maker should meet them in.
 *
 * ⚠️ Deliberately NOT an LLM. Every rule here is arithmetic over data already on
 * screen. A finding that cannot be reproduced by looking at the chart would be
 * worse than none — the reader has no way to check it.
 */
import type { MoneyMap, MmBlock, MmRow } from "./money-map";
 
export type Severity = "critical" | "warning" | "watch";
 
export interface Finding {
  id: string;
  severity: Severity;
  /** One line, in the operator's language. */
  title: string;
  /** The arithmetic behind it, so the claim can be checked against the chart. */
  evidence: string;
  /** SGD at stake — the ranking key, and why one finding outranks another. */
  stake: number;
  /** Which lens and block this belongs to, so the finding can open the flow. */
  lens: "team" | "pnl";
  blockKey?: string;
}
 
const n2 = (v: number) => Math.round(v * 100) / 100;
const sgd = (v: number) => "S$" + Math.abs(v).toLocaleString("en-SG", { minimumFractionDigits: 2, maximumFractionDigits: 2 });
const pct = (part: number, whole: number) => !whole ? "—" : (Math.abs(part / whole) * 100).toFixed(1) + "%";
 
/** Depth-first walk with the path, so a finding can say where a row sits. */
function walk(rows: MmRow[], path: string[], out: Array<{ row: MmRow; path: string[] }>){
  for (const r of rows){
    out.push({ row: r, path: [...path, r.n] });
    if (r.kids?.length) walk(r.kids, [...path, r.n], out);
  }
  return out;
}
 
/** The last two months present in a series, for a like-for-like comparison. */
function lastTwo(t: Record<string, number> | undefined, months: string[]){
  const have = months.filter((m) => (t ?? {})[m] !== undefined);
  if (have.length < 2) return null;
  const [prev, curr] = [have[have.length - 2], have[have.length - 1]];
  return { prev, curr, prevV: (t ?? {})[prev] ?? 0, currV: (t ?? {})[curr] ?? 0 };
}

/** Findings that belong to one block, for the badge on the node. */
export function findingsFor(all: Finding[], lens: "team" | "pnl", blockKey: string){
  return all.filter((f) => f.lens === lens && f.blockKey === blockKey);
