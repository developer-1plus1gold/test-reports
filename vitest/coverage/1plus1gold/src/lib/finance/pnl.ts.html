/**
 * P&L aggregation — pure logic, unit-testable without DB.
 *
 * Revenue is recognised at DELIVERY (accrual): only orders whose lifecycle is
 * 'completed' count, and the route windows them by lifecycle_completed_at so the
 * revenue lands in the period the order was delivered — the same period its COGS
 * lands (COGS is dated by inventory sold-date). Cash received for orders not yet
 * delivered is deferred revenue, not income.
 *
 * Revenue lines (delivery basis — recognised order, windowed by completion date):
 *   - Gold sales       — orders.subtotal_ipm_sgd       for paid orders
 *   - Accessories      — orders.subtotal_taxable_sgd   for paid orders
 *   - Conversion fees  — conversions.conversion_fee_sgd for paid conversions
 *   - Shipping         — orders.shipping_fee_sgd       (charged to customer)
 *   - Processing       — orders.processing_fee_sgd     (charged to customer)
 *
 * COGS — TRUE landed cost from procurement, not spot proxy:
 *   - Gold COGS = sum(inventory.cost_per_g_sgd × products.grams) for inventory
 *     rows that flipped to status='sold' inside the period. cost_per_g_sgd is
 *     the landed cost (gold + PO-side shipping + payment + insurance + other,
 *     allocated across the lines in the receipt). PO ancillary costs flow
 *     through automatically.
 *   - Accessory COGS = sum(qty × latest_landed_cost_sgd) per product. Latest
 *     PO line landed cost is the cost-basis proxy (accessories aren't tracked
 *     per-unit in the inventory table; they live in products.stock_qty).
 *
 * Operating expenses:
 *   - From ledger_entries where entry_type='expense' AND entry_date in period.
 *   - Grouped by tax category for the active jurisdiction (SG / US).
 *   - Note: PO-side costs (shipping/payment/insurance) are NOT in ledger
 *     entries — they're inside landed_cost and flow through COGS instead.
 *     Sales-side outflows (Lalamove invoice, gateway %) ARE expected as
 *     ledger entries (shipping_logistics + payment_processing_fees).
 */

/**
 * One inventory row that flipped to status='sold' inside the period.
 *
 * Dated by `sold_at`, stamped at the status transition (20260723d). It is
 * DELIBERATELY not `updated_at`: that moves on any later write, so a serial
 * correction would silently shift a sale into a different accounting period —
 * on prod one bar delivered 2026-06-30 was reporting S$887.25 of COGS in July.
 */

   *  null for conversion-sold bars (no order link) → bucketed as 'other'. */
  source_channel?: string | null;

   * (lib/finance/cost-of-conversion.ts). Counting it in both charges the whole

}

export interface PnlReport {
  periodStart:    string;
  periodEnd:      string;
  periodLabel:    string;
  jurisdiction:   Jurisdiction;
  revenue: {
    gold_sgd:        number;
    accessories_sgd: number;
    conversion_sgd:  number;
    shipping_sgd:    number;
    processing_sgd:  number;
    total_sgd:       number;
  };
  cogs: {
    gold_sgd:        number;
    accessories_sgd: number;
    total_sgd:       number;
  };
  grossProfit_sgd:    number;
  /** Marketplace commission (Σ external_channel_fee_sgd) — a selling cost that
   *  reduces net income. Not in COGS or gross profit (standard definitions kept). */
  commission_sgd:     number;
  /**
   * Shipping cushion — how much of the outbound courier cost we recovered from
   * customers vs subsidised. Most delivery is subsidised (customers are charged
   * $0 or a token fee while we pay the courier the real cost), so this surfaces
   * that gap, which the statement otherwise splits across two far-apart lines
   * (shipping revenue at the top, courier cost down in operating expenses).
   *
   *   charged  = customer shipping-fee revenue (= revenue.shipping_sgd)
   *   cost     = shipping_logistics operating expense (Lalamove/SPX/NinjaVan
   *              invoices logged to the ledger — the authoritative courier spend)
   *   cushion  = charged − cost  (NEGATIVE = we subsidised delivery)
   *   coverage = charged / cost × 100  (share of courier cost recovered from
   *              customers; null when no courier cost was booked)
   *
   * Basis note: `charged` is delivery-windowed (revenue recognised at completion)
   * while `cost` is entry-date-windowed (when the courier invoice was logged).
   * Over a period they align; a single order can straddle a month boundary.
   * Marketplace-absorbed shipping is NOT here — on those orders the buyer is
   * billed $0 and the platform nets the delivery cost from our payout, so it
   * lives in the marketplace-commission/payout path, not shipping_logistics.
   */
  shippingCushion: {
    charged_sgd:  number;
    cost_sgd:     number;
    cushion_sgd:  number;
    coverage_pct: number | null;
  };
  /**
   * Payment-processing cushion — the exact analogue of shippingCushion for the
   * payment gateway. `charged` is the processing-fee revenue we billed customers;
   * `cost` is the payment_processing_fees expense (the fee HitPay/Stripe actually
   * deducted, derived per order — see lib/finance/gateway-fee-expenses). cushion =
   * charged − cost (NEGATIVE = we absorbed the gateway cost — e.g. PayNow, which
   * charges the customer 0); coverage = charged / cost × 100.
   */
  processingCushion: {
    charged_sgd:  number;
    cost_sgd:     number;
    cushion_sgd:  number;
    coverage_pct: number | null;
  };
  /**
   * Combibar conversions, both sides on one line.
   *
   * A conversion is a SERVICE, not a sale: the metal is like-for-like, so the
   * only economics are the fee we charge and the premium we give away on the
   * outbound bar. `fee` is the conversion-fee revenue; `cost` is the derived
   * `cost_of_conversion` expense (lib/finance/cost-of-conversion). net = fee −
   * cost, negative when the premium exceeded the fee.
   *
   * Surfaced separately because the tax-line grouping would otherwise bury it
   * inside "Other operating expenses", where the one number that says what
   * conversions actually cost is unreadable.
   *
   * ⚠️ `cost` can be NEGATIVE — a bar bought when spot was lower carries a
   * realised holding gain. Not an error; see cost-of-conversion.ts §3.6.
   */
  conversionCost: {
    fee_sgd:  number;
    cost_sgd: number;
    net_sgd:  number;
  };
  /** Revenue + COGS + commission split per sales channel. Σ reconciles to totals. */
  channels:           ChannelPnl[];
  expenses: {
    lines:       CategoryLine[];
    total_sgd:   number;
  };
  netIncome_sgd:      number;
}
 
/**
 * Channel buckets, derived from the registry — NOT a local literal.
 *
 * This used to be a hardcoded allow-list, and it missed `in_person` from the day
 * that channel shipped (2026-07-17): every in-person sale's revenue and COGS was
 * folded into "Other" next to conversions, which is a different kind of thing
 * entirely. The totals still reconciled, so nothing failed loudly.
 *
 * `other` is a genuine finance-only bucket — conversions and conversion-sold
 * inventory carry no sales channel — so it stays, but now it only ever holds
 * things that really have no channel.
 */
const KNOWN_CHANNELS = new Set<string>(ALL_SALES_CHANNELS);
const CHANNEL_ORDER = [...ALL_SALES_CHANNELS, "other"];
function normChannel(c: string | null | undefined): string {
  return c && KNOWN_CHANNELS.has(c) ? c : "other";
}
 
// ── Period helpers ───────────────────────────────────────────────────────────
 
export function resolvePeriod(period: Period, at: Date): {
  start: string;
  end:   string;
  label: string;
} {
  const y = at.getUTCFullYear();
  const m = at.getUTCMonth();
 
  if (period === "month") {
    const start = new Date(Date.UTC(y, m,   1));
    const end   = new Date(Date.UTC(y, m+1, 0));
    return { start: iso(start), end: iso(end), label: monthLabel(start) };
  }
 
  if (period === "quarter") {
    const qIdx = Math.floor(m / 3);
    const start = new Date(Date.UTC(y, qIdx * 3,     1));
    const end   = new Date(Date.UTC(y, qIdx * 3 + 3, 0));
    return { start: iso(start), end: iso(end), label: `Q${qIdx + 1} ${y}` };
  }
 
  const start = new Date(Date.UTC(y, 0,  1));
  const end   = new Date(Date.UTC(y, 11, 31));
  return { start: iso(start), end: iso(end), label: `FY ${y}` };
}
 
export function stepPeriod(period: Period, anchor: Date, dir: 1 | -1): Date {
  const y = anchor.getUTCFullYear();
  const m = anchor.getUTCMonth();
  const d = anchor.getUTCDate();
  if (period === "month")   return new Date(Date.UTC(y, m + dir, Math.min(d, 28)));
  if (period === "quarter") return new Date(Date.UTC(y, m + dir * 3, Math.min(d, 28)));
  return new Date(Date.UTC(y + dir, m, Math.min(d, 28)));
}
 
function iso(d: Date): string { return d.toISOString().slice(0, 10); }
function monthLabel(d: Date): string {
  return d.toLocaleString("en-SG", { month: "long", year: "numeric", timeZone: "UTC" });
}
 
// ── Aggregation ──────────────────────────────────────────────────────────────
 
export function buildPnlReport(args: {
  period:          Period;
  at:              Date;
  jurisdiction:    Jurisdiction;
  orders:          PaidOrderRow[];
  conversions:     PaidConversionRow[];
  expenses:        ExpenseRow[];
  /** Inventory rows that flipped to 'sold' inside the period — gold + standard. */
  soldInventory:   SoldInventoryRow[];
  /** Per-line accessory cost-basis pairs for orders inside the period. */
  accessoryCogs:   AccessoryCogsRow[];
}): PnlReport {
  const { period, at, jurisdiction, orders, conversions, expenses, soldInventory, accessoryCogs } = args;
  const { start, end, label } = resolvePeriod(period, at);
 
  // Per-channel accumulator (reconciles to the consolidated totals below).
  const chan = new Map<string, { revenue: number; cogs: number; commission: number }>();
  const bumpChan = (c: string, patch: { revenue?: number; cogs?: number; commission?: number }) => {
    const e = chan.get(c) ?? { revenue: 0, cogs: 0, commission: 0 };
    e.revenue    += patch.revenue    ?? 0;
    e.cogs       += patch.cogs       ?? 0;
    e.commission += patch.commission ?? 0;
    chan.set(c, e);
  };
 
  // Revenue.
  const paidOrders = orders.filter(isOrderRecognised);
  let goldRev = 0, accRev = 0, shipRev = 0, procRev = 0, commission = 0;
  for (const o of paidOrders) {
    const ipm  = Number(o.subtotal_ipm_sgd     ?? 0);
    const tax  = Number(o.subtotal_taxable_sgd ?? 0);
    const ship = Number(o.shipping_fee_sgd     ?? 0);
    const proc = Number(o.processing_fee_sgd   ?? 0);
    const comm = Number(o.external_channel_fee_sgd ?? 0);
    goldRev += ipm; accRev += tax; shipRev += ship; procRev += proc; commission += comm;
    bumpChan(normChannel(o.source_channel), { revenue: ipm + tax + ship + proc, commission: comm });
  }
 
  let convRev = 0;
  for (const c of conversions) {
    if (!c.paid_at) continue;
    convRev += Number(c.conversion_fee_sgd ?? 0);
  }
  if (convRev) bumpChan("other", { revenue: convRev }); // conversions aren't a sales channel
 
  const totalRev = goldRev + accRev + convRev + shipRev + procRev;
 
  // COGS — landed cost from inventory (gold + standard) and PO history (accessories).
  // Gold COGS — landed cost from sold inventory rows. A consignment piece rides
  // this SAME path: its inventory row (created at receipt, cost_per_g from the
  // supplier's confirmed price) flips to sold at delivery and is counted here,
  // matched to the sale's revenue exactly like every normal gold order.
  let goldCogs = 0;
  for (const inv of soldInventory) {
    // A conversion's outbound bar is NOT a sale: the metal it carries came back
    // to us as the customer's 1g cards, and only the premium we gave away is a
    // cost. That premium is derived as `cost_of_conversion` and appears in the
    // expense register instead. Booking the bar here as well would count it
    // twice. Dating matches: both use the bar's sold_at, so the exclusion and
    // the expense always land in the same period.
    if (isConversionSoldBar(inv)) continue;
    const c = Number(inv.cost_per_g_sgd ?? 0) * Number(inv.grams ?? 0);
    goldCogs += c;
    bumpChan(normChannel(inv.source_channel), { cogs: c });
  }
  let accCogs = 0;
  for (const a of accessoryCogs) {
    const c = Number(a.quantity ?? 0) * Number(a.landed_cost_sgd ?? 0);
    accCogs += c;
    bumpChan(normChannel(a.source_channel), { cogs: c });
  }
  const totalCogs   = goldCogs + accCogs;
  const grossProfit = totalRev - totalCogs;
 
  const channels: ChannelPnl[] = CHANNEL_ORDER
    .map((c) => {
      const e = chan.get(c) ?? { revenue: 0, cogs: 0, commission: 0 };
      return {
        channel:          c,
        revenue_sgd:      roundCents(e.revenue),
        cogs_sgd:         roundCents(e.cogs),
        commission_sgd:   roundCents(e.commission),
        contribution_sgd: roundCents(e.revenue - e.cogs - e.commission),
      };
    })
    .filter((c) => c.revenue_sgd !== 0 || c.cogs_sgd !== 0 || c.commission_sgd !== 0);
 
  // Expenses grouped by tax line.
  const lineMap = new Map<string, { categories: Set<string>; amount: number }>();
  for (const e of expenses) {
    const taxLine = jurisdiction === "sg" ? e.sg_tax_category : e.us_tax_category;
    if (!taxLine) continue;
    const cur = lineMap.get(taxLine) ?? { categories: new Set(), amount: 0 };
    cur.amount += Number(e.amount_sgd ?? 0);
    if (e.category) cur.categories.add(e.category);
    lineMap.set(taxLine, cur);
  }
  const lines: CategoryLine[] = [...lineMap.entries()]
    .map(([taxLine, v]) => ({
      taxLine,
      categories: [...v.categories].sort(),
      amount_sgd: roundCents(v.amount),
    }))
    .sort((a, b) => b.amount_sgd - a.amount_sgd);
  const totalExpenses = lines.reduce((s, l) => s + l.amount_sgd, 0);
 
  // Shipping cushion — customer shipping revenue vs the courier cost booked to
  // the shipping_logistics category. Read the cost from the raw expense rows by
  // CATEGORY (not the jurisdiction-grouped tax line) so it isolates courier
  // spend cleanly on both SG ("Postage & courier") and US ("Line 27a — Other
  // (shipping)") groupings, and never sweeps in other expenses sharing a line.
  const shipCost = roundCents(
    expenses
      .filter((e) => e.category === "shipping_logistics")
      .reduce((s, e) => s + Number(e.amount_sgd ?? 0), 0),
  );
  const shipCharged  = roundCents(shipRev);
  const shipCushion  = roundCents(shipCharged - shipCost);
  const shipCoverage = shipCost > 0 ? Math.round((shipCharged / shipCost) * 1000) / 10 : null;
 
  // Payment-processing cushion — processing-fee revenue vs the derived
  // payment_processing_fees gateway cost (read by category, jurisdiction-agnostic).
  const procCost     = roundCents(
    expenses
      .filter((e) => e.category === "payment_processing_fees")
      .reduce((s, e) => s + Number(e.amount_sgd ?? 0), 0),
  );
  const procCharged  = roundCents(procRev);
  const procCushion  = roundCents(procCharged - procCost);
  const procCoverage = procCost > 0 ? Math.round((procCharged / procCost) * 1000) / 10 : null;
 
  // Cost of conversion — the premium given away on each swap, read by CATEGORY
  // (not the tax line, which lumps it in with other operating expenses).
  const convCost = roundCents(
    expenses
      .filter((e) => e.category === "cost_of_conversion")
      .reduce((s, e) => s + Number(e.amount_sgd ?? 0), 0),
  );
 
  return {
    periodStart:  start,
    periodEnd:    end,
    periodLabel:  label,
    jurisdiction,
    revenue: {
      gold_sgd:        roundCents(goldRev),
      accessories_sgd: roundCents(accRev),
      conversion_sgd:  roundCents(convRev),
      shipping_sgd:    roundCents(shipRev),
      processing_sgd:  roundCents(procRev),
      total_sgd:       roundCents(totalRev),
    },
    cogs: {
      gold_sgd:        roundCents(goldCogs),
      accessories_sgd: roundCents(accCogs),
      total_sgd:       roundCents(totalCogs),
    },
    grossProfit_sgd: roundCents(grossProfit),
    commission_sgd:  roundCents(commission),
    shippingCushion: {
      charged_sgd:  shipCharged,
      cost_sgd:     shipCost,
      cushion_sgd:  shipCushion,
      coverage_pct: shipCoverage,
    },
    processingCushion: {
      charged_sgd:  procCharged,
      cost_sgd:     procCost,
      cushion_sgd:  procCushion,
      coverage_pct: procCoverage,
    },
    conversionCost: {
      fee_sgd:  roundCents(convRev),
      cost_sgd: convCost,
      net_sgd:  roundCents(convRev - convCost),
    },
    channels,
    expenses: {
      lines,
      total_sgd: roundCents(totalExpenses),
    },
    // Commission is a selling cost outside COGS/ledger — subtract it here so net
    // income reflects the true marketplace-fee drag.
    netIncome_sgd: roundCents(grossProfit - commission - totalExpenses),
  };
}
 
// Revenue is recognised at DELIVERY (lifecycle='completed'), not at payment.
// The route already windows orders by lifecycle_completed_at, so this is the
// single membership gate; it matches COGS, which is dated by inventory sold-date.
function isOrderRecognised(o: PaidOrderRow): boolean {
  return isRevenueRecognised(o.lifecycle);
}
 
function roundCents(v: number): number {
  return Math.round(v * 100) / 100;
}
