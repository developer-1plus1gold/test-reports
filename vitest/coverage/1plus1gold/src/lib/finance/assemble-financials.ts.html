/**
 * Assemble a full month-end financial-statements dataset — the Income Statement
 * for the month, its quarter-to-date and its fiscal-year-to-date, plus a
 * point-in-time Balance Sheet as at month-end — by driving the app's OWN pure
 * finance engines (buildPnlReport / buildBalanceSheet). One place so the on-demand
 * PDF route matches the numbers the /admin Ledger screens show.
 *
 * Basis is identical to /api/admin/finance/{pnl,balance-sheet}: revenue recognised
 * at delivery (lifecycle='completed', dated by lifecycle_completed_at), gold =
 * subtotal_ipm_sgd, COGS = landed cost, internal/demo customers excluded. The balance
 * sheet is lib/finance/balance-sheet-as-of — the SAME function the Balance Sheet screen
 * calls (one assembly since 2026-11-18; the screen used to keep its own copy).
 */
// eslint-disable-next-line @typescript-eslint/no-explicit-any
type Db = any;
 
import { getInternalCustomerIds, excludeInternalOrders } from "@/lib/finance/internal-customers";
import type { Period, Jurisdiction, PnlReport } from "@/lib/finance/pnl";
import type { BalanceSheet } from "@/lib/finance/balance-sheet";
import { orderRevenue, type OrderRevenueFields } from "@/lib/finance/order-revenue";
import { buildPnlForWindow } from "@/lib/finance/pnl-window";
import { balanceSheetAsOf, BOOKS_START } from "@/lib/finance/balance-sheet-as-of";
import { sgtDayStart, sgtDayEnd } from "@/lib/finance/sgt-window";
 
export interface FinancialStatements {
  /** `heading` is the cover line ("For the month ended 31 October 2026" or, for a
   *  month still in progress, "For the period 1 to 17 November 2026"); `monthCol` is
   *  the first column's short header; `asOf` the balance-sheet date. A month that
   *  has not ended is NEVER labelled with its month-end (2026-11-17 review). */
  periodLabel:   { month: string; quarter: string; fytd: string; asOf: string; heading?: string; monthCol?: string };
  /** True when the statement stops before the month's last day (a to-date statement). */
  partial?: boolean;
  /** P&L net income from the start of the business to `asOf` — the accumulated
   *  result the balance sheet carries. Differs from FYTD only by pre-FY activity. */
  cumulativeNetIncomeSgd?: number;
  month:         PnlReport;
  quarter:       PnlReport;
  /** Results by quarter for the fiscal year: each earlier quarter in full, then the
   *  current one to date (= `quarter`). Quarters with no revenue and no expenses are
   *  left out. Optional so older callers / fixtures still render. */
  quarters?:     Array<{ label: string; start: string; end: string; pnl: PnlReport }>;
  fytd:          PnlReport;
  balanceSheet:  BalanceSheet;
  /** Cumulative expense total on the books at month-end (drives the retained-earnings
   *  note; differs from FYTD expense by any entry dated before the fiscal year). */
  expensesToDateSgd: number;
  /** June order register etc. — the completed, non-internal orders delivered in the month. */
  orderRegister: Array<{ id: string; completedAt: string | null; channel: string | null; grams: number; goldRev: number; shipping: number; grand: number }>;
}
 
function pad(n: number) { return String(n).padStart(2, "0"); }
function lastDayOfMonth(year: number, month1: number) { return new Date(Date.UTC(year, month1, 0)).getUTCDate(); }
 
/** P&L for an explicit inclusive [startDate, endDate] window — the SAME assembly
 *  as the on-screen P&L and its CSV (lib/finance/pnl-window), so the statements
 *  cannot report a different net income for the same month. */
async function pnlForWindow(db: Db, internalIds: string[], startDate: string, endDate: string, period: Period, jurisdiction: Jurisdiction): Promise<PnlReport> {
  return buildPnlForWindow(db, {
    period, jurisdiction, internalIds,
    at: new Date(sgtDayEnd(endDate)),
    window: { start: startDate, end: endDate },
  });
}
 
const MONTHS = ["January", "February", "March", "April", "May", "June", "July", "August", "September", "October", "November", "December"];
 
/** Re-exported: the balance sheet's module owns it (one inception date for everything). */
export { BOOKS_START };
 
/** The statement's end date: the requested `asOf` clamped into the month, else the
 *  month's last day. A date after month-end, or before its first day, is clamped. */
export function statementEndDate(year: number, month1: number, asOf?: string | null): { asOf: string; endDay: number; partial: boolean } {
  const last = lastDayOfMonth(year, month1);
  const monthEnd = `${year}-${pad(month1)}-${pad(last)}`;
  const monthStart = `${year}-${pad(month1)}-01`;
  if (!asOf || !/^\d{4}-\d{2}-\d{2}$/.test(asOf) || asOf >= monthEnd) return { asOf: monthEnd, endDay: last, partial: false };
  const clamped = asOf < monthStart ? monthStart : asOf;
  return { asOf: clamped, endDay: Number(clamped.slice(8, 10)), partial: true };
}
 
/** Assemble the statement set for a year + month (1-12) — to its month-end, or to
 *  `opts.asOf` for a month still in progress ("1 to 17 November"). */
export async function assembleFinancialStatements(
  db: Db, year: number, month1: number, jurisdiction: Jurisdiction = "sg", opts: { asOf?: string | null } = {},
): Promise<FinancialStatements> {
  const internalIds = await getInternalCustomerIds(db);
  const { asOf, endDay, partial } = statementEndDate(year, month1, opts.asOf);
  const qStart1  = Math.floor((month1 - 1) / 3) * 3 + 1;          // first month of this quarter
  const monthStart   = `${year}-${pad(month1)}-01`;
  const quarterStart = `${year}-${pad(qStart1)}-01`;
  const fytdStart    = `${year}-01-01`;
 
  const curQ = Math.floor((month1 - 1) / 3) + 1;
  const earlierQ = Array.from({ length: curQ - 1 }, (_, i) => {
    const q = i + 1, m1 = (q - 1) * 3 + 1, m3 = m1 + 2;
    return { q, start: `${year}-${pad(m1)}-01`, end: `${year}-${pad(m3)}-${pad(lastDayOfMonth(year, m3))}` };
  });
  const [month, quarter, fytd, cumulative, ...earlier] = await Promise.all([
    pnlForWindow(db, internalIds, monthStart, asOf, "month", jurisdiction),
    pnlForWindow(db, internalIds, quarterStart, asOf, "quarter", jurisdiction),
    pnlForWindow(db, internalIds, fytdStart, asOf, "year", jurisdiction),
    pnlForWindow(db, internalIds, BOOKS_START, asOf, "year", jurisdiction),
    ...earlierQ.map((w) => pnlForWindow(db, internalIds, w.start, w.end, "quarter", jurisdiction)),
  ]);
  const active = (p: PnlReport) => Math.abs(p.revenue.total_sgd) > 0.005 || Math.abs(p.expenses.total_sgd) > 0.005;
  const quarters = [
    ...earlierQ.map((w, i) => ({ label: `Q${w.q}`, start: w.start, end: w.end, pnl: earlier[i] })).filter((x) => active(x.pnl)),
    { label: `Q${curQ} to date`, start: quarterStart, end: asOf, pnl: quarter },
  ];
  // The balance sheet carries the P&L's own inception-to-date result, so equity and
  // the income statement cannot disagree (see buildBalanceSheet.retainedEarningsSgd).
  const bs = await balanceSheetAsOf(db, asOf, { internalIds, retainedEarningsSgd: cumulative.netIncome_sgd, jurisdiction });
 
  // Order register for the month (delivered, non-internal).
  const startTs = sgtDayStart(monthStart), endTs = sgtDayEnd(asOf);
  const { data: regRows } = await excludeInternalOrders(
    db.from("orders").select("id, lifecycle_completed_at, source_channel, total_grams, subtotal_ipm_sgd, discount_ipm_apportioned_sgd, shipping_fee_sgd, grand_total_sgd")
      .eq("lifecycle", "completed").gte("lifecycle_completed_at", startTs).lte("lifecycle_completed_at", endTs).order("lifecycle_completed_at", { ascending: true }),
    internalIds,
  );
  const orderRegister = ((regRows ?? []) as Array<Record<string, unknown>>).map((o) => ({
    id: String(o.id).slice(0, 8), completedAt: (o.lifecycle_completed_at as string | null), channel: (o.source_channel as string | null),
    grams: Number(o.total_grams ?? 0), goldRev: orderRevenue(o as OrderRevenueFields).gold, shipping: Number(o.shipping_fee_sgd ?? 0), grand: Number(o.grand_total_sgd ?? 0),
  }));
 
  const qLabel = `Q${Math.floor((month1 - 1) / 3) + 1} ${year}`;
  const mName = MONTHS[month1 - 1], mShort = mName.slice(0, 3);
  return {
    periodLabel: {
      month:   partial ? `1–${endDay} ${mName} ${year}` : `${mName} ${year}`,
      // First month of a quarter reads "Q4 2026 (Oct)", not "(Oct–Oct)".
      quarter: qStart1 === month1
        ? `${qLabel} to date (${partial ? `${mShort} 1–${endDay}` : mShort})`
        : `${qLabel} to date (${MONTHS[qStart1 - 1].slice(0, 3)}–${partial ? `${endDay} ` : ""}${mShort})`,
      fytd:    `FY${year} YTD (through ${endDay} ${mShort})`,
      asOf:    `${endDay} ${mName} ${year}`,
      heading: partial ? `For the period 1 to ${endDay} ${mName} ${year}` : `For the month ended ${endDay} ${mName} ${year}`,
      monthCol: partial ? `${mShort} 1–${endDay}` : `${mName} ${year}`,
    },
    partial,
    cumulativeNetIncomeSgd: Math.round(cumulative.netIncome_sgd * 100) / 100,
    month, quarter, quarters, fytd,
    balanceSheet: bs.sheet, expensesToDateSgd: Math.round(bs.expensesToDate * 100) / 100,
    orderRegister,
  };
}
