/**
 * Stock adjustments → P&L expense lines.
 *
 * WHY (2026-07-19, spotted by Mukesh): recording a stock adjustment lowered the
 * balance-sheet inventory asset — non-gold inventory is valued
 * `stock_qty × avg landed cost` — but nothing recorded the matching expense.
 * Writing off three damaged gift boxes made value disappear from the balance
 * sheet with no P&L entry: the loss was silently absorbed by equity instead of
 * being reported. Half of a double entry is not bookkeeping.
 *
 * This derives the missing side. Adjustments are NOT copied into
 * `ledger_entries` — they are derived at read time from the append-only
 * `stock_adjustments` table, so there is exactly one source of truth. Writing a
 * second row would let the two diverge (adjustments are immutable; ledger
 * entries are editable and deletable), which is the failure mode this whole
 * investigation started from.
 *
 * Valuation uses the SAME `getAvgLandedCostByProduct` the balance sheet uses to
 * value the asset, so the expense always equals the cost of the asset removed.
 *
 * TREATMENT BY REASON — not every adjustment is a loss:
 *   damaged / lost / write_off / other → operating expense (inventory written off)
 *   sample                             → marketing expense (a deliberate spend)
 *   found / miscount                   → NO expense. These are restatements of
 *                                        the asset: the stock was always right
 *                                        or wrong, we just recorded it better.
 *                                        Booking an expense for a corrected
 *                                        count would invent a loss that never
 *                                        happened.
 *
 * Only NEGATIVE deltas of an expense-bearing reason produce a line. A positive
 * delta (stock found) is an asset restatement — the balance sheet picks it up
 * via stock_qty and no P&L entry belongs.
 */
 
import type { SupabaseClient } from "@supabase/supabase-js";
import { getAvgLandedCostByProduct } from "@/lib/inventory/landed-cost";
import type { ExpenseRow } from "@/lib/finance/pnl";
 
interface Treatment {
  /** null = no P&L impact (asset restatement only). */
  category:        string | null;
  sg_tax_category: string | null;
  us_tax_category: string | null;
}
 
export const ADJUSTMENT_TREATMENT: Record<string, Treatment> = {
  damaged:   { category: "inventory_write_off", sg_tax_category: "Other operating expenses", us_tax_category: "Line 27a — Other (inventory write-off)" },
  lost:      { category: "inventory_write_off", sg_tax_category: "Other operating expenses", us_tax_category: "Line 27a — Other (inventory write-off)" },
  write_off: { category: "inventory_write_off", sg_tax_category: "Other operating expenses", us_tax_category: "Line 27a — Other (inventory write-off)" },
  other:     { category: "inventory_write_off", sg_tax_category: "Other operating expenses", us_tax_category: "Line 27a — Other (inventory write-off)" },
  sample:    { category: "marketing_advertising", sg_tax_category: "Advertising & marketing", us_tax_category: "Line 8 — Advertising" },
  // Restatements — the stock count changed, no economic loss occurred.
  found:     { category: null, sg_tax_category: null, us_tax_category: null },
  miscount:  { category: null, sg_tax_category: null, us_tax_category: null },
  // Units that went out on a PAID ORDER but never decremented. Emphatically NOT
  // an expense: the sale already recognised revenue AND cost (accessory COGS),
  // so billing here would charge the same units twice. Also excluded from
  // on_hand_expected — `sold_real` already accounts for them — so correcting
  // the counter closes the delta instead of moving both sides together.
  sold_unrecorded: { category: null, sg_tax_category: null, us_tax_category: null },
};
 
/** True when this reason represents a real economic loss/spend. */
export function isExpenseBearing(reasonCode: string): boolean {
  return ADJUSTMENT_TREATMENT[reasonCode]?.category != null;
}
 
/**
 * Reasons that represent a real physical MOVEMENT of stock, as opposed to a
 * RESTATEMENT of a wrong count.
 *
 * This is the same split twice over, and that is deliberate:
 *   - movement   → hits the P&L AND `on_hand_expected` (a real event happened)
 *   - restatement→ hits neither (the count was wrong; nothing moved)
 *
 * ⚠️ MUST match the `reason_code IN (...)` list in the `adj_move` CTE of
 * migration 20260719d. SQL cannot import this constant, so a test asserts the
 * two stay in sync — if you add a reason_code, update BOTH.
 */
export const MOVEMENT_REASONS: readonly string[] =
  Object.entries(ADJUSTMENT_TREATMENT)
    .filter(([, t]) => t.category != null)
    .map(([code]) => code)
    .sort();
 
/**
 * Expense rows derived from stock adjustments in [startDate, endDate] (dates,
 * inclusive — same convention as the ledger_entries query these join).
 *
 * Returns [] rather than throwing on a query failure: the P&L must still render
 * without this supplementary detail. A silent omission is logged loudly.
 */
export async function getStockAdjustmentExpenses(
  db: SupabaseClient,
  startDate: string,
  endDate: string,
): Promise<ExpenseRow[]> {
  try {
    const { data, error } = await db
      .from("stock_adjustments")
      .select("product_id, delta, reason_code, created_at")
      .gte("created_at", `${startDate}T00:00:00Z`)
      .lte("created_at", `${endDate}T23:59:59Z`);
 
    if (error) {
      console.error("[stock-adjustment-expenses] query failed:", error.message);
      return [];
    }
 
    const rows = (data ?? []) as Array<{
      product_id: string; delta: number; reason_code: string; created_at: string;
    }>;
 
    // Movement reasons bill SYMMETRICALLY: expense = -delta × cost. A negative
    // delta (units left) is an expense; a POSITIVE one is a credit reversing a
    // previous entry. That symmetry is required, not cosmetic — the ledger is
    // append-only, so the ONLY way to correct a mis-reasoned adjustment is a
    // compensating row, and if positives never credited, a mistake could never
    // be undone in the books.
    //
    // Restatement reasons (miscount, found) never bill in either direction —
    // correcting a count is not an economic event.
    const billable = rows.filter((r) => isExpenseBearing(r.reason_code));
    if (billable.length === 0) return [];
 
    const costs = await getAvgLandedCostByProduct(db, billable.map((r) => r.product_id));
 
    const out: ExpenseRow[] = [];
    for (const r of billable) {
      const perUnit = costs.get(r.product_id)?.avgPerUnit ?? 0;
      // A product with no PO receipts has no cost basis. Emitting a $0 expense
      // line would be noise pretending to be information — skip it. The unit
      // still leaves the asset via stock_qty (which is also valued at 0), so
      // the books stay consistent either way.
      if (perUnit <= 0) continue;
 
      const t = ADJUSTMENT_TREATMENT[r.reason_code];
      out.push({
        entry_date:      r.created_at.slice(0, 10),
        // -delta: units out (negative delta) → positive expense; a reversing
        // positive delta → negative expense (credit).
        amount_sgd:      Number((-r.delta * perUnit).toFixed(2)),
        category:        t.category,
        sg_tax_category: t.sg_tax_category,
        us_tax_category: t.us_tax_category,
      });
    }
    return out;
  } catch (err) {
    console.error("[stock-adjustment-expenses] unexpected failure:", err instanceof Error ? err.message : err);
    return [];
  }
}
