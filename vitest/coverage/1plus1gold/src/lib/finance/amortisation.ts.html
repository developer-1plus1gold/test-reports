/**
 * Prepaid expenses — one ledger row, recognised across N months.
 *
 * WHY (2026-09-04, Mukesh on the LinkedIn Premium Company Page annual):
 * *"I think amortizing it over 12 months is a better route."* An annual
 * subscription expensed in the month it is paid puts a year of cost into one
 * month, and here it would also have restated June — a month already issued as a
 * statements PDF — from net (S$1,517.13) to about (S$2,669).
 *
 * ONE ROW, SPREAD AT READ TIME. Twelve rows of a twelfth was rejected: a ledger
 * entry stands for a real document and a real cash movement, and splitting it
 * breaks `reconciliation.ts`, which matches ledger cash rows against a bank
 * statement. One payment left the card, so there is one row.
 *
 * ⚠️ THE HALF-ENTRY, mirrored from equipment depreciation. `buildBalanceSheet`
 * SOLVES for cash:
 *
 *     impliedCash = liabilities + equity − goldInventory − nonGoldInventory − …
 *
 * A personally-funded prepaid puts the FULL amount into the director's-loan
 * liability the day it is paid — Mukesh is owed all of it regardless of how the
 * cost is recognised — while equity falls by only one month's slice. Without a
 * PREPAID ASSET carrying the unexpired remainder, the cash plug absorbs the
 * difference and implied cash is OVERSTATED by it. And nothing fails: the sheet
 * still reports `balanced: true`, because it balances by construction. Hence
 * `unexpiredAsOf`, which the balance sheet must surface as an asset.
 *
 * Amortisation is a P&L TIMING question and never a funding one: the payable is
 * untouched by everything here.
 */
 
const n2 = (v: number) => Math.round(v * 100) / 100;
 
export interface AmortisableEntry {
  entry_date:       string;
  amount_sgd:       number | null;
  /** NULL = recognise it all on entry_date — the default, and prior behaviour. */
  amortise_months?: number | null;
  /** NULL falls back to entry_date. */
  amortise_start?:  string | null;
}
 
/** "2026-06" + 4 → "2026-10". */
function addMonths(month: string, n: number): string {
  const [y, m] = month.split("-").map(Number);
  const total = y * 12 + (m - 1) + n;
  return `${Math.floor(total / 12)}-${String((total % 12) + 1).padStart(2, "0")}`;
}
 
/** How many months this entry spreads over. 1 for anything not amortised. */
export function spreadMonths(e: AmortisableEntry): number {
  const n = Number(e.amortise_months ?? 0);
  return Number.isFinite(n) && n >= 1 ? Math.floor(n) : 1;
}
 
/** First month of recognition. */
export function startMonth(e: AmortisableEntry): string {
  return (e.amortise_start || e.entry_date || "").slice(0, 7);
}
 
/**
 * The monthly slices: `[{ month, amount_sgd }]`, summing to the full amount.
 *
 * ⚠️ Σ THEN ROUND — the FINAL month absorbs the residual. S$1,151.88 over 12 is
 * S$95.99 exactly, but S$100 over 3 is not, and Σ-of-rounded would leave a cent
 * of prepaid asset that could never be released. That drift has already caused
 * three separate bugs here (Money Map gold COGS, the supplier lens, and the
 * equipment schedule this mirrors).
 */
export function schedule(e: AmortisableEntry): Array<{ month: string; amount_sgd: number }> {
  const total = n2(Number(e.amount_sgd ?? 0));
  const start = startMonth(e);
  if (!start || total === 0) return [];
  const n = spreadMonths(e);
  if (n === 1) return [{ month: start, amount_sgd: total }];
 
  const per = n2(total / n);
  const out: Array<{ month: string; amount_sgd: number }> = [];
  let assigned = 0;
  for (let i = 0; i < n - 1; i++) {
    out.push({ month: addMonths(start, i), amount_sgd: per });
    assigned = n2(assigned + per);
  }
  out.push({ month: addMonths(start, n - 1), amount_sgd: n2(total - assigned) });
  return out;
}
 
/**
 * The slice of this entry belonging to [from, to] (ISO dates).
 *
 * A non-amortised entry recognises in the month of its entry_date, which is what
 * the P&L already does, so passing every entry through this changes nothing for
 * the rows that carry no `amortise_months`.
 */
export function recognisedInPeriod(e: AmortisableEntry, from: string, to: string): number {
  const f = from.slice(0, 7), t = to.slice(0, 7);
  return n2(schedule(e).filter((s) => s.month >= f && s.month <= t)
                       .reduce((a, s) => a + s.amount_sgd, 0));
}
 
/**
 * The unexpired remainder as of a date — the PREPAID ASSET.
 *
 * Everything after the month of `asOf` is cost paid for but not yet consumed.
 * Clamped to [0, amount] so a bad date can never invent an asset or a negative
 * one.
 */
export function unexpiredAsOf(e: AmortisableEntry, asOf: string): number {
  if (!e.amortise_months) return 0;          // not amortised → nothing prepaid
  const upto = asOf.slice(0, 7);
  const rest = schedule(e).filter((s) => s.month > upto).reduce((a, s) => a + s.amount_sgd, 0);
  return n2(Math.min(Math.max(rest, 0), n2(Number(e.amount_sgd ?? 0))));
}
 
/** Σ of the unexpired remainders — what the balance sheet carries as an asset. */
export function prepaidAssetAsOf(entries: AmortisableEntry[], asOf: string): number {
  return n2(entries.reduce((a, e) => a + unexpiredAsOf(e, asOf), 0));
}
 
/**
 * Validate the two amortisation fields coming off an admin request.
 *
 * ⚠️ THIS EXISTS BECAUSE THE WRITE PATH SILENTLY DROPPED THEM. The ledger POST
 * destructures a fixed field list and the PATCH filters through an allow-list;
 * `amortise_months` was in neither, so a client could send it, get HTTP 201, and
 * have the entry recognise its whole amount in one month anyway. Same shape as
 * the `create_order_atomic` column drift: a mapping that ignores what it does
 * not name fails without an error. Adding a column is never the whole job.
 *
 * Returns the NORMALISED values so callers write one shape: absent or null
 * clears the spread (today's default behaviour), and a start is only kept
 * alongside months — a start with no months would be inert and misleading.
 */
export function validateAmortisation(input: { amortise_months?: unknown; amortise_start?: unknown }):
  | { ok: true; months: number | null; start: string | null }
  | { ok: false; error: string } {
  const rawM = input.amortise_months;
  if (rawM === undefined || rawM === null || rawM === "") {
    return { ok: true, months: null, start: null };
  }
  // ⚠️ Type-check BEFORE coercing. `Number(true)` is 1, so a JSON `true` would
  // otherwise sail through as a one-month spread — a caught-by-its-own-test
  // hole, and exactly the kind of quiet coercion that puts a meaningless
  // schedule on a real expense.
  if (typeof rawM !== "number" && typeof rawM !== "string") {
    return { ok: false, error: "amortise_months must be a whole number of months between 1 and 600" };
  }
  const months = Number(rawM);
  if (!Number.isInteger(months) || months < 1 || months > 600) {
    return { ok: false, error: "amortise_months must be a whole number of months between 1 and 600" };
  }
  const rawS = input.amortise_start;
  if (rawS === undefined || rawS === null || rawS === "") {
    return { ok: true, months, start: null };   // falls back to entry_date
  }
  if (typeof rawS !== "string" || !/^\d{4}-\d{2}-\d{2}$/.test(rawS) || Number.isNaN(Date.parse(rawS))) {
    return { ok: false, error: "amortise_start must be a date (YYYY-MM-DD)" };
  }
  return { ok: true, months, start: rawS };
}
