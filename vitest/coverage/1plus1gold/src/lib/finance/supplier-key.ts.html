/**
 * Supplier identity — resolving THREE name spaces that do not join.
 *
 * The money map can attribute every sold bar to a supplier, but the supplier is
 * named differently depending on which table it came from:
 *
 *   purchase_orders.supplier_name   free text, typed by an operator
 *   suppliers.name                  the canonical table (used by consignment)
 *   ledger_entries.vendor           free text, mixes suppliers with utilities
 *
 * Live consequence, found on PROD 2026-09-03: `BulionKing` and `BullionKing`
 * were the same counterparty, splitting S$32.7k across two rows and quietly
 * understating a concentration finding. The row itself was corrected by
 * migration `20260903_supplier_name_typo`, but free text drifts again the next
 * time somebody types it — so the FIX IS BOTH: correct the data, and normalise
 * at read time.
 *
 * ⚠️ RESOLVE, DO NOT RENAME. `supplierKey` is a comparison key; the display name
 * stays whatever the source row says (post-migration, the correct spelling). A
 * purchase order must keep reading as the operator wrote it — that is what makes
 * it an auditable record.
 *
 * The tempting wrong fix is a `purchase_orders.supplier_id` column. That is the
 * right long-term shape, but it is a schema change on the PROCUREMENT path to
 * serve a reporting view, and its backfill needs this same matching anyway.
 */
 
/**
 * Deliberate merges, each dated and reasoned. An alias is a decision somebody
 * made, not a silent join — anything not listed here stays a separate supplier.
 *
 * Keys and values are both already-normalised keys (see `normalise`).
 */
const ALIASES: Record<string, string> = {
  // 2026-09-03 — a typo on one purchase order. Verified on PROD: both spellings
  // resolve to the same counterparty, 8 bars and S$32.7k of landed cost between
  // them. The source row was corrected by migration in the same change; this
  // entry catches the next occurrence rather than waiting for another split.
  bulionking: "bullionking",
};
 
/** Legal suffixes that carry no identity — "X Pte Ltd" and "X" are one supplier. */
const SUFFIXES = [
  "pte ltd", "pte. ltd.", "pte", "ltd", "limited", "llc", "inc", "incorporated",
  "corp", "corporation", "co", "company", "gmbh", "sa", "ag", "bv", "plc",
];
 
function normalise(raw: string): string {
  let s = raw.toLowerCase().trim();
  // Drop anything parenthesised — "(Singapore)" is a jurisdiction, not an identity.
  s = s.replace(/\([^)]*\)/g, " ");
  s = s.replace(/[.,''`"]/g, " ").replace(/[-_/&+]/g, " ");
  s = s.replace(/\s+/g, " ").trim();
  // Strip trailing legal suffixes, repeatedly: "Pte. Ltd." is two of them.
  let changed = true;
  while (changed) {
    changed = false;
    for (const suf of SUFFIXES) {
      if (s.endsWith(" " + suf)) { s = s.slice(0, -(suf.length + 1)).trim(); changed = true; }
    }
  }
  return s.replace(/\s+/g, "");
}
 
/**
 * The comparison key for a supplier name. Two names with the same key are the
 * same supplier; an empty or missing name yields "" and the caller decides what
 * to show (the money is real either way — never drop a row for want of a name).
 */
export function supplierKey(name: string | null | undefined): string {
  if (!name) return "";
  const n = normalise(name);
  return ALIASES[n] ?? n;
}
 
/**
 * Pick the display name for a group of rows that share a key.
 *
 * Longest-then-alphabetical, deliberately: a typo is usually a DROPPED letter
 * ("BulionKing" vs "BullionKing"), and the fuller spelling is also the one more
 * likely to carry the legal form somebody would recognise. Ties break
 * alphabetically so the label cannot flip between renders.
 */
export function supplierDisplayName(names: string[]): string {
  const real = names.map((n) => (n ?? "").trim()).filter(Boolean);
  if (real.length === 0) return "Unnamed supplier";
  return [...new Set(real)].sort((a, b) => b.length - a.length || a.localeCompare(b))[0];
}
 
/**
 * Names that are a BOOKKEEPING BUCKET rather than a trading relationship, and
 * must never be ranked as a supplier we could negotiate with or depend on.
 * Their cost is real and stays in every total — only the framing differs.
 */
export function isPseudoSupplier(name: string | null | undefined): boolean {
  return /^(home\s*inventory|opening\s*(stock|balance)|founder|owner)/i.test((name ?? "").trim());
}
