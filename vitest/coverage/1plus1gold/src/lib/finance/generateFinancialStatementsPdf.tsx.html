/** Results by quarter (Mukesh 2026-11-17): the trend a counterparty wants, which the
 *  month / quarter-to-date / year-to-date columns cannot show. Each earlier quarter
 *  in full, the current one to date, then FY YTD; the columns add up to FY YTD. */
function quarterlyTable(d: FinancialStatements) {
  const qs = d.quarters;
  if (!qs || qs.length < 2) return null; // first quarter of the year: the main columns already say it all
  const cols = [...qs.map((q) => ({ label: q.label, p: q.pnl })), { label: "FY YTD", p: d.fytd }];
  const other = (p: PnlReport) => p.supplierFees_sgd + p.expenses.total_sgd + p.commission_sgd + (p.marketplaceShipping_sgd ?? 0);
  const rows: Array<{ label: string; f: (p: PnlReport) => number; cls?: Style }> = [
    { label: "Total revenue", f: (p) => p.revenue.total_sgd },
    { label: "Gross profit / (loss)", f: (p) => p.grossProfit_sgd },
    { label: "Operating expenses & other", f: (p) => -other(p) },
    { label: "Net profit / (loss)", f: (p) => p.netIncome_sgd, cls: s.total },
  ];
  const cellSm = (v: number): Style[] => (isNeg(v) ? [s.cellSm, s.neg] : [s.cellSm]);
  const started = BUSINESS_CONFIG.tradingStarted;
  const startQ = qs.find((q) => started >= q.start && started <= q.end);
  const preTrading = qs.filter((q) => q.end < started);
  return (
    <View>
      <Text style={[s.h3, { marginTop: 14 }]}>Results by quarter</Text>
      <View style={[s.row, { borderBottomWidth: 1.4, borderBottomColor: C.rule2 }]}>
        <Text style={[s.lbl, s.head]}>Figures in S$</Text>
        {cols.map((c) => <Text key={c.label} style={[s.cellSm, s.head]}>{c.label}</Text>)}
      </View>
      {rows.map((r) => (
        <View key={r.label} style={r.cls ? [s.row, r.cls] : [s.row]}>
          <Text style={[s.lbl]}>{r.label}</Text>
          {cols.map((c) => <Text key={c.label} style={cellSm(r.f(c.p))}>{acc(r.f(c.p))}</Text>)}
        </View>
      ))}
      {startQ && (
        <Text style={s.note}>Trading began on {fmtLongDate(started)}. {preTrading.length > 0 ? `${preTrading.map((q) => q.label).join(" and ")} ${preTrading.length > 1 ? "are" : "is"} set-up costs before trading; ` : ""}{startQ.label} holds the first days of sales together with set-up costs.</Text>
      )}
    </View>
  );
}
const fmtLongDate = (iso: string) => new Date(`${iso}T00:00:00Z`).toLocaleDateString("en-GB", { day: "numeric", month: "long", year: "numeric", timeZone: "UTC" });

function BsRow({ label, v, total, grand }: { label: string; v: number; total?: boolean; grand?: boolean; }) {
  const rowStyle: Style[] = [s.row]; if (total) rowStyle.push(s.total); if (grand) rowStyle.push(s.grand);
  const lblStyle: Style[] = [s.lbl]; if (!total && !grand) lblStyle.push(s.indent);
  const valStyle: Style[] = [s.cell]; if (isNeg(v)) valStyle.push(s.neg);
  return (
    <View style={rowStyle}>
      <Text style={lblStyle}>{label}</Text>
      <Text style={valStyle}>{acc(v)}</Text>
    </View>
  );
}
