function ThreeRow({ label, m, q, y, cls, indent }: { label: string; m: string; q: string; y: string; cls?: Style; indent?: Style; }) {
  const cell = (t: string): Style[] => (t.startsWith("(") ? [s.cell, s.neg] : [s.cell]);
  return (
    <View style={cls ? [s.row, cls] : [s.row]}>
      <Text style={indent ? [s.lbl, indent] : [s.lbl]}>{label}</Text>
      <Text style={cell(m)}>{m}</Text>
      <Text style={cell(q)}>{q}</Text>
      <Text style={cell(y)}>{y}</Text>
    </View>
  );
}
 
function incomeStatement(d: FinancialStatements) {
  const R = [d.month, d.quarter, d.fytd];
  const rev = (f: (r: PnlReport) => number) => R.map((r) => acc(f(r))) as [string, string, string];
  const [rg, rgq, rgy] = rev((r) => r.revenue.gold_sgd);
  return (
    <View>
      <View style={[s.row, { borderBottomWidth: 1.4, borderBottomColor: C.rule2 }]}>
        <Text style={[s.lbl, s.head]}>Figures in S$</Text>
        <Text style={[s.cell, s.head]}>{d.periodLabel.month}</Text>
        <Text style={[s.cell, s.head]}>{d.periodLabel.quarter.split(" (")[0]}</Text>
        <Text style={[s.cell, s.head]}>FY YTD</Text>
      </View>
      <Text style={s.sec}>Revenue</Text>
      <ThreeRow label="Gold sales (investment precious metals)" m={rg} q={rgq} y={rgy} indent={s.indent} />
      <ThreeRow label="Accessories" {...trip(R, (r) => r.revenue.accessories_sgd)} indent={s.indent} />
      <ThreeRow label="Shipping recovered" {...trip(R, (r) => r.revenue.shipping_sgd)} indent={s.indent} />
      <ThreeRow label="Payment processing recovered" {...trip(R, (r) => r.revenue.processing_sgd)} indent={s.indent} />
      <ThreeRow label="Total revenue" {...trip(R, (r) => r.revenue.total_sgd)} cls={s.total} />
      <Text style={s.sec}>Cost of sales</Text>
      <ThreeRow label="Gold, at landed cost" {...trip(R, (r) => -r.cogs.gold_sgd)} indent={s.indent} />
      <ThreeRow label="Accessories" {...trip(R, (r) => -r.cogs.accessories_sgd)} indent={s.indent} />

      <ThreeRow label="Gross profit / (loss)" {...trip(R, (r) => r.grossProfit_sgd)} cls={s.total} />
      <ThreeRow label="Gross margin" m={pctTxt(pctOf(R[0].grossProfit_sgd, R[0].revenue.total_sgd))} q={pctTxt(pctOf(R[1].grossProfit_sgd, R[1].revenue.total_sgd))} y={pctTxt(pctOf(R[2].grossProfit_sgd, R[2].revenue.total_sgd))} indent={s.indent2} />

  );
}
