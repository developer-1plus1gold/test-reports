/**
 * Sold inventory → P&L COGS rows: the ONE select and mapper every P&L reader
 * uses (the P&L route, the financial-statements assembler, the P&L CSV).
 *
 * They used to carry three hand-copied versions. (The CSV's had drifted: it
 * never selected the conversion / Gold Yield links. It still charges those bars
 * at inventory cost on purpose — see its route — until it derives the matching
 * expenses the P&L route does.)
 */
import type { SoldInventoryRow } from "@/lib/finance/pnl";
 
export const SOLD_INVENTORY_SELECT =
  "sold_at, updated_at, cost_per_g_sgd, gold_yield_bar_id, products(grams, gst_exempt), " +
  "order_fulfilments(conversion_id, order_items(order_id, orders(source_channel)))";
 
const one = <T,>(v: T | T[] | null | undefined): T | null => (Array.isArray(v) ? v[0] ?? null : v ?? null);
 
/** Flatten one inventory row (supabase models the FK joins as arrays). */
// eslint-disable-next-line @typescript-eslint/no-explicit-any
export function toSoldInventoryRowFromInventory(i: any): SoldInventoryRow {
  const p  = one(i.products) as { grams?: number | null; gst_exempt?: boolean | null } | null;
  const of = one(i.order_fulfilments) as { conversion_id?: string | null; order_items?: unknown } | null;
  const oi = of ? (one(of.order_items as never) as { orders?: unknown } | null) : null;
  const or = oi ? (one(oi.orders as never) as { source_channel?: string | null } | null) : null;
  return {
    // sold_at is backfilled + trigger-stamped for every sold row; the
    // updated_at fallback only covers a row written before 20260723d landed.
    sold_at:           (i.sold_at ?? i.updated_at) as string,
    cost_per_g_sgd:    (i.cost_per_g_sgd ?? null) as number | null,
    grams:             p?.grams ?? null,
    source_channel:    or?.source_channel ?? null,
    conversion_id:     of?.conversion_id ?? null,
    gold_yield_bar_id: (i.gold_yield_bar_id ?? null) as string | null,
    // A piece that is not investment-grade (products.gst_exempt false, e.g. 22KT
    // jewellery) — its cost goes to the Non-IPM goods line, beside its revenue.
    non_ipm:           p?.gst_exempt === false,
  };
}
