/**
 * PSPM Semi-Annual Return aggregation — the Product & Sales information MinLaw
 * (Ministry of Law) requires from a registered PSPM dealer twice a year. Pure
 * logic, unit-testable without a DB.
 *
 * MinLaw wants, for the period: total sales (S$) + number of transactions per
 * PAYMENT MODE bucket (cash / cash-equivalent, cryptocurrency, other non-cash),
 * a grand total, the non-face-to-face (online) split, the highest-priced product
 * sold + carried (as a value → bucket), and the 3 highest-valued transactions.
 *
 * A "transaction" = a PAID, non-reversed order for PSPM (gold) products. Internal
 * (demo/test) customers are excluded by the caller. ALL our sales are online, so
 * everything is non-face-to-face.
 */
 
export type PspmPeriod = "day" | "month" | "quarter" | "half" | "year";
 
/** MinLaw payment-mode buckets. */
export type PspmBucket = "cash" | "crypto" | "other";
 
/**
 * Map our payment_method to a MinLaw bucket. COD is cash-on-delivery (cash /
 * cash-equivalent). We take no crypto today. Everything else (HitPay card/PayNow/
 * Atome/GrabPay, Stripe card/PayNow) is electronic → "other non-cash".
 */
export function pspmBucket(method: string | null | undefined): PspmBucket {
  if (method === "cod") return "cash";
  // Reserved for a future crypto gateway:
  if (method && /crypto|bitcoin|eth|usdt|xsgd|stablecoin/i.test(method)) return "crypto";
  return "other";
}
 
/** MinLaw's per-transaction "Payment Mode" dropdown label. */
export function pspmPaymentLabel(method: string | null | undefined): string {
  return pspmBucket(method) === "cash" ? "Cash"
    : pspmBucket(method) === "crypto" ? "Cryptocurrency"
    : "Electronic payments";
}
 
/** MinLaw's value buckets for "highest priced product". */
export const PSPM_VALUE_BUCKETS = [
  { min: 500_000, label: "S$500,000 and above" },
  { min: 100_000, label: "S$100,000 to < S$500,000" },
  { min: 20_000,  label: "S$20,000 to < S$100,000" },
  { min: 10_000,  label: "S$10,000 to < S$20,000" },
  { min: 5_000,   label: "S$5,000 to < S$10,000" },
  { min: 2_000,   label: "S$2,000 to < S$5,000" },
  { min: 0,       label: "S$0 to < S$2,000" },
] as const;
 
export function pspmValueBucketLabel(value: number): string {
  return (PSPM_VALUE_BUCKETS.find((b) => value >= b.min) ?? PSPM_VALUE_BUCKETS[PSPM_VALUE_BUCKETS.length - 1]).label;
}
 
export interface PspmOrderRow {
  id:                string;
  ref:               string | null;   // invoice serial / hitpay ref — for the top-3 list
  grand_total_sgd:   number | null;
  payment_method:    string | null;
  created_at:        string;
}
 
interface BucketTotal {
  sales: number;
  count: number;
  /** This bucket's sales as a % of grand-total sales (0–100, 1dp). */
  salesPct: number;
  /** This bucket's transaction count as a % of grand-total count (0–100, 1dp). */
  countPct: number;
}
 
export interface PspmReport {
  periodStart: string;
  periodEnd:   string;
  periodLabel: string;
  byMode: {
    cash:  BucketTotal;
    crypto: BucketTotal;
    other: BucketTotal;
  };
  grandTotal: BucketTotal;
  /** All sales are online → this equals the grand total; surfaced separately
   *  because MinLaw asks for it explicitly. */
  nonFaceToFace: BucketTotal;
  /** Highest single transaction value + its MinLaw bucket. */
  highestSold: { value: number; bucket: string } | null;
  /** Highest-value product we carried in inventory + its bucket (caller supplies). */
  highestCarried: { value: number; bucket: string } | null;
  /** The 3 highest-valued transactions. */
  top3: Array<{ ref: string | null; value: number; paymentLabel: string; nonFaceToFace: boolean }>;
}
 
function round2(v: number): number { return Math.round(v * 100) / 100; }
 
/**
 * Resolve [start, end] (inclusive) + a label for a period anchored at `at`.
 * Day / Month / Quarter / Half / Year. The statutory PSPM filing period is
 * SEMI-ANNUAL — "half" (H1 = Q1+Q2 = Jan–Jun, H2 = Q3+Q4 = Jul–Dec) is the
 * default and the one MinLaw actually files against; the others are for drill-down.
 */
export function resolvePspmPeriod(period: PspmPeriod, at: Date): { start: string; end: string; label: string } {
  const y = at.getUTCFullYear(), m = at.getUTCMonth(), d = at.getUTCDate();
  let start: Date, end: Date, label: string;
  if (period === "day") {
    start = new Date(Date.UTC(y, m, d));
    end   = new Date(Date.UTC(y, m, d + 1) - 1);
    label = start.toLocaleDateString("en-SG", { day: "numeric", month: "long", year: "numeric", timeZone: "UTC" });
  } else if (period === "month") {
    start = new Date(Date.UTC(y, m, 1));
    end   = new Date(Date.UTC(y, m + 1, 1) - 1);
    label = start.toLocaleString("en-SG", { month: "long", year: "numeric", timeZone: "UTC" });
  } else if (period === "quarter") {
    const q = Math.floor(m / 3);
    start = new Date(Date.UTC(y, q * 3, 1));
    end   = new Date(Date.UTC(y, q * 3 + 3, 1) - 1);
    label = `Q${q + 1} ${y}`;
  } else if (period === "half") {
    // H1 = Jan–Jun (Q1+Q2), H2 = Jul–Dec (Q3+Q4) — the semi-annual filing period.
    const h = m < 6 ? 0 : 1;
    start = new Date(Date.UTC(y, h * 6, 1));
    end   = new Date(Date.UTC(y, h * 6 + 6, 1) - 1);
    label = `H${h + 1} ${y} (${h === 0 ? "Jan–Jun" : "Jul–Dec"})`;
  } else {
    start = new Date(Date.UTC(y, 0, 1));
    end   = new Date(Date.UTC(y + 1, 0, 1) - 1);
    label = String(y);
  }
  return { start: start.toISOString(), end: end.toISOString(), label };
}
 
export function buildPspmReport(args: {
  period: PspmPeriod;
  at: Date;
  orders: PspmOrderRow[];       // already period- + paid- + internal-filtered by the caller
  highestCarriedValue?: number; // max product retail value we hold (caller computes)
}): PspmReport {
  const { period, at, orders, highestCarriedValue } = args;
  const { start, end, label } = resolvePspmPeriod(period, at);
 
  const empty = (): BucketTotal => ({ sales: 0, count: 0, salesPct: 0, countPct: 0 });
  const byMode = { cash: empty(), crypto: empty(), other: empty() };
  const grand = empty();
 
  for (const o of orders) {
    const v = Number(o.grand_total_sgd ?? 0);
    const b = pspmBucket(o.payment_method);
    byMode[b].sales += v; byMode[b].count += 1;
    grand.sales += v; grand.count += 1;
  }
 
  const sorted = [...orders].sort((a, b) => Number(b.grand_total_sgd ?? 0) - Number(a.grand_total_sgd ?? 0));
  const top3 = sorted.slice(0, 3).map((o) => ({
    ref:           o.ref,
    value:         round2(Number(o.grand_total_sgd ?? 0)),
    paymentLabel:  pspmPaymentLabel(o.payment_method),
    nonFaceToFace: true, // every 1Plus1Gold sale is placed online
  }));
 
  const highestValue = sorted.length ? Number(sorted[0].grand_total_sgd ?? 0) : 0;
 
  // Percentages are always against the grand total (0 when nothing sold).
  const pct = (part: number, whole: number) => (whole > 0 ? Math.round((part / whole) * 1000) / 10 : 0);
  const rnd = (t: BucketTotal): BucketTotal => ({
    sales:    round2(t.sales),
    count:    t.count,
    salesPct: pct(t.sales, grand.sales),
    countPct: pct(t.count, grand.count),
  });
 
  return {
    periodStart: start,
    periodEnd:   end,
    periodLabel: label,
    byMode: { cash: rnd(byMode.cash), crypto: rnd(byMode.crypto), other: rnd(byMode.other) },
    grandTotal: rnd(grand),
    nonFaceToFace: rnd(grand),  // all online
    highestSold: sorted.length ? { value: round2(highestValue), bucket: pspmValueBucketLabel(highestValue) } : null,
    highestCarried: highestCarriedValue != null
      ? { value: round2(highestCarriedValue), bucket: pspmValueBucketLabel(highestCarriedValue) }
      : null,
    top3,
  };
}
