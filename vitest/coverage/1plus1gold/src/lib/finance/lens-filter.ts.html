/**
 * Filtering a Money Map lens table by whatever is open on the diagram.
 *
 * ▶ What this can and cannot do, checked in the payload rather than assumed
 *   (2026-09-04). A lens row and a drilled node meet only where they share a
 *   NAME, so the filter works exactly as deep as the shared vocabulary goes:
 *
 *     • Customers — a customer's kids are orders and an order's kids are product
 *       LINES carrying `shortProduct(label)`, which is the same string the money
 *       tree uses for its product level. So drilling to a product filters the
 *       customer table to the people who bought it. Client-side, no server change.
 *     • Vendors — rows carry their tax lines as kids, so drilling to a tax line
 *       filters to the vendors billing against it.
 *     • Suppliers — the same walk applies wherever a supplier row carries the
 *       drilled name; where it does not, the filter simply finds nothing and the
 *       caller is expected to say so rather than show an empty table.
 *     • Marketing / reach — NOT supported. Those are session and message COUNTS
 *       with no shared key to the money tree; a name match there would be a
 *       coincidence, not a relationship.
 *
 * The match is by exact name on purpose. A substring match would quietly relate
 * "1 gram Split-Valcambi" to "1 gram Split-Valcambi ×10" and hand back a wrong
 * set that looks plausible — the failure mode this file exists to avoid.
 */
 
/** The subset of a money row this module needs. Structural, so both the lens
 *  rows and the drill nodes satisfy it without importing each other's types. */
export interface NamedRow {
  n: string;
  kids?: NamedRow[];
}
 
/**
 * The node currently narrowing the view, or null when nothing narrower than the
 * block itself is open.
 *
 * `path[0]` is the block key — opening a block is not a filter, it is what makes
 * the lens appear at all. Only a deeper selection narrows.
 */
export function selectionLabel(path: string[]): string | null {
  return path.length > 1 ? path[path.length - 1] : null;
}
 
/**
 * True when `row` IS the named node or has it somewhere beneath.
 *
 * Depth-first with a visited guard: the money tree is built fresh per request
 * and should be acyclic, but a filter that can hang the admin page on one bad
 * row is a worse failure than a filter that misses one.
 */
export function rowMatchesLabel(row: NamedRow, label: string): boolean {
  const seen = new Set<NamedRow>();
  const walk = (r: NamedRow): boolean => {
    if (seen.has(r)) return false;
    seen.add(r);
    if (r.n === label) return true;
    return (r.kids ?? []).some(walk);
  };
  return walk(row);
}
 
export interface LensFilterResult<T> {
  rows: T[];
  /** The label actually applied, or null when nothing was. */
  label: string | null;
  /** True when a selection was open but matched no row — the caller must say
   *  so, because an empty table and "no filter" look identical otherwise. */
  empty: boolean;
}
 
/**
 * Apply the open selection to a lens's rows.
 *
 * A selection that matches NOTHING returns the unfiltered rows with
 * `empty: true`, rather than an empty table. Showing zero rows would imply the
 * lens has no data, when what actually happened is that this lens does not share
 * a vocabulary with the node you opened — a different statement, and the honest
 * one to put on screen.
 */
export function applyLensFilter<T extends NamedRow>(rows: T[], label: string | null): LensFilterResult<T> {
  if (!label) return { rows, label: null, empty: false };
  const hit = rows.filter((r) => rowMatchesLabel(r, label));
  if (hit.length === 0) return { rows, label, empty: true };
  return { rows: hit, label, empty: false };
}
