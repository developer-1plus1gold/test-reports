/** "2026-03" + 4 → "2026-07". */
function addMonths(month: string, n: number): string {
  const [y, m] = month.split("-").map(Number);
  const total = (y * 12 + (m - 1)) + n;
  const yy = Math.floor(total / 12);
  const mm = (total % 12) + 1;
  return `${yy}-${String(mm).padStart(2, "0")}`;
}
 
/**
 * The monthly charges for one receipt.
 *
 * ⚠️ Σ THEN ROUND. Each month gets the rounded even share and the FINAL month
 * absorbs the residual, so the schedule sums to the landed cost to the cent.
 * Σ-of-rounded has now caused two separate bugs in this codebase (the Money Map
 * gold COGS came out 8 cents light, then the supplier lens did the same), and
 * here it would leave a few cents of asset that could never be written off.
 *
 * A receipt dated after the write-off month is expensed IN FULL in its own
 * month rather than dropped — buying equipment in 2027 must not silently
 * capitalise it forever under a policy that says nothing is carried.
 */
export function scheduleFor(
  receipt: EquipmentReceipt,
  endMonth = WRITE_OFF_MONTH,
  deMinimis = DE_MINIMIS_SGD,
): DepreciationCharge[] {
  const base = n2(Number(receipt.quantity ?? 0) * Number(receipt.unit_cost_sgd ?? 0));
  if (!(base > 0)) return [];
  const start = monthOf(receipt.received_at);
  if (!start) return [];
 
  const n = monthsBetween(start, endMonth);
  const meta = { product_id: receipt.product_id, product_label: receipt.product_label };
  // Below the capitalisation threshold, or nothing left to spread over: one
  // charge, in the month the kit arrived.
  if (base < deMinimis || n <= 1) return [{ month: start, amount_sgd: base, ...meta }];
 
  const per = n2(base / n);
  const out: DepreciationCharge[] = [];
  let assigned = 0;
  for (let i = 0; i < n - 1; i++) {
    out.push({ month: addMonths(start, i), amount_sgd: per, ...meta });
    assigned = n2(assigned + per);
  }
  out.push({ month: addMonths(start, n - 1), amount_sgd: n2(base - assigned), ...meta });
  return out;
}
 
/** Every charge for every receipt, unfiltered. */
export function allCharges(receipts: EquipmentReceipt[], endMonth = WRITE_OFF_MONTH): DepreciationCharge[] {

/**

 * so any day inside the month is a convention; the 1st keeps a monthly window
 * from splitting a charge across two periods.
 */
export function deriveEquipmentDepreciation(
  receipts: EquipmentReceipt[],
  opts: { from: string; to: string; endMonth?: string },
): ExpenseRow[] {
  const fromM = opts.from.slice(0, 7);
  const toM   = opts.to.slice(0, 7);

      sg_tax_category: "Depreciation",
      us_tax_category: "Line 13 — Depreciation",

}
 
/**
 * Accumulated depreciation up to and including the month of `asOf` — the figure
 * the balance sheet must subtract from the equipment asset.
 *
 * The caller floors the NET asset at zero rather than this figure being capped
 * here: if a write-off has already removed some of the asset via
 * `stock_adjustments`, that loss is expensed by `stock-adjustment-expenses` and
 * depreciating the full received cost on top would charge the same value twice.
 * Flooring at the caller keeps the asset honest; the residual double-count is
 * bounded by the write-off and is called out at the call site.
 */
export function accumulatedDepreciation(
  receipts: EquipmentReceipt[],
  asOf: string,
  endMonth = WRITE_OFF_MONTH,
): number {
  const upto = asOf.slice(0, 7);
  return n2(
    allCharges(receipts, endMonth)
      .filter((c) => c.month <= upto)
      .reduce((a, c) => a + c.amount_sgd, 0),
  );
}
