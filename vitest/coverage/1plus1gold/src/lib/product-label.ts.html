/**
 * Customer- and operator-readable product names.
 *
 * `products.label` is the catalogue's full name — "5 gram Gold Bar - PAMP
 * Fortuna Veriscan® (In Assay, New)". Correct and unambiguous, and too long for
 * a table cell or a card sub-line. `shortProduct` trims the boilerplate every
 * row shares ("Gold Bar - ", "(In Assay, New)") and leaves the part that
 * actually distinguishes one product from another.
 *
 * ▶ WHY THIS IS ITS OWN MODULE. It lived in `src/lib/finance/money-map.ts` until
 * 2026-09-20, because the money map was the first surface that needed it. The
 * Gold Yield dashboard needs the same string — a participant's enrolled bar had
 * been rendering as a hardcoded "CombiBar" regardless of what they actually
 * bought — and importing a finance module into a customer-facing route to reach
 * one pure string function is the wrong dependency. money-map re-exports it, so
 * its existing callers and tests are unchanged.
 */
 
/**
 * Shorten a catalogue label for display. Returns "(unnamed)" for a missing
 * label rather than an empty string, so a row with no product reads as a
 * data gap instead of silently rendering blank.
 */
export function shortProduct(label: string | null | undefined): string {
  if (!label) return "(unnamed)";
  return label
    .replace(" Gold Bar - ", " ").replace("Gold Bar - ", "")
    .replace("(In Assay, New)", "").replace("(In Assay)", "")
    .replace("Fortuna Veriscan®", "Fortuna Veriscan")
    .replace("Valcambi Suisse Gold Minted Bar", "Valcambi minted")
    .replace("Heraeus (HK) Gold Cast Bar", "Heraeus cast")
    .replace("Gold bar Nadir  Refinery-Uniquely Singapore", "Nadir · Uniquely Singapore")
    .replace("Pure Gold-Merlion", "Pure Gold Merlion").replace("Pure Gold-Vietnamese", "Pure Gold Vietnamese")
    .replace("Split-Valcambi", "Valcambi split").replace("Split-Pamp", "PAMP split")
    .replace(/ gram /g, " g ").replace(/ gram$/, " g")
    .replace(/\s+/g, " ").trim();
}
 
/**
 * The product name for a Gold Yield bar row, shortened for display.
 *
 * ⚠️ `gold_yield_bars.inventory_id` is a GY serial STRING ("GY-2026-0005"), NOT
 * a foreign key to `inventory` — so the product is reached through
 * `order_item_id`, which IS a real FK and is NOT NULL on every row. That is the
 * trap anyone reaching for the obvious path will hit.
 *
 * Returns null when the catalogue row has since been removed; both callers then
 * fall back to the weight rather than printing a product name nobody bought.
 *
 * ▶ IT IS SHARED BECAUSE THE FIRST VERSION OF THIS FIX WAS ONLY HALF APPLIED.
 * On 2026-09-20 the participant dashboard was corrected — an enrolled bar had
 * been rendering as a hardcoded "{denomination}g CombiBar" whatever the
 * customer actually bought — and the ADMIN twin, which renders the same bars
 * from `api/admin/gold-yield/overview`, was left showing the bare weight. The
 * operator found it within hours of testing. Both routes now select the same
 * join and call this, so the two cannot drift apart again; both are pinned by
 * `gold-yield-bar-label.test.ts`.
 *
 * The caller's select must carry `order_items ( products ( label ) )`.
 */
export function goldYieldBarLabel(bar: unknown): string | null {
  const label = (bar as { order_items?: { products?: { label?: string | null } | null } | null })
    ?.order_items?.products?.label;
  return label ? shortProduct(label) : null;
}
