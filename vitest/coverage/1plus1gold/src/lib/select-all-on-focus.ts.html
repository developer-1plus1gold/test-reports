/**
 * selectAllOnFocus — highlight a field's whole value when it gains focus so the
 * next keystroke REPLACES it instead of appending. Apply to quantity steppers so
 * tapping the "1" and typing "5" yields "5", not "15".
 *
 * Web/PWA/admin parity with the iOS app's global NumericKeyboardSelectAll
 * (apps/ios-customer/Sources/Shell/NumericKeyboardSelectAll.swift), which does
 * the same select-all-on-focus for numeric keyboards natively.
 *
 * The select is deferred to the next frame: some mobile browsers (iOS Safari in
 * particular) place the caret AFTER the focus event fires, which would clobber a
 * synchronous select(). requestAnimationFrame runs after that, so the selection
 * sticks. Falls back to a synchronous select() where rAF isn't available.
 */
import type { FocusEvent } from "react";
 
export function selectAllOnFocus(e: FocusEvent<HTMLInputElement>): void {
  const el = e.currentTarget;
  const run = () => {
    try { el.select(); } catch { /* detached / unsupported input type — ignore */ }
  };
  if (typeof requestAnimationFrame === "function") requestAnimationFrame(run);
  else run();
}
